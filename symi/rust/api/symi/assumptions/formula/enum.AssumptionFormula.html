Enum AssumptionFormula

pub enum AssumptionFormula {
    LogicalTrue,
    LogicalFalse,
    Predicate(AppliedPredicate),
    ScalarRelation(CanonicalScalarRelation),
    ScalarMembership(CanonicalScalarMembership),
    IntegerCongruence(CanonicalIntegerCongruence),
    LogicalNot(Box<AssumptionFormula>),
    LogicalAnd(Vec<AssumptionFormula>),
    LogicalOr(Vec<AssumptionFormula>),
}
Expand description

A canonical Boolean formula over typed predicate applications.
