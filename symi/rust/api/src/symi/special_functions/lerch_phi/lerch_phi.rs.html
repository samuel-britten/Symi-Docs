24/// Performs the advanced native `simplify_call` operation.
25pub fn simplify_call(
26    context: &Context,
27    builtin: BuiltinFunction,
28    arguments: &[ExpressionReference],
29) -> Option<ExpressionReference> {
30    if builtin != BuiltinFunction::LerchPhi || arguments.len() != 3 {
31        return None;
32    }
33    let argument = &arguments[0];
34    let order = &arguments[1];
35    let shift = &arguments[2];
36    if let Some(order) = nonpositive_integer_order(order) {
37        return nonpositive_integer_rational(context, order, argument, shift);
38    }
39    if is_zero(argument) && is_nonzero_integer(shift) {
40        let negative_order = builders::negate(context, Arc::clone(order));
41        return Some(simplify_power(context, Arc::clone(shift), negative_order));
42    }
43    if is_one(shift) {
44        let polylogarithm = simplify_function_call(
45            context,
46            FunctionIdentifier::Builtin(BuiltinFunction::Polylogarithm),
47            vec![Arc::clone(order), Arc::clone(argument)],
48        );
49        return Some(builders::divide(
50            context,
51            polylogarithm,
52            Arc::clone(argument),
53        ));
54    }
55    if is_one(argument) && is_positive_integer(order) && is_positive_integer(shift) {
56        return Some(simplify_function_call(
57            context,
58            FunctionIdentifier::Builtin(BuiltinFunction::HurwitzZeta),
59            vec![Arc::clone(order), Arc::clone(shift)],
60        ));
61    }
62    None
63}

65/// Builds the finite shift recurrence for a nonnegative exact shift count.
66pub(crate) fn shift_recurrence(
67    context: &Context,
68    argument: ExpressionReference,
69    order: ExpressionReference,
70    shift: ExpressionReference,
71    count: u64,
72) -> ExpressionReference {
73    let mut terms = Vec::with_capacity(count as usize + 1);
74    let count_expression = builders::integer(context, exact_integer::from(count as i64));
75    let shifted_parameter = simplify_sum(context, vec![Arc::clone(&shift), count_expression]);
76    let tail = simplify_function_call(
77        context,
78        FunctionIdentifier::Builtin(BuiltinFunction::LerchPhi),
79        vec![Arc::clone(&argument), Arc::clone(&order), shifted_parameter],
80    );
81    let tail_exponent = builders::integer(context, exact_integer::from(count as i64));
82    let tail_factor = simplify_power(context, Arc::clone(&argument), tail_exponent);
83    terms.push(crate::simplify::simplify_product(
84        context,
85        vec![tail_factor, tail],
86    ));
87    for index in 0..count {
88        let index_expression = builders::integer(context, exact_integer::from(index as i64));
89        let shifted = simplify_sum(context, vec![Arc::clone(&shift), index_expression]);
90        let negative_order = builders::negate(context, Arc::clone(&order));
91        let denominator = simplify_power(context, shifted, negative_order);
92        let argument_exponent = builders::integer(context, exact_integer::from(index as i64));
93        let argument_power = simplify_power(context, Arc::clone(&argument), argument_exponent);
94        terms.push(crate::simplify::simplify_product(
95            context,
96            vec![argument_power, denominator],
97        ));
98    }
99    simplify_sum(context, terms)
100}

102/// Builds the argument partial derivative of the Lerch transcendent.
103pub(crate) fn differentiate_argument(
104    context: &Context,
105    argument: &ExpressionReference,
106    order: &ExpressionReference,
107    shift: &ExpressionReference,
108) -> ExpressionReference {
109    if is_zero(argument) {
110        let one = builders::one(context);
111        let next_shift = simplify_sum(context, vec![Arc::clone(shift), one]);
112        let negative_order = builders::negate(context, Arc::clone(order));
113        return simplify_power(context, next_shift, negative_order);
114    }
115    let one = builders::one(context);
116    let previous_order = builders::subtract(context, Arc::clone(order), one);
117    let previous = simplify_function_call(
118        context,
119        FunctionIdentifier::Builtin(BuiltinFunction::LerchPhi),
120        vec![Arc::clone(argument), previous_order, Arc::clone(shift)],
121    );
122    let current = simplify_function_call(
123        context,
124        FunctionIdentifier::Builtin(BuiltinFunction::LerchPhi),
125        vec![Arc::clone(argument), Arc::clone(order), Arc::clone(shift)],
126    );
127    let shift_times_current =
128        crate::simplify::simplify_product(context, vec![Arc::clone(shift), current]);
129    let negative_shift_times_current = builders::negate(context, shift_times_current);
130    let numerator = simplify_sum(context, vec![previous, negative_shift_times_current]);
131    builders::divide(context, numerator, Arc::clone(argument))
132}

134/// Builds the shift partial derivative of the Lerch transcendent.
135pub(crate) fn differentiate_shift(
136    context: &Context,
137    argument: &ExpressionReference,
138    order: &ExpressionReference,
139    shift: &ExpressionReference,
140) -> ExpressionReference {
141    let one = builders::one(context);
142    let next_order = simplify_sum(context, vec![Arc::clone(order), one]);
143    let next = simplify_function_call(
144        context,
145        FunctionIdentifier::Builtin(BuiltinFunction::LerchPhi),
146        vec![Arc::clone(argument), next_order, Arc::clone(shift)],
147    );
148    let negative_order = builders::negate(context, Arc::clone(order));
149    crate::simplify::simplify_product(context, vec![negative_order, next])
150}

152fn nonpositive_integer_rational(
153    context: &Context,
154    order: i64,
155    argument: &ExpressionReference,
156    shift: &ExpressionReference,
157) -> Option<ExpressionReference> {
158    let magnitude = order.unsigned_abs();
159    let one = builders::one(context);
160    let negative_argument = builders::negate(context, Arc::clone(argument));
161    let one_minus_argument = simplify_sum(context, vec![one, negative_argument]);
162    let mut terms = Vec::with_capacity(magnitude as usize + 1);
163    for index in 0..=magnitude {
164        let coefficient = binomial_coefficient(magnitude, index);
165        let coefficient_expression = builders::integer(context, coefficient);
166        let shift_power = if magnitude == index {
167            builders::one(context)
168        } else {
169            let exponent =
170                builders::integer(context, exact_integer::from((magnitude - index) as i64));
171            simplify_power(context, Arc::clone(shift), exponent)
172        };
173        let moment_numerator =
174            crate::special_functions::polylogarithm::negative_integer_moment_numerator(
175                context, index, argument,
176            );
177        let complement_power = if magnitude == index {
178            builders::one(context)
179        } else {
180            let exponent =
181                builders::integer(context, exact_integer::from((magnitude - index) as i64));
182            simplify_power(context, Arc::clone(&one_minus_argument), exponent)
183        };
184        terms.push(crate::simplify::simplify_product(
185            context,
186            vec![
187                coefficient_expression,
188                shift_power,
189                moment_numerator,
190                complement_power,
191            ],
192        ));
193    }
194    let numerator = simplify_sum(context, terms);
195    let denominator_exponent =
196        builders::integer(context, exact_integer::from((magnitude + 1) as i64));
197    let denominator = simplify_power(context, one_minus_argument, denominator_exponent);
198    Some(builders::divide(context, numerator, denominator))
199}

201fn binomial_coefficient(upper: u64, lower: u64) -> exact_integer {
202    let mut value = exact_integer::ONE;
203    for index in 0..lower.min(upper - lower) {
204        value *= exact_integer::from((upper - index) as i64);
205        value /= exact_integer::from((index + 1) as i64);
206    }
207    value
208}

210fn nonpositive_integer_order(expression: &ExpressionReference) -> Option<i64> {
211    let ExpressionKind::Integer(value) = &expression.kind else {
212        return None;
213    };
214    let order = i64::try_from(value).ok()?;
215    (order <= 0 && order.unsigned_abs() <= NONPOSITIVE_INTEGER_REDUCTION_MAXIMUM_ORDER as u64)
216        .then_some(order)
217}

219fn is_zero(expression: &ExpressionReference) -> bool {
220    matches!(&expression.kind, ExpressionKind::Integer(value) if *value == exact_integer::ZERO)
221}
222
223fn is_one(expression: &ExpressionReference) -> bool {
224    matches!(&expression.kind, ExpressionKind::Integer(value) if *value == exact_integer::ONE)
225}
226
227fn is_nonzero_integer(expression: &ExpressionReference) -> bool {
228    matches!(&expression.kind, ExpressionKind::Integer(value) if *value != exact_integer::ZERO)
229}
230
231fn is_positive_integer(expression: &ExpressionReference) -> bool {
232    matches!(&expression.kind, ExpressionKind::Integer(value) if *value > exact_integer::ZERO)
233}
