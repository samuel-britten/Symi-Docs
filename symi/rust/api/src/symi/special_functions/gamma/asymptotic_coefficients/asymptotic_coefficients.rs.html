1// Exact coefficients of the shifted Stirling expansion of the analytic logarithm of Gamma,
2// DLMF 5.11.8: for fixed complex h and z -> infinity in |ph z| <= pi - delta,
3//   log Gamma(z + h) = (z + h - 1/2) log z - z + log(2 pi)/2
4//                      + sum_{k >= 1} (-1)^(k+1) B_{k+1}(h) / (k (k+1) z^k),
5// where B_n is the Bernoulli polynomial. The remainder after the order-K correction is
6// O(z^(-K-1)) uniformly in that sector. The coefficients are polynomials in the shift, so a
7// symbolic shift yields an exact polynomial expression rather than a guessed number.
8

20/// Coefficients c_j of the Bernoulli polynomial B_degree(h) = sum_j c_j h^j, lowest power
21/// first: c_j = binomial(degree, j) B_{degree - j}.
22pub(crate) fn bernoulli_polynomial_coefficients(degree: usize) -> Vec<Rational> {
23    // B_n(h) = sum_j C(n, j) B_{n-j} h^j with the B_1 = -1/2 convention of the shared table.
24    let bernoulli = bernoulli_numbers_up_to(degree);
25    let mut binomial = Integer::ONE;
26    let mut coefficients = Vec::with_capacity(degree + 1);
27    for power in 0..=degree {
28        coefficients.push(Rational::from(binomial.clone()) * bernoulli[degree - power].clone());
29        binomial =
30            binomial * Integer::from((degree - power) as i64) / Integer::from((power + 1) as i64);
31    }
32    coefficients
33}

35/// Rational factor (-1)^(order+1) / (order (order+1)) multiplying B_{order+1}(h) in the
36/// order-th correction of the shifted Stirling expansion.
37fn correction_normalization(order: usize) -> Rational {
38    // The sign alternates with the order, and the denominator is the product of the two
39    // consecutive integers order and order + 1.
40    let magnitude =
41        Rational::from_integers(Integer::ONE, Integer::from((order * (order + 1)) as i64));
42    if order % 2 == 1 {
43        magnitude
44    } else {
45        -magnitude
46    }
47}

49/// Exact order-th correction coefficient of the shifted Stirling expansion at a rational
50/// shift, (-1)^(order+1) B_{order+1}(shift) / (order (order+1)). `order` must be positive.
51pub(crate) fn log_gamma_correction_at_rational_shift(order: usize, shift: &Rational) -> Rational {
52    // Horner evaluation of the Bernoulli polynomial at the shift, then the normalization.
53    let coefficients = bernoulli_polynomial_coefficients(order + 1);
54    let mut value = Rational::ZERO;
55    for coefficient in coefficients.iter().rev() {
56        value = value * shift.clone() + coefficient.clone();
57    }
58    value * correction_normalization(order)
59}

61/// Order-th correction coefficient of the shifted Stirling expansion as an exact expression
62/// polynomial in a symbolic shift. `order` must be positive.
63pub(crate) fn log_gamma_correction_expression(
64    context: &Context,
65    order: usize,
66    shift: &ExpressionReference,
67) -> ExpressionReference {
68    // A rational shift collapses to one literal; any other shift is expanded in powers of the
69    // shift with exact rational coefficients, so two corrections at equal shifts cancel
70    // structurally once the enclosing sum is expanded.
71    if let Some(rational_shift) = rational_literal_value(shift) {
72        return builders::rational_literal(
73            context,
74            log_gamma_correction_at_rational_shift(order, &rational_shift),
75        );
76    }
77    let normalization = correction_normalization(order);
78    let terms = bernoulli_polynomial_coefficients(order + 1)
79        .into_iter()
80        .enumerate()
81        .filter(|(_, coefficient)| *coefficient != Rational::ZERO)
82        .map(|(power, coefficient)| {
83            let scaled = builders::rational_literal(context, coefficient * normalization.clone());
84            let shift_power = simplify_power(
85                context,
86                Arc::clone(shift),
87                builders::integer_value(context, power as i64),
88            );
89            simplify_product(context, vec![scaled, shift_power])
90        })
91        .collect();
92    simplify_sum(context, terms)
93}
94
95fn rational_literal_value(target: &ExpressionReference) -> Option<Rational> {
96    match &target.kind {
97        crate::expression::ExpressionKind::Integer(value) => Some(Rational::from(value.clone())),
98        crate::expression::ExpressionKind::Rational(value) => Some(value.clone()),
99        _ => None,
100    }
101}

107    fn rational(numerator: i64, denominator: i64) -> Rational {
108        Rational::from_integers(Integer::from(numerator), Integer::from(denominator))

111    #[test]
112    fn unshifted_corrections_are_the_classical_stirling_coefficients() {
113        assert_eq!(
114            log_gamma_correction_at_rational_shift(1, &Rational::ZERO),
115            rational(1, 12)
116        );
117        assert_eq!(
118            log_gamma_correction_at_rational_shift(2, &Rational::ZERO),
119            Rational::ZERO
120        );
121        assert_eq!(
122            log_gamma_correction_at_rational_shift(3, &Rational::ZERO),
123            rational(-1, 360)
124        );
125        assert_eq!(
126            log_gamma_correction_at_rational_shift(5, &Rational::ZERO),
127            rational(1, 1260)
128        );
129    }

131    #[test]
132    fn bernoulli_polynomials_match_their_closed_forms() {
133        assert_eq!(
134            bernoulli_polynomial_coefficients(2),
135            vec![rational(1, 6), rational(-1, 1), rational(1, 1)]
136        );
137        assert_eq!(
138            bernoulli_polynomial_coefficients(3),
139            vec![
140                Rational::ZERO,
141                rational(1, 2),
142                rational(-3, 2),
143                rational(1, 1)

145        );
146    }

148    #[test]
149    fn bernoulli_polynomials_satisfy_the_unit_shift_recurrence() {
150        // The functional equation Gamma(z + h + 1) = (z + h) Gamma(z + h) holds for the shifted
151        // expansion exactly when B_n(h + 1) - B_n(h) = n h^(n-1), checked here exactly.
152        for degree in 1..9usize {
153            for shift in [rational(-7, 3), Rational::ZERO, rational(5, 2)] {
154                let evaluate = |point: &Rational| {
155                    bernoulli_polynomial_coefficients(degree)
156                        .iter()
157                        .rev()
158                        .fold(Rational::ZERO, |value, coefficient| {
159                            value * point.clone() + coefficient.clone()
160                        })
161                };
162                let difference = evaluate(&(shift.clone() + Rational::ONE)) - evaluate(&shift);
163                let mut expected = Rational::from(Integer::from(degree as i64));
164                for _ in 1..degree {
165                    expected = expected * shift.clone();
166                }
167                assert_eq!(difference, expected, "degree {degree}, shift {shift:?}");
168            }
169        }
170    }

172    #[test]
173    fn symbolic_correction_specializes_to_the_rational_value() {
174        let context = Context::new();
175        let shift_symbol = builders::symbol(&context, "h");
176        for order in 1..6usize {
177            let symbolic = log_gamma_correction_expression(&context, order, &shift_symbol);
178            let mut mapping = std::collections::HashMap::new();
179            mapping.insert(
180                Arc::clone(&shift_symbol),
181                builders::rational_value(&context, 3, 4),
182            );
183            let specialized =
184                crate::expression::substitute::substitute(&context, &symbolic, &mapping);
185            let expected = builders::rational_literal(
186                &context,
187                log_gamma_correction_at_rational_shift(order, &rational(3, 4)),
188            );
189            assert_eq!(
190                crate::expand::expand(&context, &specialized),
191                expected,
192                "order {order}"
193            );
194        }
