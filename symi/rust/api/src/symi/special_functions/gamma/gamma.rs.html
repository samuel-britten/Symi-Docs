1// The gamma family — gamma, log_gamma, digamma, polygamma, beta — shared across the
2// symbolic and numeric seams. The numeric kernels are written once against the
3// private gamma_field trait (implemented for both the real and complex inexact
4// backends) so the real fast path and the complex path run identical mathematics:
5// the Stirling asymptotic series with argument-reduction recurrence and the
6// reflection formula, the production arbitrary-precision method (Arb, mpmath).
7//
8// Domain coverage (declining is always rigorous — a None here makes the verifier
9// report inconclusive rather than ever returning a wrong value):
10//   gamma     full plane minus the non-positive-integer poles (reflection for Re < 0)
11//   log_gamma the right half-plane Re(z) > 0 (principal-branch left half deferred)
12//   digamma   full plane minus the poles (reflection via pi*cot for Re <= 0)
13//   polygamma the right half-plane Re(z) > 0 for integer order >= 1 (reflection deferred)
14//   beta      via gamma(x) gamma(y) / gamma(x + y)
15

37    fn from_rational(value: &Rational, evaluator: &mut NumericEvaluator) -> Self;
38    fn pi(evaluator: &mut NumericEvaluator) -> Self;

40    fn subtract(&self, other: &Self, evaluator: &mut NumericEvaluator) -> Self;
41    fn multiply(&self, other: &Self, evaluator: &mut NumericEvaluator) -> Self;
42    fn divide(&self, other: &Self, evaluator: &mut NumericEvaluator) -> Option<Self>;
43    fn exponential(&self, evaluator: &mut NumericEvaluator) -> Self;
44    fn logarithm(&self, evaluator: &mut NumericEvaluator) -> Option<Self>;
45    fn sine(&self, evaluator: &mut NumericEvaluator) -> Self;

47    fn real_component(&self) -> RealInexactNumber;
48    fn magnitude_real(&self, evaluator: &mut NumericEvaluator) -> RealInexactNumber;

172pub(crate) fn shift_target(evaluator: &NumericEvaluator) -> i64 {
173    // Minimum real part the argument-reduction recurrence pushes the argument to
174    // before the Stirling series is applied. The asymptotic series reaches the
175    // working precision once Re(z) is a fixed fraction of the precision in bits
176    // (Arb's analysis: a real part proportional to the bit count balances the
177    // shift count against the term count); 0.16 with a constant floor is a safe
178    // margin above the 0.11 threshold where the minimal term first drops below
179    // 2^-precision.
180    let working = evaluator.working_precision() as f64;
181    (0.16 * working + 8.0) as i64
182}

184pub(crate) fn recurrence_shift(real_part: &RealInexactNumber, target: i64) -> i64 {
185    // Number of unit steps needed to raise the real part to the shift target. A
186    // real part already at or beyond the target (or too large to fit an f64) needs
187    // no shift.
188    match real_part.to_f64() {
189        Some(value) if value < target as f64 => (target as f64 - value).ceil() as i64,
190        _ => 0,
191    }
192}

194pub(crate) fn bernoulli_numbers_up_to(highest_index: usize) -> Vec<Rational> {
195    // [B_0, B_1, ..., B_{highest_index}] as exact rationals via the Akiyama-Tanigawa
196    // transformation, in the convention B_1 = -1/2 (the standard sign carried by the
197    // Bernoulli polynomials and the zeta functional values). Akiyama-Tanigawa natively
198    // yields B_1 = +1/2, corrected here so the single odd nonzero Bernoulli number is
199    // available to the zeta family; every other odd-indexed value is exactly zero and
200    // every even-indexed value is independent of the B_1 convention.
201    let mut row: Vec<Rational> = Vec::with_capacity(highest_index + 1);
202    let mut bernoulli: Vec<Rational> = Vec::with_capacity(highest_index + 1);
203    for index in 0..=highest_index {
204        row.push(Rational::from_integers(
205            Integer::from(1),
206            Integer::from((index + 1) as i64),
207        ));
208        for inner in (1..=index).rev() {
209            let difference = &row[inner - 1] - &row[inner];
210            row[inner - 1] = Rational::from(inner as i64) * difference;
211        }
212        bernoulli.push(row[0].clone());
213    }
214    if highest_index >= 1 {
215        bernoulli[1] = -bernoulli[1].clone();
216    }
217    bernoulli
218}

220/// How many even-indexed Bernoulli tables are kept for later evaluations to share.
221///
222/// A run visits only the handful of working precisions its escalation policy asks for — it
223/// rounds them to a step so that they collide — so a few entries cover a whole session; the
224/// bound is what keeps one that keeps climbing to new precisions from retaining every table
225/// it ever built.
226const RETAINED_BERNOULLI_TABLES: usize = 8;
227
228/// The even-indexed Bernoulli tables already built in this process, keyed by term count.
229///
230/// The table depends on nothing but the term count, and the term count depends on nothing but
231/// the working precision, so every gamma and zeta kernel evaluated at one precision wants the
232/// very same table. Building one is quadratic in the term count and runs on exact rationals
233/// whose numerators grow past a thousand digits, which costs far more than the asymptotic sum
234/// it feeds: the near-one Lerch and polylogarithm expansions evaluate a zeta per series term,
235/// so rebuilding the table per call dominated those kernels outright. A table is a pure
236/// accelerator — a term count has exactly one table — so a kernel that finds one here computes
237/// exactly what a kernel that built its own would, on whichever thread it runs.
238///
239/// The lock is held only to look a table up or record a new one, never across an evaluation.
240static RETAINED_EVEN_INDEXED_BERNOULLI: Mutex<Vec<(usize, Arc<[Rational]>)>> =
241    Mutex::new(Vec::new());
242
243pub(crate) fn bernoulli_even_indexed(count: usize) -> Arc<[Rational]> {
244    // [B_2, B_4, ..., B_{2*count}] as exact rationals. Only the even-indexed Bernoulli
245    // numbers enter the Stirling and digamma/polygamma asymptotic series.
246    if let Some(table) = retained_bernoulli_table(count) {
247        return table;
248    }
249    let bernoulli = bernoulli_numbers_up_to(2 * count);
250    let table: Arc<[Rational]> = (1..=count)
251        .map(|index| bernoulli[2 * index].clone())
252        .collect();
253    retain_bernoulli_table(count, &table);
254    table
255}
256
257fn retained_bernoulli_table(count: usize) -> Option<Arc<[Rational]>> {
258    let retained = RETAINED_EVEN_INDEXED_BERNOULLI
259        .lock()
260        .unwrap_or_else(|poisoned| poisoned.into_inner());
261    retained
262        .iter()
263        .find(|(retained_count, _)| *retained_count == count)
264        .map(|(_, table)| Arc::clone(table))
265}
266
267fn retain_bernoulli_table(count: usize, table: &Arc<[Rational]>) {
268    let mut retained = RETAINED_EVEN_INDEXED_BERNOULLI
269        .lock()
270        .unwrap_or_else(|poisoned| poisoned.into_inner());
271    if retained
272        .iter()
273        .any(|(retained_count, _)| *retained_count == count)
274    {
275        return;
276    }
277    if retained.len() == RETAINED_BERNOULLI_TABLES {
278        retained.remove(0);
279    }
280    retained.push((count, Arc::clone(table)));
281}

283pub(crate) fn term_count(evaluator: &NumericEvaluator) -> usize {
284    // Upper bound on Stirling / asymptotic terms. The adaptive loop stops at the
285    // asymptotic minimum well before this cap; the cap only bounds the Bernoulli
286    // table size and is set above the optimal truncation point (~pi times the
287    // shift target).
288    let working = evaluator.working_precision() as f64;
289    (0.6 * working + 12.0) as usize
290}

292fn stirling_log_gamma<Field: GammaField>(
293    large_argument: &Field,
294    bernoulli: &[Rational],
295    evaluator: &mut NumericEvaluator,
296) -> Option<Field> {
297    // log Gamma(w) = (w - 1/2) log w - w + 1/2 log(2 pi)
298    //                + sum_{k>=1} B_{2k} / (2k (2k-1) w^{2k-1}),
299    // valid for a large real part; the asymptotic sum is truncated at its
300    // smallest term.
301    let half = Field::from_rational(
302        &Rational::from_integers(Integer::from(1), Integer::from(2)),
303        evaluator,
304    );
305    let log_argument = large_argument.logarithm(evaluator)?;
306    let argument_minus_half = large_argument.subtract(&half, evaluator);
307    let principal = argument_minus_half.multiply(&log_argument, evaluator);
308    let principal = principal.subtract(large_argument, evaluator);
309    let two = Field::from_i64(2, evaluator);
310    let pi = Field::pi(evaluator);
311    let two_pi = two.multiply(&pi, evaluator);
312    let log_two_pi = two_pi.logarithm(evaluator)?;
313    let half_log_two_pi = log_two_pi.multiply(&half, evaluator);
314    let mut result = principal.add(&half_log_two_pi, evaluator);
315
316    let argument_squared = large_argument.multiply(large_argument, evaluator);
317    let mut argument_power = large_argument.clone();
318    let mut previous_magnitude: Option<RealInexactNumber> = None;
319    for (offset, bernoulli_value) in bernoulli.iter().enumerate() {
320        let index = (offset + 1) as i64;
321        let denominator = Rational::from(2 * index) * Rational::from(2 * index - 1);
322        let coefficient = Field::from_rational(&(bernoulli_value / denominator), evaluator);
323        let term = coefficient.divide(&argument_power, evaluator)?;
324        let magnitude = term.magnitude_real(evaluator);
325        if let Some(previous) = &previous_magnitude {
326            if magnitude.partial_compare(previous) != Some(Ordering::Less) {
327                break;
328            }
329        }
330        result = result.add(&term, evaluator);
331        previous_magnitude = Some(magnitude);
332        argument_power = argument_power.multiply(&argument_squared, evaluator);
333    }
334    Some(result)
335}

337fn log_gamma_value<Field: GammaField>(
338    argument: &Field,
339    bernoulli: &[Rational],
340    evaluator: &mut NumericEvaluator,
341) -> Option<Field> {
342    // Principal-branch log Gamma for Re(argument) > 0: recurrence up to a large
343    // real part, then Stirling. Declines for Re <= 0 (left half-plane deferred).
344    let real_part = argument.real_component();
345    if !is_strictly_positive(&real_part) {
346        return None;
347    }
348    let shift = recurrence_shift(&real_part, shift_target(evaluator));
349    let mut accumulated_logarithm = Field::from_i64(0, evaluator);
350    for step in 0..shift {
351        let offset = Field::from_i64(step, evaluator);
352        let shifted = argument.add(&offset, evaluator);
353        let logarithm = shifted.logarithm(evaluator)?;
354        accumulated_logarithm = accumulated_logarithm.add(&logarithm, evaluator);
355    }
356    let shift_field = Field::from_i64(shift, evaluator);
357    let large_argument = argument.add(&shift_field, evaluator);
358    let stirling = stirling_log_gamma(&large_argument, bernoulli, evaluator)?;
359    Some(stirling.subtract(&accumulated_logarithm, evaluator))
360}

362pub(crate) fn gamma_value<Field: GammaField>(
363    argument: &Field,
364    bernoulli: &[Rational],
365    evaluator: &mut NumericEvaluator,
366) -> Option<Field> {
367    // Gamma over the whole plane minus the non-positive-integer poles. The right
368    // half-plane is exp(log Gamma); the left half-plane uses the reflection
369    // Gamma(z) = pi / (sin(pi z) Gamma(1 - z)), where 1 - z lands in the right
370    // half-plane. A zero reflected denominator signals a pole and declines.
371    let real_part = argument.real_component();
372    if is_strictly_positive(&real_part) {
373        let logarithm = log_gamma_value(argument, bernoulli, evaluator)?;
374        return Some(logarithm.exponential(evaluator));
375    }
376    let one = Field::from_i64(1, evaluator);
377    let one_minus_argument = one.subtract(argument, evaluator);
378    let reflected = gamma_value(&one_minus_argument, bernoulli, evaluator)?;
379    let pi = Field::pi(evaluator);
380    let pi_argument = pi.multiply(argument, evaluator);
381    let sine = pi_argument.sine(evaluator);
382    let denominator = sine.multiply(&reflected, evaluator);
383    pi.divide(&denominator, evaluator)
384}

386fn stirling_digamma<Field: GammaField>(
387    large_argument: &Field,
388    bernoulli: &[Rational],
389    evaluator: &mut NumericEvaluator,
390) -> Option<Field> {
391    // psi(w) = log w - 1/(2w) - sum_{k>=1} B_{2k} / (2k w^{2k}) for a large real
392    // part, truncated at the smallest term.
393    let log_argument = large_argument.logarithm(evaluator)?;
394    let half = Field::from_rational(
395        &Rational::from_integers(Integer::from(1), Integer::from(2)),
396        evaluator,
397    );
398    let half_over_argument = half.divide(large_argument, evaluator)?;
399    let mut result = log_argument.subtract(&half_over_argument, evaluator);
400
401    let argument_squared = large_argument.multiply(large_argument, evaluator);
402    let mut argument_power = argument_squared.clone();
403    let mut previous_magnitude: Option<RealInexactNumber> = None;
404    for (offset, bernoulli_value) in bernoulli.iter().enumerate() {
405        let index = (offset + 1) as i64;
406        let coefficient =
407            Field::from_rational(&(bernoulli_value / Rational::from(2 * index)), evaluator);
408        let term = coefficient.divide(&argument_power, evaluator)?;
409        let magnitude = term.magnitude_real(evaluator);
410        if let Some(previous) = &previous_magnitude {
411            if magnitude.partial_compare(previous) != Some(Ordering::Less) {
412                break;
413            }
414        }
415        result = result.subtract(&term, evaluator);
416        previous_magnitude = Some(magnitude);
417        argument_power = argument_power.multiply(&argument_squared, evaluator);
418    }
419    Some(result)
420}

422fn digamma_value<Field: GammaField>(
423    argument: &Field,
424    bernoulli: &[Rational],
425    evaluator: &mut NumericEvaluator,
426) -> Option<Field> {
427    // psi over the whole plane minus the poles. Right half-plane: recurrence up
428    // then the asymptotic series. Left half-plane: reflection
429    // psi(z) = psi(1 - z) - pi cot(pi z); a zero sine signals a pole and declines.
430    let real_part = argument.real_component();
431    if is_strictly_positive(&real_part) {
432        let shift = recurrence_shift(&real_part, shift_target(evaluator));
433        let mut accumulated_reciprocal = Field::from_i64(0, evaluator);
434        for step in 0..shift {
435            let offset = Field::from_i64(step, evaluator);
436            let shifted = argument.add(&offset, evaluator);
437            let one = Field::from_i64(1, evaluator);
438            let reciprocal = one.divide(&shifted, evaluator)?;
439            accumulated_reciprocal = accumulated_reciprocal.add(&reciprocal, evaluator);
440        }
441        let shift_field = Field::from_i64(shift, evaluator);
442        let large_argument = argument.add(&shift_field, evaluator);
443        let asymptotic = stirling_digamma(&large_argument, bernoulli, evaluator)?;
444        return Some(asymptotic.subtract(&accumulated_reciprocal, evaluator));
445    }
446    let one = Field::from_i64(1, evaluator);
447    let one_minus_argument = one.subtract(argument, evaluator);
448    let reflected = digamma_value(&one_minus_argument, bernoulli, evaluator)?;
449    let pi = Field::pi(evaluator);
450    let pi_argument = pi.multiply(argument, evaluator);
451    let sine = pi_argument.sine(evaluator);
452    let cosine = pi_argument.cosine(evaluator);
453    let cotangent = cosine.divide(&sine, evaluator)?;
454    let pi_cotangent = pi.multiply(&cotangent, evaluator);
455    Some(reflected.subtract(&pi_cotangent, evaluator))
456}

458fn rising_product(start: i64, length: i64) -> Integer {
459    // (start)(start+1)...(start+length-1) as an exact integer; the empty product
460    // (length 0) is 1. Used for the (2k+m-1)!/(2k)! ratio in the polygamma series.
461    let mut total = Integer::from(1);
462    for offset in 0..length {
463        total *= Integer::from(start + offset);
464    }
465    total
466}

468pub(crate) fn factorial_integer(value: i64) -> Integer {
469    rising_product(1, value)

472fn stirling_polygamma<Field: GammaField>(
473    order: i64,
474    large_argument: &Field,
475    bernoulli: &[Rational],
476    evaluator: &mut NumericEvaluator,
477) -> Option<Field> {
478    // psi^{(m)}(w) = (-1)^{m-1} [ (m-1)!/w^m + m!/(2 w^{m+1})
479    //                + sum_{k>=1} B_{2k} (2k+m-1)!/(2k)! / w^{2k+m} ], for m >= 1.
480    let one = Field::from_i64(1, evaluator);
481    let inverse = one.divide(large_argument, evaluator)?;
482    let argument_power_m = field_power(&inverse, order as u64, evaluator);
483    let leading = Field::from_rational(&Rational::from(factorial_integer(order - 1)), evaluator)
484        .multiply(&argument_power_m, evaluator);
485    let argument_power_m_plus_one = argument_power_m.multiply(&inverse, evaluator);
486    let half = Field::from_rational(
487        &Rational::from_integers(Integer::from(1), Integer::from(2)),
488        evaluator,
489    );
490    let second = Field::from_rational(&Rational::from(factorial_integer(order)), evaluator)
491        .multiply(&half, evaluator)
492        .multiply(&argument_power_m_plus_one, evaluator);
493    let mut inner = leading.add(&second, evaluator);
494
495    let inverse_squared = inverse.multiply(&inverse, evaluator);
496    let mut inverse_power = argument_power_m_plus_one.multiply(&inverse, evaluator);
497    let mut previous_magnitude: Option<RealInexactNumber> = None;
498    for (offset, bernoulli_value) in bernoulli.iter().enumerate() {
499        let index = (offset + 1) as i64;
500        let numerator = bernoulli_value * Rational::from(rising_product(2 * index + 1, order - 1));
501        let coefficient = Field::from_rational(&numerator, evaluator);
502        let term = coefficient.multiply(&inverse_power, evaluator);
503        let magnitude = term.magnitude_real(evaluator);
504        if let Some(previous) = &previous_magnitude {
505            if magnitude.partial_compare(previous) != Some(Ordering::Less) {
506                break;
507            }
508        }
509        inner = inner.add(&term, evaluator);
510        previous_magnitude = Some(magnitude);
511        inverse_power = inverse_power.multiply(&inverse_squared, evaluator);
512    }
513    if (order - 1) % 2 == 0 {
514        Some(inner)
515    } else {
516        let negative_one = Field::from_i64(-1, evaluator);
517        Some(inner.multiply(&negative_one, evaluator))
518    }
519}

521fn field_power<Field: GammaField>(
522    base: &Field,
523    exponent: u64,
524    evaluator: &mut NumericEvaluator,
525) -> Field {
526    // base raised to a small natural exponent by repeated multiplication.
527    let mut result = Field::from_i64(1, evaluator);
528    for _ in 0..exponent {
529        result = result.multiply(base, evaluator);
530    }
531    result
532}

534fn polygamma_value<Field: GammaField>(
535    order: i64,
536    argument: &Field,
537    bernoulli: &[Rational],
538    evaluator: &mut NumericEvaluator,
539) -> Option<Field> {
540    // psi^{(m)} for integer order m >= 1 over the right half-plane via recurrence
541    // up then the asymptotic ladder. Re <= 0 is declined (reflection deferred).
542    if order < 1 {
543        return None;
544    }
545    let real_part = argument.real_component();
546    if !is_strictly_positive(&real_part) {
547        return None;
548    }
549    let shift = recurrence_shift(&real_part, shift_target(evaluator));
550    let mut accumulated = Field::from_i64(0, evaluator);
551    for step in 0..shift {
552        let offset = Field::from_i64(step, evaluator);
553        let shifted = argument.add(&offset, evaluator);
554        let one = Field::from_i64(1, evaluator);
555        let inverse = one.divide(&shifted, evaluator)?;
556        let power = field_power(&inverse, (order + 1) as u64, evaluator);
557        accumulated = accumulated.add(&power, evaluator);
558    }
559    let factorial = Field::from_rational(&Rational::from(factorial_integer(order)), evaluator);
560    let mut recurrence_correction = accumulated.multiply(&factorial, evaluator);
561    if order % 2 == 0 {
562        let negative_one = Field::from_i64(-1, evaluator);
563        recurrence_correction = recurrence_correction.multiply(&negative_one, evaluator);
564    }
565    let shift_field = Field::from_i64(shift, evaluator);
566    let large_argument = argument.add(&shift_field, evaluator);
567    let asymptotic = stirling_polygamma(order, &large_argument, bernoulli, evaluator)?;
568    Some(asymptotic.add(&recurrence_correction, evaluator))
569}

571fn beta_value<Field: GammaField>(
572    first: &Field,
573    second: &Field,
574    bernoulli: &[Rational],
575    evaluator: &mut NumericEvaluator,
576) -> Option<Field> {
577    // B(x, y) = Gamma(x) Gamma(y) / Gamma(x + y). Routing through the full-domain
578    // gamma keeps negative non-integer arguments in scope; any pole declines.
579    let gamma_first = gamma_value(first, bernoulli, evaluator)?;
580    let gamma_second = gamma_value(second, bernoulli, evaluator)?;
581    let sum = first.add(second, evaluator);
582    let gamma_sum = gamma_value(&sum, bernoulli, evaluator)?;
583    let numerator = gamma_first.multiply(&gamma_second, evaluator);
584    numerator.divide(&gamma_sum, evaluator)
585}

661/// Performs the advanced native `simplify_call` operation.
662pub fn simplify_call(
663    context: &Context,
664    builtin: BuiltinFunction,
665    arguments: &[ExpressionReference],
666) -> Option<ExpressionReference> {
667    // Symbolic auto-simplification for the gamma family: special values, the
668    // non-positive-integer poles, the polygamma(0, .) -> digamma reconciliation,
669    // the integer-offset recurrence normaliser, and the beta -> gamma reduction.
670    // A None return leaves the call unevaluated as a first-class head.
671    match builtin {
672        BuiltinFunction::Gamma if arguments.len() == 1 => simplify_gamma(context, &arguments[0]),
673        BuiltinFunction::LogGamma if arguments.len() == 1 => {
674            simplify_log_gamma(context, &arguments[0])
675        }
676        BuiltinFunction::Digamma if arguments.len() == 1 => {
677            simplify_digamma(context, &arguments[0])
678        }
679        BuiltinFunction::Polygamma if arguments.len() == 2 => {
680            simplify_polygamma(context, &arguments[0], &arguments[1])
681        }
682        BuiltinFunction::Beta if arguments.len() == 2 => {
683            simplify_beta(context, &arguments[0], &arguments[1])
684        }
685        _ => None,
686    }
687}

713fn rational_expression(context: &Context, value: &Rational) -> ExpressionReference {
714    let (numerator, denominator) = value.to_numerator_and_denominator();
715    let mut signed_numerator = Integer::from(numerator);
716    if value.sign_comparison() == Ordering::Less {
717        signed_numerator = -signed_numerator;
718    }
719    builders::rational(context, signed_numerator, Integer::from(denominator))
720}

722fn factorial_value(value: &Integer) -> Option<Integer> {
723    // Materialise value! as an exact integer for small arguments; declines beyond a
724    // bound so a huge factorial stays the compact symbolic factorial node instead.
725    let bound = i64::try_from(value).ok()?;
726    if bound < 0 || bound > 500 {
727        return None;
728    }
729    let mut result = Integer::from(1);
730    for factor in 2..=bound {
731        result *= Integer::from(factor);
732    }
733    Some(result)
734}
735
736fn signed_half_integer(value: &Rational) -> Option<Integer> {
737    // For a rational with denominator exactly 2, return the (odd) signed numerator
738    // p so the argument is p/2; declines otherwise.
739    let (numerator_natural, denominator_natural) = value.numerator_and_denominator_ref();
740    if denominator_natural != Natural::from(2u32) {
741        return None;
742    }
743    let magnitude = Integer::from(numerator_natural.clone());
744    Some(if value.sign_comparison() == Ordering::Less {
745        -magnitude
746    } else {
747        magnitude
748    })
749}

751fn simplify_gamma(
752    context: &Context,
753    argument: &ExpressionReference,
754) -> Option<ExpressionReference> {
755    match &argument.kind {
756        ExpressionKind::Integer(value) => match value.sign_comparison() {
757            Ordering::Greater => {
758                let reduced = value.clone() - Integer::from(1);
759                if let Some(factorial) = factorial_value(&reduced) {
760                    Some(builders::integer(context, factorial))
761                } else {
762                    let node = builders::integer(context, reduced);
763                    Some(builders::factorial(context, node))
764                }
765            }
766            _ => Some(undefined_expression(context)),
767        },
768        ExpressionKind::Rational(value) => {
769            let double_argument = signed_half_integer(value)?;
770            let coefficient = half_integer_gamma_coefficient(&double_argument);
771            let coefficient_expression = rational_expression(context, &coefficient);
772            let root_pi = square_root_of_pi(context);
773            Some(simplify_product(
774                context,
775                vec![coefficient_expression, root_pi],
776            ))
777        }
778        _ => gamma_recurrence(context, argument),
779    }
780}
781
782fn half_integer_gamma_coefficient(double_argument: &Integer) -> Rational {
783    // Gamma(p/2) = coefficient * sqrt(pi) for odd p, by stepping the recurrence
784    // Gamma(z+1) = z Gamma(z) between 1/2 and p/2.
785    let target = Rational::from_integers(double_argument.clone(), Integer::from(2));
786    let mut current = Rational::from_integers(Integer::from(1), Integer::from(2));
787    let mut coefficient = Rational::ONE;
788    while current < target {
789        coefficient *= &current;
790        current += Rational::ONE;
791    }
792    while current > target {
793        current -= Rational::ONE;
794        coefficient /= &current;
795    }
796    coefficient
797}

799fn simplify_log_gamma(
800    context: &Context,
801    argument: &ExpressionReference,
802) -> Option<ExpressionReference> {
803    // log_gamma(n) = log((n-1)!) for a positive integer; other arguments are left
804    // to the numeric layer so no non-principal branch is asserted symbolically.
805    if let ExpressionKind::Integer(value) = &argument.kind {
806        if value.sign_comparison() == Ordering::Greater {
807            let gamma = simplify_gamma(context, argument)?;
808            return Some(simplify_function_call(
809                context,
810                FunctionIdentifier::Builtin(BuiltinFunction::Log),
811                vec![gamma],
812            ));
813        }
814    }
815    None
816}

818fn simplify_digamma(
819    context: &Context,
820    argument: &ExpressionReference,
821) -> Option<ExpressionReference> {
822    match &argument.kind {
823        ExpressionKind::Integer(value) => match value.sign_comparison() {
824            Ordering::Greater => digamma_at_positive_integer(context, value),
825            _ => Some(undefined_expression(context)),
826        },
827        ExpressionKind::Rational(value) => digamma_at_half_integer(context, value),
828        _ => digamma_recurrence(context, argument),
829    }
830}

832fn digamma_at_positive_integer(context: &Context, value: &Integer) -> Option<ExpressionReference> {
833    // psi(n) = -euler_gamma + H_{n-1}, the (n-1)th harmonic number.
834    let bound = i64::try_from(value).ok()?;
835    if bound > 500 {
836        return None;
837    }
838    let mut harmonic = Rational::ZERO;
839    for denominator in 1..bound {
840        harmonic += Rational::from_integers(Integer::from(1), Integer::from(denominator));
841    }
842    let gamma_term = negative_euler_gamma(context);
843    if harmonic == Rational::ZERO {
844        return Some(gamma_term);
845    }
846    let harmonic_term = rational_expression(context, &harmonic);
847    Some(simplify_sum(context, vec![gamma_term, harmonic_term]))
848}
849
850fn digamma_at_half_integer(context: &Context, value: &Rational) -> Option<ExpressionReference> {
851    // psi(1/2) = -euler_gamma - 2 log 2, extended to any half-integer by stepping
852    // the recurrence psi(z+1) = psi(z) + 1/z between 1/2 and the argument.
853    let double_argument = signed_half_integer(value)?;
854    let target = Rational::from_integers(double_argument, Integer::from(2));
855    let mut current = Rational::from_integers(Integer::from(1), Integer::from(2));
856    let mut correction = Rational::ZERO;
857    while current < target {
858        correction += Rational::ONE / &current;
859        current += Rational::ONE;
860    }
861    while current > target {
862        current -= Rational::ONE;
863        correction -= Rational::ONE / &current;
864    }
865    let gamma_term = negative_euler_gamma(context);
866    let two = builders::integer_value(context, 2);
867    let log_two = simplify_function_call(
868        context,
869        FunctionIdentifier::Builtin(BuiltinFunction::Log),
870        vec![two],
871    );
872    let minus_two = builders::integer_value(context, -2);
873    let two_log_two = simplify_product(context, vec![minus_two, log_two]);
874    let mut terms = vec![gamma_term, two_log_two];
875    if correction != Rational::ZERO {
876        terms.push(rational_expression(context, &correction));
877    }
878    Some(simplify_sum(context, terms))
879}

881fn simplify_polygamma(
882    context: &Context,
883    order: &ExpressionReference,
884    argument: &ExpressionReference,
885) -> Option<ExpressionReference> {
886    let order_value = match &order.kind {
887        ExpressionKind::Integer(value) => value,
888        _ => return None,
889    };
890    if *order_value == Integer::ZERO {
891        return Some(simplify_function_call(
892            context,
893            FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
894            vec![argument.clone()],
895        ));
896    }
897    if order_value.sign_comparison() != Ordering::Greater {
898        return None;
899    }
900    if let ExpressionKind::Integer(argument_value) = &argument.kind {
901        if argument_value.sign_comparison() != Ordering::Greater {
902            return Some(undefined_expression(context));
903        }
904        if *argument_value == Integer::ONE {
905            // psi^{(m)}(1) = (-1)^{m+1} m! zeta(m+1), which
906            // folds to the pi closed form for even m+1 (psi'(1) = pi^2/6) and stays
907            // (-1)^{m+1} m! zeta(m+1) for odd m+1 (psi''(1) = -2 zeta(3)).
908            let order_value = i64::try_from(order_value).ok()?;
909            return crate::special_functions::zeta::polygamma_to_zeta(
910                context,
911                order_value,
912                argument.clone(),
913            );
914        }
915    }
916    None
917}

919fn simplify_beta(
920    context: &Context,
921    first: &ExpressionReference,
922    second: &ExpressionReference,
923) -> Option<ExpressionReference> {
924    if is_integer_one(second) {
925        return Some(reciprocal_expression(context, first.clone()));
926    }
927    if is_integer_one(first) {
928        return Some(reciprocal_expression(context, second.clone()));
929    }
930    let gamma_first = simplify_gamma(context, first)?;
931    let gamma_second = simplify_gamma(context, second)?;
932    let sum = simplify_sum(context, vec![first.clone(), second.clone()]);
933    let gamma_sum = simplify_gamma(context, &sum)?;
934    let numerator = simplify_product(context, vec![gamma_first, gamma_second]);
935    let reciprocal_sum = reciprocal_expression(context, gamma_sum);
936    Some(simplify_product(context, vec![numerator, reciprocal_sum]))
937}

939fn is_integer_one(value: &ExpressionReference) -> bool {
940    matches!(&value.kind, ExpressionKind::Integer(integer_value) if *integer_value == Integer::ONE)
941}

943fn split_integer_offset(
944    context: &Context,
945    argument: &ExpressionReference,
946) -> Option<(i64, ExpressionReference)> {
947    // Decompose a sum carrying exactly one nonzero integer term c into (c, rest),
948    // where rest is the sum of the remaining non-numeric terms. The bound keeps the
949    // recurrence product finite.
950    let children = match &argument.kind {
951        ExpressionKind::Sum(children) => children,
952        _ => return None,
953    };
954    let mut offset: Option<i64> = None;
955    let mut remainder_terms: Vec<ExpressionReference> = Vec::new();
956    for child in children {
957        if let ExpressionKind::Integer(value) = &child.kind {
958            if offset.is_some() {
959                return None;
960            }
961            offset = Some(i64::try_from(value).ok()?);
962        } else {
963            remainder_terms.push(child.clone());
964        }
965    }
966    let offset = offset?;
967    if offset == 0 || remainder_terms.is_empty() || offset.abs() > 64 {
968        return None;
969    }
970    let remainder = simplify_sum(context, remainder_terms);
971    Some((offset, remainder))
972}

974fn gamma_recurrence(
975    context: &Context,
976    argument: &ExpressionReference,
977) -> Option<ExpressionReference> {
978    // Gamma(rest + c): c > 0 multiplies the ascending factors (rest+c-1)...(rest)
979    // onto Gamma(rest); c < 0 divides by (rest-1)...(rest+c). Reduces the integer
980    // offset to zero, the canonical residue.
981    let (offset, remainder) = split_integer_offset(context, argument)?;
982    let gamma_remainder = simplify_function_call(
983        context,
984        FunctionIdentifier::Builtin(BuiltinFunction::Gamma),
985        vec![remainder.clone()],
986    );
987    if offset > 0 {
988        let mut factors = vec![gamma_remainder];
989        for step in 0..offset {
990            let shift = builders::integer_value(context, step);
991            factors.push(simplify_sum(context, vec![remainder.clone(), shift]));
992        }
993        Some(simplify_product(context, factors))
994    } else {
995        let mut denominator_factors = Vec::new();
996        for step in offset..0 {
997            let shift = builders::integer_value(context, step);
998            denominator_factors.push(simplify_sum(context, vec![remainder.clone(), shift]));
999        }
1000        let denominator = simplify_product(context, denominator_factors);
1001        let reciprocal_denominator = reciprocal_expression(context, denominator);
1002        Some(simplify_product(
1003            context,
1004            vec![gamma_remainder, reciprocal_denominator],
1005        ))
1006    }
1007}
1008
1009fn digamma_recurrence(
1010    context: &Context,
1011    argument: &ExpressionReference,
1012) -> Option<ExpressionReference> {
1013    // psi(rest + c): c > 0 adds 1/(rest)+...+1/(rest+c-1); c < 0 subtracts the
1014    // symmetric terms. Reduces the integer offset to the canonical residue.
1015    let (offset, remainder) = split_integer_offset(context, argument)?;
1016    let digamma_remainder = simplify_function_call(
1017        context,
1018        FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
1019        vec![remainder.clone()],
1020    );
1021    let mut terms = vec![digamma_remainder];
1022    if offset > 0 {
1023        for step in 0..offset {
1024            let shift = builders::integer_value(context, step);
1025            let denominator = simplify_sum(context, vec![remainder.clone(), shift]);
1026            terms.push(reciprocal_expression(context, denominator));
1027        }
1028    } else {
1029        for step in offset..0 {
1030            let shift = builders::integer_value(context, step);
1031            let denominator = simplify_sum(context, vec![remainder.clone(), shift]);
1032            let reciprocal_term = reciprocal_expression(context, denominator);
1033            terms.push(builders::negate(context, reciprocal_term));
1034        }
1035    }
1036    Some(simplify_sum(context, terms))
1037}

1263    fn even_indexed_bernoulli_tables_match_the_direct_computation() {
1264        // The even-indexed table is served from a per-term-count memo, so each count must
1265        // come back at exactly its own length and with exactly the entries a direct
1266        // computation gives — including when a shorter count is asked for after a longer
1267        // one has already been built, and when the same count is asked for twice.
1268        for count in [0usize, 1, 7, 3, 7, 20, 3] {
1269            let table = bernoulli_even_indexed(count);
1270            let direct = bernoulli_numbers_up_to(2 * count);
1271            assert_eq!(table.len(), count);
1272            for (offset, value) in table.iter().enumerate() {
1273                assert_eq!(value, &direct[2 * (offset + 1)]);
1274            }
1275        }
