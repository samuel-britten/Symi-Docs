72/// Apply the representation-stage exact diagonal values for Carlson calls.
73pub fn simplify_call(
74    context: &Context,
75    builtin: BuiltinFunction,
76    arguments: &[ExpressionReference],
77) -> Option<ExpressionReference> {
78    let expected_arity = match builtin {
79        BuiltinFunction::CarlsonRf | BuiltinFunction::CarlsonRd => 3,
80        BuiltinFunction::CarlsonRc => 2,
81        BuiltinFunction::CarlsonRj => 4,
82        _ => return None,
83    };
84    if arguments.len() != expected_arity {
85        return None;
86    }
87    if let Some(canonical_arguments) = canonical_symmetric_arguments(context, builtin, arguments) {
88        return Some(call(context, builtin, canonical_arguments));
89    }
90    if let Some(degenerate_value) = degenerate_carlson_value(context, builtin, arguments) {
91        return Some(degenerate_value);
92    }
93    if arguments.iter().all(|argument| is_one(argument)) {
94        return Some(builders::one(context));
95    }
96    if let Some(special_value) = unit_special_value(context, builtin, arguments) {
97        return Some(special_value);
98    }
99    match builtin {
100        BuiltinFunction::CarlsonRf
101            if arguments[0] == arguments[1] && arguments[1] == arguments[2] =>
102        {
103            Some(inverse_square_root(context, arguments[0].clone()))
104        }
105        BuiltinFunction::CarlsonRc if arguments[0] == arguments[1] => {
106            Some(inverse_square_root(context, arguments[0].clone()))
107        }
108        BuiltinFunction::CarlsonRd
109            if arguments[0] == arguments[1] && arguments[1] == arguments[2] =>
110        {
111            Some(inverse_three_halves_power(context, arguments[0].clone()))
112        }
113        BuiltinFunction::CarlsonRj
114            if arguments[0] == arguments[1]
115                && arguments[1] == arguments[2]
116                && arguments[2] == arguments[3] =>
117        {
118            Some(inverse_three_halves_power(context, arguments[0].clone()))
119        }
120        _ => None,
121    }
122}

124fn degenerate_carlson_value(
125    context: &Context,
126    builtin: BuiltinFunction,
127    arguments: &[ExpressionReference],
128) -> Option<ExpressionReference> {
129    match builtin {
130        BuiltinFunction::CarlsonRf
131            if arguments[0] == arguments[1] && arguments[1] != arguments[2] =>
132        {
133            Some(carlson_rc(
134                context,
135                arguments[2].clone(),
136                arguments[0].clone(),
137            ))
138        }
139        BuiltinFunction::CarlsonRf
140            if arguments[1] == arguments[2] && arguments[0] != arguments[1] =>
141        {
142            Some(carlson_rc(
143                context,
144                arguments[0].clone(),
145                arguments[1].clone(),
146            ))
147        }
148        BuiltinFunction::CarlsonRj if arguments[3] == arguments[0] => Some(carlson_rd(
149            context,
150            arguments[1].clone(),
151            arguments[2].clone(),
152            arguments[3].clone(),
153        )),
154        BuiltinFunction::CarlsonRj if arguments[3] == arguments[1] => Some(carlson_rd(
155            context,
156            arguments[0].clone(),
157            arguments[2].clone(),
158            arguments[3].clone(),
159        )),
160        BuiltinFunction::CarlsonRj if arguments[3] == arguments[2] => Some(carlson_rd(
161            context,
162            arguments[0].clone(),
163            arguments[1].clone(),
164            arguments[3].clone(),
165        )),
166        _ => None,
167    }
168}

170fn canonical_symmetric_arguments(
171    context: &Context,
172    builtin: BuiltinFunction,
173    arguments: &[ExpressionReference],
174) -> Option<Vec<ExpressionReference>> {
175    let symmetric_prefix_length = match builtin {
176        BuiltinFunction::CarlsonRf => 3,
177        BuiltinFunction::CarlsonRc => 0,
178        BuiltinFunction::CarlsonRd => 2,
179        BuiltinFunction::CarlsonRj => 3,
180        _ => return None,
181    };
182    if symmetric_prefix_length < 2 {
183        return None;
184    }
185    let mut canonical_arguments = arguments.to_vec();
186    canonical_arguments[..symmetric_prefix_length].sort_by(|left, right| {
187        crate::expression::ordering::compare(left, right, &context.symbol_table)
188    });
189    (canonical_arguments != arguments).then_some(canonical_arguments)
190}

192fn unit_special_value(
193    context: &Context,
194    builtin: BuiltinFunction,
195    arguments: &[ExpressionReference],
196) -> Option<ExpressionReference> {
197    let pi = builders::constant(context, crate::expression::BuiltinConstant::Pi);
198    let two = builders::integer(context, exact_integer::from(2));
199    let four = builders::integer(context, exact_integer::from(4));
200    match builtin {
201        BuiltinFunction::CarlsonRf
202            if is_zero(&arguments[0]) && is_one(&arguments[1]) && is_one(&arguments[2]) =>
203        {
204            Some(builders::divide(context, pi, two))
205        }
206        BuiltinFunction::CarlsonRc if is_zero(&arguments[0]) && is_one(&arguments[1]) => {
207            Some(builders::divide(context, pi, two))
208        }
209        BuiltinFunction::CarlsonRd
210            if is_zero(&arguments[0]) && is_one(&arguments[1]) && is_one(&arguments[2]) =>
211        {
212            let three = builders::integer(context, exact_integer::from(3));
213            let numerator = builders::product(context, vec![three, pi]);
214            Some(builders::divide(context, numerator, four))
215        }
216        BuiltinFunction::CarlsonRj
217            if is_zero(&arguments[0])
218                && is_one(&arguments[1])
219                && is_one(&arguments[2])
220                && is_one(&arguments[3]) =>
221        {
222            let three = builders::integer(context, exact_integer::from(3));
223            let numerator = builders::product(context, vec![three, pi]);
224            Some(builders::divide(context, numerator, four))
225        }
226        _ => None,
227    }
228}

230fn inverse_square_root(context: &Context, argument: ExpressionReference) -> ExpressionReference {
231    let exponent = builders::rational(context, exact_integer::from(-1), exact_integer::from(2));
232    builders::power(context, argument, exponent)

243/// Differentiate exact Carlson calls with respect to their scalar arguments.
244pub fn differentiate_call(
245    context: &Context,
246    builtin: BuiltinFunction,
247    arguments: &[ExpressionReference],
248    argument_derivatives: &[ExpressionReference],
249) -> Option<ExpressionReference> {
250    if builtin != BuiltinFunction::CarlsonRf
251        || arguments.len() != 3
252        || argument_derivatives.len() != 3
253    {
254        return None;
255    }
256    let negative_sixth =
257        builders::rational(context, exact_integer::from(-1), exact_integer::from(6));
258    let partials = [
259        carlson_rd(
260            context,
261            arguments[1].clone(),
262            arguments[2].clone(),
263            arguments[0].clone(),
264        ),
265        carlson_rd(
266            context,
267            arguments[0].clone(),
268            arguments[2].clone(),
269            arguments[1].clone(),
270        ),
271        carlson_rd(
272            context,
273            arguments[0].clone(),
274            arguments[1].clone(),
275            arguments[2].clone(),
276        ),
277    ];
278    let terms = partials
279        .into_iter()
280        .zip(argument_derivatives.iter())
281        .filter(|(_, derivative)| !is_zero(derivative))
282        .map(|(partial, derivative)| {
283            simplify_product(
284                context,
285                vec![negative_sixth.clone(), partial, derivative.clone()],
286            )
287        })
288        .collect::<Vec<_>>();
289    if terms.is_empty() {
290        Some(builders::zero(context))
291    } else {
292        Some(simplify_sum(context, terms))
293    }
294}
