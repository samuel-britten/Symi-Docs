10/// The normalization used by a polynomial family's generating function.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum GeneratingFunctionNormalization {
13    /// The coefficient of the auxiliary power is the polynomial itself.
14    Ordinary,
15    /// The coefficient is divided by the degree factorial.
16    Exponential,
17}
18
19/// A formal generating-function right-hand side and its coefficient normalization.
20#[derive(Clone, Debug)]
21pub struct GeneratingFunctionRelation {
22    /// The exact closed form in the auxiliary variable.
23    pub right_hand_side: ExpressionReference,
24    /// The normalization needed to recover the polynomial coefficient.
25    pub normalization: GeneratingFunctionNormalization,
26}

28/// Builds the DLMF generating-function relation for one classical family.
29pub fn generating_function(
30    context: &Context,
31    family: PolynomialFamily,
32    parameters: &[ExpressionReference],
33    argument: ExpressionReference,
34    auxiliary_variable: ExpressionReference,
35) -> Option<GeneratingFunctionRelation> {
36    if parameters.len() != family.parameter_count() {
37        return None;
38    }
39    let one = builders::one(context);
40    let two = builders::integer_value(context, 2);
41    let auxiliary_square = simplify_power(context, auxiliary_variable.clone(), two.clone());
42    let doubled_product = simplify_product(
43        context,
44        vec![two.clone(), argument.clone(), auxiliary_variable.clone()],
45    );
46    let negated_doubled_product = builders::negate(context, doubled_product);
47    let quadratic = simplify_sum(
48        context,
49        vec![one.clone(), negated_doubled_product, auxiliary_square],
50    );
51    let right_hand_side = match family {
52        PolynomialFamily::Jacobi => {
53            let square_root = simplify_function_call(
54                context,
55                FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
56                vec![quadratic],
57            );
58            let negated_auxiliary_variable = builders::negate(context, auxiliary_variable.clone());
59            let first_base = simplify_sum(
60                context,
61                vec![one.clone(), negated_auxiliary_variable, square_root.clone()],
62            );
63            let second_base = simplify_sum(
64                context,
65                vec![one.clone(), auxiliary_variable.clone(), square_root.clone()],
66            );
67            let parameter_sum =
68                simplify_sum(context, vec![parameters[0].clone(), parameters[1].clone()]);
69            let numerator = simplify_power(context, two, parameter_sum);
70            let first_power = simplify_power(context, first_base, parameters[0].clone());
71            let second_power = simplify_power(context, second_base, parameters[1].clone());
72            let denominator =
73                simplify_product(context, vec![square_root, first_power, second_power]);
74            builders::divide(context, numerator, denominator)
75        }
76        PolynomialFamily::Gegenbauer => {
77            let negated_parameter = builders::negate(context, parameters[0].clone());
78            simplify_power(context, quadratic, negated_parameter)
79        }
80        PolynomialFamily::ChebyshevFirstKind => {
81            let product = simplify_product(context, vec![argument, auxiliary_variable]);
82            let negated_product = builders::negate(context, product);
83            let numerator = simplify_sum(context, vec![one.clone(), negated_product]);
84            builders::divide(context, numerator, quadratic)
85        }
86        PolynomialFamily::ChebyshevSecondKind => builders::divide(context, one, quadratic),
87        PolynomialFamily::Legendre => {
88            let exponent = builders::rational_value(context, -1, 2);
89            simplify_power(context, quadratic, exponent)
90        }
91        PolynomialFamily::GeneralizedLaguerre => {
92            let negated_auxiliary_variable = builders::negate(context, auxiliary_variable.clone());
93            let one_minus_auxiliary =
94                simplify_sum(context, vec![one.clone(), negated_auxiliary_variable]);
95            let exponent_inner = simplify_sum(context, vec![parameters[0].clone(), one.clone()]);
96            let exponent = builders::negate(context, exponent_inner);
97            let scale = simplify_power(context, one_minus_auxiliary.clone(), exponent);
98            let product = simplify_product(context, vec![argument, auxiliary_variable]);
99            let quotient = builders::divide(context, product, one_minus_auxiliary);
100            let exponential_argument = builders::negate(context, quotient);
101            let exponential = simplify_function_call(
102                context,
103                FunctionIdentifier::Builtin(BuiltinFunction::Exp),
104                vec![exponential_argument],
105            );
106            simplify_product(context, vec![scale, exponential])
107        }
108        PolynomialFamily::Hermite => {
109            let linear = simplify_product(context, vec![two, argument, auxiliary_variable.clone()]);
110            let exponent = builders::integer_value(context, 2);
111            let square = simplify_power(context, auxiliary_variable, exponent);
112            let negated_square = builders::negate(context, square);
113            let exponential_argument = simplify_sum(context, vec![linear, negated_square]);
114            simplify_function_call(
115                context,
116                FunctionIdentifier::Builtin(BuiltinFunction::Exp),
117                vec![exponential_argument],
118            )
119        }
120    };
121    Some(GeneratingFunctionRelation {
122        right_hand_side,
123        normalization: if family == PolynomialFamily::Hermite {
124            GeneratingFunctionNormalization::Exponential
125        } else {
126            GeneratingFunctionNormalization::Ordinary
127        },
128    })
129}

131/// Expands one direct polynomial call before Taylor-series construction when safe.
132pub(crate) fn finite_taylor_input(
133    context: &Context,
134    target: &ExpressionReference,
135    variable: SymbolIdentifier,
136) -> Option<ExpressionReference> {
137    let ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(function), arguments) =
138        &target.kind
139    else {
140        return None;
141    };
142    let family = super::family_of(*function)?;
143    let analytic_argument = arguments.get(family.analytic_argument_index())?;
144    if !matches!(&analytic_argument.kind, ExpressionKind::Symbol(identifier) if *identifier == variable)
145    {
146        return None;
147    }
148    expand_call(context, *function, arguments).ok()
149}
