12/// The largest source degree exact basis conversion will eliminate.
13pub const CONVERSION_DEGREE_BUDGET: u64 = 48;
14
15/// Why one exact basis conversion could not be answered.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum ConversionDecline {
18    /// The source is not a finite polynomial in the requested variable.
19    SourceNotPolynomial,
20    /// The source degree exceeds the elimination budget.
21    SourceDegreeBudgetExhausted,
22    /// A target basis member is beyond the exact expansion budget.
23    BasisBudgetExhausted,
24    /// A target basis leading coefficient is proved zero, so the basis is degenerate.
25    BasisLeadingCoefficientZero,
26    /// A target basis leading coefficient could not be proved nonzero.
27    BasisLeadingCoefficientUnknown,
28    /// The eliminated remainder could not be proved exactly zero.
29    ReconstructionUnverified,
30    /// The requested coefficient index lies beyond the source degree.
31    CoefficientIndexBeyondDegree,
32}

34/// The exact coefficients of a finite polynomial in one classical basis.
35///
36/// Entry `k` of the returned vector is the coefficient of the basis member of
37/// degree `k`, so the source equals the sum of those coefficients against their
38/// members. The elimination is purely algebraic: it never integrates, and it is
39/// therefore valid outside the standard positive-weight parameter region as long
40/// as the basis stays nondegenerate.
41pub fn basis_coefficients(
42    context: &Context,
43    target: &ExpressionReference,
44    variable: SymbolIdentifier,
45    basis: &OrthogonalPolynomialBasis,
46) -> Result<Vec<ExpressionReference>, ConversionDecline> {
47    // Triangular elimination from the highest degree downward. Each step divides
48    // by the target member's own leading coefficient, which is proved nonzero
49    // first, so a degenerate basis declines instead of dividing by something that
50    // may vanish. The source's own leading coefficient needs no such proof: it
51    // only ever appears in a numerator, and an unknown-zero leading coefficient
52    // simply leaves the stored degree an honest upper bound.
53    let expanded = expand_orthogonal_polynomial(context, target);
54    let mut residual =
55        ExpressionCoefficientPolynomial::from_expression(context, &expanded, variable)
56            .ok_or(ConversionDecline::SourceNotPolynomial)?;
57    residual.trim_literal_zero_leading_terms();
58    let highest = residual
59        .nominal_degree()
60        .ok_or(ConversionDecline::SourceNotPolynomial)?;
61    if highest as u64 > CONVERSION_DEGREE_BUDGET {
62        return Err(ConversionDecline::SourceDegreeBudgetExhausted);
63    }
64    let members = basis_member_table(context, basis, variable, highest)?;
65    let zero = builders::zero(context);
66    let mut coefficients = vec![zero; highest + 1];
67    for degree in (0..=highest).rev() {
68        let leading = members[degree]
69            .coefficient(degree)
70            .cloned()
71            .ok_or(ConversionDecline::BasisBudgetExhausted)?;
72        match queries::is_nonzero(&leading, &context.symbol_table) {
73            TruthValue::Yes => {}
74            TruthValue::No => return Err(ConversionDecline::BasisLeadingCoefficientZero),
75            TruthValue::Unknown => {
76                return Err(ConversionDecline::BasisLeadingCoefficientUnknown);
77            }
78        }
79        let current = residual
80            .coefficient(degree)
81            .cloned()
82            .unwrap_or_else(|| builders::zero(context));
83        if is_literal_zero(&current) {
84            continue;
85        }
86        let quotient = builders::divide(context, current, leading);
87        let scaled = members[degree].scale(context, &quotient);
88        residual = ExpressionCoefficientPolynomial::subtract(context, &residual, &scaled);
89        coefficients[degree] = quotient;
90    }
91    for coefficient in residual.coefficients() {
92        if !proved_zero(context, coefficient) {
93            return Err(ConversionDecline::ReconstructionUnverified);
94        }
95    }
96    Ok(coefficients)
97}

99/// Express a finite polynomial in a requested classical orthogonal basis.
100pub fn convert_polynomial_basis(
101    context: &Context,
102    target: &ExpressionReference,
103    variable: SymbolIdentifier,
104    basis: &OrthogonalPolynomialBasis,
105) -> Result<ExpressionReference, ConversionDecline> {
106    let coefficients = basis_coefficients(context, target, variable, basis)?;
107    let argument = context.interner.intern(ExpressionKind::Symbol(variable));
108    let mut terms = Vec::with_capacity(coefficients.len());
109    for (degree, coefficient) in coefficients.iter().enumerate() {
110        if is_literal_zero(coefficient) {
111            continue;
112        }
113        let degree_expression = builders::integer_value(context, degree as i64);
114        let member = basis.member(context, &degree_expression, &argument);
115        terms.push(simplify_product(context, vec![coefficient.clone(), member]));
116    }
117    Ok(simplify_sum(context, terms))
118}

120/// Extract one exact coefficient of a finite polynomial in a requested basis.
121pub fn orthogonal_polynomial_coefficient(
122    context: &Context,
123    target: &ExpressionReference,
124    variable: SymbolIdentifier,
125    basis: &OrthogonalPolynomialBasis,
126    degree: usize,
127) -> Result<ExpressionReference, ConversionDecline> {
128    // The cheapest exact route that is guaranteed to agree with full conversion
129    // is full conversion itself: the triangular elimination that produces the
130    // requested coefficient also produces every higher one, because each depends
131    // on all of them.
132    let coefficients = basis_coefficients(context, target, variable, basis)?;
133    coefficients
134        .get(degree)
135        .cloned()
136        .ok_or(ConversionDecline::CoefficientIndexBeyondDegree)
137}

139/// Extract one exact coefficient through the weighted inner product instead.
140///
141/// This is the theorem route of the orthogonal expansion, available only where
142/// the standard orthogonality theorem applies and the existing integration layer
143/// closes the inner product exactly. It exists as an independent verifier of the
144/// triangular elimination and as a directly useful result; it is never a fallback
145/// for a declined elimination, and it never returns a value derived from an
146/// unevaluated integral.
147pub fn coefficient_by_inner_product(
148    context: &Context,
149    target: &ExpressionReference,
150    variable: SymbolIdentifier,
151    basis: &OrthogonalPolynomialBasis,
152    degree: usize,
153) -> Option<ExpressionReference> {
154    let degree_expression = builders::integer_value(context, degree as i64);
155    let OrthogonalityVerdict::OrthogonalWithNorm { squared_norm, .. } =
156        orthogonality_theorem(context, basis, &degree_expression)
157    else {
158        return None;
159    };
160    let argument = context.interner.intern(ExpressionKind::Symbol(variable));
161    let family = basis.family();
162    let parameters = basis.parameters();
163    let weight = super::metadata::weight(context, family, &parameters, &argument);
164    let member = basis.member(context, &degree_expression, &argument);
165    let member = expand_orthogonal_polynomial(context, &member);
166    let expanded = expand_orthogonal_polynomial(context, target);
167    let integrand = simplify_product(context, vec![weight, member, expanded]);
168    let integrand = crate::expand::expand(context, &integrand);
169    let (lower, upper) = super::metadata::support_interval(context, family);
170    let integral =
171        crate::integrate::definite::integrate_definite(context, &integrand, variable, lower, upper);
172    if crate::integrate::contains_integral_placeholder(&integral) {
173        return None;
174    }
175    if !matches!(
176        queries::is_finite(&integral, &context.symbol_table),
177        TruthValue::Yes
178    ) {
179        return None;
180    }
181    Some(builders::divide(context, integral, squared_norm))
182}

184fn basis_member_table(
185    context: &Context,
186    basis: &OrthogonalPolynomialBasis,
187    variable: SymbolIdentifier,
188    highest: usize,
189) -> Result<Vec<ExpressionCoefficientPolynomial>, ConversionDecline> {
190    // Every target member the elimination can need is built once, in one pass,
191    // and reused for the rest of the request.
192    let family = basis.family();
193    let parameters = basis.parameters();
194    let mut members = Vec::with_capacity(highest + 1);
195    for degree in 0..=highest {
196        if !expansion_fits_budget(family, degree as u64) {
197            return Err(ConversionDecline::BasisBudgetExhausted);
198        }
199        let coefficients = finite_coefficients(context, family, &parameters, degree as u64)
200            .ok_or(ConversionDecline::BasisBudgetExhausted)?;
201        members.push(
202            ExpressionCoefficientPolynomial::from_ascending_coefficients(variable, coefficients),
203        );
204    }
205    Ok(members)
206}

208fn proved_zero(context: &Context, candidate: &ExpressionReference) -> bool {
209    if is_literal_zero(candidate) {
210        return true;
211    }
212    let expanded = crate::expand::expand(context, candidate);
213    if is_literal_zero(&expanded) {
214        return true;
215    }
216    matches!(
217        queries::is_zero(&expanded, &context.symbol_table),
218        TruthValue::Yes
219    )
220}

222fn is_literal_zero(candidate: &ExpressionReference) -> bool {
223    matches!(&candidate.kind, ExpressionKind::Integer(value) if value == &crate::exact_arithmetic::Integer::ZERO)
