6/// An immutable descriptor of one classical orthogonal polynomial basis.
7///
8/// The descriptor is not a scalar expression: it names a family and holds the
9/// family's parameter expressions, which belong to the context that built them.
10/// It carries no cache, and every piece of family metadata it reports comes from
11/// the shared family registry rather than from a copy stored here.
12#[derive(Clone, Debug, PartialEq, Eq)]
13pub enum OrthogonalPolynomialBasis {
14    /// The Jacobi basis \(P_n^{(\alpha,\beta)}\).
15    Jacobi {
16        /// The first Jacobi parameter \(\alpha\).
17        first_parameter: ExpressionReference,
18        /// The second Jacobi parameter \(\beta\).
19        second_parameter: ExpressionReference,
20    },
21    /// The Gegenbauer basis \(C_n^{(\lambda)}\).
22    Gegenbauer {
23        /// The Gegenbauer parameter \(\lambda\).
24        parameter: ExpressionReference,
25    },
26    /// The Chebyshev basis of the first kind \(T_n\).
27    ChebyshevFirstKind,
28    /// The Chebyshev basis of the second kind \(U_n\).
29    ChebyshevSecondKind,
30    /// The Legendre basis \(P_n\).
31    Legendre,
32    /// The generalized Laguerre basis \(L_n^{(\alpha)}\).
33    GeneralizedLaguerre {
34        /// The generalized Laguerre parameter \(\alpha\).
35        parameter: ExpressionReference,
36    },
37    /// The physicists' Hermite basis \(H_n\).
38    Hermite,
39}
40
41impl OrthogonalPolynomialBasis {
42    /// The family this basis belongs to.
43    pub fn family(&self) -> PolynomialFamily {
44        match self {
45            OrthogonalPolynomialBasis::Jacobi { .. } => PolynomialFamily::Jacobi,
46            OrthogonalPolynomialBasis::Gegenbauer { .. } => PolynomialFamily::Gegenbauer,
47            OrthogonalPolynomialBasis::ChebyshevFirstKind => PolynomialFamily::ChebyshevFirstKind,
48            OrthogonalPolynomialBasis::ChebyshevSecondKind => PolynomialFamily::ChebyshevSecondKind,
49            OrthogonalPolynomialBasis::Legendre => PolynomialFamily::Legendre,
50            OrthogonalPolynomialBasis::GeneralizedLaguerre { .. } => {
51                PolynomialFamily::GeneralizedLaguerre
52            }
53            OrthogonalPolynomialBasis::Hermite => PolynomialFamily::Hermite,
54        }
55    }
56
57    /// The family parameters, in canonical call order.
58    pub fn parameters(&self) -> Vec<ExpressionReference> {
59        match self {
60            OrthogonalPolynomialBasis::Jacobi {
61                first_parameter,
62                second_parameter,
63            } => vec![first_parameter.clone(), second_parameter.clone()],
64            OrthogonalPolynomialBasis::Gegenbauer { parameter }
65            | OrthogonalPolynomialBasis::GeneralizedLaguerre { parameter } => {
66                vec![parameter.clone()]
67            }
68            _ => Vec::new(),
69        }
70    }
71
72    /// Build a descriptor from a family and its parameters in canonical call order.
73    pub fn from_family(
74        family: PolynomialFamily,
75        parameters: &[ExpressionReference],
76    ) -> Option<Self> {
77        if parameters.len() != family.parameter_count() {
78            return None;
79        }
80        Some(match family {
81            PolynomialFamily::Jacobi => OrthogonalPolynomialBasis::Jacobi {
82                first_parameter: parameters[0].clone(),
83                second_parameter: parameters[1].clone(),
84            },
85            PolynomialFamily::Gegenbauer => OrthogonalPolynomialBasis::Gegenbauer {
86                parameter: parameters[0].clone(),
87            },
88            PolynomialFamily::ChebyshevFirstKind => OrthogonalPolynomialBasis::ChebyshevFirstKind,
89            PolynomialFamily::ChebyshevSecondKind => OrthogonalPolynomialBasis::ChebyshevSecondKind,
90            PolynomialFamily::Legendre => OrthogonalPolynomialBasis::Legendre,
91            PolynomialFamily::GeneralizedLaguerre => {
92                OrthogonalPolynomialBasis::GeneralizedLaguerre {
93                    parameter: parameters[0].clone(),
94                }
95            }
96            PolynomialFamily::Hermite => OrthogonalPolynomialBasis::Hermite,
97        })
98    }
99
100    /// Rebuild this descriptor's parameter expressions inside a target context.
101    pub fn rebuild_in(&self, source: &Context, target: &mut Context) -> Option<Self> {
102        // A descriptor built against one context is meaningless in another,
103        // because an expression reference carries that context's interner
104        // identity. Moving a descriptor therefore rebuilds every parameter
105        // through the lossless serialized form, which is the established exact
106        // route between two contexts.
107        let mut parameters = Vec::new();
108        for parameter in self.parameters() {
109            let text = crate::serialize::serialize(source, &parameter);
110            let rebuilt = crate::serialize::parse::parse_serialized(target, &text).ok()?;
111            parameters.push(rebuilt);
112        }
113        Self::from_family(self.family(), &parameters)
114    }
115
116    /// Build the named call of this basis at one degree expression.
117    pub fn member(
118        &self,
119        context: &Context,
120        degree: &ExpressionReference,
121        argument: &ExpressionReference,
122    ) -> ExpressionReference {
123        let mut arguments = vec![degree.clone()];
124        arguments.extend(self.parameters());
125        arguments.push(argument.clone());
126        super::call(context, self.family().head(), arguments)
127    }
128}
