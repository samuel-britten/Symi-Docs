183/// Builds a verified antiderivative in the polynomial argument when available.
184pub fn antiderivative_in_argument(
185    context: &Context,
186    function: BuiltinFunction,
187    arguments: &[ExpressionReference],
188) -> Option<ExpressionReference> {
189    let family = family_of(function)?;
190    if arguments.len() != family.arity() {
191        return None;
192    }
193    let degree = &arguments[0];
194    let status = classify_degree(degree, &context.symbol_table);
195    let concrete_degree = status.concrete_value()?;
196    let argument = arguments[family.analytic_argument_index()].clone();
197    match family {
198        PolynomialFamily::Hermite => {
199            let successor = shift(context, degree, 1);
200            let successor_member = hermite_polynomial(context, successor.clone(), argument);
201            let two_expression = two(context);
202            let denominator = simplify_product(context, vec![two_expression, successor]);
203            Some(builders::divide(context, successor_member, denominator))
204        }
205        PolynomialFamily::Legendre if concrete_degree > &exact_integer::ZERO => {
206            let successor = shift(context, degree, 1);
207            let predecessor = shift(context, degree, -1);
208            let successor_member = legendre_polynomial(context, successor, argument.clone());
209            let predecessor_member = legendre_polynomial(context, predecessor, argument);
210            let numerator = builders::subtract(context, successor_member, predecessor_member);
211            let two_expression = two(context);
212            let doubled_degree = simplify_product(context, vec![two_expression, degree.clone()]);
213            let one = builders::one(context);
214            let denominator = simplify_sum(context, vec![doubled_degree, one]);
215            Some(builders::divide(context, numerator, denominator))
216        }
217        _ => None,
218    }
219}

221/// Applies the automatic simplifications admitted for an orthogonal polynomial call.
222pub fn simplify_call(
223    context: &Context,
224    function: BuiltinFunction,
225    arguments: &[ExpressionReference],
226) -> Option<ExpressionReference> {
227    // Automatic simplification of a named orthogonal polynomial is deliberately
228    // narrow. A proved invalid degree collapses to `undefined` under the
229    // degree-only convention; degree zero and degree one reduce to their exact
230    // closed forms; exact endpoints, exact parity reflection, degenerate
231    // parameter values, and strictly cheaper family specializations reduce as
232    // well. Nothing here ever expands a higher-degree named head, because that
233    // is a directed operation with its own size budget.
234    let family = family_of(function)?;
235    if arguments.len() != family.arity() {
236        return None;
237    }
238    let degree = arguments[0].clone();
239    let status = classify_degree(&degree, &context.symbol_table);
240    if matches!(status, DegreeStatus::Invalid) {
241        return Some(undefined(context));
242    }
243    let parameters = arguments[1..family.analytic_argument_index()].to_vec();
244    let argument = arguments[family.analytic_argument_index()].clone();
245    // A family specialization is an identity between two heads at every degree,
246    // so it does not wait on the degree classification the value rules need.
247    if let Some(value) = specialization(context, family, &parameters, &degree, &argument) {
248        return Some(value);
249    }
250    if !status.is_admissible() {
251        return None;
252    }
253    if let Some(value) = low_degree_value(context, family, &parameters, &argument, &status) {
254        return Some(value);
255    }
256    if let Some(value) = degenerate_parameter_value(context, family, &parameters, &degree, &status)
257    {
258        return Some(value);
259    }
260    if let Some(value) = endpoint_value(context, family, &parameters, &degree, &argument, &status) {
261        return Some(value);
262    }
263    parity_reflection(context, function, &parameters, &degree, &argument, &status)
264}

266fn low_degree_value(
267    context: &Context,
268    family: PolynomialFamily,
269    parameters: &[ExpressionReference],
270    argument: &ExpressionReference,
271    status: &DegreeStatus,
272) -> Option<ExpressionReference> {
273    // Degree zero is `1` in every family; degree one is the family's exact
274    // linear polynomial. Both are cheaper than the named head, so both are
275    // globally safe automatic reductions once the degree is admissible.
276    let value = status.concrete_value()?;
277    if value == &exact_integer::ZERO {
278        return Some(builders::one(context));
279    }
280    if value != &exact_integer::ONE {
281        return None;
282    }
283    match family {
284        PolynomialFamily::Jacobi
285        | PolynomialFamily::Gegenbauer
286        | PolynomialFamily::ChebyshevFirstKind
287        | PolynomialFamily::ChebyshevSecondKind
288        | PolynomialFamily::Legendre => {
289            jacobi::degree_one_value(context, family, parameters, argument)
290        }
291        PolynomialFamily::GeneralizedLaguerre => {
292            laguerre::degree_one_value(context, parameters, argument)
293        }
294        PolynomialFamily::Hermite => Some(hermite::degree_one_value(context, argument)),
295    }
296}

334fn parity_reflection(
335    context: &Context,
336    function: BuiltinFunction,
337    parameters: &[ExpressionReference],
338    degree: &ExpressionReference,
339    argument: &ExpressionReference,
340    status: &DegreeStatus,
341) -> Option<ExpressionReference> {
342    // The reflection rules canonicalize a negated argument away. Jacobi swaps
343    // its two parameters under reflection; every other family currently supported
344    // is exactly even or odd with the parity of its degree. The rewrite fires
345    // only when the argument carries a negative leading factor, so the reflected
346    // call cannot match the rule again and no cycle exists.
347    let parity = classify_degree_parity(degree, status, &context.symbol_table)?;
348    let reflected = negated_argument(context, argument)?;
349    let sign = parity_sign(context, parity);
350    let family = family_of(function)?;
351    let value = match family {
352        PolynomialFamily::Jacobi => jacobi_polynomial(
353            context,
354            degree.clone(),
355            parameters[1].clone(),
356            parameters[0].clone(),
357            reflected,
358        ),
359        PolynomialFamily::GeneralizedLaguerre => return None,
360        _ => {
361            let mut arguments = vec![degree.clone()];
362            arguments.extend(parameters.iter().cloned());
363            arguments.push(reflected);
364            call(context, function, arguments)
365        }
366    };
367    Some(simplify_product(context, vec![sign, value]))
368}

370fn specialization(
371    context: &Context,
372    family: PolynomialFamily,
373    parameters: &[ExpressionReference],
374    degree: &ExpressionReference,
375    argument: &ExpressionReference,
376) -> Option<ExpressionReference> {
377    // Only the specializations that carry no normalization prefactor at all are
378    // automatic: each replaces a parameterized head by a strictly more specific
379    // one without introducing a quotient that could degenerate.
380    match family {
381        PolynomialFamily::Jacobi | PolynomialFamily::Gegenbauer => {
382            jacobi::exact_specialization(context, family, parameters, degree, argument)
383        }
384        _ => None,
385    }
386}

388/// Builds the exact argument derivative of an orthogonal polynomial call.
389pub fn argument_derivative(
390    context: &Context,
391    function: BuiltinFunction,
392    arguments: &[ExpressionReference],
393) -> Option<ExpressionReference> {
394    // The DLMF 18.9 argument derivatives. Every rule needs a proved positive
395    // integer degree: at degree zero the polynomial is constant and simplifies
396    // before reaching here, and a degree that is merely non-negative cannot be
397    // lowered by one without risking an invalid predecessor degree.
398    let family = family_of(function)?;
399    if arguments.len() != family.arity() {
400        return None;
401    }
402    let degree = arguments[0].clone();
403    let status = classify_degree(&degree, &context.symbol_table);
404    if !classification::degree_is_positive(&degree, &status, &context.symbol_table) {
405        return None;
406    }
407    let parameters = arguments[1..family.analytic_argument_index()].to_vec();
408    let argument = arguments[family.analytic_argument_index()].clone();
409    match family {
410        PolynomialFamily::Jacobi
411        | PolynomialFamily::Gegenbauer
412        | PolynomialFamily::ChebyshevFirstKind
413        | PolynomialFamily::ChebyshevSecondKind
414        | PolynomialFamily::Legendre => {
415            jacobi::argument_derivative(context, family, &parameters, &degree, &argument)
416        }
417        PolynomialFamily::GeneralizedLaguerre => {
418            laguerre::argument_derivative(context, &parameters, &degree, &argument)
419        }
420        PolynomialFamily::Hermite => Some(hermite::argument_derivative(
421            context, &degree, &argument, &status,
422        )),
423    }
424}

426/// Builds the classical differential-equation residual of an orthogonal polynomial call.
427pub fn differential_equation_residual(
428    context: &Context,
429    function: BuiltinFunction,
430    arguments: &[ExpressionReference],
431    variable: crate::symbol::SymbolIdentifier,
432) -> Option<ExpressionReference> {
433    // The DLMF 18.8 equations, written as the residual that the named call must
434    // annihilate. The residual is built from the ordinary derivative engine, so
435    // it is an independent statement about the head rather than a restatement of
436    // the derivative table.
437    let family = family_of(function)?;
438    if arguments.len() != family.arity() {
439        return None;
440    }
441    let degree = arguments[0].clone();
442    let status = classify_degree(&degree, &context.symbol_table);
443    if !status.is_admissible() {
444        return None;
445    }
446    let parameters = arguments[1..family.analytic_argument_index()].to_vec();
447    let argument = arguments[family.analytic_argument_index()].clone();
448    let solution = call(context, function, arguments.to_vec());
449    let first = crate::expression::differentiate::differentiate(context, &solution, variable);
450    let second = crate::expression::differentiate::differentiate(context, &first, variable);
451    let (second_coefficient, first_coefficient, value_coefficient) = match family {
452        PolynomialFamily::Jacobi
453        | PolynomialFamily::Gegenbauer
454        | PolynomialFamily::ChebyshevFirstKind
455        | PolynomialFamily::ChebyshevSecondKind
456        | PolynomialFamily::Legendre => jacobi::differential_equation_coefficients(
457            context,
458            family,
459            &parameters,
460            &degree,
461            &argument,
462        ),
463        PolynomialFamily::GeneralizedLaguerre => {
464            laguerre::differential_equation_coefficients(context, &parameters, &degree, &argument)
465        }
466        PolynomialFamily::Hermite => {
467            hermite::differential_equation_coefficients(context, &degree, &argument)
468        }
469    };
470    let second_term = simplify_product(context, vec![second_coefficient, second]);
471    let first_term = simplify_product(context, vec![first_coefficient, first]);
472    let value_term = simplify_product(context, vec![value_coefficient, solution]);
473    Some(simplify_sum(
474        context,
475        vec![second_term, first_term, value_term],
476    ))
477}

479/// Rewrites a supported orthogonal polynomial call into the hypergeometric basis.
480pub fn as_hypergeometric(
481    context: &Context,
482    function: BuiltinFunction,
483    arguments: &[ExpressionReference],
484) -> Option<ExpressionReference> {
485    // The Jacobi and generalized Laguerre bridges are directed representations
486    // only. They are emitted after the shared hypergeometric parameter
487    // classifier proves the ordinary call admissible, so a singular lower
488    // parameter leaves the finite polynomial exactly as it was.
489    let family = family_of(function)?;
490    if arguments.len() != family.arity() {
491        return None;
492    }
493    let degree = arguments[0].clone();
494    let status = classify_degree(&degree, &context.symbol_table);
495    if !status.is_admissible() {
496        return None;
497    }
498    let parameters = arguments[1..family.analytic_argument_index()].to_vec();
499    let argument = arguments[family.analytic_argument_index()].clone();
500    match family {
501        PolynomialFamily::Jacobi => {
502            jacobi::as_hypergeometric(context, &parameters, &degree, &argument)
503        }
504        PolynomialFamily::GeneralizedLaguerre => {
505            laguerre::as_hypergeometric(context, &parameters, &degree, &argument)
506        }
507        _ => None,
508    }
509}

519/// The reflected argument, when the argument carries a negative leading factor.
520pub fn negated_argument(
521    context: &Context,
522    argument: &ExpressionReference,
523) -> Option<ExpressionReference> {
524    // Reflection is a canonicalization, so it only fires on an argument whose
525    // leading numeric factor is exactly negative. Negating such an argument
526    // always produces one that is not negative-leading, which is what makes the
527    // rewrite terminate.
528    if !has_negative_leading_factor(argument) {
529        return None;
530    }
531    Some(builders::negate(context, argument.clone()))
532}
533
534fn has_negative_leading_factor(argument: &ExpressionReference) -> bool {
535    match &argument.kind {
536        ExpressionKind::Integer(value) => value < &exact_integer::ZERO,
537        ExpressionKind::Rational(value) => crate::exact_arithmetic::rational_is_negative(value),
538        ExpressionKind::Product(factors) => factors.first().is_some_and(|factor| {
539            matches!(&factor.kind, ExpressionKind::Integer(value) if value < &exact_integer::ZERO)
540                || matches!(&factor.kind, ExpressionKind::Rational(value)
541                    if crate::exact_arithmetic::rational_is_negative(value))
542        }),
543        _ => false,
544    }
545}

547/// The largest concrete degree whose factorial an exact value folds to a literal.
548pub const FACTORIAL_FOLD_DEGREE_LIMIT: u64 = 64;
549
550/// The exact factorial of a degree, folded to a literal for a small concrete degree.
551pub fn degree_factorial(context: &Context, degree: &ExpressionReference) -> ExpressionReference {
552    // The ordinary factorial builder deliberately keeps a literal factorial
553    // symbolic so that the surface form stays compact. An exact special value
554    // is a different situation: it is emitted precisely because it is cheaper
555    // than the polynomial it replaces, so a small concrete degree is folded and
556    // anything larger keeps the compact symbolic node.
557    if let ExpressionKind::Integer(value) = &degree.kind {
558        if let Some(count) = classification::small_degree(value) {
559            if count <= FACTORIAL_FOLD_DEGREE_LIMIT {
560                let one = exact_integer::ONE;
561                if let Some(folded) = crate::combinatorics::rising_factorial(&one, value) {
562                    return builders::integer(context, folded);
563                }
564            }
565        }
566    }
567    builders::factorial(context, degree.clone())
568}
