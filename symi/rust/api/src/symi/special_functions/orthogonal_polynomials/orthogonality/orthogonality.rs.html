16/// The outcome of asking whether the standard orthogonality theorem applies.
17#[derive(Clone, Debug)]
18pub enum OrthogonalityVerdict {
19    /// The theorem applies, with the exact squared norm and its proved hypotheses.
20    OrthogonalWithNorm {
21        /// The exact squared norm of the member under the standard weight.
22        squared_norm: ExpressionReference,
23        /// The hypotheses that were proved, each replayable against the assumption state.
24        conditions: Vec<OrthogonalityCondition>,
25    },
26    /// The requested measure or family is not the standard orthogonality setting.
27    NotApplicable,
28    /// A required hypothesis could not be decided from the available evidence.
29    Unknown,
30}
31
32/// Decide the standard positive-weight orthogonality theorem for one basis member.
33///
34/// A `not_applicable` or `unknown` verdict says nothing at all about whether the
35/// polynomial value itself is defined: a valid polynomial call and an applicable
36/// orthogonality theorem are separate questions with separate evidence.
37pub fn orthogonality_theorem(
38    context: &Context,
39    basis: &OrthogonalPolynomialBasis,
40    degree: &ExpressionReference,
41) -> OrthogonalityVerdict {
42    let family = basis.family();
43    let parameters = basis.parameters();
44    let status = classify_degree(degree, &context.symbol_table);
45    if matches!(status, DegreeStatus::Invalid) {
46        return OrthogonalityVerdict::NotApplicable;
47    }
48    if !status.is_admissible() {
49        return OrthogonalityVerdict::Unknown;
50    }
51    let conditions = parameter_conditions(context, family, &parameters);
52    if conditions
53        .iter()
54        .any(|condition| matches!(condition.verdict, TruthValue::No))
55    {
56        return OrthogonalityVerdict::NotApplicable;
57    }
58    if conditions
59        .iter()
60        .any(|condition| !matches!(condition.verdict, TruthValue::Yes))
61    {
62        return OrthogonalityVerdict::Unknown;
63    }
64    match squared_norm(context, family, &parameters, degree, &status) {
65        Some(squared_norm) => OrthogonalityVerdict::OrthogonalWithNorm {
66            squared_norm,
67            conditions,
68        },
69        None => OrthogonalityVerdict::Unknown,
70    }
71}

73/// Recognize a standard weighted orthogonality integral and return its exact value.
74///
75/// The recognizer fires only on the family's own standard weight over its own
76/// standard interval, with both factors members of the same basis. It returns
77/// `None` — leaving the ordinary integration pipeline to run — whenever the
78/// degrees cannot be proved equal or proved unequal, or a theorem hypothesis is
79/// unproved. Nothing here samples numerically.
80pub fn weighted_orthogonality_integral(
81    context: &Context,
82    integrand: &ExpressionReference,
83    variable: SymbolIdentifier,
84    lower_bound: &ExpressionReference,
85    upper_bound: &ExpressionReference,
86) -> Option<ExpressionReference> {
87    let recognized = recognize_weighted_product(context, integrand, variable)?;
88    let (interval_lower, interval_upper) = support_interval(context, recognized.basis.family());
89    if lower_bound != &interval_lower || upper_bound != &interval_upper {
90        return None;
91    }
92    let OrthogonalityVerdict::OrthogonalWithNorm { squared_norm, .. } =
93        orthogonality_theorem(context, &recognized.basis, &recognized.first_degree)
94    else {
95        return None;
96    };
97    let difference = builders::subtract(
98        context,
99        recognized.first_degree.clone(),
100        recognized.second_degree.clone(),
101    );
102    if matches!(
103        queries::is_nonzero(&difference, &context.symbol_table),
104        TruthValue::Yes
105    ) {
106        return Some(builders::zero(context));
107    }
108    if matches!(
109        queries::is_zero(&difference, &context.symbol_table),
110        TruthValue::Yes
111    ) {
112        return Some(squared_norm);
113    }
114    None
115}

117struct RecognizedOrthogonalityIntegral {
118    basis: OrthogonalPolynomialBasis,
119    first_degree: ExpressionReference,
120    second_degree: ExpressionReference,
121}
122
123fn recognize_weighted_product(
124    context: &Context,
125    integrand: &ExpressionReference,
126    variable: SymbolIdentifier,
127) -> Option<RecognizedOrthogonalityIntegral> {
128    // The integrand must factor as two members of one basis, evaluated at the
129    // integration variable itself, times exactly the family's standard weight.
130    // The weight is compared after both sides pass through the same canonical
131    // builders, so an equivalent presentation of the same weight matches while a
132    // different measure does not.
133    let factors = match &integrand.kind {
134        ExpressionKind::Product(factors) => factors.clone(),
135        _ => vec![Arc::clone(integrand)],
136    };
137    let mut members: Vec<(OrthogonalPolynomialBasis, ExpressionReference)> = Vec::new();
138    let mut remainder: Vec<ExpressionReference> = Vec::new();
139    for factor in factors {
140        match squared_member_of_variable(&factor, variable) {
141            Some(member) if members.is_empty() => {
142                members.push(member.clone());
143                members.push(member);
144                continue;
145            }
146            _ => {}
147        }
148        match member_of_variable(&factor, variable) {
149            Some(member) if members.len() < 2 => members.push(member),
150            _ => remainder.push(factor),
151        }
152    }
153    // A single member factor is still a same-basis pair: the degree-zero member
154    // of every family currently supported is exactly one, so automatic
155    // simplification has already replaced it. Reading the missing factor as that
156    // member is the same theorem, not a weakened one.
157    if members.len() == 1 {
158        let zero = builders::zero(context);
159        let (basis, _) = members[0].clone();
160        members.push((basis, zero));
161    }
162    if members.len() != 2 {
163        return None;
164    }
165    let (first_basis, first_degree) = members[0].clone();
166    let (second_basis, second_degree) = members[1].clone();
167    if first_basis != second_basis {
168        return None;
169    }
170    let argument = context.interner.intern(ExpressionKind::Symbol(variable));
171    let expected = weight(
172        context,
173        first_basis.family(),
174        &first_basis.parameters(),
175        &argument,
176    );
177    let supplied = simplify_product(context, remainder);
178    if supplied != expected {
179        return None;
180    }
181    Some(RecognizedOrthogonalityIntegral {
182        basis: first_basis,
183        first_degree,
184        second_degree,
185    })
186}

188fn squared_member_of_variable(
189    factor: &ExpressionReference,
190    variable: SymbolIdentifier,
191) -> Option<(OrthogonalPolynomialBasis, ExpressionReference)> {
192    let ExpressionKind::Power(base, exponent) = &factor.kind else {
193        return None;
194    };
195    if !matches!(&exponent.kind, ExpressionKind::Integer(value) if value == &crate::exact_arithmetic::Integer::from(2))
196    {
197        return None;
198    }
199    member_of_variable(base, variable)
200}
201
202fn member_of_variable(
203    factor: &ExpressionReference,
204    variable: SymbolIdentifier,
205) -> Option<(OrthogonalPolynomialBasis, ExpressionReference)> {
206    let ExpressionKind::FunctionCall(function, arguments) = &factor.kind else {
207        return None;
208    };
209    let FunctionIdentifier::Builtin(builtin) = function else {
210        return None;
211    };
212    let family = family_of(*builtin)?;
213    if arguments.len() != family.arity() {
214        return None;
215    }
216    let argument = &arguments[family.analytic_argument_index()];
217    if !matches!(&argument.kind, ExpressionKind::Symbol(name) if *name == variable) {
218        return None;
219    }
220    let parameters = arguments[1..family.analytic_argument_index()].to_vec();
221    let basis = OrthogonalPolynomialBasis::from_family(family, &parameters)?;
222    Some((basis, arguments[0].clone()))
223}
