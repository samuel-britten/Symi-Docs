1// The Lambert W function W_k, shared across the symbolic and numeric seams exactly
2// as the gamma and zeta families are. W_k is the k-th branch of the inverse of
3// w -> w * exp(w): for every defined branch, W_k(z) * exp(W_k(z)) = z.
4//
5// Two surface arities exist and mean the same family: lambert_w(z) is canonical
6// shorthand for the principal branch W_0(z), and lambert_w(k, z) with an exact
7// integer k denotes W_k(z). lambert_w(0, z) folds to the unary form at
8// construction so the two heads never both appear in a normal form, mirroring
9// hurwitz_zeta(s, 1) -> zeta(s) and polygamma(0, z) -> digamma(z).
10//
11// Branch structure follows DLMF 4.13 and Corless, Gonnet, Hare, Jeffrey & Knuth
12// (1996), which also fix the cut conventions this module implements. The
13// numeric kernel follows Johansson (2017) for its seed regions and
14// refinement schedule.
15

24// Iteration and precision caps for the numeric kernel. Every one of them makes the
25// evaluator decline deterministically rather than loop or return an unconverged
26// iterate.
27const REFINEMENT_ITERATION_CAP: usize = 64;
28const PRECISION_ESCALATION_CAP: usize = 4;
29const GUARD_PRECISION_BITS: u32 = 32;
30const SERIES_TERM_CAP: usize = 60;
31
32// Coefficients of the branch-point Puiseux expansion
33// W(z) = -1 + p - p^2/3 + 11 p^3/72 - 43 p^4/540 + 769 p^5/17280,
34// p = sqrt(2(e z + 1)) (Corless, Gonnet, Hare, Jeffrey & Knuth 1996, eq. 4.22), as
35// exact numerator/denominator pairs so the multi-precision backend divides them rather
36// than inheriting an f64 quotient.
37const PUISEUX_COEFFICIENTS: [(f64, f64); 5] = [
38    (1.0, 1.0),
39    (-1.0, 3.0),
40    (11.0, 72.0),
41    (-43.0, 540.0),
42    (769.0, 17280.0),
43];
44
45// The omega constant W_0(1), used only as an initial iterate in the pocket around z = 1
46// where the asymptotic seed degenerates. It is a seed, never a returned value: Halley
47// refinement and the residual test determine the answer's accuracy.
48const OMEGA_CONSTANT: f64 = 0.567_143_290_409_783_9;

50/// Performs the advanced native `branch_index_of` operation.
51pub fn branch_index_of(arguments: &[ExpressionReference]) -> Option<Integer> {
52    // Exact branch index of a lambert_w call in either arity: the unary form is the
53    // principal branch 0, and the binary form requires an exact integer literal first
54    // argument. A non-integer or variable branch index yields None, which every
55    // branch-sensitive consumer treats as "decline", never as "assume principal".
56    match arguments.len() {
57        1 => Some(Integer::ZERO),
58        2 => match &arguments[0].kind {
59            ExpressionKind::Integer(value) => Some(value.clone()),
60            _ => None,
61        },
62        _ => None,
63    }
64}

66/// Performs the advanced native `branch_argument_of` operation.
67pub fn branch_argument_of(arguments: &[ExpressionReference]) -> Option<&ExpressionReference> {
68    // The z of W_k(z), independent of which arity carried it.
69    match arguments.len() {
70        1 => Some(&arguments[0]),
71        2 => Some(&arguments[1]),
72        _ => None,
73    }
74}

76/// Performs the advanced native `build_branch_call` operation.
77pub fn build_branch_call(
78    context: &Context,
79    branch: i64,
80    argument: ExpressionReference,
81) -> ExpressionReference {
82    // Build W_branch(argument) through the simplifying builder, choosing the canonical
83    // arity: branch 0 becomes the unary principal form, every other branch the binary
84    // form. Special values fold automatically because the builder routes through
85    // simplify_call.
86    let mut arguments = Vec::with_capacity(2);
87    if branch != 0 {
88        arguments.push(crate::expression::builders::integer_value(context, branch));
89    }
90    arguments.push(argument);
91    crate::expression::builders::function_call(
92        context,
93        FunctionIdentifier::Builtin(BuiltinFunction::LambertW),
94        arguments,
95    )
96}

98/// Performs the advanced native `simplify_call` operation.
99pub fn simplify_call(
100    context: &Context,
101    builtin: BuiltinFunction,
102    arguments: &[ExpressionReference],
103) -> Option<ExpressionReference> {
104    // Canonicalisation and exact special values for both lambert_w arities.
105    //
106    // lambert_w(0, z) rewrites to lambert_w(z) so the principal branch has exactly one
107    // normal form. The exact values fold only on the branches where they hold:
108    // W_0(0) = 0, W_0(e) = 1, W_0(-1/e) = -1 and W_-1(-1/e) = -1 (the two real branches
109    // meet at the branch point). W_k(0) for k != 0 is a logarithmic singularity, so it
110    // is deliberately left symbolic — folding it to a finite value would be wrong.
111    // A non-integer or symbolic branch index folds nothing at all.
112    if builtin != BuiltinFunction::LambertW {
113        return None;
114    }
115    if arguments.len() == 2 {
116        if let ExpressionKind::Integer(value) = &arguments[0].kind {
117            if *value == Integer::ZERO {
118                return Some(crate::expression::builders::function_call(
119                    context,
120                    FunctionIdentifier::Builtin(BuiltinFunction::LambertW),
121                    vec![std::sync::Arc::clone(&arguments[1])],
122                ));
123            }
124        }
125    }
126    let branch = branch_index_of(arguments)?;
127    let argument = branch_argument_of(arguments)?;
128    let principal = branch == Integer::ZERO;
129    if principal && is_integer_zero(argument) {
130        return Some(crate::expression::builders::zero(context));
131    }
132    if principal && is_constant_e(argument) {
133        return Some(crate::expression::builders::integer_value(context, 1));
134    }
135    let at_branch_point = is_negative_reciprocal_of_e(argument);
136    if at_branch_point && (principal || branch == Integer::from(-1)) {
137        return Some(crate::expression::builders::integer_value(context, -1));
138    }
139    None
140}

142/// Performs the advanced native `differentiate_call` operation.
143pub fn differentiate_call(
144    context: &Context,
145    arguments: &[ExpressionReference],
146) -> Option<ExpressionReference> {
147    // d/dz W_k(z) = W_k(z) / (z * (1 + W_k(z))), valid on every branch away from the
148    // branch point and from z = 0. The identity is derived from W_k(z) exp(W_k(z)) = z
149    // by implicit differentiation, so it is branch-independent in form; the branch index
150    // is carried through unchanged. Declines on a malformed branch index so a discrete
151    // branch label is never treated as a differentiable quantity.
152    let branch = branch_index_of(arguments)?;
153    let argument = branch_argument_of(arguments)?.clone();
154    let branch_value = i64::try_from(&branch).ok()?;
155    let lambert_call = build_branch_call(context, branch_value, argument.clone());
156    let one = crate::expression::builders::integer_value(context, 1);
157    let one_plus_w = crate::simplify::simplify_sum(context, vec![one, lambert_call.clone()]);
158    let denominator = crate::simplify::simplify_product(context, vec![argument, one_plus_w]);
159    let minus_one = crate::expression::builders::integer_value(context, -1);
160    let reciprocal_denominator = crate::simplify::simplify_power(context, denominator, minus_one);
161    Some(crate::simplify::simplify_product(
162        context,
163        vec![lambert_call, reciprocal_denominator],
164    ))
165}

191fn is_negative_reciprocal_of_e(reference: &ExpressionReference) -> bool {
192    // Predicate identifying the canonical form of -1/e:
193    // product([integer(-1), exp(integer(-1))]) — base-e powers canonicalize to exp
194    // calls at construction, so this structural match on an ASAE argument is exact.
195    if let ExpressionKind::Product(factors) = &reference.kind {
196        if factors.len() != 2 || !is_integer_minus_one(&factors[0]) {
197            return false;
198        }
199        if let ExpressionKind::FunctionCall(
200            FunctionIdentifier::Builtin(BuiltinFunction::Exp),
201            inner_arguments,
202        ) = &factors[1].kind
203        {
204            return inner_arguments.len() == 1 && is_integer_minus_one(&inner_arguments[0]);
205        }
206    }
207    false
208}

210/// Performs the advanced native `evaluate_real` operation.
211pub fn evaluate_real(
212    branch: i64,
213    argument: &RealInexactNumber,
214    evaluator: &mut NumericEvaluator,
215) -> Option<RealInexactNumber> {
216    // Multi-precision exactly-real Lambert W on the two branches that take real values:
217    // W_0 on [-1/e, +infinity) and W_-1 on [-1/e, 0). Every other branch, and every
218    // argument outside those intervals, declines here so the caller routes through the
219    // complex kernel.
220    //
221    // The whole computation stays in real arithmetic rather than projecting a complex
222    // result, which is what makes the real/complex coherence guarantee exact: a real
223    // branch returns a value whose imaginary part is zero by construction, not by
224    // rounding. Precision is escalated by guard digits and the result is accepted only
225    // when the defining residual and the branch's own real-range fact both hold.
226    if branch != 0 && branch != -1 {
227        return None;
228    }
229    let requested = evaluator.requested_precision_bits();
230    let mut working = requested + GUARD_PRECISION_BITS;
231    for _ in 0..PRECISION_ESCALATION_CAP {
232        let mut inner = NumericEvaluator::new(working)?;
233        let promoted = RealInexactNumber::from_big_float(argument.raw_big_float().clone(), working);
234        if !real_argument_is_in_branch_domain(branch, &promoted, &mut inner)? {
235            return None;
236        }
237        if let Some(value) = evaluate_real_at_precision(branch, &promoted, &mut inner) {
238            if real_value_is_acceptable(branch, &promoted, &value, &mut inner, requested) {
239                return Some(RealInexactNumber::from_big_float(
240                    value.raw_big_float().clone(),
241                    requested,
242                ));
243            }
244        }
245        working += requested.max(GUARD_PRECISION_BITS);
246    }
247    None
248}

250fn real_argument_is_in_branch_domain(
251    branch: i64,
252    argument: &RealInexactNumber,
253    evaluator: &mut NumericEvaluator,
254) -> Option<bool> {
255    // W_0 is real exactly on [-1/e, +infinity); W_-1 is real exactly on [-1/e, 0). An
256    // undecidable comparison returns None so the caller declines rather than guessing
257    // which side of the branch point an argument lies on.
258    let branch_point = negative_reciprocal_of_e(evaluator);
259    if argument.partial_compare(&branch_point)? == std::cmp::Ordering::Less {
260        return Some(false);
261    }
262    if branch == -1 {
263        let zero = RealInexactNumber::zero(evaluator.requested_precision_bits());
264        if argument.partial_compare(&zero)? != std::cmp::Ordering::Less {
265            return Some(false);
266        }
267    }
268    Some(true)
269}

271fn evaluate_real_at_precision(
272    branch: i64,
273    argument: &RealInexactNumber,
274    evaluator: &mut NumericEvaluator,
275) -> Option<RealInexactNumber> {
276    // One real evaluation pass at the evaluator's working precision: the exceptional
277    // points, then a seed, then Halley refinement.
278    let bits = evaluator.requested_precision_bits();
279    if argument.is_zero() {
280        return if branch == 0 {
281            Some(RealInexactNumber::zero(bits))
282        } else {
283            None
284        };
285    }
286    let branch_point = negative_reciprocal_of_e(evaluator);
287    if argument.subtract(&branch_point, evaluator).is_zero() {
288        return Some(RealInexactNumber::from_f64(-1.0, bits));
289    }
290    let seed = real_seed(branch, argument, evaluator)?;
291    refine_real_by_halley(argument, seed, evaluator)
292}

294fn real_seed(
295    branch: i64,
296    argument: &RealInexactNumber,
297    evaluator: &mut NumericEvaluator,
298) -> Option<RealInexactNumber> {
299    // Branch-aware real initial iterate, following Corless et al. section 5 — the same
300    // region split the shipped f64 principal evaluator uses, extended to W_-1:
301    //
302    //   W_0,  x < -0.3       : Puiseux series in p = sqrt(2(e x + 1));
303    //   W_0,  -0.3 <= x <= e : x itself, which sits inside Halley's basin there;
304    //   W_0,  x > e          : L1 - L2 + L2/L1 with L1 = ln x, L2 = ln L1;
305    //   W_-1, x <= -0.2      : the same Puiseux series with the negated square root,
306    //                          which is exactly the second real branch;
307    //   W_-1, -0.2 < x < 0   : L1 - L2 + L2/L1 with L1 = ln(-x), L2 = ln(-L1), the real
308    //                          form of the asymptotic seed on the lower real branch.
309    let bits = evaluator.requested_precision_bits();
310    let threshold = RealInexactNumber::from_f64(if branch == 0 { -0.3 } else { -0.2 }, bits);
311    let below_threshold = argument.partial_compare(&threshold)? == std::cmp::Ordering::Less;
312    if below_threshold {
313        return real_puiseux_seed(branch, argument, evaluator);
314    }
315    if branch == -1 {
316        let first = argument.negate().natural_logarithm(evaluator);
317        let second = first.negate().natural_logarithm(evaluator);
318        if !first.is_finite() || !second.is_finite() || first.is_zero() {
319            return None;
320        }
321        let quotient = second.divide(&first, evaluator);
322        return Some(first.subtract(&second, evaluator).add(&quotient, evaluator));
323    }
324    let e_value = RealInexactNumber::e(evaluator);
325    if argument.partial_compare(&e_value)? != std::cmp::Ordering::Greater {
326        return Some(argument.clone());
327    }
328    let first = argument.natural_logarithm(evaluator);
329    let second = first.natural_logarithm(evaluator);
330    if !first.is_finite() || !second.is_finite() || first.is_zero() {
331        return None;
332    }
333    let quotient = second.divide(&first, evaluator);
334    Some(first.subtract(&second, evaluator).add(&quotient, evaluator))
335}

337fn real_puiseux_seed(
338    branch: i64,
339    argument: &RealInexactNumber,
340    evaluator: &mut NumericEvaluator,
341) -> Option<RealInexactNumber> {
342    // W(x) = -1 + p - p^2/3 + 11 p^3/72 - 43 p^4/540 + 769 p^5/17280 with
343    // p = sqrt(2(e x + 1)) (Corless et al. eq. 4.22). The positive square root is W_0 and
344    // its negation is W_-1: the two real branches are the two signs of the same series.
345    let bits = evaluator.requested_precision_bits();
346    let e_value = RealInexactNumber::e(evaluator);
347    let one = RealInexactNumber::one(bits);
348    let two = RealInexactNumber::from_f64(2.0, bits);
349    let scaled = e_value.multiply(argument, evaluator).add(&one, evaluator);
350    let radicand = two.multiply(&scaled, evaluator);
351    if radicand.is_negative() {
352        return None;
353    }
354    let mut p = radicand.square_root(evaluator);
355    if branch == -1 {
356        p = p.negate();
357    }
358    let mut total = RealInexactNumber::from_f64(-1.0, bits);
359    let mut power = one;
360    for (numerator, denominator) in PUISEUX_COEFFICIENTS {
361        power = power.multiply(&p, evaluator);
362        let coefficient = RealInexactNumber::from_f64(numerator, bits)
363            .divide(&RealInexactNumber::from_f64(denominator, bits), evaluator);
364        total = total.add(&coefficient.multiply(&power, evaluator), evaluator);
365    }
366    Some(total)
367}

369fn refine_real_by_halley(
370    argument: &RealInexactNumber,
371    seed: RealInexactNumber,
372    evaluator: &mut NumericEvaluator,
373) -> Option<RealInexactNumber> {
374    // Halley's cubically convergent iteration for the root of f(w) = w exp(w) - x in the
375    // closed form of Corless et al. eq. 5.9. The loop stops on a vanishing step or after
376    // a fixed cap, so it always terminates; a non-finite iterate declines.
377    let bits = evaluator.requested_precision_bits();
378    let one = RealInexactNumber::one(bits);
379    let two = RealInexactNumber::from_f64(2.0, bits);
380    let mut w = seed;
381    for _ in 0..REFINEMENT_ITERATION_CAP {
382        if !w.is_finite() {
383            return None;
384        }
385        let exponential = w.exp(evaluator);
386        let value = w
387            .multiply(&exponential, evaluator)
388            .subtract(argument, evaluator);
389        if value.is_zero() {
390            return Some(w);
391        }
392        let w_plus_one = w.add(&one, evaluator);
393        let doubled = w_plus_one.add(&w_plus_one, evaluator);
394        if doubled.is_zero() {
395            return None;
396        }
397        let correction = w
398            .add(&two, evaluator)
399            .multiply(&value, evaluator)
400            .divide(&doubled, evaluator);
401        let denominator = exponential
402            .multiply(&w_plus_one, evaluator)
403            .subtract(&correction, evaluator);
404        if denominator.is_zero() || !denominator.is_finite() {
405            return None;
406        }
407        let next = w.subtract(&value.divide(&denominator, evaluator), evaluator);
408        if next.subtract(&w, evaluator).is_zero() {
409            return Some(next);
410        }
411        w = next;
412    }
413    Some(w)
414}

416fn real_value_is_acceptable(
417    branch: i64,
418    argument: &RealInexactNumber,
419    value: &RealInexactNumber,
420    evaluator: &mut NumericEvaluator,
421    requested_precision_bits: u32,
422) -> bool {
423    // Accept a real iterate only when the defining identity w exp(w) - x is within the
424    // precision-relative tolerance and the value lies in the branch's own real range:
425    // W_0 >= -1 and W_-1 <= -1, the two halves the branch point separates. The range test
426    // is what distinguishes the two branches, since both satisfy the same identity.
427    if !value.is_finite() {
428        return false;
429    }
430    let residual = value
431        .multiply(&value.exp(evaluator), evaluator)
432        .subtract(argument, evaluator)
433        .absolute_value();
434    if !residual_within_tolerance(&residual, argument, evaluator, requested_precision_bits) {
435        return false;
436    }
437    let minus_one = RealInexactNumber::from_f64(-1.0, evaluator.requested_precision_bits());
438    match value.partial_compare(&minus_one) {
439        Some(std::cmp::Ordering::Less) => branch == -1,
440        Some(_) => branch == 0,
441        None => false,
442    }
443}

445fn residual_within_tolerance(
446    residual: &RealInexactNumber,
447    argument: &RealInexactNumber,
448    evaluator: &mut NumericEvaluator,
449    requested_precision_bits: u32,
450) -> bool {
451    // The shared acceptance predicate: |residual| <= (1 + |x|) * 2^-(precision - 8). The
452    // absolute scale keeps a near-zero argument from demanding an unattainable relative
453    // accuracy; the eight-bit margin absorbs the rounding of the residual evaluation
454    // itself without ever being loose enough to admit a different branch, whose residual
455    // is of order |x|.
456    let bits = evaluator.requested_precision_bits();
457    let one = RealInexactNumber::one(bits);
458    let scale = argument.absolute_value().add(&one, evaluator);
459    let exponent = requested_precision_bits.saturating_sub(8).min(1024);
460    let mut divisor = one.clone();
461    let two = RealInexactNumber::from_f64(2.0, bits);
462    for _ in 0..exponent {
463        divisor = divisor.multiply(&two, evaluator);
464    }
465    let tolerance = scale.divide(&divisor, evaluator);
466    matches!(
467        residual.partial_compare(&tolerance),
468        Some(std::cmp::Ordering::Less) | Some(std::cmp::Ordering::Equal)
469    )
470}

472/// Performs the advanced native `evaluate_complex` operation.
473pub fn evaluate_complex(
474    branch: i64,
475    argument: &ComplexInexactNumber,
476    evaluator: &mut NumericEvaluator,
477) -> Option<ComplexInexactNumber> {
478    // Branch-indexed arbitrary-precision W_k(z), following Corless et al. (1996) and
479    // Johansson (2017): an exactly-real argument on a real branch is answered by the real
480    // kernel above; otherwise exact handling at the two exceptional points, a Puiseux
481    // expansion in sqrt(2(e z + 1)) near the shared W_0/W_-1 branch point, a convergent
482    // power series near zero for the principal branch, and the all-branch asymptotic seed
483    // L1 - L2 + L2/L1 elsewhere; then Halley refinement at guard precision.
484    //
485    // Acceptance requires a defining-identity residual below the precision margin and a
486    // branch-region test, so a converged-but-wrong-branch iterate is rejected rather than
487    // returned. Halley's iteration converges to whichever branch its seed was nearest, so
488    // convergence alone proves nothing about which W_k was computed. This is a point
489    // evaluator; it does not claim interval certification.
490    if let Some(real_argument) = argument.project_to_real() {
491        if let Some(real_value) = evaluate_real(branch, &real_argument, evaluator) {
492            return Some(ComplexInexactNumber::from_real(real_value));
493        }
494    }
495    let requested = evaluator.requested_precision_bits();
496    let mut working = requested + GUARD_PRECISION_BITS;
497    for _ in 0..PRECISION_ESCALATION_CAP {
498        let mut inner = NumericEvaluator::new(working)?;
499        let promoted = promote_precision(argument, working);
500        if let Some(value) = evaluate_complex_at_precision(branch, &promoted, &mut inner) {
501            if complex_value_is_acceptable(branch, &promoted, &value, &mut inner, requested) {
502                return Some(promote_precision(&value, requested));
503            }
504        }
505        working += requested.max(GUARD_PRECISION_BITS);
506    }
507    None
508}

510fn promote_precision(argument: &ComplexInexactNumber, precision_bits: u32) -> ComplexInexactNumber {
511    // Re-express a complex value at the given precision so subsequent operations carry
512    // guard digits rather than inheriting the caller's precision, and so the returned
513    // value is reported at the precision the caller asked for.
514    ComplexInexactNumber::new(
515        RealInexactNumber::from_big_float(
516            argument.real_part().raw_big_float().clone(),
517            precision_bits,
518        ),
519        RealInexactNumber::from_big_float(
520            argument.imaginary_part().raw_big_float().clone(),
521            precision_bits,
522        ),
523    )
524}

526fn evaluate_complex_at_precision(
527    branch: i64,
528    argument: &ComplexInexactNumber,
529    evaluator: &mut NumericEvaluator,
530) -> Option<ComplexInexactNumber> {
531    // One complex evaluation pass at the evaluator's working precision: the exceptional
532    // points, then a seed, then Halley refinement. Split out from evaluate_complex so the
533    // precision-escalation loop can repeat the whole pass at a higher precision.
534    if let Some(exact) = complex_exceptional_point_value(branch, argument, evaluator) {
535        return Some(exact);
536    }
537    let seed = complex_seed(branch, argument, evaluator)?;
538    refine_complex_by_halley(argument, seed, evaluator)
539}

541fn complex_exceptional_point_value(
542    branch: i64,
543    argument: &ComplexInexactNumber,
544    evaluator: &mut NumericEvaluator,
545) -> Option<ComplexInexactNumber> {
546    // The two points where the general machinery has no valid seed. W_0(0) = 0 exactly;
547    // W_k(0) for k != 0 is a logarithmic singularity, so it declines rather than folding
548    // to a finite value. At the branch point z = -1/e the two real branches meet at -1,
549    // which the Puiseux expansion would only approach.
550    let bits = evaluator.requested_precision_bits();
551    if argument.is_zero() {
552        return if branch == 0 {
553            Some(ComplexInexactNumber::from_real(RealInexactNumber::zero(
554                bits,
555            )))
556        } else {
557            None
558        };
559    }
560    if branch != 0 && branch != -1 {
561        return None;
562    }
563    let branch_point = ComplexInexactNumber::from_real(negative_reciprocal_of_e(evaluator));
564    if argument.subtract(&branch_point, evaluator).is_zero() {
565        return Some(ComplexInexactNumber::from_real(
566            RealInexactNumber::from_f64(-1.0, bits),
567        ));
568    }
569    None
570}

572fn complex_seed(
573    branch: i64,
574    argument: &ComplexInexactNumber,
575    evaluator: &mut NumericEvaluator,
576) -> Option<ComplexInexactNumber> {
577    // Region dispatch for the initial complex iterate:
578    //
579    //  - near the shared W_0/W_-1 branch point, the Puiseux series in
580    //    p = sqrt(2(e z + 1)), whose principal square root is W_0 and whose negation
581    //    is W_-1;
582    //  - near the origin on the principal branch, the convergent Taylor series
583    //    sum_{n>=1} (-n)^(n-1)/n! z^n;
584    //  - on the principal branch where log z is small — the pocket around z = 1 in which
585    //    the asymptotic seed degenerates — the constant W_0(1) = Omega, which sits inside
586    //    Halley's basin throughout that pocket;
587    //  - everywhere else, the all-branch asymptotic seed L1 - L2 + L2/L1 with
588    //    L1 = log z + 2 pi i k and L2 = log L1. This is the only seed carrying the branch
589    //    index, so it is what makes every integer branch reachable.
590    if (branch == 0 || branch == -1) && within_radius_of_branch_point(argument, evaluator) {
591        return complex_puiseux_seed(branch, argument, evaluator);
592    }
593    if branch == 0 && within_radius_of_origin(argument, evaluator) {
594        return complex_taylor_seed(argument, evaluator);
595    }
596    complex_asymptotic_seed(branch, argument, evaluator)
597}
598
599fn within_radius_of_branch_point(
600    argument: &ComplexInexactNumber,
601    evaluator: &mut NumericEvaluator,
602) -> bool {
603    // True inside the disc of radius 1/4 about -1/e, where the asymptotic seed is useless
604    // because the two branches collide and the Puiseux expansion converges well.
605    let branch_point = ComplexInexactNumber::from_real(negative_reciprocal_of_e(evaluator));
606    let magnitude = argument
607        .subtract(&branch_point, evaluator)
608        .magnitude(evaluator);
609    magnitude_is_below(&magnitude, 0.25, evaluator)
610}
611
612fn within_radius_of_origin(
613    argument: &ComplexInexactNumber,
614    evaluator: &mut NumericEvaluator,
615) -> bool {
616    // True inside the disc of radius 1/4 about zero, comfortably within the Taylor
617    // series' radius of convergence 1/e.
618    let magnitude = argument.magnitude(evaluator);
619    magnitude_is_below(&magnitude, 0.25, evaluator)
620}
621
622fn magnitude_is_below(
623    magnitude: &RealInexactNumber,
624    bound: f64,
625    evaluator: &NumericEvaluator,
626) -> bool {
627    // Strict comparison of a magnitude against a fixed region radius. An undecidable
628    // comparison reports false, which only moves the point to a different seed region.
629    let threshold = RealInexactNumber::from_f64(bound, evaluator.requested_precision_bits());
630    matches!(
631        magnitude.partial_compare(&threshold),
632        Some(std::cmp::Ordering::Less)
633    )
634}

636fn complex_puiseux_seed(
637    branch: i64,
638    argument: &ComplexInexactNumber,
639    evaluator: &mut NumericEvaluator,
640) -> Option<ComplexInexactNumber> {
641    // The complex form of the branch-point expansion; see real_puiseux_seed for the
642    // series and its branch-sign convention.
643    let bits = evaluator.requested_precision_bits();
644    let e_complex = ComplexInexactNumber::from_real(RealInexactNumber::e(evaluator));
645    let one = ComplexInexactNumber::from_real(RealInexactNumber::one(bits));
646    let two = ComplexInexactNumber::from_real(RealInexactNumber::from_f64(2.0, bits));
647    let shifted = e_complex.multiply(argument, evaluator).add(&one, evaluator);
648    let mut p = two.multiply(&shifted, evaluator).square_root(evaluator);
649    if branch == -1 {
650        p = p.negate();
651    }
652    let mut total = ComplexInexactNumber::from_real(RealInexactNumber::from_f64(-1.0, bits));
653    let mut power = one;
654    for (numerator, denominator) in PUISEUX_COEFFICIENTS {
655        power = power.multiply(&p, evaluator);
656        let coefficient = ComplexInexactNumber::from_real(
657            RealInexactNumber::from_f64(numerator, bits)
658                .divide(&RealInexactNumber::from_f64(denominator, bits), evaluator),
659        );
660        total = total.add(&coefficient.multiply(&power, evaluator), evaluator);
661    }
662    Some(total)
663}
664
665fn complex_taylor_seed(
666    argument: &ComplexInexactNumber,
667    evaluator: &mut NumericEvaluator,
668) -> Option<ComplexInexactNumber> {
669    // W_0(z) = sum_{n>=1} (-n)^(n-1) / n! * z^n, convergent for |z| < 1/e. Summed until a
670    // term stops changing the partial sum or the term cap is reached; the cap makes the
671    // seed bounded work rather than a convergence gamble, and an imperfect seed is
672    // harmless because Halley refinement follows.
673    let bits = evaluator.requested_precision_bits();
674    let mut total = ComplexInexactNumber::from_real(RealInexactNumber::zero(bits));
675    let mut power = ComplexInexactNumber::from_real(RealInexactNumber::one(bits));
676    let mut factorial = RealInexactNumber::one(bits);
677    for n in 1..=SERIES_TERM_CAP {
678        power = power.multiply(argument, evaluator);
679        factorial = factorial.multiply(&RealInexactNumber::from_f64(n as f64, bits), evaluator);
680        let base = RealInexactNumber::from_f64(-(n as f64), bits);
681        let mut coefficient = RealInexactNumber::one(bits);
682        for _ in 0..(n - 1) {
683            coefficient = coefficient.multiply(&base, evaluator);
684        }
685        let scale = coefficient.divide(&factorial, evaluator);
686        let term = ComplexInexactNumber::from_real(scale).multiply(&power, evaluator);
687        let next = total.add(&term, evaluator);
688        if next.subtract(&total, evaluator).is_zero() {
689            return Some(next);
690        }
691        total = next;
692    }
693    Some(total)
694}

696fn complex_asymptotic_seed(
697    branch: i64,
698    argument: &ComplexInexactNumber,
699    evaluator: &mut NumericEvaluator,
700) -> Option<ComplexInexactNumber> {
701    // W_k(z) ~ L1 - L2 + L2/L1 with L1 = log z + 2 pi i k and L2 = log L1 (Corless et al.
702    // eq. 4.19). On the principal branch L1 vanishes at z = 1 and is small in a pocket
703    // around it, where the expansion is worthless; that pocket is seeded with Omega
704    // instead. On every other branch |L1| >= 2 pi - pi = pi, so the expansion is always
705    // usable there.
706    let bits = evaluator.requested_precision_bits();
707    let logarithm = argument.natural_logarithm(evaluator)?;
708    let pi_value = RealInexactNumber::pi(evaluator);
709    let two_pi = pi_value.add(&pi_value, evaluator);
710    let offset_imaginary =
711        two_pi.multiply(&RealInexactNumber::from_f64(branch as f64, bits), evaluator);
712    let offset = ComplexInexactNumber::new(RealInexactNumber::zero(bits), offset_imaginary);
713    let first = logarithm.add(&offset, evaluator);
714    let first_magnitude = first.magnitude(evaluator);
715    if branch == 0 && !magnitude_is_at_least(&first_magnitude, 1.0, evaluator) {
716        return Some(ComplexInexactNumber::from_real(
717            RealInexactNumber::from_f64(OMEGA_CONSTANT, bits),
718        ));
719    }
720    if first.is_zero() {
721        return None;
722    }
723    let second = first.natural_logarithm(evaluator)?;
724    let quotient = second.divide(&first, evaluator)?;
725    Some(first.subtract(&second, evaluator).add(&quotient, evaluator))
726}
727
728fn magnitude_is_at_least(
729    magnitude: &RealInexactNumber,
730    bound: f64,
731    evaluator: &NumericEvaluator,
732) -> bool {
733    // Non-strict comparison of a magnitude against a fixed region radius; an undecidable
734    // comparison reports false, which only moves the point to a different seed region.
735    let threshold = RealInexactNumber::from_f64(bound, evaluator.requested_precision_bits());
736    matches!(
737        magnitude.partial_compare(&threshold),
738        Some(std::cmp::Ordering::Greater) | Some(std::cmp::Ordering::Equal)
739    )
740}

742fn refine_complex_by_halley(
743    argument: &ComplexInexactNumber,
744    seed: ComplexInexactNumber,
745    evaluator: &mut NumericEvaluator,
746) -> Option<ComplexInexactNumber> {
747    // Halley's cubically convergent iteration for the root of f(w) = w exp(w) - z, in the
748    // closed form of Corless et al. eq. 5.9. The loop stops on a vanishing step or after a
749    // fixed cap, so it always terminates; an iterate that leaves the finite plane declines.
750    let bits = evaluator.requested_precision_bits();
751    let one = ComplexInexactNumber::from_real(RealInexactNumber::one(bits));
752    let two = ComplexInexactNumber::from_real(RealInexactNumber::from_f64(2.0, bits));
753    let mut w = seed;
754    for _ in 0..REFINEMENT_ITERATION_CAP {
755        if !w.is_finite() {
756            return None;
757        }
758        let exponential = w.exponential(evaluator);
759        let value = w
760            .multiply(&exponential, evaluator)
761            .subtract(argument, evaluator);
762        if value.is_zero() {
763            return Some(w);
764        }
765        let w_plus_one = w.add(&one, evaluator);
766        let doubled = w_plus_one.add(&w_plus_one, evaluator);
767        let correction = w
768            .add(&two, evaluator)
769            .multiply(&value, evaluator)
770            .divide(&doubled, evaluator)?;
771        let denominator = exponential
772            .multiply(&w_plus_one, evaluator)
773            .subtract(&correction, evaluator);
774        let step = value.divide(&denominator, evaluator)?;
775        let next = w.subtract(&step, evaluator);
776        if next.subtract(&w, evaluator).is_zero() {
777            return Some(next);
778        }
779        w = next;
780    }
781    Some(w)
782}

784fn complex_value_is_acceptable(
785    branch: i64,
786    argument: &ComplexInexactNumber,
787    value: &ComplexInexactNumber,
788    evaluator: &mut NumericEvaluator,
789    requested_precision_bits: u32,
790) -> bool {
791    // Accept an iterate only when the defining identity W exp(W) - z is within the
792    // precision-relative tolerance and the value lies in branch k's own image region.
793    if !value.is_finite() {
794        return false;
795    }
796    let residual = value
797        .multiply(&value.exponential(evaluator), evaluator)
798        .subtract(argument, evaluator)
799        .magnitude(evaluator);
800    let argument_magnitude = argument.magnitude(evaluator);
801    if !residual_within_tolerance(
802        &residual,
803        &argument_magnitude,
804        evaluator,
805        requested_precision_bits,
806    ) {
807        return false;
808    }
809    complex_value_is_on_the_requested_branch(branch, argument, value, evaluator)
810}
811
812fn complex_value_is_on_the_requested_branch(
813    branch: i64,
814    argument: &ComplexInexactNumber,
815    value: &ComplexInexactNumber,
816    evaluator: &mut NumericEvaluator,
817) -> bool {
818    // Certify *which* branch the converged iterate belongs to, rather than merely that it
819    // solves the defining equation — Halley converges to whichever branch its seed was
820    // nearest, so the residual alone proves nothing about the label.
821    //
822    // The labelling itself supplies the test. Taking principal logarithms of
823    // w exp(w) = z gives log(w) + w = log(z) + 2*pi*i*k, and that integer k *is* the
824    // branch index in the standard counterclockwise-continuous convention of Corless et
825    // al. So recovering
826    //
827    //     k = (w + log(w) - log(z)) / (2*pi*i)
828    //
829    // and checking it rounds to the requested branch is an exact identity test, not a
830    // geometric approximation. It is also the only formulation that stays correct on the
831    // cuts, where the branch images shift: on the negative real axis W_-1(x) equals the
832    // conjugate of W_0(x), which no fixed horizontal strip contains.
833    //
834    // The acceptance window is a quarter turn: the recovered index is an exact integer up
835    // to rounding, and adjacent branches differ by a full unit, so it can never confuse
836    // neighbours.
837    let value_logarithm = match value.natural_logarithm(evaluator) {
838        Some(logarithm) => logarithm,
839        None => return false,
840    };
841    let argument_logarithm = match argument.natural_logarithm(evaluator) {
842        Some(logarithm) => logarithm,
843        None => return false,
844    };
845    let offset = value
846        .add(&value_logarithm, evaluator)
847        .subtract(&argument_logarithm, evaluator);
848    let bits = evaluator.requested_precision_bits();
849    let pi_value = RealInexactNumber::pi(evaluator);
850    let two_pi = pi_value.add(&pi_value, evaluator);
851    let period = ComplexInexactNumber::new(RealInexactNumber::zero(bits), two_pi);
852    let recovered = match offset.divide(&period, evaluator) {
853        Some(recovered) => recovered,
854        None => return false,
855    };
856    let expected = RealInexactNumber::from_f64(branch as f64, bits);
857    let real_deviation = recovered
858        .real_part()
859        .subtract(&expected, evaluator)
860        .absolute_value();
861    let imaginary_deviation = recovered.imaginary_part().absolute_value();
862    let tolerance = RealInexactNumber::from_f64(0.25, bits);
863    let within = |deviation: &RealInexactNumber| {
864        matches!(
865            deviation.partial_compare(&tolerance),
866            Some(std::cmp::Ordering::Less) | Some(std::cmp::Ordering::Equal)
867        )
868    };
869    within(&real_deviation) && within(&imaginary_deviation)
870}

872fn negative_reciprocal_of_e(evaluator: &mut NumericEvaluator) -> RealInexactNumber {
873    // The branch point -1/e at the evaluator's working precision.
874    RealInexactNumber::e(evaluator)
875        .reciprocal(evaluator)
876        .negate()
