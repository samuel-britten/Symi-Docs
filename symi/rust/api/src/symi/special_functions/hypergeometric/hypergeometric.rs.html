31const AUTOMATIC_EXPANSION_DEGREE_BUDGET: u64 = 8;

33/// The largest number of expression nodes an automatic expansion may produce.
34const AUTOMATIC_EXPANSION_NODE_BUDGET: usize = 4096;

36/// Build the generalized hypergeometric function `pFq(upper; lower; argument)`.
37pub fn hypergeometric_pfq(
38    context: &Context,
39    upper_parameters: Vec<ExpressionReference>,
40    lower_parameters: Vec<ExpressionReference>,
41    argument: ExpressionReference,
42) -> ExpressionReference {
43    // The canonical constructor. Both parameter lists become ordered group
44    // nodes that preserve order and multiplicity exactly, and the resulting
45    // call routes through the ordinary function-call simplifier so the family
46    // reductions in `simplify_call` are the only place a pFq ever collapses.
47    let upper_group = parameter_group(context, upper_parameters);
48    let lower_group = parameter_group(context, lower_parameters);
49    simplify_function_call(
50        context,
51        FunctionIdentifier::Builtin(BuiltinFunction::HypergeometricPfq),
52        vec![upper_group, lower_group, argument],
53    )
54}

102/// Automatic simplification for the generalized hypergeometric family.
103pub fn simplify_call(
104    context: &Context,
105    builtin: BuiltinFunction,
106    arguments: &[ExpressionReference],
107) -> Option<ExpressionReference> {
108    // Only the globally safe reductions of plan section 5.4 run here: an
109    // uncancellable call must survive as a first-class head, because automatic
110    // simplification is not allowed to choose a branch or to assume a
111    // parameter condition it has not proved. A None return leaves the call
112    // unevaluated.
113    if builtin != BuiltinFunction::HypergeometricPfq {
114        return None;
115    }
116    let (upper_parameters, lower_parameters, argument) = split_call_arguments(arguments)?;
117    let analysis = analyse(upper_parameters, lower_parameters);
118    if matches!(
119        analysis.admissibility,
120        AdmissibilityClassification::UndefinedParameterConfiguration
121    ) {
122        return Some(undefined_expression(context));
123    }
124    if let Some(value) = exact_value_at_zero_argument(context, &analysis, argument) {
125        return Some(value);
126    }
127    if let Some(value) = exponential_reduction(context, &analysis, argument) {
128        return Some(value);
129    }
130    if let Some(value) = reductions::gauss_unit_argument_summation(
131        context,
132        &analysis.upper_parameters,
133        &analysis.lower_parameters,
134        argument,
135    ) {
136        return Some(value);
137    }
138    if let Some(degree) = analysis.terminating_degree() {
139        if degree <= AUTOMATIC_EXPANSION_DEGREE_BUDGET {
140            if let Some(polynomial) = terminating_polynomial(
141                context,
142                &analysis.upper_parameters,
143                &analysis.lower_parameters,
144                argument,
145                degree,
146                AUTOMATIC_EXPANSION_NODE_BUDGET,
147            ) {
148                return Some(polynomial);
149            }
150        }
151    }
152    if analysis.cancelled_any_pair {
153        return Some(rebuild_call(
154            context,
155            analysis.upper_parameters,
156            analysis.lower_parameters,
157            argument.clone(),
158        ));
159    }
160    None
161}

163fn rebuild_call(
164    context: &Context,
165    upper_parameters: Vec<ExpressionReference>,
166    lower_parameters: Vec<ExpressionReference>,
167    argument: ExpressionReference,
168) -> ExpressionReference {
169    // Reconstruct a pFq call from already-reduced parameter lists. Routing back
170    // through the public constructor re-runs the family classification, which
171    // is required after cancellation because p, q, the convergence class, and
172    // the set of available reductions can all have changed.
173    hypergeometric_pfq(context, upper_parameters, lower_parameters, argument)
174}

176fn exact_value_at_zero_argument(
177    context: &Context,
178    analysis: &parameter_analysis::ParameterAnalysis,
179    argument: &ExpressionReference,
180) -> Option<ExpressionReference> {
181    // Every admissible pFq has the value one at a zero argument, because only
182    // the k = 0 term of the defining series survives and that term is one for
183    // any parameters. The classification gate matters: at an undefined or
184    // undecided parameter configuration the series has no value to take a
185    // limit of, and those cases are handled before this point or left alone.
186    if !is_exact_integer_zero(argument) {
187        return None;
188    }
189    match analysis.admissibility {
190        AdmissibilityClassification::OrdinaryAdmissible
191        | AdmissibilityClassification::Terminating { .. }
192        | AdmissibilityClassification::TerminatingBeforeDenominatorZero { .. } => {
193            Some(builders::integer(context, exact_integer::from(1)))
194        }
195        _ => None,
196    }
197}

199fn exponential_reduction(
200    context: &Context,
201    analysis: &parameter_analysis::ParameterAnalysis,
202    argument: &ExpressionReference,
203) -> Option<ExpressionReference> {
204    // 0F0(; ; z) = exp(z), DLMF 16.2.1 with both parameter lists empty: the
205    // defining series is exactly the exponential series. This also delivers
206    // 1F1(a; a; z) = exp(z), because the shared cancellation step has already
207    // removed the matching pair by the time the analysis reaches here.
208    if !analysis.upper_parameters.is_empty() || !analysis.lower_parameters.is_empty() {
209        return None;
210    }
211    Some(simplify_function_call(
212        context,
213        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
214        vec![argument.clone()],
215    ))
216}

218/// Expand a terminating generalized hypergeometric series into its polynomial.
219pub fn terminating_polynomial(
220    context: &Context,
221    upper_parameters: &[ExpressionReference],
222    lower_parameters: &[ExpressionReference],
223    argument: &ExpressionReference,
224    degree: u64,
225    node_budget: usize,
226) -> Option<ExpressionReference> {
227    // Evaluate the finite sum of DLMF 16.2.1 exactly, term by term, using the
228    // ratio between consecutive terms rather than re-expanding each Pochhammer
229    // symbol from scratch. The coefficient recurrence is
230    //
231    //     t_{k+1} / t_k = prod_i (a_i + k) / (prod_j (b_j + k) * (k + 1)),
232    //
233    // so each step costs p + q + 1 builder calls regardless of degree.
234    //
235    // Every denominator factor is proved non-zero before it is used: the loop
236    // declines outright rather than dividing by something merely not known to
237    // vanish. The node budget bounds the constructed result, and exceeding it
238    // returns None so the caller retains the compact symbolic call instead of
239    // emitting a partially expanded one.
240    //
241    // A terminating series and a truncated Maclaurin series are the same finite
242    // sum, so both go through one implementation: at the termination degree the
243    // partial sum is not an approximation but the exact value.
244    series::partial_sum(
245        context,
246        upper_parameters,
247        lower_parameters,
248        argument,
249        degree,
250        node_budget,
251    )
252}

254fn is_provably_nonzero(reference: &ExpressionReference) -> bool {
255    // Whether a Pochhammer denominator factor is exactly known not to vanish.
256    // Only exact numeric literals qualify. A symbolic factor is never assumed
257    // non-zero here, which is what stops the terminating expander from
258    // producing a polynomial for a call whose denominators may be singular.
259    match &reference.kind {
260        ExpressionKind::Integer(value) => !crate::exact_arithmetic::integer_is_zero(value),
261        ExpressionKind::Rational(value) => !crate::exact_arithmetic::rational_is_zero(value),
262        _ => false,
263    }
264}
265
266fn is_exact_integer_zero(reference: &ExpressionReference) -> bool {
267    // Whether the argument is exactly the integer zero.
268    matches!(&reference.kind, ExpressionKind::Integer(value)
269        if crate::exact_arithmetic::integer_is_zero(value))
270}

272fn undefined_expression(context: &Context) -> ExpressionReference {
273    // The project's typed undefined sentinel, used when the parameters are
274    // exactly proved to place an uncancelled pole in the defining series.
275    context.interner.intern(ExpressionKind::Constant(

280/// Whether an expression is a generalized hypergeometric call.
281pub fn is_hypergeometric_call(reference: &ExpressionReference) -> bool {
282    // A shared shape test so consumers do not each re-derive the head match.
283    matches!(
284        &reference.kind,
285        ExpressionKind::FunctionCall(
286            FunctionIdentifier::Builtin(BuiltinFunction::HypergeometricPfq),
287            _
288        )
289    )

292/// The first derivative of a generalized hypergeometric call in its argument.
293pub fn argument_derivative(
294    context: &Context,
295    arguments: &[ExpressionReference],
296) -> Option<ExpressionReference> {
297    // DLMF 16.3.1 with n = 1:
298    //
299    //     d/dz pFq(a; b; z) = (prod_i a_i / prod_j b_j) pFq(a + 1; b + 1; z).
300    //
301    // The rule is only valid when the shifted call carries the same
302    // admissibility as the original, so both are classified and a call whose
303    // denominators are not proved non-zero declines rather than emitting a
304    // formula divided by a possible zero. A parameter that itself depends on
305    // the differentiation variable is rejected by the caller, which owns the
306    // free-variable test; this routine differentiates in the argument only.
307    let (upper_parameters, lower_parameters, argument) = split_call_arguments(arguments)?;
308    let analysis = analyse(upper_parameters, lower_parameters);
309    if !matches!(
310        analysis.admissibility,
311        AdmissibilityClassification::OrdinaryAdmissible
312            | AdmissibilityClassification::Terminating { .. }
313    ) {
314        return None;
315    }
316    let mut numerator_factors: Vec<ExpressionReference> = Vec::new();
317    for parameter in upper_parameters {
318        numerator_factors.push(parameter.clone());
319    }
320    let mut denominator_factors: Vec<ExpressionReference> = Vec::new();
321    for parameter in lower_parameters {
322        if !is_provably_nonzero(parameter) {
323            return None;
324        }
325        denominator_factors.push(parameter.clone());
326    }
327    let one = builders::integer(context, exact_integer::from(1));
328    let shifted_upper: Vec<ExpressionReference> = upper_parameters
329        .iter()
330        .map(|parameter| simplify_sum(context, vec![parameter.clone(), one.clone()]))
331        .collect();
332    let shifted_lower: Vec<ExpressionReference> = lower_parameters
333        .iter()
334        .map(|parameter| simplify_sum(context, vec![parameter.clone(), one.clone()]))
335        .collect();
336    let shifted_call = hypergeometric_pfq(context, shifted_upper, shifted_lower, argument.clone());
337    let numerator = simplify_product(context, numerator_factors);
338    let denominator = simplify_product(context, denominator_factors);
339    let minus_one = builders::integer(context, exact_integer::from(-1));
340    let inverse_denominator = simplify_power(context, denominator, minus_one);
341    Some(simplify_product(
342        context,
343        vec![numerator, inverse_denominator, shifted_call],
344    ))
345}

347/// Whether a call is admissible and holomorphic on the whole finite plane.
348pub fn is_admissible_and_entire(
349    upper_parameters: &[ExpressionReference],
350    lower_parameters: &[ExpressionReference],
351) -> bool {
352    // The single fact the analytic registry consumes to classify a `pFq` call.
353    // Both halves are needed: a call whose lower parameters are not proved to
354    // keep every Pochhammer denominator away from zero has no value to be
355    // holomorphic, and a convergent series over an inadmissible parameter set is
356    // not a function. A terminating call passes on both counts, because it is a
357    // polynomial.
358    let analysis = analyse(upper_parameters, lower_parameters);
359    let admissible = matches!(
360        analysis.admissibility,
361        AdmissibilityClassification::OrdinaryAdmissible
362            | AdmissibilityClassification::Terminating { .. }
363            | AdmissibilityClassification::TerminatingBeforeDenominatorZero { .. }
364    );
365    admissible && is_entire_in_argument(upper_parameters, lower_parameters)
366}

368/// Whether the reduced call converges for every finite argument.
369pub fn is_entire_in_argument(
370    upper_parameters: &[ExpressionReference],
371    lower_parameters: &[ExpressionReference],
372) -> bool {
373    // Report the DLMF 16.2 convergence class in the form the analytic registry
374    // consumes. A terminating call is a polynomial and is entire in every
375    // class, which is why termination is checked before the count comparison.
376    let analysis = analyse(upper_parameters, lower_parameters);
377    analysis.terminating_degree().is_some()
378        || matches!(analysis.convergence, ConvergenceClass::EntireInArgument)
379}
