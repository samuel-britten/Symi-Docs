8/// The admissibility classification of one generalized hypergeometric call.
9#[derive(Debug, Clone, PartialEq, Eq)]
10pub enum AdmissibilityClassification {
11    /// No lower Pochhammer denominator vanishes anywhere in the defining series.
12    OrdinaryAdmissible,
13    /// An upper parameter is a proved non-positive integer, so the series is a
14    /// polynomial of the recorded degree.
15    Terminating {
16        /// The exact degree at which the defining series terminates.
17        degree: u64,
18    },
19    /// A lower parameter is a non-positive integer, but the series terminates
20    /// strictly before the first vanishing denominator factor is reached.
21    TerminatingBeforeDenominatorZero {
22        /// The exact degree at which the defining series terminates.
23        degree: u64,
24    },
25    /// The ordinary function has an uncancelled denominator singularity.
26    UndefinedParameterConfiguration,
27    /// Admissibility cannot be decided from the parameters as given.
28    Unknown,
29}

31/// The convergence class of a generalized hypergeometric call, by reduced sizes.
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum ConvergenceClass {
34    /// Upper count does not exceed lower count: entire in the argument.
35    EntireInArgument,
36    /// Upper count is one more than lower count: unit radius of convergence.
37    UnitDisk,
38    /// Upper count exceeds lower count by more than one: divergent unless the
39    /// series terminates.
40    DivergentUnlessTerminating,
41}

43/// The deterministic result of analysing one generalized hypergeometric call.
44#[derive(Debug, Clone)]
45pub struct ParameterAnalysis {
46    /// The upper parameters remaining after safe cancellation, in order.
47    pub upper_parameters: Vec<ExpressionReference>,
48    /// The lower parameters remaining after safe cancellation, in order.
49    pub lower_parameters: Vec<ExpressionReference>,
50    /// Whether safe cancellation removed at least one upper/lower pair.
51    pub cancelled_any_pair: bool,
52    /// The admissibility classification of the reduced call.
53    pub admissibility: AdmissibilityClassification,
54    /// The convergence class implied by the reduced parameter counts.
55    pub convergence: ConvergenceClass,
56}

58impl ParameterAnalysis {
59    /// Whether the defining series is a polynomial of a proved finite degree.
60    pub fn terminating_degree(&self) -> Option<u64> {
61        match self.admissibility {
62            AdmissibilityClassification::Terminating { degree }
63            | AdmissibilityClassification::TerminatingBeforeDenominatorZero { degree } => {
64                Some(degree)
65            }
66            _ => None,
67        }
68    }
69}

71/// Analyse one generalized hypergeometric parameter pair.
72pub fn analyse(
73    upper_parameters: &[ExpressionReference],
74    lower_parameters: &[ExpressionReference],
75) -> ParameterAnalysis {
76    // The single classifier that every other part of the family consults, so
77    // that termination, denominator singularities, and safe cancellation are
78    // decided in exactly one place. Cancellation runs first because it changes
79    // p and q, and therefore changes both the convergence class and which
80    // denominator factors still exist to vanish. Every judgement here is exact:
81    // a parameter that is not a literal integer contributes `unknown` rather
82    // than a guess, and `unknown` never collapses to an admissible answer.
83    let (upper_parameters, lower_parameters, cancelled_any_pair) =
84        cancel_common_parameters(upper_parameters, lower_parameters);
85    let convergence = classify_convergence(upper_parameters.len(), lower_parameters.len());
86    let admissibility = classify_admissibility(&upper_parameters, &lower_parameters);
87    ParameterAnalysis {
88        upper_parameters,
89        lower_parameters,
90        cancelled_any_pair,
91        admissibility,
92        convergence,
93    }
94}

96fn classify_convergence(upper_count: usize, lower_count: usize) -> ConvergenceClass {
97    // The DLMF 16.2 split by parameter counts. This describes the defining
98    // series only; a terminating call is a polynomial and converges everywhere
99    // regardless of which class it falls in.
100    match upper_count.cmp(&(lower_count + 1)) {
101        Ordering::Less => ConvergenceClass::EntireInArgument,
102        Ordering::Equal => ConvergenceClass::UnitDisk,
103        Ordering::Greater => ConvergenceClass::DivergentUnlessTerminating,
104    }
105}

107/// Return the non-negative integer `n` when the expression is exactly `-n`.
108pub fn nonpositive_integer_magnitude(reference: &ExpressionReference) -> Option<u64> {
109    // Recognise an exact non-positive integer parameter and report how far the
110    // Pochhammer symbol runs before its factor at that parameter vanishes. Only
111    // an integer literal qualifies: a symbol that merely might be a
112    // non-positive integer must never be treated as one, because that would
113    // turn an unevaluated call into a wrong polynomial.
114    let ExpressionKind::Integer(value) = &reference.kind else {
115        return None;
116    };
117    match integer_sign(value) {
118        Ordering::Greater => None,
119        _ => {
120            let magnitude = -value.clone();
121            integer_to_small_unsigned(&magnitude)
122        }
123    }
124}

126fn integer_to_small_unsigned(value: &exact_integer) -> Option<u64> {
127    // Convert an exact non-negative integer to a machine-sized count, declining
128    // when it does not fit. A termination degree beyond u64 is not a degree any
129    // caller could expand, so refusing it here keeps every downstream budget
130    // check working on a value it can actually reason about.
131    let text = crate::exact_arithmetic::integer_to_decimal(value);
132    text.parse::<u64>().ok()
133}

135fn classify_admissibility(
136    upper_parameters: &[ExpressionReference],
137    lower_parameters: &[ExpressionReference],
138) -> AdmissibilityClassification {
139    // Decide the DLMF 16.2 admissibility of the reduced parameter pair. The
140    // order of the tests is the mathematics: the series stops at the smallest
141    // non-positive integer upper parameter, so a lower-parameter singularity
142    // matters only when its own vanishing index is reached at or before that
143    // stopping point. A lower parameter whose integrality cannot be decided
144    // leaves the whole classification unknown, because admitting the call would
145    // mean dividing by something not proved non-zero.
146    let termination_degree = upper_parameters
147        .iter()
148        .filter_map(nonpositive_integer_magnitude)
149        .min();
150    let mut smallest_lower_singularity: Option<u64> = None;
151    let mut lower_integrality_undecided = false;
152    for parameter in lower_parameters {
153        match nonpositive_integer_magnitude(parameter) {
154            Some(magnitude) => {
155                smallest_lower_singularity = Some(
156                    smallest_lower_singularity
157                        .map_or(magnitude, |current: u64| current.min(magnitude)),
158                );
159            }
160            None => {
161                if !is_provably_not_a_nonpositive_integer(parameter) {
162                    lower_integrality_undecided = true;
163                }
164            }
165        }
166    }
167    match (termination_degree, smallest_lower_singularity) {
168        (Some(degree), None) => {
169            if lower_integrality_undecided {
170                AdmissibilityClassification::Unknown
171            } else {
172                AdmissibilityClassification::Terminating { degree }
173            }
174        }
175        (Some(degree), Some(singularity)) => {
176            if degree < singularity {
177                AdmissibilityClassification::TerminatingBeforeDenominatorZero { degree }
178            } else {
179                AdmissibilityClassification::UndefinedParameterConfiguration
180            }
181        }
182        (None, Some(_)) => AdmissibilityClassification::UndefinedParameterConfiguration,
183        (None, None) => {
184            if lower_integrality_undecided {
185                AdmissibilityClassification::Unknown
186            } else {
187                AdmissibilityClassification::OrdinaryAdmissible
188            }
189        }
190    }
191}

193fn is_provably_not_a_nonpositive_integer(reference: &ExpressionReference) -> bool {
194    // Whether a lower parameter is exactly proved to avoid the non-positive
195    // integers, which is what makes its Pochhammer denominator provably
196    // non-zero at every index. Only exact literals qualify. A symbol answers
197    // false here even under a positivity assumption, because this classifier
198    // deliberately takes no assumption session: the caller that owns one
199    // supplies its evidence through the assumptions layer instead.
200    match &reference.kind {
201        ExpressionKind::Integer(value) => matches!(integer_sign(value), Ordering::Greater),
202        ExpressionKind::Rational(_) => true,
203        _ => false,
204    }
205}

207fn cancel_common_parameters(
208    upper_parameters: &[ExpressionReference],
209    lower_parameters: &[ExpressionReference],
210) -> (Vec<ExpressionReference>, Vec<ExpressionReference>, bool) {
211    // Remove upper/lower parameter pairs that are structurally identical, one
212    // occurrence at a time. Multiplicity is the point: with upper [a, a] and
213    // lower [a] exactly one `a` leaves each side, so one upper `a` survives.
214    // Relative order among the survivors is preserved on both sides.
215    //
216    // A pair is cancelled only when doing so cannot change the ordinary
217    // function. Cancelling an `a` that is a non-positive integer would erase
218    // either a termination point or a denominator singularity — the two facts
219    // that decide whether the call has a value at all — so such a pair is left
220    // in place for the admissibility classifier to rule on.
221    let mut retained_lower: Vec<Option<ExpressionReference>> = lower_parameters
222        .iter()
223        .map(|item| Some(item.clone()))
224        .collect();
225    let mut retained_upper: Vec<ExpressionReference> = Vec::new();
226    let mut cancelled_any_pair = false;
227    for upper in upper_parameters {
228        if nonpositive_integer_magnitude(upper).is_some() {
229            retained_upper.push(upper.clone());
230            continue;
231        }
232        let matching_index = retained_lower.iter().position(|candidate| {
233            candidate
234                .as_ref()
235                .is_some_and(|lower| references_are_identical(lower, upper))
236        });
237        match matching_index {
238            Some(index) => {
239                retained_lower[index] = None;
240                cancelled_any_pair = true;
241            }
242            None => retained_upper.push(upper.clone()),
243        }
244    }
245    let retained_lower = retained_lower.into_iter().flatten().collect();
246    (retained_upper, retained_lower, cancelled_any_pair)
247}

249fn references_are_identical(left: &ExpressionReference, right: &ExpressionReference) -> bool {
250    // Structural identity of two parameters. Interning makes the pointer test
251    // the common case; the structural fallback keeps the comparison correct for
252    // references built in the same context before a shared intern.
253    Arc::ptr_eq(left, right) || left.kind == right.kind
254}

256/// Read the ordered parameters out of a parameter-group argument.
257pub fn parameter_group_of(reference: &ExpressionReference) -> Option<&[ExpressionReference]> {
258    // Recover one ordered group from a parameter slot. Any other shape means
259    // the call was not built through the family builders and carries no
260    // recoverable grouping, so the caller must leave it alone rather than guess
261    // where one group ends and the next begins.
262    match &reference.kind {
263        ExpressionKind::SpecialFunctionParameterVector(parameters) => Some(parameters),
264        _ => None,
265    }
266}

268/// Read both parameter groups and the argument out of a `hypergeometric_pfq` call.
269pub fn split_call_arguments(
270    arguments: &[ExpressionReference],
271) -> Option<(
272    &[ExpressionReference],
273    &[ExpressionReference],
274    &ExpressionReference,
275)> {
276    // Destructure the canonical three-slot argument list of the generalized
277    // hypergeometric head. A call that does not match this shape is malformed
278    // and every consumer declines rather than reinterpreting the slots.
279    if arguments.len() != 3 {
280        return None;
281    }
282    let upper = parameter_group_of(&arguments[0])?;
283    let lower = parameter_group_of(&arguments[1])?;
284    Some((upper, lower, &arguments[2]))
285}
