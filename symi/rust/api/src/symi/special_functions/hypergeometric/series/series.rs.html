12/// The exact Maclaurin coefficients of one generalized hypergeometric call.
13pub fn maclaurin_coefficients(
14    context: &Context,
15    upper_parameters: &[ExpressionReference],
16    lower_parameters: &[ExpressionReference],
17    order: u64,
18) -> Option<Vec<ExpressionReference>> {
19    // Generate the coefficients of DLMF 16.2.1,
20    //
21    //     c_k = (a)_k / ((b)_k k!),
22    //
23    // for k = 0 .. order, through the ratio between consecutive coefficients
24    //
25    //     c_{k+1} / c_k = prod_i (a_i + k) / (prod_j (b_j + k) (k + 1)).
26    //
27    // Using the ratio rather than rebuilding each Pochhammer symbol keeps the
28    // cost linear in the order and keeps the emitted coefficients in the reduced
29    // form the arithmetic builders produce. Every denominator factor is proved
30    // non-zero before the division: an unproved factor stops the generation
31    // outright, because a coefficient divided by something that might vanish is
32    // not a coefficient.
33    let order_index = usize::try_from(order).ok()?;
34    let mut coefficient = builders::integer(context, exact_integer::from(1));
35    let mut coefficients = Vec::with_capacity(order_index + 1);
36    coefficients.push(coefficient.clone());
37    for step in 0..order_index {
38        let step_value = builders::integer(context, exact_integer::from(step as i64));
39        let mut numerator_factors: Vec<ExpressionReference> = Vec::new();
40        for parameter in upper_parameters {
41            numerator_factors.push(simplify_sum(
42                context,
43                vec![parameter.clone(), step_value.clone()],
44            ));
45        }
46        let mut denominator_factors: Vec<ExpressionReference> = Vec::new();
47        for parameter in lower_parameters {
48            let shifted = simplify_sum(context, vec![parameter.clone(), step_value.clone()]);
49            if !is_provably_nonzero(&shifted) {
50                return None;
51            }
52            denominator_factors.push(shifted);
53        }
54        denominator_factors.push(builders::integer(
55            context,
56            exact_integer::from(step as i64 + 1),
57        ));
58        let numerator = simplify_product(context, numerator_factors);
59        let denominator = simplify_product(context, denominator_factors);
60        let minus_one = builders::integer(context, exact_integer::from(-1));
61        let inverse_denominator = simplify_power(context, denominator, minus_one);
62        coefficient = simplify_product(context, vec![coefficient, numerator, inverse_denominator]);
63        coefficients.push(coefficient.clone());
64    }
65    Some(coefficients)
66}

68/// The exact partial sum of a generalized hypergeometric series.
69pub fn partial_sum(
70    context: &Context,
71    upper_parameters: &[ExpressionReference],
72    lower_parameters: &[ExpressionReference],
73    argument: &ExpressionReference,
74    order: u64,
75    node_budget: usize,
76) -> Option<ExpressionReference> {
77    // Assemble sum_{k=0}^{order} c_k z^k from the generated coefficients. The
78    // node budget is enforced on the finished expression rather than on the
79    // order alone, because a low order over long parameter lists still produces
80    // a large polynomial. Exceeding it returns `None` so the caller keeps the
81    // compact symbolic call instead of a half-expanded one.
82    let coefficients = maclaurin_coefficients(context, upper_parameters, lower_parameters, order)?;
83    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(coefficients.len());
84    for (index, coefficient) in coefficients.into_iter().enumerate() {
85        if index == 0 {
86            terms.push(coefficient);
87            continue;
88        }
89        let exponent = builders::integer(context, exact_integer::from(index as i64));
90        let power_of_argument = simplify_power(context, argument.clone(), exponent);
91        terms.push(simplify_product(
92            context,
93            vec![coefficient, power_of_argument],
94        ));
95    }
96    let total = simplify_sum(context, terms);
97    if crate::simplify::heuristic::node_count(&total) > node_budget {
98        return None;
99    }
100    Some(total)
101}

103/// The antiderivative of a generalized hypergeometric call in its argument.
104pub fn antiderivative_in_argument(
105    context: &Context,
106    upper_parameters: &[ExpressionReference],
107    lower_parameters: &[ExpressionReference],
108    argument: &ExpressionReference,
109) -> Option<ExpressionReference> {
110    // Invert DLMF 16.3.1 at n = 1. Writing A_i = a_i - 1 and B_j = b_j - 1,
111    //
112    //     d/dz pFq(A; B; z) = (prod_i A_i / prod_j B_j) pFq(a; b; z),
113    //
114    // so the antiderivative of pFq(a; b; z) is
115    //
116    //     (prod_j (b_j - 1) / prod_i (a_i - 1)) pFq(a - 1; b - 1; z).
117    //
118    // The identity needs every a_i - 1 non-zero, or the stated quotient does not
119    // exist, and it needs the down-shifted call to be admissible in its own
120    // right, because a shift onto a non-positive integer lower parameter would
121    // move the result onto a denominator pole. Both are checked rather than
122    // assumed, and either failure declines.
123    //
124    // A terminating call is deliberately admitted here as well: its
125    // antiderivative is the down-shifted polynomial, and the automatic expander
126    // will produce that polynomial when it is within budget.
127    let analysis = analyse(upper_parameters, lower_parameters);
128    if !matches!(
129        analysis.admissibility,
130        AdmissibilityClassification::OrdinaryAdmissible
131            | AdmissibilityClassification::Terminating { .. }
132    ) {
133        return None;
134    }
135    let one = builders::integer(context, exact_integer::from(1));
136    let mut lowered_upper: Vec<ExpressionReference> = Vec::with_capacity(upper_parameters.len());
137    let mut denominator_factors: Vec<ExpressionReference> = Vec::new();
138    for parameter in upper_parameters {
139        let lowered = builders::subtract(context, parameter.clone(), one.clone());
140        if !is_provably_nonzero(&lowered) {
141            return None;
142        }
143        denominator_factors.push(lowered.clone());
144        lowered_upper.push(lowered);
145    }
146    let mut lowered_lower: Vec<ExpressionReference> = Vec::with_capacity(lower_parameters.len());
147    let mut numerator_factors: Vec<ExpressionReference> = Vec::new();
148    for parameter in lower_parameters {
149        let lowered = builders::subtract(context, parameter.clone(), one.clone());
150        if nonpositive_integer_magnitude(&lowered).is_some() {
151            return None;
152        }
153        numerator_factors.push(lowered.clone());
154        lowered_lower.push(lowered);
155    }
156    let lowered_analysis = analyse(&lowered_upper, &lowered_lower);
157    if !matches!(
158        lowered_analysis.admissibility,
159        AdmissibilityClassification::OrdinaryAdmissible
160            | AdmissibilityClassification::Terminating { .. }
161    ) {
162        return None;
163    }
164    let lowered_call = hypergeometric_pfq(context, lowered_upper, lowered_lower, argument.clone());
165    let numerator = simplify_product(context, numerator_factors);
166    let denominator = simplify_product(context, denominator_factors);
167    let coefficient = builders::divide(context, numerator, denominator);
168    Some(simplify_product(context, vec![coefficient, lowered_call]))
169}

171fn is_provably_nonzero(reference: &ExpressionReference) -> bool {
172    // Whether a factor is exactly known not to vanish. Only exact numeric
173    // literals qualify: a symbolic factor is never assumed non-zero, which is
174    // what keeps a coefficient or a quotient from being emitted over a possible
175    // zero.
176    match &reference.kind {
177        ExpressionKind::Integer(value) => !crate::exact_arithmetic::integer_is_zero(value),
178        ExpressionKind::Rational(value) => !crate::exact_arithmetic::rational_is_zero(value),
179        _ => false,
180    }
181}
