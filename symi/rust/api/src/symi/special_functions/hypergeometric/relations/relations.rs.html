12/// One member of the implemented family of contiguous relations.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub enum ContiguousRelation {
15    /// The pure three-term relation between two upper parameters and their
16    /// single unit shifts, DLMF 15.5.11 read at the level of the defining
17    /// series so that it holds for every `pFq` with at least two upper
18    /// parameters, not only for the Gauss function.
19    UpperParameterPair {
20        /// The position of the first upper parameter in its ordered group.
21        first_index: usize,
22        /// The position of the second upper parameter in its ordered group.
23        second_index: usize,
24    },
25    /// The Euler-operator relation `(theta + a_i) F = a_i F[a_i + 1]`, the
26    /// per-parameter refinement of DLMF 16.3.1.
27    UpperParameterShift {
28        /// The position of the shifted upper parameter in its ordered group.
29        index: usize,
30    },
31    /// The Euler-operator relation `(theta + b_j - 1) F = (b_j - 1) F[b_j - 1]`,
32    /// the lower-parameter companion of the previous rule.
33    LowerParameterShift {
34        /// The position of the shifted lower parameter in its ordered group.
35        index: usize,
36    },
37    /// The pure three-term relation for the confluent limit `0F1`,
38    /// `(b - 1) F[b - 1] - (b - 1) F - (z / b) F[b + 1] = 0`, obtained by
39    /// eliminating the Euler operator between the two shift relations.
40    ConfluentLimitThreeTerm,
41}

43/// The residual of the generalized hypergeometric differential equation.
44pub fn differential_equation_residual(
45    context: &Context,
46    call: &ExpressionReference,
47    variable: SymbolIdentifier,
48) -> Option<ExpressionReference> {
49    // Read the parameter groups out of a `hypergeometric_pfq` call and build the
50    // DLMF 16.8.3 residual for the call itself. A call whose argument is not the
51    // bare differentiation variable declines: the operator is stated in the
52    // function's own argument and a composed argument would need the chain rule.
53    let ExpressionKind::FunctionCall(_, arguments) = &call.kind else {
54        return None;
55    };
56    let (upper_parameters, lower_parameters, argument) = split_call_arguments(arguments)?;
57    if !argument_is_the_variable(argument, variable) {
58        return None;
59    }
60    let upper_parameters = upper_parameters.to_vec();
61    let lower_parameters = lower_parameters.to_vec();
62    Some(differential_equation_residual_of(
63        context,
64        &upper_parameters,
65        &lower_parameters,
66        call,
67        variable,
68    ))
69}

71/// The generalized hypergeometric differential-equation residual of a candidate.
72pub fn differential_equation_residual_of(
73    context: &Context,
74    upper_parameters: &[ExpressionReference],
75    lower_parameters: &[ExpressionReference],
76    candidate: &ExpressionReference,
77    variable: SymbolIdentifier,
78) -> ExpressionReference {
79    // Build the DLMF 16.8.3 operator residual
80    //
81    //     [ theta prod_j (theta + b_j - 1) - z prod_i (theta + a_i) ] w,
82    //
83    // with theta = z d/dz, applied to an arbitrary candidate solution w. Both
84    // operator products are polynomials in theta and therefore commute, so the
85    // factors may be applied in any order; they are applied innermost-first.
86    //
87    // Taking the candidate separately from the parameters is what makes the
88    // residual usable as a check on a reduced form: when a terminating call has
89    // already collapsed to its polynomial, the polynomial is still the solution
90    // of the same equation, and the residual then cancels exactly rather than
91    // merely numerically.
92    //
93    // The residual is constructed through ordinary differentiation and the
94    // ordinary arithmetic builders rather than through a closed-form shortcut.
95    // That is what makes it an independent check: a wrong argument-derivative
96    // rule shows up as a non-zero residual instead of cancelling against itself.
97    let one = builders::integer(context, exact_integer::from(1));
98    let mut left = candidate.clone();
99    for parameter in lower_parameters {
100        let shifted = builders::subtract(context, parameter.clone(), one.clone());
101        left = apply_shifted_theta(context, &left, &shifted, variable);
102    }
103    left = apply_theta(context, &left, variable);
104    let mut right = candidate.clone();
105    for parameter in upper_parameters {
106        right = apply_shifted_theta(context, &right, parameter, variable);
107    }
108    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
109    let right = simplify_product(context, vec![variable_expression, right]);
110    builders::subtract(context, left, right)
111}

113/// The residual of one implemented contiguous relation.
114pub fn contiguous_relation_residual(
115    context: &Context,
116    call: &ExpressionReference,
117    relation: ContiguousRelation,
118    variable: SymbolIdentifier,
119) -> Option<ExpressionReference> {
120    // Return an expression that is identically zero when the named relation
121    // holds for the supplied call, or `None` when the relation does not apply to
122    // this call's shape or when one of its exact side conditions is not proved.
123    //
124    // Every relation is stated so that a shifted call is built through the
125    // ordinary family constructor. That matters: the constructor re-runs
126    // admissibility classification, so a shift that lands on an undefined
127    // parameter configuration turns into the typed undefined value rather than
128    // into a silently wrong residual.
129    let ExpressionKind::FunctionCall(_, arguments) = &call.kind else {
130        return None;
131    };
132    let (upper_parameters, lower_parameters, argument) = split_call_arguments(arguments)?;
133    let upper_parameters = upper_parameters.to_vec();
134    let lower_parameters = lower_parameters.to_vec();
135    let argument = argument.clone();
136    let analysis = analyse(&upper_parameters, &lower_parameters);
137    if !matches!(
138        analysis.admissibility,
139        AdmissibilityClassification::OrdinaryAdmissible
140            | AdmissibilityClassification::Terminating { .. }
141    ) {
142        return None;
143    }
144    match relation {
145        ContiguousRelation::UpperParameterPair {
146            first_index,
147            second_index,
148        } => upper_pair_residual(
149            context,
150            call,
151            &upper_parameters,
152            &lower_parameters,
153            &argument,
154            first_index,
155            second_index,
156        ),
157        ContiguousRelation::UpperParameterShift { index } => upper_shift_residual(
158            context,
159            call,
160            &upper_parameters,
161            &lower_parameters,
162            &argument,
163            index,
164            variable,
165        ),
166        ContiguousRelation::LowerParameterShift { index } => lower_shift_residual(
167            context,
168            call,
169            &upper_parameters,
170            &lower_parameters,
171            &argument,
172            index,
173            variable,
174        ),
175        ContiguousRelation::ConfluentLimitThreeTerm => confluent_three_term_residual(
176            context,
177            call,
178            &upper_parameters,
179            &lower_parameters,
180            &argument,
181        ),
182    }
183}

185/// Every contiguous relation that applies to the shape of one call.
186pub fn applicable_contiguous_relations(call: &ExpressionReference) -> Vec<ContiguousRelation> {
187    // Enumerate the relations whose parameter-count preconditions the call
188    // satisfies. The enumeration is bounded by the parameter counts and is
189    // deliberately not a search over an unbounded lattice of integer shifts:
190    // each entry names one fixed unit shift in one fixed direction, so a rewrite
191    // built on top of this list cannot cycle.
192    let ExpressionKind::FunctionCall(_, arguments) = &call.kind else {
193        return Vec::new();
194    };
195    let Some((upper_parameters, lower_parameters, _)) = split_call_arguments(arguments) else {
196        return Vec::new();
197    };
198    let mut relations = Vec::new();
199    for index in 0..upper_parameters.len() {
200        relations.push(ContiguousRelation::UpperParameterShift { index });
201    }
202    for index in 0..lower_parameters.len() {
203        relations.push(ContiguousRelation::LowerParameterShift { index });
204    }
205    for first_index in 0..upper_parameters.len() {
206        for second_index in (first_index + 1)..upper_parameters.len() {
207            relations.push(ContiguousRelation::UpperParameterPair {
208                first_index,
209                second_index,
210            });
211        }
212    }
213    if upper_parameters.is_empty() && lower_parameters.len() == 1 {
214        relations.push(ContiguousRelation::ConfluentLimitThreeTerm);
215    }
216    relations
217}

219fn upper_pair_residual(
220    context: &Context,
221    call: &ExpressionReference,
222    upper_parameters: &[ExpressionReference],
223    lower_parameters: &[ExpressionReference],
224    argument: &ExpressionReference,
225    first_index: usize,
226    second_index: usize,
227) -> Option<ExpressionReference> {
228    // DLMF 15.5.11 read term by term. With coefficient (a)_k (b)_k / ((c)_k k!),
229    // the identity
230    //
231    //     (b - a) F + a F[a + 1] - b F[b + 1] = 0
232    //
233    // follows from a (a + 1)_k = (a)_k (a + k), because the bracket collapses to
234    // (b - a) + (a + k) - (b + k) = 0 for every k. Nothing in that cancellation
235    // touches the lower parameters, so the relation holds for every pFq with at
236    // least two upper parameters and not only for the Gauss function.
237    if first_index == second_index {
238        return None;
239    }
240    let first = upper_parameters.get(first_index)?.clone();
241    let second = upper_parameters.get(second_index)?.clone();
242    let first_shifted = shift_upper(
243        context,
244        upper_parameters,
245        lower_parameters,
246        argument,
247        first_index,
248        1,
249    );
250    let second_shifted = shift_upper(
251        context,
252        upper_parameters,
253        lower_parameters,
254        argument,
255        second_index,
256        1,
257    );
258    let difference = builders::subtract(context, second.clone(), first.clone());
259    let leading = simplify_product(context, vec![difference, call.clone()]);
260    let first_term = simplify_product(context, vec![first, first_shifted]);
261    let second_term = simplify_product(context, vec![second, second_shifted]);
262    let positive = simplify_sum(context, vec![leading, first_term]);
263    Some(builders::subtract(context, positive, second_term))
264}

266fn upper_shift_residual(
267    context: &Context,
268    call: &ExpressionReference,
269    upper_parameters: &[ExpressionReference],
270    lower_parameters: &[ExpressionReference],
271    argument: &ExpressionReference,
272    index: usize,
273    variable: SymbolIdentifier,
274) -> Option<ExpressionReference> {
275    // (theta + a_i) F(a; b; z) = a_i F(a_i + 1; b; z). Comparing coefficients of
276    // z^k, the left side contributes (k + a_i) (a)_k and the right side
277    // a_i (a)_k (a_i + k) / a_i, which are the same. The relation therefore
278    // needs no side condition beyond the shared admissibility gate.
279    if !argument_is_the_variable(argument, variable) {
280        return None;
281    }
282    let parameter = upper_parameters.get(index)?.clone();
283    let shifted_call = shift_upper(
284        context,
285        upper_parameters,
286        lower_parameters,
287        argument,
288        index,
289        1,
290    );
291    let left = apply_shifted_theta(context, call, &parameter, variable);
292    let right = simplify_product(context, vec![parameter, shifted_call]);
293    Some(builders::subtract(context, left, right))
294}

296fn lower_shift_residual(
297    context: &Context,
298    call: &ExpressionReference,
299    upper_parameters: &[ExpressionReference],
300    lower_parameters: &[ExpressionReference],
301    argument: &ExpressionReference,
302    index: usize,
303    variable: SymbolIdentifier,
304) -> Option<ExpressionReference> {
305    // (theta + b_j - 1) F(a; b; z) = (b_j - 1) F(a; b_j - 1; z), from
306    // (b_j - 1)_k = (b_j - 1) (b_j)_k / (b_j + k - 1). The rewritten Pochhammer
307    // symbol divides by b_j - 1, so the relation is stated only when that shift
308    // target is itself an admissible call; the down-shifted constructor decides
309    // that, and a shift onto a non-positive integer lower parameter is rejected
310    // here rather than being allowed to produce a division by zero.
311    if !argument_is_the_variable(argument, variable) {
312        return None;
313    }
314    let parameter = lower_parameters.get(index)?.clone();
315    let one = builders::integer(context, exact_integer::from(1));
316    let decremented = builders::subtract(context, parameter, one);
317    let shifted_call = shift_lower(
318        context,
319        upper_parameters,
320        lower_parameters,
321        argument,
322        index,
323        -1,
324    );
325    let shifted_upper = {
326        let ExpressionKind::FunctionCall(_, arguments) = &shifted_call.kind else {
327            return None;
328        };
329        let (upper, lower, _) = split_call_arguments(arguments)?;
330        analyse(upper, lower)
331    };
332    if !matches!(
333        shifted_upper.admissibility,
334        AdmissibilityClassification::OrdinaryAdmissible
335            | AdmissibilityClassification::Terminating { .. }
336    ) {
337        return None;
338    }
339    let left = apply_shifted_theta(context, call, &decremented, variable);
340    let right = simplify_product(context, vec![decremented, shifted_call]);
341    Some(builders::subtract(context, left, right))
342}

344fn confluent_three_term_residual(
345    context: &Context,
346    call: &ExpressionReference,
347    upper_parameters: &[ExpressionReference],
348    lower_parameters: &[ExpressionReference],
349    argument: &ExpressionReference,
350) -> Option<ExpressionReference> {
351    // For 0F1 the Euler operator can be eliminated between the two shift rules,
352    // because DLMF 16.3.1 gives theta F = (z / b) F[b + 1] outright when there
353    // are no upper parameters. Substituting that into
354    // (theta + b - 1) F = (b - 1) F[b - 1] leaves the pure recurrence
355    //
356    //     (b - 1) F[b - 1] - (b - 1) F - (z / b) F[b + 1] = 0,
357    //
358    // which involves no derivative at all. The down shift needs b - 1 to be an
359    // admissible lower parameter and the derivative rule needs b non-zero, and
360    // both are decided by the shifted constructors rather than assumed.
361    if !upper_parameters.is_empty() || lower_parameters.len() != 1 {
362        return None;
363    }
364    let parameter = lower_parameters[0].clone();
365    let one = builders::integer(context, exact_integer::from(1));
366    let decremented = builders::subtract(context, parameter.clone(), one);
367    let down_shifted = shift_lower(context, upper_parameters, lower_parameters, argument, 0, -1);
368    let up_shifted = shift_lower(context, upper_parameters, lower_parameters, argument, 0, 1);
369    let down_analysis = {
370        let ExpressionKind::FunctionCall(_, arguments) = &down_shifted.kind else {
371            return None;
372        };
373        let (upper, lower, _) = split_call_arguments(arguments)?;
374        analyse(upper, lower)
375    };
376    if !matches!(
377        down_analysis.admissibility,
378        AdmissibilityClassification::OrdinaryAdmissible
379            | AdmissibilityClassification::Terminating { .. }
380    ) {
381        return None;
382    }
383    let scaled_numerator = simplify_product(context, vec![argument.clone(), up_shifted]);
384    let scaled_up = builders::divide(context, scaled_numerator, parameter);
385    let first = simplify_product(context, vec![decremented.clone(), down_shifted]);
386    let second = simplify_product(context, vec![decremented, call.clone()]);
387    let partial = builders::subtract(context, first, second);
388    Some(builders::subtract(context, partial, scaled_up))
389}

391fn shift_upper(
392    context: &Context,
393    upper_parameters: &[ExpressionReference],
394    lower_parameters: &[ExpressionReference],
395    argument: &ExpressionReference,
396    index: usize,
397    offset: i64,
398) -> ExpressionReference {
399    // Rebuild the call with one upper parameter moved by an exact integer
400    // offset, leaving every other parameter and the group order untouched.
401    let shifted = shifted_parameters(context, upper_parameters, index, offset);
402    hypergeometric_pfq(
403        context,
404        shifted,
405        lower_parameters.to_vec(),
406        argument.clone(),
407    )
408}

429fn shifted_parameters(
430    context: &Context,
431    parameters: &[ExpressionReference],
432    index: usize,
433    offset: i64,
434) -> Vec<ExpressionReference> {
435    // Produce a copy of one ordered parameter group with a single entry shifted.
436    // Copying rather than sorting or reordering is what preserves the group
437    // convention the representation is built on.
438    let offset_literal = builders::integer(context, exact_integer::from(offset));
439    parameters
440        .iter()
441        .enumerate()
442        .map(|(position, parameter)| {
443            if position == index {
444                simplify_sum(context, vec![parameter.clone(), offset_literal.clone()])
445            } else {
446                parameter.clone()
447            }
448        })
449        .collect()
450}

452fn apply_theta(
453    context: &Context,
454    target: &ExpressionReference,
455    variable: SymbolIdentifier,
456) -> ExpressionReference {
457    // The Euler operator theta = z d/dz, built from ordinary differentiation so
458    // that the constructed residual exercises the differentiation engine instead
459    // of restating its answer.
460    let derivative = differentiate(context, target, variable);
461    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
462    simplify_product(context, vec![variable_expression, derivative])
463}

465fn apply_shifted_theta(
466    context: &Context,
467    target: &ExpressionReference,
468    shift: &ExpressionReference,
469    variable: SymbolIdentifier,
470) -> ExpressionReference {
471    // The operator factor (theta + shift), the single building block from which
472    // both the differential equation and the Euler-operator contiguous relations
473    // are assembled.
474    let theta_part = apply_theta(context, target, variable);
475    let shifted_part = simplify_product(context, vec![shift.clone(), target.clone()]);
476    simplify_sum(context, vec![theta_part, shifted_part])
477}

479fn argument_is_the_variable(argument: &ExpressionReference, variable: SymbolIdentifier) -> bool {
480    // Whether the call's argument is exactly the differentiation variable. The
481    // Euler-operator identities are stated in the function's own argument, so a
482    // composed argument would need the chain rule and is declined rather than
483    // being differentiated as if it were the bare variable.
484    matches!(&argument.kind, ExpressionKind::Symbol(identifier) if *identifier == variable)
485}
