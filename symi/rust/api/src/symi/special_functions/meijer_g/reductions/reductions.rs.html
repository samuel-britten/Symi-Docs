11/// Reduce a Meijer-G call to the finite sum of `pFq` terms of DLMF 16.17.2.
12pub fn reduce_to_hypergeometric_sum(
13    context: &Context,
14    parts: &MeijerGCallParts<'_>,
15) -> Option<ExpressionReference> {
16    // DLMF 16.17.2 evaluates the Mellin-Barnes integral by residues at the poles
17    // of the Gamma(b_j - s) factors with j <= m:
18    //
19    //     G^{m,n}_{p,q}(z)
20    //       = sum_{k=1..m}
21    //           prod_{j /= k, j<=m} Gamma(b_j - b_k) prod_{j<=n} Gamma(1 + b_k - a_j)
22    //           / ( prod_{j>m} Gamma(1 + b_k - b_j) prod_{j>n} Gamma(a_j - b_k) )
23    //         * z^{b_k} pF_{q-1}(1 + b_k - a; 1 + b_k - b*; (-1)^{p-m-n} z),
24    //
25    // where b* omits b_k. The hypotheses are p < q, or p = q with |z| < 1, and
26    // that no two of b_1 .. b_m differ by an integer. The last condition is what
27    // makes the poles simple: at an integer difference two pole families merge
28    // into double poles and the true expansion acquires logarithms, so the
29    // formula is not merely imprecise there but wrong.
30    //
31    // Every condition is decided exactly or the reduction declines. A symbolic
32    // lower numerator parameter leaves the pole-simplicity question open and is
33    // refused, because guessing it would be guessing which of two different
34    // expansions the function has.
35    let upper_order = parts.upper_order();
36    let lower_order = parts.lower_order();
37    let numerator_order = parts.lower_numerator_order();
38    if numerator_order == 0 || upper_order > lower_order {
39        return None;
40    }
41    if upper_order == lower_order && !argument_is_inside_the_unit_disk(parts.argument) {
42        return None;
43    }
44    let lower_numerator = parts.lower_numerator_parameters.to_vec();
45    if !pole_orders_are_provably_simple(context, &lower_numerator) {
46        return None;
47    }
48    let upper_parameters: Vec<ExpressionReference> = parts
49        .upper_numerator_parameters
50        .iter()
51        .chain(parts.upper_denominator_parameters.iter())
52        .cloned()
53        .collect();
54    let lower_parameters: Vec<ExpressionReference> = parts
55        .lower_numerator_parameters
56        .iter()
57        .chain(parts.lower_denominator_parameters.iter())
58        .cloned()
59        .collect();
60    let upper_numerator_order = parts.upper_numerator_order();
61    let sign = if (upper_order + numerator_order + upper_numerator_order) % 2 == 0 {
62        1
63    } else {
64        -1
65    };
66    let signed_argument = {
67        let literal = builders::integer(context, exact_integer::from(sign));
68        simplify_product(context, vec![literal, parts.argument.clone()])
69    };
70    let one = builders::integer(context, exact_integer::from(1));
71    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(numerator_order);
72    for index in 0..numerator_order {
73        let residue_parameter = lower_numerator[index].clone();
74        let mut numerator_factors: Vec<ExpressionReference> = Vec::new();
75        for (other, parameter) in lower_numerator.iter().enumerate() {
76            if other == index {
77                continue;
78            }
79            let difference =
80                builders::subtract(context, parameter.clone(), residue_parameter.clone());
81            numerator_factors.push(gamma_of(context, difference));
82        }
83        for parameter in parts.upper_numerator_parameters {
84            let shifted = shifted_complement(context, &one, &residue_parameter, parameter);
85            numerator_factors.push(gamma_of(context, shifted));
86        }
87        let mut denominator_factors: Vec<ExpressionReference> = Vec::new();
88        for parameter in parts.lower_denominator_parameters {
89            let shifted = shifted_complement(context, &one, &residue_parameter, parameter);
90            denominator_factors.push(gamma_of(context, shifted));
91        }
92        for parameter in parts.upper_denominator_parameters {
93            let difference =
94                builders::subtract(context, parameter.clone(), residue_parameter.clone());
95            denominator_factors.push(gamma_of(context, difference));
96        }
97        let series_upper: Vec<ExpressionReference> = upper_parameters
98            .iter()
99            .map(|parameter| shifted_complement(context, &one, &residue_parameter, parameter))
100            .collect();
101        let series_lower: Vec<ExpressionReference> = lower_parameters
102            .iter()
103            .enumerate()
104            .filter(|(position, _)| *position != index)
105            .map(|(_, parameter)| shifted_complement(context, &one, &residue_parameter, parameter))
106            .collect();
107        let series =
108            hypergeometric_pfq(context, series_upper, series_lower, signed_argument.clone());
109        let power_factor =
110            simplify_power(context, parts.argument.clone(), residue_parameter.clone());
111        let numerator = simplify_product(context, numerator_factors);
112        let denominator = simplify_product(context, denominator_factors);
113        let coefficient = builders::divide(context, numerator, denominator);
114        terms.push(simplify_product(
115            context,
116            vec![coefficient, power_factor, series],
117        ));
118    }
119    Some(simplify_sum(context, terms))
120}

122fn shifted_complement(
123    context: &Context,
124    one: &ExpressionReference,
125    residue_parameter: &ExpressionReference,
126    parameter: &ExpressionReference,
127) -> ExpressionReference {
128    // Build `1 + b_k - x`, the parameter shape every group of DLMF 16.17.2 uses.
129    let shifted = simplify_sum(context, vec![one.clone(), residue_parameter.clone()]);
130    builders::subtract(context, shifted, parameter.clone())
131}

133fn gamma_of(context: &Context, argument: ExpressionReference) -> ExpressionReference {
134    // Build one gamma factor through the ordinary call simplifier so an exact
135    // argument folds to its exact value instead of staying symbolic.
136    simplify_function_call(
137        context,
138        FunctionIdentifier::Builtin(BuiltinFunction::Gamma),
139        vec![argument],
140    )
141}

143fn pole_orders_are_provably_simple(
144    context: &Context,
145    lower_numerator: &[ExpressionReference],
146) -> bool {
147    // Whether the poles the residue expansion sums over are exactly known to be
148    // simple. They are simple when no two of b_1 .. b_m differ by an integer,
149    // and the test is exact: only literal parameters can settle it, so a single
150    // symbolic entry answers false and the reduction declines. Answering true on
151    // a guess would select the wrong expansion, not a less accurate one.
152    for parameter in lower_numerator {
153        if !matches!(
154            &parameter.kind,
155            ExpressionKind::Integer(_) | ExpressionKind::Rational(_)
156        ) {
157            return false;
158        }
159    }
160    for (index, parameter) in lower_numerator.iter().enumerate() {
161        for other in lower_numerator.iter().skip(index + 1) {
162            let difference = builders::subtract(context, other.clone(), parameter.clone());
163            if matches!(&difference.kind, ExpressionKind::Integer(_)) {
164                return false;
165            }
166        }
167    }
168    true
169}

171fn argument_is_inside_the_unit_disk(argument: &ExpressionReference) -> bool {
172    // Whether an exact rational argument is strictly inside the unit disk, the
173    // extra hypothesis DLMF 16.17.2 carries when p equals q. Only exact literals
174    // qualify; a symbolic argument leaves the question open and the reduction
175    // declines rather than assuming a region.
176    match &argument.kind {
177        ExpressionKind::Integer(value) => crate::exact_arithmetic::integer_is_zero(value),
178        ExpressionKind::Rational(value) => {
179            let magnitude = crate::exact_arithmetic::rational_absolute_value(value);
180            let difference = magnitude - crate::exact_arithmetic::rational_one();
181            crate::exact_arithmetic::rational_is_negative(&difference)
182        }
183        _ => false,
184    }
185}
