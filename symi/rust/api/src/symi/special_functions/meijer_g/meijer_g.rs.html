16/// The four ordered parameter groups and the argument of a Meijer-G call.
17pub struct MeijerGCallParts<'lifetime> {
18    /// The first `n` upper parameters.
19    pub upper_numerator_parameters: &'lifetime [ExpressionReference],
20    /// The remaining `p - n` upper parameters.
21    pub upper_denominator_parameters: &'lifetime [ExpressionReference],
22    /// The first `m` lower parameters.
23    pub lower_numerator_parameters: &'lifetime [ExpressionReference],
24    /// The remaining `q - m` lower parameters.
25    pub lower_denominator_parameters: &'lifetime [ExpressionReference],
26    /// The scalar argument.
27    pub argument: &'lifetime ExpressionReference,
28}
29
30impl MeijerGCallParts<'_> {
31    /// The DLMF order `m`, derived from the lower numerator group length.
32    pub fn lower_numerator_order(&self) -> usize {
33        self.lower_numerator_parameters.len()
34    }
35
36    /// The DLMF order `n`, derived from the upper numerator group length.
37    pub fn upper_numerator_order(&self) -> usize {
38        self.upper_numerator_parameters.len()
39    }
40
41    /// The DLMF order `p`, the total number of upper parameters.
42    pub fn upper_order(&self) -> usize {
43        self.upper_numerator_parameters.len() + self.upper_denominator_parameters.len()
44    }
45
46    /// The DLMF order `q`, the total number of lower parameters.
47    pub fn lower_order(&self) -> usize {
48        self.lower_numerator_parameters.len() + self.lower_denominator_parameters.len()
49    }
50}

52/// Build the Meijer-G function from its four ordered parameter groups.
53pub fn meijer_g(
54    context: &Context,
55    upper_numerator_parameters: Vec<ExpressionReference>,
56    upper_denominator_parameters: Vec<ExpressionReference>,
57    lower_numerator_parameters: Vec<ExpressionReference>,
58    lower_denominator_parameters: Vec<ExpressionReference>,
59    argument: ExpressionReference,
60) -> ExpressionReference {
61    // The canonical constructor, following the DLMF 16.17 partition directly.
62    // The orders m, n, p, and q are never supplied separately: they are read
63    // back from the group lengths, so no stored count can disagree with the
64    // parameters actually held. Construction chooses no contour and performs
65    // no reduction; a syntactically valid call stays symbolic.
66    let groups = vec![
67        parameter_group(context, upper_numerator_parameters),
68        parameter_group(context, upper_denominator_parameters),
69        parameter_group(context, lower_numerator_parameters),
70        parameter_group(context, lower_denominator_parameters),
71        argument,
72    ];
73    simplify_function_call(
74        context,
75        FunctionIdentifier::Builtin(BuiltinFunction::MeijerG),
76        groups,
77    )
78}

80/// Read the four ordered groups and the argument out of a Meijer-G call.
81pub fn split_call_arguments(arguments: &[ExpressionReference]) -> Option<MeijerGCallParts<'_>> {
82    // Destructure the canonical five-slot argument list. A call not matching
83    // this shape is malformed, and consumers decline rather than guessing where
84    // one parameter group ends and the next begins.
85    if arguments.len() != 5 {
86        return None;
87    }
88    Some(MeijerGCallParts {
89        upper_numerator_parameters: parameter_group_of(&arguments[0])?,
90        upper_denominator_parameters: parameter_group_of(&arguments[1])?,
91        lower_numerator_parameters: parameter_group_of(&arguments[2])?,
92        lower_denominator_parameters: parameter_group_of(&arguments[3])?,
93        argument: &arguments[4],
94    })
95}

97/// Whether the pole sets of the two numerator groups provably collide.
98pub fn numerator_poles_provably_collide(parts: &MeijerGCallParts<'_>) -> bool {
99    // The DLMF 16.17 separation restriction: the contour must separate the
100    // poles of Gamma(b_j - s) for j <= m from those of Gamma(1 - a_i + s) for
101    // i <= n. Those two pole families intersect exactly when some difference
102    // a_i - b_j is a positive integer, and no contour then exists. Only exact
103    // integer differences count as a collision; a symbolic difference leaves
104    // the question open and the call stays symbolic with an unknown judgment.
105    for upper in parts.upper_numerator_parameters {
106        for lower in parts.lower_numerator_parameters {
107            if difference_is_a_positive_integer(upper, lower) {
108                return true;
109            }
110        }
111    }
112    false
113}
114
115fn difference_is_a_positive_integer(
116    upper: &ExpressionReference,
117    lower: &ExpressionReference,
118) -> bool {
119    // Whether `upper - lower` is exactly a positive integer. The test works on
120    // integer literals only: deciding this for symbolic parameters is the job
121    // of the assumptions layer, and guessing it here would either invent a
122    // pole collision or hide a real one.
123    let (ExpressionKind::Integer(upper_value), ExpressionKind::Integer(lower_value)) =
124        (&upper.kind, &lower.kind)
125    else {
126        return false;
127    };
128    let difference = upper_value.clone() - lower_value.clone();
129    matches!(
130        crate::exact_arithmetic::integer_sign(&difference),
131        std::cmp::Ordering::Greater
132    )
133}

135/// Automatic simplification for the Meijer-G family.
136pub fn simplify_call(
137    context: &Context,
138    builtin: BuiltinFunction,
139    arguments: &[ExpressionReference],
140) -> Option<ExpressionReference> {
141    // Only exactly proved facts act here. A parameter configuration whose
142    // numerator pole sets provably collide has no admissible contour at all and
143    // is undefined; everything else stays a first-class symbolic head. No
144    // reduction to a sum of pFq terms runs automatically, because those
145    // expansions are large and their hypotheses belong to a directed request.
146    if builtin != BuiltinFunction::MeijerG {
147        return None;
148    }
149    let parts = split_call_arguments(arguments)?;
150    if numerator_poles_provably_collide(&parts) {
151        return Some(context.interner.intern(ExpressionKind::Constant(
152            crate::expression::BuiltinConstant::Undefined,
153        )));
154    }
155    None
156}

158/// Whether an expression is a Meijer-G call.
159pub fn is_meijer_g_call(reference: &ExpressionReference) -> bool {
160    // A shared shape test so consumers do not each re-derive the head match.
161    matches!(
162        &reference.kind,
163        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(BuiltinFunction::MeijerG), _)
164    )
165}

167/// Whether every parameter of a Meijer-G call is an exact literal.
168pub fn all_parameters_are_exact(parts: &MeijerGCallParts<'_>) -> bool {
169    // Report whether the reduction and validation rules may treat the whole
170    // parameter set as decidable. A single symbolic parameter is enough to make
171    // the pole-separation question undecidable, and callers use this to choose
172    // between an exact judgment and an honest unknown.
173    [
174        parts.upper_numerator_parameters,
175        parts.upper_denominator_parameters,
176        parts.lower_numerator_parameters,
177        parts.lower_denominator_parameters,
178    ]
179    .iter()
180    .all(|group| {
181        group.iter().all(|parameter| {
182            matches!(
183                &parameter.kind,
184                ExpressionKind::Integer(_) | ExpressionKind::Rational(_)
185            ) || nonpositive_integer_magnitude(parameter).is_some()
186        })
187    })
