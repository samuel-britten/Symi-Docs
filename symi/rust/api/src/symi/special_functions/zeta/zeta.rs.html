1// The Riemann zeta ζ(s) and Hurwitz zeta ζ(s, a), shared across the symbolic and
2// numeric seams exactly as the gamma family is. The numeric kernel is a single
3// Euler–Maclaurin evaluation of the Hurwitz zeta written once against the gamma
4// module's gamma_field trait (real and complex backends); the Riemann zeta is the
5// a = 1 specialisation, so one algorithm serves both over ℝ and ℂ. The symbolic
6// layer owns the classical special values — the even-integer π^{2n} closed forms,
7// ζ(0), the negative-integer functional values, the s = 1 pole, and the Hurwitz
8// reductions — all computed exactly from the shared Bernoulli table the gamma
9// kernels already ship.
10//
11// hurwitz_zeta(s, 1) folds to zeta(s) at construction so the two heads never both
12// appear in a normal form (Decision 1), mirroring polygamma(0, z) → digamma(z).

39fn field_general_power<Field: GammaField>(
40    base: &Field,
41    exponent: &Field,
42    evaluator: &mut NumericEvaluator,
43) -> Option<Field> {
44    // Principal power base^exponent = exp(exponent · log base). Declines when the
45    // base has no principal logarithm (a non-positive real base on the real
46    // backend), which is exactly where the Hurwitz direct terms would cross the cut.
47    let logarithm = base.logarithm(evaluator)?;
48    let product = exponent.multiply(&logarithm, evaluator);
49    Some(product.exponential(evaluator))
50}

52fn hurwitz_euler_maclaurin<Field: GammaField>(
53    s: &Field,
54    a: &Field,
55    bernoulli: &[Rational],
56    evaluator: &mut NumericEvaluator,
57) -> Option<(Field, u32)> {
58    // ζ(s, a) = Σ_{k≥0} (k+a)^{-s} by Euler–Maclaurin: the first N terms summed
59    // directly, then the analytic-continuation tail
60    //   (N+a)^{1-s}/(s-1) + ½(N+a)^{-s} + Σ_{k≥1} B_{2k}/(2k)! (s)_{2k-1} (N+a)^{-s-2k+1}
61    // truncated at its smallest term. N is chosen so Re(N+a) clears the Stirling
62    // shift target, exactly as the gamma kernels. Declines at the s = 1 pole and
63    // for Re(a) ≤ 0, where the direct terms would cross the principal-power cut.
64    //
65    // Reported alongside the value is how many of its leading bits cancelled away. For
66    // Re(s) < 0 the direct terms reach (N+a)^{|Re s|} while the value they add up to is
67    // very much smaller, so the leading bits of every one of them are lost; the caller
68    // decides from that count whether the working precision covered the loss.
69    let real_a = a.real_component();
70    if !is_strictly_positive(&real_a) {
71        return None;
72    }
73    let one = Field::from_i64(1, evaluator);
74    let s_minus_one = s.subtract(&one, evaluator);
75    if s_minus_one.magnitude_real(evaluator).is_zero() {
76        return None;
77    }
78    let minus_one = Field::from_i64(-1, evaluator);
79    let negated_s = s.multiply(&minus_one, evaluator);
80
81    let shift = recurrence_shift(&real_a, shift_target(evaluator));
82    let mut result = Field::from_i64(0, evaluator);
83    let mut largest_contribution = RealInexactNumber::zero(evaluator.requested_precision_bits());
84    for step in 0..shift {
85        let offset = Field::from_i64(step, evaluator);
86        let base = a.add(&offset, evaluator);
87        let term = field_general_power(&base, &negated_s, evaluator)?;
88        retain_larger_magnitude(&mut largest_contribution, &term, evaluator);
89        result = result.add(&term, evaluator);
90    }
91
92    let shift_field = Field::from_i64(shift, evaluator);
93    let large = a.add(&shift_field, evaluator);
94    let one_minus_s = one.subtract(s, evaluator);
95    let large_power_one_minus_s = field_general_power(&large, &one_minus_s, evaluator)?;
96    let integral = large_power_one_minus_s.divide(&s_minus_one, evaluator)?;
97    retain_larger_magnitude(&mut largest_contribution, &integral, evaluator);
98    result = result.add(&integral, evaluator);
99
100    let large_power_minus_s = field_general_power(&large, &negated_s, evaluator)?;
101    let half = Field::from_rational(
102        &Rational::from_integers(Integer::from(1), Integer::from(2)),
103        evaluator,
104    );
105    let half_term = half.multiply(&large_power_minus_s, evaluator);
106    retain_larger_magnitude(&mut largest_contribution, &half_term, evaluator);
107    result = result.add(&half_term, evaluator);
108
109    let large_inverse = one.divide(&large, evaluator)?;
110    let large_inverse_squared = large_inverse.multiply(&large_inverse, evaluator);
111    let mut large_inverse_power = large_inverse.clone();
112    let mut pochhammer = s.clone();
113    let mut double_factorial = Integer::from(2);
114    let mut previous_magnitude: Option<RealInexactNumber> = None;
115    for (offset, bernoulli_value) in bernoulli.iter().enumerate() {
116        let index = (offset + 1) as i64;
117        let rational_coefficient = bernoulli_value / Rational::from(double_factorial.clone());
118        let coefficient = Field::from_rational(&rational_coefficient, evaluator);
119        let term = coefficient
120            .multiply(&pochhammer, evaluator)
121            .multiply(&large_power_minus_s, evaluator)
122            .multiply(&large_inverse_power, evaluator);
123        let magnitude = term.magnitude_real(evaluator);
124        if let Some(previous) = &previous_magnitude {
125            if magnitude.partial_compare(previous) != Some(Ordering::Less) {
126                break;
127            }
128        }
129        retain_larger_magnitude(&mut largest_contribution, &term, evaluator);
130        result = result.add(&term, evaluator);
131        previous_magnitude = Some(magnitude);
132        let next_lower = Field::from_i64(2 * index - 1, evaluator);
133        let next_upper = Field::from_i64(2 * index, evaluator);
134        let factor_lower = s.add(&next_lower, evaluator);
135        let factor_upper = s.add(&next_upper, evaluator);
136        pochhammer = pochhammer
137            .multiply(&factor_lower, evaluator)
138            .multiply(&factor_upper, evaluator);
139        large_inverse_power = large_inverse_power.multiply(&large_inverse_squared, evaluator);
140        double_factorial *= Integer::from(2 * index + 1);
141        double_factorial *= Integer::from(2 * index + 2);
142    }
143    let result_magnitude = result.magnitude_real(evaluator);
144    let lost = cancelled_leading_bits(
145        &largest_contribution,
146        &result_magnitude,
147        evaluator.requested_precision_bits(),
148        evaluator,
149    );
150    Some((result, lost))
151}

153fn retain_larger_magnitude<Field: GammaField>(
154    largest: &mut RealInexactNumber,
155    candidate: &Field,
156    evaluator: &mut NumericEvaluator,
157) {
158    let magnitude = candidate.magnitude_real(evaluator);
159    if matches!(largest.partial_compare(&magnitude), Some(Ordering::Less)) {
160        *largest = magnitude;
161    }
162}

164fn cancelled_leading_bits(
165    largest_contribution: &RealInexactNumber,
166    result: &RealInexactNumber,
167    working_precision: u32,
168    evaluator: &mut NumericEvaluator,
169) -> u32 {
170    // How many leading bits the sum lost, as log2 of the ratio between the largest
171    // magnitude that entered it and the magnitude that came out. The logarithm is taken
172    // in the arbitrary-precision backend rather than in an f64 ratio, because the ratios
173    // that matter here run past the largest double: a deeply negative order makes the
174    // Euler-Maclaurin terms exceed their own sum by thousands of bits, and that count is
175    // exactly the number the caller needs. It decides working precision, never a value,
176    // and fails conservatively — a ratio that yields no finite count reports the whole
177    // working precision, which asks for the widest retry rather than the narrowest.
178    if result.is_zero() {
179        return working_precision;
180    }
181    let ratio = largest_contribution
182        .divide(result, evaluator)
183        .absolute_value();
184    let one = RealInexactNumber::one(working_precision);
185    if !matches!(ratio.partial_compare(&one), Some(Ordering::Greater)) {
186        return 0;
187    }
188    let two = RealInexactNumber::from_f64(2.0, working_precision);
189    let logarithm_of_two = two.natural_logarithm(evaluator);
190    let bits = ratio
191        .natural_logarithm(evaluator)
192        .divide(&logarithm_of_two, evaluator);
193    match bits.to_f64() {
194        Some(value) if value.is_finite() && value > 0.0 => {
195            let rounded = value.ceil();
196            if rounded >= f64::from(u32::MAX) {
197                working_precision
198            } else {
199                rounded as u32
200            }
201        }
202        Some(_) => 0,
203        None => working_precision,
204    }
205}

207/// Largest number of terms the reflected functional-equation sum may take.
208///
209/// A sum that has not closed within this many terms declines rather than returning its
210/// truncation, and the caller falls back to the Euler-Maclaurin route.
211const REFLECTION_TERM_MAXIMUM: i64 = 1024;
212
213/// The share of the term budget the route-choosing estimate is allowed to assume.
214///
215/// That estimate reads the decay of n^{Re(s)-1} alone, while the sum is answerable only to
216/// its own total, which the leading sines can drive well below the first term — sin(πs/2 +
217/// 2πnf) vanishes outright for every other n at a half-integer order and a quarter shift.
218/// Leaving three quarters of the budget in hand covers that without having to predict it.
219const REFLECTION_ESTIMATE_SHARE: i64 = 4;
220
221/// Largest shift the reflection will peel down into (0, 1].
222const REFLECTION_SHIFT_STEP_MAXIMUM: i64 = 1024;
223
224/// Bits the reflected sum's remaining tail must fall below, relative to the total it has
225/// reached, before the sum is taken to have closed, so what is dropped cannot reach the
226/// answer. The bound used is the next term's, and the tail behind it sums to a small
227/// multiple of that, which this margin absorbs.
228const REFLECTION_MARGIN_BITS: u32 = 8;
229
230/// Bits a zeta evaluation carries above the requested precision before it has measured how
231/// much its route actually lost.
232const ZETA_GUARD_BITS: u32 = 32;
233
234/// How many times a zeta evaluation may raise its working precision to cover the loss it
235/// measured before it declines.
236const ZETA_ESCALATION_BUDGET: usize = 4;
237
238/// Largest working precision a zeta evaluation will carry. A route that has not reached the
239/// requested precision by here has not shown what the value is, and declining is the only
240/// rigorous answer left.
241const ZETA_MAXIMUM_WORKING_PRECISION: u32 = 1 << 16;
242
243/// The granularity working precisions are rounded up to.
244///
245/// The asymptotic kernels are driven by a Bernoulli table whose size follows the working
246/// precision, and building one costs far more than the sum it feeds. A series that
247/// evaluates a zeta per term meets a different loss at every term, so precisions taken at
248/// the measured loss exactly would each want a table of their own; rounded up to a step,
249/// a whole run of them shares one.
250const ZETA_PRECISION_STEP: u32 = 128;
251
252fn stepped_working_precision(bits: u32) -> u32 {
253    bits.div_ceil(ZETA_PRECISION_STEP)
254        .saturating_mul(ZETA_PRECISION_STEP)
255}

257fn hurwitz_zeta_at_requested_precision<Field: GammaField>(
258    s: &Field,
259    a: &Field,
260    evaluator: &mut NumericEvaluator,
261) -> Option<Field> {
262    // The numeric zeta entry. Every attempt runs above the requested precision and reports
263    // how many of its leading bits it lost; when that loss fits inside the margin the
264    // attempt carried, the value is correct to the requested precision and is rounded back
265    // to it. Otherwise the measured loss becomes the next attempt's margin. Exhausting the
266    // budget declines rather than returning the last attempt, because a value whose leading
267    // bits all went is not the value of anything.
268    if riemann_trivial_zero(s, a) {
269        return Some(Field::from_i64(0, evaluator));
270    }
271    let requested = evaluator.requested_precision_bits();
272    let mut working = stepped_working_precision(requested.saturating_add(ZETA_GUARD_BITS));
273    for _ in 0..ZETA_ESCALATION_BUDGET {
274        if working > ZETA_MAXIMUM_WORKING_PRECISION {
275            return None;
276        }
277        let (value, lost) = hurwitz_zeta_attempt(s, a, working)?;
278        if requested.saturating_add(lost) <= working {
279            return Some(value.at_precision(requested));
280        }
281        working = stepped_working_precision(
282            requested
283                .saturating_add(lost)
284                .saturating_add(ZETA_GUARD_BITS),
285        );
286    }
287    None
288}

290fn riemann_trivial_zero<Field: GammaField>(s: &Field, a: &Field) -> bool {
291    // ζ(-2k) = 0 for every positive integer k. The reflection makes that vanishing structural
292    // — sin(πs/2) is zero at those orders and takes the whole sum with it — but a value that
293    // is exactly zero has no leading bits for the precision policy to count, and asking how
294    // many of them survived would escalate without end. The trivial zeros are named here
295    // instead, which is also the exact answer rather than an approximation to it.
296    if !s.is_real() || !a.is_real() {
297        return false;
298    }
299    let (Some(order), Some(shift)) = (s.real_component().to_f64(), a.real_component().to_f64())
300    else {
301        return false;
302    };
303    if shift != 1.0 || order >= 0.0 || order.fract() != 0.0 {
304        return false;
305    }
306    if order < -(i64::MAX as f64) {
307        return false;
308    }
309    (order as i64) % 2 == 0
310}

312fn hurwitz_zeta_attempt<Field: GammaField>(
313    s: &Field,
314    a: &Field,
315    working_precision: u32,
316) -> Option<(Field, u32)> {
317    // One evaluation of ζ(s, a) at a stated working precision, by whichever of the two
318    // routes is exact there. The Euler–Maclaurin sum is the general one; it fails only
319    // where its direct terms have outgrown the value they add up to, and that is precisely
320    // the half-plane the reflection covers, so between them the two span the argument range.
321    let mut evaluator = NumericEvaluator::new(working_precision)?;
322    let wide_s = s.at_precision(working_precision);
323    let wide_a = a.at_precision(working_precision);
324    let bernoulli = bernoulli_even_indexed(term_count(&evaluator));
325    if reflection_sum_closes(&wide_s, &wide_a, working_precision)
326        && let Some(reflected) = hurwitz_reflection(&wide_s, &wide_a, &bernoulli, &mut evaluator)
327    {
328        return Some(reflected);
329    }
330    hurwitz_euler_maclaurin(&wide_s, &wide_a, &bernoulli, &mut evaluator)
331}

333fn reflection_sum_closes<Field: GammaField>(s: &Field, a: &Field, working_precision: u32) -> bool {
334    // Whether the reflected sum reaches the working precision inside the term budget. Its
335    // terms fall like n^{Re(s)-1}, so what is left past N is bounded by N^{Re(s)}, and the
336    // route is taken only when that bound is already below the last bit at
337    // N = REFLECTION_TERM_MAXIMUM. Hurwitz's formula is stated for a real shift, so a
338    // complex one keeps to the Euler–Maclaurin route whatever the order.
339    if !a.is_real() {
340        return false;
341    }
342    let (Some(real_order), Some(real_shift)) =
343        (s.real_component().to_f64(), a.real_component().to_f64())
344    else {
345        return false;
346    };
347    if !real_order.is_finite() || real_order >= 0.0 {
348        return false;
349    }
350    if !real_shift.is_finite() || real_shift <= 0.0 {
351        return false;
352    }
353    // A shift that is a whole number peels down to exactly 1, where every sine in the sum is
354    // sin(πs/2) and what is left is ζ(1-s) — one zeta at a real part above 1, which the
355    // Euler-Maclaurin route answers with no cancellation at all. That case reflects at every
356    // negative order. Any other shift has to carry the sum itself, and is reflected only
357    // where the sum's decay closes it inside the term budget.
358    if real_shift.fract() == 0.0 {
359        return true;
360    }
361    let needed = f64::from(working_precision.saturating_add(REFLECTION_MARGIN_BITS));
362    let budget = (REFLECTION_TERM_MAXIMUM / REFLECTION_ESTIMATE_SHARE) as f64;
363    -real_order * budget.log2() >= needed
364}

366fn hurwitz_reflection<Field: GammaField>(
367    s: &Field,
368    a: &Field,
369    bernoulli: &[Rational],
370    evaluator: &mut NumericEvaluator,
371) -> Option<(Field, u32)> {
372    // Hurwitz's formula (Apostol, Introduction to Analytic Number Theory, Theorem 12.6) in
373    // the form it takes for a real shift f in (0, 1]:
374    //
375    //   ζ(s, f) = 2 Γ(1-s) (2π)^{s-1} Σ_{n≥1} sin(πs/2 + 2πnf) n^{s-1},
376    //
377    // the two Dirichlet sums of the usual statement folded into one by the sine addition
378    // formula. At f = 1 every sine is sin(πs/2) and the sum is ζ(1-s), which is the Riemann
379    // functional equation. Both sides are analytic in s, so the identity holds for complex
380    // s wherever the sum converges, and it converges like n^{Re(s)-1}: it closes in a
381    // handful of terms exactly where the Euler–Maclaurin direct terms have grown past the
382    // value and destroyed it.
383    //
384    // A shift beyond 1 is peeled down to that range first, by the definition itself:
385    // ζ(s, f) = Σ_{j<m} (f+j)^{-s} + ζ(s, f+m), so ζ(s, a) = ζ(s, f) - Σ_{j<m} (f+j)^{-s}
386    // for f = a - m.
387    let precision = evaluator.requested_precision_bits();
388    let one = Field::from_i64(1, evaluator);
389    let two = Field::from_i64(2, evaluator);
390    let whole_steps = peeled_whole_steps(&a.real_component())?;
391    let fractional = a.subtract(&Field::from_i64(whole_steps, evaluator), evaluator);
392    // The peel lands on exactly 1 precisely when the shift was a whole number to begin with.
393    let fractional_is_whole = a.real_component().to_f64()?.fract() == 0.0;
394
395    let two_pi = Field::pi(evaluator).multiply(&two, evaluator);
396    let angle_step = two_pi.multiply(&fractional, evaluator);
397    let half_order_angle = Field::pi(evaluator)
398        .multiply(s, evaluator)
399        .divide(&two, evaluator)?;
400    let exponent = s.subtract(&one, evaluator);
401
402    // With the shift peeled to exactly 1 every sine is sin(πs/2 + 2πn) = sin(πs/2), and what
403    // is left of the sum is Σ n^{s-1} = ζ(1-s). That is one zeta at a real part above 1,
404    // where the Euler-Maclaurin terms are all of one size and nothing cancels, so the whole
405    // negative half-plane is answered by a single well-conditioned evaluation. This is the
406    // Riemann functional equation, and it is the reason an integer shift needs no sum.
407    let (bracket, series_loss, terms_taken) = if fractional_is_whole {
408        let (reflected, loss) = hurwitz_zeta_attempt(&one.subtract(s, evaluator), &one, precision)?;
409        let bracket = half_order_angle
410            .sine(evaluator)
411            .multiply(&reflected, evaluator);
412        (bracket, loss, 1i64)
413    } else {
414        // sin(πs/2 + 2πnf) = sin(πs/2)cos(2πnf) + cos(πs/2)sin(2πnf) with a real shift, so
415        // |sin(πs/2)| + |cos(πs/2)| bounds every sine in the sum however far n runs. What
416        // remains past a term is therefore bounded by that times n^{Re(s)-1}, which is what
417        // decides when the sum has closed — the term's own size would not, because a sine
418        // landing on a zero says nothing whatever about the terms behind it.
419        let sine_bound = half_order_angle
420            .sine(evaluator)
421            .magnitude_real(evaluator)
422            .add(
423                &half_order_angle.cosine(evaluator).magnitude_real(evaluator),
424                evaluator,
425            );
426        let mut bracket = Field::from_i64(0, evaluator);
427        let mut largest_term_bound = RealInexactNumber::zero(precision);
428        let mut closed = false;
429        let mut terms_taken = 0i64;
430        for index in 1..=REFLECTION_TERM_MAXIMUM {
431            let index_value = Field::from_i64(index, evaluator);
432            let power = field_general_power(&index_value, &exponent, evaluator)?;
433            let angle =
434                half_order_angle.add(&angle_step.multiply(&index_value, evaluator), evaluator);
435            let term = angle.sine(evaluator).multiply(&power, evaluator);
436            bracket = bracket.add(&term, evaluator);
437            terms_taken = index;
438            let remaining = power
439                .magnitude_real(evaluator)
440                .multiply(&sine_bound, evaluator);
441            if matches!(
442                largest_term_bound.partial_compare(&remaining),
443                Some(Ordering::Less)
444            ) {
445                largest_term_bound = remaining.clone();
446            }
447            if tail_is_negligible(
448                &remaining,
449                &bracket,
450                precision.saturating_add(REFLECTION_MARGIN_BITS),
451                evaluator,
452            ) {
453                closed = true;
454                break;
455            }
456        }
457        if !closed {
458            return None;
459        }
460        // Where the leading sines vanish — as they do for every other n at a half-integer
461        // order and a quarter shift — the terms that survive sit far below the largest one
462        // the sum carried, and the rounding those larger ones left behind is that much
463        // closer to the total. Those are leading bits gone from the answer exactly as
464        // cancelled ones are, so the sum is held to the same account.
465        let bracket_magnitude = bracket.magnitude_real(evaluator);
466        let loss = cancelled_leading_bits(
467            &largest_term_bound,
468            &bracket_magnitude,
469            precision,
470            evaluator,
471        );
472        (bracket, loss, terms_taken)
473    };
474
475    let gamma = crate::special_functions::gamma::gamma_value(
476        &one.subtract(s, evaluator),
477        bernoulli,
478        evaluator,
479    )?;
480    let circle_power = field_general_power(&two_pi, &exponent, evaluator)?;
481    let mut value = two
482        .multiply(&gamma, evaluator)
483        .multiply(&circle_power, evaluator)
484        .multiply(&bracket, evaluator);
485
486    let mut largest_contribution = value.magnitude_real(evaluator);
487    let negated_s = s.multiply(&Field::from_i64(-1, evaluator), evaluator);
488    for step in 0..whole_steps {
489        let base = fractional.add(&Field::from_i64(step, evaluator), evaluator);
490        let peeled = field_general_power(&base, &negated_s, evaluator)?;
491        retain_larger_magnitude(&mut largest_contribution, &peeled, evaluator);
492        value = value.subtract(&peeled, evaluator);
493    }
494
495    let value_magnitude = value.magnitude_real(evaluator);
496    let peeling_loss = cancelled_leading_bits(
497        &largest_contribution,
498        &value_magnitude,
499        precision,
500        evaluator,
501    );
502    let angle_loss = sine_argument_bits(s, &fractional, terms_taken, evaluator)?;
503    Some((value, series_loss.max(peeling_loss).max(angle_loss)))
504}

506fn sine_argument_bits<Field: GammaField>(
507    s: &Field,
508    fractional: &Field,
509    terms_taken: i64,
510    evaluator: &mut NumericEvaluator,
511) -> Option<u32> {
512    // The sine arguments run up to |πs/2| + 2πNf, and an argument of that size is itself
513    // only known to the working precision, so the sine of it loses the bits that measure
514    // the argument. Those bits are lost to the answer exactly as cancelled ones are, and
515    // are reported the same way so the caller carries them.
516    let order_magnitude = s.magnitude_real(evaluator).to_f64()?;
517    let shift_magnitude = fractional.magnitude_real(evaluator).to_f64()?;
518    if !order_magnitude.is_finite() || !shift_magnitude.is_finite() {
519        return None;
520    }
521    let largest_angle = std::f64::consts::PI * 0.5 * order_magnitude
522        + std::f64::consts::TAU * (terms_taken as f64) * shift_magnitude;
523    let bits = (2.0 + largest_angle).log2().ceil();
524    if !bits.is_finite() || bits < 0.0 {
525        return None;
526    }
527    Some(bits as u32)
528}

530fn tail_is_negligible<Field: GammaField>(
531    remaining_bound: &RealInexactNumber,
532    total: &Field,
533    precision_bits: u32,
534    evaluator: &mut NumericEvaluator,
535) -> bool {
536    // Whether a bound on what a sum has left to add is below the last bit of what it has
537    // reached. The test is relative rather than absolute so that a sum whose terms all sit
538    // far below one is held to the same number of correct digits as one of size one.
539    let total_magnitude = total.magnitude_real(evaluator);
540    if total_magnitude.is_zero() {
541        return false;
542    }
543    let threshold = crate::special_functions::hypergeometric::numeric::negative_power_of_two(
544        precision_bits,
545        total_magnitude.precision_bits_value(),
546    );
547    let scaled = total_magnitude.multiply(&threshold, evaluator);
548    matches!(
549        remaining_bound.partial_compare(&scaled),
550        Some(Ordering::Less)
551    )
552}

554/// How many whole units a shift is lowered by before Hurwitz's formula applies to it.
555///
556/// The formula is stated for a shift in (0, 1], and a shift already there is left alone. A
557/// shift too large to peel within the term budget, or one no finite count describes,
558/// declines rather than being reduced approximately.
559fn peeled_whole_steps(real_shift: &RealInexactNumber) -> Option<i64> {
560    let value = real_shift.to_f64()?;
561    if !value.is_finite() || value <= 0.0 {
562        return None;
563    }
564    let steps = value.ceil() as i64 - 1;
565    if steps < 0 {
566        return Some(0);
567    }
568    if steps > REFLECTION_SHIFT_STEP_MAXIMUM {
569        return None;
570    }
571    Some(steps)
572}

610/// Performs the advanced native `simplify_call` operation.
611pub fn simplify_call(
612    context: &Context,
613    builtin: BuiltinFunction,
614    arguments: &[ExpressionReference],
615) -> Option<ExpressionReference> {
616    // Symbolic auto-simplification for the zeta family: the canonical
617    // hurwitz_zeta(s, 1) → zeta(s) fold, the classical special values, and the
618    // Hurwitz reductions. A None leaves the call unevaluated as a first-class head.
619    match builtin {
620        BuiltinFunction::Zeta if arguments.len() == 1 => zeta_special_value(context, &arguments[0]),
621        BuiltinFunction::HurwitzZeta if arguments.len() == 2 => {
622            hurwitz_special_value(context, &arguments[0], &arguments[1])
623        }
624        _ => None,
625    }
626}

656fn zeta_special_value(context: &Context, s: &ExpressionReference) -> Option<ExpressionReference> {
657    // ζ(1) → undefined (pole); even positive integers → the π^{2n} closed form;
658    // ζ(0) = -1/2; negative integers → -B_{m+1}/(m+1) (the trivial zeros included);
659    // odd positive integers (ζ(3), …) stay the symbolic head.
660    let value = match &s.kind {
661        ExpressionKind::Integer(value) => value,
662        _ => return None,
663    };
664    match value.sign_comparison() {
665        Ordering::Greater => {
666            if *value == Integer::ONE {
667                return Some(undefined_expression(context));
668            }
669            let argument = i64::try_from(value).ok()?;
670            if argument % 2 == 0 && argument <= EVEN_CLOSED_FORM_MAXIMUM_ARGUMENT {
671                Some(zeta_even_closed_form(context, argument / 2))
672            } else {
673                None
674            }
675        }
676        Ordering::Equal => Some(builders::rational_value(context, -1, 2)),
677        Ordering::Less => {
678            let order = -i64::try_from(value).ok()?;
679            if order > NEGATIVE_INTEGER_MAXIMUM_ORDER {
680                return None;
681            }
682            let bernoulli = bernoulli_numbers_up_to((order + 1) as usize);
683            let mut coefficient =
684                bernoulli[(order + 1) as usize].clone() / Rational::from(Integer::from(order + 1));
685            coefficient = -coefficient;
686            Some(builders::rational_literal(context, coefficient))
687        }
688    }
689}

691fn zeta_even_closed_form(context: &Context, half_argument: i64) -> ExpressionReference {
692    // ζ(2n) = (-1)^{n+1} B_{2n} (2π)^{2n} / (2 (2n)!) = (rational) · π^{2n}, the
693    // rational read off the shared Bernoulli table.
694    let double = 2 * half_argument;
695    let bernoulli = bernoulli_numbers_up_to(double as usize);
696    let two_to_double = crate::exact_arithmetic::integer_power(&Integer::from(2), double as u64);
697    let denominator = Integer::from(2) * factorial_integer(double);
698    let mut coefficient =
699        &bernoulli[double as usize] * Rational::from(two_to_double) / Rational::from(denominator);
700    if half_argument % 2 == 0 {
701        coefficient = -coefficient;
702    }
703    let coefficient_expression = builders::rational_literal(context, coefficient);
704    let pi = constant_expression(context, BuiltinConstant::Pi);
705    let exponent = builders::integer_value(context, double);
706    let pi_power = simplify_power(context, pi, exponent);
707    simplify_product(context, vec![coefficient_expression, pi_power])
708}

710fn hurwitz_special_value(
711    context: &Context,
712    s: &ExpressionReference,
713    a: &ExpressionReference,
714) -> Option<ExpressionReference> {
715    // The Hurwitz reductions. ζ(s,1) = ζ(s) is the canonical fold; non-positive
716    // integer s gives a value rational/polynomial in a (defined for every a);
717    // ζ(s,½) = (2^s-1)ζ(s); a positive integer n ≥ 2 reduces to ζ(s) minus the
718    // finite head Σ_{j<n} j^{-s}. Everything else stays the symbolic head.
719    if is_integer_one(a) {
720        return Some(zeta_head(context, s.clone()));
721    }
722    if let ExpressionKind::Integer(s_value) = &s.kind {
723        match s_value.sign_comparison() {
724            Ordering::Equal => {
725                let half = builders::rational_value(context, 1, 2);
726                let negated_a = builders::negate(context, a.clone());
727                return Some(simplify_sum(context, vec![half, negated_a]));
728            }
729            Ordering::Less => {
730                let order = -i64::try_from(s_value).ok()?;
731                return hurwitz_negative_integer(context, order, a);
732            }
733            Ordering::Greater => {}
734        }
735    }
736    if is_one_half(a) {
737        let two = builders::integer_value(context, 2);
738        let two_to_s = simplify_power(context, two, s.clone());
739        let minus_one = builders::integer_value(context, -1);
740        let factor = simplify_sum(context, vec![two_to_s, minus_one]);
741        let zeta_of_s = zeta_head(context, s.clone());
742        return Some(simplify_product(context, vec![factor, zeta_of_s]));
743    }
744    if let ExpressionKind::Integer(a_value) = &a.kind {
745        if a_value.sign_comparison() == Ordering::Greater {
746            let count = i64::try_from(a_value).ok()?;
747            if (2..=HURWITZ_POSITIVE_INTEGER_MAXIMUM).contains(&count) {
748                return Some(hurwitz_positive_integer(context, count, s));
749            }
750        }
751        return None;
752    }
753    None
754}
755
756fn hurwitz_positive_integer(
757    context: &Context,
758    count: i64,
759    s: &ExpressionReference,
760) -> ExpressionReference {
761    // ζ(s, n) = ζ(s) - Σ_{j=1}^{n-1} j^{-s} for a positive integer n ≥ 2.
762    let mut terms = vec![zeta_head(context, s.clone())];
763    for j in 1..count {
764        let base = builders::integer_value(context, j);
765        let negated_s = builders::negate(context, s.clone());
766        let power = simplify_power(context, base, negated_s);
767        terms.push(builders::negate(context, power));
768    }
769    simplify_sum(context, terms)
770}
771
772fn hurwitz_negative_integer(
773    context: &Context,
774    order: i64,
775    a: &ExpressionReference,
776) -> Option<ExpressionReference> {
777    // ζ(-m, a) = -B_{m+1}(a)/(m+1), the Bernoulli polynomial in a, for m ≥ 1.
778    let degree = order + 1;
779    if degree > HURWITZ_NEGATIVE_DEGREE_MAXIMUM {
780        return None;
781    }
782    let bernoulli = bernoulli_numbers_up_to(degree as usize);
783    let degree_rational = Rational::from(Integer::from(degree));
784    let mut terms: Vec<ExpressionReference> = Vec::new();
785    for lower in 0..=degree {
786        let bernoulli_value = &bernoulli[lower as usize];
787        if *bernoulli_value == Rational::ZERO {
788            continue;
789        }
790        let binomial = binomial_coefficient(degree, lower);
791        let mut coefficient = bernoulli_value * Rational::from(binomial);
792        coefficient = coefficient / &degree_rational;
793        coefficient = -coefficient;
794        let coefficient_expression = builders::rational_literal(context, coefficient);
795        let power = degree - lower;
796        if power == 0 {
797            terms.push(coefficient_expression);
798        } else {
799            let exponent = builders::integer_value(context, power);
800            let a_power = simplify_power(context, a.clone(), exponent);
801            terms.push(simplify_product(
802                context,
803                vec![coefficient_expression, a_power],
804            ));
805        }
806    }
807    Some(simplify_sum(context, terms))
808}

810pub(crate) fn polygamma_to_zeta(
811    context: &Context,
812    order: i64,
813    argument: ExpressionReference,
814) -> Option<ExpressionReference> {
815    // The bridge ψ^{(m)}(z) = (-1)^{m+1} m! ζ(m+1, z) for m ≥ 1. At z = 1 the
816    // Hurwitz head folds to ζ(m+1) (and then to a π form for even m+1); for a
817    // general z this is the opportunistic contraction the polygamma family exposes.
818    if order < 1 {
819        return None;
820    }
821    let order_plus_one = builders::integer_value(context, order + 1);
822    let hurwitz = hurwitz_head(context, order_plus_one, argument);
823    let mut coefficient = factorial_integer(order);
824    if order % 2 == 0 {
825        coefficient = -coefficient;
826    }
827    let coefficient_expression = builders::integer(context, coefficient);
828    Some(simplify_product(
829        context,
830        vec![coefficient_expression, hurwitz],
831    ))
832}

834fn binomial_coefficient(upper: i64, lower: i64) -> Integer {
835    // C(upper, lower) as an exact integer by the multiplicative formula; each
836    // partial product is itself a binomial coefficient, so the running division is
837    // always exact.
838    if lower < 0 || lower > upper {
839        return Integer::ZERO;
840    }
841    let mut result = Integer::from(1);
842    for step in 0..lower {
843        result *= Integer::from(upper - step);
844        result /= Integer::from(step + 1);
845    }
846    result
847}

850    matches!(&value.kind, ExpressionKind::Integer(integer_value) if *integer_value == Integer::ONE)
851}

855        &value.kind,
856        ExpressionKind::Rational(rational_value)
857            if *rational_value == Rational::from_integers(Integer::from(1), Integer::from(2))
858    )

885    /// The Bernoulli polynomial B_degree(shift) as an exact rational, from the shared
886    /// Bernoulli table: B_n(x) = Σ_k C(n, k) B_{n-k} x^k.
887    fn bernoulli_polynomial(degree: usize, shift: &Rational) -> Rational {
888        let bernoulli = bernoulli_numbers_up_to(degree);
889        let mut total = Rational::from(0i64);
890        let mut binomial = Integer::from(1);
891        let mut shift_power = Rational::from(1i64);
892        for lower in 0..=degree {
893            total = total
894                + Rational::from(binomial.clone())
895                    * bernoulli[degree - lower].clone()
896                    * shift_power.clone();
897            shift_power = shift_power * shift.clone();
898            binomial = binomial * Integer::from((degree - lower) as i64)
899                / Integer::from((lower + 1) as i64);
900        }
901        total
902    }

910    /// Whether a computed value carries the precision it was asked for, judged against the
911    /// exact value. The allowance is a few bits above the last one, which is what rounding
912    /// through a kernel's own intermediate steps costs.
913    fn reaches_requested_precision(
914        computed: &RealInexactNumber,
915        exact: &Rational,
916        precision: u32,
917        evaluator: &mut NumericEvaluator,
918    ) -> bool {
919        let exact_value = RealInexactNumber::from_rational_value(evaluator, exact);
920        if exact_value.is_zero() {
921            return computed.is_zero();
922        }
923        let relative = computed
924            .subtract(&exact_value, evaluator)
925            .divide(&exact_value, evaluator)
926            .absolute_value();
927        let allowance = crate::special_functions::hypergeometric::numeric::negative_power_of_two(
928            precision.saturating_sub(8),
929            precision,
930        );
931        matches!(
932            relative.partial_compare(&allowance),
933            Some(Ordering::Less) | Some(Ordering::Equal)
934        )
935    }
