46/// Performs the advanced native `evaluate_limit` operation.
47pub fn evaluate_limit(
48    context: &Context,
49    target: &ExpressionReference,
50    variable: SymbolIdentifier,
51    point: &ExpressionReference,
52    direction: LimitDirection,
53) -> ExpressionReference {
54    // Compute the limit of target as variable approaches point from the side
55    // given by direction. Expressions that do not depend on the variable are
56    // returned unchanged. Limits at negative infinity are reflected onto
57    // positive infinity by substituting variable -> -variable so the inner
58    // pipeline only has to specialise on a single infinity sign. Two-sided
59    // limits at finite points reduce to a pair of one-sided limits which must
60    // agree exactly under the active assumptions for the result to be definite; on
61    // disagreement an unevaluated_limit placeholder is returned.
62    if !contains_variable(target, variable) {
63        return Arc::clone(target);
64    }
65    if let Some(rewritten) = try_bound_operation_rewrite(context, target) {
66        return evaluate_limit(context, &rewritten, variable, point, direction);
67    }
68    if let Some(expanded) =
69        crate::special_functions::orthogonal_polynomials::series::finite_taylor_input(
70            context, target, variable,
71        )
72    {
73        return evaluate_limit(context, &expanded, variable, point, direction);
74    }
75    if is_negative_infinity(point) {
76        let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
77        let negated = builders::negate(context, Arc::clone(&variable_expression));
78        let mut mapping: SubstitutionMap = HashMap::new();
79        mapping.insert(variable_expression, negated);
80        let transformed = substitute(context, target, &mapping);
81        let positive_infinity = context
82            .interner
83            .intern(ExpressionKind::Constant(BuiltinConstant::Infinity));
84        let flipped = flip_direction_for_reflection(direction);
85        return evaluate_limit(context, &transformed, variable, &positive_infinity, flipped);
86    }
87    if direction == LimitDirection::TwoSided && !is_positive_infinity(point) {
88        let left = evaluate_limit(context, target, variable, point, LimitDirection::FromLeft);
89        let right = evaluate_limit(context, target, variable, point, LimitDirection::FromRight);
90        if expressions_are_exactly_equal(context, &left, &right) {
91            return left;
92        }
93        return unevaluated_limit_node(
94            context,
95            Arc::clone(target),
96            variable,
97            Arc::clone(point),
98            direction,
99        );
100    }
101    let effective_direction = if is_positive_infinity(point) {
102        LimitDirection::FromLeft
103    } else {
104        direction
105    };
106    let mut analysis = eventual_germ::GermAnalysis::new();
107    let result = limit_inner(
108        context,
109        &mut analysis,
110        target,
111        variable,
112        point,
113        effective_direction,
114        0,
115    );
116    crate::simplify::rational::presentation_normal_form(context, &result)
117}

119fn try_bound_operation_rewrite(
120    context: &Context,
121    target: &ExpressionReference,
122) -> Option<ExpressionReference> {
123    // Evaluate one reducible definite binder beneath the current limit before
124    // composing unrelated siblings. The traversal rebuilds only the ancestor
125    // path, so an expensive product or integral is not repeatedly evaluated.
126    match &target.kind {
127        ExpressionKind::Integral(
128            integrand,
129            bound_variable,
130            Some(lower_bound),
131            Some(upper_bound),
132        ) => {
133            let evaluated = integrate_definite(
134                context,
135                integrand,
136                *bound_variable,
137                Arc::clone(lower_bound),
138                Arc::clone(upper_bound),
139            );
140            (!Arc::ptr_eq(&evaluated, target)).then_some(evaluated)
141        }
142        ExpressionKind::IndexedProduct(
143            multiplicand,
144            index,
145            Some(lower_bound),
146            Some(upper_bound),
147        ) => {
148            let evaluated = product_definite(
149                context,
150                multiplicand,
151                *index,
152                Arc::clone(lower_bound),
153                Arc::clone(upper_bound),
154            );
155            (!Arc::ptr_eq(&evaluated, target)).then_some(evaluated)
156        }
157        ExpressionKind::Sum(children) => {
158            rewrite_first_bound_child(context, children, |values| simplify_sum(context, values))
159        }
160        ExpressionKind::Product(children) => {
161            rewrite_first_bound_child(context, children, |values| {
162                simplify_product(context, values)
163            })
164        }
165        ExpressionKind::Power(base, exponent) => {
166            if let Some(rewritten_base) = try_bound_operation_rewrite(context, base) {
167                return Some(builders::power(
168                    context,
169                    rewritten_base,
170                    Arc::clone(exponent),
171                ));
172            }
173            try_bound_operation_rewrite(context, exponent).map(|rewritten_exponent| {
174                builders::power(context, Arc::clone(base), rewritten_exponent)
175            })
176        }
177        _ => None,
178    }
179}

181fn rewrite_first_bound_child(
182    context: &Context,
183    children: &[ExpressionReference],
184    rebuild: impl FnOnce(Vec<ExpressionReference>) -> ExpressionReference,
185) -> Option<ExpressionReference> {
186    for (position, child) in children.iter().enumerate() {
187        if let Some(rewritten_child) = try_bound_operation_rewrite(context, child) {
188            let mut rewritten_children = children.to_vec();
189            rewritten_children[position] = rewritten_child;
190            return Some(rebuild(rewritten_children));
191        }
192    }
193    None
194}

210pub(crate) fn asymptotic_approach_from_limit(
211    point: &ExpressionReference,
212    direction: LimitDirection,
213) -> Option<AsymptoticApproach> {
214    // Single normalisation from a limit point and direction to the approach record
215    // the germ, bound and theorem layers key their certificates on. There is
216    // deliberately no second direction enum: a fact proved on one approach is never
217    // reused on another because the approach is part of every certificate.
218    if is_positive_infinity(point) {
219        return Some(AsymptoticApproach::PositiveInfinity);
220    }
221    if is_negative_infinity(point) {
222        return Some(AsymptoticApproach::NegativeInfinity);
223    }
224    Some(AsymptoticApproach::Finite {
225        point: Arc::clone(point),
226        direction: match direction {
227            LimitDirection::FromLeft => AsymptoticDirection::FromLeft,
228            LimitDirection::FromRight => AsymptoticDirection::FromRight,
229            LimitDirection::TwoSided => AsymptoticDirection::TwoSided,
230        },
231    })
232}

234fn try_germ_branch_rewrite(
235    context: &Context,
236    analysis: &mut eventual_germ::GermAnalysis,
237    target: &ExpressionReference,
238    variable: SymbolIdentifier,
239    point: &ExpressionReference,
240    direction: LimitDirection,
241) -> Option<ExpressionReference> {
242    // Local Abs and Signum reduction on the current germ. The rewrite is local to
243    // this limit problem: the original expression is never mutated and nothing is
244    // simplified globally, so a branch that only holds on one side of a point cannot
245    // escape into the caller's expression.
246    let approach = asymptotic_approach_from_limit(point, direction)?;
247    if !analysis.claim_rewrite_state(target, &approach, eventual_germ::RewriteFamily::GermBranch) {
248        return None;
249    }
250    let (rewritten, certificates) = eventual_germ::rewrite_absolute_value_and_signum(
251        analysis, context, target, variable, &approach,
252    )?;
253    for certificate in &certificates {
254        if !eventual_germ::replay_eventual_sign_certificate(context, certificate) {
255            return None;
256        }
257    }
258    Some(rewritten)
259}

261fn try_fixed_shift_gamma_limit(
262    context: &Context,
263    target: &ExpressionReference,
264    variable: SymbolIdentifier,
265    point: &ExpressionReference,
266) -> Option<ExpressionReference> {
267    // Gamma, factorial and log-Gamma values at affine arguments with fixed, possibly complex
268    // shifts, settled at positive infinity through the fixed-shift Stirling expansion with its
269    // remainders carried as bounded germs. Every certificate is replayed before the value is
270    // accepted.
271    if !is_positive_infinity(point) {
272        return None;
273    }
274    match special_function_asymptotics::try_fixed_shift_gamma_limit(context, target, variable) {
275        special_function_asymptotics::FixedShiftGammaLimit::Evaluated {
276            value,
277            certificates,
278        } => {
279            for certificate in &certificates {
280                if !special_function_asymptotics::replay_fixed_shift_gamma_certificate(
281                    context,
282                    certificate,
283                ) || !matches!(
284                    special_function_asymptotics::fixed_shift_gamma_theorem_certificate(
285                        context,
286                        certificate
287                    ),
288                    crate::asymptotic_series::AsymptoticCertificate::Theorem { .. }
289                ) {
290                    return None;
291                }
292            }
293            Some(value)
294        }
295        special_function_asymptotics::FixedShiftGammaLimit::NotApplicable
296        | special_function_asymptotics::FixedShiftGammaLimit::Declined(_) => None,
297    }
298}

300fn try_special_function_rewrite(
301    context: &Context,
302    analysis: &mut eventual_germ::GermAnalysis,
303    target: &ExpressionReference,
304    variable: SymbolIdentifier,
305    point: &ExpressionReference,
306    direction: LimitDirection,
307) -> Option<ExpressionReference> {
308    // Branch-safe factorial and Gamma normalization followed by the positive-real
309    // Stirling expansion. The rewrite is a candidate transformation only: it runs
310    // after the exact identity and simple-pole routes, and every certificate it
311    // produces is replayed before the rewritten expression is accepted.
312    let approach = asymptotic_approach_from_limit(point, direction)?;
313    if !analysis.claim_rewrite_state(
314        target,
315        &approach,
316        eventual_germ::RewriteFamily::SpecialFunction,
317    ) {
318        return None;
319    }
320    match special_function_asymptotics::try_special_function_normalization(
321        analysis, context, target, variable, &approach,
322    ) {
323        special_function_asymptotics::SpecialAsymptoticResult::Rewritten {
324            expression,
325            certificates,
326        } => {
327            for certificate in &certificates {
328                if !special_function_asymptotics::replay_special_asymptotic_certificate(
329                    context,
330                    certificate,
331                ) {
332                    return None;
333                }
334                let provenance =
335                    special_function_asymptotics::special_asymptotic_theorem_certificate(
336                        certificate,
337                    );
338                if !matches!(
339                    provenance,
340                    crate::asymptotic_series::AsymptoticCertificate::Theorem { .. }
341                ) {
342                    return None;
343                }
344            }
345            Some(expression)
346        }
347        special_function_asymptotics::SpecialAsymptoticResult::NotApplicable
348        | special_function_asymptotics::SpecialAsymptoticResult::Declined(_) => None,
349    }
350}

352fn try_bounded_perturbation(
353    context: &Context,
354    analysis: &mut eventual_germ::GermAnalysis,
355    target: &ExpressionReference,
356    variable: SymbolIdentifier,
357    point: &ExpressionReference,
358    direction: LimitDirection,
359) -> Option<ExpressionReference> {
360    // Squeeze route for a uniformly bounded perturbation of a divergent scale. It is
361    // attempted only after ordinary Gruntz classification has declined, so an
362    // expression the stronger existing path can decide keeps using that path.
363    let approach = asymptotic_approach_from_limit(point, direction)?;
364    let (value, _certificate) = eventual_germ::try_bounded_perturbation_theorem(
365        analysis, context, target, variable, &approach,
366    )?;
367    Some(match value {
368        eventual_germ::BoundedPerturbationValue::Finite(result) => result,
369        eventual_germ::BoundedPerturbationValue::PositiveInfinity => {
370            positive_infinity_expression(context)
371        }
372        eventual_germ::BoundedPerturbationValue::NegativeInfinity => {
373            negative_infinity_expression(context)
374        }
375    })
376}

378fn limit_inner(
379    context: &Context,
380    analysis: &mut eventual_germ::GermAnalysis,
381    target: &ExpressionReference,
382    variable: SymbolIdentifier,
383    point: &ExpressionReference,
384    direction: LimitDirection,
385    depth: usize,
386) -> ExpressionReference {
387    // Iterative core of the limit pipeline. Each call tries continuous symbolic
388    // substitution first, then a sequence of
389    // rewrites that reduce known indeterminate forms to a fresh limit problem
390    // dispatched back to this function with depth incremented. The depth bound
391    // prevents pathological inputs from looping forever; once exceeded the
392    // routine emits an unevaluated_limit placeholder so the caller still sees
393    // a composable expression rather than a panic.
394    if depth > MAX_RECURSION_DEPTH {
395        return unevaluated_limit_node(
396            context,
397            Arc::clone(target),
398            variable,
399            Arc::clone(point),
400            direction,
401        );
402    }
403    if !contains_variable(target, variable) {
404        return Arc::clone(target);
405    }
406    if let Some(value) = try_direct_substitution(context, target, variable, point) {
407        return value;
408    }
409    if let Some(value) = try_simple_pole_quotient_limit(context, target, variable, point, direction)
410    {
411        return value;
412    }
413    if let Some(rewrite) =
414        try_germ_branch_rewrite(context, analysis, target, variable, point, direction)
415    {
416        return limit_inner(
417            context,
418            analysis,
419            &rewrite,
420            variable,
421            point,
422            direction,
423            depth + 1,
424        );
425    }
426    if let Some(value) = try_fixed_shift_gamma_limit(context, target, variable, point) {
427        return value;
428    }
429    if let Some(rewrite) =
430        try_special_function_rewrite(context, analysis, target, variable, point, direction)
431    {
432        return limit_inner(
433            context,
434            analysis,
435            &rewrite,
436            variable,
437            point,
438            direction,
439            depth + 1,
440        );
441    }
442    if let Some(value) = try_binomial_hypergeometric_limit(context, target, variable, point) {
443        return value;
444    }
445    if let Some(value) = try_zeta_local_finite_part_limit(context, target, variable, point) {
446        return value;
447    }
448    match gruntz::try_gruntz_limit(context, target, variable, point, direction) {
449        gruntz::GruntzLimitResult::Evaluated(value) => return value,
450        gruntz::GruntzLimitResult::Declined(_) => {}
451    }
452    if let Some(value) = try_reciprocal_sine_limit(context, target, variable, point) {
453        return value;
454    }
455    if let Some(value) =
456        try_bounded_perturbation(context, analysis, target, variable, point, direction)
457    {
458        return value;
459    }
460    if let Some(value) =
461        try_vanishing_variable_logarithm_limit(context, target, variable, point, direction)
462    {
463        return value;
464    }
465    if let Some(rewrite) =
466        rewrite_eventually_positive_power(context, analysis, target, variable, point, direction)
467    {
468        return limit_inner(
469            context,
470            analysis,
471            &rewrite,
472            variable,
473            point,
474            direction,
475            depth + 1,
476        );
477    }
478    if let Some(value) =
479        try_compose_limit(context, analysis, target, variable, point, direction, depth)
480    {
481        return value;
482    }
483    if let Some(value) = try_analytic_quotient_limit(context, target, variable, point) {
484        return value;
485    }
486    if let Some(value) =
487        try_lhopital_limit(context, analysis, target, variable, point, direction, depth)
488    {
489        return value;
490    }
491    for candidate in
492        lhopital_product_candidates(context, analysis, target, variable, point, direction, depth)
493    {
494        if let Some(value) = try_lhopital_limit(
495            context,
496            analysis,
497            &candidate,
498            variable,
499            point,
500            direction,
501            depth + 1,
502        ) {
503            return value;
504        }
505    }
506    unevaluated_limit_node(
507        context,
508        Arc::clone(target),
509        variable,
510        Arc::clone(point),
511        direction,
512    )
513}

515fn try_zeta_local_finite_part_limit(
516    context: &Context,
517    target: &ExpressionReference,
518    variable: SymbolIdentifier,
519    point: &ExpressionReference,
520) -> Option<ExpressionReference> {
521    if is_positive_infinity(point) || is_negative_infinity(point) {
522        return None;
523    }
524    let ExpressionKind::Sum(terms) = &target.kind else {
525        return None;
526    };
527    for term in terms {
528        let ExpressionKind::FunctionCall(
529            FunctionIdentifier::Builtin(BuiltinFunction::Zeta),
530            arguments,
531        ) = &term.kind
532        else {
533            continue;
534        };
535        if arguments.len() != 1 {
536            continue;
537        }
538        let argument = &arguments[0];
539        let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
540        let mut substitution = HashMap::new();
541        substitution.insert(variable_expression, Arc::clone(point));
542        if !is_one(&substitute(context, argument, &substitution)) {
543            continue;
544        }
545        let slope = differentiate(context, argument, variable);
546        if contains_variable(&slope, variable)
547            || !matches!(
548                exact_expression_sign(&slope),
549                Some(Ordering::Greater | Ordering::Less)
550            )
551        {
552            continue;
553        }
554        let pole = builders::power(
555            context,
556            builders::subtract(context, Arc::clone(argument), builders::one(context)),
557            builders::integer_value(context, -1),
558        );
559        let canonical_target = builders::subtract(context, Arc::clone(term), pole);
560        if expressions_are_exactly_equal(context, target, &canonical_target) {
561            return Some(
562                context
563                    .interner
564                    .intern(ExpressionKind::Constant(BuiltinConstant::EulerGamma)),
565            );
566        }
567    }
568    None
569}

571fn try_binomial_hypergeometric_limit(
572    context: &Context,
573    target: &ExpressionReference,
574    variable: SymbolIdentifier,
575    point: &ExpressionReference,
576) -> Option<ExpressionReference> {
577    if !is_positive_infinity(point) {
578        return None;
579    }
580    let ExpressionKind::FunctionCall(
581        FunctionIdentifier::Builtin(BuiltinFunction::HypergeometricPfq),
582        arguments,
583    ) = &target.kind
584    else {
585        return None;
586    };
587    let (upper_parameters, lower_parameters, argument) =
588        crate::special_functions::hypergeometric::parameter_analysis::split_call_arguments(
589            arguments,
590        )?;
591    let binomial_power =
592        crate::special_functions::hypergeometric::reductions::binomial_power_reduction(
593            context,
594            upper_parameters,
595            lower_parameters,
596            argument,
597        )?;
598    let upper_parameter = upper_parameters.first()?;
599    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
600    let shift = builders::subtract(
601        context,
602        upper_parameter.clone(),
603        variable_expression.clone(),
604    );
605    if contains_variable(&shift, variable) {
606        return None;
607    }
608    let upper_slope = differentiate(context, upper_parameter, variable);
609    if !is_one(&upper_slope) {
610        return None;
611    }
612    let scaled_argument = simplify_product(context, vec![argument.clone(), variable_expression]);
613    if contains_variable(&scaled_argument, variable) {
614        return None;
615    }
616    let ExpressionKind::Power(base, exponent) = &binomial_power.kind else {
617        return None;
618    };
619    if contains_variable(base, variable) && contains_variable(exponent, variable) {
620        return Some(simplify_function_call(
621            context,
622            FunctionIdentifier::Builtin(BuiltinFunction::Exp),
623            vec![scaled_argument],
624        ));
625    }
626    None
627}

629fn try_direct_substitution(
630    context: &Context,
631    target: &ExpressionReference,
632    variable: SymbolIdentifier,
633    point: &ExpressionReference,
634) -> Option<ExpressionReference> {
635    // Direct substitution is valid only after the analytic layer has proved and
636    // replayed continuity of the whole expression at the point. This makes the
637    // pointwise simplification a consequence of a symbolic premise rather than
638    // an agreement between a few nearby floating-point evaluations.
639    if is_positive_infinity(point) || is_negative_infinity(point) {
640        return None;
641    }
642    if !expression_is_continuous_at_limit(context, target, variable, point)
643        && !is_proved_continuous_builtin_value(target, point)
644        && !is_proved_continuous_power_value(context, target, variable, point)
645    {
646        if !is_entire_special_function_value(target) {
647            return None;
648        }
649    }
650    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
651    let mut mapping: SubstitutionMap = HashMap::new();
652    mapping.insert(variable_expression, Arc::clone(point));
653    let substituted = substitute(context, target, &mapping);
654    if contains_undefined(&substituted) {
655        return None;
656    }
657    Some(substituted)
658}

660fn is_proved_continuous_power_value(
661    context: &Context,
662    target: &ExpressionReference,
663    variable: SymbolIdentifier,
664    point: &ExpressionReference,
665) -> bool {
666    let ExpressionKind::Power(base, exponent) = &target.kind else {
667        return false;
668    };
669    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
670    let mut mapping: SubstitutionMap = HashMap::new();
671    mapping.insert(variable_expression, Arc::clone(point));
672    let base_value = substitute(context, base, &mapping);
673    let exponent_is_real = matches!(
674        &exponent.kind,
675        ExpressionKind::Integer(_) | ExpressionKind::Rational(_)
676    );
677    exponent_is_real && exact_expression_sign(&base_value) == Some(Ordering::Greater)
678}

680fn is_proved_continuous_builtin_value(
681    target: &ExpressionReference,
682    point: &ExpressionReference,
683) -> bool {
684    let ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(function), arguments) =
685        &target.kind
686    else {
687        return false;
688    };
689    if arguments.len() != 1 {
690        return false;
691    }
692    matches!(
693        function,
694        BuiltinFunction::Sin | BuiltinFunction::Cos | BuiltinFunction::Exp
695    ) || (matches!(function, BuiltinFunction::Sqrt | BuiltinFunction::Log)
696        && exact_expression_sign(point) == Some(Ordering::Greater))
697        || (matches!(function, BuiltinFunction::Gamma)
698            && exact_expression_sign(point) == Some(Ordering::Greater))
699}

701fn is_entire_special_function_value(target: &ExpressionReference) -> bool {
702    matches!(
703        &target.kind,
704        ExpressionKind::FunctionCall(
705            FunctionIdentifier::Builtin(
706                BuiltinFunction::BesselJ
707                    | BuiltinFunction::BesselI
708                    | BuiltinFunction::SphericalBesselJ
709                    | BuiltinFunction::AiryAi
710                    | BuiltinFunction::AiryBi
711                    | BuiltinFunction::AiryAiPrime

780fn rewrite_eventually_positive_power(
781    context: &Context,
782    analysis: &mut eventual_germ::GermAnalysis,
783    target: &ExpressionReference,
784    variable: SymbolIdentifier,
785    point: &ExpressionReference,
786    direction: LimitDirection,
787) -> Option<ExpressionReference> {
788    // On a proved positive real germ, `base^exponent = exp(exponent * log(base))`
789    // is an exact identity. This is a domain-certified normalization, not an
790    // indeterminate-form guess.
791    let ExpressionKind::Power(base, exponent) = &target.kind else {
792        return None;
793    };
794    if !contains_variable(base, variable) && !contains_variable(exponent, variable) {
795        return None;
796    }
797    if proved_eventual_sign(context, analysis, base, variable, point, direction)
798        != Some(eventual_germ::EventualSign::Positive)
799    {
800        return None;
801    }
802    let logarithm = simplify_function_call(
803        context,
804        FunctionIdentifier::Builtin(BuiltinFunction::Log),
805        vec![Arc::clone(base)],
806    );
807    Some(simplify_function_call(
808        context,
809        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
810        vec![simplify_product(
811            context,
812            vec![Arc::clone(exponent), logarithm],
813        )],
814    ))
815}

817fn try_vanishing_variable_logarithm_limit(
818    context: &Context,
819    target: &ExpressionReference,
820    variable: SymbolIdentifier,
821    point: &ExpressionReference,
822    direction: LimitDirection,
823) -> Option<ExpressionReference> {
824    // The real one-sided theorem `u log(u) -> 0` for `u -> 0+` is applied only
825    // to the exact variable product. Its positivity and approach are explicit
826    // in the expression and direction; no sampled trajectory is involved.
827    if direction != LimitDirection::FromRight || !is_integer_zero_expression(point) {
828        return None;
829    }
830    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
831    let ExpressionKind::Product(factors) = &target.kind else {
832        return None;
833    };
834    if factors.len() != 2 {
835        return None;
836    }
837    let has_variable = factors
838        .iter()
839        .any(|factor| Arc::ptr_eq(factor, &variable_expression));
840    let has_logarithm = factors.iter().any(|factor| {
841        matches!(
842            &factor.kind,
843            ExpressionKind::FunctionCall(
844                FunctionIdentifier::Builtin(BuiltinFunction::Log),
845                arguments
846            ) if arguments.len() == 1 && Arc::ptr_eq(&arguments[0], &variable_expression)
847        )
848    });
849    (has_variable && has_logarithm).then(|| builders::zero(context))
850}

852fn try_reciprocal_sine_limit(
853    context: &Context,
854    target: &ExpressionReference,
855    variable: SymbolIdentifier,
856    point: &ExpressionReference,
857) -> Option<ExpressionReference> {
858    // Reparameterizing u = 1/x transforms x sin(1/x) at either infinity into
859    // sin(u)/u at u -> 0. The latter has equal first derivatives at zero, so
860    // its analytic quotient limit is exactly one.
861    if !is_positive_infinity(point) && !is_negative_infinity(point) {
862        return None;
863    }
864    let ExpressionKind::Product(factors) = &target.kind else {
865        return None;
866    };
867    if factors.len() != 2 {
868        return None;
869    }
870    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
871    let reciprocal = factors.iter().find_map(|factor| match &factor.kind {
872        ExpressionKind::FunctionCall(
873            FunctionIdentifier::Builtin(BuiltinFunction::Sin),
874            arguments,
875        ) if arguments.len() == 1 => Some(&arguments[0]),
876        _ => None,
877    })?;
878    let has_variable = factors
879        .iter()
880        .any(|factor| Arc::ptr_eq(factor, &variable_expression));
881    let reciprocal_is_variable = matches!(
882        &reciprocal.kind,
883        ExpressionKind::Power(base, exponent)
884            if Arc::ptr_eq(base, &variable_expression)
885                && matches!(&exponent.kind, ExpressionKind::Integer(value) if *value == Integer::from(-1))
886    );
887    (has_variable && reciprocal_is_variable).then(|| builders::one(context))
888}

890fn extract_quotient(
891    context: &Context,
892    target: &ExpressionReference,
893    variable: SymbolIdentifier,
894) -> Option<(ExpressionReference, ExpressionReference)> {
895    // Decompose target into (numerator, denominator) so that target equals
896    // numerator divided by denominator. Three shapes are recognised: a top
897    // level power(base, negative_exponent) acts as the reciprocal of
898    // base^(-exponent); a product with at least one factor carrying a
899    // negative exponent splits between the positive-exponent factors (the
900    // numerator) and the reciprocal of the negative-exponent factors (the
901    // denominator); and a top level function_call(Log, ...) is reported
902    // verbatim as a degenerate numerator-over-one quotient so the caller can
903    // still attempt a numerator-only rewrite path. Returns None when no
904    // denominator can be lifted out and the target therefore offers no
905    // L'Hopital handle.
906    let (numerator_factors, denominator_factors) = collect_quotient_factors(target);
907    if denominator_factors.is_empty() {
908        return None;
909    }
910    let _ = variable;
911    let numerator = assemble_product(context, numerator_factors);
912    let denominator = assemble_product(context, denominator_factors);
913    Some((numerator, denominator))
914}

916fn try_simple_pole_quotient_limit(
917    context: &Context,
918    target: &ExpressionReference,
919    variable: SymbolIdentifier,
920    point: &ExpressionReference,
921    direction: LimitDirection,
922) -> Option<ExpressionReference> {
923    if is_positive_infinity(point) || is_negative_infinity(point) {
924        return None;
925    }
926    let side_sign = one_sided_variable_difference_sign(direction)?;
927    let (numerator, denominator) = extract_quotient(context, target, variable)?;
928    let numerator_value = substitute_limit_point(context, &numerator, variable, point)?;
929    let numerator_sign = exact_expression_sign(&numerator_value)?;
930    if numerator_sign == Ordering::Equal {
931        return None;
932    }
933    let denominator_value = substitute_limit_point(context, &denominator, variable, point)?;
934    if !is_integer_zero_expression(&denominator_value) {
935        return None;
936    }
937    let derivative = differentiate(context, &denominator, variable);
938    let derivative_value = substitute_limit_point(context, &derivative, variable, point)?;
939    let derivative_sign = exact_expression_sign(&derivative_value)?;
940    if derivative_sign == Ordering::Equal {
941        return None;
942    }
943    let denominator_sign = multiply_ordering_signs(derivative_sign, side_sign);
944    let quotient_sign = multiply_ordering_signs(numerator_sign, denominator_sign);
945    match quotient_sign {
946        Ordering::Greater => Some(positive_infinity_expression(context)),
947        Ordering::Less => Some(negative_infinity_expression(context)),
948        Ordering::Equal => None,
949    }
950}

952fn try_lhopital_limit(
953    context: &Context,
954    analysis: &mut eventual_germ::GermAnalysis,
955    target: &ExpressionReference,
956    variable: SymbolIdentifier,
957    point: &ExpressionReference,
958    direction: LimitDirection,
959    depth: usize,
960) -> Option<ExpressionReference> {
961    // Apply l'Hopital's theorem only after exact child limits establish an
962    // admitted indeterminate form, analyticity proves differentiability near a
963    // finite point, and a replayed germ certificate proves that the denominator
964    // derivative is eventually nonzero. The derived quotient is then evaluated
965    // by the ordinary certified dispatcher.
966    let approach = asymptotic_approach_from_limit(point, direction)?;
967    if !analysis.claim_rewrite_state(target, &approach, eventual_germ::RewriteFamily::Lhopital) {
968        return None;
969    }
970    let (numerator, denominator) = extract_quotient(context, target, variable)?;
971    let numerator_kind = classify_proved_limit(
972        context, analysis, &numerator, variable, point, direction, depth,
973    );
974    let denominator_kind = classify_proved_limit(
975        context,
976        analysis,
977        &denominator,
978        variable,
979        point,
980        direction,
981        depth,
982    );
983    if !is_proved_lhopital_form(numerator_kind, denominator_kind)
984        || !expression_is_differentiable_on_germ(
985            context, analysis, &numerator, variable, point, direction,
986        )
987        || !expression_is_differentiable_on_germ(
988            context,
989            analysis,
990            &denominator,
991            variable,
992            point,
993            direction,
994        )
995    {
996        return None;
997    }
998    let differentiated_numerator = differentiate(context, &numerator, variable);
999    let differentiated_denominator = differentiate(context, &denominator, variable);
1000    if !matches!(
1001        proved_eventual_sign(
1002            context,
1003            analysis,
1004            &differentiated_denominator,
1005            variable,
1006            point,
1007            direction,
1008        ),
1009        Some(eventual_germ::EventualSign::Positive | eventual_germ::EventualSign::Negative)
1010    ) {
1011        return None;
1012    }
1013    let derivative_quotient = builders::divide(
1014        context,
1015        differentiated_numerator,
1016        differentiated_denominator,
1017    );
1018    let result = limit_inner(
1019        context,
1020        analysis,
1021        &derivative_quotient,
1022        variable,
1023        point,
1024        direction,
1025        depth + 1,
1026    );
1027    (!contains_unevaluated_limit(&result)).then_some(result)
1028}

1030fn try_analytic_quotient_limit(
1031    context: &Context,
1032    target: &ExpressionReference,
1033    variable: SymbolIdentifier,
1034    point: &ExpressionReference,
1035) -> Option<ExpressionReference> {
1036    // For analytic numerator and denominator, the first nonzero Taylor
1037    // derivatives determine their quotient's finite limit when the vanishing
1038    // orders agree. This is the finite-point Taylor theorem, recorded through
1039    // exact derivatives and substitutions rather than numerical sampling.
1040    if is_positive_infinity(point) || is_negative_infinity(point) {
1041        return None;
1042    }
1043    let (numerator, denominator) = extract_quotient(context, target, variable)?;
1044    if !expression_is_locally_analytic_at_limit(context, &numerator, variable, point)
1045        || !expression_is_locally_analytic_at_limit(context, &denominator, variable, point)
1046    {
1047        return None;
1048    }
1049    let (numerator_order, numerator_value) =
1050        first_nonzero_derivative_at_point(context, &numerator, variable, point)?;
1051    let (denominator_order, denominator_value) =
1052        first_nonzero_derivative_at_point(context, &denominator, variable, point)?;
1053    if numerator_order != denominator_order {
1054        return None;
1055    }
1056    Some(builders::divide(
1057        context,
1058        numerator_value,
1059        denominator_value,
1060    ))
1061}

1063fn expression_is_locally_analytic_at_limit(
1064    context: &Context,
1065    target: &ExpressionReference,
1066    variable: SymbolIdentifier,
1067    point: &ExpressionReference,
1068) -> bool {
1069    // A finite moving-endpoint integral of an analytic integrand is analytic
1070    // when both endpoints meet at the expansion point. This is the local
1071    // fundamental theorem of calculus, and admits the derivative recurrence
1072    // below without treating an arbitrary integral as an analytic expression.
1073    if expression_is_analytic_at_limit(context, target, variable, point) {
1074        return true;
1075    }
1076    match &target.kind {
1077        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
1078            return children.iter().all(|child| {
1079                expression_is_locally_analytic_at_limit(context, child, variable, point)
1080            });
1081        }
1082        ExpressionKind::Power(base, exponent) if matches!(&exponent.kind, ExpressionKind::Integer(value) if *value >= Integer::ZERO) =>
1083        {
1084            return expression_is_locally_analytic_at_limit(context, base, variable, point);
1085        }
1086        _ => {}
1087    }
1088    let ExpressionKind::Integral(integrand, bound_variable, Some(lower_bound), Some(upper_bound)) =
1089        &target.kind
1090    else {
1091        return false;
1092    };
1093    if !expression_is_analytic_at_limit(context, lower_bound, variable, point)
1094        || !expression_is_analytic_at_limit(context, upper_bound, variable, point)
1095    {
1096        return false;
1097    }
1098    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
1099    let mut substitution = HashMap::new();
1100    substitution.insert(variable_expression, Arc::clone(point));
1101    let lower_value = substitute(context, lower_bound, &substitution);
1102    let upper_value = substitute(context, upper_bound, &substitution);
1103    if !expressions_are_exactly_equal(context, &lower_value, &upper_value) {
1104        return false;
1105    }
1106    expression_is_analytic_at_limit(context, integrand, *bound_variable, &lower_value)
1107}

1109fn first_nonzero_derivative_at_point(
1110    context: &Context,
1111    target: &ExpressionReference,
1112    variable: SymbolIdentifier,
1113    point: &ExpressionReference,
1114) -> Option<(usize, ExpressionReference)> {
1115    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
1116    let mut mapping: SubstitutionMap = HashMap::new();
1117    mapping.insert(variable_expression, Arc::clone(point));
1118    let mut derivative = Arc::clone(target);
1119    for order in 0..=MAX_RECURSION_DEPTH {
1120        let value = substitute(context, &derivative, &mapping);
1121        if contains_undefined(&value) || contains_variable(&value, variable) {
1122            return None;
1123        }
1124        if exact_expression_sign(&value).is_some_and(|sign| sign != Ordering::Equal) {
1125            return Some((order, value));
1126        }
1127        derivative = differentiate(context, &derivative, variable);
1128    }
1129    None
1130}

1132fn expression_is_differentiable_on_germ(
1133    context: &Context,
1134    analysis: &mut eventual_germ::GermAnalysis,
1135    target: &ExpressionReference,
1136    variable: SymbolIdentifier,
1137    point: &ExpressionReference,
1138    direction: LimitDirection,
1139) -> bool {
1140    if expression_is_analytic_at_limit(context, target, variable, point) {
1141        return true;
1142    }
1143    match &target.kind {
1144        ExpressionKind::Symbol(identifier) => *identifier == variable,
1145        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
1146            children.iter().all(|child| {
1147                expression_is_differentiable_on_germ(
1148                    context, analysis, child, variable, point, direction,
1149                )
1150            })
1151        }
1152        ExpressionKind::Power(base, exponent) => match &exponent.kind {
1153            ExpressionKind::Integer(value) if *value >= Integer::ZERO => true,

1155                proved_eventual_sign(context, analysis, base, variable, point, direction)
1156                    .is_some_and(|sign| sign != eventual_germ::EventualSign::IdenticallyZero)
1157            }
1158            _ => false,
1159        },
1160        ExpressionKind::FunctionCall(
1161            FunctionIdentifier::Builtin(BuiltinFunction::Log),
1162            arguments,
1163        ) if arguments.len() == 1 => {
1164            proved_eventual_sign(context, analysis, &arguments[0], variable, point, direction)
1165                == Some(eventual_germ::EventualSign::Positive)
1166        }

1168            FunctionIdentifier::Builtin(
1169                BuiltinFunction::Sin
1170                | BuiltinFunction::Cos
1171                | BuiltinFunction::Tan
1172                | BuiltinFunction::Exp,
1173            ),
1174            arguments,
1175        ) if arguments.len() == 1 => expression_is_differentiable_on_germ(
1176            context,
1177            analysis,
1178            &arguments[0],
1179            variable,
1180            point,
1181            direction,
1182        ),
1183        _ => false,

1196fn classify_proved_limit(
1197    context: &Context,
1198    analysis: &mut eventual_germ::GermAnalysis,
1199    target: &ExpressionReference,
1200    variable: SymbolIdentifier,
1201    point: &ExpressionReference,
1202    direction: LimitDirection,
1203    depth: usize,
1204) -> ProvedLimitKind {
1205    if depth >= MAX_RECURSION_DEPTH {
1206        return ProvedLimitKind::Unknown;
1207    }
1208    let value = limit_inner(
1209        context,
1210        analysis,
1211        target,
1212        variable,
1213        point,
1214        direction,
1215        depth + 1,
1216    );
1217    if contains_unevaluated_limit(&value) || contains_variable(&value, variable) {
1218        return ProvedLimitKind::Unknown;
1219    }

1221        return ProvedLimitKind::Zero;
1222    }

1227        return ProvedLimitKind::NegativeInfinity;
1228    }
1229    exact_expression_sign(&value)
1230        .filter(|sign| *sign != Ordering::Equal)
1231        .map_or(ProvedLimitKind::Unknown, |_| ProvedLimitKind::FiniteNonzero)
1232}

1265fn one_sided_variable_difference_sign(direction: LimitDirection) -> Option<Ordering> {
1266    match direction {
1267        LimitDirection::FromLeft => Some(Ordering::Less),
1268        LimitDirection::FromRight => Some(Ordering::Greater),
1269        LimitDirection::TwoSided => None,
1270    }

1273fn exact_expression_sign(value: &ExpressionReference) -> Option<Ordering> {
1274    match &value.kind {
1275        ExpressionKind::Integer(integer_value) => Some(integer_value.sign_comparison()),
1276        ExpressionKind::Rational(rational_value) => Some(rational_value.sign_comparison()),
1277        ExpressionKind::Constant(BuiltinConstant::E)
1278        | ExpressionKind::Constant(BuiltinConstant::Pi) => Some(Ordering::Greater),
1279        ExpressionKind::Product(factors) => {
1280            let mut sign = Ordering::Greater;
1281            for factor in factors {
1282                sign = multiply_ordering_signs(sign, exact_expression_sign(factor)?);
1283            }
1284            Some(sign)
1285        }
1286        ExpressionKind::Power(base, exponent) => {
1287            let base_sign = exact_expression_sign(base)?;
1288            if base_sign == Ordering::Greater {
1289                return Some(Ordering::Greater);
1290            }
1291            if base_sign == Ordering::Equal {
1292                return Some(Ordering::Equal);
1293            }
1294            let exponent_value = match &exponent.kind {
1295                ExpressionKind::Integer(integer_value) => integer_value,
1296                _ => return None,
1297            };
1298            if exponent_value % Integer::from(2) == Integer::ZERO {
1299                Some(Ordering::Greater)
1300            } else {
1301                Some(Ordering::Less)
1302            }
1303        }
1304        _ => None,
1305    }
1306}
1307
1308fn multiply_ordering_signs(left: Ordering, right: Ordering) -> Ordering {
1309    if left == Ordering::Equal || right == Ordering::Equal {
1310        Ordering::Equal
1311    } else if left == right {
1312        Ordering::Greater
1313    } else {
1314        Ordering::Less
1315    }
1316}

1318fn collect_quotient_factors(
1319    target: &ExpressionReference,
1320) -> (Vec<ExpressionReference>, Vec<ExpressionReference>) {
1321    // Walk the immediate top-level structure of target and partition the
1322    // multiplicative factors into the positive-exponent group (numerator) and
1323    // the inverted-exponent group (denominator). Powers with a negative
1324    // numeric exponent move to the denominator with their exponent negated;
1325    // every other factor stays in the numerator. The decomposition is purely
1326    // structural — no algebraic rewriting is performed here.
1327    let mut numerator_parts: Vec<ExpressionReference> = Vec::new();
1328    let mut denominator_parts: Vec<ExpressionReference> = Vec::new();
1329    match &target.kind {
1330        ExpressionKind::Product(factors) => {
1331            for factor in factors {
1332                push_quotient_factor(factor, &mut numerator_parts, &mut denominator_parts);
1333            }
1334        }
1335        ExpressionKind::Power(_, _) => {
1336            push_quotient_factor(target, &mut numerator_parts, &mut denominator_parts);
1337        }
1338        _ => {
1339            numerator_parts.push(Arc::clone(target));
1340        }
1341    }
1342    (numerator_parts, denominator_parts)
1343}

1345fn push_quotient_factor(
1346    factor: &ExpressionReference,
1347    numerator_parts: &mut Vec<ExpressionReference>,
1348    denominator_parts: &mut Vec<ExpressionReference>,
1349) {
1350    // Place a single factor into the appropriate numerator or denominator
1351    // bucket. The decision is driven only by the sign of the immediate
1352    // exponent so a power(_, integer(-2)) lifts into the denominator with
1353    // exponent two while the variable-bearing base stays inside the lifted
1354    // factor.
1355    if let ExpressionKind::Power(base, exponent) = &factor.kind {
1356        if let Some(positive_exponent) = negate_numeric_if_negative(exponent) {
1357            let new_power = if is_one(&positive_exponent) {
1358                Arc::clone(base)
1359            } else {
1360                build_kind_unsimplified_power(Arc::clone(base), positive_exponent)
1361            };
1362            denominator_parts.push(new_power);
1363            return;
1364        }
1365    }
1366    numerator_parts.push(Arc::clone(factor));
1367}

1369fn negate_numeric_if_negative(exponent: &ExpressionReference) -> Option<ExpressionReference> {
1370    // If the exponent is a strictly negative numeric leaf return its negation
1371    // as a fresh expression reference so the lifted power lands in the
1372    // denominator with the correct positive exponent. Non-numeric or
1373    // non-negative exponents return None and the factor stays in the
1374    // numerator.
1375    match &exponent.kind {
1376        ExpressionKind::Integer(value) => {
1377            if *value < Integer::ZERO {
1378                let negated = -value.clone();
1379                Some(Arc::new(crate::expression::make_expression(
1380                    ExpressionKind::Integer(negated),
1381                )))
1382            } else {
1383                None
1384            }
1385        }
1386        ExpressionKind::Rational(value) => {
1387            if *value < Rational::ZERO {
1388                let negated = -value.clone();
1389                Some(Arc::new(crate::expression::make_expression(
1390                    ExpressionKind::Rational(negated),
1391                )))
1392            } else {
1393                None
1394            }
1395        }
1396        _ => None,
1397    }
1398}

1400fn build_kind_unsimplified_power(
1401    base: ExpressionReference,
1402    exponent: ExpressionReference,
1403) -> ExpressionReference {
1404    // Construct a raw power node without routing through the auto-simplifier.
1405    // The downstream pipeline only needs structural identity here; running the
1406    // simplifier could re-fold the freshly lifted denominator back into a
1407    // negative-exponent product and undo the quotient split.
1408    Arc::new(crate::expression::make_expression(ExpressionKind::Power(
1409        base, exponent,
1410    )))
1411}

1413fn assemble_product(context: &Context, factors: Vec<ExpressionReference>) -> ExpressionReference {
1414    // Multiply a list of factors back together through the auto-simplifying
1415    // product builder. An empty factor list collapses to the multiplicative
1416    // identity. Single-factor lists are returned directly so the result keeps
1417    // the cleanest possible expression shape for the downstream L'Hopital
1418    // application.
1419    if factors.is_empty() {
1420        return builders::one(context);
1421    }
1422    if factors.len() == 1 {
1423        return factors.into_iter().next().unwrap();
1424    }
1425    simplify_product(context, factors)
1426}

1428fn lhopital_product_candidates(
1429    context: &Context,
1430    analysis: &mut eventual_germ::GermAnalysis,
1431    target: &ExpressionReference,
1432    variable: SymbolIdentifier,
1433    point: &ExpressionReference,
1434    direction: LimitDirection,
1435    depth: usize,
1436) -> Vec<ExpressionReference> {
1437    // Handle the indeterminate 0 * infinity form by applying L'Hopital's rule
1438    // inline. Either half of the product may be reciprocated into the
1439    // denominator, and the two choices are genuinely different problems:
1440    // reciprocating the vanishing factor of x * Log(x) yields Log(x) / x^(-1),
1441    // whose derivative quotient collapses to -x, while reciprocating the
1442    // diverging factor yields -x * Log(x)^2, which is strictly worse. Both
1443    // candidates are returned, vanishing factor first, and the caller keeps the
1444    // first one that evaluates. Doing the rewrite inline sidesteps the
1445    // auto-simplifier folding the quotient form back into the original product,
1446    // which would cause the surrounding limit loop to spin.
1447    let factors = match &target.kind {
1448        ExpressionKind::Product(factors) => factors,
1449        _ => return Vec::new(),
1450    };
1451    let mut zero_index: Option<usize> = None;
1452    let mut infinite_index: Option<usize> = None;
1453    let factors: Vec<ExpressionReference> = factors.iter().map(Arc::clone).collect();
1454    for (index, factor) in factors.iter().enumerate() {
1455        let kind = classify_factor(context, analysis, factor, variable, point, direction, depth);
1456        match kind {
1457            ProvedLimitKind::Zero if zero_index.is_none() => zero_index = Some(index),
1458            ProvedLimitKind::PositiveInfinity | ProvedLimitKind::NegativeInfinity
1459                if infinite_index.is_none() =>
1460            {
1461                infinite_index = Some(index)
1462            }
1463            _ => {}
1464        }
1465    }
1466    let (zero_factor_index, infinite_factor_index) = match (zero_index, infinite_index) {
1467        (Some(zero), Some(infinite)) if zero != infinite => (zero, infinite),
1468        _ => return Vec::new(),
1469    };
1470    let mut candidates: Vec<ExpressionReference> = [zero_factor_index, infinite_factor_index]
1471        .into_iter()
1472        .filter_map(|reciprocated_index| {
1473            lhopital_product_rewrite(context, &factors, reciprocated_index, variable)
1474        })
1475        .collect();
1476    // Order by the size of the iterate, smallest first. Which half to reciprocate
1477    // is not decided by the roles of the factors: reciprocating the vanishing
1478    // factor of x * Log(x) at zero gives -x, while reciprocating the vanishing
1479    // factor of x * Sin(1/x) at infinity gives x^2*Sin(1/x)^2/Cos(1/x), a strictly
1480    // harder problem whose own indeterminate split recurs. The size of the derived
1481    // quotient separates the two cases; the roles do not.
1482    candidates.sort_by_key(crate::simplify::heuristic::node_count);
1483    // Only the top-level call may spend a second iterate. Exploring both branches
1484    // at every level is exponential in the search depth.
1485    if depth > 0 {
1486        candidates.truncate(1);
1487    }
1488    candidates
1489}

1491/// How deep the limit pipeline may spend an exact sub-limit on one product factor.
1492///
1493/// The exact fallbacks below are quadratic in the recursion depth, so they are
1494/// spent only near the top of the search, where the indeterminate 0 * infinity
1495/// split actually needs them. Deeper iterates fall back to the cheap probe.
1496const EXACT_FACTOR_ANALYSIS_DEPTH_CEILING: usize = 1;
1497
1498fn classify_factor(
1499    context: &Context,
1500    analysis: &mut eventual_germ::GermAnalysis,
1501    factor: &ExpressionReference,
1502    variable: SymbolIdentifier,
1503    point: &ExpressionReference,
1504    direction: LimitDirection,
1505    depth: usize,
1506) -> ProvedLimitKind {
1507    // Classify one product factor from a recursively proved symbolic limit.
1508    if depth > EXACT_FACTOR_ANALYSIS_DEPTH_CEILING {
1509        return ProvedLimitKind::Unknown;
1510    }
1511    let exact = limit_inner(
1512        context,
1513        analysis,
1514        factor,
1515        variable,
1516        point,
1517        direction,
1518        depth + 1,
1519    );
1520    if contains_unevaluated_limit(&exact) || contains_variable(&exact, variable) {
1521        return ProvedLimitKind::Unknown;
1522    }
1523    if is_positive_infinity(&exact) {
1524        return ProvedLimitKind::PositiveInfinity;
1525    }
1526    if is_negative_infinity(&exact) {
1527        return ProvedLimitKind::NegativeInfinity;
1528    }
1529    if is_integer_zero_expression(&exact) {
1530        return ProvedLimitKind::Zero;
1531    }
1532    ProvedLimitKind::Unknown
1533}

1535fn lhopital_product_rewrite(
1536    context: &Context,
1537    factors: &[ExpressionReference],
1538    reciprocated_index: usize,
1539    variable: SymbolIdentifier,
1540) -> Option<ExpressionReference> {
1541    // Build an exactly equivalent quotient from a product with a vanishing and
1542    // divergent factor. The l'Hopital theorem is applied separately, after its
1543    // premises have been certified for this quotient.
1544    let mut numerator_factors: Vec<ExpressionReference> = Vec::new();
1545    for (index, factor) in factors.iter().enumerate() {
1546        if index == reciprocated_index {
1547            continue;
1548        }
1549        numerator_factors.push(Arc::clone(factor));
1550    }
1551    let reciprocated_factor = Arc::clone(&factors[reciprocated_index]);
1552    let numerator = assemble_product(context, numerator_factors);
1553    let denominator = build_kind_unsimplified_power(
1554        reciprocated_factor,
1555        Arc::new(crate::expression::make_expression(ExpressionKind::Integer(
1556            Integer::from(-1),
1557        ))),
1558    );
1559    let _ = variable;
1560    Some(Arc::new(crate::expression::make_expression(
1561        ExpressionKind::Product(vec![numerator, denominator]),
1562    )))
1563}

1565fn proved_eventual_sign(
1566    context: &Context,
1567    analysis: &mut eventual_germ::GermAnalysis,
1568    target: &ExpressionReference,
1569    variable: SymbolIdentifier,
1570    point: &ExpressionReference,
1571    direction: LimitDirection,
1572) -> Option<eventual_germ::EventualSign> {
1573    // Exact directed sign of a germ, proved by the eventual-germ engine and
1574    // replayed independently of the search path. No floating probe is admitted
1575    // here: a probe may order candidate work, but it may never authorise a
1576    // symbolic verdict.
1577    let approach = asymptotic_approach_from_limit(point, direction)?;
1578    let certificate =
1579        eventual_germ::prove_eventual_sign(analysis, context, target, variable, &approach).ok()?;
1580    if !eventual_germ::replay_eventual_sign_certificate(context, &certificate) {
1581        return None;
1582    }
1583    Some(certificate.sign)
1584}

1586fn directed_exceptional_composition(
1587    context: &Context,
1588    analysis: &mut eventual_germ::GermAnalysis,
1589    builtin: BuiltinFunction,
1590    argument: &ExpressionReference,
1591    argument_limit: &ExpressionReference,
1592    variable: SymbolIdentifier,
1593    point: &ExpressionReference,
1594    direction: LimitDirection,
1595) -> Option<ExpressionReference> {
1596    // One-sided values of a builtin at a point where it is not continuous.
1597    // These are germ facts, not pointwise values: the principal logarithm has
1598    // no value at zero yet tends to negative infinity along any approach on
1599    // which its argument is eventually positive, and the greatest-integer
1600    // function has a value at every integer that differs from its left limit
1601    // there. The eventual sign is always read from the argument's own approach.
1602    match builtin {
1603        BuiltinFunction::Log => {
1604            if !is_integer_zero_expression(argument_limit) {
1605                return None;
1606            }
1607            let sign =
1608                proved_eventual_sign(context, analysis, argument, variable, point, direction)?;
1609            if sign != eventual_germ::EventualSign::Positive {
1610                return None;
1611            }
1612            Some(negative_infinity_expression(context))
1613        }
1614        BuiltinFunction::Floor | BuiltinFunction::Ceiling => {
1615            let ExpressionKind::Integer(value) = &argument_limit.kind else {
1616                return None;
1617            };
1618            let value = value.clone();
1619            let offset = builders::negate(context, Arc::clone(argument_limit));
1620            let displacement = simplify_sum(context, vec![Arc::clone(argument), offset]);
1621            let sign =
1622                proved_eventual_sign(context, analysis, &displacement, variable, point, direction)?;
1623            let step = match (builtin, sign) {
1624                (BuiltinFunction::Floor, eventual_germ::EventualSign::Negative) => -1,
1625                (BuiltinFunction::Floor, eventual_germ::EventualSign::Positive) => 0,
1626                (BuiltinFunction::Ceiling, eventual_germ::EventualSign::Negative) => 0,
1627                (BuiltinFunction::Ceiling, eventual_germ::EventualSign::Positive) => 1,
1628                _ => return None,
1629            };
1630            let shifted = value + Integer::from(step);
1631            Some(builders::integer(context, shifted))
1632        }
1633        BuiltinFunction::Signum => {
1634            if !is_integer_zero_expression(argument_limit) {
1635                return None;
1636            }
1637            let sign =
1638                proved_eventual_sign(context, analysis, argument, variable, point, direction)?;
1639            let value = match sign {
1640                eventual_germ::EventualSign::Positive => 1,
1641                eventual_germ::EventualSign::Negative => -1,
1642                eventual_germ::EventualSign::IdenticallyZero => return None,
1643            };
1644            Some(builders::integer_value(context, value))
1645        }
1646        _ => None,
1647    }
1648}

1650fn composition_is_continuous_at_limit(
1651    context: &Context,
1652    builtin: BuiltinFunction,
1653    argument_limit: &ExpressionReference,
1654) -> bool {
1655    // Ask the shared analytic layer whether the outer builtin is continuous at
1656    // the inner limit value. Pointwise definedness at that value is not enough:
1657    // the greatest-integer function is defined at every integer and continuous
1658    // at none of them, and the principal logarithm is defined on its own cut
1659    // and continuous across none of it.
1660    let placeholder = context
1661        .symbol_table
1662        .intern_symbol("analytic_composition_variable");
1663    let placeholder_expression = context.interner.intern(ExpressionKind::Symbol(placeholder));
1664    let call = builders::function_call(
1665        context,
1666        FunctionIdentifier::Builtin(builtin),
1667        vec![placeholder_expression],
1668    );
1669    let goal = crate::assumptions::analytic_goal::AnalyticGoal::ContinuousAtPoint {
1670        expression: call,
1671        variable: placeholder,
1672        point: Arc::clone(argument_limit),
1673    };
1674    let judgment = crate::assumptions::analytic_query::ask_in_context(&goal, context);
1675    judgment.verdict == crate::assumptions::TruthValue::Yes
1676        && judgment.discharges(goal.requested_evidence_kind())
1677        && judgment.evidence.as_ref().is_some_and(|evidence| {
1678            crate::assumptions::analytic_evidence::replay_analytic_evidence(
1679                evidence,
1680                &context.symbol_table,
1681            )
1682        })
1683}

1685fn try_compose_limit(
1686    context: &Context,
1687    analysis: &mut eventual_germ::GermAnalysis,
1688    target: &ExpressionReference,
1689    variable: SymbolIdentifier,
1690    point: &ExpressionReference,
1691    direction: LimitDirection,
1692    depth: usize,
1693) -> Option<ExpressionReference> {
1694    // Recurse into structural shapes whose limit factors through their
1695    // children. For a continuous unary builtin the result is the builtin
1696    // applied to the child's limit; for sums the result is the sum of child
1697    // limits provided no opposite-sign infinities collide; for products the
1698    // result is the product of child limits provided no 0 * infinity pairing
1699    // appears. Power composition is intentionally not handled here because
1700    // its indeterminate cases (1^infinity, 0^0, infinity^0) require the
1701    // Exp-and-Log rewrite that the surrounding loop already performs.
1702    match &target.kind {
1703        ExpressionKind::FunctionCall(function, arguments) => {
1704            if arguments.len() != 1 {
1705                return None;
1706            }
1707            let builtin = match function {
1708                FunctionIdentifier::Builtin(value) => *value,
1709                FunctionIdentifier::User(_) => return None,
1710            };
1711            let argument_limit = limit_inner(
1712                context,
1713                analysis,
1714                &arguments[0],
1715                variable,
1716                point,
1717                direction,
1718                depth + 1,
1719            );
1720            if contains_variable(&argument_limit, variable)
1721                || contains_unevaluated_limit(&argument_limit)
1722                || contains_undefined(&argument_limit)
1723            {
1724                return None;
1725            }
1726            if is_positive_infinity(&argument_limit) {
1727                return apply_builtin_at_positive_infinity(context, builtin);
1728            }
1729            if is_negative_infinity(&argument_limit) {
1730                return apply_builtin_at_negative_infinity(context, builtin);
1731            }
1732            if contains_infinity_sentinel(&argument_limit) {
1733                return None;
1734            }
1735            if let Some(value) = directed_exceptional_composition(
1736                context,
1737                analysis,
1738                builtin,
1739                &arguments[0],
1740                &argument_limit,
1741                variable,
1742                point,
1743                direction,
1744            ) {
1745                return Some(value);
1746            }
1747            if !composition_is_continuous_at_limit(context, builtin, &argument_limit) {
1748                // Composing the outer builtin through the inner limit is valid
1749                // only where that builtin is continuous. Pointwise definedness
1750                // at the inner limit is not continuity there, so an unproved
1751                // continuity obligation declines rather than publishing a value.
1752                return None;
1753            }
1754            let composed = simplify_function_call(context, *function, vec![argument_limit]);
1755            if contains_undefined(&composed) {
1756                return None;
1757            }
1758            Some(composed)
1759        }
1760        ExpressionKind::Sum(children) => {
1761            let mut child_limits: Vec<ExpressionReference> = Vec::with_capacity(children.len());
1762            let mut sees_positive_infinity = false;
1763            let mut sees_negative_infinity = false;
1764            for child in children {
1765                let child_limit = limit_inner(
1766                    context,
1767                    analysis,
1768                    child,
1769                    variable,
1770                    point,
1771                    direction,
1772                    depth + 1,
1773                );
1774                if !is_safe_composition_limit(&child_limit, variable) {
1775                    return None;
1776                }
1777                if is_positive_infinity(&child_limit) {
1778                    sees_positive_infinity = true;
1779                }
1780                if is_negative_infinity(&child_limit) {
1781                    sees_negative_infinity = true;
1782                }
1783                child_limits.push(child_limit);
1784            }
1785            if sees_positive_infinity && sees_negative_infinity {
1786                return None;
1787            }
1788            if sees_positive_infinity {
1789                return Some(positive_infinity_expression(context));
1790            }
1791            if sees_negative_infinity {
1792                return Some(negative_infinity_expression(context));
1793            }
1794            Some(simplify_sum(context, child_limits))
1795        }
1796        ExpressionKind::Product(children) => {
1797            let mut child_limits: Vec<ExpressionReference> = Vec::with_capacity(children.len());
1798            let mut sees_zero = false;
1799            let mut sees_infinity = false;
1800            for child in children {
1801                let child_limit = limit_inner(
1802                    context,
1803                    analysis,
1804                    child,
1805                    variable,
1806                    point,
1807                    direction,
1808                    depth + 1,
1809                );
1810                if !is_safe_composition_limit(&child_limit, variable) {
1811                    return None;
1812                }
1813                if is_integer_zero_expression(&child_limit) {
1814                    sees_zero = true;
1815                }
1816                if is_positive_infinity(&child_limit) || is_negative_infinity(&child_limit) {
1817                    sees_infinity = true;
1818                }
1819                child_limits.push(child_limit);
1820            }
1821            if sees_zero && sees_infinity {
1822                return None;
1823            }
1824            if sees_infinity {
1825                let mut product_sign = Ordering::Greater;
1826                for child_limit in &child_limits {
1827                    let child_sign = if is_positive_infinity(child_limit) {
1828                        Ordering::Greater
1829                    } else if is_negative_infinity(child_limit) {
1830                        Ordering::Less
1831                    } else {
1832                        exact_expression_sign(child_limit)?
1833                    };
1834                    product_sign = multiply_ordering_signs(product_sign, child_sign);
1835                }
1836                return match product_sign {
1837                    Ordering::Greater => Some(positive_infinity_expression(context)),
1838                    Ordering::Less => Some(negative_infinity_expression(context)),
1839                    Ordering::Equal => None,
1840                };
1841            }
1842            Some(simplify_product(context, child_limits))
1843        }
1844        ExpressionKind::Power(base, exponent) => {
1845            let base_limit = limit_inner(
1846                context,
1847                analysis,
1848                base,
1849                variable,
1850                point,
1851                direction,
1852                depth + 1,
1853            );
1854            let exponent_limit = limit_inner(
1855                context,
1856                analysis,
1857                exponent,
1858                variable,
1859                point,
1860                direction,
1861                depth + 1,
1862            );
1863            if !is_finite_definite(&base_limit, variable)
1864                || !is_finite_definite(&exponent_limit, variable)
1865            {
1866                return None;
1867            }
1868            if is_integer_zero_expression(&base_limit)
1869                && is_integer_zero_expression(&exponent_limit)
1870            {
1871                return None;
1872            }
1873            Some(builders::power(context, base_limit, exponent_limit))
1874        }
1875        ExpressionKind::Integral(
1876            integrand,
1877            bound_variable,
1878            Some(lower_bound),
1879            Some(upper_bound),
1880        ) => {
1881            let evaluated = integrate_definite(
1882                context,
1883                integrand,
1884                *bound_variable,
1885                Arc::clone(lower_bound),
1886                Arc::clone(upper_bound),
1887            );
1888            if Arc::ptr_eq(&evaluated, target) {
1889                return None;
1890            }
1891            Some(limit_inner(
1892                context,
1893                analysis,
1894                &evaluated,
1895                variable,
1896                point,
1897                direction,
1898                depth + 1,
1899            ))
1900        }
1901        ExpressionKind::IndexedProduct(
1902            multiplicand,
1903            index,
1904            Some(lower_bound),
1905            Some(upper_bound),
1906        ) => {
1907            let evaluated = product_definite(
1908                context,
1909                multiplicand,
1910                *index,
1911                Arc::clone(lower_bound),
1912                Arc::clone(upper_bound),
1913            );
1914            if Arc::ptr_eq(&evaluated, target) {
1915                return None;
1916            }
1917            Some(limit_inner(
1918                context,
1919                analysis,
1920                &evaluated,
1921                variable,
1922                point,
1923                direction,
1924                depth + 1,
1925            ))
1926        }
1927        _ => None,
1928    }
1929}

2182fn contains_unevaluated_limit(value: &ExpressionReference) -> bool {
2183    // Recursive scan for the unevaluated_limit placeholder marker. The
2184    // placeholder is encoded as a User-named function call so the check looks
2185    // through every function_call argument list rather than matching a
2186    // dedicated kind variant.
2187    match &value.kind {
2188        ExpressionKind::FunctionCall(FunctionIdentifier::User(_), _) => {
2189            // Conservative: any User function call is treated as opaque so a
2190            // limit value that happens to contain User functions does not
2191            // route through the composition shortcut. The dedicated check is
2192            // not strictly required because limit_inner's contract returns
2193            // either a definite value, an infinity sentinel, or the
2194            // placeholder, but matching every User-named function keeps the
2195            // gate robust against future placeholder shapes.
2196            true
2197        }
2198        ExpressionKind::Integer(_)
2199        | ExpressionKind::Rational(_)
2200        | ExpressionKind::PolynomialRoot(_)
2201        | ExpressionKind::Symbol(_)
2202        | ExpressionKind::Constant(_) => false,

2204            children.iter().any(contains_unevaluated_limit)
