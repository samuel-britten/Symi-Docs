9/// Public native data structure named `expression_display`.
10pub struct ExpressionDisplay<'lifetime> {
11    expression: &'lifetime Expression,
12    symbol_table: &'lifetime SymbolTable,
13}
14
15/// Performs the advanced native `format_expression` operation.
16pub fn format_expression<'lifetime>(
17    expression: &'lifetime Expression,
18    symbol_table: &'lifetime SymbolTable,
19) -> ExpressionDisplay<'lifetime> {
20    // Pair an expression with its symbol table for the standard library Display
21    // trait. The wrapper carries borrows only; callers must ensure both inputs
22    // outlive the formatting call. The two-argument form keeps the public surface
23    // free of trait-object indirection while letting tests format any expression
24    // they construct in line with the rest of the suite.
25    ExpressionDisplay {
26        expression,
27        symbol_table,
28    }
29}

31impl<'lifetime> fmt::Display for ExpressionDisplay<'lifetime> {
32    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
33        // Delegate to the recursive infix writer that consults the paired symbol
34        // table for user-defined names.
35        write_expression(formatter, self.expression, self.symbol_table)
36    }
37}

39fn write_expression(
40    formatter: &mut fmt::Formatter<'_>,
41    reference: &Expression,
42    table: &SymbolTable,
43) -> fmt::Result {
44    // Dispatch one expression node to the per-kind writer. Atoms render directly;
45    // compound nodes hand off to dedicated writers that know which children need
46    // parentheses and how to re-arrange the canonical layout into infix syntax.
47    match &reference.kind {
48        ExpressionKind::Integer(value) => write!(formatter, "{}", value),
49        ExpressionKind::Rational(value) => write!(formatter, "{}", value),
50        ExpressionKind::PolynomialRoot(value) => {
51            formatter.write_str("polynomial_root([")?;
52            for (index, coefficient) in value.minimal_polynomial.coefficients.iter().enumerate() {
53                if index > 0 {
54                    formatter.write_str(", ")?;
55                }
56                write!(formatter, "{}", coefficient)?;
57            }
58            write!(formatter, "], {})", value.root_index)
59        }
60        ExpressionKind::Symbol(identifier) => formatter.write_str(&table.name(*identifier)),
61        ExpressionKind::Constant(value) => formatter.write_str(constant_name(value)),
62        ExpressionKind::Sum(children) => write_sum(formatter, children, table),
63        ExpressionKind::Product(children) => write_product(formatter, children, table, true),
64        ExpressionKind::Power(base, exponent) => write_power(formatter, base, exponent, table),
65        ExpressionKind::Factorial(argument) => write_factorial(formatter, argument, table),
66        ExpressionKind::FunctionCall(function, arguments) => {
67            write_function_call(formatter, *function, arguments, table)
68        }
69        ExpressionKind::Derivative(function, orders, arguments) => {
70            write_derivative(formatter, *function, orders, arguments, table)
71        }
72        ExpressionKind::Integral(integrand, variable, lower_bound, upper_bound) => write_integral(
73            formatter,
74            integrand,
75            *variable,
76            lower_bound.as_ref(),
77            upper_bound.as_ref(),
78            table,
79        ),
80        ExpressionKind::Summation(summand, index, lower_bound, upper_bound) => write_summation(
81            formatter,
82            summand,
83            *index,
84            lower_bound.as_ref(),
85            upper_bound.as_ref(),
86            table,
87        ),
88        ExpressionKind::IndexedProduct(multiplicand, index, lower_bound, upper_bound) => {
89            write_indexed_product(
90                formatter,
91                multiplicand,
92                *index,
93                lower_bound.as_ref(),
94                upper_bound.as_ref(),
95                table,
96            )
97        }
98        ExpressionKind::LaplaceTransform(target, time_variable, frequency_variable) => {
99            write_laplace_transform(
100                formatter,
101                target,
102                *time_variable,
103                *frequency_variable,
104                table,
105            )
106        }
107        ExpressionKind::InverseLaplaceTransform(target, frequency_variable, time_variable) => {
108            write_inverse_laplace_transform(
109                formatter,
110                target,
111                *frequency_variable,
112                *time_variable,
113                table,
114            )
115        }
116        ExpressionKind::FourierTransform(target, time_variable, frequency_variable) => {
117            write_fourier_transform(
118                formatter,
119                target,
120                *time_variable,
121                *frequency_variable,
122                table,
123            )
124        }
125        ExpressionKind::InverseFourierTransform(target, frequency_variable, time_variable) => {
126            write_inverse_fourier_transform(
127                formatter,
128                target,
129                *frequency_variable,
130                *time_variable,
131                table,
132            )
133        }
134        ExpressionKind::Piecewise(branches) => write_piecewise(formatter, branches, table),
135        ExpressionKind::OrdinaryDifferentialEquation(
136            equation,
137            dependent_variable_call,
138            independent_variable,
139        ) => write_ordinary_differential_equation(
140            formatter,
141            equation,
142            dependent_variable_call,
143            *independent_variable,
144            table,
145        ),
146        ExpressionKind::RecurrenceEquation(equation, sequence_variable_call, index_variable) => {
147            write_recurrence_equation(
148                formatter,
149                equation,
150                sequence_variable_call,
151                *index_variable,
152                table,
153            )
154        }
155        ExpressionKind::Interval(lower_bound, upper_bound, lower_inclusive, upper_inclusive) => {
156            write_interval(
157                formatter,
158                lower_bound,
159                upper_bound,
160                *lower_inclusive,
161                *upper_inclusive,
162                table,
163            )
164        }
165        ExpressionKind::FiniteSet(elements) => write_finite_set(formatter, elements, table),
166        ExpressionKind::SetUnion(operands) => {
167            write_set_call(formatter, "set_union", operands, table)
168        }
169        ExpressionKind::SetIntersection(operands) => {
170            write_set_call(formatter, "set_intersection", operands, table)
171        }
172        ExpressionKind::SetComplement(left_operand, right_operand) => {
173            write_set_complement(formatter, left_operand, right_operand, table)
174        }
175        ExpressionKind::ImageSet(parameter_variable, parameter_domain, lambda_expression) => {
176            write_image_set(
177                formatter,
178                *parameter_variable,
179                parameter_domain,
180                lambda_expression,
181                table,
182            )
183        }
184        ExpressionKind::ConditionSet(parameter_variable, base_set, condition) => {
185            write_condition_set(formatter, *parameter_variable, base_set, condition, table)
186        }
187        ExpressionKind::Distribution(identifier, parameters) => {
188            write_distribution(formatter, *identifier, parameters, table)
189        }
190        ExpressionKind::RandomVariable(name, underlying_distribution) => {
191            write_random_variable(formatter, *name, underlying_distribution, table)
192        }
193        ExpressionKind::SpecialFunctionParameterVector(parameters) => {
194            write_special_function_parameter_vector(formatter, parameters, table)
195        }
196    }
197}

199fn write_special_function_parameter_vector(
200    formatter: &mut fmt::Formatter<'_>,
201    parameters: &[ExpressionReference],
202    table: &SymbolTable,
203) -> fmt::Result {
204    // Render an ordered parameter group in bracket-list form so the group
205    // boundaries of a generalized hypergeometric or Meijer-G call stay explicit
206    // and the reader can recover p, q, m, and n from the rendered lengths. The
207    // empty group renders as `[]` rather than vanishing, because a missing
208    // group and an empty one denote different functions.
209    formatter.write_str("[")?;
210    for (index, parameter) in parameters.iter().enumerate() {
211        if index > 0 {
212            formatter.write_str(", ")?;
213        }
214        write_expression(formatter, parameter, table)?;
215    }
216    formatter.write_str("]")
217}

219fn write_distribution(
220    formatter: &mut fmt::Formatter<'_>,
221    identifier: crate::statistics::identifier::DistributionIdentifier,
222    parameters: &[ExpressionReference],
223    table: &SymbolTable,
224) -> fmt::Result {
225    // Render a parametrised distribution in the call-form
226    // `name(parameter_1, parameter_2)` so the REPL round-trip stays
227    // self-evidently a distribution descriptor rather than a generic
228    // function call.
229    formatter.write_str(crate::statistics::identifier::distribution_name(identifier))?;
230    formatter.write_str("(")?;
231    for (index, parameter) in parameters.iter().enumerate() {
232        if index > 0 {
233            formatter.write_str(", ")?;
234        }
235        write_expression(formatter, parameter, table)?;
236    }
237    formatter.write_str(")")
238}

272    parameter_variable: SymbolIdentifier,
273    parameter_domain: &Expression,

290fn write_interval(
291    formatter: &mut fmt::Formatter<'_>,
292    lower_bound: &Expression,
293    upper_bound: &Expression,
294    lower_inclusive: bool,
295    upper_inclusive: bool,
296    table: &SymbolTable,
297) -> fmt::Result {
298    // Render an interval in textbook bracket notation. The opening bracket
299    // is "[" when the lower endpoint is included and "(" when excluded; the
300    // closing bracket mirrors with the upper endpoint's inclusivity.
301    formatter.write_str(if lower_inclusive { "[" } else { "(" })?;
302    write_expression(formatter, lower_bound, table)?;
303    formatter.write_str(", ")?;
304    write_expression(formatter, upper_bound, table)?;
305    formatter.write_str(if upper_inclusive { "]" } else { ")" })
306}

483fn write_piecewise(
484    formatter: &mut fmt::Formatter<'_>,
485    branches: &[(ExpressionReference, ExpressionReference)],
486    table: &SymbolTable,
487) -> fmt::Result {
488    // Render a piecewise expression as `piecewise((condition_1, value_1), ...)`.
489    // The shape parallels the integral display, prefixing the head with the
490    // builtin name and listing each branch as a parenthesised pair. The
491    // ordering of branches is significant — first-match semantics mirror
492    // SymPy's Piecewise — so the printed text preserves the stored sequence.
493    formatter.write_str("piecewise(")?;
494    for (index, (condition, value)) in branches.iter().enumerate() {
495        if index > 0 {
496            formatter.write_str(", ")?;
497        }
498        formatter.write_str("(")?;
499        write_expression(formatter, condition, table)?;
500        formatter.write_str(", ")?;
501        write_expression(formatter, value, table)?;
502        formatter.write_str(")")?;
503    }
504    formatter.write_str(")")
505}

507fn write_integral(
508    formatter: &mut fmt::Formatter<'_>,
509    integrand: &Expression,
510    variable: SymbolIdentifier,
511    lower_bound: Option<&ExpressionReference>,
512    upper_bound: Option<&ExpressionReference>,
513    table: &SymbolTable,
514) -> fmt::Result {
515    // Render an integral as a function-call-shaped surface form. The indefinite
516    // case prints `integrate(integrand, variable)`; the definite case appends
517    // the two bounds in order. The shape matches the parser's primary-rule
518    // dispatch so the printed text round-trips through parse_statement.
519    formatter.write_str("integrate(")?;
520    write_expression(formatter, integrand, table)?;
521    formatter.write_str(", ")?;
522    formatter.write_str(&table.name(variable))?;
523    if let (Some(lower), Some(upper)) = (lower_bound, upper_bound) {
524        formatter.write_str(", ")?;
525        write_expression(formatter, lower, table)?;
526        formatter.write_str(", ")?;
527        write_expression(formatter, upper, table)?;
528    }
529    formatter.write_str(")")
530}

584fn write_derivative(
585    formatter: &mut fmt::Formatter<'_>,
586    function: FunctionIdentifier,
587    orders: &[usize],
588    arguments: &[ExpressionReference],
589    table: &SymbolTable,
590) -> fmt::Result {
591    // Render an unevaluated symbolic derivative as `derivative(name, [orders], args)`.
592    // The flat surface form keeps the printed text round-trippable in principle
593    // and avoids any conflict with the `^` operator that would arise from the
594    // mathematical superscript convention.
595    formatter.write_str("derivative(")?;
596    formatter.write_str(&function_name(function, table))?;
597    formatter.write_str(", [")?;
598    for (index, order) in orders.iter().enumerate() {
599        if index > 0 {
600            formatter.write_str(", ")?;
601        }
602        write!(formatter, "{}", order)?;
603    }
604    formatter.write_str("]")?;
605    for argument in arguments {
606        formatter.write_str(", ")?;
607        write_expression(formatter, argument, table)?;
608    }
609    formatter.write_str(")")
610}

612fn constant_name(value: &BuiltinConstant) -> &'static str {
613    // Stable lowercase rendering for each builtin constant. The negative_infinity
614    // constant carries an explicit minus sign in its surface form so that the
615    // display string is unambiguous when read back.
616    match value {
617        BuiltinConstant::Pi => "pi",
618        BuiltinConstant::E => "e",
619        BuiltinConstant::EulerGamma => "euler_gamma",
620        BuiltinConstant::ImaginaryUnit => "i",
621        BuiltinConstant::Infinity => "infinity",
622        BuiltinConstant::NegativeInfinity => "-infinity",
623        BuiltinConstant::Undefined => "undefined",
624        BuiltinConstant::EmptySet => "empty_set",
625        BuiltinConstant::RealLine => "real_line",
626        BuiltinConstant::ComplexPlane => "complex_plane",
627        BuiltinConstant::IntegerSet => "integer_set",
628        BuiltinConstant::LogicalTrue => "true",
629        BuiltinConstant::LogicalFalse => "false",
630    }
631}

633fn write_sum(
634    formatter: &mut fmt::Formatter<'_>,
635    terms: &[ExpressionReference],
636    table: &SymbolTable,
637) -> fmt::Result {
638    // Print a canonical sum by walking its terms in presentation order. The first term
639    // emits its own sign (a leading minus when negative); subsequent terms emit
640    // an explicit ` + ` or ` - ` connector based on whether the term carries a
641    // negative leading factor, which is then stripped by the body writer.
642    for (index, term) in sum_terms_in_presentation_order(terms).enumerate() {
643        let negative = term_has_negative_sign(term);
644        if index == 0 {
645            if negative {
646                formatter.write_str("-")?;
647            }
648        } else {
649            formatter.write_str(if negative { " - " } else { " + " })?;
650        }
651        write_sum_term_body(formatter, term, table)?;
652    }
653    Ok(())
654}
655
656fn term_has_negative_sign(term: &Expression) -> bool {
657    // Detect whether a sum term carries a negative leading sign that the sum
658    // writer should hoist into the connector slot. Recognises bare negative
659    // numerics and products whose first canonical factor is a negative numeric.
660    match &term.kind {
661        ExpressionKind::Integer(value) => integer_is_negative(value),
662        ExpressionKind::Rational(value) => rational_is_negative(value),
663        ExpressionKind::Product(factors) => match factors.first() {
664            Some(first) => match &first.kind {
665                ExpressionKind::Integer(value) => integer_is_negative(value),
666                ExpressionKind::Rational(value) => rational_is_negative(value),
667                _ => false,
668            },
669            None => false,
670        },
671        _ => false,
672    }
673}
674
675fn write_sum_term_body(
676    formatter: &mut fmt::Formatter<'_>,
677    term: &Expression,
678    table: &SymbolTable,
679) -> fmt::Result {
680    // Write the absolute-value rendering of a sum term. Negative numerics are
681    // negated for printing; products with a negative leading factor are dispatched
682    // through the product writer with the leading sign suppressed so the sum's
683    // connector carries the only minus.
684    match &term.kind {
685        ExpressionKind::Integer(value) if integer_is_negative(value) => {
686            write!(formatter, "{}", -value.clone())
687        }
688        ExpressionKind::Rational(value) if rational_is_negative(value) => {
689            write!(formatter, "{}", -value.clone())
690        }
691        ExpressionKind::Product(factors) if term_has_negative_sign(term) => {
692            write_product(formatter, factors, table, false)
693        }
694        _ => write_expression(formatter, term, table),
695    }
696}
697
698fn write_product(
699    formatter: &mut fmt::Formatter<'_>,
700    factors: &[ExpressionReference],
701    table: &SymbolTable,
702    include_leading_sign: bool,
703) -> fmt::Result {
704    // Render a canonical product as infix multiplication and division. Splits a
705    // leading negative numeric coefficient off into a sign plus magnitude, then
706    // partitions the remaining factors into a numerator list and a denominator
707    // list. A factor of the form power(base, -1) is moved into the denominator
708    // and printed as `/base`; everything else is a numerator factor joined with
709    // `*`. When suppress_leading_sign is set, the sign is dropped from the output
710    // — used by the sum writer when the connector already carries the minus.
711    let (sign_negative, coefficient_text, factors_after_coefficient) =
712        split_leading_coefficient(factors);
713
714    let mut numerator: Vec<&Expression> = Vec::new();
715    let mut denominator: Vec<&Expression> = Vec::new();
716    for factor in factors_after_coefficient {
717        if let ExpressionKind::Power(base, exponent) = &factor.kind {
718            if is_integer_minus_one(exponent) {
719                denominator.push(base);
720                continue;
721            }
722        }
723        numerator.push(factor);
724    }
725
726    if sign_negative && include_leading_sign {
727        formatter.write_str("-")?;
728    }
729
730    let mut wrote_numerator_part = false;
731    if let Some(text) = &coefficient_text {
732        formatter.write_str(text)?;
733        wrote_numerator_part = true;
734    }
735    for factor in &numerator {
736        if wrote_numerator_part {
737            formatter.write_str("*")?;
738        }
739        write_in_product_factor_context(formatter, factor, table)?;
740        wrote_numerator_part = true;
741    }
742    if !wrote_numerator_part {
743        formatter.write_str("1")?;
744    }
745
746    for base in &denominator {
747        formatter.write_str("/")?;
748        write_in_product_factor_context(formatter, base, table)?;
749    }
750
751    Ok(())
752}
753
754fn split_leading_coefficient<'lifetime>(
755    factors: &'lifetime [ExpressionReference],
756) -> (bool, Option<String>, &'lifetime [ExpressionReference]) {
757    // Inspect the first factor of a canonical product. A leading numeric child is
758    // peeled off into a (sign, magnitude_text) pair so the body writer can place
759    // the magnitude before the rest of the factors and the sign in the slot that
760    // best fits the surrounding context. The integer literal -1 is special-cased
761    // to drop its magnitude entirely so that the sole sign remains.
762    let first = match factors.first() {
763        Some(first) => first,
764        None => return (false, None, factors),
765    };
766    match &first.kind {
767        ExpressionKind::Integer(value) if integer_is_negative_one(value) => {
768            (true, None, &factors[1..])
769        }
770        ExpressionKind::Integer(value) if integer_is_negative(value) => {
771            (true, Some(format!("{}", -value.clone())), &factors[1..])
772        }
773        ExpressionKind::Integer(value) => (false, Some(format!("{}", value)), &factors[1..]),
774        ExpressionKind::Rational(value) if rational_is_negative(value) => {
775            (true, Some(format!("{}", -value.clone())), &factors[1..])
776        }
777        ExpressionKind::Rational(value) => (false, Some(format!("{}", value)), &factors[1..]),
778        _ => (false, None, factors),
779    }
780}
781
782fn write_in_product_factor_context(
783    formatter: &mut fmt::Formatter<'_>,
784    factor: &Expression,
785    table: &SymbolTable,
786) -> fmt::Result {
787    // A product factor sits at multiplicative precedence. Sums must be wrapped
788    // because they have lower precedence; nested products do not appear thanks to
789    // ASAE flattening; powers, factorials, function calls, and atoms render in
790    // place. Negative numerics also need parentheses to avoid producing a literal
791    // such as `2*-3` whose precedence is ambiguous to a human reader.
792    let needs_parentheses =
793        matches!(&factor.kind, ExpressionKind::Sum(_)) || is_negative_numeric(factor);
794    if needs_parentheses {
795        formatter.write_str("(")?;
796    }
797    write_expression(formatter, factor, table)?;
798    if needs_parentheses {
799        formatter.write_str(")")?;
800    }
801    Ok(())
802}
803
804fn write_power(
805    formatter: &mut fmt::Formatter<'_>,
806    base: &ExpressionReference,
807    exponent: &ExpressionReference,
808    table: &SymbolTable,
809) -> fmt::Result {
810    // Render a power as either `base^exponent` or, when the exponent is exactly
811    // the integer literal -1, as `1/base`. The reciprocal form mirrors the way
812    // the product writer already prints denominator factors and gives the user
813    // the natural reading of a multiplicative inverse.
814    if is_integer_minus_one(exponent) {
815        formatter.write_str("1/")?;
816        return write_in_product_factor_context(formatter, base, table);
817    }
818    write_power_base(formatter, base, table)?;
819    formatter.write_str("^")?;
820    write_power_exponent(formatter, exponent, table)
821}
822
823fn write_power_base(
824    formatter: &mut fmt::Formatter<'_>,
825    base: &Expression,
826    table: &SymbolTable,
827) -> fmt::Result {
828    // Wrap the base of a power in parentheses whenever its kind would otherwise
829    // bind looser than the caret operator: sums, products, nested powers, and
830    // any signed or rational numeric leaf are all wrapped.
831    let needs_parentheses = matches!(
832        &base.kind,
833        ExpressionKind::Sum(_) | ExpressionKind::Product(_) | ExpressionKind::Power(_, _)
834    ) || is_signed_or_rational_numeric(base);
835    if needs_parentheses {
836        formatter.write_str("(")?;
837    }
838    write_expression(formatter, base, table)?;
839    if needs_parentheses {
840        formatter.write_str(")")?;
841    }
842    Ok(())
843}
844
845fn write_power_exponent(
846    formatter: &mut fmt::Formatter<'_>,
847    exponent: &Expression,
848    table: &SymbolTable,
849) -> fmt::Result {
850    // Wrap the exponent of a power in parentheses whenever it could be confused
851    // with a continuation of the surrounding expression. Sums, products, signed
852    // numerics, and rationals all need protection; powers are right-associative
853    // and therefore safe without parens.
854    let needs_parentheses = matches!(
855        &exponent.kind,
856        ExpressionKind::Sum(_) | ExpressionKind::Product(_)
857    ) || is_signed_or_rational_numeric(exponent);
858    if needs_parentheses {
859        formatter.write_str("(")?;
860    }
861    write_expression(formatter, exponent, table)?;
862    if needs_parentheses {
863        formatter.write_str(")")?;
864    }
865    Ok(())
866}
867
868fn write_factorial(
869    formatter: &mut fmt::Formatter<'_>,
870    argument: &Expression,
871    table: &SymbolTable,
872) -> fmt::Result {
873    // Render `argument!`. The argument is wrapped whenever it is a compound
874    // expression because the postfix `!` binds tighter than every infix operator.
875    let needs_parentheses = matches!(
876        &argument.kind,
877        ExpressionKind::Sum(_) | ExpressionKind::Product(_) | ExpressionKind::Power(_, _)
878    ) || is_signed_or_rational_numeric(argument);
879    if needs_parentheses {
880        formatter.write_str("(")?;
881    }
882    write_expression(formatter, argument, table)?;
883    if needs_parentheses {
884        formatter.write_str(")")?;
885    }
886    formatter.write_str("!")
887}
888
889fn write_function_call(
890    formatter: &mut fmt::Formatter<'_>,
891    function: FunctionIdentifier,
892    arguments: &[ExpressionReference],
893    table: &SymbolTable,
894) -> fmt::Result {
895    // Render a function application using its canonical name and a comma-separated
896    // argument list inside parentheses. Argument expressions are written without
897    // additional grouping because the call's own parentheses already isolate them.
898    formatter.write_str(&function_name(function, table))?;
899    formatter.write_str("(")?;
900    for (index, argument) in arguments.iter().enumerate() {
901        if index > 0 {
902            formatter.write_str(", ")?;
903        }
904        write_expression(formatter, argument, table)?;
905    }
906    formatter.write_str(")")
907}

909fn is_integer_minus_one(reference: &Expression) -> bool {
910    // Predicate identifying the integer literal minus one. Used to recognise both
911    // a sign-only product coefficient and a reciprocal exponent in a power.
912    match &reference.kind {
913        ExpressionKind::Integer(value) => integer_is_negative_one(value),
914        _ => false,
915    }
916}
917
918fn is_negative_numeric(reference: &Expression) -> bool {
919    // True when the expression is an integer or rational numeric leaf with a
920    // strictly negative value. Used to decide when to wrap a factor in parens.
921    match &reference.kind {
922        ExpressionKind::Integer(value) => integer_is_negative(value),
923        ExpressionKind::Rational(value) => rational_is_negative(value),
924        _ => false,
925    }
926}
927
928fn is_signed_or_rational_numeric(reference: &Expression) -> bool {
929    // True when the expression is a numeric leaf that would render with an embedded
930    // sign or slash and therefore needs parentheses inside a power, factorial, or
931    // similar tightly-binding context.
932    match &reference.kind {
933        ExpressionKind::Integer(value) => integer_is_negative(value),
934        ExpressionKind::Rational(_) => true,
935        _ => false,
936    }
937}
