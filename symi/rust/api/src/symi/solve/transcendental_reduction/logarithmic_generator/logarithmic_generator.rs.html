1// Certified logarithmic-generator substitution.
2//
3// An equation built entirely from rational constants and nonnegative rational powers of a
4// single logarithm log(g(x)) becomes a polynomial in a generator v with u = log(g(x)) =
5// v^N, where N is the least common multiple of the power denominators. The polynomial is
6// solved completely, each root is checked to be the *principal* N-th root of its u, each
7// surviving u is checked to lie in the image of the principal logarithm, and only then is
8// it lifted through g(x) = exp(u).
9//
10// Every one of those four checks is a place where a naive substitution invents roots. The
11// completeness claim is the conjunction of all of them plus the polynomial solve's own
12// certificate; any single "unknown" collapses the whole route to a decline.
13//
14// The module also owns the branch-safe log(z^a) normalisation the reduction needs. That
15// rewrite is not the generic log-power rule: by the unwinding-number theorem of Bradford,
16// Corless, Davenport, Jeffrey & Watt, K(a log z) vanishes for every nonzero z exactly when
17// the real a satisfies -1 < a <= 1, so log(z^a) = a log(z) is admissible without a
18// positivity assumption precisely on that half-open interval — which covers sqrt and
19// excludes both log(z^2) and the boundary case a = -1.

39pub(crate) fn reduce(
40    context: &Context,
41    equation: &ExpressionReference,
42    variable: SymbolIdentifier,
43    domain: SolveDomain,
44    budget: &mut TranscendentalReductionBudget,
45) -> TranscendentalReductionOutcome {
46    // Normalise every safe log(z^a), collect the single logarithmic generator, rewrite the
47    // equation as a polynomial in the rational-power generator, solve it completely, and
48    // lift each certified root.
49    let normalised = normalise_safe_logarithm_powers(context, equation, variable, true);
50    let generator = match collect_generator(context, &normalised, variable) {
51        Some(generator) => generator,
52        None => return TranscendentalReductionOutcome::Declined,
53    };
54    let powers = match collect_generator_powers(&normalised, &generator, variable) {
55        Some(powers) => powers,
56        None => return TranscendentalReductionOutcome::Declined,
57    };
58    let index = match common_power_index(&powers) {
59        Some(index) => index,
60        None => return TranscendentalReductionOutcome::Declined,
61    };
62    let (rewritten, generator_variable) =
63        match rewrite_in_generator(context, &normalised, &generator, index, variable) {
64            Some(pair) => pair,
65            None => return TranscendentalReductionOutcome::Declined,
66        };
67    if !budget.admits_node_count(crate::simplify::heuristic::node_count(&rewritten)) {
68        return TranscendentalReductionOutcome::Declined;
69    }
70    let polynomial_roots =
71        match crate::solve::vec::solve_with_verdict(context, &rewritten, generator_variable) {
72            crate::solve::vec::SolveOutcome::CompleteRoots(roots) => roots,
73            _ => return TranscendentalReductionOutcome::Declined,
74        };
75    if !budget.admits_candidates(polynomial_roots.len()) {
76        return TranscendentalReductionOutcome::Declined;
77    }
78    budget.record_candidates(polynomial_roots.len());
79    let mut lifted_pieces: Vec<ExpressionReference> = Vec::new();
80    let mut representatives: Vec<ExpressionReference> = Vec::new();
81    for root in polynomial_roots {
82        let logarithm_value = match certify_generator_root(context, &root, index) {
83            Some(value) => value,
84            None => return TranscendentalReductionOutcome::Declined,
85        };
86        match principal_logarithm_image_verdict(context, &logarithm_value, domain) {
87            TruthValue::Yes => {}
88            TruthValue::No => continue,
89            TruthValue::Unknown => return TranscendentalReductionOutcome::Declined,
90        }
91        let exponentiated =
92            build_unary_call(context, BuiltinFunction::Exp, Arc::clone(&logarithm_value));
93        let lift_equation =
94            builders::subtract(context, Arc::clone(&generator), Arc::clone(&exponentiated));
95        let piece = crate::solve::solveset::solveset_for_reduction(
96            context,
97            &lift_equation,
98            variable,
99            domain,
100            budget,
101        );
102        if crate::solve::transcendental_reduction::composition::is_declined_set(&piece) {
103            return TranscendentalReductionOutcome::Declined;
104        }
105        if let Some(elements) = crate::solve::vec::solution_set_to_vec(&piece) {
106            for element in &elements {
107                if !crate::solve::transcendental_reduction::candidate_is_defined(
108                    context, equation, variable, element,
109                ) {
110                    return TranscendentalReductionOutcome::Declined;
111                }
112            }
113            representatives.extend(elements);
114        }
115        lifted_pieces.push(piece);
116    }
117    let solution_set = merge_pieces(context, lifted_pieces);
118    let representatives = crate::solve::transcendental_reduction::deduplicate(representatives);
119    TranscendentalReductionOutcome::Complete {
120        solution_set,
121        representative_roots: representatives,
122    }
123}

125fn merge_pieces(context: &Context, pieces: Vec<ExpressionReference>) -> ExpressionReference {
126    // Union of the per-root inverse images. An empty candidate list is a proof of
127    // emptiness here: every generator root was classified and every one was rejected.
128    match pieces.len() {
129        0 => empty_set(context),
130        1 => pieces.into_iter().next().unwrap(),
131        _ => {
132            let mut finite_elements: Vec<ExpressionReference> = Vec::new();
133            let mut others: Vec<ExpressionReference> = Vec::new();
134            for piece in pieces {
135                match crate::solve::vec::solution_set_to_vec(&piece) {
136                    Some(elements) => finite_elements.extend(elements),
137                    None => others.push(piece),
138                }
139            }
140            if others.is_empty() {
141                let elements = crate::solve::transcendental_reduction::deduplicate(finite_elements);
142                return if elements.is_empty() {
143                    empty_set(context)
144                } else {
145                    finite_set(context, elements)
146                };
147            }
148            if !finite_elements.is_empty() {
149                let elements = crate::solve::transcendental_reduction::deduplicate(finite_elements);
150                others.push(finite_set(context, elements));
151            }
152            set_union(context, others)
153        }
154    }
155}

157pub(crate) fn normalise_safe_logarithm_powers(
158    context: &Context,
159    expression: &ExpressionReference,
160    variable: SymbolIdentifier,
161    nonzero_obligation_is_discharged: bool,
162) -> ExpressionReference {
163    // Rewrite log(z^a) -> a*log(z) wherever the unwinding number is provably zero, and
164    // leave every other log(power) untouched. Applied bottom-up so a nested occurrence is
165    // normalised before its parent is examined.
166    //
167    // nonzero_obligation_is_discharged says whether the caller guarantees the z != 0 side
168    // condition some other way. The solve path sets it: the equation already contains
169    // log(z^a), so it is undefined at z = 0 to begin with, and every candidate root is
170    // re-checked against the *original* equation, which rejects any root the rewrite
171    // could only have introduced. A directed-simplifier caller leaves it false and gets
172    // the strict rule, which rewrites only a provably nonzero base.
173    let rebuilt = match &expression.kind {
174        ExpressionKind::Sum(children) => {
175            let mapped = children
176                .iter()
177                .map(|child| {
178                    normalise_safe_logarithm_powers(
179                        context,
180                        child,
181                        variable,
182                        nonzero_obligation_is_discharged,
183                    )
184                })
185                .collect();
186            crate::simplify::simplify_sum(context, mapped)
187        }
188        ExpressionKind::Product(children) => {
189            let mapped = children
190                .iter()
191                .map(|child| {
192                    normalise_safe_logarithm_powers(
193                        context,
194                        child,
195                        variable,
196                        nonzero_obligation_is_discharged,
197                    )
198                })
199                .collect();
200            crate::simplify::simplify_product(context, mapped)
201        }
202        ExpressionKind::Power(base, exponent) => {
203            let new_base = normalise_safe_logarithm_powers(
204                context,
205                base,
206                variable,
207                nonzero_obligation_is_discharged,
208            );
209            let new_exponent = normalise_safe_logarithm_powers(
210                context,
211                exponent,
212                variable,
213                nonzero_obligation_is_discharged,
214            );
215            crate::simplify::simplify_power(context, new_base, new_exponent)
216        }
217        ExpressionKind::FunctionCall(function, arguments) => {
218            let function = *function;
219            let mapped: Vec<ExpressionReference> = arguments
220                .iter()
221                .map(|argument| {
222                    normalise_safe_logarithm_powers(
223                        context,
224                        argument,
225                        variable,
226                        nonzero_obligation_is_discharged,
227                    )
228                })
229                .collect();
230            builders::function_call(context, function, mapped)
231        }
232        _ => return Arc::clone(expression),
233    };
234    match safe_logarithm_power_rewrite(context, &rebuilt, nonzero_obligation_is_discharged) {
235        Some(rewritten) => rewritten,
236        None => rebuilt,
237    }
238}

240pub(crate) fn safe_logarithm_power_rewrite(
241    context: &Context,
242    expression: &ExpressionReference,
243    nonzero_obligation_is_discharged: bool,
244) -> Option<ExpressionReference> {
245    // log(z^a) = a*log(z) - 2*pi*i*K(a*log z). The correction K vanishes for every nonzero
246    // z exactly when the real a satisfies -1 < a <= 1 (Bradford, Corless, Davenport,
247    // Jeffrey & Watt), so the rewrite is admitted for:
248    //
249    //   - an exact rational a with -1 < a <= 1, carrying the z != 0 obligation; or
250    //   - any exact real a when z is provably positive, since then a*log z is real.
251    //
252    // a = -1 is rejected without a positivity proof: on the negative real cut, negating
253    // log z crosses the half-open strip boundary. Integer exponents such as a = 2 are
254    // rejected for the same reason — this is deliberately not the generic log-power rule.
255    //
256    // The positivity route's realness requirement on `a` is not optional: for a purely
257    // imaginary a, a*log(z) is purely imaginary and can leave the fundamental strip
258    // entirely, so an exponent of unknown domain is refused there just as it is on the
259    // rational path.
260    let arguments = match &expression.kind {
261        ExpressionKind::FunctionCall(
262            FunctionIdentifier::Builtin(BuiltinFunction::Log),
263            arguments,
264        ) if arguments.len() == 1 => arguments,
265        _ => return None,
266    };
267    let (base, exponent) = match &arguments[0].kind {
268        ExpressionKind::Power(base, exponent) => (Arc::clone(base), Arc::clone(exponent)),
269        _ => return None,
270    };
271    let base_is_positive =
272        crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &base)
273            == TruthValue::Yes
274            && crate::assumptions::range_evidence::proves_real(&context.symbol_table, &exponent);
275    if !base_is_positive {
276        let value = rational_value_of(&exponent)?;
277        if !unwinding_number_is_zero(&value) {
278            return None;
279        }
280        if !nonzero_obligation_is_discharged
281            && !crate::expression::queries::is_provably_nonzero(context, &base)
282        {
283            return None;
284        }
285    }
286    let logarithm = build_unary_call(context, BuiltinFunction::Log, base);
287    Some(builders::product(context, vec![exponent, logarithm]))
288}

290fn unwinding_number_is_zero(exponent: &Rational) -> bool {
291    // The exact admissibility interval -1 < a <= 1 of the unwinding theorem.
292    let minus_one = Rational::from_integers(Integer::from(-1), Integer::from(1));
293    let one = Rational::from_integers(Integer::from(1), Integer::from(1));
294    *exponent > minus_one && *exponent <= one
295}

297fn rational_value_of(expression: &ExpressionReference) -> Option<Rational> {
298    match &expression.kind {
299        ExpressionKind::Integer(value) => {
300            Some(Rational::from_integers(value.clone(), Integer::from(1)))
301        }
302        ExpressionKind::Rational(value) => Some(value.clone()),
303        _ => None,
304    }

307fn collect_generator(
308    context: &Context,
309    equation: &ExpressionReference,
310    variable: SymbolIdentifier,
311) -> Option<ExpressionReference> {
312    // Select the one base logarithmic generator log(g(x)). Every variable-bearing
313    // logarithm must be the same call after normalisation, and the variable must occur
314    // only inside it; a second inequivalent generator, or a bare variable outside every
315    // logarithm, declines.
316    let mut found: Option<ExpressionReference> = None;
317    if !collect_generator_recursive(equation, variable, &mut found) {
318        return None;
319    }
320    let generator_call = found?;
321    let inner = match &generator_call.kind {
322        ExpressionKind::FunctionCall(_, arguments) if arguments.len() == 1 => {
323            Arc::clone(&arguments[0])
324        }
325        _ => return None,
326    };
327    let _ = context;
328    Some(inner)
329}

331fn collect_generator_recursive(
332    expression: &ExpressionReference,
333    variable: SymbolIdentifier,
334    found: &mut Option<ExpressionReference>,
335) -> bool {
336    // Walk the equation. A variable-bearing logarithm becomes (or must match) the
337    // generator and is not descended into; any other variable-bearing leaf or call is a
338    // rejection, because it would leave an occurrence of x outside the generator.
339    if is_variable_free(expression, variable) {
340        return true;
341    }
342    if let ExpressionKind::FunctionCall(
343        FunctionIdentifier::Builtin(BuiltinFunction::Log),
344        arguments,
345    ) = &expression.kind
346    {
347        if arguments.len() == 1 {
348            match found {
349                Some(existing) => return Arc::ptr_eq(existing, expression),
350                None => {
351                    *found = Some(Arc::clone(expression));
352                    return true;
353                }
354            }
355        }
356        return false;
357    }
358    match &expression.kind {
359        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => children
360            .iter()
361            .all(|child| collect_generator_recursive(child, variable, found)),
362        ExpressionKind::Power(base, exponent) => {
363            if contains_variable(exponent, variable) {
364                return false;
365            }
366            collect_generator_recursive(base, variable, found)
367        }
368        _ => false,
369    }
370}

372fn collect_generator_powers(
373    equation: &ExpressionReference,
374    generator: &ExpressionReference,
375    variable: SymbolIdentifier,
376) -> Option<Vec<Rational>> {
377    // Every exponent the generator logarithm appears under. The outer equation must be
378    // built from rational constants, sums, products, and nonnegative rational powers of
379    // u = log(g(x)); a negative power would put u in a denominator, where the principal
380    // root classification below has nothing to say.
381    let mut powers: Vec<Rational> = Vec::new();
382    if !collect_powers_recursive(
383        equation,
384        generator,
385        variable,
386        &Rational::from_integers(Integer::from(1), Integer::from(1)),
387        &mut powers,
388    ) {
389        return None;
390    }
391    if powers.is_empty() {
392        return None;
393    }
394    Some(powers)
395}

397fn collect_powers_recursive(
398    expression: &ExpressionReference,
399    generator: &ExpressionReference,
400    variable: SymbolIdentifier,
401    accumulated: &Rational,
402    powers: &mut Vec<Rational>,
403) -> bool {
404    if is_variable_free(expression, variable) {
405        return true;
406    }
407    if let ExpressionKind::FunctionCall(
408        FunctionIdentifier::Builtin(BuiltinFunction::Log),
409        arguments,
410    ) = &expression.kind
411    {
412        if arguments.len() == 1 && Arc::ptr_eq(&arguments[0], generator) {
413            let zero = Rational::from_integers(Integer::from(0), Integer::from(1));
414            if *accumulated < zero {
415                return false;
416            }
417            powers.push(accumulated.clone());
418            return true;
419        }
420        return false;
421    }
422    match &expression.kind {
423        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => children
424            .iter()
425            .all(|child| collect_powers_recursive(child, generator, variable, accumulated, powers)),
426        ExpressionKind::Power(base, exponent) => {
427            let exponent_value = match rational_value_of(exponent) {
428                Some(value) => value,
429                None => return false,
430            };
431            let combined = accumulated.clone() * exponent_value;
432            collect_powers_recursive(base, generator, variable, &combined, powers)
433        }
434        _ => false,
435    }
436}

438fn common_power_index(powers: &[Rational]) -> Option<i64> {
439    // The least common multiple of the exponent denominators, capped so a pathological
440    // denominator cannot demand an astronomically high-degree polynomial.
441    let mut index = crate::exact_arithmetic::Natural::from(1u32);
442    for power in powers {
443        let parts = crate::exact_arithmetic::rational_into_parts(power.clone());
444        index = crate::exact_arithmetic::natural_least_common_multiple(&index, &parts.denominator);
445    }
446    let value = i64::try_from(&crate::exact_arithmetic::natural_to_integer(index)).ok()?;
447    (value >= 1 && value <= MAXIMUM_COMMON_RATIONAL_POWER_INDEX).then_some(value)
448}

450fn rewrite_in_generator(
451    context: &Context,
452    equation: &ExpressionReference,
453    generator: &ExpressionReference,
454    index: i64,
455    variable: SymbolIdentifier,
456) -> Option<(ExpressionReference, SymbolIdentifier)> {
457    // Replace u = log(g(x)) by v^index throughout, producing a polynomial in the fresh
458    // generator symbol v. The substitution is exact by definition of v: it introduces no
459    // branch choice of its own, and the principal-root check below is what decides which
460    // v values correspond to a genuine u.
461    //
462    // The rewrite composes exponents arithmetically rather than leaving (v^index)^a for
463    // the simplifier: index was chosen as the least common multiple of the exponent
464    // denominators precisely so that index*a is an integer for every a appearing, and the
465    // simplifier could not perform that composition anyway — (v^2)^(1/2) is not v without
466    // a sign proof about v, which is exactly the branch question deferred to
467    // certify_generator_root.
468    let generator_symbol = context.symbol_table.fresh("logarithm_generator");
469    let generator_expression = context
470        .interner
471        .intern(ExpressionKind::Symbol(generator_symbol));
472    let one = Rational::from_integers(Integer::from(1), Integer::from(1));
473    let rewritten = rewrite_recursive(
474        context,
475        equation,
476        generator,
477        &generator_expression,
478        index,
479        variable,
480        &one,
481    )?;
482    if contains_variable(&rewritten, variable) {
483        return None;
484    }
485    Some((rewritten, generator_symbol))
486}

488fn rewrite_recursive(
489    context: &Context,
490    expression: &ExpressionReference,
491    generator: &ExpressionReference,
492    generator_expression: &ExpressionReference,
493    index: i64,
494    variable: SymbolIdentifier,
495    accumulated_exponent: &Rational,
496) -> Option<ExpressionReference> {
497    // Rebuild the equation with every occurrence of u = log(g(x)) replaced by v^index,
498    // carrying the exponent a subtree sits under so that u^a becomes v^(index*a) directly.
499    // A variable-free subtree raised to that exponent is rebuilt unchanged, because the
500    // enclosing power node already accounted for it.
501    if is_variable_free(expression, variable) {
502        return Some(Arc::clone(expression));
503    }
504    if let ExpressionKind::FunctionCall(
505        FunctionIdentifier::Builtin(BuiltinFunction::Log),
506        arguments,
507    ) = &expression.kind
508    {
509        if arguments.len() == 1 && Arc::ptr_eq(&arguments[0], generator) {
510            let index_value = Rational::from_integers(Integer::from(index), Integer::from(1));
511            let total = accumulated_exponent.clone() * index_value;
512            if !crate::exact_arithmetic::rational_is_integer(&total) {
513                return None;
514            }
515            let parts = crate::exact_arithmetic::rational_into_parts(total);
516            let exponent_value = i64::try_from(&parts.numerator).ok()?;
517            let exponent = builders::integer_value(context, exponent_value);
518            return Some(crate::simplify::simplify_power(
519                context,
520                Arc::clone(generator_expression),
521                exponent,
522            ));
523        }
524        return None;
525    }
526    match &expression.kind {
527        ExpressionKind::Sum(children) => {
528            let mut mapped = Vec::with_capacity(children.len());
529            for child in &children.clone() {
530                mapped.push(rewrite_recursive(
531                    context,
532                    child,
533                    generator,
534                    generator_expression,
535                    index,
536                    variable,
537                    accumulated_exponent,
538                )?);
539            }
540            Some(crate::simplify::simplify_sum(context, mapped))
541        }
542        ExpressionKind::Product(children) => {
543            let mut mapped = Vec::with_capacity(children.len());
544            for child in &children.clone() {
545                mapped.push(rewrite_recursive(
546                    context,
547                    child,
548                    generator,
549                    generator_expression,
550                    index,
551                    variable,
552                    accumulated_exponent,
553                )?);
554            }
555            Some(crate::simplify::simplify_product(context, mapped))
556        }
557        ExpressionKind::Power(base, exponent) => {
558            let base = Arc::clone(base);
559            let exponent_value = rational_value_of(exponent)?;
560            let combined = accumulated_exponent.clone() * exponent_value;
561            rewrite_recursive(
562                context,
563                &base,
564                generator,
565                generator_expression,
566                index,
567                variable,
568                &combined,
569            )
570        }
571        _ => None,
572    }
573}

575fn certify_generator_root(
576    context: &Context,
577    root: &ExpressionReference,
578    index: i64,
579) -> Option<ExpressionReference> {
580    // A polynomial root v corresponds to a genuine logarithm value only when v is the
581    // *principal* index-th root of u = v^index. This is the finite principal-complex
582    // companion to the shipped real common-power path; it introduces no complex CAD.
583    //
584    // Exact real nonnegative roots pass immediately: the principal root of a nonnegative
585    // real is the nonnegative real root. Everything else — a negative real, an algebraic
586    // number, a symbolic expression — is undecided by this bounded test, so the whole
587    // completeness claim declines rather than admitting a possibly-wrong branch.
588    let index_expression = builders::integer_value(context, index);
589    let logarithm_value =
590        crate::simplify::simplify_power(context, Arc::clone(root), index_expression);
591    if index == 1 {
592        return Some(logarithm_value);
593    }
594    let simplified = crate::simplify::simplify(context, Arc::clone(root));
595    match crate::assumptions::range_evidence::negative_verdict(&context.symbol_table, &simplified) {
596        TruthValue::No => Some(logarithm_value),
597        _ => None,
598    }
599}

601fn principal_logarithm_image_verdict(
602    context: &Context,
603    logarithm_value: &ExpressionReference,
604    domain: SolveDomain,
605) -> TruthValue {
606    // A retained u must lie in the image of the principal logarithm: -pi < Im u <= pi. A
607    // real-domain generator additionally requires u real. Exact real values pass both
608    // tests immediately (their imaginary part is zero); anything the exact machinery
609    // cannot classify returns unknown, which declines the completeness claim.
610    let is_real_value =
611        crate::assumptions::range_evidence::proves_real(&context.symbol_table, logarithm_value);
612    if is_real_value {
613        return TruthValue::Yes;
614    }
615    if domain == SolveDomain::Real {
616        return TruthValue::Unknown;
617    }
618    TruthValue::Unknown
619}

625    fn symbol_with(
626        context: &mut Context,
627        name: &str,
628        positive: bool,
629        nonzero: bool,
630        real: bool,
631    ) -> ExpressionReference {
632        // A symbol carrying exactly the assumptions a case needs. Symbols are used rather
633        // than numeric literals because ASAE folds a literal power away before the rewrite
634        // ever sees it — log(3^1) is log(3), which has no power node to rewrite.
635        let identifier = context.symbol_table.intern_symbol(name);
636        let mut predicates = Vec::new();
637        if real || positive {
638            predicates.push(crate::assumptions::CurrentScalarPredicate::Real);
639        }
640        if positive {
641            predicates.push(crate::assumptions::CurrentScalarPredicate::Positive);
642        }
643        if nonzero {
644            predicates.push(crate::assumptions::CurrentScalarPredicate::Nonzero);
645        }
646        context
647            .symbol_table
648            .declare_symbol_predicates(identifier, &predicates);
649        context.interner.intern(ExpressionKind::Symbol(identifier))
650    }

652    fn logarithm_of_power(
653        context: &mut Context,
654        base: ExpressionReference,
655        exponent_text: &str,
656    ) -> ExpressionReference {
657        let exponent = crate::serialize::parse::parse_serialized(context, exponent_text)
658            .expect("exponent must parse");
659        let power = crate::simplify::simplify_power(context, base, exponent);
660        build_unary_call(context, BuiltinFunction::Log, power)
661    }

663    #[test]
664    fn unwinding_zero_exponents_rewrite_over_a_nonzero_base() {
665        // The exact admissibility interval is -1 < a <= 1: those exponents make the
666        // unwinding correction K(a log z) vanish for every nonzero z, so log(z^a) really
667        // is a log(z) there — with no positivity assumption needed. The interval's upper
668        // endpoint a = 1 is not listed because ASAE folds z^1 to z at construction, so
669        // there is no power node left for the rewrite to see; the identity holds there
670        // trivially.
671        for exponent in [
672            "rational(\"1\", \"2\")",
673            "rational(\"-1\", \"2\")",
674            "rational(\"3\", \"4\")",
675            "rational(\"-99\", \"100\")",
676        ] {
677            let mut context = Context::new();
678            let base = symbol_with(&mut context, "z", false, true, false);
679            let call = logarithm_of_power(&mut context, base, exponent);
680            assert!(
681                safe_logarithm_power_rewrite(&mut context, &call, false).is_some(),
682                "exponent {exponent} should be admissible over a nonzero base"
683            );
684        }
685    }

689        // a = -1 sits exactly on the excluded boundary — on the negative real cut,
