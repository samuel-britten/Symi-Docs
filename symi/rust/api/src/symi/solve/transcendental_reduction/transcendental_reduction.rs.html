1// The shared transcendental-equation reduction layer.
2//
3// Both solve entries route through this one module so a reduction's soundness and its
4// completeness are decided once. The Vec-shaped `solve` and the set-shaped `solveset`
5// answer different questions — `Exp(x) = x` has infinitely many complex solutions, so
6// returning the single sound representative -W_0(-1) is useful while wrapping it in a
7// finite set would be false — and the outcome type below carries both pieces so neither
8// caller has to infer the other's answer.
9//
10// The layer never uses numerical evidence to decide anything. A transformation is
11// admitted because an exact identity licenses it, and a completeness claim is the
12// conjunction of every stage's certificate: a complete downstream polynomial solve does
13// not repair an implication-only upstream rewrite. Every resource limit, unproved
14// premise, unclassified candidate, and incomplete recursive inverse image yields
15// `declined`, which asserts nothing — never an empty set, which asserts a great deal.
16

41// Public behaviour budgets. Changing one changes what the solver will and will not
42// answer, so each is pinned here rather than tuned at a call site; a profiling change
43// needs a documented plan amendment, not an accidental unbounded retry.
44pub(crate) const MAXIMUM_REDUCTION_DEPTH: usize = 8;
45pub(crate) const MAXIMUM_REWRITTEN_NODE_COUNT: usize = 512;
46pub(crate) const MAXIMUM_COMMON_RATIONAL_POWER_INDEX: i64 = 32;
47pub(crate) const MAXIMUM_FINITE_CANDIDATES: usize = 64;

49// The reserved name of the real-domain realness probe. The space makes it unreachable from

51const REAL_DOMAIN_PROBE_NAME: &str = "real domain probe";

53#[derive(Debug)]
54pub(crate) enum TranscendentalReductionOutcome {
55    // What a reduction proved.
56    //
57    // `complete` carries a solution set that is the *whole* solution set over the
58    // selected domain, together with any closed-form representatives worth reporting
59    // through the Vec-shaped entry. Every representative is a member of the set.
60    //
61    // `verified_roots` carries roots that are each provably a solution and nothing more:
62    // no certificate says these are all of them. A set-shaped caller must never lift
63    // them into a finite set, because that shape is a completeness claim.
64    //
65    // `declined` asserts nothing at all.
66    Complete {
67        solution_set: ExpressionReference,
68        representative_roots: Vec<ExpressionReference>,
69    },
70    VerifiedRoots(Vec<ExpressionReference>),
71    Declined,
72}

74#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
75pub(crate) struct ReductionMeasure {
76    // The lexicographic termination measure, over the three fields below. A recursive
77    // solve is admitted only when the sub-equation's measure is strictly smaller, and
78    // the three components are non-negative integers, so any chain of reductions is
79    // finite. Visited-state detection is a second guard, not this proof.
80    pub(crate) transcendental_calls: usize,
81    pub(crate) variable_occurrences: usize,
82    pub(crate) nodes: usize,
83}

85// Identifies one dispatcher sub-problem within a single top-level solve: the equation by
86// its hash-consed pointer identity, the solve variable, the domain, and the two pieces of
87// mutable state a cached answer could otherwise silently outlive — the durable assumption
88// revision and the owning context. A key is never carried between top-level solves, so a
89// stale revision or context can only ever miss, never hit with a wrong answer.
90pub(crate) type DispatcherMemoKey = (usize, SymbolIdentifier, SolveDomain, u64, u64);

92pub(crate) struct TranscendentalReductionBudget {
93    // Resources shared by every reduction on one solve path. Threaded by reference
94    // through the set-shaped entries so a reduction that re-enters solveset cannot
95    // silently restart its own allowance.
96    remaining_depth: usize,
97    original_node_count: usize,
98    introduced_candidates: usize,
99    integer_family_active: bool,
100    visited: Vec<(usize, SymbolIdentifier, SolveDomain)>,
101    measure: Option<ReductionMeasure>,
102    dispatcher_memo: HashMap<DispatcherMemoKey, ExpressionReference>,
103}
104
105impl TranscendentalReductionBudget {
106    pub(crate) fn new(equation: &ExpressionReference) -> Self {
107        // A fresh allowance for one top-level solve.
108        TranscendentalReductionBudget {
109            remaining_depth: MAXIMUM_REDUCTION_DEPTH,
110            original_node_count: node_count(equation),
111            introduced_candidates: 0,
112            integer_family_active: false,
113            visited: Vec::new(),
114            measure: None,
115            dispatcher_memo: HashMap::new(),
116        }
117    }
118
119    pub(crate) fn dispatcher_memo_lookup(
120        &self,
121        key: &DispatcherMemoKey,
122    ) -> Option<ExpressionReference> {
123        self.dispatcher_memo.get(key).map(Arc::clone)
124    }
125
126    pub(crate) fn dispatcher_memo_insert(
127        &mut self,
128        key: DispatcherMemoKey,
129        result: ExpressionReference,
130    ) {
131        self.dispatcher_memo.insert(key, result);
132    }
133
134    pub(crate) fn admits_node_count(&self, count: usize) -> bool {
135        // Reject a rewrite before constructing it when its node estimate exceeds the cap,
136        // measured against both the absolute cap and the original equation's size.
137        count <= MAXIMUM_REWRITTEN_NODE_COUNT && count <= self.original_node_count * 64 + 64
138    }
139
140    pub(crate) fn admits_candidates(&self, count: usize) -> bool {
141        // Cap the number of finite candidates a single path may lift.
142        self.introduced_candidates + count <= MAXIMUM_FINITE_CANDIDATES
143    }
144
145    pub(crate) fn record_candidates(&mut self, count: usize) {
146        self.introduced_candidates += count;
147    }
148
149    pub(crate) fn integer_family_is_available(&self) -> bool {
150        // At most one newly introduced integer family per reduction path: a second one
151        // would need a two-parameter set expression that is not yet implemented.
152        !self.integer_family_active
153    }
154
155    pub(crate) fn claim_integer_family(&mut self) {
156        self.integer_family_active = true;
157    }
158
159    fn enter(
160        &mut self,
161        equation: &ExpressionReference,
162        variable: SymbolIdentifier,
163        domain: SolveDomain,
164    ) -> Option<ReductionFrame> {
165        // Admit one reduction step. The step is refused when the depth is exhausted, when
166        // this exact (equation, variable, domain) state has already been reduced on this
167        // path, or when the measure did not strictly decrease since the enclosing step.
168        if self.remaining_depth == 0 {
169            return None;
170        }
171        let state = (Arc::as_ptr(equation) as usize, variable, domain);
172        if self.visited.contains(&state) {
173            return None;
174        }
175        let measure = measure_of(equation, variable);
176        if let Some(enclosing) = self.measure {
177            if measure >= enclosing {
178                return None;
179            }
180        }
181        self.remaining_depth -= 1;
182        self.visited.push(state);
183        let restored = self.measure;
184        self.measure = Some(measure);
185        Some(ReductionFrame {
186            restored_measure: restored,
187        })
188    }
189
190    fn leave(&mut self, frame: ReductionFrame) {
191        // Release one reduction step, restoring the enclosing measure so a sibling branch
192        // is judged against the same bound this one was.
193        self.remaining_depth += 1;
194        self.visited.pop();
195        self.measure = frame.restored_measure;
196    }
197}

203pub(crate) fn measure_of(
204    equation: &ExpressionReference,
205    variable: SymbolIdentifier,
206) -> ReductionMeasure {
207    // Lexicographic size of a reduction state: how many variable-bearing transcendental
208    // calls it still contains, how many times the variable occurs, and how large it is.
209    ReductionMeasure {
210        transcendental_calls: count_variable_bearing_transcendental_calls(equation, variable),
211        variable_occurrences: count_variable_occurrences(equation, variable),
212        nodes: node_count(equation),
213    }
214}
215
216fn count_variable_bearing_transcendental_calls(
217    equation: &ExpressionReference,
218    variable: SymbolIdentifier,
219) -> usize {
220    // Count the transcendental function applications and variable-exponent powers whose
221    // argument carries the variable. These are exactly the obstructions the reduction
222    // layer exists to remove, so removing one is real progress.
223    record_counter!(expression_traversals);
224    let mut total = 0;
225    if !crate::solve::vec::contains_variable(equation, variable) {
226        return 0;
227    }
228    match &equation.kind {
229        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) => {
230            if is_transcendental_builtin(*builtin) {
231                total += 1;
232            }
233            for argument in arguments {
234                total += count_variable_bearing_transcendental_calls(argument, variable);
235            }
236        }
237        ExpressionKind::Power(base, exponent) => {
238            if crate::solve::vec::contains_variable(exponent, variable) {
239                total += 1;
240            }
241            total += count_variable_bearing_transcendental_calls(base, variable);
242            total += count_variable_bearing_transcendental_calls(exponent, variable);
243        }
244        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
245            for child in children {
246                total += count_variable_bearing_transcendental_calls(child, variable);
247            }
248        }
249        ExpressionKind::FunctionCall(FunctionIdentifier::User(_), arguments) => {
250            for argument in arguments {
251                total += count_variable_bearing_transcendental_calls(argument, variable);
252            }
253        }
254        _ => {}
255    }
256    total
257}

261    use BuiltinFunction::*;

274            | Tanh

287fn count_variable_occurrences(equation: &ExpressionReference, variable: SymbolIdentifier) -> usize {
288    // One full walk of the tree, counted as one traversal by the instrumentation
289    // layer; the recursion itself is `count_variable_occurrences_within`.
290    record_counter!(expression_traversals);
291    count_variable_occurrences_within(equation, variable)
292}
293
294fn count_variable_occurrences_within(
295    equation: &ExpressionReference,
296    variable: SymbolIdentifier,
297) -> usize {
298    // Count the leaf occurrences of the variable.
299    match &equation.kind {
300        ExpressionKind::Symbol(identifier) => usize::from(*identifier == variable),
301        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => children
302            .iter()
303            .map(|child| count_variable_occurrences_within(child, variable))
304            .sum(),
305        ExpressionKind::Power(base, exponent) => {
306            count_variable_occurrences_within(base, variable)
307                + count_variable_occurrences_within(exponent, variable)
308        }
309        ExpressionKind::FunctionCall(_, arguments) => arguments
310            .iter()
311            .map(|argument| count_variable_occurrences_within(argument, variable))
312            .sum(),
313        ExpressionKind::Factorial(argument) => {
314            count_variable_occurrences_within(argument, variable)
315        }
316        _ => 0,
317    }
318}

320pub(crate) fn reduce(
321    context: &Context,
322    equation: &ExpressionReference,
323    variable: SymbolIdentifier,
324    domain: SolveDomain,
325    budget: &mut TranscendentalReductionBudget,
326    profile: &EquationProfile,
327) -> TranscendentalReductionOutcome {
328    // The shared entry. Runs every *applicable* registered reducer on one admitted step
329    // and arbitrates between their outcomes; an exhausted budget, a repeated state, or a
330    // measure that failed to decrease declines before any reducer runs.
331    //
332    // Applicability is read off the equation profile, which is one walk of the tree the
333    // caller has already paid for. Each gate below is a necessary condition for its
334    // reducer to return anything but a decline, so a closed gate costs the equation
335    // nothing and can only ever skip a route that was going to decline. That is what
336    // keeps a polynomial equation from paying for four transcendental recognisers, and in
337    // particular what stops the logarithmic-generator route from rewriting the whole tree
338    // before discovering there is no logarithm in it.
339    record_counter!(transcendental_reduction_entries);
340    let exponential_or_logarithmic = profile.admits_an_exponential_or_logarithmic_reduction();
341    let logarithmic_generator_applies = profile.admits_a_logarithmic_generator();
342    let power_equality_applies = is_two_term_sum(equation);
343    if !exponential_or_logarithmic && !logarithmic_generator_applies && !power_equality_applies {
344        return TranscendentalReductionOutcome::Declined;
345    }
346    let frame = match budget.enter(equation, variable, domain) {
347        Some(frame) => frame,
348        None => return TranscendentalReductionOutcome::Declined,
349    };
350    let mut outcomes: Vec<TranscendentalReductionOutcome> = Vec::with_capacity(4);
351    if exponential_or_logarithmic {
352        record_counter!(transcendental_reducer_entries);
353        outcomes.push(composition::reduce(
354            context, equation, variable, domain, budget,
355        ));
356        record_counter!(transcendental_reducer_entries);
357        outcomes.push(lambert::reduce(context, equation, variable, domain, budget));
358    }
359    if logarithmic_generator_applies {
360        record_counter!(transcendental_reducer_entries);
361        outcomes.push(logarithmic_generator::reduce(
362            context, equation, variable, domain, budget,
363        ));
364    }
365    if power_equality_applies {
366        record_counter!(transcendental_reducer_entries);
367        outcomes.push(power_equality::reduce(
368            context, equation, variable, domain, budget,
369        ));
370    }
371    budget.leave(frame);
372    arbitrate(context, outcomes)
373}

375fn is_two_term_sum(equation: &ExpressionReference) -> bool {
376    // The applicability gate of the same-base power reducer, which recognises only a sum
377    // of exactly two scaled powers. It is a shape question about the root node, so it is
378    // answered without a walk and without the profile: the reducer's premise is a
379    // structural property the profile's family counts do not express — a same-base pair
380    // such as (x + 1)^(1/2) - (x + 1)^(1/3) carries no transcendental call at all.
381    matches!(&equation.kind, ExpressionKind::Sum(children) if children.len() == 2)
382}

384fn arbitrate(
385    context: &Context,
386    outcomes: Vec<TranscendentalReductionOutcome>,
387) -> TranscendentalReductionOutcome {
388    // Combine the reducers' verdicts (§10.3):
389    //   1. a complete set outranks verified roots;
390    //   2. among complete sets, prefer the lower weighted expression cost;
391    //   3. among verified root lists, union every exactly verified root;
392    //   4. a decline never erases another route's sound result;
393    //   5. no route upgrades another route's verified roots into a completeness claim.
394    let mut best_complete: Option<(ExpressionReference, Vec<ExpressionReference>)> = None;
395    let mut verified: Vec<ExpressionReference> = Vec::new();
396    for outcome in outcomes {
397        match outcome {
398            TranscendentalReductionOutcome::Complete {
399                solution_set,
400                representative_roots,
401            } => {
402                let replace = match &best_complete {
403                    None => true,
404                    Some((current, _)) => node_count(&solution_set) < node_count(current),
405                };
406                if replace {
407                    best_complete = Some((solution_set, representative_roots));
408                }
409            }
410            TranscendentalReductionOutcome::VerifiedRoots(roots) => verified.extend(roots),
411            TranscendentalReductionOutcome::Declined => {}
412        }
413    }
414    if let Some((solution_set, representative_roots)) = best_complete {
415        return TranscendentalReductionOutcome::Complete {
416            solution_set,
417            representative_roots,
418        };
419    }
420    let deduplicated = deduplicate(verified);
421    if deduplicated.is_empty() {
422        TranscendentalReductionOutcome::Declined
423    } else {
424        TranscendentalReductionOutcome::VerifiedRoots(normalize(context, deduplicated))
425    }
426}

428pub(crate) fn deduplicate(roots: Vec<ExpressionReference>) -> Vec<ExpressionReference> {
429    // Collapse repeated roots through hash-cons pointer identity, preserving order.
430    let mut seen: Vec<ExpressionReference> = Vec::new();
431    for root in roots {
432        if !seen.iter().any(|existing| Arc::ptr_eq(existing, &root)) {
433            seen.push(root);
434        }
435    }
436    seen
437}

439fn normalize(context: &Context, roots: Vec<ExpressionReference>) -> Vec<ExpressionReference> {
440    // Presentation pass matching the Vec-shaped solver's own: the bounded rational normal
441    // form never grows a root, so identity and ordering semantics are unchanged.
442    roots
443        .into_iter()
444        .map(|root| crate::simplify::rational::presentation_normal_form(context, &root))
445        .collect()
446}

461pub(crate) fn candidate_reduces_to_exact_zero(
462    context: &Context,
463    equation: &ExpressionReference,
464    variable: SymbolIdentifier,
465    candidate: &ExpressionReference,
466) -> bool {
467    // Exact verification of a finite candidate: substitute it into the *original*
468    // equation and require the residual to be canonically zero. A residual that
469    // simplifies to `undefined` or an infinity rejects the candidate — those are the
470    // extraneous roots a naive rearrangement invents — and a residual that stays
471    // symbolic is undecided, which also rejects. No numeric probe participates: this
472    // predicate is the exact acceptance boundary the stricter power route
473    // relies on.
474    let residual = substitute_variable(context, equation, variable, candidate);
475    let simplified = crate::simplify::simplify(context, residual);
476    is_exact_zero(&simplified)
477}

479fn is_exact_zero(expression: &ExpressionReference) -> bool {
480    // Canonical zero, and nothing that merely looks like it.
481    matches!(
482        &expression.kind,
483        ExpressionKind::Integer(value)
484            if *value == crate::exact_arithmetic::Integer::ZERO
485    )
486}

488pub(crate) fn candidate_is_defined(
489    context: &Context,
490    equation: &ExpressionReference,
491    variable: SymbolIdentifier,
492    candidate: &ExpressionReference,
493) -> bool {
494    // Reject a candidate whose substitution makes the original equation undefined or
495    // infinite before any further reasoning treats it as a root.
496    let residual = substitute_variable(context, equation, variable, candidate);
497    let simplified = crate::simplify::simplify(context, residual);
498    !is_undefined_or_infinite(&simplified)
499}

512pub(crate) fn solution_set_is_complete_finite(
513    solution_set: &ExpressionReference,
514) -> Option<Vec<ExpressionReference>> {
515    // A complete solution set that happens to be finite can be flattened into the
516    // Vec-shaped answer without losing anything. Every other complete shape — an image
517    // set, a union of families — cannot, and returns None so the caller falls back to
518    // representatives.
519    match &solution_set.kind {
520        ExpressionKind::FiniteSet(elements) => Some(elements.clone()),
521        ExpressionKind::Constant(BuiltinConstant::EmptySet) => Some(Vec::new()),
522        _ => None,
523    }
524}

548pub(crate) enum ExactSign {
549    Positive,
550    Zero,
551    Negative,
552    Unknown,
553}
554
555pub(crate) fn exact_sign_of(context: &Context, expression: &ExpressionReference) -> ExactSign {
556    // Sign decided by exact means only: a canonical numeric literal, or the assumption
557    // queries, whose product and sum rules are sufficient conditions rather than numeric
558    // estimates. No floating-point comparison may prove a sign condition inside this layer,
559    // so an undecided sign becomes `unknown` and the caller declines.
560    let simplified = crate::simplify::simplify(context, Arc::clone(expression));
561    if is_exact_zero(&simplified) {
562        return ExactSign::Zero;
563    }
564    if crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &simplified)
565        == crate::assumptions::TruthValue::Yes
566    {
567        return ExactSign::Positive;
568    }
569    let negated = builders::negate(context, simplified);
570    if crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &negated)
571        == crate::assumptions::TruthValue::Yes
572    {
573        return ExactSign::Negative;
574    }
575    ExactSign::Unknown
576}

578pub(crate) fn is_real_over_domain(
579    context: &Context,
580    expression: &ExpressionReference,
581    variable: SymbolIdentifier,
582    domain: SolveDomain,
583) -> bool {
584    // Realness of an expression under the domain the solve is being conducted over. Over
585    // the complex plane this is plain realness. Over the reals the caller has already
586    // restricted the solve variable to ℝ, so the variable is replaced by a real-assumed
587    // placeholder before the query and only the *rest* of the expression has to prove
588    // itself real. Without this the real-domain arms would decline every equation in an
589    // unannotated variable, which is exactly the common case.
590    //
591    // The placeholder is one reserved symbol per context rather than a fresh one per call:
592    // it is an internal device that is substituted away immediately, so minting a new
593    // symbol for every realness question would grow the symbol table without bound across a
594    // long session. Its name contains a space, which the parser cannot produce, so
595    // interning it can never capture or shadow a User symbol. Its realness is recorded at
596    // the binding site rather than as a durable assertion: a probe is scaffolding, and a
597    // durable write would move the assumption revision on every realness question, throwing
598    // away the compiled query session the rest of the solve is about to use.
599    if domain == SolveDomain::Complex {
600        return crate::assumptions::range_evidence::proves_real(&context.symbol_table, expression);
601    }
602    let placeholder = context.symbol_table.intern_symbol(REAL_DOMAIN_PROBE_NAME);
603    context.symbol_table.declare_binder_predicates(
604        placeholder,
605        &[crate::assumptions::CurrentScalarPredicate::Real],
606    );
607    let placeholder_expression = context.interner.intern(ExpressionKind::Symbol(placeholder));
608    let probed = substitute_variable(context, expression, variable, &placeholder_expression);
609    let result = crate::assumptions::range_evidence::proves_real(&context.symbol_table, &probed);
610    result
611}

618    #[test]
619    fn a_dispatcher_memo_hit_requires_every_key_component_to_match() {
620        // The memo key is (equation pointer, variable, domain, assumption revision, context
621        // identifier), and every component is there to invalidate a stale entry: a changed
622        // domain or variable is a different question, a changed revision means the
623        // assumption base the cached answer was proved against has moved, and a changed
624        // context identifier means the cached answer was not even computed for this
625        // context. A hit therefore requires an exact match on all five, which this pins
626        // directly against the budget rather than through a full solve.
627        let mut context = Context::new();
628        let equation = integer_value(&mut context, 0);
629        let mut budget = TranscendentalReductionBudget::new(&equation);
630        let variable = SymbolIdentifier(0);
631        let result = integer_value(&mut context, 1);
632        let pointer = Arc::as_ptr(&equation) as usize;
633        let key: DispatcherMemoKey = (pointer, variable, SolveDomain::Real, 0, 0);
634        budget.dispatcher_memo_insert(key, Arc::clone(&result));
635        assert!(budget.dispatcher_memo_lookup(&key).is_some());
636
637        let mismatched_domain: DispatcherMemoKey = (pointer, variable, SolveDomain::Complex, 0, 0);
638        assert!(budget.dispatcher_memo_lookup(&mismatched_domain).is_none());
639
640        let mismatched_variable: DispatcherMemoKey =
641            (pointer, SymbolIdentifier(1), SolveDomain::Real, 0, 0);
642        assert!(
643            budget
644                .dispatcher_memo_lookup(&mismatched_variable)
645                .is_none()
646        );
647
648        let mismatched_revision: DispatcherMemoKey = (pointer, variable, SolveDomain::Real, 1, 0);
649        assert!(
650            budget
651                .dispatcher_memo_lookup(&mismatched_revision)
652                .is_none(),
653            "a changed assumption revision must miss, since the cached answer was proved \
654             against the assumption base the earlier revision denotes"
655        );
656
657        let mismatched_context: DispatcherMemoKey = (pointer, variable, SolveDomain::Real, 0, 1);
658        assert!(
659            budget.dispatcher_memo_lookup(&mismatched_context).is_none(),
660            "a changed context identifier must miss, since the cached answer was not computed \
661             for this context"
662        );
663    }
