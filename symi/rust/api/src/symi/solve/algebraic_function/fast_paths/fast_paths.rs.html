21pub(super) fn solve_with_fast_path(
22    context: &Context,
23    equation: &ExpressionReference,
24    variable: SymbolIdentifier,
25) -> Option<ExpressionReference> {
26    super::absolute_value::solve_absolute_value_by_sign_partition(context, equation, variable)
27        .or_else(|| solve_isolated_radical(context, equation, variable))
28        .or_else(|| solve_common_power(context, equation, variable))

31fn solve_common_power(
32    context: &Context,
33    equation: &ExpressionReference,
34    variable: SymbolIdentifier,
35) -> Option<ExpressionReference> {
36    let common_index = collect_common_power_index(equation, variable, 1)?;
37    if common_index < 2 || common_index > 32 {
38        return None;
39    }
40    let generator_variable = context.symbol_table.fresh("root_generator");
41    let transformed = transform_to_common_power(
42        context,
43        equation,
44        variable,
45        generator_variable,
46        common_index,
47    )?;
48    expression_to_univariate(&transformed, generator_variable)?;
49    let zero = builders::zero(context);
50    let equality = builders::equal(context, transformed, Arc::clone(&zero));
51    let generator = context
52        .interner
53        .intern(ExpressionKind::Symbol(generator_variable));
54    let nonnegative = builders::greater_than_or_equal(context, Arc::clone(&generator), zero);
55    let formula = builders::function_call(
56        context,
57        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
58        vec![equality, nonnegative],
59    );
60    let generator_set = materialize_univariate_set(context, &formula, generator_variable)?;
61    match &generator_set.kind {
62        ExpressionKind::Constant(crate::expression::BuiltinConstant::EmptySet) => {
63            Some(generator_set)
64        }
65        ExpressionKind::FiniteSet(roots) => {
66            let exponent = builders::integer_value(context, common_index as i64);
67            let mapped = roots
68                .iter()
69                .map(|root| builders::power(context, Arc::clone(root), Arc::clone(&exponent)))
70                .collect();
71            Some(finite_set(context, mapped))
72        }
73        _ => None,
74    }
75}

77fn collect_common_power_index(
78    target: &ExpressionReference,
79    variable: SymbolIdentifier,
80    current: u64,
81) -> Option<u64> {
82    match &target.kind {
83        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) => Some(current),
84        ExpressionKind::Symbol(found) if *found == variable => Some(current),
85        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
86            children.iter().try_fold(current, |index, child| {
87                collect_common_power_index(child, variable, index)
88            })
89        }
90        ExpressionKind::Power(base, exponent) => match &exponent.kind {
91            ExpressionKind::Integer(value) => {
92                let exponent_value = i64::try_from(value).ok()?;
93                if exponent_value < 0 {
94                    return None;
95                }
96                collect_common_power_index(base, variable, current)
97            }
98            ExpressionKind::Rational(value) if matches!(&base.kind, ExpressionKind::Symbol(found) if *found == variable) =>
99            {
100                let parts = rational_into_parts(value.clone());
101                let numerator = i64::try_from(parts.numerator).ok()?;
102                let denominator = u64::try_from(parts.denominator).ok()?;
103                if numerator < 0 {
104                    return None;
105                }
106                checked_least_common_multiple(current, denominator)
107            }
108            _ => None,
109        },
110        ExpressionKind::FunctionCall(
111            FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
112            arguments,
113        ) if arguments.len() == 1
114            && matches!(&arguments[0].kind, ExpressionKind::Symbol(found) if *found == variable) =>
115        {
116            checked_least_common_multiple(current, 2)
117        }
118        _ => None,
119    }
120}

122fn transform_to_common_power(
123    context: &Context,
124    target: &ExpressionReference,
125    variable: SymbolIdentifier,
126    generator_variable: SymbolIdentifier,
127    common_index: u64,
128) -> Option<ExpressionReference> {
129    match &target.kind {
130        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) => Some(Arc::clone(target)),
131        ExpressionKind::Symbol(found) if *found == variable => {
132            let generator = context
133                .interner
134                .intern(ExpressionKind::Symbol(generator_variable));
135            let exponent = builders::integer_value(context, common_index as i64);
136            Some(builders::power(context, generator, exponent))
137        }
138        ExpressionKind::Sum(children) => {
139            let transformed = children
140                .iter()
141                .map(|child| {
142                    transform_to_common_power(
143                        context,
144                        child,
145                        variable,
146                        generator_variable,
147                        common_index,
148                    )
149                })
150                .collect::<Option<Vec<_>>>()?;
151            Some(builders::sum(context, transformed))
152        }
153        ExpressionKind::Product(children) => {
154            let transformed = children
155                .iter()
156                .map(|child| {
157                    transform_to_common_power(
158                        context,
159                        child,
160                        variable,
161                        generator_variable,
162                        common_index,
163                    )
164                })
165                .collect::<Option<Vec<_>>>()?;
166            Some(builders::product(context, transformed))
167        }
168        ExpressionKind::Power(base, exponent) => match &exponent.kind {
169            ExpressionKind::Integer(value) => {
170                let exponent_value = i64::try_from(value).ok()?;
171                if exponent_value < 0 {
172                    return None;
173                }
174                let transformed_base = transform_to_common_power(
175                    context,
176                    base,
177                    variable,
178                    generator_variable,
179                    common_index,
180                )?;
181                Some(builders::power(
182                    context,
183                    transformed_base,
184                    Arc::clone(exponent),
185                ))
186            }
187            ExpressionKind::Rational(value) if matches!(&base.kind, ExpressionKind::Symbol(found) if *found == variable) =>
188            {
189                let parts = rational_into_parts(value.clone());
190                let numerator = i64::try_from(parts.numerator).ok()?;
191                let denominator = u64::try_from(parts.denominator).ok()?;
192                if numerator < 0 || common_index % denominator != 0 {
193                    return None;
194                }
195                let scaled = numerator.checked_mul((common_index / denominator) as i64)?;
196                let generator = context
197                    .interner
198                    .intern(ExpressionKind::Symbol(generator_variable));
199                let exponent = builders::integer_value(context, scaled);
200                Some(builders::power(context, generator, exponent))
201            }
202            _ => None,
203        },
204        ExpressionKind::FunctionCall(
205            FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
206            arguments,
207        ) if arguments.len() == 1
208            && matches!(&arguments[0].kind, ExpressionKind::Symbol(found) if *found == variable) =>
209        {
210            let generator = context
211                .interner
212                .intern(ExpressionKind::Symbol(generator_variable));
213            let exponent = builders::integer_value(context, (common_index / 2) as i64);
214            Some(builders::power(context, generator, exponent))
215        }
216        _ => None,
217    }
218}

220fn checked_least_common_multiple(left: u64, right: u64) -> Option<u64> {
221    let mut first = left;
222    let mut second = right;
223    while second != 0 {
224        let remainder = first % second;
225        first = second;
226        second = remainder;
227    }
228    left.checked_div(first)?.checked_mul(right)
229}

231fn solve_isolated_radical(
232    context: &Context,
233    equation: &ExpressionReference,
234    variable: SymbolIdentifier,
235) -> Option<ExpressionReference> {
236    let terms = match &equation.kind {
237        ExpressionKind::Sum(terms) => terms.clone(),
238        _ => vec![Arc::clone(equation)],
239    };
240    let mut radical = None;
241    let mut remaining = Vec::new();
242    for term in terms {
243        if let Some(found) = isolate_radical_factor(context, &term, variable) {
244            if radical.is_some() {
245                return None;
246            }
247            radical = Some(found);
248        } else {
249            if super::contains_real_algebraic_function_node(&term) {
250                return None;
251            }
252            remaining.push(term);
253        }
254    }
255    let radical = radical?;
256    if radical.numerator.unsigned_abs() != 1 || radical.denominator < 2 {
257        return None;
258    }
259    expression_to_univariate(&radical.base, variable)?;
260    let remainder = builders::sum(context, remaining);
261    expression_to_univariate(&remainder, variable)?;
262    let negative_remainder = builders::negate(context, remainder);
263    let right = builders::divide(context, negative_remainder, radical.coefficient);
264    expression_to_univariate(&right, variable)?;
265    let denominator = builders::integer_value(context, radical.denominator as i64);
266    let right_power = builders::power(context, Arc::clone(&right), denominator);
267    let zero = builders::zero(context);
268    let base_nonnegative =
269        builders::greater_than_or_equal(context, Arc::clone(&radical.base), Arc::clone(&zero));
270    let (equation_constraint, range_constraint) = if radical.numerator > 0 {
271        let equality_polynomial = builders::subtract(context, radical.base, right_power);
272        let equation_constraint = builders::equal(context, equality_polynomial, Arc::clone(&zero));
273        let range_constraint = builders::greater_than_or_equal(context, right, zero);
274        (equation_constraint, range_constraint)
275    } else {
276        let product = builders::product(context, vec![radical.base, right_power]);
277        let one = builders::integer_value(context, 1);
278        let equality_polynomial = builders::subtract(context, product, one);
279        let equation_constraint = builders::equal(context, equality_polynomial, Arc::clone(&zero));
280        let range_constraint = builders::greater_than(context, right, zero);
281        (equation_constraint, range_constraint)
282    };
283    let formula = builders::function_call(
284        context,
285        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
286        vec![equation_constraint, range_constraint, base_nonnegative],
287    );
288    materialize_univariate_set(context, &formula, variable)
289}

291fn isolate_radical_factor(
292    context: &Context,
293    term: &ExpressionReference,
294    variable: SymbolIdentifier,
295) -> Option<IsolatedRadical> {
296    if let Some((base, numerator, denominator)) = rational_power_parts(term) {
297        let coefficient = builders::integer_value(context, 1);
298        return Some(IsolatedRadical {
299            coefficient,
300            base,
301            numerator,
302            denominator,
303        });
304    }
305    let factors = match &term.kind {
306        ExpressionKind::Product(factors) => factors,
307        _ => return None,
308    };
309    let mut radical_parts = None;
310    let mut coefficient_factors = Vec::new();
311    for factor in factors {
312        if let Some(parts) = rational_power_parts(factor) {
313            if radical_parts.is_some() {
314                return None;
315            }
316            radical_parts = Some(parts);
317        } else {
318            if contains_variable(factor, variable) {
319                return None;
320            }
321            coefficient_factors.push(Arc::clone(factor));
322        }
323    }
324    let (base, numerator, denominator) = radical_parts?;
325    let coefficient = builders::product(context, coefficient_factors);
326    Some(IsolatedRadical {
327        coefficient,
328        base,
329        numerator,
330        denominator,
331    })
332}

334fn rational_power_parts(target: &ExpressionReference) -> Option<(ExpressionReference, i64, u64)> {
335    match &target.kind {
336        ExpressionKind::Power(base, exponent) => {
337            let value = match &exponent.kind {
338                ExpressionKind::Rational(value) => value,
339                _ => return None,
340            };
341            let parts = rational_into_parts(value.clone());
342            let numerator = i64::try_from(parts.numerator).ok()?;
343            let denominator = u64::try_from(parts.denominator).ok()?;
344            Some((Arc::clone(base), numerator, denominator))
345        }
346        ExpressionKind::FunctionCall(
347            FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
348            arguments,
349        ) if arguments.len() == 1 => Some((Arc::clone(&arguments[0]), 1, 2)),
350        _ => None,
351    }
