62pub(crate) fn lower_real_algebraic_equation(
63    context: &Context,
64    equation: &ExpressionReference,
65    solve_variable: SymbolIdentifier,
66) -> Option<LoweredRealAlgebraicEquation> {
67    let planned_root_indices = collect_root_indices(equation)?;
68    let mut state = LoweringState {
69        context,
70        solve_variable,
71        memoized_nodes: HashMap::new(),
72        algebraic_constants: HashMap::new(),
73        planned_root_indices,
74        radical_generators: HashMap::new(),
75        constraints: Vec::new(),
76        equational_constraints: Vec::new(),
77        auxiliary_variables: Vec::new(),
78        guarded_denominators: HashSet::new(),
79        required_denominators: HashSet::new(),
80        guarded_nonnegative_values: HashSet::new(),
81        required_nonnegative_values: HashSet::new(),
82    };
83    let lowered = state.lower_node(equation)?;
84    state.require_nonzero_denominator(&lowered.denominator);
85    let zero = builders::zero(state.context);
86    let equation_constraint = builders::equal(state.context, Arc::clone(&lowered.numerator), zero);
87    state
88        .equational_constraints
89        .push(Arc::clone(&equation_constraint));
90    state.constraints.push(equation_constraint);
91    let matrix = conjunction(state.context, state.constraints.clone());
92    let mut variables = vec![solve_variable];
93    variables.extend(state.auxiliary_variables.iter().copied());
94    let normalised = formula_from_expression(state.context, &matrix, Vec::new(), &variables)?;
95    let work_estimate = estimate_work(&normalised.matrix, &variables)?;
96    let ledger_is_complete = state.required_denominators == state.guarded_denominators
97        && state.required_nonnegative_values == state.guarded_nonnegative_values;
98    if !ledger_is_complete || !work_is_admitted(work_estimate) {
99        return None;
100    }
101    Some(LoweredRealAlgebraicEquation {
102        matrix: normalised.matrix,
103        variables,
104        number_of_free_variables: 1,
105        auxiliary_variables: state.auxiliary_variables,
106        equational_constraints: state.equational_constraints,
107        work_estimate,
108        ledger_is_complete,
109    })
110}

112impl LoweringState<'_> {
113    fn lower_node(&mut self, target: &ExpressionReference) -> Option<LoweredRationalFunction> {
114        if let Some(lowered) = self.memoized_nodes.get(target) {
115            return Some(lowered.clone());
116        }
117        let lowered = match &target.kind {
118            ExpressionKind::Integer(_) | ExpressionKind::Rational(_) => {
119                self.from_polynomial(Arc::clone(target))
120            }
121            ExpressionKind::Symbol(variable) if *variable == self.solve_variable => {
122                self.from_polynomial(Arc::clone(target))
123            }
124            ExpressionKind::PolynomialRoot(number) if number.is_real() => {
125                self.lower_algebraic_constant(target, number)?
126            }
127            ExpressionKind::Sum(children) => self.lower_sum(children)?,
128            ExpressionKind::Product(children) => self.lower_product(children)?,
129            ExpressionKind::Power(base, exponent) => self.lower_power(base, exponent)?,
130            ExpressionKind::FunctionCall(
131                FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
132                arguments,
133            ) if arguments.len() == 1 => {
134                let exponent = builders::rational_value(self.context, 1, 2);
135                self.lower_power(&arguments[0], &exponent)?
136            }
137            ExpressionKind::FunctionCall(
138                FunctionIdentifier::Builtin(BuiltinFunction::Abs),
139                arguments,
140            ) if arguments.len() == 1 => self.lower_absolute_value(&arguments[0])?,
141            _ => return None,
142        };
143        self.memoized_nodes
144            .insert(Arc::clone(target), lowered.clone());
145        Some(lowered)
146    }
147
148    fn from_polynomial(&mut self, value: ExpressionReference) -> LoweredRationalFunction {
149        LoweredRationalFunction {
150            numerator: value,
151            denominator: builders::integer_value(self.context, 1),
152        }
153    }
154
155    fn lower_sum(&mut self, children: &[ExpressionReference]) -> Option<LoweredRationalFunction> {
156        let zero = builders::zero(self.context);
157        let mut result = self.from_polynomial(zero);
158        for child in children {
159            let lowered = self.lower_node(child)?;
160            let left = builders::product(
161                self.context,
162                vec![result.numerator, Arc::clone(&lowered.denominator)],
163            );
164            let right = builders::product(
165                self.context,
166                vec![lowered.numerator, Arc::clone(&result.denominator)],
167            );
168            let numerator = builders::sum(self.context, vec![left, right]);
169            let denominator =
170                builders::product(self.context, vec![result.denominator, lowered.denominator]);
171            result = LoweredRationalFunction {
172                numerator,
173                denominator,
174            };
175            self.require_nonzero_denominator(&result.denominator);
176        }
177        Some(result)
178    }
179
180    fn lower_product(
181        &mut self,
182        children: &[ExpressionReference],
183    ) -> Option<LoweredRationalFunction> {
184        let one = builders::integer_value(self.context, 1);
185        let mut result = self.from_polynomial(one);
186        for child in children {
187            let lowered = self.lower_node(child)?;
188            let numerator =
189                builders::product(self.context, vec![result.numerator, lowered.numerator]);
190            let denominator =
191                builders::product(self.context, vec![result.denominator, lowered.denominator]);
192            result = LoweredRationalFunction {
193                numerator,
194                denominator,
195            };
196            self.require_nonzero_denominator(&result.denominator);
197        }
198        Some(result)
199    }
200
201    fn lower_power(
202        &mut self,
203        base: &ExpressionReference,
204        exponent: &ExpressionReference,
205    ) -> Option<LoweredRationalFunction> {
206        match &exponent.kind {
207            ExpressionKind::Integer(value) => {
208                let exponent_value = i64::try_from(value).ok()?;
209                self.lower_integer_power(base, exponent_value)
210            }
211            ExpressionKind::Rational(value) => {
212                let parts = rational_into_parts(value.clone());
213                let numerator = i64::try_from(parts.numerator).ok()?;
214                let denominator = u64::try_from(parts.denominator).ok()?;
215                self.lower_rational_power(base, numerator, denominator)
216            }
217            _ => None,
218        }
219    }
220
221    fn lower_integer_power(
222        &mut self,
223        base: &ExpressionReference,
224        exponent: i64,
225    ) -> Option<LoweredRationalFunction> {
226        if exponent.unsigned_abs() > MAXIMUM_POWER_MAGNITUDE as u64 {
227            return None;
228        }
229        let lowered = self.lower_node(base)?;
230        if exponent >= 0 {
231            let exponent_expression = builders::integer_value(self.context, exponent);
232            let numerator = builders::power(
233                self.context,
234                lowered.numerator,
235                Arc::clone(&exponent_expression),
236            );
237            let denominator =
238                builders::power(self.context, lowered.denominator, exponent_expression);
239            self.require_nonzero_denominator(&denominator);
240            return Some(LoweredRationalFunction {
241                numerator,
242                denominator,
243            });
244        }
245        self.require_nonzero_denominator(&lowered.numerator);
246        let magnitude = builders::integer_value(self.context, -exponent);
247        let numerator = builders::power(self.context, lowered.denominator, Arc::clone(&magnitude));
248        let denominator = builders::power(self.context, lowered.numerator, magnitude);
249        self.require_nonzero_denominator(&denominator);
250        Some(LoweredRationalFunction {
251            numerator,
252            denominator,
253        })
254    }
255
256    fn lower_rational_power(
257        &mut self,
258        base: &ExpressionReference,
259        numerator_exponent: i64,
260        denominator_exponent: u64,
261    ) -> Option<LoweredRationalFunction> {
262        if denominator_exponent < 2
263            || numerator_exponent.unsigned_abs() > MAXIMUM_POWER_MAGNITUDE as u64
264            || denominator_exponent > MAXIMUM_POWER_MAGNITUDE as u64
265        {
266            return None;
267        }
268        let generator_index = *self
269            .planned_root_indices
270            .get(base)
271            .unwrap_or(&denominator_exponent);
272        if generator_index > MAXIMUM_POWER_MAGNITUDE as u64 {
273            return None;
274        }
275        let root = if let Some((root, _)) = self.radical_generators.get(base) {
276            Arc::clone(root)
277        } else {
278            let lowered_base = self.lower_node(base)?;
279            self.require_nonzero_denominator(&lowered_base.denominator);
280            let base_sign_polynomial = builders::product(
281                self.context,
282                vec![
283                    Arc::clone(&lowered_base.numerator),
284                    Arc::clone(&lowered_base.denominator),
285                ],
286            );
287            self.require_nonnegative(base_sign_polynomial);
288            let root = self.fresh_auxiliary("radical");
289            self.require_nonnegative(Arc::clone(&root));
290            let root_index = builders::integer_value(self.context, generator_index as i64);
291            let root_power = builders::power(self.context, Arc::clone(&root), root_index);
292            let defining_left = builders::product(
293                self.context,
294                vec![root_power, Arc::clone(&lowered_base.denominator)],
295            );
296            let defining = builders::subtract(
297                self.context,
298                defining_left,
299                Arc::clone(&lowered_base.numerator),
300            );
301            self.require_equality(defining);
302            self.radical_generators
303                .insert(Arc::clone(base), (Arc::clone(&root), generator_index));
304            root
305        };
306        let scale = generator_index.checked_div(denominator_exponent)?;
307        if scale.checked_mul(denominator_exponent)? != generator_index {
308            return None;
309        }
310        let magnitude = numerator_exponent.unsigned_abs().checked_mul(scale)? as i64;
311        let magnitude_expression = builders::integer_value(self.context, magnitude);
312        let powered_root = builders::power(self.context, Arc::clone(&root), magnitude_expression);
313        if numerator_exponent >= 0 {
314            return Some(self.from_polynomial(powered_root));
315        }
316        self.require_nonzero_denominator(&root);
317        Some(LoweredRationalFunction {
318            numerator: builders::integer_value(self.context, 1),
319            denominator: powered_root,
320        })
321    }
322
323    fn lower_absolute_value(
324        &mut self,
325        argument: &ExpressionReference,
326    ) -> Option<LoweredRationalFunction> {
327        let lowered = self.lower_node(argument)?;
328        self.require_nonzero_denominator(&lowered.denominator);
329        let absolute = self.fresh_auxiliary("absolute_value");
330        self.require_nonnegative(Arc::clone(&absolute));
331        let two = builders::integer_value(self.context, 2);
332        let absolute_squared =
333            builders::power(self.context, Arc::clone(&absolute), Arc::clone(&two));
334        let denominator_squared = builders::power(
335            self.context,
336            Arc::clone(&lowered.denominator),
337            Arc::clone(&two),
338        );
339        let numerator_squared = builders::power(self.context, lowered.numerator, two);
340        let left = builders::product(self.context, vec![absolute_squared, denominator_squared]);
341        let defining = builders::subtract(self.context, left, numerator_squared);
342        self.require_equality(defining);
343        Some(self.from_polynomial(absolute))
344    }

346    fn lower_algebraic_constant(
347        &mut self,
348        original: &ExpressionReference,
349        number: &Arc<crate::algebraic_numbers::absolute_number::AlgebraicNumber>,
350    ) -> Option<LoweredRationalFunction> {
351        if let Some(existing) = self.algebraic_constants.get(original) {
352            return Some(existing.clone());
353        }
354        let auxiliary = self.fresh_auxiliary("algebraic_constant");
355        let mut terms = Vec::new();
356        for (exponent, coefficient) in number.minimal_polynomial.coefficients.iter().enumerate() {
357            if crate::exact_arithmetic::integer_is_zero(coefficient) {
358                continue;
359            }
360            let coefficient_expression = builders::integer(self.context, coefficient.clone());
361            let term = if exponent == 0 {
362                coefficient_expression
363            } else {
364                let exponent_expression = builders::integer_value(self.context, exponent as i64);
365                let power =
366                    builders::power(self.context, Arc::clone(&auxiliary), exponent_expression);
367                builders::product(self.context, vec![coefficient_expression, power])
368            };
369            terms.push(term);
370        }
371        let defining = builders::sum(self.context, terms);
372        self.require_equality(defining);
373        let interval = match &number.isolating_region {
374            AlgebraicIsolatingRegion::Real(interval) => interval,
375            AlgebraicIsolatingRegion::NonReal(_) => return None,
376        };
377        if interval.is_exact() {
378            let value = builders::rational_literal(self.context, interval.lower.clone());
379            let bound = builders::equal(self.context, Arc::clone(&auxiliary), value);
380            self.require_relation(bound);
381        } else {
382            let lower = builders::rational_literal(self.context, interval.lower.clone());
383            let upper = builders::rational_literal(self.context, interval.upper.clone());
384            let lower_constraint = builders::less_than(self.context, lower, Arc::clone(&auxiliary));
385            self.require_relation(lower_constraint);
386            let upper_constraint = builders::less_than(self.context, Arc::clone(&auxiliary), upper);
387            self.require_relation(upper_constraint);
388        }
389        let lowered = self.from_polynomial(auxiliary);
390        self.algebraic_constants
391            .insert(Arc::clone(original), lowered.clone());
392        Some(lowered)
393    }

395    fn fresh_auxiliary(&mut self, prefix: &str) -> ExpressionReference {
396        let variable = self.context.symbol_table.fresh(prefix);
397        self.auxiliary_variables.push(variable);
398        self.context
399            .interner
400            .intern(ExpressionKind::Symbol(variable))
401    }
402
403    fn require_nonzero_denominator(&mut self, denominator: &ExpressionReference) {
404        if is_one(denominator) {
405            return;
406        }
407        self.required_denominators.insert(Arc::clone(denominator));
408        if self.guarded_denominators.insert(Arc::clone(denominator)) {
409            let zero = builders::zero(self.context);
410            let guard = builders::not_equal(self.context, Arc::clone(denominator), zero);
411            self.constraints.push(guard);
412        }
413    }
414
415    fn require_nonnegative(&mut self, value: ExpressionReference) {
416        self.required_nonnegative_values.insert(Arc::clone(&value));
417        if self.guarded_nonnegative_values.insert(Arc::clone(&value)) {
418            let zero = builders::zero(self.context);
419            let guard = builders::greater_than_or_equal(self.context, value, zero);
420            self.constraints.push(guard);
421        }
422    }
423
424    fn require_equality(&mut self, polynomial: ExpressionReference) {
425        let zero = builders::zero(self.context);
426        let equality = builders::equal(self.context, polynomial, zero);
427        self.equational_constraints.push(Arc::clone(&equality));
428        self.constraints.push(equality);
429    }
430
431    fn require_relation(&mut self, relation: ExpressionReference) {
432        self.constraints.push(relation);
433    }
434}

436fn collect_root_indices(target: &ExpressionReference) -> Option<HashMap<ExpressionReference, u64>> {
437    let mut indices = HashMap::new();
438    let mut pending = vec![target];
439    while let Some(current) = pending.pop() {
440        match &current.kind {
441            ExpressionKind::Power(base, exponent) => {
442                if let ExpressionKind::Rational(value) = &exponent.kind {
443                    let parts = rational_into_parts(value.clone());
444                    let denominator = u64::try_from(parts.denominator).ok()?;
445                    let previous = indices.get(base).copied().unwrap_or(1);
446                    indices.insert(
447                        Arc::clone(base),
448                        checked_least_common_multiple(previous, denominator)?,
449                    );
450                }
451                pending.push(base);
452                pending.push(exponent);
453            }
454            ExpressionKind::FunctionCall(
455                FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
456                arguments,
457            ) if arguments.len() == 1 => {
458                let base = &arguments[0];
459                let previous = indices.get(base).copied().unwrap_or(1);
460                indices.insert(
461                    Arc::clone(base),
462                    checked_least_common_multiple(previous, 2)?,
463                );
464                pending.push(base);
465            }
466            ExpressionKind::FunctionCall(_, arguments)
467            | ExpressionKind::Sum(arguments)
468            | ExpressionKind::Product(arguments) => pending.extend(arguments.iter()),
469            _ => {}
470        }
471    }
472    Some(indices)
473}

475fn checked_least_common_multiple(left: u64, right: u64) -> Option<u64> {
476    let mut first = left;
477    let mut second = right;
478    while second != 0 {
479        let remainder = first % second;
480        first = second;
481        second = remainder;
482    }
483    left.checked_div(first)?.checked_mul(right)
484}

486fn conjunction(context: &Context, constraints: Vec<ExpressionReference>) -> ExpressionReference {
487    if constraints.is_empty() {
488        return builders::constant(context, crate::expression::BuiltinConstant::LogicalTrue);
489    }
490    if constraints.len() == 1 {
491        return constraints.into_iter().next().unwrap();
492    }
493    builders::function_call(
494        context,
495        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
496        constraints,
497    )
498}

500fn estimate_work(
501    matrix: &ExpressionReference,
502    variables: &[SymbolIdentifier],
503) -> Option<AlgebraicEquationWorkEstimate> {
504    let atoms = atom_polynomials(matrix);
505    let mut maximum_degree = 0;
506    let mut monomial_count: usize = 0;
507    for atom in atoms.iter() {
508        let polynomial_value = expression_to_polynomial(atom, variables)?;
509        let (degree, monomials) = polynomial_statistics(&polynomial_value);
510        maximum_degree = maximum_degree.max(degree);
511        monomial_count = monomial_count.checked_add(monomials)?;
512    }
513    let number_of_auxiliary_variables = variables.len().saturating_sub(1);
514    let projection_growth = 4usize.checked_pow(number_of_auxiliary_variables as u32)?;
515    monomial_count = monomial_count.checked_mul(projection_growth)?;
516    Some(AlgebraicEquationWorkEstimate {
517        number_of_auxiliary_variables,
518        maximum_polynomial_degree: maximum_degree,
519        number_of_relation_atoms: atoms.len(),
520        estimated_monomial_count: monomial_count,
521    })
522}
523
524fn polynomial_statistics(value: &Polynomial) -> (usize, usize) {
525    let mut maximum_degree = 0;
526    let mut monomial_count = 0;
527    for (exponent, coefficient) in value.terms.iter() {
528        match coefficient {
529            PolynomialCoefficient::Nested(nested) => {
530                let (nested_degree, nested_monomials) = polynomial_statistics(nested);
531                maximum_degree = maximum_degree.max(exponent.saturating_add(nested_degree));
532                monomial_count += nested_monomials;
533            }
534            _ => {
535                maximum_degree = maximum_degree.max(*exponent);
536                monomial_count += 1;
537            }
538        }
539    }
540    (maximum_degree, monomial_count)
541}
542
543fn work_is_admitted(estimate: AlgebraicEquationWorkEstimate) -> bool {
544    estimate.number_of_auxiliary_variables <= MAXIMUM_AUXILIARY_VARIABLES
545        && estimate.number_of_relation_atoms <= MAXIMUM_RELATION_ATOMS
546        && estimate.maximum_polynomial_degree <= MAXIMUM_POLYNOMIAL_DEGREE
547        && estimate.estimated_monomial_count <= MAXIMUM_ESTIMATED_MONOMIALS
548}

550fn is_one(value: &ExpressionReference) -> bool {
551    matches!(&value.kind, ExpressionKind::Integer(integer) if *integer == exact_integer::from(1))
552}
