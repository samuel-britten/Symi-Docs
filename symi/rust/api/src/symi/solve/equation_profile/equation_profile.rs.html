1// The structural summary of one equation, derived by a single walk of its tree.
2//
3// Every arm of the equation-set dispatcher needs the same handful of structural facts:
4// how often the variable occurs, how many transcendental applications wrap it, whether
5// there is an absolute value, a rational power, a negative rational power. Deriving each
6// of them with its own recursive predicate meant one full traversal per fact per arm per
7// recursion level, all of them re-deriving what the previous arm had just computed.
8//
9// The profile is computed once for an `(equation, variable)` pair and carried by
10// reference through the arms, so a fact costs one walk for the whole descent rather than
11// one walk per question.
12//
13// A profile is a summary used to decide whether an arm *can* apply, never to decide what
14// the answer is: every arm that runs still proves its own premises. The counts are
15// therefore deliberately over-inclusive — the walk descends into every child of every
16// node kind, including binder bodies whose bound variable shadows the solve variable — so
17// a gate can only ever admit a route that would have declined, never skip one that would
18// have answered.
19
20use crate::expression::{ExpressionKind, ExpressionReference};
21use crate::function::{BuiltinFunction, FunctionIdentifier};
22use crate::instrumentation::record_counter;
23use crate::symbol::SymbolIdentifier;
24
25pub(crate) struct EquationProfile {
26    // The variable the equation was profiled for, so a profile can be checked against
27    // the variable an arm is solving for rather than silently answering about another.
28    variable: SymbolIdentifier,
29    /// Whether the solve variable occurs in the equation, under the binder-aware rule the
30    /// solvers use. This one field is authoritative rather than over-inclusive.
31    contains_variable: bool,
32    /// Leaf occurrences of the solve variable.
33    pub(crate) variable_occurrences: usize,
34    /// Nodes visited by the walk.
35    pub(crate) nodes: usize,
36    /// Transcendental applications and variable-exponent powers around the variable.
37    /// These are the obstructions the shared reduction layer exists to remove.
38    pub(crate) transcendental_calls: usize,
39    /// Circular, hyperbolic and inverse-circular applications around the variable.
40    pub(crate) trigonometric_calls: usize,
41    /// Exponential forms around the variable: `Exp(u)` with a variable-bearing `u`, and
42    /// any power whose exponent carries the variable.
43    pub(crate) exponential_forms: usize,
44    /// Principal-logarithm applications around the variable.
45    pub(crate) logarithm_calls: usize,
46    /// A power whose exponent is a rational literal occurs anywhere in the tree.
47    pub(crate) contains_rational_power: bool,
48    /// A power whose exponent is a negative rational literal occurs anywhere.
49    pub(crate) contains_negative_rational_power: bool,
50    /// An absolute-value or square-root application occurs anywhere in the tree.
51    pub(crate) contains_absolute_value_or_square_root: bool,
52}
53
54impl EquationProfile {
55    pub(crate) fn of(equation: &ExpressionReference, variable: SymbolIdentifier) -> Self {
56        // The single walk. One traversal is recorded for the whole profile, which is the
57        // property the cost gate asserts: an arm asks the profile, it does not re-walk.
58        record_counter!(equation_profile_builds);
59        record_counter!(expression_traversals);
60        let mut profile = EquationProfile {
61            variable,
62            contains_variable: crate::solve::vec::contains_variable(equation, variable),
63            variable_occurrences: 0,
64            nodes: 0,
65            transcendental_calls: 0,
66            trigonometric_calls: 0,
67            exponential_forms: 0,
68            logarithm_calls: 0,
69            contains_rational_power: false,
70            contains_negative_rational_power: false,
71            contains_absolute_value_or_square_root: false,
72        };
73        profile.walk(equation);
74        profile
75    }
76
77    /// True when the solve variable occurs anywhere in the equation.
78    pub(crate) fn contains_variable(&self) -> bool {
79        self.contains_variable
80    }
81
82    /// True when the equation carries a shape the real algebraic route recognises: an
83    /// absolute value, a square root, or any rational power.
84    pub(crate) fn contains_real_algebraic_function_node(&self) -> bool {
85        self.contains_absolute_value_or_square_root || self.contains_rational_power
86    }
87
88    /// True when the equal-composition and Lambert reducers could apply: both rearrange
89    /// the equation around an exponential or a principal logarithm, and neither can
90    /// recognise anything built from any other transcendental family.
91    pub(crate) fn admits_an_exponential_or_logarithmic_reduction(&self) -> bool {
92        self.exponential_forms > 0 || self.logarithm_calls > 0
93    }
94
95    /// True when the logarithmic-generator reducer could apply. Its generator is a
96    /// logarithm of a variable-bearing argument, and its `log(z^a)` normalisation only
97    /// rewrites logarithms that are already present, so no logarithm means no generator.
98    pub(crate) fn admits_a_logarithmic_generator(&self) -> bool {
99        self.logarithm_calls > 0
100    }
101
102    fn walk(&mut self, expression: &ExpressionReference) -> bool {
103        // Returns whether this subtree carries the solve variable, so a parent classes
104        // its own node without a second descent. Every count above is accumulated on the
105        // way back up from exactly one visit per node.
106        self.nodes += 1;
107        let mut carries_variable = false;
108        match &expression.kind {
109            ExpressionKind::Symbol(identifier) => {
110                if *identifier == self.variable {
111                    self.variable_occurrences += 1;
112                    carries_variable = true;
113                }
114            }
115            ExpressionKind::Integer(_)
116            | ExpressionKind::Rational(_)
117            | ExpressionKind::PolynomialRoot(_)
118            | ExpressionKind::Constant(_) => {}
119            ExpressionKind::Power(base, exponent) => {
120                if let ExpressionKind::Rational(value) = &exponent.kind {
121                    self.contains_rational_power = true;
122                    if crate::exact_arithmetic::rational_is_negative(value) {
123                        self.contains_negative_rational_power = true;
124                    }
125                }
126                let base_carries = self.walk(base);
127                let exponent_carries = self.walk(exponent);
128                if exponent_carries {
129                    self.transcendental_calls += 1;
130                    self.exponential_forms += 1;
131                }
132                carries_variable = base_carries || exponent_carries;
133            }
134            ExpressionKind::FunctionCall(function, arguments)
135            | ExpressionKind::Derivative(function, _, arguments) => {
136                let function = *function;
137                for argument in arguments {
138                    carries_variable |= self.walk(argument);
139                }
140                if let FunctionIdentifier::Builtin(builtin) = function {
141                    if matches!(builtin, BuiltinFunction::Abs | BuiltinFunction::Sqrt) {
142                        self.contains_absolute_value_or_square_root = true;
143                    }
144                    if carries_variable {
145                        self.classify_variable_bearing_call(builtin);
146                    }
147                }
148            }
149            _ => {
150                for child in child_expressions(expression) {
151                    carries_variable |= self.walk(&child);
152                }
153            }
154        }
155        carries_variable
156    }
157
158    fn classify_variable_bearing_call(&mut self, builtin: BuiltinFunction) {
159        // Class one builtin application whose argument carries the variable. The family
160        // counts overlap with `transcendental_calls` deliberately: that count answers
161        // "is there anything here for the reduction stage at all", while the family
162        // counts are the applicability gates of the individual reducers.
163        use BuiltinFunction::*;
164        match builtin {
165            Exp => {
166                self.transcendental_calls += 1;
167                self.exponential_forms += 1;
168            }
169            Log => {
170                self.transcendental_calls += 1;
171                self.logarithm_calls += 1;
172            }
173            LambertW => {
174                self.transcendental_calls += 1;
175            }
176            Sin | Cos | Tan | Sec | Csc | Cot | Sinh | Cosh | Tanh | Sech | Csch | Coth
177            | Arcsin | Arccos | Arctan | Arcsinh | Arccosh | Arctanh => {
178                self.transcendental_calls += 1;
179                self.trigonometric_calls += 1;
180            }
181            _ => {}
182        }
183    }
184}

186fn child_expressions(expression: &ExpressionReference) -> Vec<ExpressionReference> {
187    // Structural children of the node kinds the walk does not case out itself. Binder
188    // bodies are included without regard to shadowing, which is what makes the counts
189    // over-inclusive and therefore safe as gates.
190    use std::sync::Arc;
191    match &expression.kind {
192        ExpressionKind::Sum(children)
193        | ExpressionKind::Product(children)
194        | ExpressionKind::FiniteSet(children)
195        | ExpressionKind::SetUnion(children)
196        | ExpressionKind::SetIntersection(children)
197        | ExpressionKind::Distribution(_, children)
198        | ExpressionKind::SpecialFunctionParameterVector(children) => children.clone(),
199        ExpressionKind::Interval(lower_bound, upper_bound, _, _)
200        | ExpressionKind::SetComplement(lower_bound, upper_bound) => {
201            vec![Arc::clone(lower_bound), Arc::clone(upper_bound)]
202        }
203        ExpressionKind::Factorial(argument)
204        | ExpressionKind::LaplaceTransform(argument, _, _)
205        | ExpressionKind::InverseLaplaceTransform(argument, _, _)
206        | ExpressionKind::FourierTransform(argument, _, _)
207        | ExpressionKind::InverseFourierTransform(argument, _, _)
208        | ExpressionKind::RandomVariable(_, argument) => vec![Arc::clone(argument)],
209        ExpressionKind::Integral(body, _, lower_bound, upper_bound)
210        | ExpressionKind::Summation(body, _, lower_bound, upper_bound)
211        | ExpressionKind::IndexedProduct(body, _, lower_bound, upper_bound) => {
212            let mut children = vec![Arc::clone(body)];
213            children.extend(lower_bound.iter().cloned());
214            children.extend(upper_bound.iter().cloned());
215            children
216        }
217        ExpressionKind::Piecewise(branches) => branches
218            .iter()
219            .flat_map(|(condition, value)| [Arc::clone(condition), Arc::clone(value)])
220            .collect(),
221        ExpressionKind::OrdinaryDifferentialEquation(left, right, _)
222        | ExpressionKind::RecurrenceEquation(left, right, _) => {
223            vec![Arc::clone(left), Arc::clone(right)]
224        }
225        ExpressionKind::ImageSet(_, body, set) | ExpressionKind::ConditionSet(_, body, set) => {
226            vec![Arc::clone(body), Arc::clone(set)]
227        }
228        _ => Vec::new(),
229    }
