16/// Public native enumeration named `relation_kind`.
17pub enum RelationKind {
18    /// Represents the `less` case.
19    Less,
20    /// Represents the `less_or_equal` case.
21    LessOrEqual,
22    /// Represents the `greater` case.
23    Greater,
24    /// Represents the `greater_or_equal` case.
25    GreaterOrEqual,
26    /// Represents the `not_equal` case.
27    NotEqual,
28}

30/// Performs the advanced native `solve_relation_as_set` operation.
31pub fn solve_relation_as_set(
32    context: &Context,
33    relation: &ExpressionReference,
34    variable: SymbolIdentifier,
35) -> Option<ExpressionReference> {
36    // Recognise a top-level inequality relation `left R right` and return the
37    // real solution set of `{ x : left(x) R right(x) }`. Returns None when the
38    // expression is not one of the recognised relation builtins, so the caller
39    // can fall through to its equation handling. When the relation is recognised
40    // but does not reduce to a univariate rational polynomial inequality, a
41    // rigorous condition_set placeholder is returned rather than a guessed
42    // interval.
43    let (left, right, kind) = match_relation(relation)?;
44    let difference = builders::subtract(context, left, right);
45    match solve_polynomial_inequality(context, &difference, variable, kind) {
46        Some(solution) => Some(solution),
47        None => {
48            let universe = real_line(context);
49            Some(condition_set(
50                context,
51                variable,
52                universe,
53                Arc::clone(relation),
54            ))
55        }
56    }
57}

59/// Performs the advanced native `solve_relation_set_multivariate` operation.
60pub fn solve_relation_set_multivariate(
61    context: &Context,
62    relation: &ExpressionReference,
63    variables: &[SymbolIdentifier],
64) -> Option<ExpressionReference> {
65    // The multivariate side of solving over inequalities (a real-quantifier-elimination
66    // consumer): a Boolean combination of polynomial relations over ℝⁿ routes to the
67    // CAD-backed semialgebraic solver, which returns the quantifier-free description of
68    // the real solution set. The univariate `solve_relation_as_set` above is unchanged
69    // and remains the fast path for one variable; this is the additive seam for two or
70    // more free variables, where the single-variable interval assembly does not apply.
71    crate::real_quantifier_elimination::solve_semialgebraic(context, relation, variables)
72}

74/// Performs the advanced native `solve_polynomial_inequality` operation.
75pub fn solve_polynomial_inequality(
76    context: &Context,
77    difference: &ExpressionReference,
78    variable: SymbolIdentifier,
79    kind: RelationKind,
80) -> Option<ExpressionReference> {
81    // Solve `{ x : difference(x) R 0 }` for a difference that is a univariate
82    // polynomial over the rationals. Returns None when the difference is not such
83    // a polynomial (a symbolic-coefficient or transcendental input), so the
84    // caller can emit a condition_set placeholder. The constant cases (the zero
85    // polynomial and a nonzero constant) are answered directly; otherwise the
86    // solution is assembled from the real-root sign partition.
87    let polynomial_value = expression_to_univariate(difference, variable)?;
88    if is_zero(&polynomial_value) {
89        return Some(zero_difference_solution(context, kind));
90    }
91    if degree(&polynomial_value).unwrap_or(0) == 0 {
92        let partition = solve_sign_partition(&polynomial_value)?;
93        let constant_sign = partition.region_signs[0];
94        return Some(constant_solution(context, constant_sign, kind));
95    }
96    let partition = solve_sign_partition(&polynomial_value)?;
97    Some(build_solution_set(context, &partition, variable, kind))
98}

100fn build_solution_set(
101    context: &Context,
102    partition: &SignPartition,
103    variable: SymbolIdentifier,
104    kind: RelationKind,
105) -> ExpressionReference {
106    // Assemble the solution set from the sign partition. Each open region whose
107    // sign satisfies the relation contributes an open interval (a half-line at the
108    // ends); the non-strict relations additionally admit every real root as a
109    // point, and the not-equal relation is the real line with the roots removed.
110    let root_count = partition.roots.len();
111    if matches!(kind, RelationKind::NotEqual) {
112        return not_equal_solution(context, partition, variable);
113    }
114    let mut pieces: Vec<ExpressionReference> = Vec::new();
115    for region in 0..=root_count {
116        if !sign_satisfies(partition.region_signs[region], kind) {
117            continue;
118        }
119        let lower = if region == 0 {
120            builders::constant(context, BuiltinConstant::NegativeInfinity)
121        } else {
122            root_endpoint(context, partition, variable, region - 1)
123        };
124        let upper = if region == root_count {
125            builders::constant(context, BuiltinConstant::Infinity)
126        } else {
127            root_endpoint(context, partition, variable, region)
128        };
129        pieces.push(interval(context, lower, upper, false, false));
130    }
131    if matches!(
132        kind,
133        RelationKind::LessOrEqual | RelationKind::GreaterOrEqual
134    ) {
135        let root_points: Vec<ExpressionReference> = (0..root_count)
136            .map(|index| root_endpoint(context, partition, variable, index))
137            .collect();
138        if !root_points.is_empty() {
139            pieces.push(finite_set(context, root_points));
140        }
141    }
142    if pieces.is_empty() {
143        return empty_set(context);
144    }
145    if pieces.len() == 1 {
146        return pieces.into_iter().next().unwrap();
147    }
148    set_union(context, pieces)
149}

151fn not_equal_solution(
152    context: &Context,
153    partition: &SignPartition,
154    variable: SymbolIdentifier,
155) -> ExpressionReference {
156    // `difference != 0` over the reals is the whole line with the real roots
157    // removed.
158    let root_count = partition.roots.len();
159    if root_count == 0 {
160        return real_line(context);
161    }
162    let root_points: Vec<ExpressionReference> = (0..root_count)
163        .map(|index| root_endpoint(context, partition, variable, index))
164        .collect();
165    let removed = finite_set(context, root_points);
166    let universe = real_line(context);
167    set_complement(context, universe, removed)
168}

170fn root_endpoint(
171    context: &Context,
172    partition: &SignPartition,
173    variable: SymbolIdentifier,
174    index: usize,
175) -> ExpressionReference {
176    // The expression naming the index-th smallest real root. Rational roots are
177    // emitted as exact rational literals; irrational roots become an order-aware
178    // real_root(squarefree, variable, index) placeholder.
179    let interval_value = &partition.roots[index];
180    if interval_value.is_exact() {
181        return builders::rational_literal(context, interval_value.lower.clone());
182    }
183    real_root_node(context, &partition.squarefree, variable, index)
184}

186pub(crate) fn real_root_node(
187    context: &Context,
188    squarefree: &crate::polynomial::Polynomial,
189    variable: SymbolIdentifier,
190    real_root_index: usize,
191) -> ExpressionReference {
192    // Build the placeholder real_root(p, x, k) denoting the k-th smallest real
193    // root of the squarefree polynomial p. Encoded as a user function call, in
194    // the same self-describing style as polynomial_root, so it round-trips
195    // through the interner and the display and substitution machinery without a
196    // new ExpressionKind variant.
197    let function_symbol = context.symbol_table.intern_symbol("real_root");
198    let function = FunctionIdentifier::User(function_symbol);
199    let polynomial_expression = polynomial_to_expression(context, squarefree);
200    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
201    let index_expression = builders::integer_value(context, real_root_index as i64);
202    builders::function_call(
203        context,
204        function,
205        vec![polynomial_expression, variable_expression, index_expression],
206    )
207}

209fn sign_satisfies(sign: i32, kind: RelationKind) -> bool {
210    // Whether an open region of the given sign belongs to the solution of the
211    // relation against zero. The boundary (sign zero) never arises on an open
212    // region, where the polynomial is nonzero by construction.
213    match kind {
214        RelationKind::Greater | RelationKind::GreaterOrEqual => sign > 0,
215        RelationKind::Less | RelationKind::LessOrEqual => sign < 0,
216        RelationKind::NotEqual => sign != 0,
217    }
218}

220fn zero_difference_solution(context: &Context, kind: RelationKind) -> ExpressionReference {
221    // The solution when the two sides are identically equal, so the relation is
222    // tested as `0 R 0`.
223    let holds = match kind {
224        RelationKind::Less | RelationKind::Greater | RelationKind::NotEqual => false,
225        RelationKind::LessOrEqual | RelationKind::GreaterOrEqual => true,
226    };
227    if holds {
228        real_line(context)
229    } else {
230        empty_set(context)
231    }
232}
233
234fn constant_solution(context: &Context, sign: i32, kind: RelationKind) -> ExpressionReference {
235    // The solution when the difference is a nonzero constant of the given sign:
236    // the relation holds either everywhere or nowhere.
237    if sign_satisfies(sign, kind) {
238        real_line(context)
239    } else {
240        empty_set(context)
241    }
242}

244fn match_relation(
245    relation: &ExpressionReference,
246) -> Option<(ExpressionReference, ExpressionReference, RelationKind)> {
247    // Decompose a binary relation builtin call into its two sides and the
248    // relation kind. `left R right` is reduced by the caller to `(left - right) R
249    // 0`, so the relation kind carried here applies unchanged to the difference.
250    if let ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) =
251        &relation.kind
252    {
253        if arguments.len() != 2 {
254            return None;
255        }
256        let kind = match builtin {
257            BuiltinFunction::LessThan => RelationKind::Less,
258            BuiltinFunction::LessThanOrEqual => RelationKind::LessOrEqual,
259            BuiltinFunction::GreaterThan => RelationKind::Greater,
260            BuiltinFunction::GreaterThanOrEqual => RelationKind::GreaterOrEqual,
261            BuiltinFunction::NotEqual => RelationKind::NotEqual,
262            _ => return None,
263        };
264        return Some((Arc::clone(&arguments[0]), Arc::clone(&arguments[1]), kind));
265    }
266    None
