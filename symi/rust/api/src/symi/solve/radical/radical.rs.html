15/// Performs the advanced native `cubic_roots` operation.
16pub fn cubic_roots(
17    context: &Context,
18    factor: &Polynomial,
19    _variable: SymbolIdentifier,
20) -> Option<Vec<ExpressionReference>> {
21    // Closed-form radical solution of a numeric cubic a*x^3 + b*x^2 + c*x + d
22    // via the branch-robust Cardano form. The three roots are
23    // x_k = -(1/(3a)) * [ b + xi^k * C + delta_0 / (xi^k * C) ] for k = 0,1,2,
24    // where C = cube_root((delta_1 +- sqrt(delta_1^2 - 4*delta_0^3)) / 2) with
25    // the sign chosen so C is nonzero, delta_0 = b^2 - 3ac,
26    // delta_1 = 2b^3 - 9abc + 27a^2 d, and xi = (-1 + sqrt(-3))/2 a primitive
27    // cube root of unity. Because the formula pairs C with delta_0/C, the three
28    // roots are correct for ANY fixed branch of the cube root, so the numeric
29    // evaluator's principal complex cube root is safe. Returns None when the
30    // factor is not a degree-3 polynomial with numeric coefficients.
31    let coefficients = numeric_dense_coefficients(factor, 3)?;
32    let coefficient_a = &coefficients[3];
33    let coefficient_b = &coefficients[2];
34    let coefficient_c = &coefficients[1];
35    let coefficient_d = &coefficients[0];
36    if is_zero(coefficient_a) {
37        return None;
38    }
39    let three = integer_number(3);
40    let delta_0 = subtract(
41        &multiply(coefficient_b, coefficient_b),
42        &multiply(&three, &multiply(coefficient_a, coefficient_c)),
43    );
44    let two_b_cubed = multiply(
45        &integer_number(2),
46        &multiply(&multiply(coefficient_b, coefficient_b), coefficient_b),
47    );
48    let nine_a_b_c = multiply(
49        &integer_number(9),
50        &multiply(coefficient_a, &multiply(coefficient_b, coefficient_c)),
51    );
52    let twenty_seven_a_squared_d = multiply(
53        &integer_number(27),
54        &multiply(&multiply(coefficient_a, coefficient_a), coefficient_d),
55    );
56    let delta_1 = add(
57        &subtract(&two_b_cubed, &nine_a_b_c),
58        &twenty_seven_a_squared_d,
59    );
60    let three_a = multiply(&three, coefficient_a);
61    let negative_reciprocal_three_a = negate(&divide(&integer_number(1), &three_a)?);
62    if is_zero(&delta_0) && is_zero(&delta_1) {
63        // Triple root -b/(3a): cannot arise from an irreducible cubic but is
64        // guarded so the returned multiset always has cardinality three.
65        let triple_root_number = multiply(&negative_reciprocal_three_a, coefficient_b);
66        let triple_root = number_expression(context, &triple_root_number);
67        return Some(vec![
68            Arc::clone(&triple_root),
69            Arc::clone(&triple_root),
70            triple_root,
71        ]);
72    }
73    let radicand = subtract(
74        &multiply(&delta_1, &delta_1),
75        &multiply(
76            &integer_number(4),
77            &multiply(&delta_0, &multiply(&delta_0, &delta_0)),
78        ),
79    );
80    let use_plus_sign = if is_zero(&delta_0) {
81        !is_negative(&delta_1)
82    } else {
83        true
84    };
85    let square_root_radicand = square_root_of_number(context, &radicand);
86    let signed_square_root = if use_plus_sign {
87        square_root_radicand
88    } else {
89        builders::negate(context, square_root_radicand)
90    };
91    let delta_1_expression = number_expression(context, &delta_1);
92    let inner_numerator = builders::sum(context, vec![delta_1_expression, signed_square_root]);
93    let two_expression = builders::integer_value(context, 2);
94    let inner = builders::divide(context, inner_numerator, two_expression);
95    let cube_root_term = cube_root_expression(context, inner);
96    let unity = cube_root_of_unity(context);
97    let two_for_square = builders::integer_value(context, 2);
98    let unity_squared = builders::power(context, Arc::clone(&unity), two_for_square);
99    let coefficient_b_expression = number_expression(context, coefficient_b);
100    let delta_0_expression = number_expression(context, &delta_0);
101    let negative_reciprocal_expression = number_expression(context, &negative_reciprocal_three_a);
102    let mut roots: Vec<ExpressionReference> = Vec::with_capacity(3);
103    for root_index in 0..3usize {
104        let unity_power = match root_index {
105            0 => None,
106            1 => Some(Arc::clone(&unity)),
107            _ => Some(Arc::clone(&unity_squared)),
108        };
109        let unity_times_cube_root = match unity_power {
110            None => Arc::clone(&cube_root_term),
111            Some(power_value) => {
112                builders::product(context, vec![power_value, Arc::clone(&cube_root_term)])
113            }
114        };
115        let delta_0_over_term = builders::divide(
116            context,
117            Arc::clone(&delta_0_expression),
118            Arc::clone(&unity_times_cube_root),
119        );
120        let bracket = builders::sum(
121            context,
122            vec![
123                Arc::clone(&coefficient_b_expression),
124                unity_times_cube_root,
125                delta_0_over_term,
126            ],
127        );
128        let root = builders::product(
129            context,
130            vec![Arc::clone(&negative_reciprocal_expression), bracket],
131        );
132        roots.push(root);
133    }
134    Some(roots)
135}

137/// Performs the advanced native `quartic_roots` operation.
138pub fn quartic_roots(
139    context: &Context,
140    factor: &Polynomial,
141    variable: SymbolIdentifier,
142) -> Option<Vec<ExpressionReference>> {
143    // Closed-form radical solution of a numeric quartic via Ferrari's method.
144    // The quartic is made monic (coefficients B,C,D,E) and depressed by the
145    // substitution x = t - B/4 into t^4 + p t^2 + q t + r. When q = 0 the
146    // depressed quartic is biquadratic (quadratic in t^2) and is solved by two
147    // nested square roots; otherwise the resolvent cubic
148    // 8 m^3 + 8p m^2 + (2p^2 - 8r) m - q^2 is solved (its constant term -q^2 is
149    // nonzero, so every resolvent root is nonzero), one root m is taken, and the
150    // depressed quartic factors into the two quadratics
151    //   t^2 +- sqrt(2m) t + (p/2 + m -+ q/(2 sqrt(2m))),
152    // each solved by the quadratic formula. The four roots are returned in the
153    // fixed order (quadratic one: +, -; quadratic two: +, -). Returns None when
154    // the factor is not a degree-4 polynomial with numeric coefficients.
155    let coefficients = numeric_dense_coefficients(factor, 4)?;
156    let coefficient_a = &coefficients[4];
157    if is_zero(coefficient_a) {
158        return None;
159    }
160    let monic_b = divide(&coefficients[3], coefficient_a)?;
161    let monic_c = divide(&coefficients[2], coefficient_a)?;
162    let monic_d = divide(&coefficients[1], coefficient_a)?;
163    let monic_e = divide(&coefficients[0], coefficient_a)?;
164    let monic_b_squared = multiply(&monic_b, &monic_b);
165    let monic_b_cubed = multiply(&monic_b_squared, &monic_b);
166    let monic_b_fourth = multiply(&monic_b_squared, &monic_b_squared);
167    let depressed_p = subtract(
168        &monic_c,
169        &multiply(&rational_number(3, 8), &monic_b_squared),
170    );
171    let depressed_q = add(
172        &subtract(
173            &monic_d,
174            &multiply(&rational_number(1, 2), &multiply(&monic_b, &monic_c)),
175        ),
176        &multiply(&rational_number(1, 8), &monic_b_cubed),
177    );
178    let depressed_r = add(
179        &subtract(
180            &add(
181                &monic_e,
182                &multiply(
183                    &rational_number(1, 16),
184                    &multiply(&monic_b_squared, &monic_c),
185                ),
186            ),
187            &multiply(&rational_number(1, 4), &multiply(&monic_b, &monic_d)),
188        ),
189        &negate(&multiply(&rational_number(3, 256), &monic_b_fourth)),
190    );
191    let shift = multiply(&rational_number(1, 4), &monic_b);
192    let shift_expression = number_expression(context, &shift);
193    let depressed_roots = if is_zero(&depressed_q) {
194        biquadratic_depressed_roots(context, &depressed_p, &depressed_r)?
195    } else {
196        ferrari_depressed_roots(context, &depressed_p, &depressed_q, &depressed_r, variable)?
197    };
198    let roots = depressed_roots
199        .into_iter()
200        .map(|depressed_root| {
201            builders::subtract(context, depressed_root, Arc::clone(&shift_expression))
202        })
203        .collect();
204    Some(roots)
205}

207fn biquadratic_depressed_roots(
208    context: &Context,
209    depressed_p: &Number,
210    depressed_r: &Number,
211) -> Option<Vec<ExpressionReference>> {
212    // Roots of the depressed biquadratic t^4 + p t^2 + r = 0. Setting u = t^2,
213    // u = (-p +- sqrt(p^2 - 4r))/2, and each u contributes the pair t = +-sqrt(u).
214    let discriminant = subtract(
215        &multiply(depressed_p, depressed_p),
216        &multiply(&integer_number(4), depressed_r),
217    );
218    let square_root_discriminant = square_root_of_number(context, &discriminant);
219    let negative_p_expression = number_expression(context, &negate(depressed_p));
220    let two_expression = builders::integer_value(context, 2);
221    let u_plus_numerator = builders::sum(
222        context,
223        vec![
224            Arc::clone(&negative_p_expression),
225            Arc::clone(&square_root_discriminant),
226        ],
227    );
228    let u_plus = builders::divide(context, u_plus_numerator, Arc::clone(&two_expression));
229    let negative_square_root = builders::negate(context, Arc::clone(&square_root_discriminant));
230    let u_minus_numerator =
231        builders::sum(context, vec![negative_p_expression, negative_square_root]);
232    let u_minus = builders::divide(context, u_minus_numerator, two_expression);
233    let root_from_u_plus = square_root_expression(context, u_plus);
234    let root_from_u_minus = square_root_expression(context, u_minus);
235    let negative_root_from_u_plus = builders::negate(context, Arc::clone(&root_from_u_plus));
236    let negative_root_from_u_minus = builders::negate(context, Arc::clone(&root_from_u_minus));
237    Some(vec![
238        root_from_u_plus,
239        negative_root_from_u_plus,
240        root_from_u_minus,
241        negative_root_from_u_minus,
242    ])
243}

245fn ferrari_depressed_roots(
246    context: &Context,
247    depressed_p: &Number,
248    depressed_q: &Number,
249    depressed_r: &Number,
250    variable: SymbolIdentifier,
251) -> Option<Vec<ExpressionReference>> {
252    // Roots of the depressed quartic t^4 + p t^2 + q t + r = 0 with q != 0 via
253    // the resolvent cubic. A nonzero resolvent root m gives sqrt(2m) and the two
254    // quadratic factors whose roots are the four depressed roots.
255    let two_p_squared = multiply(&integer_number(2), &multiply(depressed_p, depressed_p));
256    let eight_r = multiply(&integer_number(8), depressed_r);
257    let resolvent_dense = vec![
258        negate(&multiply(depressed_q, depressed_q)),
259        subtract(&two_p_squared, &eight_r),
260        multiply(&integer_number(8), depressed_p),
261        integer_number(8),
262    ];
263    let resolvent = univariate_from_dense(variable, resolvent_dense);
264    let resolvent_roots = cubic_roots(context, &resolvent, variable)?;
265    let resolvent_root = resolvent_roots.into_iter().next()?;
266    let two_for_two_m = builders::integer_value(context, 2);
267    let two_m = builders::product(context, vec![two_for_two_m, Arc::clone(&resolvent_root)]);
268    let square_root_two_m = square_root_expression(context, two_m);
269    let two_for_denominator = builders::integer_value(context, 2);
270    let two_square_root_two_m = builders::product(
271        context,
272        vec![two_for_denominator, Arc::clone(&square_root_two_m)],
273    );
274    let depressed_q_expression = number_expression(context, depressed_q);
275    let q_over_two_square_root =
276        builders::divide(context, depressed_q_expression, two_square_root_two_m);
277    let negative_q_over_two_square_root =
278        builders::negate(context, Arc::clone(&q_over_two_square_root));
279    let half_p_expression =
280        number_expression(context, &multiply(&rational_number(1, 2), depressed_p));
281    let constant_one = builders::sum(
282        context,
283        vec![
284            Arc::clone(&half_p_expression),
285            Arc::clone(&resolvent_root),
286            negative_q_over_two_square_root,
287        ],
288    );
289    let constant_two = builders::sum(
290        context,
291        vec![half_p_expression, resolvent_root, q_over_two_square_root],
292    );
293    let negative_square_root_two_m = builders::negate(context, Arc::clone(&square_root_two_m));
294    let (first_plus, first_minus) =
295        monic_quadratic_roots(context, Arc::clone(&square_root_two_m), constant_one);
296    let (second_plus, second_minus) =
297        monic_quadratic_roots(context, negative_square_root_two_m, constant_two);
298    Some(vec![first_plus, first_minus, second_plus, second_minus])
299}

301fn monic_quadratic_roots(
302    context: &Context,
303    linear_coefficient: ExpressionReference,
304    constant_coefficient: ExpressionReference,
305) -> (ExpressionReference, ExpressionReference) {
306    // The two roots of the monic quadratic t^2 + (linear) t + (constant) = 0
307    // via the quadratic formula t = (-(linear) +- sqrt((linear)^2 - 4 constant))/2.
308    let two_for_power = builders::integer_value(context, 2);
309    let linear_squared = builders::power(context, Arc::clone(&linear_coefficient), two_for_power);
310    let four_expression = builders::integer_value(context, 4);
311    let four_constant = builders::product(context, vec![four_expression, constant_coefficient]);
312    let discriminant = builders::subtract(context, linear_squared, four_constant);
313    let square_root_discriminant = square_root_expression(context, discriminant);
314    let negative_linear = builders::negate(context, linear_coefficient);
315    let two_expression = builders::integer_value(context, 2);
316    let root_plus_numerator = builders::sum(
317        context,
318        vec![
319            Arc::clone(&negative_linear),
320            Arc::clone(&square_root_discriminant),
321        ],
322    );
323    let root_plus = builders::divide(context, root_plus_numerator, Arc::clone(&two_expression));
324    let negative_square_root = builders::negate(context, square_root_discriminant);
325    let root_minus_numerator = builders::sum(context, vec![negative_linear, negative_square_root]);
326    let root_minus = builders::divide(context, root_minus_numerator, two_expression);
327    (root_plus, root_minus)
328}

330/// Performs the advanced native `binomial_roots` operation.
331pub fn binomial_roots(
332    context: &Context,
333    factor: &Polynomial,
334    _variable: SymbolIdentifier,
335) -> Option<Vec<ExpressionReference>> {
336    // All n roots of an irreducible binomial a*x^n + k (n >= 3): writing
337    // x^n = c with c = -k/a, the roots are the n-th roots of c. For a numeric
338    // (always real-rational) c the roots are built around the real nonnegative
339    // radical |c|^(1/n) with sign-aware angles — angle 2*pi*j/n when c > 0 and
340    // (2*j+1)*pi/n when c < 0 — so the on-axis roots collapse to manifestly real
341    // +/- |c|^(1/n) form rather than the complex principal branch c^(1/n) times a
342    // root of unity. This matters downstream: the real root of x^3 + 2 is the
343    // real -(2^(1/3)), and emitting it as (-2)^(1/3)*exp(2*pi*i/3) — numerically
344    // real but structurally carrying the imaginary unit — made the real-line
345    // filters in solveset / singularities drop it, so solveset(x^3+2) and
346    // singularities(1/(x^3+2)) wrongly returned the empty set. For a symbolic c
347    // (sign undecidable) the construction falls back to the principal-branch form
348    // c^(1/n) * exp(2*pi*i*j/n), still rigorous because every n-th root of c is
349    // c^(1/n) times an n-th root of unity with no branch-selection hazard (the
350    // gate lift). Returns None unless the factor is exactly two terms at
351    // exponents n and 0 with reifiable coefficients.
352    let degree_value = degree(factor)?;
353    if degree_value < 3 {
354        return None;
355    }
356    if factor.terms.len() != 2 {
357        return None;
358    }
359    let leading_coefficient = factor
360        .terms
361        .iter()
362        .find(|(exponent, _)| *exponent == degree_value)
363        .map(|(_, coefficient)| coefficient)?;
364    let constant_coefficient = factor
365        .terms
366        .iter()
367        .find(|(exponent, _)| *exponent == 0)
368        .map(|(_, coefficient)| coefficient)?;
369    if is_coefficient_algebraic(leading_coefficient)
370        || is_coefficient_algebraic(constant_coefficient)
371    {
372        return None;
373    }
374    if coefficient_is_zero(leading_coefficient) || coefficient_is_zero(constant_coefficient) {
375        return None;
376    }
377    if let Some(roots) = numeric_binomial_roots(
378        context,
379        leading_coefficient,
380        constant_coefficient,
381        degree_value,
382    ) {
383        return Some(roots);
384    }
385    let constant_expression =
386        binomial_constant_expression(context, leading_coefficient, constant_coefficient);
387    let inverse_degree = builders::rational(
388        context,
389        Integer::from(1),
390        Integer::from(degree_value as i64),
391    );
392    let principal_root = builders::power(context, constant_expression, inverse_degree);
393    let mut roots: Vec<ExpressionReference> = Vec::with_capacity(degree_value);
394    for root_index in 0..degree_value {
395        if root_index == 0 {
396            roots.push(Arc::clone(&principal_root));
397            continue;
398        }
399        let unity_power = root_of_unity_power(context, root_index, degree_value);
400        let root = builders::product(context, vec![Arc::clone(&principal_root), unity_power]);
401        roots.push(root);
402    }
403    Some(roots)
404}

406fn numeric_binomial_roots(
407    context: &Context,
408    leading_coefficient: &PolynomialCoefficient,
409    constant_coefficient: &PolynomialCoefficient,
410    degree_value: usize,
411) -> Option<Vec<ExpressionReference>> {
412    // The n-th roots of a numeric (real-rational) radicand c = -k/a, built around
413    // the real nonnegative radical |c|^(1/n) so on-axis roots come out in real
414    // +/- |c|^(1/n) form. The angle of root j (in units of pi) is 2*j/n when
415    // c > 0 and (2*j+1)/n when c < 0; whenever that angle is an integer the root
416    // lies on the real axis (+radical for an even multiple of pi, -radical for an
417    // odd one) and is emitted without the imaginary unit, so the downstream
418    // real-line filters keep the genuinely real roots. Returns None when either
419    // coefficient is non-numeric so the caller falls back to the symbolic
420    // principal-branch construction.
421    let leading_number = numeric_coefficient(leading_coefficient)?;
422    let constant_number = numeric_coefficient(constant_coefficient)?;
423    let radicand = divide(&negate(&constant_number), &leading_number)
424        .expect("numeric_binomial_roots: leading coefficient is nonzero");
425    let radicand_is_negative = is_negative(&radicand);
426    let magnitude = if radicand_is_negative {
427        negate(&radicand)
428    } else {
429        radicand
430    };
431    let magnitude_expression = number_expression(context, &magnitude);
432    let inverse_degree = builders::rational(
433        context,
434        Integer::from(1),
435        Integer::from(degree_value as i64),
436    );
437    let real_radical = builders::power(context, magnitude_expression, inverse_degree);
438    let degree_as_integer = degree_value as i64;
439    let mut roots: Vec<ExpressionReference> = Vec::with_capacity(degree_value);
440    for root_index in 0..degree_value {
441        let angle_numerator = if radicand_is_negative {
442            2 * root_index as i64 + 1
443        } else {
444            2 * root_index as i64
445        };
446        if angle_numerator % degree_as_integer == 0 {
447            let half_turns = angle_numerator / degree_as_integer;
448            if half_turns.rem_euclid(2) == 0 {
449                roots.push(Arc::clone(&real_radical));
450            } else {
451                let negated = builders::negate(context, Arc::clone(&real_radical));
452                roots.push(negated);
453            }
454            continue;
455        }
456        let unity_power =
457            complex_exponential_pi_multiple(context, angle_numerator, degree_as_integer);
458        let root = builders::product(context, vec![Arc::clone(&real_radical), unity_power]);
459        roots.push(root);
460    }
461    Some(roots)
462}

464fn complex_exponential_pi_multiple(
465    context: &Context,
466    angle_numerator: i64,
467    angle_denominator: i64,
468) -> ExpressionReference {
469    // The unit-modulus complex number exp(i * (angle_numerator/angle_denominator)
470    // * pi), built in complex exponential form so the numeric evaluator's
471    // principal exp closes it over the complex plane. The rational angle
472    // coefficient reduces through the auto-simplifying builder.
473    let angle_coefficient = builders::rational(
474        context,
475        Integer::from(angle_numerator),
476        Integer::from(angle_denominator),
477    );
478    let pi = builders::constant(context, BuiltinConstant::Pi);
479    let imaginary_unit = builders::constant(context, BuiltinConstant::ImaginaryUnit);
480    let exponent = builders::product(context, vec![angle_coefficient, pi, imaginary_unit]);
481    builders::function_call(
482        context,
483        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
484        vec![exponent],
485    )
486}

488fn root_of_unity_power(
489    context: &Context,
490    root_index: usize,
491    degree_value: usize,
492) -> ExpressionReference {
493    // The primitive-n-th-root-of-unity power exp(2*pi*i*root_index/degree),
494    // built in complex exponential form so the numeric evaluator's principal
495    // exp closes it over the complex plane.
496    let angle_coefficient = builders::rational(
497        context,
498        Integer::from(2 * root_index as i64),
499        Integer::from(degree_value as i64),
500    );
501    let pi = builders::constant(context, BuiltinConstant::Pi);
502    let imaginary_unit = builders::constant(context, BuiltinConstant::ImaginaryUnit);
503    let exponent = builders::product(context, vec![angle_coefficient, pi, imaginary_unit]);
504    builders::function_call(
505        context,
506        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
507        vec![exponent],
508    )
509}

511fn cube_root_of_unity(context: &Context) -> ExpressionReference {
512    // The primitive cube root of unity xi = (-1 + sqrt(-3))/2, built in the
513    // imaginary-unit-free surd shape the quadratic solver already uses.
514    let minus_three = builders::integer_value(context, -3);
515    let square_root_minus_three = square_root_expression(context, minus_three);
516    let minus_one = builders::integer_value(context, -1);
517    let numerator = builders::sum(context, vec![minus_one, square_root_minus_three]);
518    let two = builders::integer_value(context, 2);
519    builders::divide(context, numerator, two)
520}

522fn square_root_of_number(context: &Context, value: &Number) -> ExpressionReference {
523    // The principal square root of a numeric value, emitted as value^(1/2) so a
524    // perfect square folds to its rational root and any other value (including a
525    // negative one, whose root is complex) stays in canonical surd form.
526    let value_expression = number_expression(context, value);
527    square_root_expression(context, value_expression)
528}

530fn square_root_expression(context: &Context, value: ExpressionReference) -> ExpressionReference {
531    // The principal square root of an expression as value^(1/2).
532    let one_half = builders::rational_value(context, 1, 2);
533    builders::power(context, value, one_half)
534}
535
536fn cube_root_expression(context: &Context, value: ExpressionReference) -> ExpressionReference {
537    // The principal cube root of an expression as value^(1/3).
538    let one_third = builders::rational_value(context, 1, 3);
539    builders::power(context, value, one_third)
540}
541
542fn number_expression(context: &Context, value: &Number) -> ExpressionReference {
543    // Lift a numeric value into its canonical interned expression form.
544    match value {
545        Number::Integer(integer_value) => builders::integer(context, integer_value.clone()),
546        Number::Rational(rational_value) => {
547            builders::rational_literal(context, rational_value.clone())
548        }
549    }
550}
551
552fn numeric_dense_coefficients(factor: &Polynomial, expected_degree: usize) -> Option<Vec<Number>> {
553    // Dense ascending-degree numeric coefficient vector of a univariate factor,
554    // or None when the factor degree differs from expected_degree or any
555    // coefficient is non-numeric. Index i holds the coefficient of x^i.
556    if degree(factor)? != expected_degree {
557        return None;
558    }
559    let mut coefficients = vec![integer_number(0); expected_degree + 1];
560    for (exponent, coefficient) in factor.terms.iter() {
561        if *exponent > expected_degree {
562            return None;
563        }
564        coefficients[*exponent] = numeric_coefficient(coefficient)?;
565    }
566    Some(coefficients)
567}
568
569fn numeric_coefficient(coefficient: &PolynomialCoefficient) -> Option<Number> {
570    // Project a polynomial coefficient to a numeric value, or None for the
571    // nested and algebraic variants the numeric radical solvers do not accept.
572    match coefficient {
573        PolynomialCoefficient::Numeric(value) => Some(value.clone()),
574        _ => None,
575    }
576}
577
578fn is_coefficient_algebraic(coefficient: &PolynomialCoefficient) -> bool {
579    // The algebraic-element coefficient variant has no expression-layer bridge
580    // yet, so the radical solvers decline it rather than reify a placeholder.
581    matches!(coefficient, PolynomialCoefficient::Algebraic(_))
582}
583
584fn coefficient_is_zero(coefficient: &PolynomialCoefficient) -> bool {
585    // True when a coefficient is the zero of its ring. Canonical polynomials do
586    // not store zero coefficients as terms, but the binomial solver guards the
587    // leading/constant coefficients defensively before dividing by them.
588    match coefficient {
589        PolynomialCoefficient::Numeric(value) => is_zero(value),
590        PolynomialCoefficient::Nested(nested) => crate::polynomial::is_zero(nested),
591        PolynomialCoefficient::Algebraic(_) => false,
592    }
593}
594
595fn binomial_constant_expression(
596    context: &Context,
597    leading_coefficient: &PolynomialCoefficient,
598    constant_coefficient: &PolynomialCoefficient,
599) -> ExpressionReference {
600    // The value c = -constant/leading whose n-th roots solve the binomial. The
601    // numeric case reduces to an exact rational; the symbolic case reifies both
602    // coefficients to the expression layer and divides through the
603    // auto-simplifying builders.
604    if let (Some(leading_number), Some(constant_number)) = (
605        numeric_coefficient(leading_coefficient),
606        numeric_coefficient(constant_coefficient),
607    ) {
608        let value = divide(&negate(&constant_number), &leading_number)
609            .expect("binomial_constant_expression: leading coefficient is nonzero");
610        return number_expression(context, &value);
611    }
612    let leading_expression = coefficient_to_expression(context, leading_coefficient);
613    let constant_expression = coefficient_to_expression(context, constant_coefficient);
614    let negated_constant = builders::negate(context, constant_expression);
615    builders::divide(context, negated_constant, leading_expression)
616}

619    // The numeric value of a small integer literal.
620    Number::Integer(Integer::from(value))
621}
