21/// Performs the advanced native `integer` operation.
22pub fn integer(context: &Context, value: exact_integer) -> ExpressionReference {
23    // Build an integer leaf. The interner ensures repeated calls with equal
24    // values share storage.
25    context.interner.intern(ExpressionKind::Integer(value))
26}

28/// Performs the advanced native `rational` operation.
29pub fn rational(
30    context: &Context,
31    numerator: exact_integer,
32    denominator: exact_integer,
33) -> ExpressionReference {
34    // Build a rational leaf normalised to lowest terms. Constructing through the
35    // number module enforces the integer-vs-rational invariant: a denominator of
36    // one collapses back to the integer variant before interning, preserving
37    // hash-cons identity between e.g. rational(4, 2) and integer(2).
38    let value =
39        number::from_integers(numerator, denominator).expect("rational builder: zero denominator");
40    match value {
41        number_value::Integer(integer_value) => integer(context, integer_value),
42        number_value::Rational(rational_value) => context
43            .interner
44            .intern(ExpressionKind::Rational(rational_value)),
45    }
46}

94    // simplifier collapses double negations and folds numeric coefficients.

111/// Performs the advanced native `divide` operation.
112pub fn divide(
113    context: &Context,
114    dividend: ExpressionReference,
115    divisor: ExpressionReference,
116) -> ExpressionReference {
117    // Division is rewritten as multiplication by the reciprocal. There is no
118    // dedicated division node in the ASAE representation. A literal zero divisor
119    // is propagated through simplify_power as the undefined sentinel.
120    let minus_one = integer(context, exact_integer::from(-1));
121    let reciprocal = simplify_power(context, divisor, minus_one);
122    simplify_product(context, vec![dividend, reciprocal])
123}

232/// Performs the advanced native `factorial` operation.
233pub fn factorial(context: &Context, argument: ExpressionReference) -> ExpressionReference {
234    // Build a factorial node. Automatic simplification only collapses
235    // the trivially-defined cases factorial(0) = 1 and factorial(1) = 1;
236    // any other concrete integer argument is interned as a symbolic
237    // factorial node so the surface representation stays compact even
238    // when the numeric value would be unrepresentable in machine
239    // arithmetic. Negative-integer arguments collapse to the symbolic
240    // undefined sentinel since the real-valued factorial does not
241    // extend to negative integers. Numeric evaluation of a symbolic
242    // factorial is the responsibility of the inexact-arithmetic layer
243    // which can decide on its own whether to materialise the bignum
244    // value.
245    use crate::expression::{BuiltinConstant, ExpressionKind};
246    use std::cmp::Ordering as cmp_ordering;
247    if let ExpressionKind::Integer(value) = &argument.kind {
248        match integer_sign(value) {
249            cmp_ordering::Less => {
250                return context
251                    .interner
252                    .intern(ExpressionKind::Constant(BuiltinConstant::Undefined));
253            }
254            cmp_ordering::Equal => return one(context),
255            cmp_ordering::Greater => {
256                if integer_is_one(value) {
257                    return one(context);
258                }
259            }
260        }
261    }
262    context.interner.intern(ExpressionKind::Factorial(argument))
263}

627pub fn rational_literal(context: &Context, value: exact_rational) -> ExpressionReference {
628    // Build an interned rational value, applying the
629    // integer-collapse invariant so a rational with denominator one becomes an
630    // integer leaf. The sign is captured before the move into the

635    rational(
636        context,
637        parts.numerator,
638        natural_to_integer(parts.denominator),
639    )
