14/// Performs the advanced native `differentiate` operation.
15pub fn differentiate(
16    context: &Context,
17    target: &ExpressionReference,
18    with_respect_to: SymbolIdentifier,
19) -> ExpressionReference {
20    crate::instrumentation::record_counter!(differentiation_requests);
21    // Compute the symbolic derivative of an expression with respect to a single
22    // variable. The implementation is total over the supported expression grammar
23    // and deterministic — no heuristic search is involved. Differentiation of an
24    // unknown user function emits a symbolic derivative placeholder so that the
25    // chain rule remains composable across calls. The traversal memoises on the
26    // pair (expression pointer, variable) so that the chain rule's repeated
27    // references to the same subexpression resolve to a single recursive call.
28    let mut memo = ExpressionTraversalMemo::new(context);
29    differentiate_recursive(context, target, with_respect_to, &mut memo)
30}

32/// Differentiate one target inside a traversal a caller already owns.
33///
34/// A caller that asks for several derivatives of one operator — the entries of a gradient,
35/// jacobian, hessian or curl — shares one traversal rather than starting an unrelated memo per
36/// entry. The keys are the same keys [`differentiate`] mints, so the shared subderivatives of the
37/// entries resolve to one recursive call however the caller groups them.
38pub(crate) fn differentiate_within_traversal(
39    context: &Context,
40    target: &ExpressionReference,
41    with_respect_to: SymbolIdentifier,
42    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
43) -> ExpressionReference {
44    crate::instrumentation::record_counter!(differentiation_requests);
45    differentiate_recursive(context, target, with_respect_to, memo)
46}

48/// Performs the advanced native `partial_derivative` operation.
49pub fn partial_derivative(
50    context: &Context,
51    target: &ExpressionReference,
52    orders: &[(SymbolIdentifier, usize)],
53) -> ExpressionReference {
54    // Apply repeated single-variable differentiation according to a list of
55    // (variable, order) pairs. When the input expression contains no
56    // non-differentiable shapes such as abs or signum, Schwarz's theorem permits
57    // freely reordering the partials; we exploit this by sorting the request so
58    // that two callers asking for the same mixed partial in different orders see
59    // pointer-equal results from the memo cache. When abs or signum is in scope
60    // we honour the user's ordering verbatim because Schwarz no longer applies
61    // through a non-smooth subexpression.
62    let permits_canonical_order = !contains_non_smooth_function(target);
63    let effective_orders: Vec<(SymbolIdentifier, usize)> = if permits_canonical_order {
64        let mut sorted: Vec<(SymbolIdentifier, usize)> = orders.to_vec();
65        sorted.sort_by_key(|(identifier, _)| identifier.0);
66        sorted
67    } else {
68        orders.to_vec()
69    };
70    let mut current = Arc::clone(target);
71    for (variable, order) in effective_orders {
72        for _ in 0..order {
73            current = differentiate(context, &current, variable);
74        }
75    }
76    current
77}

79fn differentiate_recursive(
80    context: &Context,
81    target: &ExpressionReference,
82    with_respect_to: SymbolIdentifier,
83    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
84) -> ExpressionReference {
85    // Inner recursive entry. Splits on the ExpressionKind variant and dispatches
86    // to a per-shape rule. Numeric leaves and unrelated symbols differentiate to
87    // zero immediately; the variable being differentiated against returns one;
88    // composite shapes combine recursive results through the auto-simplifying
89    // builders so the returned expression is always in ASAE form.
90    let memo_key = memo.key(
91        context,
92        target,
93        ExpressionOperationFamily::Differentiation,
94        u64::from(with_respect_to.0),
95        1,
96    );
97    if let Some(cached) = memo.lookup(&memo_key) {
98        crate::instrumentation::record_counter!(expression_memo_hits);
99        return cached;
100    }
101    let result = match &target.kind {
102        ExpressionKind::Integer(_)
103        | ExpressionKind::Rational(_)
104        | ExpressionKind::PolynomialRoot(_) => intern_zero(context),
105        ExpressionKind::Constant(_) => intern_zero(context),
106        ExpressionKind::Symbol(identifier) => {
107            if *identifier == with_respect_to {
108                intern_one(context)
109            } else {
110                intern_zero(context)
111            }
112        }
113        ExpressionKind::Sum(children) => {
114            differentiate_sum(context, target, children, with_respect_to, memo)
115        }
116        ExpressionKind::Product(factors) => {
117            differentiate_product(context, target, factors, with_respect_to, memo)
118        }
119        ExpressionKind::Power(base, exponent) => {
120            differentiate_power(context, base, exponent, with_respect_to, memo)
121        }
122        ExpressionKind::Factorial(argument) => {
123            differentiate_factorial(context, argument, with_respect_to, memo)
124        }
125        ExpressionKind::FunctionCall(function, arguments) => differentiate_function_call(
126            context,
127            target,
128            *function,
129            arguments,
130            with_respect_to,
131            memo,
132        ),
133        ExpressionKind::Derivative(function, orders, arguments) => differentiate_derivative_node(
134            context,
135            target,
136            *function,
137            orders,
138            arguments,
139            with_respect_to,
140            memo,
141        ),
142        ExpressionKind::Integral(integrand, variable, lower_bound, upper_bound) => {
143            differentiate_integral(
144                context,
145                integrand,
146                *variable,
147                lower_bound.as_ref(),
148                upper_bound.as_ref(),
149                with_respect_to,
150                memo,
151            )
152        }
153        ExpressionKind::Summation(summand, index, lower_bound, upper_bound) => {
154            differentiate_summation(
155                context,
156                summand,
157                *index,
158                lower_bound.as_ref(),
159                upper_bound.as_ref(),
160                with_respect_to,
161                memo,
162            )
163        }
164        ExpressionKind::IndexedProduct(multiplicand, index, lower_bound, upper_bound) => {
165            differentiate_indexed_product(
166                context,
167                multiplicand,
168                *index,
169                lower_bound.as_ref(),
170                upper_bound.as_ref(),
171                with_respect_to,
172                memo,
173            )
174        }
175        ExpressionKind::LaplaceTransform(_, _, _)
176        | ExpressionKind::InverseLaplaceTransform(_, _, _)
177        | ExpressionKind::FourierTransform(_, _, _)
178        | ExpressionKind::InverseFourierTransform(_, _, _) => {
179            differentiate_laplace_binder(context, target, with_respect_to)
180        }
181        ExpressionKind::Piecewise(branches) => {
182            differentiate_piecewise(context, target, branches, with_respect_to, memo)
183        }
184        ExpressionKind::OrdinaryDifferentialEquation(_, _, _)
185        | ExpressionKind::RecurrenceEquation(_, _, _) => {
186            // An unsolved ordinary_differential_equation or
187            // recurrence_equation placeholder is a symbolic problem
188            // statement rather than an analytic expression, so its
189            // derivative is not well defined. This placeholder
190            // discipline mirrors the integral rule: a refused
191            // differentiation returns the undefined sentinel so no
192            // heuristic guess is fabricated in place of a
193            // chain-rule placeholder.
194            intern_undefined(context)
195        }
196        ExpressionKind::Interval(_, _, _, _)
197        | ExpressionKind::FiniteSet(_)
198        | ExpressionKind::SetUnion(_)
199        | ExpressionKind::SetIntersection(_)
200        | ExpressionKind::SetComplement(_, _)
201        | ExpressionKind::ImageSet(_, _, _)
202        | ExpressionKind::ConditionSet(_, _, _)
203        | ExpressionKind::Distribution(_, _)
204        | ExpressionKind::RandomVariable(_, _)
205        | ExpressionKind::SpecialFunctionParameterVector(_) => {
206            // Set-tier expressions and probability-measure descriptors are not
207            // point-valued analytic objects; pointwise differentiation does not
208            // apply. Return the undefined sentinel under the same rigour
209            // contract that governs the ordinary_differential_equation /
210            // recurrence_equation arms. (The score-function derivative against
211            // a distribution parameter is a future-round extension.)
212            intern_undefined(context)
213        }
214    };
215    let published = memo.publish_if_absent(memo_key, result);
216    crate::instrumentation::record_counter!(expression_memo_publications);
217    published
218}

220fn differentiate_laplace_binder(
221    context: &Context,
222    target: &ExpressionReference,
223    with_respect_to: SymbolIdentifier,
224) -> ExpressionReference {
225    // Differentiate a laplace_transform or inverse_laplace_transform binder.
226    // The two transform variants both bind one variable internally, so a
227    // derivative with respect to that bound variable is identically zero
228    // (the transform result does not depend on the integration dummy).
229    // For any other differentiation variable, we cannot in general commute
230    // the derivative with the transform without consuming the Bromwich
231    // contour machinery (not yet implemented), so the result is the
232    // symbolic derivative placeholder over a user function "differentiate"
233    // — kept structurally so the chain rule can compose around it.
234    use crate::expression::ExpressionKind;
235    use crate::function::FunctionIdentifier;
236    match &target.kind {
237        ExpressionKind::LaplaceTransform(_, time_variable, _) => {
238            if *time_variable == with_respect_to {
239                return crate::expression::builders::zero(context);
240            }
241        }
242        ExpressionKind::InverseLaplaceTransform(_, frequency_variable, _) => {
243            if *frequency_variable == with_respect_to {
244                return crate::expression::builders::zero(context);
245            }
246        }
247        ExpressionKind::FourierTransform(_, time_variable, _) => {
248            if *time_variable == with_respect_to {
249                return crate::expression::builders::zero(context);
250            }
251        }
252        ExpressionKind::InverseFourierTransform(_, frequency_variable, _) => {
253            if *frequency_variable == with_respect_to {
254                return crate::expression::builders::zero(context);
255            }
256        }
257        _ => return intern_undefined(context),
258    }
259    let function_symbol = context.symbol_table.intern_symbol("derivative");
260    let function = FunctionIdentifier::User(function_symbol);
261    let variable_expression = context
262        .interner
263        .intern(ExpressionKind::Symbol(with_respect_to));
264    let order = crate::expression::builders::integer_value(context, 1);
265    crate::expression::builders::function_call(
266        context,
267        function,
268        vec![Arc::clone(target), variable_expression, order],
269    )
270}

272fn differentiate_sum(
273    context: &Context,
274    parent: &ExpressionReference,
275    children: &[ExpressionReference],
276    with_respect_to: SymbolIdentifier,
277    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
278) -> ExpressionReference {
279    // Linearity: the derivative of a sum is the sum of the derivatives. The
280    // result vector is handed to simplify_sum so that any zero terms vanish and
281    // any like terms collapse without an extra normalisation pass.
282    let derivatives =
283        memo.ordered_child_map(context, parent, children, "expression", |child, memo| {
284            differentiate_recursive(context, child, with_respect_to, memo)
285        });
286    simplify_sum(context, derivatives)
287}

289fn differentiate_product(
290    context: &Context,
291    parent: &ExpressionReference,
292    factors: &[ExpressionReference],
293    with_respect_to: SymbolIdentifier,
294    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
295) -> ExpressionReference {
296    // Generalised n-ary product rule: differentiating f_1 * f_2 * ... * f_n with
297    // respect to x yields the sum over i of (df_i / dx) * (product of all f_j for
298    // j != i). Each summand is constructed through the simplifying builders so
299    // that any zero derivative drops out via simplify_product collapse and any
300    // common factor across summands is recombined by the eventual simplify_sum.
301    let factor_derivatives =
302        memo.ordered_child_map(context, parent, factors, "expression", |factor, memo| {
303            differentiate_recursive(context, factor, with_respect_to, memo)
304        });
305    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(factors.len());
306    for (index, factor_derivative) in factor_derivatives.into_iter().enumerate() {
307        if is_integer_zero(&factor_derivative) {
308            continue;
309        }
310        let mut term_factors: Vec<ExpressionReference> = Vec::with_capacity(factors.len());
311        term_factors.push(factor_derivative);
312        for (other_index, other_factor) in factors.iter().enumerate() {
313            if other_index != index {
314                term_factors.push(Arc::clone(other_factor));
315            }
316        }
317        terms.push(simplify_product(context, term_factors));
318    }
319    if terms.is_empty() {
320        return intern_zero(context);
321    }
322    simplify_sum(context, terms)
323}

325fn differentiate_power(
326    context: &Context,
327    base: &ExpressionReference,
328    exponent: &ExpressionReference,
329    with_respect_to: SymbolIdentifier,
330    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
331) -> ExpressionReference {
332    // General power rule: d/dx u^v = u^v * (v' * log(u) + v * u' / u). Two
333    // shortcuts apply when one side is independent of x: a constant exponent
334    // collapses the expression to v * u^(v - 1) * u', and a constant base
335    // collapses it to u^v * log(u) * v'. The fully general formula is reserved
336    // for the case where both sides depend on x because it introduces a log node
337    // that the constant-exponent shortcut would otherwise leak unnecessarily.
338    let base_derivative = differentiate_recursive(context, base, with_respect_to, memo);
339    let exponent_derivative = differentiate_recursive(context, exponent, with_respect_to, memo);
340    let base_is_constant = is_integer_zero(&base_derivative);
341    let exponent_is_constant = is_integer_zero(&exponent_derivative);
342    if base_is_constant && exponent_is_constant {
343        return intern_zero(context);
344    }
345    if exponent_is_constant {
346        let one = intern_one(context);
347        let negated_one = negate_expression(context, one);
348        let new_exponent = simplify_sum(context, vec![Arc::clone(exponent), negated_one]);
349        let new_power = simplify_power(context, Arc::clone(base), new_exponent);
350        return simplify_product(
351            context,
352            vec![Arc::clone(exponent), new_power, base_derivative],
353        );
354    }
355    if base_is_constant {
356        let log_base = simplify_function_call(
357            context,
358            FunctionIdentifier::Builtin(BuiltinFunction::Log),
359            vec![Arc::clone(base)],
360        );
361        let pow_node = simplify_power(context, Arc::clone(base), Arc::clone(exponent));
362        return simplify_product(context, vec![pow_node, log_base, exponent_derivative]);
363    }
364    let log_base = simplify_function_call(
365        context,
366        FunctionIdentifier::Builtin(BuiltinFunction::Log),
367        vec![Arc::clone(base)],
368    );
369    let logarithmic_term = simplify_product(context, vec![exponent_derivative, log_base]);
370    let reciprocal_base = reciprocal_of(context, Arc::clone(base));
371    let chain_term = simplify_product(
372        context,
373        vec![Arc::clone(exponent), base_derivative, reciprocal_base],
374    );
375    let bracket = simplify_sum(context, vec![logarithmic_term, chain_term]);
376    let pow_node = simplify_power(context, Arc::clone(base), Arc::clone(exponent));
377    simplify_product(context, vec![pow_node, bracket])
378}

380fn differentiate_factorial(
381    context: &Context,
382    argument: &ExpressionReference,
383    with_respect_to: SymbolIdentifier,
384    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
385) -> ExpressionReference {
386    // Treat factorial as the analytic continuation gamma(x + 1). The chain rule
387    // therefore yields digamma(argument + 1) * factorial(argument) * argument'.
388    // When the argument does not depend on the variable of differentiation the
389    // chain term is zero and the entire derivative collapses, which keeps the
390    // common case d/dx factorial(integer_constant) = 0 free of any digamma
391    // residue.
392    let argument_derivative = differentiate_recursive(context, argument, with_respect_to, memo);
393    if is_integer_zero(&argument_derivative) {
394        return intern_zero(context);
395    }
396    let one = intern_one(context);
397    let argument_plus_one = simplify_sum(context, vec![Arc::clone(argument), one]);
398    let digamma_call = simplify_function_call(
399        context,
400        FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
401        vec![argument_plus_one],
402    );
403    let factorial_node = context
404        .interner
405        .intern(ExpressionKind::Factorial(Arc::clone(argument)));
406    simplify_product(
407        context,
408        vec![digamma_call, factorial_node, argument_derivative],
409    )
410}

412fn differentiate_function_call(
413    context: &Context,
414    parent: &ExpressionReference,
415    function: FunctionIdentifier,
416    arguments: &[ExpressionReference],
417    with_respect_to: SymbolIdentifier,
418    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
419) -> ExpressionReference {
420    // Apply the chain rule against the function's known derivative when one is
421    // available, otherwise fall back to emitting symbolic derivative placeholder
422    // nodes that future passes can inspect or rewrite. A single-argument builtin
423    // takes the fast path through builtin_unary_derivative; everything else —
424    // multi-argument builtins as well as user-defined functions — routes through
425    // the multivariate placeholder construction so that the chain rule remains
426    // sound across any function arity. The combinatorial builtins
427    // have no continuous extension implemented here (the gamma-routed analytic
428    // continuation lands with the gamma family), so they receive an early
429    // dedicated check: when every argument is independent of the differentiation
430    // variable the derivative is rigorously zero, and otherwise the call returns
431    // undefined rather than fabricate a placeholder.
432    if let FunctionIdentifier::Builtin(builtin) = function {
433        if is_combinatorial_builtin(builtin) {
434            for argument in arguments {
435                let argument_derivative =
436                    differentiate_recursive(context, argument, with_respect_to, memo);
437                if !is_integer_zero(&argument_derivative) {
438                    return intern_undefined(context);
439                }
440            }
441            return intern_zero(context);
442        }
443        if arguments.len() == 1 {
444            let argument = &arguments[0];
445            let argument_derivative =
446                differentiate_recursive(context, argument, with_respect_to, memo);
447            if is_integer_zero(&argument_derivative) {
448                return intern_zero(context);
449            }
450            if let Some(inner) = builtin_unary_derivative(context, builtin, argument) {
451                return simplify_product(context, vec![inner, argument_derivative]);
452            }
453        }
454        if builtin == BuiltinFunction::HypergeometricPfq {
455            if let Some(derivative) =
456                hypergeometric_argument_derivative(context, arguments, with_respect_to, memo)
457            {
458                return derivative;
459            }
460        }
461        if builtin == BuiltinFunction::MeijerG {
462            return undifferentiated_call_placeholder(context, function, arguments);
463        }
464        if builtin == BuiltinFunction::LerchPhi {
465            if let Some(derivative) =
466                lerch_phi_derivative(context, arguments, with_respect_to, memo)
467            {
468                return derivative;
469            }
470        }
471        if builtin == BuiltinFunction::CarlsonRf && arguments.len() == 3 {
472            let argument_derivatives = arguments
473                .iter()
474                .map(|argument| differentiate_recursive(context, argument, with_respect_to, memo))
475                .collect::<Vec<_>>();
476            if let Some(derivative) = crate::special_functions::carlson_elliptic::differentiate_call(
477                context,
478                builtin,
479                arguments,
480                &argument_derivatives,
481            ) {
482                return derivative;
483            }
484        }
485        if builtin == BuiltinFunction::EllipticPiIncomplete && arguments.len() == 3 {
486            let characteristic_derivative =
487                differentiate_recursive(context, &arguments[0], with_respect_to, memo);
488            let amplitude_derivative =
489                differentiate_recursive(context, &arguments[1], with_respect_to, memo);
490            let parameter_derivative =
491                differentiate_recursive(context, &arguments[2], with_respect_to, memo);
492            if is_integer_zero(&characteristic_derivative) && is_integer_zero(&parameter_derivative)
493            {
494                if is_integer_zero(&amplitude_derivative) {
495                    return intern_zero(context);
496                }
497                let sine = simplify_function_call(
498                    context,
499                    FunctionIdentifier::Builtin(BuiltinFunction::Sin),
500                    vec![Arc::clone(&arguments[1])],
501                );
502                let two = integer_value(context, 2);
503                let sine_square = simplify_power(context, sine, two);
504                let characteristic_product = simplify_product(
505                    context,
506                    vec![Arc::clone(&arguments[0]), sine_square.clone()],
507                );
508                let parameter_product =
509                    simplify_product(context, vec![Arc::clone(&arguments[2]), sine_square]);
510                let first_one = intern_one(context);
511                let negative_characteristic_product =
512                    negate_expression(context, characteristic_product);
513                let first_factor =
514                    simplify_sum(context, vec![first_one, negative_characteristic_product]);
515                let radicand_one = intern_one(context);
516                let negative_parameter_product = negate_expression(context, parameter_product);
517                let radicand =
518                    simplify_sum(context, vec![radicand_one, negative_parameter_product]);
519                let half = rational(context, Integer::ONE, Integer::from(2));
520                let square_root = simplify_power(context, radicand, half);
521                let denominator = simplify_product(context, vec![first_factor, square_root]);
522                let reciprocal = reciprocal_of(context, denominator);
523                return simplify_product(context, vec![reciprocal, amplitude_derivative]);
524            }
525        }
526        if matches!(
527            builtin,
528            BuiltinFunction::AiryAi
529                | BuiltinFunction::AiryBi
530                | BuiltinFunction::AiryAiPrime
531                | BuiltinFunction::AiryBiPrime
532        ) {
533            let argument = &arguments[0];
534            let argument_derivative =
535                differentiate_recursive(context, argument, with_respect_to, memo);
536            if is_integer_zero(&argument_derivative) {
537                return intern_zero(context);
538            }
539            if let Some(inner) = crate::special_functions::airy::argument_derivative(
540                context,
541                builtin,
542                Arc::clone(argument),
543            ) {
544                return simplify_product(context, vec![inner, argument_derivative]);
545            }
546        }
547        if crate::special_functions::orthogonal_polynomials::family_of(builtin).is_some() {
548            if let Some(derivative) = orthogonal_polynomial_argument_derivative(
549                context,
550                builtin,
551                arguments,
552                with_respect_to,
553                memo,
554            ) {
555                return derivative;
556            }
557        }
558        if arguments.len() == 2 {
559            if let Some(derivative) =
560                builtin_binary_derivative(context, builtin, arguments, with_respect_to, memo)
561            {
562                return derivative;
563            }
564        }
565    }
566    emit_chain_rule_via_placeholder(context, parent, function, arguments, with_respect_to, memo)
567}

569fn lerch_phi_derivative(
570    context: &Context,
571    arguments: &[ExpressionReference],
572    with_respect_to: SymbolIdentifier,
573    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
574) -> Option<ExpressionReference> {
575    let (argument, order, shift) = (arguments.first()?, arguments.get(1)?, arguments.get(2)?);
576    let order_derivative = differentiate_recursive(context, order, with_respect_to, memo);
577    if !is_integer_zero(&order_derivative) {
578        return None;
579    }
580    let argument_derivative = differentiate_recursive(context, argument, with_respect_to, memo);
581    let shift_derivative = differentiate_recursive(context, shift, with_respect_to, memo);
582    if is_integer_zero(&argument_derivative) && is_integer_zero(&shift_derivative) {
583        return Some(intern_zero(context));
584    }
585    let mut terms = Vec::with_capacity(2);
586    if !is_integer_zero(&argument_derivative) {
587        let argument_partial = crate::special_functions::lerch_phi::differentiate_argument(
588            context, argument, order, shift,
589        );
590        terms.push(simplify_product(
591            context,
592            vec![argument_partial, argument_derivative],
593        ));
594    }
595    if !is_integer_zero(&shift_derivative) {
596        let shift_partial = crate::special_functions::lerch_phi::differentiate_shift(
597            context, argument, order, shift,
598        );
599        terms.push(simplify_product(
600            context,
601            vec![shift_partial, shift_derivative],
602        ));
603    }
604    Some(simplify_sum(context, terms))
605}

607fn orthogonal_polynomial_argument_derivative(
608    context: &Context,
609    builtin: BuiltinFunction,
610    arguments: &[ExpressionReference],
611    with_respect_to: SymbolIdentifier,
612    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
613) -> Option<ExpressionReference> {
614    // Chain-rule the DLMF 18.9 argument derivative of a classical orthogonal
615    // polynomial. The degree and any family parameters are ordinary slots that
616    // may themselves depend on the variable; closed-form differentiation in
617    // those slots is not yet implemented, so a moving degree or parameter
618    // declines here and the ordinary multivariate placeholder path takes over
619    // rather than silently treating the slot as constant.
620    let family = crate::special_functions::orthogonal_polynomials::family_of(builtin)?;
621    if arguments.len() != family.arity() {
622        return None;
623    }
624    for slot in &arguments[..family.analytic_argument_index()] {
625        let slot_derivative = differentiate_recursive(context, slot, with_respect_to, memo);
626        if !is_integer_zero(&slot_derivative) {
627            return None;
628        }
629    }
630    let argument = &arguments[family.analytic_argument_index()];
631    let argument_derivative = differentiate_recursive(context, argument, with_respect_to, memo);
632    if is_integer_zero(&argument_derivative) {
633        return Some(intern_zero(context));
634    }
635    let inner = crate::special_functions::orthogonal_polynomials::argument_derivative(
636        context, builtin, arguments,
637    )?;
638    Some(simplify_product(context, vec![inner, argument_derivative]))
639}

641fn hypergeometric_argument_derivative(
642    context: &Context,
643    arguments: &[ExpressionReference],
644    with_respect_to: SymbolIdentifier,
645    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
646) -> Option<ExpressionReference> {
647    // Chain-rule the DLMF 16.3.1 argument-derivative identity for a
648    // generalized hypergeometric call. Parameter differentiation is not yet
649    // implemented, so a parameter group that mentions the differentiation
650    // variable routes to the undifferentiated placeholder instead of receiving
651    // the argument-only formula, which is not valid when a parameter varies.
652    let (upper_group, lower_group, argument) =
653        (arguments.first()?, arguments.get(1)?, arguments.get(2)?);
654    let parameters_vary = [upper_group, lower_group].into_iter().any(|group| {
655        crate::expression::free_variables::free_variables(group).contains(&with_respect_to)
656    });
657    if parameters_vary {
658        return Some(undifferentiated_call_placeholder(
659            context,
660            FunctionIdentifier::Builtin(BuiltinFunction::HypergeometricPfq),
661            arguments,
662        ));
663    }
664    let argument_derivative = differentiate_recursive(context, argument, with_respect_to, memo);
665    if is_integer_zero(&argument_derivative) {
666        return Some(intern_zero(context));
667    }
668    let inner = crate::special_functions::hypergeometric::argument_derivative(context, arguments)?;
669    Some(simplify_product(context, vec![inner, argument_derivative]))
670}

672fn undifferentiated_call_placeholder(
673    context: &Context,
674    function: FunctionIdentifier,
675    arguments: &[ExpressionReference],
676) -> ExpressionReference {
677    // Emit the derivative placeholder that marks a call whose derivative was
678    // not carried out. The ordinary multivariate placeholder sum cannot be used
679    // for the generalized hypergeometric and Meijer-G heads, because their
680    // parameter slots hold ordered groups rather than scalars and there is no
681    // scalar inner derivative to multiply a partial by. The bare placeholder
682    // asserts no value at all, which is what makes it safe here: it records
683    // that the operation declined rather than reporting a result.
684    let orders = vec![0usize; arguments.len()];
685    let cloned_arguments: Vec<ExpressionReference> = arguments.iter().map(Arc::clone).collect();
686    context.interner.intern(ExpressionKind::Derivative(
687        function,
688        orders,
689        cloned_arguments,
690    ))
691}

693fn builtin_binary_derivative(
694    context: &Context,
695    builtin: BuiltinFunction,
696    arguments: &[ExpressionReference],
697    with_respect_to: SymbolIdentifier,
698    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
699) -> Option<ExpressionReference> {
700    // Closed-form derivatives of the two-argument gamma-family builtins.
701    //   d/dx polygamma(m, z) = polygamma(m + 1, z) * z'   (the order m is constant)
702    //   d/dx beta(x, y) = beta(x, y) [ (psi(x) - psi(x+y)) x'
703    //                                  + (psi(y) - psi(x+y)) y' ]
704    //   d/dx hurwitz_zeta(s, a) = -s * hurwitz_zeta(s + 1, a) * a'   (the s-derivative
705    //                                  of zeta is non-elementary, so any dependence of s
706    //                                  on the variable declines to the placeholder path)
707    //   d/dx lambert_w(k, z) = W_k(z) / (z (1 + W_k(z))) * z'   (the branch index k is
708    //                                  a discrete label, so any dependence of k on the
709    //                                  variable declines to the placeholder path)
710    // Declines (None) when the discrete order argument itself depends on the
711    // variable, leaving the multivariate placeholder path to handle it.
712    match builtin {
713        BuiltinFunction::EllipticF | BuiltinFunction::EllipticEIncomplete => {
714            let amplitude = &arguments[0];
715            let parameter = &arguments[1];
716            let parameter_derivative =
717                differentiate_recursive(context, parameter, with_respect_to, memo);
718            if !is_integer_zero(&parameter_derivative) {
719                return None;
720            }
721            let amplitude_derivative =
722                differentiate_recursive(context, amplitude, with_respect_to, memo);
723            if is_integer_zero(&amplitude_derivative) {
724                return Some(intern_zero(context));
725            }
726            let sine = simplify_function_call(
727                context,
728                FunctionIdentifier::Builtin(BuiltinFunction::Sin),
729                vec![Arc::clone(amplitude)],
730            );
731            let two = integer_value(context, 2);
732            let sine_square = simplify_power(context, sine, two);
733            let parameter_product =
734                simplify_product(context, vec![Arc::clone(parameter), sine_square]);
735            let one = intern_one(context);
736            let negated_parameter_product = negate_expression(context, parameter_product);
737            let radicand = simplify_sum(context, vec![one, negated_parameter_product]);
738            let half = rational(context, Integer::ONE, Integer::from(2));
739            let square_root = simplify_power(context, radicand, half);
740            let inner = if builtin == BuiltinFunction::EllipticF {
741                reciprocal_of(context, square_root)
742            } else {
743                square_root
744            };
745            Some(simplify_product(context, vec![inner, amplitude_derivative]))
746        }
747        BuiltinFunction::JacobiSn
748        | BuiltinFunction::JacobiCn
749        | BuiltinFunction::JacobiDn
750        | BuiltinFunction::JacobiAmplitude => {
751            let parameter_derivative =
752                differentiate_recursive(context, &arguments[1], with_respect_to, memo);
753            if !is_integer_zero(&parameter_derivative) {
754                return None;
755            }
756            let argument_derivative =
757                differentiate_recursive(context, &arguments[0], with_respect_to, memo);
758            if is_integer_zero(&argument_derivative) {
759                return Some(intern_zero(context));
760            }
761            let inner = crate::special_functions::jacobi_elliptic::differentiate_call(
762                context, builtin, arguments,
763            )?;
764            Some(simplify_product(context, vec![inner, argument_derivative]))
765        }
766        BuiltinFunction::EllipticPi => {
767            let characteristic = &arguments[0];
768            let parameter = &arguments[1];
769            let characteristic_derivative =
770                differentiate_recursive(context, characteristic, with_respect_to, memo);
771            let parameter_derivative =
772                differentiate_recursive(context, parameter, with_respect_to, memo);
773            if is_integer_zero(&characteristic_derivative) && is_integer_zero(&parameter_derivative)
774            {
775                return Some(intern_zero(context));
776            }
777            None
778        }
779        BuiltinFunction::LambertW => {
780            let branch = &arguments[0];
781            let inner_argument = &arguments[1];
782            let branch_derivative = differentiate_recursive(context, branch, with_respect_to, memo);
783            if !is_integer_zero(&branch_derivative) {
784                return None;
785            }
786            let argument_derivative =
787                differentiate_recursive(context, inner_argument, with_respect_to, memo);
788            if is_integer_zero(&argument_derivative) {
789                return Some(intern_zero(context));
790            }
791            let inner =
792                crate::special_functions::lambert_w::differentiate_call(context, arguments)?;
793            Some(simplify_product(context, vec![inner, argument_derivative]))
794        }
795        BuiltinFunction::BesselJ
796        | BuiltinFunction::BesselY
797        | BuiltinFunction::BesselI
798        | BuiltinFunction::BesselK
799        | BuiltinFunction::HankelFirstKind
800        | BuiltinFunction::HankelSecondKind => {
801            let order_derivative =
802                differentiate_recursive(context, &arguments[0], with_respect_to, memo);
803            if !is_integer_zero(&order_derivative) {
804                return None;
805            }
806            let argument_derivative =
807                differentiate_recursive(context, &arguments[1], with_respect_to, memo);
808            if is_integer_zero(&argument_derivative) {
809                return Some(intern_zero(context));
810            }
811            let inner = crate::special_functions::bessel::argument_derivative(
812                context,
813                builtin,
814                Arc::clone(&arguments[0]),
815                Arc::clone(&arguments[1]),
816            )?;
817            Some(simplify_product(context, vec![inner, argument_derivative]))
818        }
819        BuiltinFunction::SphericalBesselJ
820        | BuiltinFunction::SphericalBesselY
821        | BuiltinFunction::SphericalHankelFirstKind
822        | BuiltinFunction::SphericalHankelSecondKind => {
823            let order_derivative =
824                differentiate_recursive(context, &arguments[0], with_respect_to, memo);
825            if !is_integer_zero(&order_derivative) {
826                return None;
827            }
828            let argument_derivative =
829                differentiate_recursive(context, &arguments[1], with_respect_to, memo);
830            if is_integer_zero(&argument_derivative) {
831                return Some(intern_zero(context));
832            }
833            let inner = crate::special_functions::bessel::argument_derivative(
834                context,
835                builtin,
836                Arc::clone(&arguments[0]),
837                Arc::clone(&arguments[1]),
838            )?;
839            Some(simplify_product(context, vec![inner, argument_derivative]))
840        }
841        BuiltinFunction::HurwitzZeta => {
842            let s = &arguments[0];
843            let a = &arguments[1];
844            let s_derivative = differentiate_recursive(context, s, with_respect_to, memo);
845            if !is_integer_zero(&s_derivative) {
846                return None;
847            }
848            let a_derivative = differentiate_recursive(context, a, with_respect_to, memo);
849            if is_integer_zero(&a_derivative) {
850                return Some(intern_zero(context));
851            }
852            let one = intern_one(context);
853            let next_s = simplify_sum(context, vec![Arc::clone(s), one]);
854            let next_zeta = simplify_function_call(
855                context,
856                FunctionIdentifier::Builtin(BuiltinFunction::HurwitzZeta),
857                vec![next_s, Arc::clone(a)],
858            );
859            let negative_s = negate_expression(context, Arc::clone(s));
860            Some(simplify_product(
861                context,
862                vec![negative_s, next_zeta, a_derivative],
863            ))
864        }
865        BuiltinFunction::Polylogarithm => {
866            let order = &arguments[0];
867            let argument = &arguments[1];
868            let order_derivative = differentiate_recursive(context, order, with_respect_to, memo);
869            if !is_integer_zero(&order_derivative) {
870                return None;
871            }
872            let argument_derivative =
873                differentiate_recursive(context, argument, with_respect_to, memo);
874            if is_integer_zero(&argument_derivative) {
875                return Some(intern_zero(context));
876            }
877            let inner = crate::special_functions::polylogarithm::differentiate_argument(
878                context, order, argument,
879            );
880            Some(simplify_product(context, vec![inner, argument_derivative]))
881        }
882        BuiltinFunction::Polygamma => {
883            let order = &arguments[0];
884            let inner_argument = &arguments[1];
885            let order_derivative = differentiate_recursive(context, order, with_respect_to, memo);
886            if !is_integer_zero(&order_derivative) {
887                return None;
888            }
889            let argument_derivative =
890                differentiate_recursive(context, inner_argument, with_respect_to, memo);
891            if is_integer_zero(&argument_derivative) {
892                return Some(intern_zero(context));
893            }
894            let one = intern_one(context);
895            let next_order = simplify_sum(context, vec![Arc::clone(order), one]);
896            let next_polygamma = simplify_function_call(
897                context,
898                FunctionIdentifier::Builtin(BuiltinFunction::Polygamma),
899                vec![next_order, Arc::clone(inner_argument)],
900            );
901            Some(simplify_product(
902                context,
903                vec![next_polygamma, argument_derivative],
904            ))
905        }
906        BuiltinFunction::Beta => {
907            let first = &arguments[0];
908            let second = &arguments[1];
909            let first_derivative = differentiate_recursive(context, first, with_respect_to, memo);
910            let second_derivative = differentiate_recursive(context, second, with_respect_to, memo);
911            if is_integer_zero(&first_derivative) && is_integer_zero(&second_derivative) {
912                return Some(intern_zero(context));
913            }
914            let beta_call = simplify_function_call(
915                context,
916                FunctionIdentifier::Builtin(BuiltinFunction::Beta),
917                vec![Arc::clone(first), Arc::clone(second)],
918            );
919            let sum = simplify_sum(context, vec![Arc::clone(first), Arc::clone(second)]);
920            let digamma_sum = simplify_function_call(
921                context,
922                FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
923                vec![sum],
924            );
925            let mut bracket_terms: Vec<ExpressionReference> = Vec::new();
926            if !is_integer_zero(&first_derivative) {
927                let digamma_first = simplify_function_call(
928                    context,
929                    FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
930                    vec![Arc::clone(first)],
931                );
932                let negated_sum = negate_expression(context, Arc::clone(&digamma_sum));
933                let factor = simplify_sum(context, vec![digamma_first, negated_sum]);
934                bracket_terms.push(simplify_product(context, vec![factor, first_derivative]));
935            }
936            if !is_integer_zero(&second_derivative) {
937                let digamma_second = simplify_function_call(
938                    context,
939                    FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
940                    vec![Arc::clone(second)],
941                );
942                let negated_sum = negate_expression(context, Arc::clone(&digamma_sum));
943                let factor = simplify_sum(context, vec![digamma_second, negated_sum]);
944                bracket_terms.push(simplify_product(context, vec![factor, second_derivative]));
945            }
946            let bracket = simplify_sum(context, bracket_terms);
947            Some(simplify_product(context, vec![beta_call, bracket]))
948        }
949        _ => None,
950    }
951}

953fn is_combinatorial_builtin(builtin: BuiltinFunction) -> bool {
954    // Membership predicate for the combinatorial family. The
955    // family is closed under an integer-only domain: any
956    // argument-dependent differentiation against any member of this set
957    // refuses (returns undefined). The continuous
958    // extensions via gamma will lift the refusal in a later round.
959    matches!(
960        builtin,
961        BuiltinFunction::Binomial
962            | BuiltinFunction::FallingFactorial
963            | BuiltinFunction::RisingFactorial
964            | BuiltinFunction::StirlingFirst
965            | BuiltinFunction::StirlingSecond
966            | BuiltinFunction::BellNumber
967            | BuiltinFunction::CatalanNumber
968            | BuiltinFunction::FibonacciNumber
969            | BuiltinFunction::PartitionCount
970    )
971}

982fn differentiate_derivative_node(
983    context: &Context,
984    parent: &ExpressionReference,
985    function: FunctionIdentifier,
986    orders: &[usize],
987    arguments: &[ExpressionReference],
988    with_respect_to: SymbolIdentifier,
989    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
990) -> ExpressionReference {
991    // Multivariate chain rule on an existing derivative placeholder. For each
992    // argument slot whose derivative is non-zero, raise the corresponding order
993    // by one and multiply by the inner derivative. The result is the sum of
994    // these per-slot contributions, mirroring the expansion of d/dx D_alpha f
995    // where alpha is the existing multi-index.
996    let argument_derivatives = memo.ordered_child_map(
997        context,
998        parent,
999        arguments,
1000        "expression",
1001        |argument, memo| differentiate_recursive(context, argument, with_respect_to, memo),
1002    );
1003    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(arguments.len());
1004    for (index, argument_derivative) in argument_derivatives.into_iter().enumerate() {
1005        if is_integer_zero(&argument_derivative) {
1006            continue;
1007        }
1008        let mut bumped_orders: Vec<usize> = orders.to_vec();
1009        bumped_orders[index] += 1;
1010        let bumped_arguments: Vec<ExpressionReference> = arguments.iter().map(Arc::clone).collect();
1011        let bumped_derivative = context.interner.intern(ExpressionKind::Derivative(
1012            function,
1013            bumped_orders,
1014            bumped_arguments,
1015        ));
1016        terms.push(simplify_product(
1017            context,
1018            vec![bumped_derivative, argument_derivative],
1019        ));
1020    }
1021    if terms.is_empty() {
1022        return intern_zero(context);
1023    }
1024    simplify_sum(context, terms)
1025}

1027fn differentiate_integral(
1028    context: &Context,
1029    integrand: &ExpressionReference,
1030    variable: SymbolIdentifier,
1031    lower_bound: Option<&ExpressionReference>,
1032    upper_bound: Option<&ExpressionReference>,
1033    with_respect_to: SymbolIdentifier,
1034    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
1035) -> ExpressionReference {
1036    // Differentiate an integral node. Three cases are recognised, all of them
1037    // mathematically rigorous: differentiating an indefinite integral with
1038    // respect to its bound variable applies the fundamental theorem of
1039    // calculus and returns the integrand directly; differentiating a definite
1040    // integral with respect to a parameter that is not the bound variable
1041    // applies the Leibniz integral rule and produces the integral of the
1042    // partial derivative plus the boundary contributions; differentiating an
1043    // indefinite integral with respect to a parameter that is not the bound
1044    // variable cannot in general be reduced — the FTC argument does not extend
1045    // without bounds — so we emit an unevaluated derivative placeholder that
1046    // future passes can inspect or further reduce.
1047    let is_indefinite = lower_bound.is_none() && upper_bound.is_none();
1048    if is_indefinite && variable == with_respect_to {
1049        return Arc::clone(integrand);
1050    }
1051    if is_indefinite {
1052        return emit_indefinite_integral_derivative_placeholder(
1053            context,
1054            integrand,
1055            variable,
1056            with_respect_to,
1057        );
1058    }
1059    let lower = lower_bound.unwrap();
1060    let upper = upper_bound.unwrap();
1061    let integrand_partial = differentiate_recursive(context, integrand, with_respect_to, memo);
1062    let integral_of_partial = crate::expression::builders::integral(
1063        context,
1064        integrand_partial,
1065        variable,
1066        Some(Arc::clone(lower)),
1067        Some(Arc::clone(upper)),
1068    );
1069    let upper_derivative = differentiate_recursive(context, upper, with_respect_to, memo);
1070    let lower_derivative = differentiate_recursive(context, lower, with_respect_to, memo);
1071    let mut substitution_upper: HashMap<ExpressionReference, ExpressionReference> = HashMap::new();
1072    let bound_symbol = context.interner.intern(ExpressionKind::Symbol(variable));
1073    substitution_upper.insert(Arc::clone(&bound_symbol), Arc::clone(upper));
1074    let integrand_at_upper =
1075        crate::expression::substitute::substitute(context, integrand, &substitution_upper);
1076    let mut substitution_lower: HashMap<ExpressionReference, ExpressionReference> = HashMap::new();
1077    substitution_lower.insert(Arc::clone(&bound_symbol), Arc::clone(lower));
1078    let integrand_at_lower =
1079        crate::expression::substitute::substitute(context, integrand, &substitution_lower);
1080    let upper_contribution = simplify_product(context, vec![integrand_at_upper, upper_derivative]);
1081    let lower_contribution = simplify_product(context, vec![integrand_at_lower, lower_derivative]);
1082    let negated_lower_contribution = negate_expression(context, lower_contribution);
1083    simplify_sum(
1084        context,
1085        vec![
1086            integral_of_partial,
1087            upper_contribution,
1088            negated_lower_contribution,
1089        ],
1090    )
1091}

1093fn differentiate_summation(
1094    context: &Context,
1095    summand: &ExpressionReference,
1096    index: SymbolIdentifier,
1097    lower_bound: Option<&ExpressionReference>,
1098    upper_bound: Option<&ExpressionReference>,
1099    with_respect_to: SymbolIdentifier,
1100    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
1101) -> ExpressionReference {
1102    // Differentiate a summation node. For parameters distinct from the
1103    // summation index the Leibniz rule for sums applies: the derivative of
1104    // a finite sum is the sum of derivatives. The boundary contributions
1105    // that show up in the integral analogue are absent here because the
1106    // sum's discrete bounds shift the indexing rather than feed a
1107    // continuous integrand into a boundary substitution; differentiation
1108    // with respect to a bound is not generally a well-defined operation on
1109    // a discrete sum and is refused. Differentiation with respect to the
1110    // bound index itself is rejected — the bound index is not a
1111    // free variable so its derivative carries no meaning.
1112    if index == with_respect_to {
1113        return intern_undefined(context);
1114    }
1115    if let Some(lower) = lower_bound {
1116        if free_variables_of(lower).contains(&with_respect_to) {
1117            return intern_undefined(context);
1118        }
1119    }
1120    if let Some(upper) = upper_bound {
1121        if free_variables_of(upper).contains(&with_respect_to) {
1122            return intern_undefined(context);
1123        }
1124    }
1125    let summand_derivative = differentiate_recursive(context, summand, with_respect_to, memo);
1126    crate::expression::builders::summation(
1127        context,
1128        summand_derivative,
1129        index,
1130        lower_bound.map(Arc::clone),
1131        upper_bound.map(Arc::clone),
1132    )
1133}

1135fn differentiate_indexed_product(
1136    context: &Context,
1137    multiplicand: &ExpressionReference,
1138    index: SymbolIdentifier,
1139    lower_bound: Option<&ExpressionReference>,
1140    upper_bound: Option<&ExpressionReference>,
1141    with_respect_to: SymbolIdentifier,
1142    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
1143) -> ExpressionReference {
1144    // Differentiate an indexed_product by logarithmic differentiation. For a
1145    // parameter distinct from the bound index the identity
1146    //   d/dx ∏_k f(k, x) = (∏_k f(k, x)) · Σ_k (∂_x f / f)
1147    // holds: differentiating log ∏ = Σ log f term by term and multiplying back
1148    // by the product itself. The derivative carries an embedded summation node
1149    // alongside the product, both over the same range. Differentiation with
1150    // respect to the bound index itself, or against an index-dependent bound, is
1151    // not well defined on a discrete product and is refused with the undefined
1152    // sentinel, matching differentiate_summation. The indefinite (anti-quotient)
1153    // case has no matching finite Σ over which the identity reads, so it is also
1154    // refused.
1155    if index == with_respect_to {
1156        return intern_undefined(context);
1157    }
1158    let (lower, upper) = match (lower_bound, upper_bound) {
1159        (Some(lower), Some(upper)) => (lower, upper),
1160        _ => return intern_undefined(context),
1161    };
1162    if free_variables_of(lower).contains(&with_respect_to)
1163        || free_variables_of(upper).contains(&with_respect_to)
1164    {
1165        return intern_undefined(context);
1166    }
1167    let multiplicand_derivative =
1168        differentiate_recursive(context, multiplicand, with_respect_to, memo);
1169    let minus_one = crate::expression::builders::integer_value(context, -1);
1170    let reciprocal_multiplicand =
1171        crate::expression::builders::power(context, Arc::clone(multiplicand), minus_one);
1172    let logarithmic_summand = crate::expression::builders::product(
1173        context,
1174        vec![multiplicand_derivative, reciprocal_multiplicand],
1175    );
1176    let logarithmic_sum = crate::expression::builders::summation(
1177        context,
1178        logarithmic_summand,
1179        index,
1180        Some(Arc::clone(lower)),
1181        Some(Arc::clone(upper)),
1182    );
1183    let product_node = crate::expression::builders::indexed_product(
1184        context,
1185        Arc::clone(multiplicand),
1186        index,
1187        Some(Arc::clone(lower)),
1188        Some(Arc::clone(upper)),
1189    );
1190    crate::expression::builders::product(context, vec![product_node, logarithmic_sum])
1191}

1200fn differentiate_piecewise(
1201    context: &Context,
1202    parent: &ExpressionReference,
1203    branches: &[(ExpressionReference, ExpressionReference)],
1204    with_respect_to: SymbolIdentifier,
1205    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
1206) -> ExpressionReference {
1207    // Differentiate a piecewise expression. The derivative is taken inside each
1208    // branch's value while the branch's condition is left unchanged: the
1209    // condition selects which open interval applies and is not itself a
1210    // function of the differentiation variable. The result is mathematically
1211    // valid on the open intervals each branch denotes — non-differentiability
1212    // at the boundary between branches inherits the same abs/signum
1213    // discipline used elsewhere and is documented at the call site rather
1214    // than guarded at runtime.
1215    let values: Vec<ExpressionReference> = branches
1216        .iter()
1217        .map(|(_, value)| Arc::clone(value))
1218        .collect();
1219    let derivatives =
1220        memo.ordered_child_map(context, parent, &values, "expression", |value, memo| {
1221            differentiate_recursive(context, value, with_respect_to, memo)
1222        });
1223    let rebuilt: Vec<(ExpressionReference, ExpressionReference)> = branches
1224        .iter()
1225        .zip(derivatives)
1226        .map(|((condition, _), derivative)| (Arc::clone(condition), derivative))
1227        .collect();
1228    crate::simplify::simplify_piecewise(context, rebuilt)
1229}

1231fn emit_indefinite_integral_derivative_placeholder(
1232    context: &Context,
1233    integrand: &ExpressionReference,
1234    variable: SymbolIdentifier,
1235    with_respect_to: SymbolIdentifier,
1236) -> ExpressionReference {
1237    // Build the symbolic derivative placeholder for an indefinite integral
1238    // differentiated with respect to a parameter other than its bound
1239    // variable. The placeholder is encoded as a user-function call whose name
1240    // includes the bound variable so that subsequent passes can tell the
1241    // unevaluated derivative apart from the integrand itself. The construction
1242    // intentionally avoids producing a syntactic answer because the FTC does
1243    // not apply: the underlying primitive may depend on the parameter through
1244    // a constant of integration that the integral notation hides.
1245    let symbol_name = context.symbol_table.name(variable).to_string();
1246    let placeholder_name = format!("indefinite_integral_derivative_{}", symbol_name);
1247    let function_identifier_value = crate::function::FunctionIdentifier::User(
1248        context.symbol_table.intern_symbol(&placeholder_name),
1249    );
1250    let placeholder_integral = context.interner.intern(ExpressionKind::Integral(
1251        Arc::clone(integrand),
1252        variable,
1253        None,
1254        None,
1255    ));
1256    let parameter_expression = context
1257        .interner
1258        .intern(ExpressionKind::Symbol(with_respect_to));
1259    context.interner.intern(ExpressionKind::Derivative(
1260        function_identifier_value,
1261        vec![1, 0],
1262        vec![parameter_expression, placeholder_integral],
1263    ))
1264}

1266fn emit_chain_rule_via_placeholder(
1267    context: &Context,
1268    parent: &ExpressionReference,
1269    function: FunctionIdentifier,
1270    arguments: &[ExpressionReference],
1271    with_respect_to: SymbolIdentifier,
1272    memo: &mut ExpressionTraversalMemo<'_, ExpressionReference>,
1273) -> ExpressionReference {
1274    // Construct the multivariate chain rule sum over derivative placeholders. Each
1275    // argument slot contributes a term D_i f * (d argument_i / dx); arguments
1276    // whose derivative is zero are dropped from the sum. The base orders vector
1277    // is filled with zeros and incremented at the slot for each contribution so
1278    // that the placeholder's interpretation matches the partial-index convention.
1279    let argument_derivatives = memo.ordered_child_map(
1280        context,
1281        parent,
1282        arguments,
1283        "expression",
1284        |argument, memo| differentiate_recursive(context, argument, with_respect_to, memo),
1285    );
1286    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(arguments.len());
1287    for (index, argument_derivative) in argument_derivatives.into_iter().enumerate() {
1288        if is_integer_zero(&argument_derivative) {
1289            continue;
1290        }
1291        let mut orders = vec![0usize; arguments.len()];
1292        orders[index] = 1;
1293        let cloned_arguments: Vec<ExpressionReference> = arguments.iter().map(Arc::clone).collect();
1294        let placeholder = context.interner.intern(ExpressionKind::Derivative(
1295            function,
1296            orders,
1297            cloned_arguments,
1298        ));
1299        terms.push(simplify_product(
1300            context,
1301            vec![placeholder, argument_derivative],
1302        ));
1303    }
1304    if terms.is_empty() {
1305        return intern_zero(context);
1306    }
1307    simplify_sum(context, terms)
1308}

1738fn error_function_derivative_factor(
1739    context: &Context,
1740    argument: &ExpressionReference,
1741) -> ExpressionReference {
1742    // Construct the chain-rule factor (2/sqrt(pi)) * exp(-x^2) shared between
1743    // the derivatives of error_function and error_function_complementary. The
1744    // pi term goes through the builtin_constant variant and the sqrt through
1745    // the simplifying sqrt builder so the result is canonical.
1746    let two = integer_value(context, 2);
1747    let argument_squared = simplify_power(context, Arc::clone(argument), Arc::clone(&two));
1748    let negated_square = negate_expression(context, argument_squared);
1749    let exp_call = simplify_function_call(
1750        context,
1751        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
1752        vec![negated_square],
1753    );
1754    let pi_constant = context.interner.intern(ExpressionKind::Constant(
1755        crate::expression::BuiltinConstant::Pi,
1756    ));
1757    let sqrt_pi = simplify_function_call(
1758        context,
1759        FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
1760        vec![pi_constant],
1761    );
1762    let inverse_sqrt_pi = reciprocal_of(context, sqrt_pi);
1763    simplify_product(context, vec![two, inverse_sqrt_pi, exp_call])
1764}

1768    argument: ExpressionReference,
1769) -> ExpressionReference {
1770    // Construct sqrt(1 - argument^2), the canonical denominator that appears in
1771    // the derivatives of arcsin and arccos. Built once here to avoid repeating
1772    // the expression construction inside two derivative-table entries.
1773    let two = integer_value(context, 2);
1774    let argument_squared = simplify_power(context, argument, two);
1775    let one = intern_one(context);
1776    let negated_square = negate_expression(context, argument_squared);
1777    let bracket = simplify_sum(context, vec![one, negated_square]);
1778    simplify_function_call(
1779        context,
1780        FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
1781        vec![bracket],
1782    )
1783}

1821    // Build argument^(-1) through the simplify_power builder. Used as a building
1822    // block whenever a derivative formula divides by an existing expression.
1823    let minus_one = integer(context, Integer::from(-1));
1824    simplify_power(context, argument, minus_one)
1825}

1827fn absolute_value_of(context: &Context, argument: ExpressionReference) -> ExpressionReference {
1828    // Wrap an expression in the `abs` builtin so the resulting call participates
1829    // in the canonical-form pipeline. Used by the GG_addendum_3 derivatives of
1830    // arcsec / arccsc / arccsch whose closed form carries an |x| factor that
1831    // pins the sign of the surrounding radical.
1832    simplify_function_call(
1833        context,
1834        FunctionIdentifier::Builtin(BuiltinFunction::Abs),
1835        vec![argument],
1836    )
1837}

1839fn negate_expression(context: &Context, argument: ExpressionReference) -> ExpressionReference {
1840    // Multiply an expression by integer -1 through simplify_product so that the
1841    // result is in canonical form. Equivalent to the public `negate` builder
1842    // but reused inline here to keep the derivative rules self-contained.
1843    let minus_one = integer(context, Integer::from(-1));
1844    simplify_product(context, vec![minus_one, argument])
1845}

1863fn is_integer_zero(reference: &ExpressionReference) -> bool {
1864    // Predicate identifying the integer literal zero. Used to short-circuit any
1865    // chain-rule contribution whose inner derivative collapses to zero so that
1866    // we do not fabricate a multiplicative chain that the simplifier would only
1867    // remove again on the next pass.
1868    matches!(&reference.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
1869}

1871fn contains_non_smooth_function(target: &ExpressionReference) -> bool {
1872    // Detect whether a subtree contains a builtin that is not differentiable
1873    // everywhere on its domain. Currently abs and signum are the only such
1874    // builtins; the predicate guards the partial-derivative reordering pass so
1875    // that Schwarz's theorem is only invoked on smooth subexpressions.
1876    use ExpressionKind::*;
1877    match &target.kind {
1878        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => false,
1879        Sum(children) | Product(children) => children.iter().any(contains_non_smooth_function),
1880        Power(base, exponent) => {
1881            contains_non_smooth_function(base) || contains_non_smooth_function(exponent)
1882        }
1883        Factorial(argument) => contains_non_smooth_function(argument),
1884        FunctionCall(function, arguments) => {
1885            if matches!(
1886                function,
1887                FunctionIdentifier::Builtin(BuiltinFunction::Abs)
1888                    | FunctionIdentifier::Builtin(BuiltinFunction::Signum)
1889            ) {
1890                return true;
1891            }
1892            arguments.iter().any(contains_non_smooth_function)
1893        }
1894        Derivative(_, _, arguments) => arguments.iter().any(contains_non_smooth_function),
1895        Integral(integrand, _, lower_bound, upper_bound) => {
1896            if contains_non_smooth_function(integrand) {
1897                return true;
1898            }
1899            if let Some(lower) = lower_bound {
1900                if contains_non_smooth_function(lower) {
1901                    return true;
1902                }
1903            }
1904            if let Some(upper) = upper_bound {
1905                if contains_non_smooth_function(upper) {
1906                    return true;
1907                }
1908            }
1909            false
1910        }
1911        Summation(summand, _, lower_bound, upper_bound) => {
1912            if contains_non_smooth_function(summand) {
1913                return true;
1914            }
1915            if let Some(lower) = lower_bound {
1916                if contains_non_smooth_function(lower) {
1917                    return true;
1918                }
1919            }
1920            if let Some(upper) = upper_bound {
1921                if contains_non_smooth_function(upper) {
1922                    return true;
1923                }
1924            }
1925            false
1926        }
1927        LaplaceTransform(target, _, _)
1928        | InverseLaplaceTransform(target, _, _)
1929        | FourierTransform(target, _, _)
1930        | InverseFourierTransform(target, _, _) => contains_non_smooth_function(target),
1931        Piecewise(_) => true,
1932        OrdinaryDifferentialEquation(equation, dependent_variable_call, _)
1933        | RecurrenceEquation(equation, dependent_variable_call, _) => {
1934            contains_non_smooth_function(equation)
1935                || contains_non_smooth_function(dependent_variable_call)
1936        }
1937        _ => true,
1938    }
1939}
