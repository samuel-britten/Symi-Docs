16/// A deterministic structural profile of an immutable expression DAG.
17#[derive(Clone, Debug, Default, Eq, PartialEq)]
18pub(crate) struct ExpressionDagProfile {
19    unique_node_count: usize,
20    edge_count: usize,
21    shared_node_count: usize,
22    maximum_depth: usize,
23    maximum_child_width: usize,
24    estimated_output_node_count: usize,
25    estimated_memory_bytes: usize,
26}

28impl ExpressionDagProfile {

36        self.edge_count

42    }
43
44    /// Return the longest canonical path depth in the reachable DAG.
45    pub(crate) const fn maximum_depth(&self) -> usize {
46        self.maximum_depth
47    }
48
49    /// Return the largest number of direct children on one node.
50    pub(crate) const fn maximum_child_width(&self) -> usize {
51        self.maximum_child_width
52    }

57    }

70/// Build one profile by visiting each reachable allocation once and retaining canonical depth data.
71pub(crate) fn profile(target: &ExpressionReference) -> ExpressionDagProfile {
72    profile_within(target, usize::MAX)
73}

75fn profile_within(target: &ExpressionReference, node_limit: usize) -> ExpressionDagProfile {
76    // The profile is a measurement of the graph, so it borrows the graph rather than holding it:
77    // the root reference the caller already owns keeps every reachable node alive, and the walk
78    // needs a node's address and its children rather than a share of its ownership. The visited
79    // set and the incoming-edge tally have identical keys, so the edge tally alone carries both
80    // the sharing count and the unique-node count.
81    let mut profile = ExpressionDagProfile::default();
82    let mut incoming_edges = HashMap::<*const Expression, usize>::new();
83    let mut pending: Vec<(&Expression, usize)> = vec![(target.as_ref(), 0usize)];
84    while let Some((node, depth)) = pending.pop() {
85        let pointer = node as *const Expression;
86        let incoming_count = {
87            let incoming_count = incoming_edges.entry(pointer).or_insert(0);
88            *incoming_count = incoming_count.saturating_add(1);
89            *incoming_count
90        };
91        profile.edge_count = profile.edge_count.saturating_add(1);
92        profile.maximum_depth = profile.maximum_depth.max(depth);
93        if incoming_count != 1 {
94            continue;
95        }
96        if incoming_edges.len() >= node_limit {
97            continue;
98        }
99        let first_child_position = pending.len();
100        let child_depth = depth.saturating_add(1);
101        for_each_child(node, |child| {
102            pending.push((child.as_ref(), child_depth));
103        });
104        profile.maximum_child_width = profile
105            .maximum_child_width
106            .max(pending.len() - first_child_position);
107        pending[first_child_position..].reverse();
108    }
109    profile.edge_count = profile.edge_count.saturating_sub(1);
110    profile.unique_node_count = incoming_edges.len();
111    profile.shared_node_count = incoming_edges
112        .values()
113        .filter(|incoming_count| **incoming_count > 1)
114        .count();
115    profile.estimated_output_node_count =
116        profile.unique_node_count.saturating_add(profile.edge_count);
117    profile.estimated_memory_bytes = profile
118        .estimated_output_node_count
119        .saturating_mul(std::mem::size_of::<Expression>() + std::mem::size_of::<usize>());
120    record_counter!(expression_dag_profiles);
121    record_counter!(expression_dag_unique_nodes, profile.unique_node_count);
122    record_counter!(expression_dag_edges, profile.edge_count);
123    record_counter!(expression_dag_shared_nodes, profile.shared_node_count);
124    record_counter!(
125        expression_dag_output_nodes,
126        profile.estimated_output_node_count
127    );
128    observe_peak(
129        |counters| &mut counters.expression_dag_peak_memory_bytes,
130        profile.estimated_memory_bytes,
131    );
132    profile
133}

135/// The node ceiling that bounds the cost-gate profile traversal.
136const BOUNDED_PROFILE_NODE_LIMIT: usize = 512;

138/// Profile a DAG for the cost gate, stopping once enough nodes prove the input is large.
139pub(crate) fn bounded_profile(target: &ExpressionReference) -> ExpressionDagProfile {
140    // The full profile visits every reachable allocation, which is the right measure for a
141    // report but the wrong one for a gate consulted at every node of a recursive traversal.
142    // The gate only needs to know whether the subtree is large, so the walk stops at a fixed
143    // node ceiling and reports the ceiling. That keeps the decision deterministic — it depends
144    // only on the expression, never on timing or worker count — while bounding its cost.
145    profile_within(target, BOUNDED_PROFILE_NODE_LIMIT)
146}

148/// The nodes a subtree must present before it is worth profiling for the cost gate.
149///
150/// The gate runs at every node of a recursive traversal, and the traversal's own per-node work is
151/// a handful of interner probes, so the gate has to be cheaper than that or it becomes the cost
152/// it is trying to avoid. This is the width of the cheap probe below: walking this many nodes is
153/// comparable to interning one, and no subtree smaller than this can clear any expression cutoff.
154const COST_GATE_PROBE_NODE_LIMIT: usize = 24;
155
156thread_local! {
157    /// The work units of the smallest subtree the cost gate has already refused above this point
158    /// of a recursive traversal, or no bound when the traversal is not inside a refused subtree.
159    ///
160    /// A descendant's subtree is a subgraph of its ancestor's, and the estimate is monotone in the
161    /// nodes and edges a subtree presents, so a coordinator inside a refused subtree cannot
162    /// present more work than the refusal measured. When the refusal was exact — and it is, since
163    /// the bounded walk stops early only once it has *cleared* a cutoff — every class whose cutoff
164    /// is at or above that bound is answerable without walking anything at all. Without this, a
165    /// deep traversal profiles the DAG below every one of its nodes and the gate becomes the cost
166    /// it exists to avoid.
167    ///
168    /// The bound belongs to the thread because it describes one descent: a worker running a
169    /// distributed subtree has its own descent and inherits no bound from the coordinator that
170    /// handed the subtree out.
171    static ENCLOSING_REFUSED_WORK_UNITS: Cell<u64> = const { Cell::new(u64::MAX) };
172}

174/// Run one serial descent under the bound a cost refusal established for it.
175fn within_refused_subtree<ValueType>(
176    refused_work_units: u64,
177    descent: impl FnOnce() -> ValueType,
178) -> ValueType {
179    let restored = ENCLOSING_REFUSED_WORK_UNITS.get();
180    ENCLOSING_REFUSED_WORK_UNITS.set(restored.min(refused_work_units));
181    let value = descent();
182    ENCLOSING_REFUSED_WORK_UNITS.set(restored);
183    value
184}

186/// Answer the cost gate's question — is this subtree big enough to distribute? — as cheaply as it
187/// can be answered.
188///
189/// The second value reports whether the walk saw the whole subtree. A truncated walk understates a
190/// large subtree, which is exactly what the gate wants — it only has to know that the subtree is
191/// big — but it is not a bound anything else may be concluded from.
192fn profile_to_clear_cutoff(
193    target: &ExpressionReference,
194    cutoff: u64,
195) -> (ExpressionDagProfile, bool) {
196    // Two steps. A narrow probe first: it stops after a couple of dozen nodes, and a subtree that
197    // ends before the probe does is measured exactly and is far under any expression cutoff, so
198    // the gate is settled without a full walk. Only a subtree that fills the probe — which is
199    // already rare, and rarer still further down a traversal — is walked properly, and even then
200    // the walk stops as soon as it has seen enough nodes for the estimate to clear the cutoff.
201    // The estimate is monotone in the node count, so stopping early can only understate a large
202    // subtree, never overstate a small one, and the decision stays a function of the expression.
203    let probe = profile_within(target, COST_GATE_PROBE_NODE_LIMIT);
204    if probe.unique_node_count() < COST_GATE_PROBE_NODE_LIMIT {
205        return (probe, true);
206    }
207    let sufficient_node_count = usize::try_from(cutoff.div_ceil(8).saturating_add(1))
208        .unwrap_or(BOUNDED_PROFILE_NODE_LIMIT)
209        .min(BOUNDED_PROFILE_NODE_LIMIT);
210    let profile = profile_within(target, sufficient_node_count);
211    let complete = profile.unique_node_count() < sufficient_node_count;
212    (profile, complete)
213}

215/// Visit children in the exact order used by the expression's structural hash and serializer.
216///
217/// The children are handed to the visitor borrowed from the parent they already belong to. Every
218/// node strongly owns its children, so a borrow of the parent keeps each child alive for as long
219/// as the visitor can see it, and a traversal that only needs to read a child — to hash it, to
220/// count it, to look its summary up by address — has no reason to pay a reference-count pair per
221/// edge for the privilege. This is the walk that the cost gate runs at every node of a recursive
222/// traversal, so an edge that costs an atomic increment and decrement is a cost the whole session
223/// pays for a value nothing keeps.
224pub(crate) fn for_each_child<'expression, VisitorType>(
225    target: &'expression Expression,
226    mut visitor: VisitorType,
227) where
228    VisitorType: FnMut(&'expression ExpressionReference),
229{
230    use ExpressionKind::*;
231    match &target.kind {
232        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => {}
233        Sum(values)
234        | Product(values)
235        | FiniteSet(values)
236        | SetUnion(values)
237        | SetIntersection(values)
238        | SpecialFunctionParameterVector(values) => {
239            values.iter().for_each(&mut visitor);
240        }
241        Power(base, exponent) => {
242            visitor(base);
243            visitor(exponent);
244        }
245        Factorial(argument) | RandomVariable(_, argument) => visitor(argument),
246        FunctionCall(_, arguments) | Derivative(_, _, arguments) | Distribution(_, arguments) => {
247            arguments.iter().for_each(&mut visitor);
248        }
249        Integral(integrand, _, lower_bound, upper_bound)
250        | Summation(integrand, _, lower_bound, upper_bound)
251        | IndexedProduct(integrand, _, lower_bound, upper_bound) => {
252            visitor(integrand);
253            if let Some(lower_bound) = lower_bound {
254                visitor(lower_bound);
255            }
256            if let Some(upper_bound) = upper_bound {
257                visitor(upper_bound);
258            }
259        }
260        LaplaceTransform(target, _, _)
261        | InverseLaplaceTransform(target, _, _)
262        | FourierTransform(target, _, _)
263        | InverseFourierTransform(target, _, _) => visitor(target),
264        Piecewise(branches) => {
265            for (condition, value) in branches {
266                visitor(condition);
267                visitor(value);
268            }
269        }
270        OrdinaryDifferentialEquation(equation, dependent_variable_call, _)
271        | RecurrenceEquation(equation, dependent_variable_call, _) => {
272            visitor(equation);
273            visitor(dependent_variable_call);
274        }
275        Interval(lower_bound, upper_bound, _, _) | SetComplement(lower_bound, upper_bound) => {
276            visitor(lower_bound);
277            visitor(upper_bound);
278        }
279        ImageSet(_, parameter_domain, lambda_expression)
280        | ConditionSet(_, parameter_domain, lambda_expression) => {
281            visitor(parameter_domain);
282            visitor(lambda_expression);
283        }
284    }
285}

287/// Return owned children in the exact order used by the structural hash and serializer.
288pub(crate) fn child_references(target: &ExpressionReference) -> Vec<ExpressionReference> {
289    let mut children = Vec::new();
290    for_each_child(target, |child| children.push(Arc::clone(child)));
291    children
292}

294/// Return the distinct nodes reachable from a root in deterministic first-seen order.
295pub(crate) fn unique_nodes(target: &ExpressionReference) -> Vec<ExpressionReference> {
296    // One reference-count pair per distinct node, not one per edge: the walk itself borrows, and
297    // only a node that is being handed to the caller is cloned. The children of a node are pushed
298    // in order and the newly pushed run is reversed, which reproduces exactly the first-seen order
299    // a reversed push produced when the frontier held owned references.
300    let mut nodes = Vec::new();
301    let mut seen = HashSet::<*const Expression>::new();
302    let mut pending: Vec<&ExpressionReference> = vec![target];
303    while let Some(reference) = pending.pop() {
304        if !seen.insert(Arc::as_ptr(reference)) {
305            continue;
306        }
307        let first_child_position = pending.len();
308        for_each_child(reference, |child| pending.push(child));
309        pending[first_child_position..].reverse();
310        nodes.push(Arc::clone(reference));
311    }
312    nodes
313}

326/// Map independent expression children through the operation's deterministic execution policy.
327pub(crate) fn ordered_child_map<ValueType, FunctionType>(
328    context: &Context,
329    parent: &ExpressionReference,
330    children: &[ExpressionReference],
331    work_class_identifier: &'static str,
332    function: FunctionType,
333) -> Vec<ValueType>
334where
335    ValueType: Send,
336    FunctionType: Fn(&ExpressionReference) -> ValueType + Send + Sync,
337{
338    record_counter!(expression_dag_route_probes);
339    // The cost gate must not cost more than the work it gates. This runs at every node of a
340    // recursive traversal, so each of the questions below is asked in increasing order of cost
341    // and the first one that settles the route returns before the next is paid for: a node with
342    // fewer than two children has nothing to distribute, a sequential operation or a pool with
343    // no second worker has nowhere to distribute it, and only then is the subtree walked. The
344    // walk itself stops as soon as it has seen enough nodes to clear the work class's cutoff,
345    // so a large subtree costs no more to gate than a borderline one, and without these guards
346    // the traversal would profile the DAG below every node and become quadratic.
347    let availability = crate::parallelism::runtime_availability();
348    let worker_count = availability.worker_count.max(1);
349    let independent_unit_count = children.len();
350    if independent_unit_count < 2
351        || worker_count < 2
352        || !availability.active
353        || context.operation().execution_route() == ExecutionRoute::Serial
354    {
355        record_counter!(expression_serial_coordinators);
356        return ordered_map(ExecutionRoute::Serial, children, function);
357    }
358    let Some(work_unit_cutoff) = crate::parallelism::work_cutoff(
359        work_class_identifier,
360        context.operation().admitted_serial_only_work_classes(),
361        worker_count,
362    ) else {
363        record_counter!(expression_serial_coordinators);
364        return ordered_map(ExecutionRoute::Serial, children, function);
365    };
366    // A cutoff no subtree inside this refused one can reach is answered from the bound the
367    // refusal left behind, without walking anything.
368    if work_unit_cutoff >= ENCLOSING_REFUSED_WORK_UNITS.get() {
369        record_counter!(expression_serial_coordinators);
370        return ordered_map(ExecutionRoute::Serial, children, function);
371    }
372    let (profile, profile_is_complete) = profile_to_clear_cutoff(parent, work_unit_cutoff);
373    let domain_work_units = (profile.unique_node_count() as u64)
374        .saturating_mul(8)
375        .saturating_add((profile.edge_count() as u64).saturating_mul(2));
376    let predicted_task_local_bytes = profile.estimated_memory_bytes().saturating_add(
377        children
378            .len()
379            .saturating_mul(std::mem::size_of::<ExpressionReference>()),
380    );
381    let work_estimate = WorkEstimate::new(
382        domain_work_units,
383        independent_unit_count,
384        predicted_task_local_bytes,
385        profile.maximum_depth(),
386        independent_unit_count.min(worker_count),
387        work_class_identifier,
388    );
389    let decision = context.operation().execution_decision_silent(
390        &work_estimate,
391        independent_unit_count.min(worker_count),
392        crate::parallelism::runtime_memory_capacity(predicted_task_local_bytes),
393    );
394    match decision.route {
395        ExecutionRoute::Parallel => {
396            record_counter!(expression_parallel_coordinators);
397            ordered_map(ExecutionRoute::Parallel, children, function)
398        }
399        ExecutionRoute::Serial => {
400            record_counter!(expression_serial_coordinators);
401            // A refusal taken on a complete measurement of this subtree bounds every
402            // coordinator the serial descent below it will reach. A truncated one measures
403            // nothing about them and leaves no bound behind.
404            if profile_is_complete {
405                within_refused_subtree(domain_work_units, || {
406                    ordered_map(ExecutionRoute::Serial, children, function)
407                })
408            } else {
409                ordered_map(ExecutionRoute::Serial, children, function)
410            }
411        }
412    }

442    #[test]
443    fn a_bounded_profile_reports_whether_it_saw_the_whole_subtree() {
444        use super::{BOUNDED_PROFILE_NODE_LIMIT, profile_to_clear_cutoff};
445        let mut context = Context::new();
446        let x = symbol(&mut context, "x");
447        let one = integer_value(&mut context, 1);
448        let small = sum(&mut context, vec![x, one]);
449        let (small_profile, complete) = profile_to_clear_cutoff(&small, 4_096);
450        assert!(
451            complete,
452            "a subtree the walk finished is a complete measurement"
453        );
454        assert_eq!(small_profile.unique_node_count(), 3);
455
456        // A sum wider than the walk's own ceiling: the gate learns that the subtree is large and
457        // nothing else, so the measurement is not a bound on anything.
458        let mut leaves = Vec::new();
459        for value in 0..BOUNDED_PROFILE_NODE_LIMIT + 8 {
460            leaves.push(symbol(&mut context, &format!("symbol_{value}")));
461        }
462        let wide = sum(&mut context, leaves);
463        let (_, complete) = profile_to_clear_cutoff(&wide, u64::MAX);
464        assert!(!complete, "a truncated walk must not be read as a bound");
465    }

468    fn a_refused_subtree_answers_a_wider_cutoff_without_walking_again() {
469        use super::{ENCLOSING_REFUSED_WORK_UNITS, within_refused_subtree};
470        assert_eq!(ENCLOSING_REFUSED_WORK_UNITS.get(), u64::MAX);
471        within_refused_subtree(64, || {
472            assert_eq!(ENCLOSING_REFUSED_WORK_UNITS.get(), 64);
473            // A tighter refusal inside a looser one keeps the tighter bound, and a looser one
474            // inside a tighter one does not loosen it.

483        assert_eq!(ENCLOSING_REFUSED_WORK_UNITS.get(), u64::MAX);
