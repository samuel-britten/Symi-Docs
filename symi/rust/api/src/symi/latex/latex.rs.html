56pub struct LatexOptions {

58    // content-independent canonical choice — never a heuristic that inspects the
59    // expression and switches notation for readability (that belongs in the

65    /// Stores the `natural_logarithm_style` value.
66    pub natural_logarithm_style: NaturalLogarithmStyle,
67    /// Stores the `multiletter_symbol_style` value.
68    pub multiletter_symbol_style: MultiletterSymbolStyle,
69}

106pub fn to_latex(reference: &ExpressionReference, table: &SymbolTable) -> String {
107    // Entry point for callers that accept the canonical defaults. Delegates to
108    // the options-aware renderer with `latex_options::default()`.
109    to_latex_with_options(reference, table, &LatexOptions::default())
110}

112/// Performs the advanced native `to_latex_with_options` operation.
113pub fn to_latex_with_options(
114    reference: &ExpressionReference,
115    table: &SymbolTable,
116    options: &LatexOptions,
117) -> String {
118    // Render an expression to LaTeX, then apply the presentation flags as
119    // post-processing passes over the rendered string. The passes are written
120    // to be unambiguous on this renderer's own output: `\exp\left(...\right)`,
121    // `\log\left`, and `\mathit{...}` each have exactly one source in the
122    // renderer, so the targeted rewrites cannot touch unrelated notation.
123    let mut rendered = render_expression(reference, table);
124    if options.exponential_style == ExponentialStyle::ECaret {
125        rendered = rewrite_exponential_to_caret(&rendered);
126    }
127    if options.natural_logarithm_style == NaturalLogarithmStyle::Ln {
128        rendered = rendered.replace("\\log\\left", "\\ln\\left");
129    }
130    if options.multiletter_symbol_style == MultiletterSymbolStyle::Upright {
131        rendered = rendered.replace("\\mathit{", "\\mathrm{");
132    }
133    rendered
134}

136fn rewrite_exponential_to_caret(input: &str) -> String {
137    // Rewrite every `\exp\left(BODY\right)` into `e^{BODY}`. The renderer always
138    // wraps the single exp argument in `\left(...\right)`, so the opening marker
139    // is fixed; matching_right_paren finds the balanced close, which handles
140    // nested exponentials and any inner delimiters. Repeats until no marker
141    // remains so nested `\exp(\exp(...))` fully converts.
142    let mut current = input.to_owned();
143    loop {
144        let start = match current.find("\\exp\\left(") {
145            Some(position) => position,
146            None => return current,
147        };
148        let body_start = start + "\\exp\\left(".len();
149        let body_end = match matching_right_parenthesis(&current, body_start) {
150            Some(position) => position,
151            None => return current,
152        };
153        let body = current[body_start..body_end].to_owned();
154        let after = body_end + "\\right)".len();
155        current = format!("{}e^{{{}}}{}", &current[..start], body, &current[after..]);
156    }
157}

159fn matching_right_parenthesis(input: &str, from: usize) -> Option<usize> {
160    // Given a position just past a `\left(`, return the byte index of the
161    // `\right)` that closes it, tracking nested `\left(` / `\right)` depth. Only
162    // round `\left(` / `\right)` pairs adjust the depth; other delimiters
163    // (`\left|`, `\left[`, ...) are balanced among themselves and never contain
164    // an unmatched `\right)` in this renderer's output.
165    let mut depth = 1usize;
166    let mut index = from;
167    while index < input.len() {
168        let remainder = &input[index..];
169        if remainder.starts_with("\\left(") {
170            depth += 1;
171            index += "\\left(".len();
172        } else if remainder.starts_with("\\right)") {
173            depth -= 1;
174            if depth == 0 {
175                return Some(index);
176            }
177            index += "\\right)".len();
178        } else {
179            index += remainder.chars().next().map(char::len_utf8).unwrap_or(1);
180        }
181    }
182    None
183}

185fn render_expression(reference: &Expression, table: &SymbolTable) -> String {
186    // Dispatch one expression node to the per-kind renderer. Atoms render
187    // directly; compound nodes hand off to dedicated renderers that know which
188    // children need grouping and how to arrange the canonical layout into
189    // mathematical LaTeX.
190    match &reference.kind {
191        ExpressionKind::Integer(value) => format!("{}", value),
192        ExpressionKind::Rational(value) => render_rational(value),
193        ExpressionKind::PolynomialRoot(value) => {
194            let coefficients = value
195                .minimal_polynomial
196                .coefficients
197                .iter()
198                .map(ToString::to_string)
199                .collect::<Vec<_>>()
200                .join(", ");
201            format!(
202                "\\operatorname{{Root}}\\left([{}], {}\\right)",
203                coefficients, value.root_index
204            )
205        }
206        ExpressionKind::Symbol(identifier) => render_symbol(*identifier, table),
207        ExpressionKind::Constant(value) => render_constant(value),
208        ExpressionKind::Sum(children) => render_sum(children, table),
209        ExpressionKind::Product(factors) => render_product(factors, table, true),
210        ExpressionKind::Power(base, exponent) => render_power(base, exponent, table),
211        ExpressionKind::Factorial(argument) => render_factorial(argument, table),
212        ExpressionKind::FunctionCall(function, arguments) => {
213            render_function_call(*function, arguments, table)
214        }
215        ExpressionKind::Derivative(function, orders, arguments) => {
216            render_derivative(*function, orders, arguments, table)
217        }
218        ExpressionKind::Integral(integrand, variable, lower_bound, upper_bound) => render_integral(
219            integrand,
220            *variable,
221            lower_bound.as_ref(),
222            upper_bound.as_ref(),
223            table,
224        ),
225        ExpressionKind::Summation(summand, index, lower_bound, upper_bound) => render_summation(
226            summand,
227            *index,
228            lower_bound.as_ref(),
229            upper_bound.as_ref(),
230            table,
231        ),
232        ExpressionKind::IndexedProduct(multiplicand, index, lower_bound, upper_bound) => {
233            render_indexed_product(
234                multiplicand,
235                *index,
236                lower_bound.as_ref(),
237                upper_bound.as_ref(),
238                table,
239            )
240        }
241        ExpressionKind::LaplaceTransform(target, _time_variable, frequency_variable) => {
242            render_laplace_transform(target, *frequency_variable, table)
243        }
244        ExpressionKind::InverseLaplaceTransform(target, _frequency_variable, time_variable) => {
245            render_inverse_laplace_transform(target, *time_variable, table)
246        }
247        ExpressionKind::FourierTransform(target, _time_variable, frequency_variable) => {
248            render_fourier_transform(target, *frequency_variable, table)
249        }
250        ExpressionKind::InverseFourierTransform(target, _frequency_variable, time_variable) => {
251            render_inverse_fourier_transform(target, *time_variable, table)
252        }
253        ExpressionKind::Piecewise(branches) => render_piecewise(branches, table),
254        ExpressionKind::OrdinaryDifferentialEquation(
255            equation,
256            dependent_variable_call,
257            independent_variable,
258        ) => render_ordinary_differential_equation(
259            equation,
260            dependent_variable_call,
261            *independent_variable,
262            table,
263        ),
264        ExpressionKind::RecurrenceEquation(equation, sequence_variable_call, index_variable) => {
265            render_recurrence_equation(equation, sequence_variable_call, *index_variable, table)
266        }
267        ExpressionKind::Interval(lower_bound, upper_bound, lower_inclusive, upper_inclusive) => {
268            render_interval(
269                lower_bound,
270                upper_bound,
271                *lower_inclusive,
272                *upper_inclusive,
273                table,
274            )
275        }
276        ExpressionKind::FiniteSet(elements) => render_finite_set(elements, table),
277        ExpressionKind::SetUnion(operands) => render_set_union(operands, table),
278        ExpressionKind::SetIntersection(operands) => render_set_intersection(operands, table),
279        ExpressionKind::SetComplement(left_operand, right_operand) => {
280            render_set_complement(left_operand, right_operand, table)
281        }
282        ExpressionKind::ImageSet(parameter_variable, parameter_domain, lambda_expression) => {
283            render_image_set(
284                *parameter_variable,
285                parameter_domain,
286                lambda_expression,
287                table,
288            )
289        }
290        ExpressionKind::ConditionSet(parameter_variable, base_set, condition) => {
291            render_condition_set(*parameter_variable, base_set, condition, table)
292        }
293        ExpressionKind::Distribution(identifier, parameters) => {
294            render_distribution(*identifier, parameters, table)
295        }
296        ExpressionKind::RandomVariable(name, underlying_distribution) => {
297            render_random_variable(*name, underlying_distribution, table)
298        }
299        ExpressionKind::SpecialFunctionParameterVector(parameters) => {
300            render_special_function_parameter_vector(parameters, table)
301        }
302    }
303}

305fn render_special_function_parameter_vector(
306    parameters: &[ExpressionReference],
307    table: &SymbolTable,
308) -> String {
309    // Render one ordered parameter group as a comma-separated row. A bare group
310    // is only reachable when a caller renders it outside its owning call; the
311    // hypergeometric and Meijer-G renderers lay their groups out over the two
312    // rows of the conventional notation and do not route through here.
313    parameters
314        .iter()
315        .map(|parameter| render_expression(parameter, table))
316        .collect::<Vec<String>>()
317        .join(", ")
318}

320fn render_distribution(
321    identifier: crate::statistics::identifier::DistributionIdentifier,
322    parameters: &[ExpressionReference],
323    table: &SymbolTable,
324) -> String {
325    // Render a parametrised distribution as `\mathrm{Normal}(0, 1)`.
326    let name = crate::statistics::identifier::distribution_name(identifier);
327    let display_name = capitalise(name);
328    let mut out = format!("\\mathrm{{{}}}\\left(", display_name);
329    for (index, parameter) in parameters.iter().enumerate() {
330        if index > 0 {
331            out.push_str(", ");
332        }
333        out.push_str(&render_expression(parameter, table));
334    }
335    out.push_str("\\right)");
336    out
337}

354fn capitalise(name: &str) -> String {
355    // Convert a snake-case distribution name to a title-case form suitable for
356    // LaTeX rendering. `negative_binomial` becomes `NegativeBinomial`,
357    // `chi_squared` becomes `ChiSquared`, `student_t` becomes `StudentT`.
358    let mut output = String::with_capacity(name.len());
359    let mut next_upper = true;
360    for character in name.chars() {
361        if character == '_' {
362            next_upper = true;
363            continue;
364        }
365        if next_upper {
366            for upper in character.to_uppercase() {
367                output.push(upper);
368            }
369            next_upper = false;
370        } else {
371            output.push(character);
372        }
373    }
374    output
375}

416fn render_interval(
417    lower_bound: &Expression,
418    upper_bound: &Expression,
419    lower_inclusive: bool,
420    upper_inclusive: bool,
421    table: &SymbolTable,
422) -> String {
423    // Render an interval as the textbook LaTeX bracket pair, mirroring the
424    // Display arm's `[a, b]` / `(a, b)` choice via `\left[` / `\left(` and
425    // the matching close bracket.
426    let opener = if lower_inclusive {
427        "\\left["
428    } else {
429        "\\left("
430    };
431    let closer = if upper_inclusive {
432        "\\right]"
433    } else {
434        "\\right)"
435    };
436    format!(
437        "{}{}, {}{}",
438        opener,
439        render_expression(lower_bound, table),
440        render_expression(upper_bound, table),
441        closer,
442    )
443}

445fn render_finite_set(elements: &[ExpressionReference], table: &SymbolTable) -> String {
446    // Render a finite set as `\left\{a, b, c\right\}`. Children are emitted
447    // in canonical (sorted) order, comma-separated.
448    let mut out = String::from("\\left\\{");
449    for (index, element) in elements.iter().enumerate() {
450        if index > 0 {
451            out.push_str(", ");
452        }
453        out.push_str(&render_expression(element, table));
454    }
455    out.push_str("\\right\\}");
456    out
457}

494fn render_recurrence_equation(
495    equation: &Expression,
496    sequence_variable_call: &Expression,
497    index_variable: SymbolIdentifier,
498    table: &SymbolTable,
499) -> String {
500    // Render an unsolved recurrence_equation placeholder as the textbook
501    // surface form `\mathrm{Rec}\!\left\{equation = 0,\, y(n);\, n\right\}`,
502    // paralleling the sibling render_ordinary_differential_equation writer
503    // and using the `\mathrm{Rec}` head so the rendering is distinguishable
504    // from an ODE at a glance.
505    let equation_latex = render_expression(equation, table);
506    let sequence_call_latex = render_expression(sequence_variable_call, table);
507    let index_name = table.name(index_variable);
508    let index_latex = render_variable_name(&index_name);
509    format!(
510        "\\mathrm{{Rec}}\\!\\left\\{{{} = 0,\\, {};\\, {}\\right\\}}",
511        equation_latex, sequence_call_latex, index_latex,
512    )
513}

515fn render_ordinary_differential_equation(
516    equation: &Expression,
517    dependent_variable_call: &Expression,
518    independent_variable: SymbolIdentifier,
519    table: &SymbolTable,
520) -> String {
521    // Render an unsolved ordinary_differential_equation placeholder as the
522    // textbook surface form `\mathrm{ODE}\!\left\{equation = 0,\, y(x);\, x\right\}`.
523    // The `= 0` reflects the convention that the equation slot stores the
524    // canonical LHS - RHS = 0 form; the dependent-variable call is the
525    // function being solved for; the independent-variable name follows the
526    // semicolon to distinguish it from the equation's free symbols.
527    let equation_latex = render_expression(equation, table);
528    let dependent_call_latex = render_expression(dependent_variable_call, table);
529    let independent_name = table.name(independent_variable);
530    let independent_latex = render_variable_name(&independent_name);
531    format!(
532        "\\mathrm{{ODE}}\\!\\left\\{{{} = 0,\\, {};\\, {}\\right\\}}",
533        equation_latex, dependent_call_latex, independent_latex,
534    )
535}

537fn render_laplace_transform(
538    target: &Expression,
539    frequency_variable: SymbolIdentifier,
540    table: &SymbolTable,
541) -> String {
542    // Render a forward Laplace transform as
543    //   \mathcal{L}\!\left\{target\right\}\!\left(s\right)
544    // matching the textbook surface used by Doetsch and Schiff. The bound
545    // time variable does not appear because the standard mathematical
546    // notation suppresses it; the outer parameter argument is the
547    // frequency variable.
548    let target_latex = render_expression(target, table);
549    let frequency_name = table.name(frequency_variable);
550    let frequency_latex = render_variable_name(&frequency_name);
551    format!(
552        "\\mathcal{{L}}\\!\\left\\{{{}\\right\\}}\\!\\left({}\\right)",
553        target_latex, frequency_latex
554    )
555}

612fn render_rational(value: &exact_rational) -> String {
613    // Render a rational number as \frac{numerator}{denominator}. A negative
614    // rational carries a leading minus sign outside the fraction because LaTeX
615    // renders the minus best that way. The numerator and denominator returned
616    // by to_numerator_and_denominator are both absolute (unsigned) values.
617    let parts = rational_parts(value);
618    let sign = if integer_is_negative(&parts.numerator) {
619        "-"
620    } else {
621        ""
622    };
623    let numerator_magnitude = integer_absolute_magnitude(&parts.numerator);
624    format!(
625        "{}\\frac{{{}}}{{{}}}",
626        sign,
627        natural_to_decimal(&numerator_magnitude),
628        natural_to_decimal(&parts.denominator)
629    )
630}

632fn render_symbol(identifier: SymbolIdentifier, table: &SymbolTable) -> String {
633    // Variables are italic: single ASCII letters render as-is (italic by
634    // default in math mode), Greek-letter names map to their LaTeX command,
635    // and everything else uses \mathit{...} to keep multi-letter names italic.

639fn render_variable_name(name: &str) -> String {
640    // Dispatch a variable name to its canonical LaTeX form. Greek names (e.g.
641    // "alpha") map to the corresponding command; single ASCII letters render
642    // bare, except the letters the parser reads as constants when bare (`i` the
643    // imaginary unit, `e` Euler's number), which use the \mathit{...} escape so
644    // they read back as the literal symbol rather than the constant; all other
645    // multi-letter names use \mathit for italic upright text.
646    if let Some(greek_command) = greek_latex_command(name) {
647        return greek_command.to_owned();
648    }
649    let mut characters = name.chars();
650    if let (Some(single_character), None) = (characters.next(), characters.next()) {
651        if single_character.is_ascii_alphabetic() && !letter_is_a_bare_constant(single_character) {
652            return name.to_owned();
653        }
654    }
655    format!("\\mathit{{{}}}", escape_latex_identifier(name))
656}

658fn letter_is_a_bare_constant(letter: char) -> bool {
659    // The single letters the LaTeX parser resolves to a constant when they
660    // appear bare: `i` (imaginary unit) and `e` (Euler's number). A symbol with
661    // one of these names must render through the \mathit{...} escape so it round
662    // trips to the symbol rather than the constant. `d` is excluded — the walker
663    // already treats a bare `d` as an ordinary symbol.
664    letter == 'i' || letter == 'e'
665}

728fn render_sum(terms: &[ExpressionReference], table: &SymbolTable) -> String {
729    // Print a canonical sum by walking its terms in presentation order. The first
730    // term emits its own sign; subsequent terms emit an explicit + or − connector
731    // based on whether the term carries a negative leading factor, which is
732    // then stripped by the body renderer.
733    let mut result = String::new();
734    for (index, term) in sum_terms_in_presentation_order(terms).enumerate() {
735        let negative = term_has_negative_sign(term);
736        if index == 0 {
737            if negative {
738                result.push('-');
739            }
740        } else if negative {
741            result.push_str(" - ");
742        } else {
743            result.push_str(" + ");
744        }
745        result.push_str(&render_sum_term_body(term, table));
746    }
747    result
748}

750fn term_has_negative_sign(term: &Expression) -> bool {
751    // Detect whether a sum term carries a negative leading sign. Mirrors the
752    // identical predicate in display.rs.
753    match &term.kind {
754        ExpressionKind::Integer(value) => integer_is_negative(value),
755        ExpressionKind::Rational(value) => rational_is_negative(value),
756        ExpressionKind::Product(factors) => match factors.first() {
757            Some(first) => match &first.kind {
758                ExpressionKind::Integer(value) => integer_is_negative(value),
759                ExpressionKind::Rational(value) => rational_is_negative(value),
760                _ => false,
761            },
762            None => false,
763        },
764        _ => false,
765    }
766}

768fn render_sum_term_body(term: &Expression, table: &SymbolTable) -> String {
769    // Write the absolute-value rendering of a sum term. Negative numerics are
770    // negated for printing; products with a negative leading factor are
771    // dispatched through the product renderer with the leading sign suppressed
772    // so the sum's connector carries the only minus.
773    match &term.kind {
774        ExpressionKind::Integer(value) if integer_is_negative(value) => {
775            format!("{}", -value.clone())
776        }
777        ExpressionKind::Rational(value) if rational_is_negative(value) => {
778            render_rational(&(-value.clone()))
779        }
780        ExpressionKind::Product(factors) if term_has_negative_sign(term) => {
781            render_product(factors, table, false)
782        }
783        _ => render_expression(term, table),
784    }
785}

787fn render_product(
788    factors: &[ExpressionReference],
789    table: &SymbolTable,
790    include_leading_sign: bool,
791) -> String {
792    // Render a canonical product. Splits a leading numeric coefficient off into
793    // a sign plus magnitude, then partitions the remaining factors into numerator
794    // and denominator lists. Factors of the form power(base, -1) go into the
795    // denominator and the whole product is wrapped in \frac{...}{...}. When
796    // there are no denominator factors, multiplies numerator parts with \cdot.
797    // product(-1, x) renders as -x because the magnitude-1 coefficient is
798    // suppressed and only its sign is emitted.
799    let (sign_negative, coefficient_text, factors_after_coefficient) =
800        split_leading_coefficient(factors);
801
802    let mut numerator: Vec<&Expression> = Vec::new();
803    let mut denominator: Vec<&Expression> = Vec::new();
804    for factor in factors_after_coefficient {
805        if let ExpressionKind::Power(base, exponent) = &factor.kind {
806            if is_integer_minus_one(exponent) {
807                denominator.push(base);
808                continue;
809            }
810        }
811        numerator.push(factor);
812    }
813
814    let sign = if sign_negative && include_leading_sign {
815        "-"
816    } else {
817        ""
818    };
819
820    if denominator.is_empty() {
821        let mut parts: Vec<String> = Vec::new();
822        if let Some(text) = &coefficient_text {
823            parts.push(text.clone());
824        }
825        for factor in &numerator {
826            parts.push(render_in_product_factor_context(factor, table));
827        }
828        if parts.is_empty() {
829            format!("{}1", sign)
830        } else {
831            format!("{}{}", sign, join_product_factors(&parts))
832        }
833    } else {
834        let mut numerator_parts: Vec<String> = Vec::new();
835        if let Some(text) = &coefficient_text {
836            numerator_parts.push(text.clone());
837        }
838        for factor in &numerator {
839            numerator_parts.push(render_in_frac_factor_context(factor, table));
840        }
841        let numerator_str = if numerator_parts.is_empty() {
842            "1".to_owned()
843        } else {
844            join_product_factors(&numerator_parts)
845        };
846        let denominator_str = denominator
847            .iter()
848            .map(|base| render_in_frac_factor_context(base, table))
849            .collect::<Vec<_>>()
850            .join(" ");
851        format!("{}\\frac{{{}}}{{{}}}", sign, numerator_str, denominator_str)
852    }
853}
854
855fn join_product_factors(parts: &[String]) -> String {
856    // Join multiplicative factors. Implicit juxtaposition (a single space) is the
857    // default, but a factor that begins with a number placed immediately after a
858    // preceding factor would be misread as one numeral rather than a product:
859    // \frac{1}{2} 2^{...} reads as the mixed number "two and a half", and 2 3
860    // reads as twenty-three. Such boundaries get an explicit \cdot so the product
861    // is unambiguous, matching standard CAS typography.
862    let mut joined = String::new();
863    for (index, part) in parts.iter().enumerate() {
864        if index > 0 {
865            if begins_with_number(part) {
866                joined.push_str(" \\cdot ");
867            } else {
868                joined.push(' ');
869            }
870        }
871        joined.push_str(part);
872    }
873    joined
874}

876fn begins_with_number(rendered: &str) -> bool {
877    // A rendered factor reads as a number when it opens with a digit or with a
878    // displayed fraction (\frac{...}{...}). These are exactly the leading forms
879    // that would fuse with a preceding factor under implicit multiplication.
880    rendered.starts_with(|character: char| character.is_ascii_digit())
881        || rendered.starts_with("\\frac")
882}

884fn split_leading_coefficient<'lifetime>(
885    factors: &'lifetime [ExpressionReference],
886) -> (bool, Option<String>, &'lifetime [ExpressionReference]) {
887    // Inspect the first factor of a canonical product. A leading numeric child
888    // is peeled off into a (sign, magnitude_text) pair. The integer literal -1
889    // is special-cased to drop its magnitude entirely. Mirrors the identical
890    // helper in display.rs.
891    let first = match factors.first() {
892        Some(first) => first,
893        None => return (false, None, factors),
894    };
895    match &first.kind {
896        ExpressionKind::Integer(value) if integer_is_negative_one(value) => {
897            (true, None, &factors[1..])
898        }
899        ExpressionKind::Integer(value) if integer_is_negative(value) => {
900            (true, Some(format!("{}", -value.clone())), &factors[1..])
901        }
902        ExpressionKind::Integer(value) => (false, Some(format!("{}", value)), &factors[1..]),
903        ExpressionKind::Rational(value) if rational_is_negative(value) => (
904            true,
905            Some(render_rational(&(-value.clone()))),
906            &factors[1..],
907        ),
908        ExpressionKind::Rational(value) => (false, Some(render_rational(value)), &factors[1..]),
909        _ => (false, None, factors),
910    }
911}

913fn render_in_product_factor_context(factor: &Expression, table: &SymbolTable) -> String {
914    // A product factor sits at multiplicative precedence. Sums must be wrapped
915    // because they have lower precedence. Negative numerics also need grouping
916    // to avoid producing a leading minus that looks like a binary minus.
917    let needs_grouping =
918        matches!(&factor.kind, ExpressionKind::Sum(_)) || is_negative_numeric(factor);
919    if needs_grouping {
920        format!("\\left({}\\right)", render_expression(factor, table))
921    } else {
922        render_expression(factor, table)
923    }
924}

926fn render_in_frac_factor_context(factor: &Expression, table: &SymbolTable) -> String {
927    // Inside a \frac numerator or denominator the fraction bar already provides
928    // grouping, so sums do not need extra parentheses. Negative numerics in the
929    // denominator do need parentheses to keep the minus sign unambiguous.
930    let needs_grouping = is_negative_numeric(factor);
931    if needs_grouping {
932        format!("\\left({}\\right)", render_expression(factor, table))
933    } else {
934        render_expression(factor, table)
935    }
936}

938fn render_power(
939    base: &ExpressionReference,
940    exponent: &ExpressionReference,
941    table: &SymbolTable,
942) -> String {
943    // Render a power. When the exponent is exactly -1 the whole expression
944    // is a multiplicative inverse and is rendered as \frac{1}{base}. When the
945    // base is a function call, dispatch to render_power_of_function_call which
946    // applies the trig convention (\sin^{2}(x)) vs the wrapping convention
947    // ((\ln(x))^{2}). Otherwise the LaTeX caret syntax is used with the base
948    // wrapped whenever its kind would otherwise parse incorrectly.
949    if is_integer_minus_one(exponent) {
950        let base_latex = render_in_frac_factor_context(base, table);
951        return format!("\\frac{{1}}{{{}}}", base_latex);
952    }
953    if let ExpressionKind::FunctionCall(function, arguments) = &base.kind {
954        return render_power_of_function_call(*function, arguments, exponent, table);
955    }
956    let base_latex = render_power_base(base, table);
957    let exponent_latex = render_power_exponent(exponent, table);
958    format!("{}^{{{}}}", base_latex, exponent_latex)
959}

961fn render_power_of_function_call(
962    function: FunctionIdentifier,
963    arguments: &[ExpressionReference],
964    exponent: &Expression,
965    table: &SymbolTable,
966) -> String {
967    // Trig and hyperbolic functions (including arc- inverses) use the convention
968    // \sin^{2}\left(x\right) — exponent between the function name and the argument.
969    // All other functions wrap the entire call: \left(\ln\left(x\right)\right)^{2}.
970    let exponent_latex = render_power_exponent(exponent, table);
971    if is_trig_function(function) {
972        let function_name = match function {
973            FunctionIdentifier::Builtin(builtin) => builtin_latex_name(builtin).to_owned(),
974            FunctionIdentifier::User(identifier) => {
975                return format!(
976                    "\\left({}\\right)^{{{}}}",
977                    render_user_function_call(identifier, arguments, table),
978                    exponent_latex
979                );
980            }
981        };
982        let argument_list = arguments
983            .iter()
984            .map(|argument| render_expression(argument, table))
985            .collect::<Vec<_>>()
986            .join(", ");
987        format!(
988            "{}^{{{}}}\\left({}\\right)",
989            function_name, exponent_latex, argument_list
990        )
991    } else {
992        let call_latex = render_function_call(function, arguments, table);
993        format!("\\left({}\\right)^{{{}}}", call_latex, exponent_latex)
994    }
995}

1001    match function {
1002        FunctionIdentifier::Builtin(builtin) => matches!(
1003            builtin,

1006                | BuiltinFunction::Tan
1007                | BuiltinFunction::Sec
1008                | BuiltinFunction::Csc
1009                | BuiltinFunction::Cot
1010                | BuiltinFunction::Sinh
1011                | BuiltinFunction::Cosh
1012                | BuiltinFunction::Tanh
1013                | BuiltinFunction::Sech
1014                | BuiltinFunction::Csch
1015                | BuiltinFunction::Coth
1016                | BuiltinFunction::Arcsin
1017                | BuiltinFunction::Arccos
1018                | BuiltinFunction::Arctan
1019                | BuiltinFunction::Arcsec
1020                | BuiltinFunction::Arccsc
1021                | BuiltinFunction::Arccot
1022                | BuiltinFunction::Arcsinh
1023                | BuiltinFunction::Arccosh
1024                | BuiltinFunction::Arctanh
1025                | BuiltinFunction::Arcsech
1026                | BuiltinFunction::Arccsch

1030    }
1031}

1033fn render_power_base(base: &Expression, table: &SymbolTable) -> String {
1034    // Wrap the base of a power in \left(\right) whenever its kind would
1035    // otherwise bind looser than the caret: sums, products, nested powers, and
1036    // any signed or rational numeric leaf are all wrapped.
1037    let needs_grouping = matches!(
1038        &base.kind,
1039        ExpressionKind::Sum(_) | ExpressionKind::Product(_) | ExpressionKind::Power(_, _)
1040    ) || is_signed_or_rational_numeric(base);
1041    if needs_grouping {
1042        format!("\\left({}\\right)", render_expression(base, table))
1043    } else {
1044        render_expression(base, table)
1045    }
1046}

1048fn render_power_exponent(exponent: &Expression, table: &SymbolTable) -> String {
1049    // The exponent already sits inside LaTeX braces, so no extra parentheses
1050    // are needed for sums or products. Only signed numerics could be confusing
1051    // because a leading minus in a LaTeX superscript is fine — so actually

1463fn render_hypergeometric_pfq(arguments: &[ExpressionReference], table: &SymbolTable) -> String {
1464    // Render {}_pF_q with the upper parameters over the lower ones and the
1465    // argument after a semicolon, following DLMF 16.2. The sub- and
1466    // superscripts are the actual group lengths, so the rendered p and q can
1467    // never disagree with the parameters shown.
1468    let (Some(upper), Some(lower)) = (
1469        arguments.first().and_then(parameter_group_of),
1470        arguments.get(1).and_then(parameter_group_of),
1471    ) else {
1472        return render_standard_call("\\operatorname{hypergeometric\\_pfq}", arguments, table);
1473    };
1474    let Some(argument) = arguments.get(2) else {
1475        return render_standard_call("\\operatorname{hypergeometric\\_pfq}", arguments, table);
1476    };
1477    format!(
1478        "{{}}_{{{}}}F_{{{}}}\\left(\\begin{{matrix}}{}\\\\{}\\end{{matrix}};{}\\right)",
1479        upper.len(),
1480        lower.len(),
1481        render_parameter_row(upper, table),
1482        render_parameter_row(lower, table),
1483        render_expression(argument, table)
1484    )
1485}

1487fn render_meijer_g(arguments: &[ExpressionReference], table: &SymbolTable) -> String {
1488    // Render G_{p,q}^{m,n} following DLMF 16.17. The four stored groups are
1489    // concatenated back into the two displayed rows, while m, n, p, and q are
1490    // computed from the group lengths so the displayed orders always describe
1491    // the parameters actually held.
1492    let groups: Option<Vec<&[ExpressionReference]>> = (0..4)
1493        .map(|index| arguments.get(index).and_then(parameter_group_of))
1494        .collect();
1495    let (Some(groups), Some(argument)) = (groups, arguments.get(4)) else {
1496        return render_standard_call("\\operatorname{meijer\\_g}", arguments, table);
1497    };
1498    let upper_numerator_count = groups[0].len();
1499    let lower_numerator_count = groups[2].len();
1500    let upper_row = render_parameter_row(&[groups[0], groups[1]].concat(), table);
1501    let lower_row = render_parameter_row(&[groups[2], groups[3]].concat(), table);
1502    format!(
1503        "G_{{{},{}}}^{{{},{}}}\\left({}\\,\\middle|\\,\\begin{{matrix}}{}\\\\{}\\end{{matrix}}\\right)",
1504        groups[0].len() + groups[1].len(),
1505        groups[2].len() + groups[3].len(),
1506        lower_numerator_count,
1507        upper_numerator_count,
1508        render_expression(argument, table),
1509        upper_row,
1510        lower_row
1511    )
1512}

1514fn render_orthogonal_polynomial(
1515    function: BuiltinFunction,
1516    arguments: &[ExpressionReference],
1517    table: &SymbolTable,
1518) -> String {
1519    // The conventional DLMF Chapter 18 notation: the degree is a subscript, the
1520    // family parameters are a parenthesized superscript, and the argument stays
1521    // in the call parentheses. The superscript is what tells a Jacobi `P` apart
1522    // from a Legendre `P`, so a parameterized family always renders one, and a
1523    // call whose argument count does not match its family falls back to the
1524    // ordinary named form rather than mislabelling its slots.
1525    let Some(family) = crate::special_functions::orthogonal_polynomials::family_of(function) else {
1526        return render_standard_call(builtin_latex_name(function), arguments, table);
1527    };
1528    if arguments.len() != family.arity() {
1529        return render_standard_call(builtin_latex_name(function), arguments, table);
1530    }
1531    let parameters = arguments[1..family.analytic_argument_index()]
1532        .iter()
1533        .map(|parameter| render_expression(parameter, table))
1534        .collect::<Vec<_>>()
1535        .join(",");
1536    let superscript = if parameters.is_empty() {
1537        String::new()
1538    } else {
1539        format!("^{{({})}}", parameters)
1540    };
1541    format!(
1542        "{}_{{{}}}{}\\left({}\\right)",
1543        builtin_latex_name(function),
1544        render_expression(&arguments[0], table),
1545        superscript,
1546        render_expression(&arguments[family.analytic_argument_index()], table)
1547    )
1548}

1654fn comparison_link_of(reference: &ExpressionReference) -> Option<ComparisonLink<'_>> {
1655    // View an expression as one link of a comparison chain: a two-argument call to
1656    // a comparison builtin that chains. Anything else — a connective, a `\neq`, a
1657    // non-comparison call — yields None and so blocks the chained rendering.
1658    match &reference.kind {
1659        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments)
1660            if arguments.len() == 2 && comparison_chain_direction(*builtin).is_some() =>
1661        {
1662            Some(ComparisonLink {
1663                operator: *builtin,
1664                left: &arguments[0],
1665                right: &arguments[1],
1666            })
1667        }
1668        _ => None,
1669    }

1672fn comparison_chain_order(links: &[ComparisonLink<'_>]) -> Option<Vec<usize>> {
1673    // Order the links of a conjunction into the single path they spell, or None
1674    // when they do not spell exactly one. The path starts at the only left operand
1675    // that is no link's right operand, and each step takes the only link starting
1676    // where the previous one ended; a fork, a break, or a cycle leaves links
1677    // unvisited or the start ambiguous, and rejects the chain. Operands are
1678    // compared by pointer because the interner hash-conses, so structural equality
1679    // is pointer equality.
1680    let is_shared = |candidate: &ExpressionReference| {
1681        links.iter().any(|link| Arc::ptr_eq(link.right, candidate))
1682    };
1683    let mut starts = links
1684        .iter()
1685        .enumerate()
1686        .filter(|(_, link)| !is_shared(link.left));
1687    let (first, start_link) = starts.next()?;
1688    if starts.next().is_some() {
1689        return None;
1690    }
1691    let mut order = vec![first];
1692    let mut end = start_link.right;
1693    while order.len() < links.len() {
1694        let mut continuations = links
1695            .iter()
1696            .enumerate()
1697            .filter(|(index, link)| !order.contains(index) && Arc::ptr_eq(link.left, end));
1698        let (next, next_link) = continuations.next()?;
1699        if continuations.next().is_some() {
1700            return None;
1701        }
1702        order.push(next);
1703        end = next_link.right;
1704    }
1705    Some(order)
1706}

1709    arguments: &[ExpressionReference],
1710    table: &SymbolTable,
1711) -> Option<String> {
1712    // Render a conjunction of comparisons back as the chained relation notation
1713    // (`1 < x < 4`) when its conjuncts link end to start into one path whose
1714    // directions agree — the inverse of the parser's chain folding, so the chained
1715    // input form round-trips. Auto-simplification sorts a logical_and's operands,
1716    // so the written order is gone and the path is rebuilt here. None means the
1717    // conjunction is not one such chain and the caller renders it with \land.
1718    if arguments.len() < 2 {
1719        return None;
1720    }
1721    let links = arguments
1722        .iter()
1723        .map(comparison_link_of)
1724        .collect::<Option<Vec<_>>>()?;
1725    let operators = links.iter().map(|link| link.operator).collect::<Vec<_>>();
1726    if !is_chainable_comparison_sequence(&operators) {
1727        return None;
1728    }
1729    let order = comparison_chain_order(&links)?;
1730    let mut rendered = render_expression(links[order[0]].left, table);
1731    for index in order {
1732        let link = &links[index];
1733        rendered.push(' ');
1734        rendered.push_str(comparison_latex_operator(link.operator)?);
1735        rendered.push(' ');
1736        rendered.push_str(&render_expression(link.right, table));
1737    }
1738    Some(rendered)
1739}

1741fn render_infix_connective(
1742    arguments: &[ExpressionReference],
1743    table: &SymbolTable,
1744    operator: &str,
1745) -> String {
1746    // Render an n-ary logical connective as an infix chain joined by the operator,
1747    // parenthesising any operand that is itself a logical compound so the grouping
1748    // survives. A degenerate single argument renders bare; an empty argument list
1749    // falls back to a standard call so a malformed node still renders.
1750    if arguments.is_empty() {
1751        return render_standard_call(operator, arguments, table);
1752    }
1753    let separator = format!(" {} ", operator);
1754    arguments
1755        .iter()
1756        .map(|argument| render_logical_operand(argument, table))
1757        .collect::<Vec<_>>()
1758        .join(&separator)
1759}

1763    // the operand is a logical compound.
1764    if arguments.len() == 1 {
1765        format!("\\lnot {}", render_logical_operand(&arguments[0], table))
1766    } else {
1767        render_standard_call("\\lnot", arguments, table)
1768    }
1769}

1771fn render_logical_operand(reference: &ExpressionReference, table: &SymbolTable) -> String {
1772    // Render a connective operand, wrapping it in parentheses when it is itself a
1773    // logical compound (another connective) so the rendered precedence is correct.
1774    if is_logical_compound(reference) {
1775        format!("\\left({}\\right)", render_expression(reference, table))
1776    } else {
1777        render_expression(reference, table)
1778    }
1779}

1800    identifier: SymbolIdentifier,
1801    arguments: &[ExpressionReference],
1802    table: &SymbolTable,
1803) -> String {

1805    // All other user functions render as \mathrm{name}\left(args\right).

1819fn render_derivative_placeholder(arguments: &[ExpressionReference], table: &SymbolTable) -> String {
1820    // Render the deferred-differentiation placeholder from execute.rs. The
1821    // placeholder layout is (expression, variable_symbol, order_integer).
1822    // Renders as \frac{d^n expression}{d variable^n}\!\left(variable\right)
1823    // for a univariate ordinary derivative. Falls back to the generic
1824    // \mathrm{derivative}\left(...\right) form when the layout is unexpected.
1825    if arguments.len() != 3 {
1826        return render_standard_call("\\mathrm{derivative}", arguments, table);
1827    }
1828    let expression_latex = render_expression(&arguments[0], table);
1829    let variable_latex = render_expression(&arguments[1], table);
1830    let order_str = match &arguments[2].kind {
1831        ExpressionKind::Integer(value) => {
1832            if let Ok(signed) = i64::try_from(value) {
1833                signed.to_string()
1834            } else {
1835                return render_standard_call("\\mathrm{derivative}", arguments, table);
1836            }
1837        }
1838        _ => return render_standard_call("\\mathrm{derivative}", arguments, table),
1839    };
1840    let order: i64 = order_str.parse().unwrap_or(1);
1841    if order == 1 {
1842        format!(
1843            "\\frac{{\\mathrm{{d}}\\,{}}}{{\\mathrm{{d}}\\,{}}}\\!\\left({}\\right)",
1844            expression_latex, variable_latex, variable_latex
1845        )
1846    } else {
1847        format!(
1848            "\\frac{{\\mathrm{{d}}^{{{}}}\\,{}}}{{\\mathrm{{d}}\\,{}^{{{}}}}}\\!\\left({}\\right)",
1849            order, expression_latex, variable_latex, order, variable_latex
1850        )
1851    }
1852}

1854fn render_limit_placeholder(arguments: &[ExpressionReference], table: &SymbolTable) -> String {
1855    // Render the unevaluated-limit placeholder from limit.rs. The placeholder
1856    // layout is (target, variable_symbol, point, direction_marker) where
1857    // direction_marker is 0 (two-sided), -1 (from left), or 1 (from right).
1858    // Falls back to the generic form when the layout is unexpected.
1859    if arguments.len() != 4 {
1860        return render_standard_call("\\lim", arguments, table);
1861    }
1862    let target_latex = render_expression(&arguments[0], table);
1863    let variable_latex = render_expression(&arguments[1], table);
1864    let point_latex = render_expression(&arguments[2], table);
1865    let direction_suffix = match &arguments[3].kind {
1866        ExpressionKind::Integer(value) => {
1867            if let Ok(signed) = i64::try_from(value) {
1868                match signed {
1869                    0 => "",
1870                    -1 => "^{-}",
1871                    1 => "^{+}",
1872                    _ => return render_standard_call("\\lim", arguments, table),
1873                }
1874            } else {
1875                return render_standard_call("\\lim", arguments, table);
1876            }
1877        }
1878        _ => return render_standard_call("\\lim", arguments, table),
1879    };
1880    format!(
1881        "\\lim_{{{} \\to {}{}}} {}",
1882        variable_latex, point_latex, direction_suffix, target_latex
1883    )
1884}

1923fn render_derivative(
1924    function: FunctionIdentifier,
1925    orders: &[usize],
1926    arguments: &[ExpressionReference],
1927    table: &SymbolTable,
1928) -> String {
1929    // Render an unevaluated derivative AST node (the Faa di Bruno chain-rule
1930    // placeholder). Uses Leibniz notation. For a single non-zero order entry,
1931    // uses ordinary derivative d/dx; for multiple non-zero entries, uses partial
1932    // derivative notation \partial/\partial x. The univariate order-1 case
1933    // collapses to \frac{d\,f}{d\,x}.
1934    let function_latex = function_latex_name_for_derivative(function, table);
1935    let argument_latexes: Vec<String> = arguments
1936        .iter()
1937        .map(|a| render_expression(a, table))
1938        .collect();
1939    let argument_list = argument_latexes.join(", ");
1940
1941    let nonzero_indices: Vec<usize> = orders
1942        .iter()
1943        .enumerate()
1944        .filter(|&(_, order)| *order > 0)
1945        .map(|(index, _)| index)
1946        .collect();
1947    let total_order: usize = orders.iter().sum();
1948
1949    if nonzero_indices.len() == 1 {
1950        let index = nonzero_indices[0];
1951        let order = orders[index];
1952        let variable = &argument_latexes[index];
1953        if order == 1 {
1954            format!(
1955                "\\frac{{\\mathrm{{d}}\\,{}}}{{\\mathrm{{d}}\\,{}}}\\!\\left({}\\right)",
1956                function_latex, variable, argument_list
1957            )
1958        } else {
1959            format!(
1960                "\\frac{{\\mathrm{{d}}^{{{}}}\\,{}}}{{\\mathrm{{d}}\\,{}^{{{}}}}}\\!\\left({}\\right)",
1961                order, function_latex, variable, order, argument_list
1962            )
1963        }
1964    } else {
1965        let denominator_parts: Vec<String> = nonzero_indices
1966            .iter()
1967            .map(|&index| {
1968                let variable = &argument_latexes[index];
1969                let order = orders[index];
1970                if order == 1 {
1971                    format!("\\partial\\,{}", variable)
1972                } else {
1973                    format!("\\partial\\,{}^{{{}}}", variable, order)
1974                }
1975            })
1976            .collect();
1977        let denominator = denominator_parts.join("\\,");
1978        if total_order == 1 {
1979            format!(
1980                "\\frac{{\\partial\\,{}}}{{{}}}\\!\\left({}\\right)",
1981                function_latex, denominator, argument_list
1982            )
1983        } else {
1984            format!(
1985                "\\frac{{\\partial^{{{}}}\\,{}}}{{{}}}\\!\\left({}\\right)",
1986                total_order, function_latex, denominator, argument_list
1987            )
1988        }
1989    }
1990}

1992fn function_latex_name_for_derivative(function: FunctionIdentifier, table: &SymbolTable) -> String {
1993    // Resolve a function identifier to its LaTeX name for use inside a
1994    // derivative notation. Builtins use the standard \cmd form; user functions
1995    // use \mathrm{name}.
1996    match function {
1997        FunctionIdentifier::Builtin(builtin) => builtin_latex_name(builtin).to_owned(),
1998        FunctionIdentifier::User(identifier) => {
1999            let name = table.name(identifier);
2000            format!("\\mathrm{{{}}}", escape_latex_identifier(&name))
2001        }
2002    }
2003}

2007    variable: SymbolIdentifier,
2008    lower_bound: Option<&ExpressionReference>,
2009    upper_bound: Option<&ExpressionReference>,
2010    table: &SymbolTable,
2011) -> String {
2012    // Render an integral AST node. The indefinite case produces \int integrand
2013    // \, d\,variable; the definite case adds _{lower}^{upper} bounds.

2020            let upper_latex = render_expression(upper, table);

2022                "\\int_{{{}}}^{{{}}} {} \\, \\mathrm{{d}}\\,{}",
2023                lower_latex, upper_latex, integrand_latex, variable_latex
2024            )
2025        }
2026        _ => {
2027            format!(
2028                "\\int {} \\, \\mathrm{{d}}\\,{}",
2029                integrand_latex, variable_latex
2030            )
2031        }
2032    }

2092fn render_piecewise(
2093    branches: &[(ExpressionReference, ExpressionReference)],
2094    table: &SymbolTable,
2095) -> String {
2096    // Render a piecewise expression as a LaTeX cases environment.
2097    // The order of branches is significant (first-match semantics).
2098    let mut result = "\\begin{cases}".to_owned();
2099    for (index, (condition, value)) in branches.iter().enumerate() {
2100        if index > 0 {
2101            result.push_str(" \\\\");
2102        }
2103        let value_latex = render_expression(value, table);
2104        let condition_latex = render_expression(condition, table);
2105        result.push_str(&format!(
2106            " {} & \\text{{if }}\\,{}",
2107            value_latex, condition_latex
2108        ));
2109    }
2110    result.push_str(" \\end{cases}");
2111    result
2112}
