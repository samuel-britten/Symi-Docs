10/// Performs the advanced native `negate` operation.
11pub fn negate(element: &AlgebraicElement) -> AlgebraicElement {
12    // Coefficient-wise additive inverse. The dense representation makes this a
13    // linear-time copy with sign flips on every slot.
14    AlgebraicElement {
15        field: Arc::clone(&element.field),
16        representative: element
17            .representative
18            .iter()
19            .map(|coefficient| -coefficient.clone())
20            .collect(),
21    }
22}

24/// Performs the advanced native `add` operation.
25pub fn add(left: &AlgebraicElement, right: &AlgebraicElement) -> AlgebraicElement {
26    // Coefficient-wise addition. Both operands must live in the same field;
27    // when they don't, the debug_assert fires in development builds and the
28    // result silently uses the left field in release builds.
29    debug_assert!(Arc::ptr_eq(&left.field, &right.field));
30    let degree = left.field.degree;
31    let mut representative: Vec<exact_rational> = Vec::with_capacity(degree);
32    for index in 0..degree {
33        representative
34            .push(left.representative[index].clone() + right.representative[index].clone());
35    }
36    AlgebraicElement {
37        field: Arc::clone(&left.field),
38        representative,
39    }
40}

42/// Performs the advanced native `subtract` operation.
43pub fn subtract(left: &AlgebraicElement, right: &AlgebraicElement) -> AlgebraicElement {
44    // Coefficient-wise subtraction. Implemented directly rather than as
45    // add(left, negate(right)) so that the intermediate negation does not
46    // allocate.
47    debug_assert!(Arc::ptr_eq(&left.field, &right.field));
48    let degree = left.field.degree;
49    let mut representative: Vec<exact_rational> = Vec::with_capacity(degree);
50    for index in 0..degree {
51        representative
52            .push(left.representative[index].clone() - right.representative[index].clone());
53    }
54    AlgebraicElement {
55        field: Arc::clone(&left.field),
56        representative,
57    }
58}

60/// Performs the advanced native `multiply` operation.
61pub fn multiply(left: &AlgebraicElement, right: &AlgebraicElement) -> AlgebraicElement {
62    // Polynomial multiplication in Q[generator] followed by reduction modulo
63    // the defining polynomial. The intermediate product has degree at most
64    // 2·(field.degree − 1); the reduction collapses it back to length
65    // field.degree.
66    debug_assert!(Arc::ptr_eq(&left.field, &right.field));
67    let degree = left.field.degree;
68    let mut raw_product: Vec<exact_rational> =
69        vec![rational_zero(); 2 * degree.saturating_sub(1) + 1];
70    if degree == 0 {
71        return zero_element(&left.field);
72    }
73    for (left_index, left_coefficient) in left.representative.iter().enumerate() {
74        if *left_coefficient == rational_zero() {
75            continue;
76        }
77        for (right_index, right_coefficient) in right.representative.iter().enumerate() {
78            if *right_coefficient == rational_zero() {
79                continue;
80            }
81            let combined_index = left_index + right_index;
82            raw_product[combined_index] = raw_product[combined_index].clone()
83                + left_coefficient.clone() * right_coefficient.clone();
84        }
85    }
86    let reduced = reduce_modulo_defining_polynomial(&left.field, raw_product);
87    AlgebraicElement {
88        field: Arc::clone(&left.field),
89        representative: reduced,
90    }
91}

93/// Performs the advanced native `power_integer` operation.
94pub fn power_integer(
95    base: &AlgebraicElement,
96    exponent: &exact_integer,
97) -> Option<AlgebraicElement> {
98    // Raise an algebraic element to an integer exponent. Negative exponents
99    // first invert the base; the zero exponent returns the multiplicative
100    // identity. Magnitudes that exceed u64::MAX abort with None — in practice
101    // the limit is far above any realistic algebraic computation.
102    use std::cmp::Ordering;
103    match integer_sign(exponent) {
104        Ordering::Equal => Some(one_element(&base.field)),
105        Ordering::Greater => {
106            let magnitude = integer_to_u64(exponent)?;
107            Some(repeated_squaring(base, magnitude))
108        }
109        Ordering::Less => {
110            let inverted = inverse(base)?;
111            let positive_exponent = -exponent.clone();
112            let magnitude = integer_to_u64(&positive_exponent)?;
113            Some(repeated_squaring(&inverted, magnitude))
114        }
115    }
116}

118fn repeated_squaring(base: &AlgebraicElement, exponent: u64) -> AlgebraicElement {
119    // Square-and-multiply exponentiation over Q(α). The accumulator starts at
120    // one_element; on each iteration the low bit of the exponent decides
121    // whether the current square is folded into the result.
122    if exponent == 0 {
123        return one_element(&base.field);
124    }
125    if exponent == 1 {
126        return base.clone();
127    }
128    let mut result = one_element(&base.field);
129    let mut current = base.clone();
130    let mut remaining = exponent;
131    while remaining > 0 {
132        if remaining & 1 == 1 {
133            result = multiply(&result, &current);
134        }
135        remaining >>= 1;
136        if remaining > 0 {
137            current = multiply(&current, &current);
138        }
139    }
140    result
141}

143/// Performs the advanced native `inverse` operation.
144pub fn inverse(element: &AlgebraicElement) -> Option<AlgebraicElement> {
145    // Multiplicative inverse via the extended Euclidean algorithm on Q[α]
146    // between the element's representative and the defining polynomial. The
147    // defining polynomial is irreducible by construction, so the gcd of any
148    // non-zero representative with it is a non-zero constant; dividing
149    // through by that constant yields the Bezout coefficient that is the
150    // inverse. None is returned only for the zero element.
151    if is_zero(element) {
152        return None;
153    }
154    let defining = defining_polynomial_dense_rationals(&element.field)?;
155    let (gcd_vector, multiplier_left) =
156        extended_euclidean_dense(element.representative.clone(), defining);
157    let constant = gcd_vector.first().cloned()?;
158    if constant == rational_zero() {
159        return None;
160    }
161    let inverse_constant = rational_one() / constant;
162    let scaled: Vec<exact_rational> = multiplier_left
163        .into_iter()
164        .map(|coefficient| coefficient * inverse_constant.clone())
165        .collect();
166    let reduced = reduce_modulo_defining_polynomial(&element.field, scaled);
167    Some(AlgebraicElement {
168        field: Arc::clone(&element.field),
169        representative: reduced,
170    })
171}

173/// Performs the advanced native `divide` operation.
174pub fn divide(left: &AlgebraicElement, right: &AlgebraicElement) -> Option<AlgebraicElement> {
175    // Field division. Returns None precisely when the divisor is the zero
176    // element; otherwise computes multiply(left, inverse(right)).
177    debug_assert!(Arc::ptr_eq(&left.field, &right.field));
178    let inverted = inverse(right)?;
179    Some(multiply(left, &inverted))
180}

182/// Performs the advanced native `scale_by_rational` operation.
183pub fn scale_by_rational(element: &AlgebraicElement, factor: &exact_rational) -> AlgebraicElement {
184    // Multiply every coefficient in the representative by a rational scalar.
185    // The shortcut path through scalar multiplication is taken whenever the
186    // factor enters via the polynomial-coefficient bridge.
187    if *factor == rational_zero() {
188        return zero_element(&element.field);
189    }
190    AlgebraicElement {
191        field: Arc::clone(&element.field),
192        representative: element
193            .representative
194            .iter()
195            .map(|coefficient| coefficient.clone() * factor.clone())
196            .collect(),
197    }
198}

200fn reduce_modulo_defining_polynomial(
201    field: &Arc<NumberField>,
202    mut working: Vec<exact_rational>,
203) -> Vec<exact_rational> {
204    // Polynomial remainder against the defining polynomial. Operates in place
205    // on a working dense vector by repeatedly cancelling the leading term:
206    // subtracting (working_leading / 1) · (generator^j · defining_polynomial)
207    // from the working vector, where j is the degree gap. Returns a length-
208    // field.degree dense vector.
209    let degree = field.degree;
210    if working.len() <= degree {
211        working.resize(degree, rational_zero());
212        return working;
213    }
214    let defining = match defining_polynomial_dense_rationals(field) {
215        Some(value) => value,
216        None => {
217            working.truncate(degree);
218            return working;
219        }
220    };
221    while working.len() > degree {
222        let leading_index = working.len() - 1;
223        let leading_coefficient = working[leading_index].clone();
224        working.pop();
225        if leading_coefficient == rational_zero() {
226            continue;
227        }
228        let offset = leading_index - degree;
229        for inner_index in 0..degree {
230            working[offset + inner_index] = working[offset + inner_index].clone()
231                - leading_coefficient.clone() * defining[inner_index].clone();
232        }
233    }
234    if working.len() < degree {
235        working.resize(degree, rational_zero());
236    }
237    working
238}

240fn extended_euclidean_dense(
241    left: Vec<exact_rational>,
242    right: Vec<exact_rational>,
243) -> (Vec<exact_rational>, Vec<exact_rational>) {
244    // Iterative extended Euclidean algorithm on dense rational coefficient
245    // vectors. Returns (gcd, multiplier_left) such that
246    // multiplier_left * left + multiplier_right * right == gcd, with the
247    // multiplier_right discarded because the inverse computation does not
248    // need it. All polynomials are stored ascending-degree.
249    let mut old_remainder = trim_trailing_zeros(left);
250    let mut remainder = trim_trailing_zeros(right);
251    let mut old_multiplier = vec![rational_one()];
252    let mut multiplier: Vec<exact_rational> = Vec::new();
253    while !remainder.is_empty() {
254        let (quotient, new_remainder) = dense_divide(&old_remainder, &remainder);
255        let next_multiplier =
256            subtract_dense(&old_multiplier, &dense_multiply(&quotient, &multiplier));
257        old_remainder = remainder;
258        remainder = trim_trailing_zeros(new_remainder);
259        old_multiplier = multiplier;
260        multiplier = trim_trailing_zeros(next_multiplier);
261    }
262    (old_remainder, old_multiplier)
263}

265fn dense_divide(
266    dividend: &[exact_rational],
267    divisor: &[exact_rational],
268) -> (Vec<exact_rational>, Vec<exact_rational>) {
269    // Long division of dense ascending-degree rational vectors. Returns
270    // (quotient, remainder) with remainder of degree strictly less than the
271    // divisor's degree. The divisor must be non-empty; callers handle the
272    // zero-divisor case before invoking this helper.
273    let divisor_degree = divisor.len() - 1;
274    let leading_divisor = divisor[divisor_degree].clone();
275    let mut working = dividend.to_vec();
276    let mut quotient: Vec<exact_rational> = Vec::new();
277    while working.len() > divisor_degree && !working.is_empty() {
278        let working_degree = working.len() - 1;
279        if working_degree < divisor_degree {
280            break;
281        }
282        let working_leading = working[working_degree].clone();
283        let coefficient_ratio = working_leading / leading_divisor.clone();
284        let exponent = working_degree - divisor_degree;
285        if quotient.len() <= exponent {
286            quotient.resize(exponent + 1, rational_zero());
287        }
288        quotient[exponent] = coefficient_ratio.clone();
289        for inner_index in 0..=divisor_degree {
290            working[exponent + inner_index] = working[exponent + inner_index].clone()
291                - coefficient_ratio.clone() * divisor[inner_index].clone();
292        }
293        while let Some(last) = working.last() {
294            if *last == rational_zero() {
295                working.pop();
296            } else {
297                break;
298            }
299        }
300    }
301    (quotient, working)
302}

304fn dense_multiply(left: &[exact_rational], right: &[exact_rational]) -> Vec<exact_rational> {
305    // Standard schoolbook multiplication on dense ascending-degree vectors.
306    // Returns an empty vector when either operand is the zero polynomial.
307    if left.is_empty() || right.is_empty() {
308        return Vec::new();
309    }
310    let mut product: Vec<exact_rational> = vec![rational_zero(); left.len() + right.len() - 1];
311    for (left_index, left_coefficient) in left.iter().enumerate() {
312        if *left_coefficient == rational_zero() {
313            continue;
314        }
315        for (right_index, right_coefficient) in right.iter().enumerate() {
316            if *right_coefficient == rational_zero() {
317                continue;
318            }
319            product[left_index + right_index] = product[left_index + right_index].clone()
320                + left_coefficient.clone() * right_coefficient.clone();
321        }
322    }
323    product
324}

326fn subtract_dense(left: &[exact_rational], right: &[exact_rational]) -> Vec<exact_rational> {
327    // Coefficient-wise subtraction with implicit zero padding so the two
328    // operands need not have the same length.
329    let length = left.len().max(right.len());
330    let mut result: Vec<exact_rational> = Vec::with_capacity(length);
331    for index in 0..length {
332        let left_value = left.get(index).cloned().unwrap_or(rational_zero());
333        let right_value = right.get(index).cloned().unwrap_or(rational_zero());
334        result.push(left_value - right_value);
335    }
336    result
337}

339fn trim_trailing_zeros(mut value: Vec<exact_rational>) -> Vec<exact_rational> {
340    // Drop trailing zero entries so the dense vector's length equals the
341    // polynomial degree plus one. The Euclidean loop terminates when this
342    // function returns the empty vector for the current remainder.
343    while let Some(last) = value.last() {
344        if *last == rational_zero() {
345            value.pop();
346        } else {
347            break;
348        }
349    }
350    value
351}

360/// Performs the advanced native `add_scalar` operation.
361pub fn add_scalar(element: &AlgebraicElement, value: exact_rational) -> AlgebraicElement {
362    // Add a rational scalar to the constant slot of the representative without
363    // allocating a temporary algebraic_element for the scalar.
364    if value == rational_zero() {
365        return element.clone();
366    }
367    let mut representative = element.representative.clone();
368    if !representative.is_empty() {
369        representative[0] = representative[0].clone() + value;
370    }
371    AlgebraicElement {
372        field: Arc::clone(&element.field),
373        representative,
374    }
375}
