70impl PrimitiveIntegerPolynomial {
71    /// Provides the `from_rational_coefficients` operation on this native type.
72    pub fn from_rational_coefficients(
73        coefficients: &[exact_rational],
74    ) -> Option<PrimitiveIntegerPolynomial> {
75        // Normalize a nonconstant rational polynomial by clearing denominators,
76        // removing integer content, and choosing positive leading coefficient.
77        let mut normalized_rationals = coefficients.to_vec();
78        while normalized_rationals
79            .last()
80            .is_some_and(crate::exact_arithmetic::rational_is_zero)
81        {
82            normalized_rationals.pop();
83        }
84        if normalized_rationals.len() < 2 {
85            return None;
86        }
87        let common_denominator =
88            normalized_rationals
89                .iter()
90                .fold(natural_one(), |current, coefficient| {
91                    natural_least_common_multiple(
92                        &current,
93                        &rational_parts(coefficient).denominator,
94                    )
95                });
96        let mut integers: Vec<exact_integer> = normalized_rationals
97            .iter()
98            .map(|coefficient| {
99                let parts = rational_parts(coefficient);
100                let multiplier = &common_denominator / &parts.denominator;
101                parts.numerator * natural_to_integer(multiplier)
102            })
103            .collect();
104        let content = integers
105            .iter()
106            .fold(natural_zero(), |current, coefficient| {
107                if integer_is_zero(coefficient) {
108                    current
109                } else if current == natural_zero() {
110                    integer_greatest_common_divisor(coefficient, coefficient)
111                } else {
112                    let next = integer_greatest_common_divisor(
113                        &natural_to_integer(current.clone()),
114                        coefficient,
115                    );
116                    next
117                }
118            });
119        if content != natural_zero() && content != natural_one() {
120            let signed_content = natural_to_integer(content);
121            for coefficient in integers.iter_mut() {
122                *coefficient /= &signed_content;
123            }
124        }
125        if crate::exact_arithmetic::integer_sign(integers.last().unwrap()) == Ordering::Less {
126            for coefficient in integers.iter_mut() {
127                *coefficient = -coefficient.clone();
128            }
129        }
130        Some(PrimitiveIntegerPolynomial {
131            coefficients: integers,
132        })
133    }
134
135    /// Provides the `degree` operation on this native type.
136    pub fn degree(&self) -> usize {
137        self.coefficients.len() - 1
138    }
139
140    /// Provides the `rational_coefficients` operation on this native type.
141    pub fn rational_coefficients(&self) -> Vec<exact_rational> {
142        self.coefficients
143            .iter()
144            .cloned()
145            .map(exact_rational::from)
146            .collect()
147    }
148}

150impl ComplexIsolatingRectangle {
151    /// Provides the `checked` operation on this native type.
152    pub fn checked(
153        real_lower: exact_rational,
154        real_upper: exact_rational,
155        imaginary_lower: exact_rational,
156        imaginary_upper: exact_rational,
157    ) -> Option<ComplexIsolatingRectangle> {
158        // Construct a nondegenerate rational rectangle; root certification is a
159        // separate operation performed by the exact rectangle counter.
160        if real_lower >= real_upper || imaginary_lower >= imaginary_upper {
161            return None;
162        }
163        Some(ComplexIsolatingRectangle {
164            real_lower,
165            real_upper,
166            imaginary_lower,
167            imaginary_upper,
168        })
169    }
170}

172impl AlgebraicNumber {
173    /// Provides the `is_real` operation on this native type.
174    pub fn is_real(&self) -> bool {
175        matches!(self.isolating_region, AlgebraicIsolatingRegion::Real(_))
176    }
177
178    /// Provides the `rational_value` operation on this native type.
179    pub fn rational_value(&self) -> Option<exact_rational> {
180        // An irreducible algebraic number is rational exactly in degree one.
181        if self.minimal_polynomial.degree() != 1 {
182            return None;
183        }
184        let constant = exact_rational::from(self.minimal_polynomial.coefficients[0].clone());
185        let leading = exact_rational::from(self.minimal_polynomial.coefficients[1].clone());
186        Some(-constant / leading)
187    }
188
189    /// Provides the `exact_real_sign` operation on this native type.
190    pub fn exact_real_sign(&self) -> Option<i32> {
191        // Decide the sign of a real embedded root by an exact sign-at-root query.
192        let interval = match &self.isolating_region {
193            AlgebraicIsolatingRegion::Real(interval) => interval,
194            AlgebraicIsolatingRegion::NonReal(_) => return None,
195        };
196        let numeric: Vec<Number> = self
197            .minimal_polynomial
198            .coefficients
199            .iter()
200            .cloned()
201            .map(Number::Integer)
202            .collect();
203        let defining = univariate_from_dense(SymbolIdentifier(u32::MAX - 3), numeric);
204        let coordinate = univariate_from_dense(
205            SymbolIdentifier(u32::MAX - 3),
206            vec![Number::Integer(0.into()), Number::Integer(1.into())],
207        );
208        sign_at_real_root(&defining, interval, &coordinate)
209    }
210
211    /// Provides the `is_zero` operation on this native type.
212    pub fn is_zero(&self) -> bool {
213        self.rational_value()
214            .is_some_and(|value| crate::exact_arithmetic::rational_is_zero(&value))
215    }
