9/// The result of an explicitly requested numerical definite integral.
10#[derive(Debug, Clone)]
11pub enum NumericDefiniteIntegrationOutcome {
12    /// A sequence of independent Gauss--Legendre rules reached the requested tolerance.
13    Evaluated {
14        /// The highest-order quadrature value.
15        value: ComplexInexactNumber,
16        /// The magnitude of the final two quadrature values.  This is an estimate, not an enclosure.
17        estimated_absolute_error: RealInexactNumber,
18        /// The arithmetic precision used for the reported value.
19        precision_bits: u32,
20        /// The number of nodes in the accepted rule.
21        node_count: usize,
22    },
23    /// The integrand, bounds, or convergence check was outside this finite-interval method.
24    Declined,
25}
26
27/// Numerically integrates an explicitly requested finite-interval integral.
28pub fn integrate_finite_interval(
29    context: &Context,
30    integrand: &ExpressionReference,
31    variable: SymbolIdentifier,
32    lower_bound: &ExpressionReference,
33    upper_bound: &ExpressionReference,
34    precision_bits: u32,
35) -> NumericDefiniteIntegrationOutcome {
36    let requested_precision = precision_bits.max(32);
37    let Some(lower_value) = evaluate_to_complex_inexact(context, lower_bound, requested_precision)
38    else {
39        return NumericDefiniteIntegrationOutcome::Declined;
40    };
41    let Some(upper_value) = evaluate_to_complex_inexact(context, upper_bound, requested_precision)
42    else {
43        return NumericDefiniteIntegrationOutcome::Declined;
44    };
45    let Some(lower_real) = lower_value.project_to_real() else {
46        return NumericDefiniteIntegrationOutcome::Declined;
47    };
48    let Some(upper_real) = upper_value.project_to_real() else {
49        return NumericDefiniteIntegrationOutcome::Declined;
50    };
51    if !lower_real.to_f64().is_some_and(f64::is_finite)
52        || !upper_real.to_f64().is_some_and(f64::is_finite)
53    {
54        return NumericDefiniteIntegrationOutcome::Declined;
55    }
56
57    let working_precision = requested_precision.saturating_add(32);
58    // The rule-difference is an error estimate, not a rigorous enclosure.  Cap its requested
59    // scale at 64 bits: demanding more digits from an estimate made by two related rules would
60    // falsely advertise a proof of the quadrature remainder.
61    let tolerance = 2.0_f64.powi(-((requested_precision / 2).min(64) as i32));
62    let mut previous_value = None;
63    for node_count in [16_usize, 32, 64, 128, 256] {
64        let Some(value) = gauss_legendre_rule(
65            context,
66            integrand,
67            variable,
68            &lower_real,
69            &upper_real,
70            working_precision,
71            node_count,
72        ) else {
73            return NumericDefiniteIntegrationOutcome::Declined;
74        };
75        if let Some(previous) = previous_value {
76            let evaluator = match NumericEvaluator::new(working_precision) {
77                Some(evaluator) => evaluator,
78                None => return NumericDefiniteIntegrationOutcome::Declined,
79            };
80            let error = value.subtract(&previous, &evaluator).magnitude(&evaluator);
81            if error.to_f64().is_some_and(|estimate| estimate <= tolerance) {
82                return NumericDefiniteIntegrationOutcome::Evaluated {
83                    value,
84                    estimated_absolute_error: error,
85                    precision_bits: working_precision,
86                    node_count,
87                };
88            }
89        }
90        previous_value = Some(value);
91    }
92    NumericDefiniteIntegrationOutcome::Declined
93}

95fn gauss_legendre_rule(
96    context: &Context,
97    integrand: &ExpressionReference,
98    variable: SymbolIdentifier,
99    lower_bound: &RealInexactNumber,
100    upper_bound: &RealInexactNumber,
101    precision_bits: u32,
102    node_count: usize,
103) -> Option<ComplexInexactNumber> {
104    let mut evaluator = NumericEvaluator::new(precision_bits)?;
105    let half = RealInexactNumber::from_f64(0.5, precision_bits);
106    let interval_width = upper_bound.subtract(lower_bound, &evaluator);
107    let mut total = ComplexInexactNumber::from_real(RealInexactNumber::zero(precision_bits));
108    for root_index in 0..node_count / 2 {
109        let (node, weight) =
110            gauss_legendre_node_and_weight(root_index, node_count, &mut evaluator)?;
111        let (positive_point, positive_jacobian) =
112            endpoint_smoothed_point(lower_bound, &interval_width, &node, &half, &evaluator);
113        let negative_node = node.negate();
114        let (negative_point, negative_jacobian) = endpoint_smoothed_point(
115            lower_bound,
116            &interval_width,
117            &negative_node,
118            &half,
119            &evaluator,
120        );
121        let positive_value =
122            evaluate_integrand(context, integrand, variable, positive_point, precision_bits)?;
123        let negative_value =
124            evaluate_integrand(context, integrand, variable, negative_point, precision_bits)?;
125        let positive_contribution = positive_value.multiply(
126            &ComplexInexactNumber::from_real(positive_jacobian),
127            &evaluator,
128        );
129        let negative_contribution = negative_value.multiply(
130            &ComplexInexactNumber::from_real(negative_jacobian),
131            &evaluator,
132        );
133        total = total.add(
134            &positive_contribution
135                .add(&negative_contribution, &evaluator)
136                .multiply(&ComplexInexactNumber::from_real(weight), &evaluator),
137            &evaluator,
138        );
139    }
140    Some(total)
141}

143fn endpoint_smoothed_point(
144    lower_bound: &RealInexactNumber,
145    interval_width: &RealInexactNumber,
146    node: &RealInexactNumber,
147    half: &RealInexactNumber,
148    evaluator: &NumericEvaluator,
149) -> (RealInexactNumber, RealInexactNumber) {
150    // The rational map u^8 / (u^8 + (1-u)^8) has eighth-order contact with both
151    // endpoints.  It keeps finite logarithmic and algebraic endpoint behaviour away from
152    // Gauss--Legendre's polynomial-error bottleneck without sampling a singular endpoint.
153    let one = RealInexactNumber::one(evaluator.requested_precision_bits());
154    let parameter = node.add(&one, evaluator).multiply(half, evaluator);
155    let complement = one.subtract(&parameter, evaluator);
156    let parameter_power = integer_power(&parameter, 8, evaluator);
157    let complement_power = integer_power(&complement, 8, evaluator);
158    let denominator = parameter_power.add(&complement_power, evaluator);
159    let quotient = parameter_power.divide(&denominator, evaluator);
160    let point = lower_bound.add(&interval_width.multiply(&quotient, evaluator), evaluator);
161    let numerator = RealInexactNumber::from_f64(4.0, evaluator.requested_precision_bits())
162        .multiply(&integer_power(&parameter, 7, evaluator), evaluator)
163        .multiply(&integer_power(&complement, 7, evaluator), evaluator);
164    let jacobian = interval_width.multiply(
165        &numerator.divide(&denominator.multiply(&denominator, evaluator), evaluator),
166        evaluator,
167    );
168    (point, jacobian)
169}

171fn integer_power(
172    base: &RealInexactNumber,
173    exponent: usize,
174    evaluator: &NumericEvaluator,
175) -> RealInexactNumber {
176    let mut result = RealInexactNumber::one(evaluator.requested_precision_bits());
177    for _ in 0..exponent {
178        result = result.multiply(base, evaluator);
179    }
180    result
181}

195fn gauss_legendre_node_and_weight(
196    root_index: usize,
197    node_count: usize,
198    evaluator: &mut NumericEvaluator,
199) -> Option<(RealInexactNumber, RealInexactNumber)> {
200    let precision = evaluator.requested_precision_bits();
201    let mut node = RealInexactNumber::from_f64(
202        (std::f64::consts::PI * (root_index as f64 + 0.75) / (node_count as f64 + 0.5)).cos(),
203        precision,
204    );
205    for _ in 0..24 {
206        let (polynomial, derivative) =
207            legendre_polynomial_and_derivative(node.clone(), node_count, evaluator);
208        node = node.subtract(&polynomial.divide(&derivative, evaluator), evaluator);
209    }
210    let (_, derivative) = legendre_polynomial_and_derivative(node.clone(), node_count, evaluator);
211    let one = RealInexactNumber::one(precision);
212    let two = RealInexactNumber::from_f64(2.0, precision);
213    let denominator = one
214        .subtract(&node.multiply(&node, evaluator), evaluator)
215        .multiply(&derivative.multiply(&derivative, evaluator), evaluator);
216    Some((node, two.divide(&denominator, evaluator)))
217}

219fn legendre_polynomial_and_derivative(
220    node: RealInexactNumber,
221    degree: usize,
222    evaluator: &NumericEvaluator,
223) -> (RealInexactNumber, RealInexactNumber) {
224    let precision = evaluator.requested_precision_bits();
225    if degree == 0 {
226        return (
227            RealInexactNumber::one(precision),
228            RealInexactNumber::zero(precision),
229        );
230    }
231    let mut preceding_polynomial = RealInexactNumber::one(precision);
232    let mut current_polynomial = node.clone();
233    let mut preceding_derivative = RealInexactNumber::zero(precision);
234    let mut current_derivative = RealInexactNumber::one(precision);
235    for current_degree in 2..=degree {
236        let leading = RealInexactNumber::from_f64((2 * current_degree - 1) as f64, precision);
237        let trailing = RealInexactNumber::from_f64((current_degree - 1) as f64, precision);
238        let divisor = RealInexactNumber::from_f64(current_degree as f64, precision);
239        let next_polynomial = leading
240            .multiply(&node, evaluator)
241            .multiply(&current_polynomial, evaluator)
242            .subtract(
243                &trailing.multiply(&preceding_polynomial, evaluator),
244                evaluator,
245            )
246            .divide(&divisor, evaluator);
247        let next_derivative = leading
248            .multiply(
249                &current_polynomial.add(&node.multiply(&current_derivative, evaluator), evaluator),
250                evaluator,
251            )
252            .subtract(
253                &trailing.multiply(&preceding_derivative, evaluator),
254                evaluator,
255            )
256            .divide(&divisor, evaluator);
257        preceding_polynomial = current_polynomial;
258        current_polynomial = next_polynomial;
259        preceding_derivative = current_derivative;
260        current_derivative = next_derivative;
261    }
262    (current_polynomial, current_derivative)
263}

274    #[test]
275    fn integrates_a_polynomial_and_reversed_bounds() {
276        let context = Context::new();
277        let variable = symbol(&context, "x");
278        let variable_identifier = match variable.kind {
279            ExpressionKind::Symbol(identifier) => identifier,
280            _ => panic!("builder must create a symbol"),
281        };
282        let integrand = power(&context, variable, integer_value(&context, 2));
283        let lower = integer_value(&context, 1);
284        let upper = integer_value(&context, 0);
285        let NumericDefiniteIntegrationOutcome::Evaluated { value, .. } = integrate_finite_interval(
286            &context,
287            &integrand,
288            variable_identifier,
289            &lower,
290            &upper,
291            96,
292        ) else {
293            panic!("smooth polynomial must converge");
294        };
295        assert!((value.real_part().to_f64().expect("finite value") + 1.0 / 3.0).abs() < 1e-24);
296        assert!(value.imaginary_part().is_zero());
297    }

299    #[test]
300    fn integrates_complex_values_without_symbolic_integration() {
301        let context = Context::new();
302        let variable = symbol(&context, "x");
303        let variable_identifier = match variable.kind {
304            ExpressionKind::Symbol(identifier) => identifier,
305            _ => panic!("builder must create a symbol"),
306        };
307        let imaginary_unit = context
308            .interner
309            .intern(ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit));

311        let lower = integer_value(&context, 0);
312        let upper = integer_value(&context, 1);
313        let NumericDefiniteIntegrationOutcome::Evaluated { value, .. } = integrate_finite_interval(
314            &context,
315            &integrand,
316            variable_identifier,
317            &lower,
318            &upper,
319            96,
320        ) else {
321            panic!("smooth complex integral must converge");
322        };
323        assert!(value.real_part().is_zero());
324        assert!((value.imaginary_part().to_f64().expect("finite value") - 0.5).abs() < 1e-24);
325    }
