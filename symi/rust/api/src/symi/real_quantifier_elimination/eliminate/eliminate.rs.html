12/// Performs the advanced native `eliminate_quantifiers` operation.
13pub fn eliminate_quantifiers(
14    context: &Context,
15    formula: &TarskiFormula,
16) -> Option<ExpressionReference> {
17    // The top-level real quantifier-elimination driver (Decisions 4, 5, 6). It
18    // collects the matrix's atom polynomials as the CAD's sign-invariant input, routes the
19    // top-level equality conjuncts into the CAD problem's equational-constraint slot
20    // (Decision 5), runs
21    // the pruned partial CAD, propagates the whole quantifier prefix innermost-out, and
22    // constructs the verified free-variable solution formula. The result is a
23    // quantifier-free expression equivalent to `formula` over ℝ (a Boolean constant for a
24    // fully-quantified sentence). None outside ℚ[x₁,…,xₙ], on an honest
25    // real-algebraic-number-layer lift decline,
26    // or when the solution formula cannot be certified (the conservative decline).
27    //
28    // Equational-constraint routing is sound here precisely because the equality conjunct
29    // is required of every solution: off the variety the equality atom is false, so the
30    // whole conjunction is false on every off-variety cell (the constraint, being in the
31    // EC projection, is sign-invariant per cell), and the EC contract guarantees
32    // sign-invariance for the remaining atoms on the variety — exactly where a solution can
33    // lie. The solution formula is verified exact at every cell sample point regardless.
34    record_counter!(quantifier_elimination_invocations);
35    let atoms = atom_polynomials(&formula.matrix);
36    let mut problem = CadProblem::new(context, atoms.clone(), formula.variables.clone())?;
37    let equational_polynomials = top_level_equational_polynomials(&formula.matrix);
38    problem.equational_constraints = equational_polynomials
39        .iter()
40        .filter_map(|polynomial| atoms.iter().position(|atom| Arc::ptr_eq(atom, polynomial)))
41        .collect();
42    let decomposition = partial_cylindrical_algebraic_decomposition(
43        context,
44        &problem,
45        &formula.prefix,
46        &formula.matrix,
47    )?;
48    construct_solution_formula(context, &decomposition)
49}
