16/// Performs the advanced native `construct_solution_formula` operation.
17pub fn construct_solution_formula(
18    context: &Context,
19    decomposition: &PartialDecomposition,
20) -> Option<ExpressionReference> {
21    // Build the quantifier-free solution formula in the free variables from a partial CAD.
22    // For a fully-quantified sentence the result is the single Boolean constant.
23    // For a formula with free variables, each true free cell is described by the
24    // conjunction of its sign conditions on the free-variable projection polynomials and
25    // the cells are disjoined (Hong's simple solution formula). The construction is then
26    // VERIFIED exact: the formula is evaluated at every free cell's sample point and must
27    // match the cell's quantified truth. Because every atom is a free projection
28    // polynomial, the formula's truth is constant on each (sign-invariant) free cell, so
29    // agreement at the sample points certifies agreement everywhere. On a mismatch
30    // (projection-definability failed — Brown's incremental augmentation is not yet
31    // implemented) the construction declines (None) rather than
32    // returning a possibly-inexact formula, so the QE output is never wrong.
33    if let Some(truth) = decomposition.sentence_truth {
34        return Some(logic::constant(context, truth));
35    }
36    let projection_polynomials = free_projection_polynomials(context, decomposition);
37    let formula = build_disjunction(context, decomposition, &projection_polynomials)?;
38    if !verify_solution_formula(context, &formula, decomposition) {
39        return None;
40    }
41    Some(formula)
42}

44fn free_projection_polynomials(
45    context: &Context,
46    decomposition: &PartialDecomposition,
47) -> Vec<ExpressionReference> {
48    // The projection polynomials that are functions of the free variables only — the
49    // levels 1..number_free of the tower — as deduplicated expressions. These are the
50    // only polynomials whose sign conditions can appear in a free-space solution formula,
51    // and each is sign-invariant on every free cell (it defines that cell's level or an
52    // ancestor's). For the no-quantifier case (number_free = n) this is the whole tower,
53    // including the input atom polynomials.
54    let mut expressions: Vec<ExpressionReference> = Vec::new();
55    for level in 0..decomposition.number_free {
56        for polynomial in decomposition.tower[level].iter() {
57            let expression = polynomial_to_expression(context, polynomial);
58            if !expressions
59                .iter()
60                .any(|existing| Arc::ptr_eq(existing, &expression))
61            {
62                expressions.push(expression);
63            }
64        }
65    }
66    expressions
67}

69fn build_disjunction(
70    context: &Context,
71    decomposition: &PartialDecomposition,
72    projection_polynomials: &[ExpressionReference],
73) -> Option<ExpressionReference> {
74    // The disjunction over the true free cells of each cell's sign-condition conjunction.
75    // No true cell yields logical_false; every cell true yields logical_true (the cells
76    // partition free space); otherwise the sign-condition DNF, built through the
77    // auto-simplifying logical builders.
78    let true_cells: Vec<&RealAlgebraicPoint> = decomposition
79        .free_cell_truths
80        .iter()
81        .filter(|(_, truth)| *truth)
82        .map(|(point, _)| point)
83        .collect();
84    if true_cells.is_empty() {
85        return Some(logic::constant(context, false));
86    }
87    if true_cells.len() == decomposition.free_cell_truths.len() {
88        return Some(logic::constant(context, true));
89    }
90    let mut clauses: Vec<ExpressionReference> = Vec::with_capacity(true_cells.len());
91    for point in true_cells {
92        clauses.push(sign_condition_conjunction(
93            context,
94            projection_polynomials,
95            point,
96        )?);
97    }
98    Some(builders::function_call(
99        context,
100        FunctionIdentifier::Builtin(BuiltinFunction::LogicalOr),
101        clauses,
102    ))
103}

105fn sign_condition_conjunction(
106    context: &Context,
107    projection_polynomials: &[ExpressionReference],
108    point: &RealAlgebraicPoint,
109) -> Option<ExpressionReference> {
110    // The conjunction of sign conditions `pᵢ Rᵢ 0` describing the free cell containing
111    // `point`: each projection polynomial's exact sign at the sample point fixes its
112    // relation (> 0, = 0, or < 0). None when a sign is undecided (the honest decline).
113    let mut atoms: Vec<ExpressionReference> = Vec::with_capacity(projection_polynomials.len());
114    for polynomial in projection_polynomials.iter() {
115        let sign = sign_of_polynomial_at_real_point(polynomial, point)?;
116        atoms.push(sign_condition_atom(context, polynomial, sign));
117    }
118    Some(builders::function_call(
119        context,
120        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
121        atoms,
122    ))
123}

125fn sign_condition_atom(
126    context: &Context,
127    polynomial: &ExpressionReference,
128    sign: i32,
129) -> ExpressionReference {
130    // The relation atom `polynomial R 0` for the polynomial's sign: > 0 for a positive
131    // sign, = 0 for zero, < 0 for negative.
132    let relation = if sign > 0 {
133        AtomicRelation::Greater
134    } else if sign < 0 {
135        AtomicRelation::Less
136    } else {
137        AtomicRelation::Equal
138    };
139    let zero = builders::integer(context, crate::exact_arithmetic::integer_zero());
140    builders::function_call(
141        context,
142        FunctionIdentifier::Builtin(relation.to_builtin()),
143        vec![Arc::clone(polynomial), zero],
144    )
145}

147fn verify_solution_formula(
148    context: &Context,
149    formula: &ExpressionReference,
150    decomposition: &PartialDecomposition,
151) -> bool {
152    // The rigor net (Decision 4): the constructed formula must reproduce every free
153    // cell's quantified truth at its sample point. Because the formula's atoms are
154    // sign-invariant free projection polynomials, sample-point agreement certifies
155    // agreement on the whole cell, hence everywhere. A single mismatch (or an undecided
156    // evaluation) fails verification and forces the honest decline.
157    // Each cell's check reads the finished formula and its own immutable sample point, so the
158    // checks are independent and may be decided as one batch; they are then read back in cell
159    // order, and every one of them must succeed. A check that fails is a failed verification
160    // however many others passed, and the first failing cell in canonical order is the one that
161    // ends the verification, so a later cell finishing first changes nothing.
162    let route = verification_route(context, decomposition);
163    let cancellation = context.operation().cancellation_token();
164    let verdicts = ordered_map(route, &decomposition.free_cell_truths, |(point, truth)| {
165        if cancellation.is_cancelled() {
166            return false;
167        }
168        matches!(evaluate_formula_at_point(formula, point), Some(value) if value == *truth)
169    });
170    verdicts.into_iter().all(|verdict| verdict)
171}

173fn verification_route(context: &Context, decomposition: &PartialDecomposition) -> ExecutionRoute {
174    // The decision for the verification batch. Each cell costs one evaluation of the whole
175    // solution formula at an algebraic sample point — an exact sign decision for every sign
176    // condition the formula carries — so the work is the cell count times the number of free
177    // projection polynomials the construction used; task-local memory is the one truth a check
178    // returns.
179    if !context
180        .operation()
181        .may_fork_work_class("formula_atom_evaluation")
182    {
183        return ExecutionRoute::Serial;
184    }
185    let atom_count: usize = decomposition
186        .tower
187        .iter()
188        .take(decomposition.number_free)
189        .map(|level| level.len())
190        .sum();
191    let cell_count = decomposition.free_cell_truths.len();
192    let domain_work_units = (cell_count as u64).saturating_mul(atom_count as u64 * 4);
193    let predicted_task_local_bytes = atom_count.saturating_mul(64).max(256);
194    let worker_count = crate::parallelism::maximum_operation_concurrency();
195    let requested_permits = cell_count.min(worker_count.max(1));
196    context
197        .operation()
198        .execution_decision(
199            &WorkEstimate::new(
200                domain_work_units,
201                cell_count,
202                predicted_task_local_bytes,
203                0,
204                requested_permits,
205                "formula_atom_evaluation",
206            ),
207            requested_permits,
208            crate::parallelism::runtime_memory_capacity(predicted_task_local_bytes),
209        )
210        .route
211}
