377/// Public native enumeration named `comparison_direction`.
378pub enum ComparisonDirection {
379    /// Represents the `increasing` case.
380    Increasing,
381    /// Represents the `decreasing` case.
382    Decreasing,
383    /// Represents the `equality` case.
384    Equality,
385}
386
387/// Performs the advanced native `comparison_chain_direction` operation.
388pub fn comparison_chain_direction(function: BuiltinFunction) -> Option<ComparisonDirection> {
389    // Classify a comparison builtin by the direction it runs in when it appears in
390    // a chained relation such as `1 < x < 4`. `not_equal` is deliberately absent:
391    // the conventional reading of `a \neq b \neq c` is pairwise distinctness, which
392    // is not the conjunction of adjacent comparisons, so it never chains.
393    match function {
394        BuiltinFunction::LessThan | BuiltinFunction::LessThanOrEqual => {
395            Some(ComparisonDirection::Increasing)
396        }
397        BuiltinFunction::GreaterThan | BuiltinFunction::GreaterThanOrEqual => {
398            Some(ComparisonDirection::Decreasing)
399        }
400        BuiltinFunction::Equal => Some(ComparisonDirection::Equality),
401        _ => None,
402    }
403}
404
405/// Performs the advanced native `is_chainable_comparison_sequence` operation.
406pub fn is_chainable_comparison_sequence(operators: &[BuiltinFunction]) -> bool {
407    // Whether a sequence of comparison builtins may be written as one chain. Every
408    // operator must be chainable, and the strict directions present must agree:
409    // `1 < x \leq 4` and `a = b < c` chain, `1 < x > 3` does not. This is the one
410    // rule shared by the LaTeX parser, which accepts such chains, and the LaTeX
411    // renderer, which folds a conjunction back into one — keeping them in step is
412    // what makes the chained form round-trip.
413    let mut strict_direction = None;
414    for operator in operators {
415        match comparison_chain_direction(*operator) {
416            None => return false,
417            Some(ComparisonDirection::Equality) => {}
418            Some(direction) => match strict_direction {
419                Some(established) if established != direction => return false,
420                _ => strict_direction = Some(direction),
421            },
422        }
423    }
424    true
425}

427/// Performs the advanced native `function_name` operation.
428pub fn function_name(function: FunctionIdentifier, table: &SymbolTable) -> ResolvedName {
429    // Resolve any FunctionIdentifier — builtin or user-defined — back to its name
430    // string. User functions are looked up through the supplied symbol_table.
431    match function {
432        FunctionIdentifier::Builtin(builtin) => ResolvedName::Builtin(builtin_name(builtin)),
433        FunctionIdentifier::User(identifier) => ResolvedName::Published(table.name(identifier)),
434    }
435}

443        assert_eq!(builtin_name(BuiltinFunction::Sin), "sin");

450        let table = SymbolTable::new();
451        let identifier = table.intern_symbol("my_function");
452        let resolved = function_name(FunctionIdentifier::User(identifier), &table);
453        assert_eq!(resolved, "my_function");
454    }

456    #[test]
457    fn comparison_chain_direction_classifies_the_chainable_operators() {
458        assert_eq!(
459            comparison_chain_direction(BuiltinFunction::LessThan),
460            Some(ComparisonDirection::Increasing)
461        );
462        assert_eq!(
463            comparison_chain_direction(BuiltinFunction::LessThanOrEqual),
464            Some(ComparisonDirection::Increasing)
465        );
466        assert_eq!(
467            comparison_chain_direction(BuiltinFunction::GreaterThan),
468            Some(ComparisonDirection::Decreasing)
469        );
470        assert_eq!(
471            comparison_chain_direction(BuiltinFunction::GreaterThanOrEqual),
472            Some(ComparisonDirection::Decreasing)
473        );
474        assert_eq!(
475            comparison_chain_direction(BuiltinFunction::Equal),
476            Some(ComparisonDirection::Equality)
477        );
478        assert_eq!(comparison_chain_direction(BuiltinFunction::NotEqual), None);
479        assert_eq!(comparison_chain_direction(BuiltinFunction::Sin), None);
480    }

484        assert!(is_chainable_comparison_sequence(&[
485            BuiltinFunction::LessThan,
486            BuiltinFunction::LessThanOrEqual,

488        assert!(is_chainable_comparison_sequence(&[
489            BuiltinFunction::GreaterThanOrEqual,
490            BuiltinFunction::GreaterThan,
491        ]));
492        assert!(is_chainable_comparison_sequence(&[
493            BuiltinFunction::Equal,
494            BuiltinFunction::Equal,
495        ]));
496    }
