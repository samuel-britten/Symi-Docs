22/// The shipping disposition of one work class.
23///
24/// Automatic execution schedules a boundary only after that boundary has demonstrated a stable
25/// end-to-end benefit over the final optimized serial implementation. A boundary that is correct
26/// but unprofitable is not an incomplete implementation: it stays in the tree, stays directly
27/// exercisable, and stays out of reach of the automatic route. Making that a state rather than a
28/// cutoff nobody can clear is the difference between a policy a reader can check and a magic
29/// number they have to reverse-engineer.
30#[derive(Clone, Copy, Debug, Eq, PartialEq)]
31pub(crate) enum WorkClassAdmission {
32    /// A measured cost model admits automatic parallel execution once the estimate reaches the
33    /// cutoff and the host publishes enough workers for the measured gain to exist.
34    Automatic {
35        /// The work units the estimate must present before the coordinator may form tasks.
36        work_unit_cutoff: u64,
37        /// The concurrent tasks one operation must be allowed before this class may form them.
38        ///
39        /// A gain is a measurement taken at a worker count, and several of these classes are
40        /// interner-bound: they gain on four workers and lose on two, because two workers split
41        /// the same contention over half the throughput. Admitting such a class at every capacity
42        /// would ship a measured loss when the per-operation cap permits fewer concurrent tasks
43        /// than the measurement did, so the useful concurrency the gain was measured at is part
44        /// of the disposition rather than an assumption about the host.
45        minimum_operation_concurrency: usize,
46        /// Identifies the enduring serial workload and scaling dimension that justified promotion.
47        serial_baseline: SerialBaselineEvidence,
48    },
49    /// The parallel implementation may be exercised directly, but automatic execution cannot
50    /// select it. A class whose scaling evidence is absent, stale, or negative is in this state.
51    SerialOnly,
52}

527/// Initialize the owned process-wide Rayon pool with distinct pool and operation capacities.
528fn initialize_native_parallelism_with_operation_concurrency(
529    requested_worker_count: usize,
530    requested_maximum_operation_concurrency: usize,
531) -> Result<RuntimeInitialization, RuntimeInitializationError> {
532    #[cfg(not(target_arch = "wasm32"))]
533    {
534        // A rejected argument is the caller's mistake, not a runtime failure: publishing a
535        // failure record here would retire the process-wide pool before it ever existed and
536        // leave every later well-formed request answering `initialization_failed`.
537        if requested_worker_count == 0
538            || requested_maximum_operation_concurrency == 0
539            || requested_maximum_operation_concurrency > requested_worker_count
540        {
541            return Err(RuntimeInitializationError::InvalidWorkerCount);
542        }
543
544        if let Some(runtime_state) = NATIVE_RUNTIME.get() {
545            return match runtime_state {
546                NativeRuntimeState::Active {
547                    worker_count,
548                    maximum_operation_concurrency,
549                    ..
550                } if *worker_count == requested_worker_count
551                    && *maximum_operation_concurrency
552                        == requested_maximum_operation_concurrency =>
553                {
554                    Ok(RuntimeInitialization::AlreadyInitialized)
555                }
556                NativeRuntimeState::Active { worker_count, .. } => {
557                    Err(RuntimeInitializationError::ConfigurationMismatch {
558                        configured_worker_count: *worker_count,
559                        requested_worker_count,
560                    })
561                }
562                NativeRuntimeState::InitializationFailed => {
563                    Err(RuntimeInitializationError::InitializationFailed)
564                }
565            };
566        }
567
568        let runtime_state = match ThreadPoolBuilder::new()
569            .num_threads(requested_worker_count)
570            .build()
571        {
572            Ok(pool) => NativeRuntimeState::Active {
573                pool,
574                worker_count: requested_worker_count,
575                maximum_operation_concurrency: requested_maximum_operation_concurrency,
576            },
577            Err(_) => NativeRuntimeState::InitializationFailed,
578        };
579        let was_published = NATIVE_RUNTIME.set(runtime_state).is_ok();
580        if !was_published {
581            return initialize_native_parallelism_with_operation_concurrency(
582                requested_worker_count,
583                requested_maximum_operation_concurrency,
584            );
585        }
586        match NATIVE_RUNTIME.get() {
587            Some(NativeRuntimeState::Active { .. }) => Ok(RuntimeInitialization::Initialized),
588            Some(NativeRuntimeState::InitializationFailed) | None => {
589                Err(RuntimeInitializationError::InitializationFailed)
590            }
591        }
592    }
593    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
594    {
595        let _ = (
596            requested_worker_count,
597            requested_maximum_operation_concurrency,
598        );
599        Err(RuntimeInitializationError::RequiresAsyncBootstrap)
600    }
601    #[cfg(all(target_arch = "wasm32", not(target_feature = "atomics")))]
602    {
603        let _ = (
604            requested_worker_count,
605            requested_maximum_operation_concurrency,
606        );
607        Err(RuntimeInitializationError::NotCompiled)
608    }
609}

611/// Register the Rayon global pool after the atomic WebAssembly bootstrap promise resolves.
612#[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
613pub(crate) fn register_wasm_parallelism(
614    requested_worker_count: usize,
615) -> Result<RuntimeInitialization, RuntimeInitializationError> {
616    if requested_worker_count == 0 {
617        return Err(RuntimeInitializationError::InvalidWorkerCount);
618    }
619    if let Some(runtime_state) = NATIVE_RUNTIME.get() {
620        return match runtime_state {
621            NativeRuntimeState::WasmActive { worker_count }
622                if *worker_count == requested_worker_count =>
623            {
624                Ok(RuntimeInitialization::AlreadyInitialized)
625            }
626            NativeRuntimeState::WasmActive { worker_count } => {
627                Err(RuntimeInitializationError::ConfigurationMismatch {
628                    configured_worker_count: *worker_count,
629                    requested_worker_count,
630                })
631            }
632            NativeRuntimeState::Active { worker_count, .. } => {
633                Err(RuntimeInitializationError::ConfigurationMismatch {
634                    configured_worker_count: *worker_count,
635                    requested_worker_count,
636                })
637            }
638            NativeRuntimeState::InitializationFailed => {
639                Err(RuntimeInitializationError::InitializationFailed)
640            }
641        };
642    }
643    if NATIVE_RUNTIME
644        .set(NativeRuntimeState::WasmActive {
645            worker_count: requested_worker_count,
646        })
647        .is_ok()
648    {
649        Ok(RuntimeInitialization::Initialized)
650    } else {
651        register_wasm_parallelism(requested_worker_count)
652    }
653}

771/// Identifies a logical task independently of its worker or completion order.
772#[derive(Clone, Debug, Eq, Hash, Ord, PartialEq, PartialOrd)]
773pub(crate) struct LogicalTaskIdentifier {
774    path: Vec<u64>,
775}
776
777impl LogicalTaskIdentifier {
778    /// Construct the root identifier for one operation.
779    pub(crate) fn root(operation_identifier: u64) -> Self {
780        LogicalTaskIdentifier {
781            path: vec![operation_identifier],
782        }
783    }
784
785    /// Derive a child identifier in canonical task order.
786    pub(crate) fn child(&self, child_identifier: u64) -> Self {
787        let mut path = self.path.clone();
788        path.push(child_identifier);
789        LogicalTaskIdentifier { path }
790    }
791
792    /// Return the nesting depth of this task.
793    pub(crate) fn depth(&self) -> usize {
794        self.path.len().saturating_sub(1)
795    }
796
797    /// Return the complete logical path used to derive deterministic child state.
798    pub(crate) fn path(&self) -> &[u64] {
799        &self.path
800    }
801
802    /// Name each input's task in turn, reusing one allocation for every child identifier.
803    ///
804    /// A serial coordinator visits its children one after another, so only one child identifier
805    /// is ever live. Cloning the parent path per child would allocate once per task for a name
806    /// most tasks only read, which a coordinator that never forks should not pay for. The child
807    /// is presented by reference, its position is replaced for the next one, and the identifier
808    /// is restored to the parent when the walk ends, so the parent is exactly what it was and
809    /// each child sees the same name it would have been given its own allocation.
810    pub(crate) fn for_each_indexed_child<Input, Value>(
811        &mut self,
812        inputs: &[Input],
813        mut function: impl FnMut(&Input, &LogicalTaskIdentifier) -> Value,
814    ) -> Vec<Value> {
815        self.path.push(0);
816        let position = self.path.len() - 1;
817        let values = inputs
818            .iter()
819            .enumerate()
820            .map(|(index, input)| {
821                self.path[position] = index as u64;
822                function(input, self)
823            })
824            .collect();
825        self.path.pop();
826        values
827    }
828}

984/// The result of applying the deterministic execution policy.
985#[derive(Clone, Copy, Debug, Eq, PartialEq)]

987    /// Selects the backend route.
988    pub(crate) route: ExecutionRoute,
989    /// Explains a serial route, or is absent for a parallel decision.
990    pub(crate) serial_fallback_reason: Option<SerialFallbackReason>,
991    /// Reports the selected work-class cutoff, or its absence for a serial-only class.
992    pub(crate) selected_cutoff: Option<u64>,
993}
994
995/// Apply the serial-first execution policy to one work estimate.
996pub(crate) fn execution_decision(
997    execution_mode: ExecutionMode,
998    runtime_availability: RuntimeAvailability,
999    work_estimate: &WorkEstimate,
1000    available_permits: usize,
1001    available_memory_bytes: Option<usize>,
1002    cancellation: &CancellationToken,
1003    admitted_serial_only_work_classes: &[&str],
1004) -> ExecutionDecision {
1005    let useful_worker_count = runtime_availability
1006        .worker_count
1007        .min(runtime_availability.maximum_operation_concurrency);
1008    let selected_cutoff = work_cutoff(
1009        work_estimate.work_class_identifier,
1010        admitted_serial_only_work_classes,
1011        useful_worker_count,
1012    );
1013    let cost_gate_reason = match selected_cutoff {
1014        None => Some(SerialFallbackReason::SerialOnlyWorkClass),
1015        Some(cutoff) if work_estimate.domain_work_units < cutoff => {
1016            Some(SerialFallbackReason::BelowCostCutoff)
1017        }
1018        Some(_) => None,
1019    };
1020    let serial_fallback_reason = if execution_mode == ExecutionMode::Sequential {
1021        Some(SerialFallbackReason::SequentialMode)
1022    } else if cancellation.is_cancelled() {
1023        Some(SerialFallbackReason::Cancelled)
1024    } else if !runtime_availability.active {
1025        Some(SerialFallbackReason::RuntimeUnavailable)
1026    } else if runtime_availability.worker_count <= 1 {
1027        Some(SerialFallbackReason::SingleWorker)
1028    } else if work_estimate.independent_unit_count < 2 {
1029        Some(SerialFallbackReason::NoIndependentUnits)
1030    } else if cost_gate_reason.is_some() {
1031        cost_gate_reason
1032    } else if work_estimate.requested_permits == 0
1033        || available_permits < work_estimate.requested_permits
1034    {
1035        Some(SerialFallbackReason::NoPermits)
1036    } else if available_memory_bytes
1037        .is_some_and(|available| work_estimate.predicted_task_local_bytes > available)
1038    {
1039        Some(SerialFallbackReason::MemoryBudgetExceeded)
1040    } else {
1041        None
1042    };
1043
1044    crate::instrumentation::record_work_class_route(
1045        work_estimate.work_class_identifier,
1046        serial_fallback_reason.is_none(),
1047    );
1048
1049    ExecutionDecision {
1050        route: if serial_fallback_reason.is_some() {
1051            ExecutionRoute::Serial
1052        } else {
1053            ExecutionRoute::Parallel
1054        },
1055        serial_fallback_reason,
1056        selected_cutoff,
1057    }
1058}

1060/// Narrow an already-resolved route by one coordinator's own work estimate.
1061///
1062/// A route resolved at an operation's outermost coordinator says only that this operation is
1063/// allowed to run in parallel. It says nothing about whether the nested coordinator now forming
1064/// tasks — a Hensel level's factor corrections, one image's distinct-degree splits — has enough
1065/// work to pay for the pool. Inheriting the outer route unchanged skips that question entirely
1066/// and schedules trivial task batches; this asks it again, against the nested coordinator's own
1067/// work class, without needing the operation the outer decision was taken from.
1068///
1069/// It carries no admission list, so it serves only a class the shipping policy already admits. A
1070/// serial-only nested coordinator is refused here whatever its enclosing operation admits, which
1071/// is why every caller's work class carries a measured cutoff.
1072pub(crate) fn narrow_route_to_estimate(
1073    route: ExecutionRoute,
1074    work_estimate: &WorkEstimate,
1075    cancellation: &CancellationToken,
1076) -> ExecutionRoute {
1077    if route != ExecutionRoute::Parallel {
1078        return ExecutionRoute::Serial;
1079    }
1080    execution_decision(
1081        ExecutionMode::Automatic,
1082        runtime_availability(),
1083        work_estimate,
1084        work_estimate.requested_permits,
1085        None,
1086        cancellation,
1087        NO_ADMITTED_SERIAL_ONLY_WORK_CLASSES,
1088    )
1089    .route
1090}

1136/// Return the shipping disposition recorded for one work class.
1137///
1138/// A class with no entry is serial-only rather than admitted at a default cutoff: an unnamed
1139/// boundary has no scaling evidence, and automatic execution schedules a boundary only after that
1140/// boundary has demonstrated a benefit over the optimized serial route.
1141///
1142/// `admitted_serial_only_work_classes` is the operation's own list of serial-only classes it may
1143/// nonetheless run in parallel, which is how a disabled implementation stays under test. It is
1144/// empty in every shipped build.
1145pub(crate) fn work_class_admission(
1146    work_class_identifier: &str,
1147    admitted_serial_only_work_classes: &[&str],
1148) -> WorkClassAdmission {
1149    let recorded = WORK_CLASS_ADMISSIONS
1150        .iter()
1151        .find_map(|(known_identifier, admission)| {
1152            (*known_identifier == work_class_identifier).then_some(*admission)
1153        })
1154        .unwrap_or(WorkClassAdmission::SerialOnly);
1155    match recorded {
1156        WorkClassAdmission::Automatic { .. } => recorded,
1157        WorkClassAdmission::SerialOnly
1158            if admitted_serial_only_work_classes
1159                .iter()
1160                .any(|identifier| *identifier == work_class_identifier) =>
1161        {
1162            WorkClassAdmission::Automatic {
1163                work_unit_cutoff: 0,
1164                // The seam exists to reach a disabled boundary's parallel route, including on the
1165                // capacity its class would refuse, so it carries the lowest capacity a pool has.
1166                minimum_operation_concurrency: 2,
1167                serial_baseline: SerialBaselineEvidence {
1168                    benchmark_workload_identifier: "internal_parallel_route_test",
1169                    scaling_dimension: "directly supplied work estimate",
1170                },
1171            }
1172        }
1173        WorkClassAdmission::SerialOnly => recorded,
1174    }
1175}

1177/// Report whether one work class could reach the parallel backend at all.
1178///
1179/// This is the coordinator's first question, and it is answered from a handful of field reads and
1180/// one lookup in the shipping policy: the mode the operation runs under, the runtime the host
1181/// published, whether the operation has been cancelled, and whether the class is admitted to
1182/// automatic execution at any cutoff. None of those depends on the size of the work, so a class
1183/// this refuses could not have been admitted by any estimate.
1184///
1185/// A coordinator asks this before it forms anything. The estimate handed to [`execution_decision`]
1186/// is not free — the cost proxies are traversals of the level, the formula or the subproblem about
1187/// to run, and the peer records, budget reservations and outcome envelopes built around it exist
1188/// only so that a pool can be handed work. A serial-only class pays for all of it and then
1189/// discards it at the first branch of the policy, once per level of every recursive descent.
1190/// Asking here means the serial route pays for the mathematics alone.
1191pub(crate) fn work_class_may_fork(
1192    execution_mode: ExecutionMode,
1193    cancellation: &CancellationToken,
1194    work_class_identifier: &str,
1195    admitted_serial_only_work_classes: &[&str],
1196) -> bool {
1197    let runtime_availability = runtime_availability();
1198    execution_mode == ExecutionMode::Automatic
1199        && !cancellation.is_cancelled()
1200        && runtime_availability.worker_count > 1
1201        && runtime_availability.maximum_operation_concurrency > 1
1202        && matches!(
1203            work_class_admission(work_class_identifier, admitted_serial_only_work_classes),
1204            WorkClassAdmission::Automatic {
1205                minimum_operation_concurrency,
1206                ..
1207            } if runtime_availability.worker_count
1208                .min(runtime_availability.maximum_operation_concurrency)
1209                >= minimum_operation_concurrency
1210        )
1211}

1213/// Return the cutoff a work class is admitted at on a pool of this width, or nothing when the
1214/// disposition is out of reach: the class is serial-only, or the pool is narrower than the
1215/// capacity its measured gain needs.
1216pub(crate) fn work_cutoff(
1217    work_class_identifier: &str,
1218    admitted_serial_only_work_classes: &[&str],

1231fn run_instrumented_task<Value, Function>(function: Function) -> (Value, InstrumentationCounters)
1232where
1233    Function: FnOnce() -> Value,
1234{
1235    // The frame is owned by a guard rather than by this function body, so a task that leaves by
1236    // unwinding removes its own frame on the way out and the worker it ran on is left with the
1237    // stack it started with. Without that, the next task on the same worker would record into
1238    // the abandoned frame and the counters of an unrelated operation would move.
1239    let frame = instrumentation::TaskFrame::enter();
1240    let value = function();
1241    (value, frame.leave())
1242}

1244fn merge_task_results<Value>(task_results: Vec<(Value, InstrumentationCounters)>) -> Vec<Value> {
1245    task_results
1246        .into_iter()
1247        .map(|(value, counters)| {
1248            instrumentation::merge(&counters);
1249            value
1250        })
1251        .collect()
1252}

1254/// Downgrade a route that has nothing to run in parallel.
1255///
1256/// Handing one task to the pool costs a job submission and a latch wait to buy no concurrency at
1257/// all, and a coordinator that forms narrow batches — one modular image at a time, one evaluation
1258/// point at a time — would otherwise pay that on every batch. The policy already declines such
1259/// work as `NoIndependentUnits`; this applies the same rule to a batch the coordinator formed
1260/// after the decision.
1261fn route_for_task_count(route: ExecutionRoute, task_count: usize) -> ExecutionRoute {
1262    if task_count < 2 {
1263        return ExecutionRoute::Serial;
1264    }
1265    route
1266}

1268fn execute_ordered<Input, Value, Function>(
1269    route: ExecutionRoute,
1270    inputs: &[Input],
1271    function: Function,
1272) -> Vec<(Value, InstrumentationCounters)>
1273where
1274    Input: Sync,
1275    Value: Send,
1276    Function: Fn(&Input) -> Value + Send + Sync,
1277{
1278    let route = route_for_task_count(route, inputs.len());
1279    // The single-threaded WebAssembly build has no parallel route to take.
1280    #[cfg(all(target_arch = "wasm32", not(target_feature = "atomics")))]
1281    let _ = route;
1282    #[cfg(any(not(target_arch = "wasm32"), target_feature = "atomics"))]
1283    if let Some(outputs) = with_active_pool(route, |pool| {
1284        pool.install(|| {
1285            inputs
1286                .par_iter()
1287                .map(|input| run_instrumented_task(|| function(input)))
1288                .collect()
1289        })
1290    }) {
1291        return outputs;
1292    }
1293    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
1294    if wasm_runtime_is_active(route) {
1295        return inputs
1296            .par_iter()
1297            .map(|input| run_instrumented_task(|| function(input)))
1298            .collect();
1299    }
1300    inputs
1301        .iter()
1302        .map(|input| run_instrumented_task(|| function(input)))
1303        .collect()
1304}

1353/// Map values in input order while retaining the same closure bounds as a future parallel path.
1354pub(crate) fn ordered_map<Input, Value, Function>(
1355    route: ExecutionRoute,
1356    inputs: &[Input],
1357    function: Function,
1358) -> Vec<Value>
1359where
1360    Input: Sync,
1361    Value: Send,
1362    Function: Fn(&Input) -> Value + Send + Sync,
1363{
1364    // A serial batch never leaves this thread, so it needs none of the per-item machinery that
1365    // carries a task's counters home: no instrumentation frame, no counter record travelling
1366    // beside each value, and no merge. Items record straight into the caller's frame, exactly as
1367    // they did before these primitives came to sit between a parent expression node and its
1368    // children. The counter record is a large struct, and the expression traversals now reach
1369    // this once per node rather than once per modular image, so entering and leaving a frame per
1370    // child is worth not paying for a batch that was never going to be distributed.
1371    if route_for_task_count(route, inputs.len()) != ExecutionRoute::Parallel {
1372        return inputs.iter().map(function).collect();
1373    }
1374    merge_task_results(execute_ordered(route, inputs, function))
1375}

1377/// Return the first canonically produced value, evaluating no later input once one is found.
1378///
1379/// A serial route stops at the first input whose task yields a value, which is what a sequential
1380/// loop with an early return did. A parallel route evaluates the formed batch and then takes the
1381/// first value in input order, so the two routes agree on which value is returned and neither
1382/// depends on completion order.
1383pub(crate) fn ordered_find_first<Input, Value, Function>(
1384    route: ExecutionRoute,
1385    inputs: &[Input],
1386    function: Function,
1387) -> Option<Value>
1388where
1389    Input: Sync,
1390    Value: Send,
1391    Function: Fn(&Input) -> Option<Value> + Send + Sync,
1392{
1393    if route_for_task_count(route, inputs.len()) != ExecutionRoute::Parallel {
1394        return inputs.iter().find_map(function);
1395    }
1396    ordered_map(route, inputs, function)
1397        .into_iter()
1398        .flatten()
1399        .next()
1400}

1432/// Map fallible task outcomes in input order without discarding later outcomes.
1433pub(crate) fn ordered_try_map<Input, Value, Error, Function>(
1434    route: ExecutionRoute,
1435    inputs: &[Input],
1436    cancellation: &CancellationToken,
1437    function: Function,
1438) -> Vec<TaskOutcome<Value, Error>>
1439where
1440    Input: Sync,
1441    Value: Send,
1442    Error: Send,
1443    Function: Fn(&Input) -> TaskOutcome<Value, Error> + Send + Sync,
1444{
1445    // A serial batch takes the same shortcut `ordered_map` takes: it never leaves this thread, so
1446    // it needs no per-item instrumentation frame, no counter record carried beside each outcome,
1447    // and no merge afterwards. The counter record is a large struct, and the coordinators that
1448    // reach here — one cylindrical level's parents, one level's cells, one fiber's siblings —
1449    // form a batch per level of every decomposition, so the record is paid for once per cell of
1450    // a forced-sequential run that was never going to distribute any of them.
1451    if route_for_task_count(route, inputs.len()) != ExecutionRoute::Parallel {
1452        return inputs
1453            .iter()
1454            .map(|input| {
1455                if cancellation.is_cancelled() {
1456                    TaskOutcome::Cancelled
1457                } else {
1458                    function(input)
1459                }
1460            })
1461            .collect();
1462    }
1463    merge_task_results(execute_ordered(route, inputs, |input| {
1464        if cancellation.is_cancelled() {
1465            TaskOutcome::Cancelled
1466        } else {
1467            function(input)
1468        }
1469    }))
1470}

1520/// Reduce values with a fixed adjacent-pair binary tree.
1521pub(crate) fn deterministic_reduce<Value, Function>(
1522    route: ExecutionRoute,
1523    values: Vec<Value>,
1524    function: Function,
1525) -> Option<Value>
1526where
1527    Value: Send,
1528    Function: Fn(Value, Value) -> Value + Send + Sync,
1529{
1530    // A serial reduction folds the same adjacent-pair tree directly. The association is the
1531    // shared semantics of both routes and is unchanged — the merge is not assumed associative —
1532    // but a level that is never handed out needs none of the machinery that carries a task's
1533    // counters home: no instrumentation frame per pair, no counter record travelling beside each
1534    // merged value, and no merge afterwards. The record is a large struct, and a reduction of n
1535    // values performs n - 1 merges, so a forced-sequential matrix product was paying for one per
1536    // entry of every level of the tree.
1537    if route != ExecutionRoute::Parallel {
1538        return reduce_level_pairs_serially(values, &function);
1539    }
1540    let mut current_level = values;
1541    while current_level.len() > 1 {
1542        #[cfg(any(not(target_arch = "wasm32"), target_feature = "atomics"))]
1543        match parallel_reduce_level(route, current_level, &function) {
1544            Ok(next_level) => {
1545                current_level = next_level;
1546                continue;
1547            }
1548            Err(values) => current_level = values,
1549        }
1550        let mut task_results = Vec::with_capacity(current_level.len().div_ceil(2));
1551        let mut values = current_level.into_iter();
1552        while let Some(left_value) = values.next() {
1553            let Some(right_value) = values.next() else {
1554                task_results.push((left_value, InstrumentationCounters::new()));
1555                break;
1556            };
1557            let (value, counters) = run_instrumented_task(|| function(left_value, right_value));
1558            task_results.push((value, counters));
1559        }
1560        current_level = merge_task_results(task_results);
1561    }
1562    current_level.into_iter().next()
1563}

1565/// Fold the adjacent-pair tree of a serial reduction, forming no task records.
1566fn reduce_level_pairs_serially<Value, Function>(
1567    values: Vec<Value>,
1568    function: &Function,
1569) -> Option<Value>
1570where
1571    Function: Fn(Value, Value) -> Value,
1572{
1573    let mut current_level = values;
1574    while current_level.len() > 1 {
1575        let mut next_level = Vec::with_capacity(current_level.len().div_ceil(2));
1576        let mut values = current_level.into_iter();
1577        while let Some(left_value) = values.next() {
1578            match values.next() {
1579                Some(right_value) => next_level.push(function(left_value, right_value)),
1580                None => {
1581                    next_level.push(left_value);
1582                    break;
1583                }
1584            }
1585        }
1586        current_level = next_level;
1587    }
1588    current_level.into_iter().next()
1589}

1591#[cfg(any(not(target_arch = "wasm32"), target_feature = "atomics"))]
1592fn parallel_reduce_level<Value, Function>(
1593    route: ExecutionRoute,
1594    values: Vec<Value>,
1595    function: &Function,
1596) -> Result<Vec<Value>, Vec<Value>>
1597where
1598    Value: Send,
1599    Function: Fn(Value, Value) -> Value + Send + Sync,
1600{
1601    // A level of two values is one pair, so it has no more independent work than a level of one.
1602    if route_for_task_count(route, values.len() / 2) != ExecutionRoute::Parallel {
1603        return Err(values);
1604    }
1605    #[cfg(not(all(target_arch = "wasm32", target_feature = "atomics")))]
1606    {
1607        let Some(NativeRuntimeState::Active { pool, .. }) = NATIVE_RUNTIME.get() else {
1608            return Err(values);
1609        };
1610        let task_results = pool.install(|| {
1611            values
1612                .into_par_iter()
1613                .chunks(2)
1614                .filter_map(|values| {
1615                    let mut values = values.into_iter();
1616                    let Some(left_value) = values.next() else {
1617                        return None;
1618                    };
1619                    Some(match values.next() {
1620                        Some(right_value) => {
1621                            run_instrumented_task(|| function(left_value, right_value))
1622                        }
1623                        None => (left_value, InstrumentationCounters::new()),
1624                    })
1625                })
1626                .collect()
1627        });
1628        return Ok(merge_task_results(task_results));
1629    }
1630    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
1631    {
1632        if wasm_runtime_is_active(route) {
1633            let task_results = values
1634                .into_par_iter()
1635                .chunks(2)
1636                .filter_map(|values| {
1637                    let mut values = values.into_iter();
1638                    let Some(left_value) = values.next() else {
1639                        return None;
1640                    };
1641                    Some(match values.next() {
1642                        Some(right_value) => {
1643                            run_instrumented_task(|| function(left_value, right_value))
1644                        }
1645                        None => (left_value, InstrumentationCounters::new()),
1646                    })
1647                })
1648                .collect();
1649            return Ok(merge_task_results(task_results));
1650        }
1651        Err(values)
1652    }
1653}

1655/// Run two independent serial branches while preserving their left/right identities.
1656pub(crate) fn join<LeftValue, RightValue, LeftFunction, RightFunction>(
1657    route: ExecutionRoute,
1658    left_function: LeftFunction,
1659    right_function: RightFunction,
1660) -> (LeftValue, RightValue)
1661where
1662    LeftValue: Send,
1663    RightValue: Send,
1664    LeftFunction: FnOnce() -> LeftValue + Send,
1665    RightFunction: FnOnce() -> RightValue + Send,
1666{
1667    // A branch may execute on a worker that is not the caller, so each one records into its
1668    // own capture and the caller merges both in left-then-right order. Counts are summed and
1669    // maxima taken, so the merged snapshot is the same whichever thread ran which branch.
1670    // The single-threaded WebAssembly build has no parallel route to take.
1671    #[cfg(all(target_arch = "wasm32", not(target_feature = "atomics")))]
1672    let _ = route;
1673    #[cfg(any(not(target_arch = "wasm32"), target_feature = "atomics"))]
1674    if route == ExecutionRoute::Parallel
1675        && let Some(NativeRuntimeState::Active { pool, .. }) = NATIVE_RUNTIME.get()
1676    {
1677        let (left, right) = pool.join(
1678            || run_instrumented_task(left_function),
1679            || run_instrumented_task(right_function),
1680        );
1681        return merge_branch_results(left, right);
1682    }
1683    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
1684    if wasm_runtime_is_active(route) {
1685        let (left, right) = rayon::join(
1686            || run_instrumented_task(left_function),
1687            || run_instrumented_task(right_function),
1688        );
1689        return merge_branch_results(left, right);
1690    }
1691    merge_branch_results(
1692        run_instrumented_task(left_function),

1709/// Run two branches with logical identifiers derived from their common parent.
1710pub(crate) fn join_with_identifiers<LeftValue, RightValue, LeftFunction, RightFunction>(
1711    route: ExecutionRoute,
1712    parent_identifier: &LogicalTaskIdentifier,
1713    left_function: LeftFunction,
1714    right_function: RightFunction,
1715) -> (LeftValue, RightValue)
1716where
1717    LeftValue: Send,
1718    RightValue: Send,
1719    LeftFunction: FnOnce(&LogicalTaskIdentifier) -> LeftValue + Send,
1720    RightFunction: FnOnce(&LogicalTaskIdentifier) -> RightValue + Send,
1721{
1722    let left_identifier = parent_identifier.child(0);
1723    let right_identifier = parent_identifier.child(1);
1724    // The single-threaded WebAssembly build has no parallel route to take.
1725    #[cfg(all(target_arch = "wasm32", not(target_feature = "atomics")))]
1726    let _ = route;
1727    #[cfg(any(not(target_arch = "wasm32"), target_feature = "atomics"))]
1728    if route == ExecutionRoute::Parallel
1729        && let Some(NativeRuntimeState::Active { pool, .. }) = NATIVE_RUNTIME.get()
1730    {
1731        let (left, right) = pool.join(
1732            || run_instrumented_task(|| left_function(&left_identifier)),
1733            || run_instrumented_task(|| right_function(&right_identifier)),
1734        );
1735        return merge_branch_results(left, right);
1736    }
1737    #[cfg(all(target_arch = "wasm32", target_feature = "atomics"))]
1738    if wasm_runtime_is_active(route) {
1739        let (left, right) = rayon::join(
1740            || run_instrumented_task(|| left_function(&left_identifier)),
1741            || run_instrumented_task(|| right_function(&right_identifier)),
1742        );
1743        return merge_branch_results(left, right);
1744    }
1745    merge_branch_results(
1746        run_instrumented_task(|| left_function(&left_identifier)),
1747        run_instrumented_task(|| right_function(&right_identifier)),
1748    )
1749}

2064/// Select the canonical mathematical result from task outcomes independent of completion order.
2065pub(crate) fn select_search_outcome<Value, Evidence, Decline>(
2066    mut outcomes: Vec<IndexedSearchTaskOutcome<Value, Evidence, Decline>>,
2067) -> SearchResolution<Value, Evidence, Decline>
2068where
2069    Value: Clone,
2070    Evidence: Clone,
2071    Decline: Clone,
2072{
2073    outcomes.sort_by(|left, right| {
2074        left.identifier
2075            .canonical_order()
2076            .cmp(&right.identifier.canonical_order())
2077    });
2078    let mut candidates = Vec::new();
2079    let mut empty_evidence = Vec::new();
2080    let mut empty_conditions = SearchConditionLedger::empty();
2081    let mut declines = Vec::new();
2082    let mut explored_boundaries = Vec::new();
2083    let mut has_resource_exhaustion = false;
2084    let mut has_cancellation = false;
2085
2086    for indexed_outcome in outcomes {
2087        match indexed_outcome.outcome {
2088            SearchTaskOutcome::Candidate {
2089                value,
2090                evidence,
2091                conditions,
2092                completeness_contribution,
2093                quality_key,
2094            } => candidates.push((
2095                quality_key,
2096                indexed_outcome.identifier,
2097                value,
2098                evidence,
2099                conditions,
2100                completeness_contribution,
2101            )),
2102            SearchTaskOutcome::ProvedEmpty {
2103                evidence,
2104                conditions,
2105            } => {
2106                empty_evidence.push(evidence);
2107                empty_conditions.merge(&conditions);
2108            }
2109            SearchTaskOutcome::Declined {
2110                reason,
2111                explored_boundary,
2112            } => {
2113                declines.push(reason);
2114                explored_boundaries.push(explored_boundary);
2115            }
2116            SearchTaskOutcome::ResourceExhausted => has_resource_exhaustion = true,
2117            SearchTaskOutcome::Cancelled => has_cancellation = true,
2118        }
2119    }
2120
2121    // Every candidate that ran contributes to the ledger, not only the one selected: a caller
2122    // reading the ledger is asking which parts of the original domain were searched, and the
2123    // domains the losing candidates covered were searched just as much as the winner's. The
2124    // outcomes arrive in canonical task order, so the ledger is in that order too.
2125    let contributions = candidates
2126        .iter()
2127        .map(|(_, _, _, _, _, contribution)| contribution.clone())
2128        .collect::<Vec<_>>();
2129    if let Some((quality_key, identifier, value, evidence, conditions, contribution)) =
2130        candidates.into_iter().min_by(|left, right| {
2131            (left.0.as_slice(), left.1.canonical_order())
2132                .cmp(&(right.0.as_slice(), right.1.canonical_order()))
2133        })
2134    {
2135        // The selected candidate may only be called complete when its own domain claim is
2136        // exhaustive and nothing else in the batch left part of the problem unsearched. A
2137        // decline, a resource boundary, or a cancellation each leaves a domain unaccounted for,
2138        // and none of them is a proof that the domain held nothing.
2139        let complete = contribution.domain_covered
2140            && contribution.exhaustive
2141            && declines.is_empty()
2142            && !has_resource_exhaustion
2143            && !has_cancellation;
2144        return SearchResolution::Candidate {
2145            identifier,
2146            value,
2147            evidence,
2148            conditions,
2149            completeness: SearchCompletenessLedger {
2150                contributions,
2151                exhaustive: complete,
2152            },
2153            quality_key,
2154        };
2155    }
2156    if has_cancellation {
2157        return SearchResolution::Cancelled;
2158    }
2159    if has_resource_exhaustion {
2160        return SearchResolution::ResourceExhausted;
2161    }
2162    if !declines.is_empty() {
2163        return SearchResolution::Declined {
2164            reasons: declines,
2165            explored_boundaries,
2166        };
2167    }
2168    if empty_evidence.is_empty() {
2169        SearchResolution::Declined {
2170            reasons: Vec::new(),
2171            explored_boundaries: vec!["no search tasks"],
2172        }
2173    } else {
2174        SearchResolution::ProvedEmpty {
2175            evidence: empty_evidence,
2176            conditions: empty_conditions,
2177        }
2178    }
2179}

2181/// Confirm that formed batches are peer groups supplied in canonical priority order.
2182fn validate_search_batches<Input>(
2183    batches: &[SearchPriorityBatch<Input>],
2184) -> Result<(), SearchBatchError> {
2185    let mut previous_priority_tier = None;
2186    for batch in batches {
2187        if !batch.has_one_priority_tier() {
2188            let task_priority_tier = batch
2189                .tasks
2190                .iter()
2191                .find(|task| task.identifier.priority_tier != batch.priority_tier)
2192                .map(|task| task.identifier.priority_tier)
2193                .unwrap_or(batch.priority_tier);
2194            return Err(SearchBatchError::MixedPriorityTier {
2195                declared_priority_tier: batch.priority_tier,
2196                task_priority_tier,
2197            });
2198        }
2199        if let Some(previous_priority_tier) = previous_priority_tier
2200            && batch.priority_tier < previous_priority_tier
2201        {
2202            return Err(SearchBatchError::OutOfOrderPriorityTier {
2203                previous_priority_tier,
2204                current_priority_tier: batch.priority_tier,
2205            });
2206        }
2207        previous_priority_tier = Some(batch.priority_tier);
2208    }
2209    Ok(())
2210}

2336    identify: Identify,
2337    cancellation: &CancellationToken,
2338    function: Function,
2339    settlement: Settlement,
2340) -> Vec<IndexedSearchTaskOutcome<Value, Evidence, Decline>>
2341where
2342    Input: Clone + Sync,
2343    Value: Send,
2344    Evidence: Send,
2345    Decline: Send,
2346    Identify: Fn(usize, &Input) -> SearchTaskIdentifier,
2347    Function: Fn(&Input, &SearchTaskIdentifier) -> SearchTaskOutcome<Value, Evidence, Decline>
2348        + Send
2349        + Sync,
2350    Settlement: Fn(&SearchTaskOutcome<Value, Evidence, Decline>) -> SearchSettlement,
2351{
2352    if route_for_task_count(route, inputs.len()) == ExecutionRoute::Parallel {
2353        let tasks = inputs
2354            .iter()
2355            .enumerate()
2356            .map(|(branch_index, input)| {
2357                SearchTask::new(identify(branch_index, input), input.clone())
2358            })
2359            .collect::<Vec<_>>();
2360        // A parallel run executes the whole formed batch and selects canonically, so the
2361        // settlement rule — which exists to let a serial fold stop early — does not apply to it.
2362        let batch = SearchPriorityBatch::new(priority_tier, tasks);
2363        record_search_batch_route(route, batch.tasks.len());
2364        let mut outcomes = run_search_batch(route, &batch, cancellation, &function);
2365        sort_search_outcomes(&mut outcomes);
2366        return outcomes;
2367    }
2368    record_search_batch_route(route, inputs.len());
2369    let mut outcomes = Vec::new();
2370    let mut executed_task_count = 0;
2371    for (branch_index, input) in inputs.iter().enumerate() {
2372        let identifier = identify(branch_index, input);
2373        debug_assert_eq!(
2374            identifier.priority_tier, priority_tier,
2375            "a peer group's tasks share its declared priority tier"
2376        );
2377        let outcome = if cancellation.is_cancelled() {
2378            SearchTaskOutcome::Cancelled
2379        } else {
2380            function(input, &identifier)
2381        };
2382        executed_task_count += 1;
2383        let settled = settlement(&outcome) == SearchSettlement::SettlesBatch;
2384        outcomes.push(IndexedSearchTaskOutcome {
2385            identifier,
2386            outcome,
2387        });
2388        if settled {
2389            break;
2390        }
2391    }
2392    record_counter!(
2393        search_settled_tasks,
2394        inputs.len().saturating_sub(executed_task_count)
2395    );
2396    sort_search_outcomes(&mut outcomes);
2397    outcomes
2398}

2400/// Record which backend one formed batch was handed to, so the boundary can be measured.
2401fn record_search_batch_route(route: ExecutionRoute, task_count: usize) {
2402    if route_for_task_count(route, task_count) == ExecutionRoute::Parallel {
2403        record_counter!(search_parallel_batches);
2404    } else {
2405        record_counter!(search_serial_batches);
2406    }
2407}
2408/// Describes whether an outcome already settles a batch's canonical result.
2409///
2410/// A search coordinator that selects the first canonical task able to answer has no use for the
2411/// outcomes of later peers: the mathematics of the selection makes them unreachable. Declaring
2412/// that with this predicate lets a serial run stop at the settling task, which is exactly what
2413/// the sequential reference did, while a parallel run still executes the whole formed batch and
2414/// selects canonically. Both routes therefore return the same result, and neither depends on
2415/// completion order.
2416#[derive(Clone, Copy, Debug, Eq, PartialEq)]
2417pub(crate) enum SearchSettlement {
2418    /// The canonical result is decided and later peers cannot change it.
2419    SettlesBatch,
2420    /// Later peers must still run because they can contribute to the result.
2421    RequiresRemainingPeers,
2422}

2424/// Execute priority batches, stopping a serial batch at its first canonically settling outcome.
2425///
2426/// `settlement` is consulted in canonical task order and must be a mathematical property of the
2427/// outcome alone, never of when it arrived. Tasks skipped by a settled serial batch are reported
2428/// with no outcome at all rather than as declines, so a caller can never read a task that did not
2429/// run as a mathematical claim.
2430pub(crate) fn execute_search_batches_with_settlement<
2431    Input,
2432    Value,
2433    Evidence,
2434    Decline,
2435    Function,
2436    Settlement,
2437>(
2438    route: ExecutionRoute,
2439    batches: &[SearchPriorityBatch<Input>],
2440    cancellation: &CancellationToken,
2441    function: Function,
2442    settlement: Settlement,
2443) -> Result<Vec<IndexedSearchTaskOutcome<Value, Evidence, Decline>>, SearchBatchError>
2444where
2445    Input: Sync,
2446    Value: Send,
2447    Evidence: Send,
2448    Decline: Send,
2449    Function: Fn(&Input, &SearchTaskIdentifier) -> SearchTaskOutcome<Value, Evidence, Decline>
2450        + Send
2451        + Sync,
2452    Settlement: Fn(&SearchTaskOutcome<Value, Evidence, Decline>) -> SearchSettlement,
2453{
2454    validate_search_batches(batches)?;
2455    let mut outcomes = Vec::new();
2456    for batch in batches {
2457        record_search_batch_route(route, batch.tasks.len());
2458        if route_for_task_count(route, batch.tasks.len()) == ExecutionRoute::Parallel {
2459            outcomes.extend(run_search_batch(route, batch, cancellation, &function));
2460            continue;
2461        }
2462        let mut executed_task_count = 0;
2463        for task in &batch.tasks {
2464            let outcome = if cancellation.is_cancelled() {
2465                SearchTaskOutcome::Cancelled
2466            } else {
2467                function(&task.input, &task.identifier)
2468            };
2469            executed_task_count += 1;
2470            let settled = settlement(&outcome) == SearchSettlement::SettlesBatch;
2471            outcomes.push(IndexedSearchTaskOutcome {
2472                identifier: task.identifier.clone(),
2473                outcome,
2474            });
2475            if settled {
2476                break;
2477            }
2478        }
2479        record_counter!(
2480            search_settled_tasks,
2481            batch.tasks.len().saturating_sub(executed_task_count)
2482        );
2483    }
2484    sort_search_outcomes(&mut outcomes);
2485    Ok(outcomes)
2486}

2612/// A deterministic operation budget shared by task-local reservations.
2613#[derive(Clone, Debug)]
2614pub(crate) struct DeterministicBudget {
2615    limit: u64,
2616    mode: BudgetMode,
2617    charges_may_overlap: bool,
2618    state: Arc<BudgetState>,
2619}
2620
2621impl DeterministicBudget {
2622    /// Construct a budget with a fixed limit and charging mode.
2623    ///
2624    /// The budget is built for a route that may fork, so every charge synchronizes with the
2625    /// charges of other workers. [`DeterministicBudget::new_for_execution_mode`] is how a route
2626    /// that cannot fork asks for the cheaper ledger.
2627    pub(crate) fn new(limit: u64, mode: BudgetMode) -> Self {
2628        DeterministicBudget::new_with_overlap(limit, mode, true)
2629    }
2630
2631    /// Construct a budget whose charging cost matches the execution mode it will run under.
2632    ///
2633    /// A forced-sequential operation charges only from the thread that opened it, so the
2634    /// compare-and-exchange loop a shared ledger runs at every charge buys nothing there: the
2635    /// only writer is the caller. The storage stays atomic either way, so nothing is unsound if
2636    /// a route charges from somewhere the mode did not anticipate; the mode selects only how
2637    /// much synchronization each charge pays for, and both report the same total.
2638    pub(crate) fn new_for_execution_mode(
2639        limit: u64,
2640        mode: BudgetMode,
2641        execution_mode: crate::operation::ExecutionMode,
2642    ) -> Self {
2643        DeterministicBudget::new_with_overlap(
2644            limit,
2645            mode,
2646            execution_mode == crate::operation::ExecutionMode::Automatic,
2647        )
2648    }
2649
2650    fn new_with_overlap(limit: u64, mode: BudgetMode, charges_may_overlap: bool) -> Self {
2651        DeterministicBudget {
2652            limit,
2653            mode,
2654            charges_may_overlap,
2655            state: Arc::new(BudgetState {
2656                consumed: AtomicU64::new(0),
2657                reserved: AtomicU64::new(0),
2658            }),
2659        }
2660    }
2661
2662    /// Charge exhaustive work if the resulting total remains within the limit.
2663    pub(crate) fn charge(&self, work_units: u64) -> bool {
2664        if self.mode != BudgetMode::Exhaustive {
2665            return false;
2666        }
2667        self.charge_ledger(work_units)
2668    }
2669
2670    /// Charge serial work against the committed portion of a coordinator budget.
2671    pub(crate) fn charge_serial(&self, work_units: u64) -> bool {
2672        if self.mode != BudgetMode::Coordinator {
2673            return false;
2674        }
2675        self.charge_ledger(work_units)
2676    }
2677
2678    fn charge_ledger(&self, work_units: u64) -> bool {
2679        if self.charges_may_overlap {
2680            return charge_atomic(&self.state.consumed, work_units, self.limit);
2681        }
2682        let current = self.state.consumed.load(Ordering::Relaxed);
2683        let Some(next) = current.checked_add(work_units) else {
2684            return false;
2685        };
2686        if next > self.limit {
2687            return false;
2688        }
2689        self.state.consumed.store(next, Ordering::Relaxed);
2690        true
2691    }
2692
2693    /// Return the work committed at completed batch boundaries.
2694    pub(crate) fn consumed(&self) -> u64 {
2695        self.state.consumed.load(Ordering::Acquire)
2696    }
2697
2698    /// Reserve all canonical child budgets or return no partial reservation.
2699    pub(crate) fn reserve_batch(
2700        &self,
2701        child_budgets: &[u64],
2702    ) -> Result<Vec<BudgetReservation>, BudgetReservationError> {
2703        if self.mode != BudgetMode::Coordinator {
2704            return Err(BudgetReservationError::WrongMode);
2705        }
2706        let requested = child_budgets
2707            .iter()
2708            .try_fold(0u64, |total, budget| total.checked_add(*budget))
2709            .ok_or(BudgetReservationError::Overflow)?;
2710        loop {
2711            let consumed = self.state.consumed.load(Ordering::Acquire);
2712            let reserved = self.state.reserved.load(Ordering::Acquire);
2713            let available = self
2714                .limit
2715                .checked_sub(consumed)
2716                .and_then(|remaining| remaining.checked_sub(reserved))
2717                .ok_or(BudgetReservationError::InsufficientBudget)?;
2718            if requested > available {
2719                return Err(BudgetReservationError::InsufficientBudget);
2720            }
2721            if self
2722                .state
2723                .reserved
2724                .compare_exchange(
2725                    reserved,
2726                    reserved + requested,
2727                    Ordering::AcqRel,
2728                    Ordering::Acquire,
2729                )
2730                .is_ok()
2731            {
2732                return Ok(child_budgets
2733                    .iter()
2734                    .map(|budget| BudgetReservation {
2735                        allocated: *budget,
2736                        consumed: 0,
2737                    })
2738                    .collect());
2739            }
2740        }
2741    }
2742
2743    /// Reconcile one canonical reservation batch and return its accounting summary.
2744    pub(crate) fn reconcile_batch(
2745        &self,
2746        reservations: &[BudgetReservation],
2747    ) -> BudgetReconciliation {
2748        let allocated = reservations
2749            .iter()
2750            .map(|reservation| reservation.allocated)
2751            .sum::<u64>();
2752        let consumed = reservations
2753            .iter()
2754            .map(|reservation| reservation.consumed)
2755            .sum::<u64>();
2756        let returned = allocated.saturating_sub(consumed);
2757        self.state.reserved.fetch_sub(allocated, Ordering::AcqRel);
2758        self.state.consumed.fetch_add(consumed, Ordering::AcqRel);
2759        BudgetReconciliation {
2760            consumed_work_units: consumed,
2761            returned_work_units: returned,
2762        }
2763    }
2764
2765    /// Return the fixed limit this budget charges against.
2766    pub(crate) const fn ceiling(&self) -> u64 {
2767        self.limit
2768    }
2769
2770    /// Return the uncommitted budget available for the next canonical batch.
2771    pub(crate) fn remaining(&self) -> u64 {
2772        self.limit
2773            .saturating_sub(self.state.consumed.load(Ordering::Acquire))
2774            .saturating_sub(self.state.reserved.load(Ordering::Acquire))
2775    }
2776}
2777
2778fn charge_atomic(counter: &AtomicU64, work_units: u64, limit: u64) -> bool {
2779    loop {
2780        let current = counter.load(Ordering::Acquire);
2781        let Some(next) = current.checked_add(work_units) else {
2782            return false;
2783        };
2784        if next > limit {
2785            return false;
2786        }
2787        if counter
2788            .compare_exchange(current, next, Ordering::AcqRel, Ordering::Acquire)
2789            .is_ok()
2790        {
2791            return true;
2792        }
2793    }
2794}

3366        fn append_permutations(
3367            remaining_values: &mut Vec<usize>,
3368            current_permutation: &mut Vec<usize>,
3369            permutations: &mut Vec<Vec<usize>>,
3370        ) {
3371            if remaining_values.is_empty() {
3372                permutations.push(current_permutation.clone());
3373                return;
3374            }
3375            for value_index in 0..remaining_values.len() {
3376                let value = remaining_values.remove(value_index);
3377                current_permutation.push(value);
3378                append_permutations(remaining_values, current_permutation, permutations);
3379                current_permutation.pop();
3380                remaining_values.insert(value_index, value);
3381            }
3382        }
3383
3384        let mut remaining_values = (0..number_of_values).collect::<Vec<_>>();
3385        let mut all_permutations = Vec::new();
3386        append_permutations(
3387            &mut remaining_values,
3388            &mut Vec::new(),
3389            &mut all_permutations,
