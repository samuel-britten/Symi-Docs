34/// Public native enumeration named `parse_error`.
35pub enum ParseError {
36    /// Represents the `unexpected_character` case.
37    UnexpectedCharacter(char, usize),
38    /// Represents the `unexpected_token` case.
39    UnexpectedToken(String, usize),
40    /// Represents the `unexpected_end_of_input` case.
41    UnexpectedEndOfInput,
42    /// Represents the `expected_identifier` case.
43    ExpectedIdentifier(usize),
44    /// Represents the `expected_right_parenthesis` case.
45    ExpectedRightParenthesis(usize),
46    /// Represents the `integer_parse_error` case.
47    IntegerParseError(String),
48    /// Represents the `unknown_meta_command` case.
49    UnknownMetaCommand(String),
50    /// Represents the `unknown_assumption_flag` case.
51    UnknownAssumptionFlag(String),
52    /// Represents the `expected_assumption_flag` case.
53    ExpectedAssumptionFlag,
54}

56impl fmt::Display for ParseError {
57    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
58        // Render a parse error as a single human-readable line. Positions are
59        // reported as zero-based byte offsets into the original source string.
60        match self {
61            ParseError::UnexpectedCharacter(character, position) => write!(
62                formatter,
63                "unexpected character {:?} at position {}",
64                character, position
65            ),
66            ParseError::UnexpectedToken(description, position) => write!(
67                formatter,
68                "unexpected token {} at position {}",
69                description, position
70            ),
71            ParseError::UnexpectedEndOfInput => formatter.write_str("unexpected end of input"),
72            ParseError::ExpectedIdentifier(position) => {
73                write!(formatter, "expected identifier at position {}", position)
74            }
75            ParseError::ExpectedRightParenthesis(position) => write!(
76                formatter,
77                "expected closing parenthesis at position {}",
78                position
79            ),
80            ParseError::IntegerParseError(text) => {
81                write!(formatter, "could not parse integer literal {:?}", text)
82            }
83            ParseError::UnknownMetaCommand(name) => {
84                write!(formatter, "unknown meta-command {:?}", name)
85            }
86            ParseError::UnknownAssumptionFlag(name) => {
87                write!(formatter, "unknown assumption flag {:?}", name)
88            }
89            ParseError::ExpectedAssumptionFlag => {
90                formatter.write_str(":assume requires at least one flag")
91            }
92        }
93    }
94}

97enum Token {
98    IntegerLiteral(String, usize),
99    Identifier(String, usize),
100    Plus(usize),
101    Minus(usize),
102    Star(usize),
103    Slash(usize),
104    Caret(usize),
105    LeftParenthesis(usize),
106    RightParenthesis(usize),
107    Comma(usize),
108    Equals(usize),
109    Colon(usize),
110    EndOfInput(usize),
111}
112
113fn token_position(value: &Token) -> usize {
114    // Report the byte offset associated with a token. Used to surface positional
115    // information in parse errors so the user can locate problems by column.
116    match value {
117        Token::IntegerLiteral(_, position)
118        | Token::Identifier(_, position)
119        | Token::Plus(position)
120        | Token::Minus(position)
121        | Token::Star(position)
122        | Token::Slash(position)
123        | Token::Caret(position)
124        | Token::LeftParenthesis(position)
125        | Token::RightParenthesis(position)
126        | Token::Comma(position)
127        | Token::Equals(position)
128        | Token::Colon(position)
129        | Token::EndOfInput(position) => *position,
130    }
131}
132
133fn token_description(value: &Token) -> String {
134    // Describe a token for error messages. Atoms quote their text; punctuation
135    // names the symbol; the sentinel reports end-of-input.
136    match value {
137        Token::IntegerLiteral(text, _) => format!("integer {:?}", text),
138        Token::Identifier(name, _) => format!("identifier {:?}", name),
139        Token::Plus(_) => "'+'".to_string(),
140        Token::Minus(_) => "'-'".to_string(),
141        Token::Star(_) => "'*'".to_string(),
142        Token::Slash(_) => "'/'".to_string(),
143        Token::Caret(_) => "'^'".to_string(),
144        Token::LeftParenthesis(_) => "'('".to_string(),
145        Token::RightParenthesis(_) => "')'".to_string(),
146        Token::Comma(_) => "','".to_string(),
147        Token::Equals(_) => "'='".to_string(),
148        Token::Colon(_) => "':'".to_string(),
149        Token::EndOfInput(_) => "end of input".to_string(),
150    }
151}

153fn tokenize(source: &str) -> Result<Vec<Token>, ParseError> {
154    // Convert source text into a flat token vector with an explicit end_of_input
155    // sentinel at the end. The single-pass scanner recognises operators, integer
156    // literals, and identifiers; whitespace is skipped. Non-ASCII characters and
157    // unrecognised punctuation surface as unexpected_character errors.
158    let bytes = source.as_bytes();
159    let mut tokens: Vec<Token> = Vec::new();
160    let mut position: usize = 0;
161    while position < bytes.len() {
162        let byte = bytes[position];
163        if byte.is_ascii_whitespace() {
164            position += 1;
165            continue;
166        }
167        let start = position;
168        match byte {
169            b'+' => {
170                tokens.push(Token::Plus(start));
171                position += 1;
172            }
173            b'-' => {
174                tokens.push(Token::Minus(start));
175                position += 1;
176            }
177            b'*' => {
178                tokens.push(Token::Star(start));
179                position += 1;
180            }
181            b'/' => {
182                tokens.push(Token::Slash(start));
183                position += 1;
184            }
185            b'^' => {
186                tokens.push(Token::Caret(start));
187                position += 1;
188            }
189            b'(' => {
190                tokens.push(Token::LeftParenthesis(start));
191                position += 1;
192            }
193            b')' => {
194                tokens.push(Token::RightParenthesis(start));
195                position += 1;
196            }
197            b',' => {
198                tokens.push(Token::Comma(start));
199                position += 1;
200            }
201            b'=' => {
202                tokens.push(Token::Equals(start));
203                position += 1;
204            }
205            b':' => {
206                tokens.push(Token::Colon(start));
207                position += 1;
208            }
209            digit if digit.is_ascii_digit() => {
210                while position < bytes.len() && bytes[position].is_ascii_digit() {
211                    position += 1;
212                }
213                let text = source[start..position].to_string();
214                tokens.push(Token::IntegerLiteral(text, start));
215            }
216            alpha if alpha.is_ascii_alphabetic() || alpha == b'_' => {
217                while position < bytes.len()
218                    && (bytes[position].is_ascii_alphanumeric() || bytes[position] == b'_')
219                {
220                    position += 1;
221                }
222                let name = source[start..position].to_string();
223                tokens.push(Token::Identifier(name, start));
224            }
225            other => {
226                return Err(ParseError::UnexpectedCharacter(other as char, start));
227            }
228        }
229    }
230    tokens.push(Token::EndOfInput(bytes.len()));
231    Ok(tokens)
232}

234/// Performs the advanced native `parse_statement` operation.
235pub fn parse_statement(
236    source: &str,
237    context: &Context,
238    bindings: &HashMap<String, ExpressionReference>,
239) -> Result<Statement, ParseError> {
240    // Top-level entry point. Tokenises the source, dispatches to the meta-command
241    // path on a leading colon, recognises an assignment as IDENT '=' expression,
242    // and otherwise parses a single expression. The result must consume every
243    // token; trailing input is rejected as an unexpected token.
244    let tokens = tokenize(source)?;
245    let mut state = ParserState {
246        tokens,
247        position: 0,
248        context,
249        bindings,
250    };
251    let result = if matches!(state.peek(), Token::Colon(_)) {
252        state.advance();
253        state.parse_meta_command()?
254    } else if state.is_assignment_lookahead() {
255        state.parse_assignment()?
256    } else {
257        let value = state.parse_expression()?;
258        Statement::Expression(value)
259    };
260    state.expect_end_of_input()?;
261    Ok(result)
262}

264struct ParserState<'session> {
265    tokens: Vec<Token>,
266    position: usize,
267    context: &'session Context,
268    bindings: &'session HashMap<String, ExpressionReference>,
269}
270
271impl<'session> ParserState<'session> {
272    fn peek(&self) -> &Token {
273        // Inspect the current token without consuming it. Positions past the end
274        // are represented by the end_of_input sentinel that tokenize always emits.
275        &self.tokens[self.position]
276    }
277
278    fn peek_ahead(&self, offset: usize) -> &Token {
279        // Look ahead an arbitrary number of tokens. Bounded by the sentinel so the
280        // returned value always exists.
281        let index = (self.position + offset).min(self.tokens.len() - 1);
282        &self.tokens[index]
283    }
284
285    fn advance(&mut self) {
286        // Consume the current token, advancing the cursor by one slot. Stops at
287        // the sentinel so further advances are no-ops.
288        if self.position + 1 < self.tokens.len() {
289            self.position += 1;
290        }
291    }
292
293    fn expect_end_of_input(&self) -> Result<(), ParseError> {
294        // Confirm the cursor sits on the end_of_input sentinel. Surplus tokens
295        // surface as unexpected_token so the user sees a precise complaint.
296        match self.peek() {
297            Token::EndOfInput(_) => Ok(()),
298            other => Err(ParseError::UnexpectedToken(
299                token_description(other),
300                token_position(other),
301            )),
302        }
303    }
304
305    fn expect_identifier(&mut self) -> Result<String, ParseError> {
306        // Consume the current token as an identifier and return its name. Any
307        // other token, including end_of_input, raises an expected_identifier error
308        // that records the offending position.
309        match self.peek() {
310            Token::Identifier(name, _) => {
311                let extracted = name.clone();
312                self.advance();
313                Ok(extracted)
314            }
315            other => Err(ParseError::ExpectedIdentifier(token_position(other))),
316        }
317    }
318
319    fn is_assignment_lookahead(&self) -> bool {
320        // Recognise the IDENT '=' prefix that distinguishes an assignment statement
321        // from a bare expression. Looks two tokens ahead without committing.
322        matches!(self.peek(), Token::Identifier(_, _))
323            && matches!(self.peek_ahead(1), Token::Equals(_))
324    }
325
326    fn parse_assignment(&mut self) -> Result<Statement, ParseError> {
327        // Consume IDENT '=' expression and return an assignment statement. The
328        // identifier is taken verbatim; binding lookup happens in the REPL when
329        // the resulting expression is later referenced.
330        let name = self.expect_identifier()?;
331        match self.peek() {
332            Token::Equals(_) => self.advance(),
333            other => {
334                return Err(ParseError::UnexpectedToken(
335                    token_description(other),
336                    token_position(other),
337                ));
338            }
339        }
340        let value = self.parse_expression()?;
341        Ok(Statement::Assignment(name, value))
342    }
343
344    fn parse_meta_command(&mut self) -> Result<Statement, ParseError> {
345        // Dispatch on the meta-command name that follows the leading colon. The
346        // grammar admits :quit, :show IDENT, and :assume IDENT FLAG+.
347        let command = self.expect_identifier()?;
348        match command.as_str() {
349            "quit" => Ok(Statement::Quit),
350            "show" => {
351                let name = self.expect_identifier()?;
352                Ok(Statement::Show(name))
353            }
354            "assume" => {
355                let name = self.expect_identifier()?;
356                let mut flags: Vec<AssumptionFlag> = Vec::new();
357                while !matches!(self.peek(), Token::EndOfInput(_)) {
358                    let flag_name = self.expect_identifier()?;
359                    flags.push(parse_assumption_flag(&flag_name)?);
360                }
361                if flags.is_empty() {
362                    return Err(ParseError::ExpectedAssumptionFlag);
363                }
364                Ok(Statement::Assume(name, flags))
365            }
366            _ => Err(ParseError::UnknownMetaCommand(command)),
367        }
368    }
369
370    fn parse_expression(&mut self) -> Result<ExpressionReference, ParseError> {
371        // Parse the additive layer. Successive '+' and '-' operators chain
372        // left-to-right; subtraction is rewritten through the subtract builder so
373        // automatic simplification produces the canonical sum form directly.
374        let mut left = self.parse_term()?;
375        loop {
376            match self.peek() {
377                Token::Plus(_) => {
378                    self.advance();
379                    let right = self.parse_term()?;
380                    left = sum(self.context, vec![left, right]);
381                }
382                Token::Minus(_) => {
383                    self.advance();
384                    let right = self.parse_term()?;
385                    left = subtract(self.context, left, right);
386                }
387                _ => break,
388            }
389        }
390        Ok(left)
391    }
392
393    fn parse_term(&mut self) -> Result<ExpressionReference, ParseError> {
394        // Parse the multiplicative layer. '*' and '/' chain left-to-right; division
395        // is routed through the divide builder so a denominator becomes a
396        // negative-exponent power factor in the canonical product.
397        let mut left = self.parse_factor()?;
398        loop {
399            match self.peek() {
400                Token::Star(_) => {
401                    self.advance();
402                    let right = self.parse_factor()?;
403                    left = product(self.context, vec![left, right]);
404                }
405                Token::Slash(_) => {
406                    self.advance();
407                    let right = self.parse_factor()?;
408                    left = divide(self.context, left, right);
409                }
410                _ => break,
411            }
412        }
413        Ok(left)
414    }
415
416    fn parse_factor(&mut self) -> Result<ExpressionReference, ParseError> {
417        // Parse leading '+' or '-' as unary operators over the exponentiation layer,
418        // so a sign binds more loosely than '^' and '-x^2' is '-(x^2)'. Unary minus
419        // negates through the builder so the result is canonicalized; unary plus only
420        // consumes the token. Recursive so that '--x' folds back to 'x'.
421        match self.peek() {
422            Token::Minus(_) => {
423                self.advance();
424                let operand = self.parse_factor()?;
425                Ok(negate(self.context, operand))
426            }
427            Token::Plus(_) => {
428                self.advance();
429                self.parse_factor()
430            }
431            _ => self.parse_power(),
432        }
433    }
434
435    fn parse_power(&mut self) -> Result<ExpressionReference, ParseError> {
436        // Parse the exponentiation layer. '^' binds right-associatively, and the
437        // exponent is a full signed factor, so '2^3^2' is '2^(3^2)' and 'x^-2' is
438        // 'x^(-2)', while the base is a primary that carries no sign of its own.
439        let base = self.parse_primary()?;
440        if matches!(self.peek(), Token::Caret(_)) {
441            self.advance();
442            let exponent = self.parse_factor()?;
443            Ok(power(self.context, base, exponent))
444        } else {
445            Ok(base)
446        }
447    }
448
449    fn parse_primary(&mut self) -> Result<ExpressionReference, ParseError> {
450        // Parse an atomic expression: a numeric literal, an identifier (possibly
451        // followed by a parenthesised argument list to form a function call), or
452        // a parenthesised sub-expression. Anything else is an unexpected token.
453        match self.peek().clone() {
454            Token::IntegerLiteral(text, _) => {
455                self.advance();
456                let value = integer_from_decimal(&text)
457                    .ok_or_else(|| ParseError::IntegerParseError(text))?;
458                Ok(integer(self.context, value))
459            }
460            Token::Identifier(name, _) => {
461                self.advance();
462                if matches!(self.peek(), Token::LeftParenthesis(_)) {
463                    self.advance();
464                    let arguments = self.parse_argument_list()?;
465                    match self.peek() {
466                        Token::RightParenthesis(_) => self.advance(),
467                        other => {
468                            return Err(ParseError::ExpectedRightParenthesis(token_position(
469                                other,
470                            )));
471                        }
472                    }
473                    if name == "integrate" {
474                        return build_integrate_call(self.context, arguments, &name);
475                    }
476                    if name == "summation" {
477                        return build_summation_call(self.context, arguments, &name);
478                    }
479                    if name == "indexed_product" {
480                        return build_product_call(self.context, arguments, &name);
481                    }
482                    if name == "laplace_transform" {
483                        return build_laplace_transform_call(self.context, arguments, &name);
484                    }
485                    if name == "inverse_laplace_transform" {
486                        return build_inverse_laplace_transform_call(
487                            self.context,
488                            arguments,
489                            &name,
490                        );
491                    }
492                    if name == "fourier_transform" {
493                        return build_fourier_transform_call(self.context, arguments, &name);
494                    }
495                    if name == "inverse_fourier_transform" {
496                        return build_inverse_fourier_transform_call(
497                            self.context,
498                            arguments,
499                            &name,
500                        );
501                    }
502                    let function = resolve_function_name(self.context, &name);
503                    Ok(function_call(self.context, function, arguments))
504                } else {
505                    Ok(resolve_identifier(self.context, &name, self.bindings))
506                }
507            }
508            Token::LeftParenthesis(_) => {
509                self.advance();
510                let inner = self.parse_expression()?;
511                match self.peek() {
512                    Token::RightParenthesis(_) => {
513                        self.advance();
514                        Ok(inner)
515                    }
516                    other => Err(ParseError::ExpectedRightParenthesis(token_position(other))),
517                }
518            }
519            Token::EndOfInput(_) => Err(ParseError::UnexpectedEndOfInput),
520            other => Err(ParseError::UnexpectedToken(
521                token_description(&other),
522                token_position(&other),
523            )),
524        }
525    }
526
527    fn parse_argument_list(&mut self) -> Result<Vec<ExpressionReference>, ParseError> {
528        // Parse a comma-separated argument list, stopping at the closing
529        // parenthesis. An empty list is permitted so callers can express zero-arg
530        // function applications. The closing parenthesis itself is consumed by the
531        // caller.
532        if matches!(self.peek(), Token::RightParenthesis(_)) {
533            return Ok(Vec::new());
534        }
535        let mut arguments: Vec<ExpressionReference> = Vec::new();
536        arguments.push(self.parse_expression()?);
537        while matches!(self.peek(), Token::Comma(_)) {
538            self.advance();
539            arguments.push(self.parse_expression()?);
540        }
541        Ok(arguments)
542    }
543}

545fn resolve_function_name(context: &Context, name: &str) -> FunctionIdentifier {
546    // Map a textual function name to a FunctionIdentifier. Built-in names take
547    // precedence so that the parser produces the same identifier the simplifier
548    // recognises for trivial-identity reductions; any other name becomes a
549    // user-defined function backed by a fresh symbol identifier.
550    //
551    // The canonical spellings come from the serialized grammar's name table, so
552    // this parser and the serializer always agree on which names denote builtins:
553    // a name the serializer emits for a builtin must never come back as an
554    // undefined function of the same name. The extra arms below are short
555    // aliases this surface accepts in addition to the canonical spelling.
556    if let Some(builtin) = crate::serialize::builtin_function_from_name(name) {
557        return FunctionIdentifier::Builtin(builtin);
558    }
559    match name {
560        "asinh" => FunctionIdentifier::Builtin(BuiltinFunction::Arcsinh),
561        "acosh" => FunctionIdentifier::Builtin(BuiltinFunction::Arccosh),
562        "atanh" => FunctionIdentifier::Builtin(BuiltinFunction::Arctanh),
563        "asech" => FunctionIdentifier::Builtin(BuiltinFunction::Arcsech),
564        "acsch" => FunctionIdentifier::Builtin(BuiltinFunction::Arccsch),
565        "acoth" => FunctionIdentifier::Builtin(BuiltinFunction::Arccoth),
566        "asin" => FunctionIdentifier::Builtin(BuiltinFunction::Arcsin),
567        "acos" => FunctionIdentifier::Builtin(BuiltinFunction::Arccos),
568        "atan" => FunctionIdentifier::Builtin(BuiltinFunction::Arctan),
569        "asec" => FunctionIdentifier::Builtin(BuiltinFunction::Arcsec),
570        "acsc" => FunctionIdentifier::Builtin(BuiltinFunction::Arccsc),
571        "acot" => FunctionIdentifier::Builtin(BuiltinFunction::Arccot),
572        "ceil" => FunctionIdentifier::Builtin(BuiltinFunction::Ceiling),
573        _ => FunctionIdentifier::User(context.symbol_table.intern_symbol(name)),
574    }
575}

577fn resolve_identifier(
578    context: &Context,
579    name: &str,
580    bindings: &HashMap<String, ExpressionReference>,
581) -> ExpressionReference {
582    // Resolve a bare identifier in three layers: a session-level binding takes
583    // precedence (the user typed `x = ...` earlier and now writes `x`), then the
584    // reserved constant names, then a fresh or recurring symbol leaf.
585    if let Some(value) = bindings.get(name) {
586        return Arc::clone(value);
587    }
588    match name {
589        "pi" => context
590            .interner
591            .intern(ExpressionKind::Constant(BuiltinConstant::Pi)),
592        "e" => context
593            .interner
594            .intern(ExpressionKind::Constant(BuiltinConstant::E)),
595        "euler_gamma" => context
596            .interner
597            .intern(ExpressionKind::Constant(BuiltinConstant::EulerGamma)),
598        "i" => context
599            .interner
600            .intern(ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit)),
601        "infinity" => context
602            .interner
603            .intern(ExpressionKind::Constant(BuiltinConstant::Infinity)),
604        "undefined" => context
605            .interner
606            .intern(ExpressionKind::Constant(BuiltinConstant::Undefined)),
607        "true" => context
608            .interner
609            .intern(ExpressionKind::Constant(BuiltinConstant::LogicalTrue)),
610        "false" => context
611            .interner
612            .intern(ExpressionKind::Constant(BuiltinConstant::LogicalFalse)),
613        _ => symbol(context, name),
614    }
615}

617fn build_integrate_call(
618    context: &Context,
619    arguments: Vec<ExpressionReference>,
620    function_name: &str,
621) -> Result<ExpressionReference, ParseError> {
622    // Construct an integral expression from the argument list of an
623    // `integrate(...)` primary call. Two valid shapes are accepted:
624    // `integrate(integrand, variable)` for indefinite integration and
625    // `integrate(integrand, variable, lower, upper)` for definite integration.
626    // The variable argument must be a bare symbol because the integrand binder
627    // requires a symbol identifier; any other shape is rejected as an
628    // unexpected token.
629    let argument_count = arguments.len();
630    if argument_count != 2 && argument_count != 4 {
631        return Err(ParseError::UnexpectedToken(
632            format!(
633                "call to {} with {} arguments",
634                function_name, argument_count
635            ),
636            0,
637        ));
638    }
639    let mut iterator = arguments.into_iter();
640    let integrand = iterator.next().unwrap();
641    let variable_expression = iterator.next().unwrap();
642    let variable_identifier = match &variable_expression.kind {
643        ExpressionKind::Symbol(identifier) => *identifier,
644        _ => {
645            return Err(ParseError::UnexpectedToken(
646                format!("variable position in {} call", function_name),
647                0,
648            ));
649        }
650    };
651    let lower_bound = iterator.next();
652    let upper_bound = iterator.next();
653    Ok(integral(
654        context,
655        integrand,
656        variable_identifier,
657        lower_bound,
658        upper_bound,
659    ))
660}

662fn build_laplace_transform_call(
663    context: &Context,
664    arguments: Vec<ExpressionReference>,
665    function_name: &str,
666) -> Result<ExpressionReference, ParseError> {
667    // Construct a laplace_transform expression from the argument list of
668    // a `laplace_transform(...)` primary call. The accepted shape is
669    // `laplace_transform(target, time_variable, frequency_variable)`
670    // where both variable arguments must be bare symbols; any other
671    // shape is rejected as an unexpected token. Routes through the
672    // eager builder so the forward table and algebraic rules fire at parse time.
673    let argument_count = arguments.len();
674    if argument_count != 3 {
675        return Err(ParseError::UnexpectedToken(
676            format!(
677                "call to {} with {} arguments",
678                function_name, argument_count
679            ),
680            0,
681        ));
682    }

684    let target = iterator.next().unwrap();
685    let time_expression = iterator.next().unwrap();
686    let frequency_expression = iterator.next().unwrap();
687    let time_identifier = match &time_expression.kind {
688        ExpressionKind::Symbol(identifier) => *identifier,
689        _ => {
690            return Err(ParseError::UnexpectedToken(
691                format!("time variable position in {} call", function_name),
692                0,
693            ));
694        }
695    };
696    let frequency_identifier = match &frequency_expression.kind {
697        ExpressionKind::Symbol(identifier) => *identifier,
698        _ => {
699            return Err(ParseError::UnexpectedToken(
700                format!("frequency variable position in {} call", function_name),
701                0,
702            ));
703        }
704    };
705    Ok(laplace_transform(
706        context,
707        target,
708        time_identifier,
709        frequency_identifier,
710    ))
711}

862fn build_summation_call(
863    context: &Context,
864    arguments: Vec<ExpressionReference>,
865    function_name: &str,
866) -> Result<ExpressionReference, ParseError> {
867    // Construct a summation expression from the argument list of a
868    // `summation(...)` primary call. Two valid shapes are accepted:
869    // `summation(summand, index)` for the indefinite (anti-difference)
870    // sum and `summation(summand, index, lower, upper)` for the definite
871    // sum. The index argument must be a bare symbol because the binder
872    // requires a symbol identifier; any other shape is rejected as an
873    // unexpected token.
874    let argument_count = arguments.len();
875    if argument_count != 2 && argument_count != 4 {
876        return Err(ParseError::UnexpectedToken(
877            format!(
878                "call to {} with {} arguments",
879                function_name, argument_count
880            ),
881            0,
882        ));
883    }
884    let mut iterator = arguments.into_iter();
885    let summand = iterator.next().unwrap();
886    let index_expression = iterator.next().unwrap();
887    let index_identifier = match &index_expression.kind {
888        ExpressionKind::Symbol(identifier) => *identifier,
889        _ => {
890            return Err(ParseError::UnexpectedToken(
891                format!("index position in {} call", function_name),
892                0,
893            ));
894        }
895    };
896    let lower_bound = iterator.next();
897    let upper_bound = iterator.next();
898    Ok(summation(
899        context,
900        summand,
901        index_identifier,
902        lower_bound,
903        upper_bound,
904    ))
905}

907fn build_product_call(
908    context: &Context,
909    arguments: Vec<ExpressionReference>,
910    function_name: &str,
911) -> Result<ExpressionReference, ParseError> {
912    // Construct an indexed_product expression from the argument list of an
913    // `indexed_product(...)` primary call. Two valid shapes are accepted:
914    // `indexed_product(multiplicand, index)` for the indefinite (anti-quotient)
915    // product and `indexed_product(multiplicand, index, lower, upper)` for the
916    // definite product. The index argument must be a bare symbol because the
917    // binder requires a symbol identifier; any other shape is rejected as an
918    // unexpected token. This parallels build_summation_call exactly.
919    let argument_count = arguments.len();
920    if argument_count != 2 && argument_count != 4 {
921        return Err(ParseError::UnexpectedToken(
922            format!(
923                "call to {} with {} arguments",
924                function_name, argument_count
925            ),
926            0,
927        ));
928    }
929    let mut iterator = arguments.into_iter();
930    let multiplicand = iterator.next().unwrap();
931    let index_expression = iterator.next().unwrap();
932    let index_identifier = match &index_expression.kind {
933        ExpressionKind::Symbol(identifier) => *identifier,
934        _ => {
935            return Err(ParseError::UnexpectedToken(
936                format!("index position in {} call", function_name),
937                0,
938            ));
939        }
940    };
941    let lower_bound = iterator.next();
942    let upper_bound = iterator.next();
943    Ok(indexed_product(
944        context,
945        multiplicand,
946        index_identifier,
947        lower_bound,
948        upper_bound,
949    ))
950}

952fn parse_assumption_flag(name: &str) -> Result<AssumptionFlag, ParseError> {
953    // Recognise the flag vocabulary accepted by the :assume meta-command. Flags are
954    // spelled exactly as the registered scalar predicates, so the whole predicate
955    // vocabulary — including even, odd, prime, and algebraic — is reachable without
956    // a second name table that could drift from the registry.
957    match crate::assumptions::PredicateIdentifier::from_name(name) {
958        Some(crate::assumptions::PredicateIdentifier::CurrentScalar(predicate)) => Ok(predicate),
959        _ => Err(ParseError::UnknownAssumptionFlag(name.to_string())),
960    }
961}
