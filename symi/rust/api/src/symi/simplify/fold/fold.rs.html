10/// Performs the advanced native `intern_number` operation.
11pub fn intern_number(context: &Context, value: Number) -> ExpressionReference {
12    // Materialize a number value as an interned expression_reference. The integer
13    // and rational variants of ExpressionKind shadow the corresponding number
14    // variants, so the conversion is a direct dispatch with no extra normalization
15    // (number values are themselves already canonicalized to lowest terms).
16    match value {
17        Number::Integer(integer_value) => context
18            .interner
19            .intern(ExpressionKind::Integer(integer_value)),
20        Number::Rational(rational_value) => context
21            .interner
22            .intern(ExpressionKind::Rational(rational_value)),
23    }
24}

26/// Performs the advanced native `extract_number` operation.
27pub fn extract_number(reference: &ExpressionReference) -> Option<Number> {
28    // If the supplied expression is a numeric leaf, return its number value as an
29    // owned clone. Returns None for every non-numeric kind, signaling that the
30    // operand must be processed symbolically.
31    match &reference.kind {
32        ExpressionKind::Integer(value) => Some(Number::Integer(value.clone())),
33        ExpressionKind::Rational(value) => Some(Number::Rational(value.clone())),
34        _ => None,
35    }
36}

38/// Performs the advanced native `integer_one` operation.
39pub fn integer_one() -> Number {
40    // Multiplicative identity used as the seed value when folding numeric factors
41    // of a product.
42    Number::Integer(Integer::from(1))
43}
44
45/// Performs the advanced native `integer_zero` operation.
46pub fn integer_zero() -> Number {
47    // Additive identity used as the seed value when folding numeric terms of a
48    // sum.
49    Number::Integer(Integer::from(0))
50}

52/// Performs the advanced native `extract_sum_coefficient` operation.
53pub fn extract_sum_coefficient(
54    context: &Context,
55    operand: ExpressionReference,
56) -> (Number, ExpressionReference) {
57    // Decompose a non-numeric sum operand into (coefficient, term-base) such that
58    // the original operand equals coefficient * term-base under the canonical
59    // representation. A product whose first factor is numeric peels that numeric
60    // off and reassembles the remaining factors into the term-base. Any other
61    // shape is its own term-base with an implicit coefficient of one.
62    match &operand.kind {
63        ExpressionKind::Product(factors) => {
64            if let Some(first) = factors.first() {
65                if let Some(numeric) = extract_number(first) {
66                    let rest: Vec<ExpressionReference> =
67                        factors[1..].iter().map(Arc::clone).collect();
68                    let base = if rest.len() == 1 {
69                        rest.into_iter().next().unwrap()
70                    } else {
71                        context.interner.intern(ExpressionKind::Product(rest))
72                    };
73                    return (numeric, base);
74                }
75            }
76            (integer_one(), operand)
77        }
78        _ => (integer_one(), operand),
79    }
80}

82/// Performs the advanced native `extract_product_split` operation.
83pub fn extract_product_split(
84    context: &Context,
85    operand: ExpressionReference,
86) -> (ExpressionReference, ExpressionReference) {
87    // Decompose a product operand into (base, exponent) for like-base grouping. A
88    // power node yields its base and exponent directly; an exp call is viewed as
89    // the power e^argument so the product machinery combines exp(x) * exp(2x) the
90    // same way it combines 2^x * 2^(1-x) (simplify_power re-canonicalizes the
91    // recombined base-e power back to an exp call). Any other shape is its own
92    // base with an interned exponent of one, which lets the caller use the
93    // standard sum-of-exponents combination path uniformly.
94    match &operand.kind {
95        ExpressionKind::Power(base, exponent) => (Arc::clone(base), Arc::clone(exponent)),
96        ExpressionKind::FunctionCall(
97            FunctionIdentifier::Builtin(BuiltinFunction::Exp),
98            arguments,
99        ) if arguments.len() == 1 => {
100            let base_e = context
101                .interner
102                .intern(ExpressionKind::Constant(BuiltinConstant::E));
103            (base_e, Arc::clone(&arguments[0]))
104        }
105        _ => {
106            let one = context
107                .interner
108                .intern(ExpressionKind::Integer(Integer::from(1)));
109            (operand, one)
110        }
111    }
112}
