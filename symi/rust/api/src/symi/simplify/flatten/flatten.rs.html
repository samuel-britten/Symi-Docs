5/// Performs the advanced native `flatten_sum` operation.
6pub fn flatten_sum(operands: Vec<ExpressionReference>) -> Vec<ExpressionReference> {
7    // Splice the children of any sum operand directly into the result, producing a
8    // flat list with no sum-of-sums structure. Non-sum operands are passed through
9    // unchanged. Operates on a single level: callers rely on the ASAE invariant
10    // that any nested sum has itself already been flattened, so a single pass
11    // suffices to remove all sum-of-sum nesting.
12    let mut result: Vec<ExpressionReference> = Vec::with_capacity(operands.len());
13    for operand in operands {
14        match &operand.kind {
15            ExpressionKind::Sum(children) => {
16                for child in children {
17                    result.push(Arc::clone(child));
18                }
19            }
20            _ => result.push(operand),
21        }
22    }
23    result
24}

26/// Performs the advanced native `flatten_product` operation.
27pub fn flatten_product(operands: Vec<ExpressionReference>) -> Vec<ExpressionReference> {
28    // Splice the children of any product operand directly into the result. Mirrors
29    // flatten_sum for products. Single-level pass justified by the same ASAE
30    // invariant for nested products.
31    let mut result: Vec<ExpressionReference> = Vec::with_capacity(operands.len());
32    for operand in operands {
33        match &operand.kind {
34            ExpressionKind::Product(children) => {
35                for child in children {
36                    result.push(Arc::clone(child));
37                }
38            }
39            _ => result.push(operand),
40        }
41    }
42    result
43}

51    use crate::expression::ExpressionKind;
52    use crate::interner::Context;

54    fn integer_expression(context: &Context, value: i64) -> ExpressionReference {
55        context
56            .interner
57            .intern(ExpressionKind::Integer(Integer::from(value)))
58    }
59
60    fn symbol_expression(context: &Context, name: &str) -> ExpressionReference {
61        let identifier = context.symbol_table.intern_symbol(name);
62        context.interner.intern(ExpressionKind::Symbol(identifier))
63    }
64
65    #[test]
66    fn flatten_sum_unwraps_one_nested_sum() {
67        let mut context = Context::new();
68        let a = symbol_expression(&mut context, "a");
69        let b = symbol_expression(&mut context, "b");
70        let c = symbol_expression(&mut context, "c");
71        let inner = context
72            .interner
73            .intern(ExpressionKind::Sum(vec![Arc::clone(&a), Arc::clone(&b)]));
74        let result = flatten_sum(vec![inner, Arc::clone(&c)]);
75        assert_eq!(result.len(), 3);
76        assert_eq!(result[0].kind, a.kind);
77        assert_eq!(result[1].kind, b.kind);
78        assert_eq!(result[2].kind, c.kind);
79    }
80
81    #[test]
82    fn flatten_product_passes_non_products_through() {
83        let mut context = Context::new();
84        let two = integer_expression(&mut context, 2);
85        let x = symbol_expression(&mut context, "x");
86        let result = flatten_product(vec![Arc::clone(&two), Arc::clone(&x)]);
87        assert_eq!(result.len(), 2);
88        assert!(Arc::ptr_eq(&result[0], &two));
89        assert!(Arc::ptr_eq(&result[1], &x));
90    }
