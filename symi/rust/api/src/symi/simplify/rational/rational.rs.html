13const NODE_BUDGET: usize = 1_500;

16/// Performs the advanced native `cancel` operation.
17pub fn cancel(context: &Context, target: &ExpressionReference) -> ExpressionReference {
18    // Rational-function normal form: lift the expression to a single quotient of
19    // multivariate polynomials, divide numerator and denominator by their GCD, and
20    // rebuild. Non-polynomial kernels (sin(x), exp(x), x^(1/2), pi, ...) are
21    // abstracted to fresh symbols before lifting and substituted back afterwards —
22    // the generalized-symbol trick — so (sin(x)^2 - 1)/(sin(x) + 1) cancels exactly
23    // like (t^2 - 1)/(t + 1). Returns the input unchanged when the expression is
24    // not rational in its kernels, contains no division to normalize, or exceeds
25    // the node budget (this is a directed rewriter, so a bounded no-op is always
26    // safe).
27    rational_normal_form(context, target)
28}

30/// Performs the advanced native `together` operation.
31pub fn together(context: &Context, target: &ExpressionReference) -> ExpressionReference {
32    // Combine a sum of quotients over a single common denominator. The fraction
33    // lift used by the shared normal form inherently places every term over the
34    // common denominator and its canonicalizer divides out the GCD, so `together`
35    // and `cancel` coincide on this representation; both names are exposed because
36    // they are distinct user intents shared across mainstream CAS vocabularies.
37    rational_normal_form(context, target)
38}

40/// Performs the advanced native `presentation_normal_form` operation.
41pub fn presentation_normal_form(
42    context: &Context,
43    target: &ExpressionReference,
44) -> ExpressionReference {
45    // Cheap, bounded normal-form pass for solver outputs at the operation seams
46    // (integrate, solve, limit, ODE/recurrence solutions): the rational normal
47    // form only — numeric-radical canonicalization already happens at
48    // construction, and the full heuristic ladder is deliberately not run here.
49    // The result is kept only when its weighted complexity does not grow, so a
50    // seam can never make an output worse than the raw form.
51    let cancelled = cancel(context, target);
52    if Arc::ptr_eq(&cancelled, target) {
53        return cancelled;
54    }
55    if crate::simplify::heuristic::weighted_complexity(&cancelled)
56        <= crate::simplify::heuristic::weighted_complexity(target)
57    {
58        cancelled
59    } else {
60        Arc::clone(target)
61    }
62}

64/// Performs the advanced native `factor_with_kernels` operation.
65pub fn factor_with_kernels(context: &Context, target: &ExpressionReference) -> ExpressionReference {
66    // Kernel-aware factorisation: abstract every maximal transcendental kernel to
67    // a fresh symbol, factor the resulting polynomial over Q, and substitute the
68    // kernels back, so sin(x)^4 - cos(x)^4 factors to
69    // (sin^2 + cos^2)(sin + cos)(sin - cos) and the trigonometric passes can
70    // finish the collapse. Falls back to the input on any non-polynomial residue
71    // or when the node budget is exceeded.
72    if crate::simplify::heuristic::node_count(target) > NODE_BUDGET {
73        return Arc::clone(target);
74    }
75    let mut kernel_map = KernelAbstraction::new();
76    let abstracted = abstract_kernels(context, target, &mut kernel_map);
77    let factored = crate::factor::factor(context, &abstracted, None);
78    substitute_kernels_back(context, &factored, &kernel_map)
79}

81fn rational_normal_form(context: &Context, target: &ExpressionReference) -> ExpressionReference {
82    // Shared engine for cancel/together. Pipeline: budget guard, kernel
83    // abstraction, fraction lift through the rational_function layer (which
84    // canonicalizes via multivariate GCD), reification as numerator/denominator
85    // expressions, kernel back-substitution.
86    if crate::simplify::heuristic::node_count(target) > NODE_BUDGET {
87        return Arc::clone(target);
88    }
89    let mut kernel_map = KernelAbstraction::new();
90    assign_kernels_canonically(context, target, &mut kernel_map);
91    let abstracted = abstract_kernels(context, target, &mut kernel_map);
92    let mut variable_order = extract_variables_in_order(&abstracted, None);
93    if variable_order.is_empty() {
94        return Arc::clone(target);
95    }
96    // The variable order must be canonical, not appearance order: the head of the
97    // list becomes the main variable of the lifted fraction and therefore decides
98    // the reified grouping, and appearance order is not stable across cancel's own
99    // rewrites — an appearance-derived order made cancel non-idempotent. Ascending
100    // symbol identifier is the nesting the polynomial arithmetic layer itself
101    // normalises to, so requesting the same chain keeps the conversion and the
102    // arithmetic in agreement and the lifted representation canonical.
103    variable_order.sort_by_key(|identifier| identifier.0);
104    let fraction = match lift_to_fraction(&abstracted, &variable_order) {
105        Some(value) => value,
106        None => return Arc::clone(target),
107    };
108    let numerator_expression = polynomial_to_expression(context, &fraction.numerator);
109    let denominator_expression = polynomial_to_expression(context, &fraction.denominator);
110    let rebuilt =
111        crate::expression::builders::divide(context, numerator_expression, denominator_expression);
112    let restored = substitute_kernels_back(context, &rebuilt, &kernel_map);
113    restored
114}

116struct KernelAbstraction {
117    replacements: SubstitutionMap,
118    inverse: Vec<(ExpressionReference, ExpressionReference)>,
119    next_index: usize,
120}
121
122impl KernelAbstraction {
123    fn new() -> Self {
124        KernelAbstraction {
125            replacements: HashMap::new(),
126            inverse: Vec::new(),
127            next_index: 0,
128        }
129    }
130
131    fn symbol_for(
132        &mut self,
133        context: &Context,
134        kernel: &ExpressionReference,
135    ) -> ExpressionReference {
136        // Return the fresh stand-in symbol for a kernel, creating one on first
137        // encounter. Hash-consing makes structurally equal kernels pointer-equal,
138        // so the map keys on the expression reference directly.
139        if let Some(existing) = self.replacements.get(kernel) {
140            return Arc::clone(existing);
141        }
142        let name = format!("__rational_kernel_{}", self.next_index);
143        self.next_index += 1;
144        let identifier = context.symbol_table.intern_symbol(&name);
145        let stand_in = context.interner.intern(ExpressionKind::Symbol(identifier));
146        self.replacements
147            .insert(Arc::clone(kernel), Arc::clone(&stand_in));
148        self.inverse.push((stand_in.clone(), Arc::clone(kernel)));
149        stand_in
150    }
151}

153fn assign_kernels_canonically(
154    context: &Context,
155    target: &ExpressionReference,
156    kernels: &mut KernelAbstraction,
157) {
158    // Pre-pass that collects every maximal kernel and assigns the stand-in
159    // symbols in canonical (Cohen ordering) sequence rather than encounter
160    // sequence. Encounter order is not stable across rewrites of the same value
161    // — cancel's own output can present the kernels in a different order — and
162    // since the polynomial layer nests by stand-in identifier, an
163    // order-dependent assignment made cancel oscillate between two equal
164    // outputs instead of reaching a fixed point.
165    let mut collected: Vec<ExpressionReference> = Vec::new();
166    collect_kernels(target, &mut collected);
167    collected.sort_by(|left, right| {
168        crate::expression::ordering::compare(left, right, &context.symbol_table)
169    });
170    for kernel in collected.iter() {
171        let _ = kernels.symbol_for(context, kernel);
172    }
173}

175fn collect_kernels(target: &ExpressionReference, collected: &mut Vec<ExpressionReference>) {
176    // Gather the maximal non-rational kernels in the same shape discipline as
177    // abstract_kernels, without rewriting anything.
178    use ExpressionKind::*;
179    match &target.kind {
180        Integer(_) | Rational(_) | Symbol(_) => {}
181        Sum(children) | Product(children) => {
182            for child in children.iter() {
183                collect_kernels(child, collected);
184            }
185        }
186        Power(base, exponent) => match &exponent.kind {
187            Integer(_) => collect_kernels(base, collected),
188            _ => {
189                if !collected
190                    .iter()
191                    .any(|existing| existing.kind == target.kind)
192                {
193                    collected.push(Arc::clone(target));
194                }
195            }
196        },
197        _ => {
198            if !collected
199                .iter()
200                .any(|existing| existing.kind == target.kind)
201            {
202                collected.push(Arc::clone(target));
203            }
204        }
205    }
206}

208fn abstract_kernels(
209    context: &Context,
210    target: &ExpressionReference,
211    kernels: &mut KernelAbstraction,
212) -> ExpressionReference {
213    // Replace every maximal non-rational kernel with a fresh symbol. The rational
214    // skeleton — sums, products, integer powers, numeric and symbol leaves — is
215    // traversed; any other node (function calls, constants like pi and e,
216    // factorials, fractional or symbolic powers, binders, piecewise, sets) is
217    // treated as an opaque kernel in its entirety. Substituting kernels with
218    // independent symbols is equality-preserving for any value assignment, so the
219    // GCD cancellation downstream remains a sound identity.
220    use ExpressionKind::*;
221    match &target.kind {
222        Integer(_) | Rational(_) | Symbol(_) => Arc::clone(target),
223        Sum(children) => {
224            let rebuilt: Vec<ExpressionReference> = children
225                .iter()
226                .map(|child| abstract_kernels(context, child, kernels))
227                .collect();
228            crate::simplify::simplify_sum(context, rebuilt)
229        }
230        Product(children) => {
231            let rebuilt: Vec<ExpressionReference> = children
232                .iter()
233                .map(|child| abstract_kernels(context, child, kernels))
234                .collect();
235            crate::simplify::simplify_product(context, rebuilt)
236        }
237        Power(base, exponent) => match &exponent.kind {
238            Integer(_) => {
239                let rebuilt_base = abstract_kernels(context, base, kernels);
240                crate::simplify::simplify_power(context, rebuilt_base, Arc::clone(exponent))
241            }
242            _ => kernels.symbol_for(context, target),
243        },
244        _ => kernels.symbol_for(context, target),
245    }
246}

248fn substitute_kernels_back(
249    context: &Context,
250    target: &ExpressionReference,
251    kernels: &KernelAbstraction,
252) -> ExpressionReference {
253    // Invert the kernel abstraction by substituting each stand-in symbol with its
254    // original kernel expression.
255    if kernels.inverse.is_empty() {
256        return Arc::clone(target);
257    }
258    let mut mapping: SubstitutionMap = HashMap::new();
259    for (stand_in, kernel) in kernels.inverse.iter() {
260        mapping.insert(Arc::clone(stand_in), Arc::clone(kernel));
261    }
262    substitute(context, target, &mapping)
263}

265fn lift_to_fraction(
266    target: &ExpressionReference,
267    variable_order: &[SymbolIdentifier],
268) -> Option<RationalFunction> {
269    // Lift a kernel-free expression into a canonical quotient of multivariate
270    // polynomials. Nodes that are already polynomial lift directly; sums and
271    // products fold through the canonicalizing rational_function arithmetic;
272    // negative integer powers lift the base and invert. Anything else aborts the
273    // lift so the caller can leave the input untouched.
274    if let Some(polynomial_value) = expression_to_polynomial(target, variable_order) {
275        return Some(RationalFunction::from_polynomial(polynomial_value));
276    }
277    use ExpressionKind::*;
278    match &target.kind {
279        Sum(children) => {
280            let mut total = RationalFunction::zero(variable_order[0]);
281            for child in children.iter() {
282                let lifted = lift_to_fraction(child, variable_order)?;
283                total = total.add(&lifted);
284            }
285            Some(total)
286        }
287        Product(children) => {
288            let mut total = RationalFunction::one(variable_order[0]);
289            for child in children.iter() {
290                let lifted = lift_to_fraction(child, variable_order)?;
291                total = total.multiply(&lifted);
292            }
293            Some(total)
294        }
295        Power(base, exponent) => {
296            let exponent_value = match &exponent.kind {
297                Integer(value) => i64::try_from(value).ok()?,
298                _ => return None,
299            };
300            if exponent_value.abs() > POWER_REPEAT_LIMIT {
301                return None;
302            }
303            let lifted_base = lift_to_fraction(base, variable_order)?;
304            let positive_form = if exponent_value < 0 {
305                lifted_base.reciprocal()?
306            } else {
307                lifted_base
308            };
309            let mut accumulated = RationalFunction::one(variable_order[0]);
310            for _ in 0..exponent_value.unsigned_abs() {
311                accumulated = accumulated.multiply(&positive_form);
312            }
313            Some(accumulated)
314        }
315        _ => None,
316    }
317}

330    fn cancel_collapses_difference_of_squares_quotient() {
331        let mut context = Context::new();
332        let x = symbol(&mut context, "x");
333        let two = integer_value(&mut context, 2);
334        let x_squared = power(&mut context, Arc::clone(&x), two);
335        let minus_one = integer_value(&mut context, -1);
336        let numerator = sum(&mut context, vec![x_squared, minus_one]);
337        let one = integer_value(&mut context, 1);
338        let denominator = sum(&mut context, vec![Arc::clone(&x), one]);
339        let quotient = divide(&mut context, numerator, denominator);
340        let result = cancel(&mut context, &quotient);
341        let minus_one_again = integer_value(&mut context, -1);
342        let expected = sum(&mut context, vec![Arc::clone(&x), minus_one_again]);
343        assert!(
344            Arc::ptr_eq(&result, &expected),
345            "expected x - 1, got {}",
346            serialize(&context, &result)
347        );
