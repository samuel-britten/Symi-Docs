25    pub(crate) original: ExpressionReference,

27    pub(crate) expanded_square: ExpressionReference,
28    pub(crate) root_sign: BinomialSquareSign,

31pub(crate) fn discover_expanded_square_witness(
32    context: &Context,
33    radicand: &ExpressionReference,
34) -> Option<ExpandedSquareWitness> {
35    // Recognise a canonical three-term sum that is the exact expansion of a
36    // binomial square, returning the binomial as a replayed witness. The search is
37    // fixed and tiny: the three ways of choosing the two diagonal terms times the
38    // two binomial signs, six full candidates at most. A candidate is admitted
39    // only when both chosen terms have principal square roots that square back to
40    // them exactly and the expansion of the candidate square is the interned
41    // original sum, so the middle term, its coefficient, its sign, and every
42    // algebraic relation among the generators (sqrt(r)^2 = r and the like) are all
43    // verified by replay rather than by an ad hoc 2*a*b pattern — which also makes
44    // the recognizer independent of ASAE child ordering. The witness proves
45    // squareness only; the principal-power branch guard belongs to the consumer.
46    // Discovery is depth-limited because rooting a diagonal term re-enters power
47    // simplification, which may meet another three-term sum.
48    let children = match &radicand.kind {
49        ExpressionKind::Sum(children) if children.len() == 3 => children.clone(),
50        _ => return None,
51    };
52    let _discovery_level =
53        context.enter_expanded_square_discovery(EXPANDED_SQUARE_DISCOVERY_DEPTH_LIMIT)?;
54    search_expanded_square_candidates(context, radicand, &children)
55}

57fn search_expanded_square_candidates(
58    context: &Context,
59    radicand: &ExpressionReference,
60    children: &[ExpressionReference],
61) -> Option<ExpandedSquareWitness> {
62    // Deterministic candidate sweep in canonical order: diagonal pairs (0,1),
63    // (0,2), (1,2), each with the addition sign before the subtraction sign. The
64    // first candidate whose replay succeeds wins, so the recognizer is a function
65    // of the interned input alone. Each term is rooted once and the result reused
66    // across the pairs it appears in, so a three-term sum costs three root
67    // attempts and at most six square replays.
68    let term_roots: Vec<Option<ExpressionReference>> = children
69        .iter()
70        .map(|child| exact_term_square_root(context, child))
71        .collect();
72    let diagonal_pairs = [(0_usize, 1_usize), (0, 2), (1, 2)];
73    for (first_index, second_index) in diagonal_pairs {
74        let (first_root, second_root) = match (&term_roots[first_index], &term_roots[second_index])
75        {
76            (Some(first), Some(second)) => (Arc::clone(first), Arc::clone(second)),
77            _ => continue,
78        };
79        for root_sign in [
80            BinomialSquareSign::Addition,
81            BinomialSquareSign::Subtraction,
82        ] {
83            let signed_second = match root_sign {
84                BinomialSquareSign::Addition => Arc::clone(&second_root),
85                BinomialSquareSign::Subtraction => {
86                    let minus_one = builders::integer_value(context, -1);
87                    crate::simplify::simplify_product(
88                        context,
89                        vec![minus_one, Arc::clone(&second_root)],
90                    )
91                }
92            };
93            let candidate = crate::simplify::simplify_sum(
94                context,
95                vec![Arc::clone(&first_root), signed_second],
96            );
97            let two = builders::integer_value(context, 2);
98            let squared = crate::simplify::simplify_power(context, Arc::clone(&candidate), two);
99            let expanded = expand(context, &squared);
100            if !Arc::ptr_eq(&expanded, radicand) {
101                continue;
102            }
103            return Some(ExpandedSquareWitness {
104                original: Arc::clone(radicand),
105                root: candidate,
106                expanded_square: expanded,
107                root_sign,
108            });
109        }
110    }
111    None
112}

114fn exact_term_square_root(
115    context: &Context,
116    term: &ExpressionReference,
117) -> Option<ExpressionReference> {
118    // The principal square root of one summand, admitted only when squaring it
119    // reproduces the summand as the very same interned node. Branch-safe by
120    // construction: the root is whatever ordinary power simplification produces
121    // under the current assumptions, and the exact square-back test is what makes
122    // the candidate usable, so no unproved radical ever enters a candidate.
123    let half = builders::rational_value(context, 1, 2);
124    let root = crate::simplify::simplify_power(context, Arc::clone(term), half);
125    let two = builders::integer_value(context, 2);
126    let squared = crate::simplify::simplify_power(context, Arc::clone(&root), two);
127    if Arc::ptr_eq(&squared, term) {
128        Some(root)
129    } else {
130        None
131    }
132}

134pub(crate) fn expanded_square_absolute_value(
135    context: &Context,
136    radicand: &ExpressionReference,
137) -> Option<ExpressionReference> {
138    // The square-root consumer of an expanded-square witness: sqrt(u^2) -> abs(u)
139    // for a witness proved real. Realness is the whole guard the identity needs on
140    // the reals, and the absolute value is then handed to the ordinary function
141    // simplification, whose exact sign procedure folds abs(u) to u or -u whenever
142    // the sign is decided. A witness whose realness is unknown declines, because
143    // the principal square root of a complex square is not its absolute value.
144    let witness = discover_expanded_square_witness(context, radicand)?;
145    if !crate::assumptions::range_evidence::proves_real(&context.symbol_table, &witness.root) {
146        return None;
147    }
148    Some(crate::simplify::simplify_function_call(
149        context,
150        FunctionIdentifier::Builtin(BuiltinFunction::Abs),
151        vec![witness.root],
152    ))
153}

155pub(crate) fn expanded_square_nested_power(
156    context: &Context,
157    radicand: &ExpressionReference,
158    exponent: &ExpressionReference,
159) -> Option<ExpressionReference> {
160    // The general-exponent consumer of an expanded-square witness:
161    // (u^2)^a -> u^(2a) for a witness proved strictly positive. Positivity, not
162    // mere realness, is what the principal-power composition law needs: the
163    // expanded base being a square makes it nonnegative, but a zero or
164    // sign-unknown witness leaves both the branch and the definedness of a
165    // symbolic non-integer exponent undecided. Rebuilding through ordinary power
166    // simplification is deliberate — it lets like-base exponent collection in the
167    // enclosing product finish the job, so no target-shaped exponent is ever
168    // recognised here.
169    let witness = discover_expanded_square_witness(context, radicand)?;
170    if crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &witness.root)
171        != TruthValue::Yes
172    {
173        return None;
174    }
175    let two = builders::integer_value(context, 2);
176    let doubled_exponent =
177        crate::simplify::simplify_product(context, vec![two, Arc::clone(exponent)]);
178    Some(crate::simplify::simplify_power(
179        context,
180        witness.root,
181        doubled_exponent,
182    ))
183}

185pub(crate) struct PerfectSquareRadicalRewrite {
186    pub(crate) rewritten: ExpressionReference,
187    pub(crate) required_formula: ExpressionReference,
188    pub(crate) introduced_absolute_arguments: Vec<ExpressionReference>,
189}

191pub(crate) fn rewrite_global_perfect_square_radical(
192    context: &Context,
193    radicand: &ExpressionReference,
194) -> Option<PerfectSquareRadicalRewrite> {
195    let factors = match &radicand.kind {
196        ExpressionKind::Product(factors) => factors.clone(),
197        _ => vec![radicand.clone()],
198    };
199    let half = builders::rational_value(context, 1, 2);
200    let mut extracted = Vec::new();
201    let mut residual = Vec::new();
202    let mut introduced_absolute_arguments = Vec::new();
203    for factor in factors {
204        if let Some((base, half_exponent)) = even_power_factor(&factor) {
205            if crate::assumptions::range_evidence::proves_real(&context.symbol_table, &base) {
206                let absolute = builders::function_call(
207                    context,
208                    FunctionIdentifier::Builtin(BuiltinFunction::Abs),
209                    vec![base.clone()],
210                );
211                let half_exponent_expression = context
212                    .interner
213                    .intern(ExpressionKind::Integer(half_exponent));
214                extracted.push(builders::power(context, absolute, half_exponent_expression));
215                introduced_absolute_arguments.push(base);
216                continue;
217            }
218        }
219        if matches!(
220            factor.kind,
221            ExpressionKind::Integer(_) | ExpressionKind::Rational(_)
222        ) && crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &factor)
223            == TruthValue::Yes
224        {
225            let root = builders::power(context, factor.clone(), half.clone());
226            if !matches!(
227                &root.kind,
228                ExpressionKind::Power(base, exponent)
229                    if base.kind == factor.kind && exponent.kind == half.kind
230            ) {
231                extracted.push(root);
232                continue;
233            }
234        }
235        residual.push(factor);
236    }
237    if extracted.is_empty() {
238        return None;
239    }
240    if !residual.is_empty() {
241        let remaining_radicand = builders::product(context, residual);
242        extracted.push(builders::power(context, remaining_radicand, half));
243    }
244    let rewritten = builders::product(context, extracted);
245    Some(PerfectSquareRadicalRewrite {
246        rewritten,
247        required_formula: context
248            .interner
249            .intern(ExpressionKind::Constant(BuiltinConstant::LogicalTrue)),
250        introduced_absolute_arguments,
251    })
252}

254fn even_power_factor(factor: &ExpressionReference) -> Option<(ExpressionReference, exact_integer)> {
255    let ExpressionKind::Power(base, exponent) = &factor.kind else {
256        return None;
257    };
258    let ExpressionKind::Integer(integer_exponent) = &exponent.kind else {
259        return None;
260    };
261    if integer_exponent <= &exact_integer::ZERO
262        || integer_exponent.clone() % exact_integer::from(2) != exact_integer::ZERO
263    {
264        return None;
265    }
266    let half_exponent = integer_exponent.clone() / exact_integer::from(2);
267    Some((base.clone(), half_exponent))
268}

292    #[test]
293    fn a_discovered_witness_replays_to_its_own_original() {
294        // The witness contract: the recorded expansion is the interned original sum, and
295        // squaring the recorded root reproduces it. Both are checked here so a future
296        // change that returns a root without replaying it fails loudly.
297        let mut context = Context::new();
298        positive_symbol(&mut context, "r");
299        let radicand = build(
300            &mut context,
301            "sum([integer(\"1\"), product([integer(\"4\"), power(symbol(\"r\"), rational(\"1\", \"2\"))]), product([integer(\"4\"), symbol(\"r\")])])",
302        );
303        let witness =
304            discover_expanded_square_witness(&mut context, &radicand).expect("witness exists");
305        assert_eq!(witness.root_sign, BinomialSquareSign::Addition);
306        assert!(Arc::ptr_eq(&witness.original, &radicand));
307        assert!(Arc::ptr_eq(&witness.expanded_square, &radicand));
308        let two = builders::integer_value(&mut context, 2);
309        let squared = crate::simplify::simplify_power(&mut context, Arc::clone(&witness.root), two);
310        let expanded = expand(&mut context, &squared);
311        assert!(
312            Arc::ptr_eq(&expanded, &radicand),
313            "witness root {} does not square back",
314            serialize(&mut context, &witness.root)
315        );
316    }
