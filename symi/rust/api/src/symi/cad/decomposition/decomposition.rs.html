21/// Public native enumeration named `cell_kind`.
22pub enum CellKind {
23    // A cell that is a section in this level's variable: the sample coordinate is a
24    // root of the named projection factor (an index into the level's projection
25    // set, or into the input set at level 1 where the base polynomials are the
26    // level-1 projection set). Zero-dimensional in this variable.
27    /// Represents the `section` case.
28    Section {
29        #[doc = "Stores the `defining_polynomial` value."]
30        defining_polynomial: usize,
31    },
32    // A cell that is a sector in this level's variable: an open interval with a
33    // rational sample coordinate. Full-dimensional in this variable.
34    /// Represents the `sector` case.
35    Sector,
36}

39/// Public native data structure named `cad_cell`.
40pub struct CadCell {
41    // One cell of the cylindrical tree (Decision 6). `level` is the number of fixed
42    // coordinates (1..=n); `index` is the cylindrical address, one position per
43    // level; `sample_point` is a `real_algebraic_point` whose coordinates
44    // co-reside in one real-embedded Q(γ); `kind` tags the cell as a section (with
45    // the defining polynomial) or a sector in this level's variable; `parent` is the
46    // arena index of the base cell this stack rose over (None at level 1); and
47    // `sign_vector` records the sign of each input polynomial at the sample point,
48    // filled for full-dimensional (level-n) cells where every coordinate is fixed.
49    /// Stores the `level` value.
50    pub level: usize,
51    /// Stores the `index` value.
52    pub index: Vec<usize>,
53    /// Stores the `sample_point` value.
54    pub sample_point: RealAlgebraicPoint,
55    /// Stores the `kind` value.
56    pub kind: CellKind,
57    /// Stores the `parent` value.
58    pub parent: Option<usize>,
59    /// Stores the `sign_vector` value.
60    pub sign_vector: Vec<i32>,
61}

64/// Public native data structure named `cylindrical_algebraic_decomposition`.
65pub struct CylindricalAlgebraicDecomposition {
66    // The CAD output (Decision 6): the variable order, the input polynomials (level-n
67    // projection set), the projection tower (entry k-1 → the level-k projection set),
68    // and the cell arena with the cylindrical tree carried by each cell's `parent`
69    // link. The read-only provenance fields below (Decision 8) add: `projection_operator_used`
70    // names the operator the decomposition was built with ("brown", "lazard", or
71    // "equational_constraint"); `contract` records whether it is a full sign-invariant
72    // CAD or an equational-constraint CAD with the restricted on-variety guarantee; and
73    // `constraint_count` is the number of leading input polynomials that are the
74    // designated constraints (0 for the full path), so a consumer can read which full
75    // cells lie on the constraint variety.
76    /// Stores the `variables` value.
77    pub variables: Vec<SymbolIdentifier>,
78    /// Stores the `input_polynomials` value.
79    pub input_polynomials: Vec<Polynomial>,
80    /// Stores the `projection_tower` value.
81    pub projection_tower: Vec<Vec<Polynomial>>,
82    /// Stores the `cells` value.
83    pub cells: Vec<CadCell>,
84    /// Stores the `projection_operator_used` value.
85    pub projection_operator_used: String,
86    /// Stores the `contract` value.
87    pub contract: CadContract,
88    /// Stores the `constraint_count` value.
89    pub constraint_count: usize,
90}

92impl CylindricalAlgebraicDecomposition {
93    /// Provides the `full_dimensional_cells` operation on this native type.
94    pub fn full_dimensional_cells(&self) -> Vec<usize> {
95        // The arena indices of the full-dimensional (level-n) cells — the cells that
96        // actually partition ℝⁿ. The lower-level cells are the projection structure
97        // the lifting rose through. The real-quantifier-elimination solution-formula
98        // construction reads these.
99        let number_of_variables = self.variables.len();
100        (0..self.cells.len())
101            .filter(|&index| self.cells[index].level == number_of_variables)
102            .collect()
103    }
104
105    /// Provides the `cell_dimension` operation on this native type.
106    pub fn cell_dimension(&self, cell_index: usize) -> usize {
107        // The geometric dimension of a cell as a subset of ℝⁿ: the number of
108        // coordinates along its cylindrical stack parentage that are sectors (open
109        // intervals), each contributing one degree of freedom; sections contribute
110        // none. Walks the parent chain from the cell to the base.
111        let mut dimension = 0;
112        let mut current = Some(cell_index);
113        while let Some(index) = current {
114            if matches!(self.cells[index].kind, CellKind::Sector) {
115                dimension += 1;
116            }
117            current = self.cells[index].parent;
118        }
119        dimension
120    }
121
122    /// Provides the `on_variety_cells` operation on this native type.
123    pub fn on_variety_cells(&self) -> Vec<usize> {
124        // The arena indices of the full-dimensional cells that lie on the constraint
125        // variety V(constraints) — every designated constraint signs to zero there
126        // (Decision 4). These are the cells the EC contract guarantees are
127        // sign-invariant for the non-constraint polynomials; off-variety cells carry no
128        // guarantee. Empty for a full sign-invariant CAD (no constraints declared).
129        let number_of_variables = self.variables.len();
130        (0..self.cells.len())
131            .filter(|&index| self.cells[index].level == number_of_variables)
132            .filter(|&index| {
133                (0..self.constraint_count)
134                    .all(|constraint| self.cells[index].sign_vector[constraint] == 0)
135            })
136            .collect()
137    }
138}

140/// Performs the advanced native `sample_point_as_floats` operation.
141pub fn sample_point_as_floats(point: &RealAlgebraicPoint) -> Vec<f64> {
142    // A numeric *view* of a sample point's coordinates as floats, for the binding
143    // accessors and the test oracle — never a decision. Each coordinate
144    // Σ cᵢ γ^i is evaluated at a high-precision rational approximation of the
145    // embedding generator γ; the trivial (rational) field contributes its constant
146    // coordinate directly because its generator value is zero.
147    let generator_value = real_algebraic_number_to_float(&point.embedding.generator_value);
148    point
149        .coordinates
150        .iter()
151        .map(|element| {
152            let mut value = 0.0_f64;
153            let mut power = 1.0_f64;
154            for coefficient in element.representative.iter() {
155                value += rational_to_float(coefficient) * power;
156                power *= generator_value;
157            }
158            value
159        })
160        .collect()
161}
162
163fn real_algebraic_number_to_float(value: &RealAlgebraicNumber) -> f64 {
164    // The midpoint of a tightly-refined isolating interval, as a float.
165    let (lower, upper) = evaluate_to_interval(value, 100);
166    rational_to_float(&(&(lower + upper) / exact_rational::from(2)))
167}

169fn rational_to_float(value: &exact_rational) -> f64 {
170    nearest_f64_from_rational(value)
171}

173/// Performs the advanced native `cylindrical_algebraic_decomposition` operation.
174pub fn cylindrical_algebraic_decomposition(
175    context: &Context,
176    problem: &CadProblem,
177) -> Option<CylindricalAlgebraicDecomposition> {
178    // The driver (Decisions 1, 2, 4, 5, 6, 8) — the single operator dispatch swap over the
179    // unchanged base/lift/cell-assembly. With an empty equational-constraint slot it
180    // routes through the full-sign-invariant fast path (Brown, guarded by
181    // is_well_oriented with an unconditional Lazard fallback); with a non-empty slot it
182    // routes through the equational-constraint path (Gröbner-preconditioned reduced
183    // projection, the restricted on-variety contract). Both are thin loops over the
184    // same lift so the real-quantifier-elimination partial CAD can replace the loop
185    // with a pruned traversal. None
186    // outside ℚ[x₁,…,xₙ] (Decision 7) or when the real-algebraic-number layer's exact
187    // arithmetic cannot co-reside a
188    // lifted sample point's coordinates (an honest decline, never a wrong cell).
189    let variables = &problem.variables;
190    let number_of_variables = variables.len();
191    if number_of_variables == 0 {
192        return None;
193    }
194    if problem.equational_constraints.is_empty() {
195        full_sign_invariant_decomposition(context, problem)
196    } else {
197        equational_constraint_decomposition(context, problem)
198    }
199}

201fn full_sign_invariant_decomposition(
202    context: &Context,
203    problem: &CadProblem,
204) -> Option<CylindricalAlgebraicDecomposition> {
205    // The transparent fast default (Decisions 2, 3): Brown when well-oriented, the
206    // complete Lazard core otherwise. The Brown decomposition is built and its
207    // well-orientedness verified by re-reading every projection factor at every
208    // positive-dimensional cell's sample point; on any nullification the whole problem
209    // is recomputed with `lazard_projection`, so a Brown CAD is never emitted unverified.
210    let variables = &problem.variables;
211    let number_of_variables = variables.len();
212    let reversed_order: Vec<SymbolIdentifier> = variables.iter().rev().cloned().collect();
213    let mut input_polynomials: Vec<Polynomial> = Vec::with_capacity(problem.polynomials.len());
214    for expression in problem.polynomials.iter() {
215        input_polynomials.push(expression_to_polynomial(expression, &reversed_order)?);
216    }
217    let (projection_tower, cells) =
218        build_tower_and_cells(context, &BrownProjection, &input_polynomials, variables)?;
219    let (projection_tower, mut cells, operator_used) =
220        if is_well_oriented(context, &projection_tower, &cells) {
221            (projection_tower, cells, "brown")
222        } else {
223            let (lazard_tower, lazard_cells) =
224                build_tower_and_cells(context, &LazardProjection, &input_polynomials, variables)?;
225            (lazard_tower, lazard_cells, "lazard")
226        };
227    fill_sign_vectors(
228        context,
229        &mut cells,
230        &problem.polynomials,
231        number_of_variables,
232    )?;
233    Some(CylindricalAlgebraicDecomposition {
234        variables: variables.clone(),
235        input_polynomials,
236        projection_tower,
237        cells,
238        projection_operator_used: operator_used.to_string(),
239        contract: CadContract::FullSignInvariant,
240        constraint_count: 0,
241    })
242}

244fn equational_constraint_decomposition(
245    context: &Context,
246    problem: &CadProblem,
247) -> Option<CylindricalAlgebraicDecomposition> {
248    // The equational-constraint path (Decision 4). Gröbner-precondition the constraint
249    // set, build the EC input list (constraints first, then reduced non-constraints),
250    // and decompose with the reduced equational-constraint operator. The same
251    // well-orientedness guard protects the EC projection: on a nullification the whole
252    // problem is recomputed with the complete Lazard operator over the same inputs,
253    // which is a full sign-invariant CAD and so trivially honours the on-variety EC
254    // contract. An inconsistent constraint set (empty variety) yields the complete
255    // Lazard CAD of the original polynomials, on which no cell lies on the empty variety.
256    let variables = &problem.variables;
257    let number_of_variables = variables.len();
258    let setup = equational_constraint_setup(context, problem)?;
259    if setup.empty_variety {
260        return empty_variety_decomposition(context, problem);
261    }
262    let input_polynomials = setup.input_polynomials.clone();
263    let (projection_tower, cells) =
264        build_tower_and_cells(context, &setup.operator, &input_polynomials, variables)?;
265    let (projection_tower, mut cells, operator_used) =
266        if is_well_oriented(context, &projection_tower, &cells) {
267            (projection_tower, cells, "equational_constraint")
268        } else {
269            let (lazard_tower, lazard_cells) =
270                build_tower_and_cells(context, &LazardProjection, &input_polynomials, variables)?;
271            (lazard_tower, lazard_cells, "lazard")
272        };
273    let sign_expressions: Vec<ExpressionReference> = input_polynomials
274        .iter()
275        .map(|value| polynomial_to_expression(context, value))
276        .collect();
277    fill_sign_vectors(context, &mut cells, &sign_expressions, number_of_variables)?;
278    Some(CylindricalAlgebraicDecomposition {
279        variables: variables.clone(),
280        input_polynomials,
281        projection_tower,
282        cells,
283        projection_operator_used: operator_used.to_string(),
284        contract: CadContract::EquationalConstraintOnVariety,
285        constraint_count: setup.constraint_count,
286    })
287}

289fn empty_variety_decomposition(
290    context: &Context,
291    problem: &CadProblem,
292) -> Option<CylindricalAlgebraicDecomposition> {
293    // The degenerate EC case: the constraint variety is empty (the Gröbner basis is
294    // {1}). The EC-CAD is then vacuously sign-invariant on the empty variety, so the
295    // complete Lazard CAD of the original polynomials — carrying no on-variety cell
296    // (no point makes every constraint vanish) — is a correct, honest answer. Labelled
297    // with the EC contract and a zero constraint count so `on_variety_cells` is empty.
298    let variables = &problem.variables;
299    let number_of_variables = variables.len();
300    let reversed_order: Vec<SymbolIdentifier> = variables.iter().rev().cloned().collect();
301    let mut input_polynomials: Vec<Polynomial> = Vec::with_capacity(problem.polynomials.len());
302    for expression in problem.polynomials.iter() {
303        input_polynomials.push(expression_to_polynomial(expression, &reversed_order)?);
304    }
305    let (projection_tower, mut cells) =
306        build_tower_and_cells(context, &LazardProjection, &input_polynomials, variables)?;
307    fill_sign_vectors(
308        context,
309        &mut cells,
310        &problem.polynomials,
311        number_of_variables,
312    )?;
313    Some(CylindricalAlgebraicDecomposition {
314        variables: variables.clone(),
315        input_polynomials,
316        projection_tower,
317        cells,
318        projection_operator_used: "lazard".to_string(),
319        contract: CadContract::EquationalConstraintOnVariety,
320        constraint_count: 0,
321    })
322}

324/// Performs the advanced native `decompose_with_operator` operation.
325pub fn decompose_with_operator(
326    context: &Context,
327    problem: &CadProblem,
328    operator: &dyn ProjectionOperator,
329    operator_label: &str,
330) -> Option<CylindricalAlgebraicDecomposition> {
331    // Build the full sign-invariant CAD of the problem with a *caller-chosen* operator
332    // and no well-orientedness fallback — the seam the real-quantifier-elimination
333    // partial-CAD driver and the
334    // Brown-vs-Lazard cross-check use to drive this module's base/lift with one specific
335    // operator (Decision 5). The default `cylindrical_algebraic_decomposition` entry
336    // point routes through the selection policy instead; this is the explicit-operator
337    // form. Signs every input polynomial at every full cell; the contract is the full
338    // sign-invariant one (the caller is responsible for the operator's validity, e.g.
339    // by passing `lazard_projection` or a Brown operator it has guarded).
340    let variables = &problem.variables;
341    let number_of_variables = variables.len();
342    if number_of_variables == 0 {
343        return None;
344    }
345    let reversed_order: Vec<SymbolIdentifier> = variables.iter().rev().cloned().collect();
346    let mut input_polynomials: Vec<Polynomial> = Vec::with_capacity(problem.polynomials.len());
347    for expression in problem.polynomials.iter() {
348        input_polynomials.push(expression_to_polynomial(expression, &reversed_order)?);
349    }
350    let (projection_tower, mut cells) =
351        build_tower_and_cells(context, operator, &input_polynomials, variables)?;
352    fill_sign_vectors(
353        context,
354        &mut cells,
355        &problem.polynomials,
356        number_of_variables,
357    )?;
358    Some(CylindricalAlgebraicDecomposition {
359        variables: variables.clone(),
360        input_polynomials,
361        projection_tower,
362        cells,
363        projection_operator_used: operator_label.to_string(),
364        contract: CadContract::FullSignInvariant,
365        constraint_count: 0,
366    })
367}

369fn build_tower_and_cells(
370    context: &Context,
371    operator: &dyn ProjectionOperator,
372    input_polynomials: &[Polynomial],
373    variables: &[SymbolIdentifier],
374) -> Option<(Vec<Vec<Polynomial>>, Vec<CadCell>)> {
375    // The tower-then-base-then-lift loop, factored out so the full and
376    // equational-constraint
377    // paths drive the identical base/lift/cell-assembly with whichever operator the
378    // policy selected. One cylindrical level is completed before the next begins: the
379    // parents of the level are fixed in canonical cell order before any of them is
380    // lifted, each parent task returns its complete ordered stack, and the coordinator
381    // concatenates the stacks by parent and then by child, so the arena the next level
382    // reads is the one the parent order names. None on a lift decline (the
383    // real-algebraic-number layer cannot co-reside a sample point) — a parent that
384    // declines makes the whole decomposition decline and is never read as a parent
385    // with no children.
386    let projection_tower = project_tower(context, operator, input_polynomials, variables);
387    let mut cells: Vec<CadCell> = base_phase(context, &projection_tower[0], variables[0])?;
388    for level in 1..variables.len() {
389        let new_variable = variables[level];
390        let level_polynomials: &[Polynomial] = &projection_tower[level];
391        let parents: Vec<(usize, CadCell)> = (0..cells.len())
392            .filter(|&index| cells[index].level == level)
393            .map(|index| (index, cells[index].clone()))
394            .collect();
395        let route = lifting_route(context, level_polynomials, parents.len(), level);
396        let cancellation = context.operation().cancellation_token();
397        let stacks = ordered_try_map(
398            route,
399            &parents,
400            &cancellation,
401            |(base_index, base_cell)| -> TaskOutcome<Vec<CadCell>, ()> {
402                match try_lift_over_cell(
403                    context,
404                    level_polynomials,
405                    new_variable,
406                    base_cell,
407                    *base_index,
408                ) {
409                    Some(stack) => TaskOutcome::Value(stack),
410                    None => TaskOutcome::Declined,
411                }
412            },
413        );
414        for stack in stacks.into_iter() {
415            match stack {
416                TaskOutcome::Value(stack) => cells.extend(stack),
417                _ => return None,
418            }
419        }
420    }
421    Some((projection_tower, cells))
422}

424fn lifting_route(
425    context: &Context,
426    level_polynomials: &[Polynomial],
427    parent_count: usize,
428    level: usize,
429) -> ExecutionRoute {
430    // One cylindrical level's execution decision. A parent's stack costs the Lazard
431    // evaluation of every level factor at its sample point — proportional to the
432    // factor's term count and to the number of coordinates already fixed — followed by
433    // the isolation of the resulting fiber's roots over the sample point's field. The
434    // level's work is that per-parent cost times the number of parents. Task-local
435    // memory is charged at the fiber and sample-point extension one parent holds.
436    //
437    // A forced-sequential operation discards every estimate at the first branch of the execution
438    // decision, so the cost proxy below — which reads the level, the tower, or the record list it
439    // is estimating — is only formed when the answer can still be the parallel route.
440    if !context.operation().may_fork_work_class("cad_lifting") {
441        return ExecutionRoute::Serial;
442    }
443    let per_parent_work_units = level_polynomials
444        .iter()
445        .map(|polynomial| {
446            (polynomial.terms.len() as u64)
447                .saturating_mul(degree(polynomial).unwrap_or(0) as u64 + 1)
448        })
449        .fold(0_u64, |total, units| total.saturating_add(units))
450        .saturating_mul(level as u64);
451    let domain_work_units = per_parent_work_units.saturating_mul(parent_count as u64);
452    let predicted_task_local_bytes = level_polynomials
453        .len()
454        .saturating_add(1)
455        .saturating_mul(512)
456        .saturating_mul(level.max(1));
457    let worker_count = crate::parallelism::maximum_operation_concurrency();
458    let requested_permits = parent_count.min(worker_count.max(1));
459    context
460        .operation()
461        .execution_decision(
462            &WorkEstimate::new(
463                domain_work_units,
464                parent_count,
465                predicted_task_local_bytes,
466                0,
467                requested_permits,
468                "cad_lifting",
469            ),
470            requested_permits,
471            crate::parallelism::runtime_memory_capacity(predicted_task_local_bytes),
472        )
473        .route
474}

476fn fill_sign_vectors(
477    context: &Context,
478    cells: &mut [CadCell],
479    sign_expressions: &[ExpressionReference],
480    number_of_variables: usize,
481) -> Option<()> {
482    // Fill each full-dimensional cell's sign vector by signing every input polynomial
483    // at its sample point through the real-algebraic-number layer. Each full cell's
484    // vector is decided from its own immutable sample point, so the cells are signed as
485    // one batch in cylindrical order and written back in that order. None when a sign is
486    // undetermined (a sample point the real-algebraic-number layer cannot resolve),
487    // the honest decline — a cell whose sign is undetermined declines the whole
488    // decomposition however many other cells resolved.
489    let full_cells: Vec<usize> = (0..cells.len())
490        .filter(|&index| cells[index].level == number_of_variables)
491        .collect();
492    let route = sign_evaluation_route(context, full_cells.len(), sign_expressions.len());
493    let cancellation = context.operation().cancellation_token();
494    let sign_vectors = ordered_try_map(
495        route,
496        &full_cells,
497        &cancellation,
498        |&index| -> TaskOutcome<Vec<i32>, ()> {
499            let mut sign_vector: Vec<i32> = Vec::with_capacity(sign_expressions.len());
500            for expression in sign_expressions.iter() {
501                match sign_of_polynomial_at_real_point(expression, &cells[index].sample_point) {
502                    Some(sign) => sign_vector.push(sign),
503                    None => return TaskOutcome::Declined,
504                }
505            }
506            TaskOutcome::Value(sign_vector)
507        },
508    );
509    let mut resolved: Vec<Vec<i32>> = Vec::with_capacity(full_cells.len());
510    for outcome in sign_vectors.into_iter() {
511        match outcome {
512            TaskOutcome::Value(sign_vector) => resolved.push(sign_vector),
513            _ => return None,
514        }
515    }
516    for (&index, sign_vector) in full_cells.iter().zip(resolved.into_iter()) {
517        cells[index].sign_vector = sign_vector;
518    }
519    Some(())
520}

522fn sign_evaluation_route(
523    context: &Context,
524    full_cell_count: usize,
525    expression_count: usize,
526) -> ExecutionRoute {
527    // The decision for signing the full-dimensional cells. Each cell costs one exact
528    // sign decision per input polynomial at an algebraic sample point, so the work is
529    // the cell count times the polynomial count; task-local memory is the one sign
530    // vector a cell builds. Signing is the same per-cell evaluation the lifting
531    // performs and is gated on the same class.
532    //
533    // A forced-sequential operation discards every estimate at the first branch of the execution
534    // decision, so the cost proxy below — which reads the level, the tower, or the record list it
535    // is estimating — is only formed when the answer can still be the parallel route.
536    if !context.operation().may_fork_work_class("cad_lifting") {
537        return ExecutionRoute::Serial;
538    }
539    let domain_work_units = (full_cell_count as u64).saturating_mul(expression_count as u64 * 4);
540    let predicted_task_local_bytes = expression_count.saturating_mul(64).max(256);
541    let worker_count = crate::parallelism::maximum_operation_concurrency();
542    let requested_permits = full_cell_count.min(worker_count.max(1));
543    context
544        .operation()
545        .execution_decision(
546            &WorkEstimate::new(
547                domain_work_units,
548                full_cell_count,
549                predicted_task_local_bytes,
550                0,
551                requested_permits,
552                "cad_lifting",
553            ),
554            requested_permits,
