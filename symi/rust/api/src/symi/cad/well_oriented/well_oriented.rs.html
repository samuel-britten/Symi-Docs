8/// Performs the advanced native `is_well_oriented` operation.
9pub fn is_well_oriented(
10    context: &Context,
11    projection_tower: &[Vec<Polynomial>],
12    cells: &[CadCell],
13) -> bool {
14    // The Brown soundness predicate (Decision 3). A Brown decomposition is provably
15    // sign-invariant exactly when the input is *well-oriented*: no projection factor
16    // nullifies (vanishes identically in the lifted variable) at the sample point of a
17    // positive-dimensional cell of the CAD of the lower-dimensional space. This
18    // re-reads each level's projection factors at each such cell's sample point via
19    // the plain evaluation `evaluate_factor_at_sample` (the same flatten/substitute
20    // path `lift_over_cell` uses, minus the Lazard valuation that would repair a
21    // nullification) and reports a failure on any nullification.
22    //
23    // The factors lifted to introduce xₘ₊₁ over a level-m cell are
24    // projection_tower[m] (main variable xₘ₊₁); a cell is positive-dimensional when
25    // its parent chain holds at least one sector. Returns false (→ Lazard fallback)
26    // both on a detected nullification and, conservatively, on an evaluation that
27    // cannot be performed, so the policy is never unsound.
28    let number_of_variables = projection_tower.len();
29    let candidates: Vec<usize> = (0..cells.len())
30        .filter(|&index| {
31            let level = cells[index].level;
32            level != 0 && level < number_of_variables
33        })
34        .filter(|&index| cell_geometric_dimension(cells, index) != 0)
35        .collect();
36    let route = well_orientedness_route(context, projection_tower, candidates.len());
37    if route != ExecutionRoute::Parallel {
38        return candidates
39            .iter()
40            .all(|&index| cell_is_well_oriented(projection_tower, &cells[index]));
41    }
42    ordered_map(route, &candidates, |&index| {
43        cell_is_well_oriented(projection_tower, &cells[index])
44    })
45    .into_iter()
46    .all(|verdict| verdict)
47}

49fn cell_is_well_oriented(projection_tower: &[Vec<Polynomial>], cell: &CadCell) -> bool {
50    // One cell's share of the predicate: every factor lifted over this cell's level is
51    // read at its sample point, and the cell fails on a nullification or on an
52    // evaluation that cannot be performed. The cell owns its sample point and the
53    // level's factors are immutable, so the verdicts are independent and the
54    // conjunction over them does not depend on the order they are decided in.
55    projection_tower[cell.level].iter().all(|factor| {
56        matches!(
57            evaluate_factor_at_sample(factor, factor.main_variable, &cell.sample_point),
58            Some(fiber) if !fiber.is_empty()
59        )
60    })
61}

63fn well_orientedness_route(
64    context: &Context,
65    projection_tower: &[Vec<Polynomial>],
66    candidate_count: usize,
67) -> ExecutionRoute {
68    // The guard's execution decision. A candidate cell costs one plain evaluation per
69    // factor of its level, so the work is the candidate count times the largest level's
70    // factor work; task-local memory is the one fiber an evaluation builds. The guard
71    // reads only immutable level data and is charged on the projection class it
72    // verifies.
73    //
74    // A forced-sequential operation discards every estimate at the first branch of the execution
75    // decision, so the cost proxy below — which reads the level, the tower, or the record list it
76    // is estimating — is only formed when the answer can still be the parallel route.
77    if !context.operation().may_fork_work_class("cad_projection") {
78        return ExecutionRoute::Serial;
79    }
80    let level_work_units = projection_tower
81        .iter()
82        .map(|level| {
83            level
84                .iter()
85                .map(|polynomial| polynomial.terms.len() as u64)
86                .fold(0_u64, |total, units| total.saturating_add(units))
87        })
88        .max()
89        .unwrap_or(0);
90    let domain_work_units = level_work_units
91        .saturating_mul(candidate_count as u64)
92        .saturating_mul(8);
93    let predicted_task_local_bytes = (level_work_units as usize).saturating_mul(64).max(256);
94    let worker_count = crate::parallelism::maximum_operation_concurrency();
95    let requested_permits = candidate_count.min(worker_count.max(1));
96    context
97        .operation()
98        .execution_decision(
99            &WorkEstimate::new(
100                domain_work_units,
101                candidate_count,
102                predicted_task_local_bytes,
103                0,
104                requested_permits,
105                "cad_projection",
106            ),
107            requested_permits,
108            crate::parallelism::runtime_memory_capacity(predicted_task_local_bytes),
109        )
110        .route
111}

113fn cell_geometric_dimension(cells: &[CadCell], index: usize) -> usize {
114    // The geometric dimension of a cell as a subset of ℝⁿ: the number of sector
115    // coordinates along its cylindrical-stack parentage (sections contribute none).
116    // Mirrors `cylindrical_algebraic_decomposition::cell_dimension` but over a raw
117    // cell slice so the guard reads the in-progress arena before the decomposition
118    // value is assembled.
119    let mut dimension = 0;
120    let mut current = Some(index);
121    while let Some(arena_index) = current {
122        if matches!(cells[arena_index].kind, CellKind::Sector) {
123            dimension += 1;
124        }
125        current = cells[arena_index].parent;
126    }
127    dimension
128}
