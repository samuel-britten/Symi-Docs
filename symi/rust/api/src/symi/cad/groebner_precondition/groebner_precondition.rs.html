14/// Public native data structure named `preconditioned_constraints`.
15pub struct PreconditionedConstraints {
16    // The Gröbner-preconditioned equational-constraint set (Decision 4), recursive
17    // polynomials in the CAD's reversed variable order (main variable xₙ), ready to
18    // drive the equational-constraint projection.
19    //
20    // `constraint_polynomials` is the reduced Gröbner basis of the constraint ideal —
21    // the canonical, propagation-friendly generating set, possibly a single polynomial
22    // for one constraint or a smaller basis when several share a factor.
23    // `other_polynomials` is each non-constraint reduced modulo that basis (its normal
24    // form), which equals the original polynomial on the variety V(constraints) and so
25    // preserves every on-variety sign while shrinking the input. `empty_variety` is set
26    // when the basis is {1}: the constraints are inconsistent, V(constraints) = ∅, and
27    // the equational-constraint CAD is trivial.
28    /// Stores the `constraint_polynomials` value.
29    pub constraint_polynomials: Vec<Polynomial>,
30    /// Stores the `other_polynomials` value.
31    pub other_polynomials: Vec<Polynomial>,
32    /// Stores the `empty_variety` value.
33    pub empty_variety: bool,
34}

36/// Performs the advanced native `precondition_constraints` operation.
37pub fn precondition_constraints(
38    context: &Context,
39    polynomials: &[ExpressionReference],
40    constraints: &[usize],
41    variables: &[SymbolIdentifier],
42) -> Option<PreconditionedConstraints> {
43    // Reduce the raw equational constraints to a reduced Gröbner basis under the CAD
44    // elimination order (`groebner_basis`, transparently F4-accelerated),
45    // and reduce every non-constraint polynomial modulo that basis
46    // (`multivariate_normal_form`). The CAD eliminates xₙ first, so the ring's variable
47    // order is the reversed list [xₙ,…,x₁] and the elimination order ranks the
48    // higher-projected variables above the lower ones — a propagation-friendly basis.
49    //
50    // None when any constraint is non-ℚ-polynomial (inheriting `groebner_basis`'s clean
51    // decline). `empty_variety` is set when the basis is {1} (the constraints are
52    // inconsistent). The returned polynomials carry the reversed CAD order (main
53    // variable xₙ) so they slot directly into the projection tower. Takes the problem's
54    // expression polynomials (not recursive `polynomial`s) because that is the form the
55    // Gröbner surface consumes; the recursive conversion happens here once.
56    let constraint_set: std::collections::BTreeSet<usize> = constraints.iter().copied().collect();
57    let constraint_expressions: Vec<ExpressionReference> = constraints
58        .iter()
59        .map(|&index| polynomials[index].clone())
60        .collect();
61    let reversed_order: Vec<SymbolIdentifier> = variables.iter().rev().cloned().collect();
62
63    let ring = PolynomialRing::new(reversed_order.clone(), elimination_order(0));
64    let basis_expressions = groebner_basis(context, &ring, &constraint_expressions)?;
65
66    if basis_is_unit(&basis_expressions, &reversed_order) {
67        return Some(PreconditionedConstraints {
68            constraint_polynomials: Vec::new(),
69            other_polynomials: Vec::new(),
70            empty_variety: true,
71        });
72    }
73
74    let mut constraint_polynomials: Vec<Polynomial> = Vec::with_capacity(basis_expressions.len());
75    for expression in basis_expressions.iter() {
76        constraint_polynomials.push(expression_to_polynomial(expression, &reversed_order)?);
77    }
78
79    let basis_distributed: Vec<_> = basis_expressions
80        .iter()
81        .map(|expression| expression_to_distributed_polynomial(expression, &reversed_order))
82        .collect::<Option<Vec<_>>>()?;
83
84    let mut other_polynomials: Vec<Polynomial> = Vec::new();
85    for (index, expression) in polynomials.iter().enumerate() {
86        if constraint_set.contains(&index) {
87            continue;
88        }
89        let distributed = expression_to_distributed_polynomial(expression, &reversed_order)?;
90        let reduced = multivariate_normal_form(&distributed, &basis_distributed, ring.order);
91        let reduced_expression =
92            distributed_polynomial_to_expression(context, &reduced, &reversed_order);
93        let reduced_polynomial = expression_to_polynomial(&reduced_expression, &reversed_order)?;
94        other_polynomials.push(reduced_polynomial);
95    }
96
97    Some(PreconditionedConstraints {
98        constraint_polynomials,
99        other_polynomials,
100        empty_variety: false,
101    })
102}

104fn basis_is_unit(basis: &[ExpressionReference], variables: &[SymbolIdentifier]) -> bool {
105    // A reduced Gröbner basis equal to {1} (a nonzero constant) certifies an empty
106    // variety: the constraint ideal is the whole ring. Detected structurally over the
107    // distributed form — the basis is a single nonzero polynomial whose only term is
108    // the constant monomial (every exponent zero), so no projection variable survives.
109    if basis.len() != 1 {
110        return false;
111    }
112    match expression_to_distributed_polynomial(&basis[0], variables) {
113        Some(value) => {
114            !distributed::is_zero(&value)
115                && value.terms.len() == 1
116                && value.terms[0]
117                    .0
118                    .exponents
119                    .iter()
120                    .all(|exponent| *exponent == 0)
121        }
122        None => false,
123    }
124}
