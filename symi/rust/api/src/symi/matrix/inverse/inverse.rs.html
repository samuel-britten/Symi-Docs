11fn nonzero_entry_count(matrix_value: &Matrix) -> usize {
12    // How many entries of the matrix are not the literal zero. This is a structural read, not an
13    // assumption query: it is a cost estimate, and an entry the estimate counts that later proves
14    // to be zero costs the batch nothing but a slightly generous prediction.
15    matrix_value
16        .entries
17        .iter()
18        .filter(|entry| {
19            !matches!(&entry.kind, crate::expression::ExpressionKind::Integer(value)
20                if *value == crate::exact_arithmetic::Integer::ZERO)
21        })
22        .count()
23}

25/// Performs the advanced native `cofactor_matrix` operation.
26pub fn cofactor_matrix(context: &Context, matrix_value: &Matrix) -> MatrixResult<Matrix> {
27    // Build the cofactor matrix of a square input. The (row, column)
28    // cofactor entry is (-1)^(row + column) times the determinant of
29    // the (n - 1) x (n - 1) minor obtained by striking out the
30    // matching row and column from the input. Minors are dispatched
31    // back through Berkowitz so the entire cofactor pass stays
32    // division-free.
33    if !matrix_value.is_square() {
34        return Err(MatrixError::InverseRequiresSquare {
35            rows: matrix_value.rows,
36            columns: matrix_value.columns,
37        });
38    }
39    let dimension = matrix_value.rows;
40    // A cofactor task is one minor determinant, and what that determinant costs is set by how
41    // many entries the minor actually holds rather than by the matrix's shape: the minors of a
42    // diagonal matrix collapse immediately, and handing them out costs more than taking them.
43    // The estimate is therefore the entries the batch reads — one row of minors per row of the
44    // matrix, each reading the nonzero entries outside its own row and column.
45    let route = cofactor_route(context, matrix_value, dimension);
46    let row_indices = (0..dimension).collect::<Vec<_>>();
47    let rows = ordered_map(route, &row_indices, |row_index| {
48        (0..dimension)
49            .map(|column_index| {
50                let minor_matrix = minor(matrix_value, *row_index, column_index);
51                let minor_determinant = determinant(context, &minor_matrix);
52                let sign_is_negative = (*row_index + column_index) % 2 == 1;
53                minor_determinant.map(|value| {
54                    if sign_is_negative {
55                        builders::negate(context, value)
56                    } else {
57                        value
58                    }
59                })
60            })
61            .collect::<Vec<_>>()
62    });
63    let mut entries: Vec<ExpressionReference> = Vec::with_capacity(dimension * dimension);
64    for row in rows {
65        for entry in row {
66            entries.push(entry?);
67        }
68    }
69    Ok(Matrix {
70        rows: dimension,
71        columns: dimension,
72        entries,
73    })
74}

76/// Performs the advanced native `adjugate` operation.
77pub fn adjugate(context: &Context, matrix_value: &Matrix) -> MatrixResult<Matrix> {
78    // The adjugate (classical adjoint) is the transpose of the
79    // cofactor matrix. Used directly by the inverse pipeline below.
80    let cofactor = cofactor_matrix(context, matrix_value)?;
81    let route = adjugate_transpose_route(context, &cofactor);
82    Ok(crate::matrix::arithmetic::transpose_with_route(
83        route, &cofactor,
84    ))
85}

87/// Choose the route for one cofactor pass.
88///
89/// Counting the nonzero entries is a pass over the whole matrix, so a class the shipping policy
90/// cannot admit is answered from its disposition first and the matrix is read only by the
91/// cofactors themselves.
92fn cofactor_route(context: &Context, matrix_value: &Matrix, dimension: usize) -> ExecutionRoute {
93    if !context.operation().may_fork_work_class("matrix_cofactor") {
94        return ExecutionRoute::Serial;
95    }
96    let nonzero_entry_count = nonzero_entry_count(matrix_value);
97    let worker_count = crate::parallelism::maximum_operation_concurrency();
98    let requested_permits = dimension.min(worker_count.max(1));
99    context
100        .operation()
101        .execution_decision(
102            &WorkEstimate::new(
103                (dimension as u64)
104                    .saturating_mul(nonzero_entry_count as u64)
105                    .saturating_mul(dimension.saturating_sub(1) as u64),
106                dimension,
107                dimension.saturating_mul(512),
108                0,
109                requested_permits,
110                "matrix_cofactor",
111            ),
112            requested_permits,
113            crate::parallelism::runtime_memory_capacity(dimension.saturating_mul(512)),
114        )
115        .route
116}

118/// Choose the route for the transpose that turns a cofactor matrix into the adjugate.
119fn adjugate_transpose_route(context: &Context, cofactor: &Matrix) -> ExecutionRoute {
120    if !context.operation().may_fork_work_class("matrix_arithmetic") {
121        return ExecutionRoute::Serial;
122    }
123    let worker_count = crate::parallelism::maximum_operation_concurrency();
124    let requested_permits = cofactor.rows.min(worker_count.max(1));
125    context
126        .operation()
127        .execution_decision(
128            &WorkEstimate::new(
129                cofactor.entries.len() as u64,
130                cofactor.rows,
131                cofactor.columns.saturating_mul(128),
132                0,
133                requested_permits,
134                "matrix_arithmetic",
135            ),
136            requested_permits,
137            crate::parallelism::runtime_memory_capacity(cofactor.columns.saturating_mul(128)),
138        )
139        .route
140}

142/// Performs the advanced native `inverse_via_adjugate` operation.
143pub fn inverse_via_adjugate(context: &Context, matrix_value: &Matrix) -> MatrixResult<Matrix> {
144    // Build the inverse of a square matrix from `adjugate / determinant`.
145    // The determinant is computed first so that a provably zero
146    // determinant short-circuits to `MatrixError::SingularMatrix` and
147    // an undecidable verdict reports `MatrixError::SingularityIndeterminate`
148    // rather than silently producing a divide-by-zero entry. When the
149    // determinant is provably nonzero, every adjugate entry is divided
150    // by it via the simplifying `divide` builder so the entries of the
151    // resulting matrix are canonical.
152    if !matrix_value.is_square() {
153        return Err(MatrixError::InverseRequiresSquare {
154            rows: matrix_value.rows,
155            columns: matrix_value.columns,
156        });
157    }
158    match crate::matrix::ask_matrix_predicate(context, matrix_value, MatrixPredicate::Invertible) {
159        TruthValue::No => return Err(MatrixError::SingularMatrix),
160        TruthValue::Unknown => {
161            return Err(MatrixError::SingularityIndeterminate);
162        }
163        TruthValue::Yes => {}
164    }
165    let det = determinant(context, matrix_value)?;
166    let adjugate_matrix = adjugate(context, matrix_value)?;
167    let mut entries: Vec<ExpressionReference> = Vec::with_capacity(adjugate_matrix.entries.len());
168    for entry in adjugate_matrix.entries.iter() {
169        let scaled = builders::divide(context, entry.clone(), det.clone());
170        entries.push(scaled);
171    }
172    Ok(Matrix {
173        rows: adjugate_matrix.rows,
174        columns: adjugate_matrix.columns,
175        entries,
176    })
177}

179/// Performs the advanced native `inverse_with_certified_determinant` operation.
180pub fn inverse_with_certified_determinant(
181    context: &Context,
182    matrix_value: &Matrix,
183    certified_determinant: &ExpressionReference,
184) -> MatrixResult<Matrix> {
185    // The inverse of a matrix whose determinant the caller has already reduced to a form the
186    // assumptions prove nonzero. `adjugate(M) M = det(M) I` is a polynomial identity in the
187    // entries and holds over any commutative ring, so a nonzero determinant is the whole of what
188    // inversion needs; the invertibility predicate additionally asks whether the entries are
189    // finite complex numbers, which a free symbol can never establish and which the identity does
190    // not require. The supplied determinant is not taken on trust: it is proved nonzero here, and
191    // it is proved to be the determinant of this matrix, so a caller cannot smuggle in an
192    // invertibility it does not have.
193    if !matrix_value.is_square() {
194        return Err(MatrixError::InverseRequiresSquare {
195            rows: matrix_value.rows,
196            columns: matrix_value.columns,
197        });
198    }
199    if is_nonzero(certified_determinant, &context.symbol_table) != TruthValue::Yes {
200        return Err(MatrixError::SingularityIndeterminate);
201    }
202    let computed = determinant(context, matrix_value)?;
203    let difference = crate::simplify::simplify(
204        context,
205        builders::subtract(context, computed, certified_determinant.clone()),
206    );
207    if crate::zero_equivalence::zero_equivalent(context, &difference, None)
208        != crate::zero_equivalence::ZeroEquivalenceVerdict::TrueValue
209    {
210        return Err(MatrixError::SingularityIndeterminate);
211    }
212    let adjugate_matrix = adjugate(context, matrix_value)?;
213    Ok(Matrix {
214        rows: adjugate_matrix.rows,
215        columns: adjugate_matrix.columns,
216        entries: adjugate_matrix
217            .entries
218            .iter()
219            .map(|entry| builders::divide(context, entry.clone(), certified_determinant.clone()))
220            .collect(),
221    })
222}

224fn minor(matrix_value: &Matrix, strike_row: usize, strike_column: usize) -> Matrix {
225    // Build the (n - 1) x (n - 1) minor by striking out the named row
226    // and column. Pure index arithmetic — no simplification is needed
227    // because the entries themselves are copied verbatim.
228    let dimension = matrix_value.rows;
229    let new_dimension = dimension - 1;
230    let mut entries: Vec<ExpressionReference> = Vec::with_capacity(new_dimension * new_dimension);
231    for row_index in 0..dimension {
232        if row_index == strike_row {
233            continue;
234        }
235        for column_index in 0..dimension {
236            if column_index == strike_column {
237                continue;
238            }
239            entries.push(
240                matrix_value
241                    .entry_unchecked(row_index, column_index)
242                    .clone(),
243            );
244        }
245    }
246    Matrix {
247        rows: new_dimension,
248        columns: new_dimension,
249        entries,
250    }
251}

254/// Public native data structure named `lower_upper_decomposition`.
255pub struct LowerUpperDecomposition {
256    /// Stores the `lower` value.
257    pub lower: Matrix,
258    /// Stores the `upper` value.
259    pub upper: Matrix,
260    /// Stores the `row_permutation` value.
261    pub row_permutation: Vec<usize>,
262}
263
264impl LowerUpperDecomposition {
265    /// Return a replayable certificate for this LU result and its source matrix.
266    pub fn postcondition_evidence(
267        &self,
268        context: &Context,
269        source: &Matrix,
270    ) -> Option<crate::assumptions::matrix_operation_evidence::MatrixOperationEvidence> {
271        let input =
272            crate::assumptions::matrix_subject::MatrixSubject::from_matrix(context, source).ok()?;
273        let lower =
274            crate::assumptions::matrix_subject::MatrixSubject::from_matrix(context, &self.lower)
275                .ok()?;
276        let upper =
277            crate::assumptions::matrix_subject::MatrixSubject::from_matrix(context, &self.upper)
278                .ok()?;
279        Some(
280            crate::assumptions::matrix_operation_evidence::MatrixOperationEvidence::LowerUpper {
281                input,
282                lower,
283                upper,
284                row_permutation: self.row_permutation.clone(),
285            },
286        )
287    }
288}

290/// Performs the advanced native `lower_upper_decomposition_with_partial_pivoting` operation.
291pub fn lower_upper_decomposition_with_partial_pivoting(
292    context: &Context,
293    matrix_value: &Matrix,
294) -> MatrixResult<LowerUpperDecomposition> {
295    // Standard partial-pivoting LU decomposition. Pivot selection
296    // matches the Bareiss path's discipline: prefer the candidate row
297    // with the smallest expression hash among rows whose pivot entry is
298    // provably nonzero, refuse with `pivot_indeterminate` when no
299    // candidate is provably nonzero but at least one is undecidable.
300    // The permutation vector records the row swaps performed during
301    // pivoting so a future numeric backend can replay them.
302    if !matrix_value.is_square() {
303        return Err(MatrixError::InverseRequiresSquare {
304            rows: matrix_value.rows,
305            columns: matrix_value.columns,
306        });
307    }
308    let dimension = matrix_value.rows;
309    let mut upper_entries = matrix_value.entries.clone();
310    let mut lower_entries: Vec<ExpressionReference> =
311        vec![builders::zero(context); dimension * dimension];
312    let one = builders::one(context);
313    for diagonal_index in 0..dimension {
314        lower_entries[diagonal_index * dimension + diagonal_index] = one.clone();
315    }
316    let mut row_permutation: Vec<usize> = (0..dimension).collect();
317    for column_index in 0..dimension {
318        let chosen_row_option =
319            choose_pivot_row_for_lu(context, &upper_entries, dimension, column_index)?;
320        let chosen_row = match chosen_row_option {
321            None => continue,
322            Some(row) => row,
323        };
324        if chosen_row != column_index {
325            for inner_column in 0..dimension {
326                upper_entries.swap(
327                    column_index * dimension + inner_column,
328                    chosen_row * dimension + inner_column,
329                );
330            }
331            for inner_column in 0..column_index {
332                lower_entries.swap(
333                    column_index * dimension + inner_column,
334                    chosen_row * dimension + inner_column,
335                );
336            }
337            row_permutation.swap(column_index, chosen_row);
338        }
339        let pivot_value = upper_entries[column_index * dimension + column_index].clone();
340        for lower_row in (column_index + 1)..dimension {
341            let head_entry = upper_entries[lower_row * dimension + column_index].clone();
342            let multiplier = builders::divide(context, head_entry, pivot_value.clone());
343            lower_entries[lower_row * dimension + column_index] = multiplier.clone();
344            for inner_column in column_index..dimension {
345                let pivot_row_entry =
346                    upper_entries[column_index * dimension + inner_column].clone();
347                let scaled_pivot_entry =
348                    builders::product(context, vec![multiplier.clone(), pivot_row_entry]);
349                let lower_entry = upper_entries[lower_row * dimension + inner_column].clone();
350                let updated = builders::subtract(context, lower_entry, scaled_pivot_entry);
351                upper_entries[lower_row * dimension + inner_column] = updated;
352            }
353        }
354    }
355    let result = LowerUpperDecomposition {
356        lower: Matrix {
357            rows: dimension,
358            columns: dimension,
359            entries: lower_entries,
360        },
361        upper: Matrix {
362            rows: dimension,
363            columns: dimension,
364            entries: upper_entries,
365        },
366        row_permutation,
367    };
368    if !verify_lower_upper_postcondition(context, matrix_value, &result)? {
369        return Err(MatrixError::LowerUpperVerificationFailed);
370    }
371    Ok(result)
372}

374fn verify_lower_upper_postcondition(
375    context: &Context,
376    source: &Matrix,
377    decomposition: &LowerUpperDecomposition,
378) -> MatrixResult<bool> {
379    let permuted_source = Matrix {
380        rows: source.rows,
381        columns: source.columns,
382        entries: decomposition
383            .row_permutation
384            .iter()
385            .flat_map(|row| {
386                (0..source.columns).map(move |column| source.entry_unchecked(*row, column).clone())
387            })
388            .collect(),
389    };
390    let product =
391        crate::matrix::matrix_multiply(context, &decomposition.lower, &decomposition.upper)?;
392    Ok(product.shape() == permuted_source.shape() && product.entries == permuted_source.entries)
393}

395fn choose_pivot_row_for_lu(
396    context: &Context,
397    entries: &[ExpressionReference],
398    dimension: usize,
399    pivot_index: usize,
400) -> MatrixResult<Option<usize>> {
401    // LU-specific pivot selection that mirrors the Bareiss path's
402    // discipline: prefer a provably nonzero candidate with the
403    // smallest expression hash, refuse when no provably nonzero
404    // candidate exists but at least one undecidable candidate does.
405    let mut best: Option<(u64, usize)> = None;
406    let mut saw_unknown = false;
407    for candidate_row in pivot_index..dimension {
408        let entry = &entries[candidate_row * dimension + pivot_index];
409        match is_nonzero(entry, &context.symbol_table) {
410            TruthValue::Yes => {
411                let entry_hash = entry.hash;
412                match best {
413                    None => best = Some((entry_hash, candidate_row)),
414                    Some((existing_hash, existing_row))
415                        if (entry_hash, candidate_row) < (existing_hash, existing_row) =>
416                    {
417                        best = Some((entry_hash, candidate_row));
418                    }
419                    _ => {}
420                }
421            }
422            TruthValue::No => {}
423            TruthValue::Unknown => saw_unknown = true,
424        }
425    }
426    match best {
427        Some((_, chosen)) => Ok(Some(chosen)),
428        None if saw_unknown => Err(MatrixError::PivotIndeterminate {
429            row: pivot_index,
430            column: pivot_index,
431        }),
432        None => Ok(None),
433    }
