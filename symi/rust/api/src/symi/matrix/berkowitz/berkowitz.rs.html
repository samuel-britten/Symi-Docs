6/// Performs the advanced native `determinant` operation.
7pub fn determinant(context: &Context, matrix_value: &Matrix) -> MatrixResult<ExpressionReference> {
8    // Compute the determinant by reading the constant term off the
9    // characteristic-polynomial coefficient vector and adjusting the sign
10    // for the dimension. Berkowitz's recurrence is division-free, runs in
11    // O(n^4) ring operations, and stays rigorous over any commutative
12    // coefficient ring — including the symbolic-expression coefficient
13    // layer that may not form an integral domain. The determinant is
14    // (-1)^n times the constant term of the characteristic polynomial as
15    // produced by this implementation (the polynomial is built as
16    // det(t * I - M), so the constant term carries a sign of (-1)^n
17    // relative to det(M)).
18    if !matrix_value.is_square() {
19        return Err(MatrixError::DeterminantRequiresSquare {
20            rows: matrix_value.rows,
21            columns: matrix_value.columns,
22        });
23    }
24    let coefficients = characteristic_polynomial_coefficients(context, matrix_value)?;
25    let constant_term = coefficients[0].clone();
26    let dimension = matrix_value.rows;
27    if dimension % 2 == 0 {
28        Ok(constant_term)
29    } else {
30        Ok(builders::negate(context, constant_term))
31    }
32}

34/// Performs the advanced native `characteristic_polynomial_coefficients` operation.
35pub fn characteristic_polynomial_coefficients(
36    context: &Context,
37    matrix_value: &Matrix,
38) -> MatrixResult<Vec<ExpressionReference>> {
39    // Run Berkowitz's algorithm and return the coefficient vector of the
40    // monic characteristic polynomial det(t * I - matrix_value) ordered
41    // from the constant term at index zero to the leading coefficient at
42    // index n. The recurrence assembles a sequence of Toeplitz matrices
43    // T_k (one per principal submatrix size k = 1, 2, ..., n) and folds
44    // them by repeated multiplication against the running coefficient
45    // vector. Division-free throughout — every intermediate is built via
46    // the simplifying `sum` and `product` builders so symbolic
47    // coefficient layers compose naturally.
48    if !matrix_value.is_square() {
49        return Err(MatrixError::CharacteristicPolynomialRequiresSquare {
50            rows: matrix_value.rows,
51            columns: matrix_value.columns,
52        });
53    }
54    let dimension = matrix_value.rows;
55    if dimension == 0 {
56        return Err(MatrixError::EmptyMatrix);
57    }
58    let one = builders::one(context);
59    let mut running_vector: Vec<ExpressionReference> = vec![one];
60    for principal_size in 1..=dimension {
61        let toeplitz_column = build_toeplitz_column(context, matrix_value, principal_size);
62        running_vector = multiply_toeplitz(context, &toeplitz_column, &running_vector);
63    }
64    let mut coefficients: Vec<ExpressionReference> = running_vector;
65    coefficients.reverse();
66    Ok(coefficients)
67}

69fn build_toeplitz_column(
70    context: &Context,
71    matrix_value: &Matrix,
72    principal_size: usize,
73) -> Vec<ExpressionReference> {
74    // Construct the (principal_size + 1)-entry generating column of the
75    // Toeplitz matrix T_k associated with the k-th step of Berkowitz's
76    // recurrence (k = principal_size). The principal-leading submatrix
77    // of size principal_size lives in the top-left corner of
78    // matrix_value; let M be that submatrix, let R be the
79    // (principal_size - 1)-long row vector consisting of the entries
80    // M[principal_size - 1, 0 .. principal_size - 1], let C be the
81    // (principal_size - 1)-long column vector M[0 .. principal_size - 1,
82    // principal_size - 1], and let a be the scalar
83    // M[principal_size - 1, principal_size - 1]. Then the generating
84    // column of T_k is (1, -a, -R * C, -R * M_sub * C, ..., -R * M_sub^{k - 2} * C)
85    // where M_sub is the (principal_size - 1) x (principal_size - 1)
86    // submatrix sitting in the top-left of M.
87    let one = builders::one(context);
88    let minus_one = builders::integer_value(context, -1);
89    let leading_index = principal_size - 1;
90    let mut column: Vec<ExpressionReference> = Vec::with_capacity(principal_size + 1);
91    column.push(one);
92    let diagonal_entry = matrix_value
93        .entry_unchecked(leading_index, leading_index)
94        .clone();
95    let negated_diagonal = builders::product(context, vec![minus_one.clone(), diagonal_entry]);
96    column.push(negated_diagonal);
97    if principal_size == 1 {
98        return column;
99    }
100    let inner_size = principal_size - 1;
101    let column_vector: Vec<ExpressionReference> = (0..inner_size)
102        .map(|row_index| {
103            matrix_value
104                .entry_unchecked(row_index, leading_index)
105                .clone()
106        })
107        .collect();
108    let row_vector: Vec<ExpressionReference> = (0..inner_size)
109        .map(|column_index| {
110            matrix_value
111                .entry_unchecked(leading_index, column_index)
112                .clone()
113        })
114        .collect();
115    let inner_matrix_entries: Vec<ExpressionReference> = {
116        let mut buffer = Vec::with_capacity(inner_size * inner_size);
117        for row_index in 0..inner_size {
118            for column_index in 0..inner_size {
119                buffer.push(
120                    matrix_value
121                        .entry_unchecked(row_index, column_index)
122                        .clone(),
123                );
124            }
125        }
126        buffer
127    };
128    let mut current_column = column_vector;
129    for _ in 0..(principal_size - 1) {
130        let inner_product = vector_inner_product(context, &row_vector, &current_column);
131        let negated = builders::product(context, vec![minus_one.clone(), inner_product]);
132        column.push(negated);
133        current_column =
134            matrix_times_vector(context, &inner_matrix_entries, inner_size, &current_column);
135    }
136    column
137}

139fn vector_inner_product(
140    context: &Context,
141    left: &[ExpressionReference],
142    right: &[ExpressionReference],
143) -> ExpressionReference {
144    // Standard inner product. Builds a list of pairwise products and
145    // wraps the result through the simplifying `sum` builder. Both
146    // operands must have equal length; callers in this module ensure
147    // that invariant before calling.
148    assert_eq!(
149        left.len(),
150        right.len(),
151        "vector_inner_product: operand lengths must match",
152    );
153    if left.is_empty() {
154        return builders::zero(context);
155    }
156    let terms: Vec<ExpressionReference> = left
157        .iter()
158        .zip(right.iter())
159        .map(|(left_entry, right_entry)| {
160            builders::product(context, vec![left_entry.clone(), right_entry.clone()])
161        })
162        .collect();
163    builders::sum(context, terms)
164}

166fn matrix_times_vector(
167    context: &Context,
168    matrix_entries: &[ExpressionReference],
169    matrix_size: usize,
170    vector: &[ExpressionReference],
171) -> Vec<ExpressionReference> {
172    // Multiply a square matrix (passed as a flat row-major slice of
173    // expression references plus its side length) against a column
174    // vector. Used inside the Berkowitz Toeplitz-column builder.
175    assert_eq!(
176        vector.len(),
177        matrix_size,
178        "matrix_times_vector: vector length must match matrix size",
179    );
180    let mut result: Vec<ExpressionReference> = Vec::with_capacity(matrix_size);
181    for row_index in 0..matrix_size {
182        let mut term_list: Vec<ExpressionReference> = Vec::with_capacity(matrix_size);
183        for column_index in 0..matrix_size {
184            let entry = matrix_entries[row_index * matrix_size + column_index].clone();
185            let factor = vector[column_index].clone();
186            term_list.push(builders::product(context, vec![entry, factor]));
187        }
188        result.push(builders::sum(context, term_list));
189    }
190    result
191}

193fn multiply_toeplitz(
194    context: &Context,
195    toeplitz_column: &[ExpressionReference],
196    operand: &[ExpressionReference],
197) -> Vec<ExpressionReference> {
198    // Multiply the (k + 1) x k lower-triangular Toeplitz matrix whose
199    // generating column is `toeplitz_column` against the length-k input
200    // vector `operand`. The Berkowitz recurrence applies this
201    // multiplication at each step k = 1, 2, ..., n, growing the running
202    // vector by one entry per step so that after step n the vector has
203    // length n + 1 and contains the characteristic-polynomial
204    // coefficients (in reverse order, from leading to constant).
205    // Result vector has length column_length = k + 1, matching the
206    // number of rows in the Toeplitz matrix; previous closed-form
207    // attempts at deriving the length from operand_length + column_length
208    // produce one spurious trailing entry at every step past k = 1.
209    let column_length = toeplitz_column.len();
210    let operand_length = operand.len();
211    let result_length = column_length;
212    let mut result: Vec<ExpressionReference> = Vec::with_capacity(result_length);
213    for output_index in 0..result_length {
214        let mut term_list: Vec<ExpressionReference> = Vec::new();
215        for column_index in 0..operand_length {
216            if output_index < column_index {
217                continue;
218            }
219            let row_offset = output_index - column_index;
220            if row_offset >= column_length {
221                continue;
222            }
223            let toeplitz_entry = toeplitz_column[row_offset].clone();
224            let operand_entry = operand[column_index].clone();
225            term_list.push(builders::product(
226                context,
227                vec![toeplitz_entry, operand_entry],
228            ));
229        }
230        if term_list.is_empty() {
231            result.push(builders::zero(context));
232        } else {
233            result.push(builders::sum(context, term_list));
234        }
235    }
236    result
237}
