53pub(crate) fn rational_base_field(context: &Context) -> MatrixResult<Arc<NumberField>> {
54    let generator = context
55        .symbol_table
56        .fresh("matrix_rational_field_generator");
57    let defining = Polynomial {
58        main_variable: generator,
59        terms: vec![(
60            1,
61            PolynomialCoefficient::Numeric(Number::Integer(integer_one())),
62        )],
63    };
64    make_number_field(generator, &context.algebraic_field_interner, defining)
65        .ok_or(MatrixError::ExactEntryUnsupported)
66}

68pub(crate) fn normalize_matrix(
69    context: &Context,
70    matrix_value: &Matrix,
71) -> MatrixResult<(Arc<NumberField>, ExactMatrix)> {
72    if !matrix_value.is_square() {
73        return Err(MatrixError::JordanRequiresSquare {
74            rows: matrix_value.rows,
75            columns: matrix_value.columns,
76        });
77    }
78    if matrix_value.rows == 0 {
79        return Err(MatrixError::EmptyMatrix);
80    }
81    if matrix_value.rows > MAXIMUM_EXACT_MATRIX_DIMENSION {
82        return Err(MatrixError::ResourceBudgetExhausted);
83    }
84    let field = match discover_radical_field(context, &matrix_value.entries)? {
85        Some(value) => value,
86        None => rational_base_field(context)?,
87    };
88    let mut entries = Vec::with_capacity(matrix_value.entries.len());
89    for entry in &matrix_value.entries {
90        let converted = expression_to_algebraic_element(context, entry, &field)
91            .ok_or(MatrixError::ExactEntryUnsupported)?;
92        entries.push(converted);
93    }
94    Ok((
95        field,
96        ExactMatrix {
97            rows: matrix_value.rows,
98            columns: matrix_value.columns,
99            entries,
100        },
101    ))
102}

104fn discover_radical_field(
105    context: &Context,
106    entries: &[ExpressionReference],
107) -> MatrixResult<Option<Arc<NumberField>>> {
108    let mut candidate: Option<(exact_rational, usize)> = None;
109    for entry in entries {
110        collect_radical(entry, &mut candidate)?;
111    }
112    let Some((radicand, degree)) = candidate else {
113        return Ok(None);
114    };
115    let generator = context
116        .symbol_table
117        .fresh("matrix_algebraic_field_generator");
118    let constant = if rational_is_integer(&radicand) {
119        Number::Integer(rational_to_integer(&radicand).ok_or(MatrixError::ExactEntryUnsupported)?)
120    } else {
121        Number::Rational(radicand.clone())
122    };
123    let defining = Polynomial {
124        main_variable: generator,
125        terms: vec![
126            (
127                degree,
128                PolynomialCoefficient::Numeric(Number::Integer(integer_one())),
129            ),
130            (
131                0,
132                PolynomialCoefficient::Numeric(crate::number::negate(&constant)),
133            ),
134        ],
135    };
136    let field = make_number_field(generator, &context.algebraic_field_interner, defining)
137        .ok_or(MatrixError::ExactEntryUnsupported)?;
138    Ok(Some(field))
139}

141fn collect_radical(
142    expression: &ExpressionReference,
143    candidate: &mut Option<(exact_rational, usize)>,
144) -> MatrixResult<()> {
145    match &expression.kind {
146        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) => Ok(()),
147        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
148            for child in children {
149                collect_radical(child, candidate)?;
150            }
151            Ok(())
152        }
153        ExpressionKind::Power(base, exponent) => {
154            if let Some((radicand, degree)) = pure_radical(base, exponent) {
155                if let Some(existing) = candidate {
156                    if existing.0 != radicand || existing.1 != degree {
157                        return Err(MatrixError::ExactEntryUnsupported);
158                    }
159                } else {
160                    *candidate = Some((radicand, degree));
161                }
162                Ok(())
163            } else if matches!(exponent.kind, ExpressionKind::Integer(_)) {
164                collect_radical(base, candidate)
165            } else {
166                Err(MatrixError::ExactEntryUnsupported)
167            }
168        }
169        _ => Err(MatrixError::ExactEntryUnsupported),
170    }
171}

173fn pure_radical(
174    base: &ExpressionReference,
175    exponent: &ExpressionReference,
176) -> Option<(exact_rational, usize)> {
177    let radicand = match &base.kind {
178        ExpressionKind::Integer(value) => exact_rational::from(value.clone()),
179        ExpressionKind::Rational(value) => value.clone(),
180        _ => return None,
181    };
182    let exponent_value = match &exponent.kind {
183        ExpressionKind::Rational(value) => value,
184        _ => return None,
185    };
186    let parts = rational_into_parts(exponent_value.clone());
187    if integer_absolute_magnitude(&parts.numerator) != natural_one() {
188        return None;
189    }
190    Some((
191        radicand,
192        usize::try_from(natural_to_u64(&parts.denominator)?).ok()?,
193    ))
194}

212pub(crate) fn multiply_matrix(left: &ExactMatrix, right: &ExactMatrix) -> ExactMatrix {
213    let field = &left.entries[0].field;
214    let mut result = zero_matrix(field, left.rows, right.columns);
215    for row in 0..left.rows {
216        for column in 0..right.columns {
217            let mut value = zero_element(field);
218            for inner in 0..left.columns {
219                value = add(
220                    &value,
221                    &multiply(left.entry(row, inner), right.entry(inner, column)),
222                );
223            }
224            result.entries[row * right.columns + column] = value;
225        }
226    }
227    result
228}

230pub(crate) fn subtract_scalar_identity(
231    value: &ExactMatrix,
232    scalar: &AlgebraicElement,
233) -> ExactMatrix {
234    let mut result = value.clone();
235    for index in 0..value.rows {
236        let position = index * value.columns + index;
237        result.entries[position] = subtract(&result.entries[position], scalar);
238    }
239    result
240}

242pub(crate) fn power_matrix(value: &ExactMatrix, exponent: usize) -> ExactMatrix {
243    if exponent == 0 {
244        return identity_matrix(&value.entries[0].field, value.rows);
245    }
246    let mut result = identity_matrix(&value.entries[0].field, value.rows);
247    let mut current = value.clone();
248    let mut remaining = exponent;
249    while remaining > 0 {
250        if remaining & 1 == 1 {
251            result = multiply_matrix(&result, &current);
252        }
253        remaining >>= 1;
254        if remaining > 0 {
255            current = multiply_matrix(&current, &current);
256        }
257    }
258    result
259}

265pub(crate) fn kernel_basis(value: &ExactMatrix) -> Vec<Vec<AlgebraicElement>> {
266    let (pivots, reduced) = elimination(value);
267    let field = &value.entries[0].field;
268    let free_columns: Vec<usize> = (0..value.columns)
269        .filter(|column| !pivots.contains(column))
270        .collect();
271    let mut basis = Vec::with_capacity(free_columns.len());
272    for free_column in free_columns {
273        let mut vector = vec![zero_element(field); value.columns];
274        vector[free_column] = one_element(field);
275        for (pivot_row, pivot_column) in pivots.iter().enumerate().rev() {
276            vector[*pivot_column] = negate(&reduced.entry(pivot_row, free_column));
277        }
278        basis.push(vector);
279    }
280    basis
281}

283fn elimination(value: &ExactMatrix) -> (Vec<usize>, ExactMatrix) {
284    let mut reduced = value.clone();
285    let mut pivots = Vec::new();
286    let mut pivot_row = 0usize;
287    for column in 0..value.columns {
288        let Some(candidate_row) =
289            (pivot_row..value.rows).find(|row| !is_zero(reduced.entry(*row, column)))
290        else {
291            continue;
292        };
293        if candidate_row != pivot_row {
294            for swap_column in 0..value.columns {
295                reduced.entries.swap(
296                    pivot_row * value.columns + swap_column,
297                    candidate_row * value.columns + swap_column,
298                );
299            }
300        }
301        let pivot = reduced.entry(pivot_row, column).clone();
302        for scale_column in column..value.columns {
303            let position = pivot_row * value.columns + scale_column;
304            reduced.entries[position] = divide(&reduced.entries[position], &pivot).unwrap();
305        }
306        for row in 0..value.rows {
307            if row == pivot_row {
308                continue;
309            }
310            let factor = reduced.entry(row, column).clone();
311            if is_zero(&factor) {
312                continue;
313            }
314            for reduce_column in column..value.columns {
315                let position = row * value.columns + reduce_column;
316                let correction = multiply(&factor, reduced.entry(pivot_row, reduce_column));
317                reduced.entries[position] = subtract(&reduced.entries[position], &correction);
318            }
319        }
320        pivots.push(column);
321        pivot_row += 1;
322        if pivot_row == value.rows {
323            break;
324        }
325    }
326    (pivots, reduced)
327}

329pub(crate) fn columns_matrix(
330    field: &Arc<NumberField>,
331    rows: usize,
332    columns: &[Vec<AlgebraicElement>],
333) -> ExactMatrix {
334    if columns.is_empty() {
335        return zero_matrix(field, rows, 0);
336    }
337    let mut entries = Vec::with_capacity(rows * columns.len());
338    for row in 0..rows {
339        for column in columns {
340            entries.push(column[row].clone());
341        }
342    }
343    ExactMatrix {
344        rows,
345        columns: columns.len(),
346        entries,
347    }
348}

350pub(crate) fn multiply_vector(
351    value: &ExactMatrix,
352    vector: &[AlgebraicElement],
353) -> Vec<AlgebraicElement> {
354    let field = &value.entries[0].field;
355    let mut result = vec![zero_element(field); value.rows];
356    for row in 0..value.rows {
357        for column in 0..value.columns {
358            result[row] = add(
359                &result[row],
360                &multiply(value.entry(row, column), &vector[column]),
361            );
362        }
363    }
364    result
365}

377pub(crate) fn berkowitz_coefficients(value: &ExactMatrix) -> Vec<AlgebraicElement> {
378    let field = &value.entries[0].field;
379    let mut running = vec![one_element(field)];
380    for principal_size in 1..=value.rows {
381        let leading = principal_size - 1;
382        let mut column = vec![one_element(field), negate(value.entry(leading, leading))];
383        if principal_size > 1 {
384            let inner_size = principal_size - 1;
385            let row_vector: Vec<_> = (0..inner_size)
386                .map(|column| value.entry(leading, column).clone())
387                .collect();
388            let mut column_vector: Vec<_> = (0..inner_size)
389                .map(|row| value.entry(row, leading).clone())
390                .collect();
391            for _ in 0..principal_size - 1 {
392                let mut inner_product = zero_element(field);
393                for index in 0..inner_size {
394                    inner_product = add(
395                        &inner_product,
396                        &multiply(&row_vector[index], &column_vector[index]),
397                    );
398                }
399                column.push(negate(&inner_product));
400                let mut next = vec![zero_element(field); inner_size];
401                for row in 0..inner_size {
402                    for inner in 0..inner_size {
403                        next[row] = add(
404                            &next[row],
405                            &multiply(value.entry(row, inner), &column_vector[inner]),
406                        );
407                    }
408                }
409                column_vector = next;
410            }
411        }
412        let mut next = vec![zero_element(field); column.len()];
413        for output in 0..column.len() {
414            for input in 0..running.len() {
415                if output >= input {
416                    next[output] = add(
417                        &next[output],
418                        &multiply(&column[output - input], &running[input]),
419                    );
420                }
421            }
422        }
423        running = next;
424    }
425    running.reverse();
426    running
427}

429pub(crate) fn coefficients_polynomial(
430    variable: crate::symbol::SymbolIdentifier,
431    coefficients: &[AlgebraicElement],
432) -> Polynomial {
433    let mut terms: Vec<_> = coefficients
434        .iter()
435        .enumerate()
436        .filter(|(_, coefficient)| !is_zero(coefficient))
437        .map(|(exponent, coefficient)| {
438            (
439                exponent,
440                PolynomialCoefficient::Algebraic(coefficient.clone()),
441            )
442        })
443        .collect();
444    terms.reverse();
445    Polynomial {
446        main_variable: variable,
447        terms,
448    }
449}

451pub(crate) fn reify_element(
452    context: &Context,
453    element: &AlgebraicElement,
454    selected_root: Option<&ExpressionReference>,
455) -> ExpressionReference {
456    let Some(root) = selected_root else {
457        return algebraic_element_to_expression(context, element);
458    };
459    let mut terms = Vec::new();
460    for (exponent, coefficient) in element.representative.iter().enumerate() {
461        if *coefficient == rational_zero() {
462            continue;
463        }
464        let coefficient_expression =
465            crate::expression::builders::rational_literal(context, coefficient.clone());
466        if exponent == 0 {
467            terms.push(coefficient_expression);
468        } else {
469            let exponent_expression =
470                crate::expression::builders::integer_value(context, exponent as i64);
471            let power =
472                crate::expression::builders::power(context, root.clone(), exponent_expression);
473            terms.push(crate::expression::builders::product(
474                context,
475                vec![coefficient_expression, power],
476            ));
477        }
478    }
479    crate::expression::builders::sum(context, terms)
480}

482pub(crate) fn rational_sign(value: &AlgebraicElement) -> Option<RootClassification> {
483    let rational = rational_value(value)?;
484    if rational == rational_zero() {
485        Some(RootClassification::Zero)
486    } else if rational > rational_zero() {
487        Some(RootClassification::PositiveReal)
488    } else {
489        Some(RootClassification::NegativeReal)
490    }
491}
