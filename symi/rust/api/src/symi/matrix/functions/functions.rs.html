19/// Return a replayable postcondition certificate for an exact matrix function result.
20pub fn matrix_function_postcondition_evidence(
21    context: &Context,
22    input: &Matrix,
23    result: &Matrix,
24    function: crate::assumptions::matrix_operation_evidence::MatrixFunction,
25) -> Option<crate::assumptions::matrix_operation_evidence::MatrixOperationEvidence> {
26    let input =
27        crate::assumptions::matrix_subject::MatrixSubject::from_matrix(context, input).ok()?;
28    let result =
29        crate::assumptions::matrix_subject::MatrixSubject::from_matrix(context, result).ok()?;
30    Some(
31        crate::assumptions::matrix_operation_evidence::MatrixOperationEvidence::Function {
32            input,
33            result,
34            function,
35        },
36    )
37}

39#[derive(Clone)]
40enum PrimaryMatrixFunction {
41    Exponential,
42    /// The exponential of a scalar multiple of the matrix, carrying that scalar.
43    ///
44    /// This is not the exponential of a second matrix: the scalar commutes with everything, so
45    /// `exp(s A)` is the primary function `z -> exp(s z)` evaluated on `A`, and its Jordan block
46    /// coefficients are the derivatives of that function at the eigenvalue. Keeping the scalar
47    /// here rather than forming the product `s A` first is what lets the eigenvalues stay exact
48    /// when the scalar is a free symbol, which is the whole point of the operation. Its `k`-th
49    /// derivative at an eigenvalue is `s^k exp(s lambda)`, which the block coefficient divides by
50    /// `k!`, so on the nilpotent part the block reads `exp(s lambda) sum_k s^k N^k / k!`.
51    ScaledExponential(ExpressionReference),
52    Logarithm,
53    SquareRoot,
54    Sine,
55    Cosine,
56    HyperbolicSine,
57    HyperbolicCosine,
58}

112fn evaluate(
113    context: &Context,
114    matrix_value: &Matrix,
115    function: PrimaryMatrixFunction,
116) -> MatrixResult<Matrix> {
117    let decomposition = certified_decomposition(context, matrix_value)?;
118    let block_function = block_diagonal_function(context, &decomposition, &function)?;
119    if matrix_value.rows == 1 {
120        if matches!(&function, PrimaryMatrixFunction::SquareRoot) {
121            let square = matrix_multiply(context, &block_function, &block_function)?;
122            verify_matrices_equal(context, &square, matrix_value)?;
123        }
124        return Ok(block_function);
125    }
126    let route = function_route(context, 2, matrix_value.rows);
127    let (inverse, left_product) = join(
128        route,
129        || matrix_inverse(context, &decomposition.public.similarity),
130        || matrix_multiply(context, &decomposition.public.similarity, &block_function),
131    );
132    let inverse = inverse?;
133    let left_product = left_product?;
134    let result = matrix_multiply(context, &left_product, &inverse)?;
135    verify_reconstruction(context, &decomposition, &block_function, &result)?;
136    if matches!(&function, PrimaryMatrixFunction::SquareRoot) {
137        let square = matrix_multiply(context, &result, &result)?;
138        verify_matrices_equal(context, &square, matrix_value)?;
139    }
140    Ok(result)
141}

143fn evaluate_entire_pair(
144    context: &Context,
145    matrix_value: &Matrix,
146    requested_function: PrimaryMatrixFunction,
147) -> MatrixResult<Matrix> {
148    let decomposition = certified_decomposition(context, matrix_value)?;
149    let trigonometric = matches!(
150        &requested_function,
151        PrimaryMatrixFunction::Sine | PrimaryMatrixFunction::Cosine
152    );
153    let first_function = if trigonometric {
154        PrimaryMatrixFunction::Sine
155    } else {
156        PrimaryMatrixFunction::HyperbolicSine
157    };
158    let second_function = if trigonometric {
159        PrimaryMatrixFunction::Cosine
160    } else {
161        PrimaryMatrixFunction::HyperbolicCosine
162    };
163    let route = function_route(context, 2, matrix_value.rows);
164    let (first_block, second_block) = join(
165        route,
166        || block_diagonal_function(context, &decomposition, &first_function),
167        || block_diagonal_function(context, &decomposition, &second_function),
168    );
169    let first_block = first_block?;
170    let second_block = second_block?;
171    verify_entire_pair_identity(context, &first_block, &second_block, trigonometric)?;
172    let requested_block = if matches!(
173        &requested_function,
174        PrimaryMatrixFunction::Sine | PrimaryMatrixFunction::HyperbolicSine
175    ) {
176        &first_block
177    } else {
178        &second_block
179    };
180    reconstruct(context, &decomposition, requested_block)
181}

183fn reconstruct(
184    context: &Context,
185    decomposition: &CertifiedJordanDecomposition,
186    block_function: &Matrix,
187) -> MatrixResult<Matrix> {
188    if decomposition.public.jordan_form.rows == 1 {
189        return Ok(block_function.clone());
190    }
191    let route = function_route(context, 2, decomposition.public.jordan_form.rows);
192    let (inverse, left_product) = join(
193        route,
194        || matrix_inverse(context, &decomposition.public.similarity),
195        || matrix_multiply(context, &decomposition.public.similarity, block_function),
196    );
197    let inverse = inverse?;
198    let left_product = left_product?;
199    let result = matrix_multiply(context, &left_product, &inverse)?;
200    verify_reconstruction(context, decomposition, block_function, &result)?;
201    Ok(result)
202}

204fn verify_entire_pair_identity(
205    context: &Context,
206    first: &Matrix,
207    second: &Matrix,
208    trigonometric: bool,
209) -> MatrixResult<()> {
210    let route = function_route(context, 2, first.rows);
211    let (first_square, second_square) = join(
212        route,
213        || matrix_multiply(context, first, first),
214        || matrix_multiply(context, second, second),
215    );
216    let first_square = first_square?;
217    let second_square = second_square?;
218    let combined = if trigonometric {
219        matrix_add(context, &first_square, &second_square)?
220    } else {
221        matrix_subtract(context, &second_square, &first_square)?
222    };
223    let identity = matrix_identity(context, first.rows)?;
224    verify_matrices_equal(context, &combined, &identity)
225}

227fn block_diagonal_function(
228    context: &Context,
229    decomposition: &CertifiedJordanDecomposition,
230    function: &PrimaryMatrixFunction,
231) -> MatrixResult<Matrix> {
232    let dimension = decomposition.public.jordan_form.rows;
233    let route = function_route(context, decomposition.blocks.len(), dimension);
234    let coefficients = ordered_map(route, &decomposition.blocks, |block| {
235        block_coefficients(
236            context,
237            &block.root.expression,
238            block.root.classification,
239            block.vectors.len(),
240            function,
241        )
242    });
243    let zero = builders::zero(context);
244    let mut entries = vec![zero; dimension * dimension];
245    let mut offset = 0usize;
246    for (block, block_coefficients) in decomposition.blocks.iter().zip(coefficients) {
247        let block_coefficients = block_coefficients?;
248        for row in 0..block.vectors.len() {
249            for column in row..block.vectors.len() {
250                entries[(offset + row) * dimension + offset + column] =
251                    block_coefficients[column - row].clone();
252            }
253        }
254        offset += block.vectors.len();
255    }
256    Ok(Matrix {
257        rows: dimension,
258        columns: dimension,
259        entries,
260    })
261}

263fn block_coefficients(
264    context: &Context,
265    eigenvalue: &ExpressionReference,
266    classification: RootClassification,
267    size: usize,
268    function: &PrimaryMatrixFunction,
269) -> MatrixResult<Vec<ExpressionReference>> {
270    match function {
271        PrimaryMatrixFunction::Exponential => {
272            let exponential = builders::function_call(
273                context,
274                FunctionIdentifier::Builtin(BuiltinFunction::Exp),
275                vec![eigenvalue.clone()],
276            );
277            let mut coefficients = Vec::with_capacity(size);
278            let mut factorial = integer_one();
279            for order in 0..size {
280                if order > 0 {
281                    factorial *= exact_integer::from(order);
282                }
283                let divisor = builders::integer(context, factorial.clone());
284                coefficients.push(builders::divide(context, exponential.clone(), divisor));
285            }
286            Ok(coefficients)
287        }
288        PrimaryMatrixFunction::ScaledExponential(scalar) => {
289            let rate = builders::product(context, vec![scalar.clone(), eigenvalue.clone()]);
290            let exponential = builders::function_call(
291                context,
292                FunctionIdentifier::Builtin(BuiltinFunction::Exp),
293                vec![rate],
294            );
295            let mut coefficients = Vec::with_capacity(size);
296            let mut factorial = integer_one();
297            for order in 0..size {
298                if order > 0 {
299                    factorial *= exact_integer::from(order);
300                }
301                let power = if order == 0 {
302                    builders::one(context)
303                } else {
304                    let exponent = builders::integer_value(context, order as i64);
305                    builders::power(context, scalar.clone(), exponent)
306                };
307                let divisor = builders::integer(context, factorial.clone());
308                let scaled = builders::product(context, vec![exponential.clone(), power]);
309                coefficients.push(builders::divide(context, scaled, divisor));
310            }
311            Ok(coefficients)
312        }
313        PrimaryMatrixFunction::Logarithm => {
314            if matches!(
315                classification,
316                RootClassification::NegativeReal | RootClassification::Zero
317            ) {
318                return Err(MatrixError::PrincipalBranchCut);
319            }
320            let logarithm = builders::function_call(
321                context,
322                FunctionIdentifier::Builtin(BuiltinFunction::Log),
323                vec![eigenvalue.clone()],
324            );
325            let mut coefficients = vec![logarithm];
326            for order in 1..size {
327                let exponent = builders::integer_value(context, order as i64);
328                let power = builders::power(context, eigenvalue.clone(), exponent);
329                let denominator_integer = builders::integer_value(context, order as i64);
330                let denominator = builders::product(context, vec![denominator_integer, power]);
331                let numerator =
332                    builders::integer_value(context, if order % 2 == 1 { 1 } else { -1 });
333                coefficients.push(builders::divide(context, numerator, denominator));
334            }
335            Ok(coefficients)
336        }
337        PrimaryMatrixFunction::SquareRoot => {
338            if classification == RootClassification::NegativeReal {
339                return Err(MatrixError::PrincipalBranchCut);
340            }
341            if classification == RootClassification::Zero {
342                if size > 1 {
343                    return Err(MatrixError::NonanalyticZeroBlock);
344                }
345                return Ok(vec![builders::zero(context)]);
346            }
347            let mut coefficients = Vec::with_capacity(size);
348            let mut binomial = rational_one();
349            for order in 0..size {
350                if order > 0 {
351                    binomial = binomial
352                        * (rational_from_integers(integer_one(), exact_integer::from(2)).unwrap()
353                            - exact_rational::from(order - 1))
354                        / exact_rational::from(order);
355                }
356                let exponent = rational_from_integers(
357                    integer_one() - exact_integer::from(2 * order),
358                    exact_integer::from(2),
359                )
360                .unwrap();
361                let exponent_expression = builders::rational_literal(context, exponent);
362                let power = builders::power(context, eigenvalue.clone(), exponent_expression);
363                let coefficient = builders::rational_literal(context, binomial.clone());
364                coefficients.push(builders::product(context, vec![coefficient, power]));
365            }
366            Ok(coefficients)
367        }
368        PrimaryMatrixFunction::Sine => periodic_coefficients(
369            context,
370            eigenvalue,
371            size,
372            &[
373                (BuiltinFunction::Sin, false),
374                (BuiltinFunction::Cos, false),
375                (BuiltinFunction::Sin, true),
376                (BuiltinFunction::Cos, true),
377            ],
378        ),
379        PrimaryMatrixFunction::Cosine => periodic_coefficients(
380            context,
381            eigenvalue,
382            size,
383            &[
384                (BuiltinFunction::Cos, false),
385                (BuiltinFunction::Sin, true),
386                (BuiltinFunction::Cos, true),
387                (BuiltinFunction::Sin, false),
388            ],
389        ),
390        PrimaryMatrixFunction::HyperbolicSine => periodic_coefficients(
391            context,
392            eigenvalue,
393            size,
394            &[
395                (BuiltinFunction::Sinh, false),
396                (BuiltinFunction::Cosh, false),
397            ],
398        ),
399        PrimaryMatrixFunction::HyperbolicCosine => periodic_coefficients(
400            context,
401            eigenvalue,
402            size,
403            &[
404                (BuiltinFunction::Cosh, false),
405                (BuiltinFunction::Sinh, false),
406            ],
407        ),
408    }
409}

411fn periodic_coefficients(
412    context: &Context,
413    eigenvalue: &ExpressionReference,
414    size: usize,
415    derivative_cycle: &[(BuiltinFunction, bool)],
416) -> MatrixResult<Vec<ExpressionReference>> {
417    let mut coefficients = Vec::with_capacity(size);
418    let mut factorial = integer_one();
419    for order in 0..size {
420        if order > 0 {
421            factorial *= exact_integer::from(order);
422        }
423        let (function, negative) = derivative_cycle[order % derivative_cycle.len()];
424        let value = builders::function_call(
425            context,
426            FunctionIdentifier::Builtin(function),
427            vec![eigenvalue.clone()],
428        );
429        let signed_value = if negative {
430            builders::negate(context, value)
431        } else {
432            value
433        };
434        let divisor = builders::integer(context, factorial.clone());
435        coefficients.push(builders::divide(context, signed_value, divisor));
436    }
437    Ok(coefficients)
438}

440fn verify_reconstruction(
441    context: &Context,
442    decomposition: &CertifiedJordanDecomposition,
443    block_function: &Matrix,
444    result: &Matrix,
445) -> MatrixResult<()> {
446    let route = function_route(context, 2, result.rows);
447    let (left, right) = join(
448        route,
449        || matrix_multiply(context, &decomposition.public.similarity, block_function),
450        || matrix_multiply(context, result, &decomposition.public.similarity),
451    );
452    let left = left?;
453    let right = right?;
454    verify_matrices_equal(context, &left, &right)
455}
456
457fn verify_matrices_equal(context: &Context, left: &Matrix, right: &Matrix) -> MatrixResult<()> {
458    if left.shape() != right.shape() {
459        return Err(MatrixError::MatrixFunctionVerificationFailed);
460    }
461    let route = function_route(context, left.entries.len(), left.rows.max(left.columns));
462    let indices = (0..left.entries.len()).collect::<Vec<_>>();
463    let differences = ordered_map(route, &indices, |entry_index| {
464        let difference = builders::subtract(
465            context,
466            left.entries[*entry_index].clone(),
467            right.entries[*entry_index].clone(),
468        );
469        crate::simplify::simplify(context, difference)
470    });
471    if differences
472        .iter()
473        .any(|difference| is_nonzero(difference, &context.symbol_table) != TruthValue::No)
474    {
475        return Err(MatrixError::MatrixFunctionVerificationFailed);
476    }
477    Ok(())
478}

480fn function_route(
481    context: &Context,
482    independent_unit_count: usize,
483    dimension: usize,
484) -> ExecutionRoute {
485    if !context.operation().may_fork_work_class("matrix_function") {
486        return ExecutionRoute::Serial;
487    }
488    let worker_count = crate::parallelism::maximum_operation_concurrency();
489    let requested_permits = independent_unit_count.min(worker_count.max(1));
490    context
491        .operation()
492        .execution_decision(
493            &WorkEstimate::new(
494                (independent_unit_count as u64)
495                    .saturating_mul(dimension as u64)
496                    .saturating_mul(dimension as u64),
497                independent_unit_count,
498                dimension.saturating_mul(dimension).saturating_mul(256),
499                0,
500                requested_permits,
501                "matrix_function",
502            ),
503            requested_permits,
504            crate::parallelism::runtime_memory_capacity(
505                dimension.saturating_mul(dimension).saturating_mul(256),
506            ),
507        )
508        .route
