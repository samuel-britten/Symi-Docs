93pub(crate) fn certified_decomposition(
94    context: &Context,
95    matrix_value: &Matrix,
96) -> MatrixResult<CertifiedJordanDecomposition> {
97    let (base_field, base_matrix) = normalize_matrix(context, matrix_value)?;
98    let variable = context.symbol_table.fresh("matrix_jordan_eigenvalue");
99    let coefficients = berkowitz_coefficients(&base_matrix);
100    let characteristic = coefficients_polynomial(variable, &coefficients);
101    let factors: Vec<(usize, Polynomial)> = if base_field.degree == 1 {
102        let rational_characteristic = algebraic_polynomial_to_rational(&characteristic)?;
103        crate::polynomial::factorization::factor_univariate_over_integers_with_operation(
104            &rational_characteristic,
105            context.operation(),
106        )
107    } else {
108        factor_in_algebraic_extension(&characteristic, &base_field)
109            .into_iter()
110            .map(|factor| (factor.multiplicity, factor.factor))
111            .collect()
112    };
113    if factors.is_empty() {
114        return Err(MatrixError::RootDescriptorUncertified);
115    }
116    let factor_route = decomposition_route(
117        context,
118        factors.len(),
119        (matrix_value.rows as u64)
120            .saturating_mul(matrix_value.rows as u64)
121            .saturating_mul(factors.len() as u64),
122        matrix_value
123            .rows
124            .saturating_mul(matrix_value.rows)
125            .saturating_mul(256),
126    );
127    let factor_results = ordered_map(factor_route, &factors, |(multiplicity, factor)| {
128        root_data_for_factor(context, &base_field, &base_matrix, *multiplicity, factor)
129    });
130    let mut root_data = Vec::new();
131    for result in factor_results {
132        root_data.extend(result?);
133    }
134    root_data.sort_by(|(left, _, _), (right, _, _)| {
135        classification_order(left.classification)
136            .cmp(&classification_order(right.classification))
137            .then(left.slot.cmp(&right.slot))
138    });
139    let chain_route = decomposition_route(
140        context,
141        root_data.len(),
142        (matrix_value.rows as u64)
143            .saturating_mul(matrix_value.rows as u64)
144            .saturating_mul(root_data.len() as u64),
145        matrix_value
146            .rows
147            .saturating_mul(matrix_value.rows)
148            .saturating_mul(512),
149    );
150    let chain_results = ordered_map(
151        chain_route,
152        &root_data,
153        |(root, selected_matrix, eigenvalue)| {
154            chains_for_root(selected_matrix, root.clone(), eigenvalue, chain_route)
155        },
156    );
157    let mut certified_blocks = Vec::new();
158    for result in chain_results {
159        certified_blocks.extend(result?);
160    }
161    if certified_blocks
162        .iter()
163        .map(|block| block.vectors.len())
164        .sum::<usize>()
165        != matrix_value.rows
166    {
167        return Err(MatrixError::JordanKernelGrowthInconsistent);
168    }
169    certified_blocks.sort_by(|left, right| {
170        classification_order(left.root.classification)
171            .cmp(&classification_order(right.root.classification))
172            .then(left.root.slot.cmp(&right.root.slot))
173            .then(right.vectors.len().cmp(&left.vectors.len()))
174    });
175    let decomposition = assemble_decomposition(context, matrix_value.rows, certified_blocks)?;
176    verify_similarity_certificate(context, matrix_value, &decomposition.public)?;
177    Ok(decomposition)
178}

180/// Choose the route for the entrywise similarity check of one Jordan certificate.
181fn certificate_route(
182    context: &Context,
183    independent_unit_count: usize,
184    source_rows: usize,
185) -> ExecutionRoute {
186    if !context
187        .operation()
188        .may_fork_work_class("matrix_certificate")
189    {
190        return ExecutionRoute::Serial;
191    }
192    let worker_count = crate::parallelism::maximum_operation_concurrency();
193    let requested_permits = independent_unit_count.min(worker_count.max(1));
194    context
195        .operation()
196        .execution_decision(
197            &WorkEstimate::new(
198                (independent_unit_count as u64).saturating_mul(source_rows as u64),
199                independent_unit_count,
200                256,
201                0,
202                requested_permits,
203                "matrix_certificate",
204            ),
205            requested_permits,
206            crate::parallelism::runtime_memory_capacity(256),
207        )
208        .route
209}

242fn root_data_for_factor(
243    context: &Context,
244    base_field: &Arc<crate::algebraic_numbers::field::NumberField>,
245    base_matrix: &ExactMatrix,
246    multiplicity: usize,
247    factor: &Polynomial,
248) -> MatrixResult<Vec<(ExactRoot, ExactMatrix, AlgebraicElement)>> {
249    let factor_degree = degree(factor).ok_or(MatrixError::RootDescriptorUncertified)?;
250    if factor_degree == 1 {
251        let constant = factor
252            .terms
253            .iter()
254            .find(|(exponent, _)| *exponent == 0)
255            .and_then(|(_, coefficient)| coefficient_to_element(coefficient, base_field))
256            .unwrap_or_else(|| crate::algebraic_numbers::zero_element(base_field));
257        let eigenvalue = negate(&constant);
258        let classification = rational_sign(&eigenvalue)
259            .or_else(|| classify_base_generator(&eigenvalue))
260            .ok_or(MatrixError::RootClassificationUnsupported)?;
261        let expression = reify_element(context, &eigenvalue, None);
262        return Ok(vec![(
263            ExactRoot {
264                field: Arc::clone(base_field),
265                expression,
266                classification,
267                multiplicity,
268                slot: 0,
269            },
270            base_matrix.clone(),
271            eigenvalue,
272        )]);
273    }
274    if base_field.degree != 1 {
275        return Err(MatrixError::RootDescriptorUncertified);
276    }
277    let rational_factor = if factor
278        .terms
279        .iter()
280        .all(|(_, coefficient)| matches!(coefficient, PolynomialCoefficient::Numeric(_)))
281    {
282        factor.clone()
283    } else {
284        algebraic_polynomial_to_rational(factor)?
285    };
286    let selected = selected_roots(context, &rational_factor, multiplicity)?;
287    selected
288        .into_iter()
289        .map(|(root, eigenvalue)| {
290            let promoted_entries = base_matrix
291                .entries
292                .iter()
293                .map(|entry| {
294                    rational_value(entry)
295                        .map(|value| rational_element_in(&root.field, value))
296                        .ok_or(MatrixError::ExactEntryUnsupported)
297                })
298                .collect::<MatrixResult<Vec<_>>>()?;
299            Ok((
300                root,
301                ExactMatrix {
302                    rows: base_matrix.rows,
303                    columns: base_matrix.columns,
304                    entries: promoted_entries,
305                },
306                eigenvalue,
307            ))
308        })
309        .collect()
310}

312fn classify_base_generator(element: &AlgebraicElement) -> Option<RootClassification> {
313    if is_zero(element) {
314        return Some(RootClassification::Zero);
315    }
316    let embedding_count =
317        crate::real_algebraic_numbers::embedding::real_embedding_count(&element.field);
318    if embedding_count == 0 {
319        return None;
320    }
321    let embedding =
322        crate::real_algebraic_numbers::embedding::embed_real(&element.field, embedding_count - 1)?;
323    let sign = crate::real_algebraic_numbers::embedding::sign_of_element(&embedding, element);
324    if sign > 0 {
325        Some(RootClassification::PositiveReal)
326    } else if sign < 0 {
327        Some(RootClassification::NegativeReal)
328    } else {
329        Some(RootClassification::Zero)
330    }
331}

333fn classification_order(value: RootClassification) -> u8 {
334    match value {
335        RootClassification::NegativeReal => 0,
336        RootClassification::Zero => 1,
337        RootClassification::PositiveReal => 2,
338        RootClassification::Nonreal => 3,
339    }

342fn algebraic_polynomial_to_rational(value: &Polynomial) -> MatrixResult<Polynomial> {
343    let mut terms = Vec::with_capacity(value.terms.len());
344    for (exponent, coefficient) in &value.terms {
345        let element = match coefficient {
346            PolynomialCoefficient::Algebraic(element) => element,
347            _ => return Err(MatrixError::RootDescriptorUncertified),
348        };
349        let rational = rational_value(element).ok_or(MatrixError::RootDescriptorUncertified)?;
350        let numeric = if crate::exact_arithmetic::rational_denominator(&rational) == natural_one() {
351            Number::Integer(
352                crate::exact_arithmetic::rational_to_integer(&rational)
353                    .ok_or(MatrixError::RootDescriptorUncertified)?,
354            )
355        } else {
356            Number::Rational(rational)
357        };
358        terms.push((*exponent, PolynomialCoefficient::Numeric(numeric)));
359    }
360    Ok(Polynomial {
361        main_variable: value.main_variable,
362        terms,
363    })
364}

366fn selected_roots(
367    context: &Context,
368    factor: &Polynomial,
369    multiplicity: usize,
370) -> MatrixResult<Vec<(ExactRoot, AlgebraicElement)>> {
371    let factor_degree = degree(factor).ok_or(MatrixError::RootDescriptorUncertified)?;
372    if factor_degree != 2 {
373        return Err(MatrixError::RootClassificationUnsupported);
374    }
375    let coefficient = |exponent: usize| -> MatrixResult<exact_rational> {
376        let Some((_, value)) = factor.terms.iter().find(|(power, _)| *power == exponent) else {
377            return Ok(rational_zero());
378        };
379        match value {
380            PolynomialCoefficient::Numeric(Number::Integer(value)) => {
381                Ok(exact_rational::from(value.clone()))
382            }
383            PolynomialCoefficient::Numeric(Number::Rational(value)) => Ok(value.clone()),
384            _ => Err(MatrixError::RootDescriptorUncertified),
385        }
386    };
387    let leading = coefficient(2)?;
388    let linear = coefficient(1)? / leading.clone();
389    let constant = coefficient(0)? / leading;
390    let discriminant = linear.clone() * linear.clone() - exact_rational::from(4) * constant.clone();
391    let generator = context.symbol_table.fresh("matrix_selected_root");
392    let renamed = Polynomial {
393        main_variable: generator,
394        terms: factor.terms.clone(),
395    };
396    let field = make_number_field(generator, &context.algebraic_field_interner, renamed)
397        .ok_or(MatrixError::RootDescriptorUncertified)?;
398    let generator_element = crate::algebraic_numbers::generator_element(&field);
399    let half = builders::rational(context, integer_one(), exact_integer::from(2));
400    let discriminant_expression = builders::rational_literal(context, discriminant.clone());
401    let square_root = builders::power(context, discriminant_expression, half);
402    let negative_linear = builders::rational_literal(context, -linear.clone());
403    let two = builders::integer_value(context, 2);
404    let lower_numerator = builders::subtract(context, negative_linear.clone(), square_root.clone());
405    let upper_numerator = builders::sum(context, vec![negative_linear, square_root]);
406    let lower_expression = builders::divide(context, lower_numerator, two.clone());
407    let upper_expression = builders::divide(context, upper_numerator, two);
408    let classifications = quadratic_classifications(&linear, &constant, &discriminant);
409    Ok(vec![
410        (
411            ExactRoot {
412                field: Arc::clone(&field),
413                expression: lower_expression,
414                classification: classifications.0,
415                multiplicity,
416                slot: 0,
417            },
418            generator_element.clone(),
419        ),
420        (
421            ExactRoot {
422                field: Arc::clone(&field),
423                expression: upper_expression,
424                classification: classifications.1,
425                multiplicity,
426                slot: 1,
427            },
428            generator_element,
429        ),
430    ])
431}

433fn quadratic_classifications(
434    linear: &exact_rational,
435    constant: &exact_rational,
436    discriminant: &exact_rational,
437) -> (RootClassification, RootClassification) {
438    if *discriminant < rational_zero() {
439        return (RootClassification::Nonreal, RootClassification::Nonreal);
440    }
441    if *constant < rational_zero() {
442        return (
443            RootClassification::NegativeReal,
444            RootClassification::PositiveReal,
445        );
446    }
447    if *constant == rational_zero() {
448        if *linear > rational_zero() {
449            return (RootClassification::NegativeReal, RootClassification::Zero);
450        }
451        return (RootClassification::Zero, RootClassification::PositiveReal);
452    }
453    if *linear > rational_zero() {
454        (
455            RootClassification::NegativeReal,
456            RootClassification::NegativeReal,
457        )
458    } else {
459        (
460            RootClassification::PositiveReal,
461            RootClassification::PositiveReal,
462        )
463    }
464}

466fn chains_for_root(
467    matrix_value: &ExactMatrix,
468    root: ExactRoot,
469    eigenvalue: &AlgebraicElement,
470    verification_route: ExecutionRoute,
471) -> MatrixResult<Vec<CertifiedJordanBlock>> {
472    let shifted = subtract_scalar_identity(matrix_value, eigenvalue);
473    let mut kernels: Vec<Vec<Vec<AlgebraicElement>>> = vec![Vec::new()];
474    for exponent in 1..=root.multiplicity + 1 {
475        kernels.push(kernel_basis(&power_matrix(&shifted, exponent)));
476        if kernels[exponent].len() == root.multiplicity {
477            if exponent == root.multiplicity + 1 {
478                break;
479            }
480            kernels.push(kernels[exponent].clone());
481            break;
482        }
483    }
484    if kernels.last().map(Vec::len) != Some(root.multiplicity) {
485        return Err(MatrixError::JordanKernelGrowthInconsistent);
486    }
487    let maximum = kernels.len() - 2;
488    let mut exact_counts = vec![0usize; maximum + 1];
489    for size in 1..=maximum {
490        let current = kernels[size].len();
491        let previous = kernels[size - 1].len();
492        let next = kernels.get(size + 1).map(Vec::len).unwrap_or(current);
493        let at_least = current
494            .checked_sub(previous)
495            .ok_or(MatrixError::JordanKernelGrowthInconsistent)?;
496        let next_at_least = next
497            .checked_sub(current)
498            .ok_or(MatrixError::JordanKernelGrowthInconsistent)?;
499        exact_counts[size] = at_least
500            .checked_sub(next_at_least)
501            .ok_or(MatrixError::JordanKernelGrowthInconsistent)?;
502    }
503    if exact_counts
504        .iter()
505        .enumerate()
506        .map(|(size, count)| size * count)
507        .sum::<usize>()
508        != root.multiplicity
509    {
510        return Err(MatrixError::JordanKernelGrowthInconsistent);
511    }
512    let field = &root.field;
513    let mut blocks = Vec::new();
514    for size in (1..=maximum).rev() {
515        let expected = exact_counts[size];
516        if expected == 0 {
517            continue;
518        }
519        let mut span = kernels[size - 1].clone();
520        if let Some(next_kernel) = kernels.get(size + 1) {
521            for vector in next_kernel {
522                span.push(multiply_vector(&shifted, vector));
523            }
524        }
525        let mut current_rank = rank(&columns_matrix(field, matrix_value.rows, &span));
526        let mut selected = Vec::new();
527        for candidate in &kernels[size] {
528            let mut extended = span.clone();
529            extended.push(candidate.clone());
530            let extended_rank = rank(&columns_matrix(field, matrix_value.rows, &extended));
531            if extended_rank > current_rank {
532                span.push(candidate.clone());
533                selected.push(candidate.clone());
534                current_rank = extended_rank;
535                if selected.len() == expected {
536                    break;
537                }
538            }
539        }
540        if selected.len() != expected {
541            return Err(MatrixError::JordanChainConstructionFailed);
542        }
543        let candidates = selected
544            .into_iter()
545            .map(|terminal| {
546                let mut reversed = vec![terminal];
547                while reversed.len() < size {
548                    let next = multiply_vector(&shifted, reversed.last().unwrap());
549                    reversed.push(next);
550                }
551                reversed.reverse();
552                reversed
553            })
554            .collect::<Vec<_>>();
555        let verified_candidates = ordered_map(verification_route, &candidates, |vectors| {
556            if is_zero_vector(&vectors[0]) {
557                return Err(MatrixError::JordanChainConstructionFailed);
558            }
559            for index in 0..vectors.len() {
560                let image = multiply_vector(&shifted, &vectors[index]);
561                if index == 0 {
562                    if !is_zero_vector(&image) {
563                        return Err(MatrixError::JordanSimilarityVerificationFailed);
564                    }
565                } else if !vector_equals(&image, &vectors[index - 1]) {
566                    return Err(MatrixError::JordanSimilarityVerificationFailed);
567                }
568            }
569            Ok(CertifiedJordanBlock {
570                root: root.clone(),
571                vectors: vectors.clone(),
572            })
573        });
574        for candidate in verified_candidates {
575            blocks.push(candidate?);
576        }
577    }
578    let all_vectors: Vec<_> = blocks
579        .iter()
580        .flat_map(|block| block.vectors.clone())
581        .collect();
582    if rank(&columns_matrix(field, matrix_value.rows, &all_vectors)) != root.multiplicity {
583        return Err(MatrixError::JordanChainConstructionFailed);
584    }
585    Ok(blocks)
586}

588fn verify_similarity_certificate(
589    context: &Context,
590    source: &Matrix,
591    decomposition: &MatrixJordanDecomposition,
592) -> MatrixResult<()> {
593    let left = crate::matrix::matrix_multiply(context, source, &decomposition.similarity)?;
594    let right = crate::matrix::matrix_multiply(
595        context,
596        &decomposition.similarity,
597        &decomposition.jordan_form,
598    )?;
599    if left.shape() != right.shape() {
600        return Err(MatrixError::JordanSimilarityVerificationFailed);
601    }
602    let route = certificate_route(context, left.entries.len(), source.rows);
603    let entries = (0..left.entries.len()).collect::<Vec<_>>();
604    let checks = ordered_map(route, &entries, |entry_index| {
605        let difference = crate::expression::builders::subtract(
606            context,
607            left.entries[*entry_index].clone(),
608            right.entries[*entry_index].clone(),
609        );
610        crate::simplify::simplify(context, difference)
611    });
612    if checks
613        .iter()
614        .any(|difference| !crate::matrix::expression_is_zero(context, difference))
615    {
616        return Err(MatrixError::JordanSimilarityVerificationFailed);
617    }
618    Ok(())
619}

621fn assemble_decomposition(
622    context: &Context,
623    dimension: usize,
624    blocks: Vec<CertifiedJordanBlock>,
625) -> MatrixResult<CertifiedJordanDecomposition> {
626    let zero = builders::zero(context);
627    let one = builders::one(context);
628    let mut similarity_entries = vec![zero.clone(); dimension * dimension];
629    let mut jordan_entries = vec![zero; dimension * dimension];
630    let mut public_blocks = Vec::with_capacity(blocks.len());
631    let mut column_offset = 0usize;
632    for block in &blocks {
633        for (inner_column, vector) in block.vectors.iter().enumerate() {
634            for row in 0..dimension {
635                similarity_entries[row * dimension + column_offset + inner_column] =
636                    reify_element(context, &vector[row], Some(&block.root.expression));
637            }
638            jordan_entries
639                [(column_offset + inner_column) * dimension + column_offset + inner_column] =
640                block.root.expression.clone();
641            if inner_column + 1 < block.vectors.len() {
642                jordan_entries[(column_offset + inner_column) * dimension
643                    + column_offset
644                    + inner_column
645                    + 1] = one.clone();
646            }
647        }
648        public_blocks.push(MatrixJordanBlock {
649            eigenvalue: block.root.expression.clone(),
650            size: block.vectors.len(),
651        });
652        column_offset += block.vectors.len();
653    }
654    let public = MatrixJordanDecomposition {
655        similarity: Matrix {
656            rows: dimension,
657            columns: dimension,
658            entries: similarity_entries,
659        },
660        jordan_form: Matrix {
661            rows: dimension,
662            columns: dimension,
663            entries: jordan_entries,
664        },
665        blocks: public_blocks,
666    };
667    Ok(CertifiedJordanDecomposition { public, blocks })
