15/// Performs the advanced native `integer_determinant` operation.
16pub fn integer_determinant(
17    size: usize,
18    entries: &[crate::exact_arithmetic::Integer],
19) -> crate::exact_arithmetic::Integer {
20    use crate::exact_arithmetic::{integer_one, integer_zero};
21
22    assert_eq!(entries.len(), size * size);
23    if size == 0 {
24        return integer_one();
25    }
26    let mut working = entries.to_vec();
27    let mut previous_pivot = integer_one();
28    let mut sign = integer_one();
29    for pivot_index in 0..(size - 1) {
30        let Some(chosen_row) =
31            (pivot_index..size).find(|row| working[*row * size + pivot_index] != integer_zero())
32        else {
33            return integer_zero();
34        };
35        if chosen_row != pivot_index {
36            for column in 0..size {
37                working.swap(pivot_index * size + column, chosen_row * size + column);
38            }
39            sign = -sign;
40        }
41        let pivot = working[pivot_index * size + pivot_index].clone();
42        for row in (pivot_index + 1)..size {
43            for column in (pivot_index + 1)..size {
44                let numerator = &pivot * &working[row * size + column]
45                    - &working[row * size + pivot_index] * &working[pivot_index * size + column];
46                working[row * size + column] = numerator / &previous_pivot;
47            }
48        }
49        previous_pivot = pivot;
50    }
51    sign * working[(size - 1) * size + size - 1].clone()
52}

63/// Performs the advanced native `row_reduce` operation.
64pub fn row_reduce(context: &Context, matrix_value: &Matrix) -> MatrixResult<Matrix> {
65    // Public entry point producing a row-echelon-form matrix via Bareiss
66    // fraction-free Gaussian elimination. The accompanying pivot-column
67    // list lives inside the `row_echelon_form` struct returned by the
68    // inner driver — this surface returns only the elimination result so
69    // it composes with the existing `matrix` type at call sites.
70    Ok(eliminate(context, matrix_value)?.matrix)
71}

73/// Performs the advanced native `rank` operation.
74pub fn rank(context: &Context, matrix_value: &Matrix) -> MatrixResult<usize> {
75    // Public entry point for the matrix rank: the number of pivot
76    // columns surfaced by Bareiss elimination.

80/// Performs the advanced native `nullspace_basis` operation.
81pub fn nullspace_basis(
82    context: &Context,
83    matrix_value: &Matrix,
84) -> MatrixResult<Vec<Vec<ExpressionReference>>> {
85    // Public entry point for a basis of the nullspace of the matrix.
86    // The basis is recovered by running Bareiss elimination, completing
87    // it into reduced row-echelon form via a back-elimination pass, and
88    // for each free column reading off the column-vector witness
89    // obtained by setting that free variable to one and every other
90    // free variable to zero. The returned vectors have length equal to
91    // the original column count.
92    let echelon = eliminate(context, matrix_value)?;
93    let reduced = back_eliminate(context, echelon)?;
94    let column_count = matrix_value.columns;
95    let mut pivot_row_of_column: Vec<Option<usize>> = vec![None; column_count];
96    for (pivot_index, column_index) in reduced.pivot_columns.iter().enumerate() {
97        pivot_row_of_column[*column_index] = Some(pivot_index);
98    }
99    let mut basis: Vec<Vec<ExpressionReference>> = Vec::new();
100    for free_column in 0..column_count {
101        if pivot_row_of_column[free_column].is_some() {
102            continue;
103        }
104        let mut vector: Vec<ExpressionReference> = Vec::with_capacity(column_count);
105        let zero = builders::zero(context);
106        for slot in 0..column_count {
107            if slot == free_column {
108                vector.push(builders::one(context));
109            } else {
110                vector.push(zero.clone());
111            }
112        }
113        for (pivot_index, pivot_column) in reduced.pivot_columns.iter().enumerate() {
114            let coefficient = reduced
115                .matrix
116                .entry_unchecked(pivot_index, free_column)
117                .clone();
118            let pivot_value = reduced
119                .matrix
120                .entry_unchecked(pivot_index, *pivot_column)
121                .clone();
122            let quotient = builders::divide(context, coefficient, pivot_value);
123            let negated = builders::negate(context, quotient);
124            vector[*pivot_column] = negated;
125        }
126        basis.push(vector);
127    }
128    Ok(basis)
129}

131/// Performs the advanced native `eliminate` operation.
132pub fn eliminate(context: &Context, matrix_value: &Matrix) -> MatrixResult<RowEchelonForm> {
133    // Bareiss fraction-free Gaussian elimination with deterministic
134    // partial pivoting. The classical two-step recurrence
135    // M[i, j] := (M[k, k] * M[i, j] − M[i, k] * M[k, j]) / M[k - 1, k - 1]
136    // produces an exact division at every step over any integral
137    // domain; for the general symbolic coefficient layer used here, the
138    // simplifying `divide` builder keeps the result canonical even when
139    // exact cancellation is not visible to the simplifier. Pivot
140    // selection prefers the entry with the smallest canonical
141    // expression hash among rows for which `is_nonzero` returns
142    // `truth_value::yes`. Rows whose entry is `truth_value::no` are
143    // skipped; if every remaining candidate is `truth_value::unknown`
144    // the routine aborts with `pivot_indeterminate` — no heuristic guess
145    // is fabricated.
146    let row_count = matrix_value.rows;
147    let column_count = matrix_value.columns;
148    if row_count == 0 || column_count == 0 {
149        return Err(MatrixError::EmptyMatrix);
150    }
151    let mut entries = matrix_value.entries.clone();
152    let mut previous_pivot = builders::one(context);
153    let mut pivot_row: usize = 0;
154    let mut pivot_columns: Vec<usize> = Vec::new();
155    for column_index in 0..column_count {
156        if pivot_row >= row_count {
157            break;
158        }
159        let chosen_row_option = choose_pivot_row(
160            context,
161            &mut entries,
162            row_count,
163            column_count,
164            pivot_row,
165            column_index,
166        )?;
167        let chosen_row = match chosen_row_option {
168            None => continue,
169            Some(row) => row,
170        };
171        if chosen_row != pivot_row {
172            swap_rows(&mut entries, column_count, pivot_row, chosen_row);
173        }
174        let pivot_value = entries[pivot_row * column_count + column_index].clone();
175        let lower_rows = ((pivot_row + 1)..row_count).collect::<Vec<_>>();
176        let route = elimination_route();
177        let pivot_row_entries =
178            entries[pivot_row * column_count..(pivot_row + 1) * column_count].to_vec();
179        let lower_row_inputs = lower_rows
180            .iter()
181            .map(|lower_row| {
182                entries[lower_row * column_count..(lower_row + 1) * column_count].to_vec()
183            })
184            .collect::<Vec<_>>();
185        let updated_rows = ordered_map(route, &lower_row_inputs, |lower_entries| {
186            let mut updated = lower_entries.clone();
187            for inner_column in (column_index + 1)..column_count {
188                let m_ij = lower_entries[inner_column].clone();
189                let m_ik = lower_entries[column_index].clone();
190                let m_kj = pivot_row_entries[inner_column].clone();
191                let pivot_times_entry = builders::product(context, vec![pivot_value.clone(), m_ij]);
192                let cross_product = builders::product(context, vec![m_ik, m_kj]);
193                let numerator = builders::subtract(context, pivot_times_entry, cross_product);
194                updated[inner_column] =
195                    builders::divide(context, numerator, previous_pivot.clone());
196            }
197            updated[column_index] = builders::zero(context);
198            updated
199        });
200        for (lower_row, updated) in lower_rows.into_iter().zip(updated_rows) {
201            entries[lower_row * column_count..(lower_row + 1) * column_count]
202                .clone_from_slice(&updated);
203        }
204        previous_pivot = pivot_value;
205        pivot_columns.push(column_index);
206        pivot_row += 1;
207    }
208    let result_matrix = Matrix {
209        rows: row_count,
210        columns: column_count,
211        entries,
212    };
213    Ok(RowEchelonForm {
214        matrix: result_matrix,
215        pivot_columns,
216    })
217}

219fn back_eliminate(context: &Context, echelon: RowEchelonForm) -> MatrixResult<RowEchelonForm> {
220    // Complete a Bareiss row-echelon form into reduced row-echelon form
221    // by eliminating above each pivot. The pass walks the pivot list in
222    // reverse so each elimination only touches entries already settled
223    // by previous reductions. The resulting matrix has its pivot
224    // columns zeroed everywhere except at the pivot row itself, which
225    // is what `nullspace_basis` needs in order to recover the free
226    // parameters directly.
227    let row_count = echelon.matrix.rows;
228    let column_count = echelon.matrix.columns;
229    let mut entries = echelon.matrix.entries;
230    for (pivot_index, pivot_column) in echelon.pivot_columns.iter().enumerate().rev() {
231        let pivot_value = entries[pivot_index * column_count + *pivot_column].clone();
232        let upper_rows = (0..pivot_index).collect::<Vec<_>>();
233        let route = elimination_route();
234        let pivot_row_entries =
235            entries[pivot_index * column_count..(pivot_index + 1) * column_count].to_vec();
236        let upper_row_inputs = upper_rows
237            .iter()
238            .map(|upper_row| {
239                entries[upper_row * column_count..(upper_row + 1) * column_count].to_vec()
240            })
241            .collect::<Vec<_>>();
242        let updated_rows = ordered_map(route, &upper_row_inputs, |upper_entries| {
243            let leading_entry = upper_entries[*pivot_column].clone();
244            if is_provably_zero(&leading_entry, context) {
245                return upper_entries.clone();
246            }
247            let mut updated = upper_entries.clone();
248            for inner_column in 0..column_count {
249                if inner_column == *pivot_column {
250                    updated[inner_column] = builders::zero(context);
251                    continue;
252                }
253                let upper_entry = upper_entries[inner_column].clone();
254                let lower_entry = pivot_row_entries[inner_column].clone();
255                let scaled_upper =
256                    builders::product(context, vec![pivot_value.clone(), upper_entry]);
257                let scaled_lower =
258                    builders::product(context, vec![leading_entry.clone(), lower_entry]);
259                updated[inner_column] = builders::subtract(context, scaled_upper, scaled_lower);
260            }
261            updated
262        });
263        for (upper_row, updated) in upper_rows.into_iter().zip(updated_rows) {
264            entries[upper_row * column_count..(upper_row + 1) * column_count]
265                .clone_from_slice(&updated);
266        }
267    }
268    let reduced_matrix = Matrix {
269        rows: row_count,
270        columns: column_count,
271        entries,
272    };
273    Ok(RowEchelonForm {
274        matrix: reduced_matrix,
275        pivot_columns: echelon.pivot_columns,
276    })
277}

279fn choose_pivot_row(
280    context: &Context,
281    entries: &mut [ExpressionReference],
282    row_count: usize,
283    column_count: usize,
284    pivot_row: usize,
285    column_index: usize,
286) -> MatrixResult<Option<usize>> {
287    // Walk the candidate rows beneath the current pivot row, classifying
288    // each entry as provably zero, provably nonzero, or
289    // assumption-indeterminate. When at least one candidate is provably
290    // nonzero, return the one with the smallest expression hash among
291    // those — ties are broken by the row index (smaller wins) so the
292    // result is fully deterministic. When every candidate is provably
293    // zero, return None and let the driver advance to the next column.
294    // When no candidate is provably nonzero but at least one is
295    // assumption-indeterminate, refuse the operation.
296    let mut best: Option<(u64, usize)> = None;
297    let mut saw_unknown = false;
298    for candidate_row in pivot_row..row_count {
299        let entry_index = candidate_row * column_count + column_index;
300        let initial_status = is_nonzero(&entries[entry_index], &context.symbol_table);
301        let status = if initial_status == TruthValue::Unknown {
302            let normalized = crate::simplify::simplify(context, entries[entry_index].clone());
303            let normalized_status = is_nonzero(&normalized, &context.symbol_table);
304            entries[entry_index] = normalized;
305            normalized_status
306        } else {
307            initial_status
308        };
309        match status {
310            TruthValue::Yes => {
311                let entry_hash = entries[entry_index].hash;
312                match best {
313                    None => best = Some((entry_hash, candidate_row)),
314                    Some((existing_hash, existing_row))
315                        if (entry_hash, candidate_row) < (existing_hash, existing_row) =>
316                    {
317                        best = Some((entry_hash, candidate_row));
318                    }
319                    _ => {}
320                }
321            }
322            TruthValue::No => {}
323            TruthValue::Unknown => saw_unknown = true,
324        }
325    }
326    match best {
327        Some((_, chosen)) => Ok(Some(chosen)),
328        None if saw_unknown => Err(MatrixError::PivotIndeterminate {
329            row: pivot_row,
330            column: column_index,
331        }),
332        None => Ok(None),
333    }
334}

336fn swap_rows(
337    entries: &mut [ExpressionReference],
338    column_count: usize,
339    first_row: usize,
340    second_row: usize,
341) {
342    // Swap two whole rows of the flat row-major buffer. Used by the
343    // pivoting step when the chosen pivot is not already in the current
344    // pivot row.
345    for column_index in 0..column_count {
346        entries.swap(
347            first_row * column_count + column_index,
348            second_row * column_count + column_index,
349        );
350    }
351}

353fn is_provably_zero(reference: &ExpressionReference, context: &Context) -> bool {
354    // Quick check used by the back-elimination pass to skip rows whose
355    // leading entry is already zero. Defers to the existing nonzero
356    // query and only treats `truth_value::no` as zero; an unknown
357    // verdict keeps the elimination going so any non-trivial entry is
358    // still cleared.
359    use crate::expression::ExpressionKind;
360    if let ExpressionKind::Integer(value) = &reference.kind {
361        if *value == crate::exact_arithmetic::integer_zero() {
362            return true;
363        }
364    }
365    // Fall back to the assumption query in case of symbolic zeros that
366    // the literal-integer check missed.
367    matches!(is_nonzero(reference, &context.symbol_table), TruthValue::No,)
368}
