58struct RationalMatrix {
59    rows: usize,
60    columns: usize,
61    entries: Vec<Number>,
62}
63
64impl RationalMatrix {
65    fn zero(rows: usize, columns: usize) -> Self {
66        Self {
67            rows,
68            columns,
69            entries: vec![Number::Integer(0.into()); rows * columns],
70        }
71    }
72
73    fn identity(size: usize) -> Self {
74        let mut result = Self::zero(size, size);
75        for index in 0..size {
76            result.entries[index * size + index] = Number::Integer(1.into());
77        }
78        result
79    }
80
81    fn entry(&self, row: usize, column: usize) -> &Number {
82        &self.entries[row * self.columns + column]
83    }
84
85    fn entry_mut(&mut self, row: usize, column: usize) -> &mut Number {
86        &mut self.entries[row * self.columns + column]
87    }
88}

90fn extract_rational_matrix(matrix_value: &Matrix) -> MatrixResult<RationalMatrix> {
91    if !matrix_value.is_square() {
92        return Err(MatrixError::RationalCanonicalRequiresSquare {
93            rows: matrix_value.rows,
94            columns: matrix_value.columns,
95        });
96    }
97    if matrix_value.rows == 0 {
98        return Err(MatrixError::EmptyMatrix);
99    }
100    let mut entries = Vec::with_capacity(matrix_value.entries.len());
101    for row in 0..matrix_value.rows {
102        for column in 0..matrix_value.columns {
103            entries.push(match &matrix_value.entry_unchecked(row, column).kind {
104                ExpressionKind::Integer(value) => Number::Integer(value.clone()),
105                ExpressionKind::Rational(value) => Number::Rational(value.clone()),
106                _ => return Err(MatrixError::NonrationalEntry { row, column }),
107            });
108        }
109    }
110    Ok(RationalMatrix {
111        rows: matrix_value.rows,
112        columns: matrix_value.columns,
113        entries,
114    })
115}

117fn rational_multiply(left: &RationalMatrix, right: &RationalMatrix) -> Option<RationalMatrix> {
118    if left.columns != right.rows {
119        return None;
120    }
121    let mut result = RationalMatrix::zero(left.rows, right.columns);
122    for row in 0..left.rows {
123        for column in 0..right.columns {
124            let mut value = Number::Integer(0.into());
125            for inner in 0..left.columns {
126                value = number_arithmetic::add(
127                    &value,
128                    &number_arithmetic::multiply(
129                        left.entry(row, inner),
130                        right.entry(inner, column),
131                    ),
132                );
133            }
134            *result.entry_mut(row, column) = value;
135        }
136    }
137    Some(result)
138}

140fn rational_matrix_vector_multiply(value: &RationalMatrix, vector: &[Number]) -> Vec<Number> {
141    (0..value.rows)
142        .map(|row| {
143            (0..value.columns).fold(Number::Integer(0.into()), |sum, column| {
144                number_arithmetic::add(
145                    &sum,
146                    &number_arithmetic::multiply(value.entry(row, column), &vector[column]),
147                )
148            })
149        })
150        .collect()
151}

153fn rational_rank(value: &RationalMatrix) -> usize {
154    let mut working = value.clone();
155    let mut rank = 0;
156    for column in 0..working.columns {
157        let Some(pivot_row) = (rank..working.rows)
158            .find(|row| !number_arithmetic::is_zero(working.entry(*row, column)))
159        else {
160            continue;
161        };
162        for entry_column in 0..working.columns {
163            working.entries.swap(
164                rank * working.columns + entry_column,
165                pivot_row * working.columns + entry_column,
166            );
167        }
168        let pivot = working.entry(rank, column).clone();
169        for row in rank + 1..working.rows {
170            if number_arithmetic::is_zero(working.entry(row, column)) {
171                continue;
172            }
173            let factor = number_arithmetic::divide(working.entry(row, column), &pivot).unwrap();
174            for target_column in column..working.columns {
175                let replacement = number_arithmetic::subtract(
176                    working.entry(row, target_column),
177                    &number_arithmetic::multiply(&factor, working.entry(rank, target_column)),
178                );
179                *working.entry_mut(row, target_column) = replacement;
180            }
181        }
182        rank += 1;
183        if rank == working.rows {
184            break;
185        }
186    }
187    rank
188}

190fn polynomial_source(variable: SymbolIdentifier, value: &RationalMatrix) -> PolynomialMatrix {
191    let mut entries = Vec::with_capacity(value.rows * value.columns);
192    for row in 0..value.rows {
193        for column in 0..value.columns {
194            let constant = number_arithmetic::negate(value.entry(row, column));
195            let coefficients = if row == column {
196                vec![constant, Number::Integer(1.into())]
197            } else {
198                vec![constant]
199            };
200            entries.push(univariate_from_dense(variable, coefficients));
201        }
202    }
203    PolynomialMatrix::new(value.rows, value.columns, entries)
204}

206fn polynomial_coefficients(value: &Polynomial) -> Option<Vec<Number>> {
207    let maximum_degree = degree(value).unwrap_or(0);
208    let mut coefficients = vec![Number::Integer(0.into()); maximum_degree + 1];
209    for (exponent, coefficient) in &value.terms {
210        coefficients[*exponent] = coefficient_to_number(coefficient)?;
211    }
212    Some(coefficients)
213}

215fn product_of_polynomials(
216    context: &Context,
217    variable: SymbolIdentifier,
218    values: &[Polynomial],
219) -> Polynomial {
220    let route = decomposition_route(context, values.len(), values.len().saturating_mul(128));
221    deterministic_reduce(route, values.to_vec(), |left, right| {
222        multiply(&left, &right)
223    })
224    .unwrap_or_else(|| univariate_from_dense(variable, vec![Number::Integer(1.into())]))
225}

227fn berkowitz_polynomial(
228    context: &Context,
229    matrix_value: &Matrix,
230    variable: SymbolIdentifier,
231) -> MatrixResult<Polynomial> {
232    let coefficients = characteristic_polynomial_coefficients(context, matrix_value)?;
233    let numeric = coefficients
234        .into_iter()
235        .map(|value| match &value.kind {
236            ExpressionKind::Integer(integer_value) => Ok(Number::Integer(integer_value.clone())),
237            ExpressionKind::Rational(rational_value) => {
238                Ok(Number::Rational(rational_value.clone()))
239            }
240            _ => Err(MatrixError::PolynomialSmithVerificationFailed),
241        })
242        .collect::<MatrixResult<Vec<_>>>()?;
243    Ok(univariate_from_dense(variable, numeric))
244}

246fn companion_matrix(factors: &[Polynomial]) -> MatrixResult<RationalMatrix> {
247    let size: usize = factors.iter().map(|value| degree(value).unwrap_or(0)).sum();
248    let mut result = RationalMatrix::zero(size, size);
249    let mut offset = 0;
250    for factor in factors {
251        let coefficients = polynomial_coefficients(factor)
252            .ok_or(MatrixError::PolynomialSmithVerificationFailed)?;
253        let block_size = degree(factor).unwrap_or(0);
254        for index in 0..block_size.saturating_sub(1) {
255            *result.entry_mut(offset + index + 1, offset + index) = Number::Integer(1.into());
256        }
257        for row in 0..block_size {
258            *result.entry_mut(offset + row, offset + block_size - 1) =
259                number_arithmetic::negate(&coefficients[row]);
260        }
261        offset += block_size;
262    }
263    Ok(result)
264}

266fn evaluate_polynomial_vector_at_matrix(
267    matrix_value: &RationalMatrix,
268    polynomial_vector: &[Polynomial],
269) -> MatrixResult<Vec<Number>> {
270    let maximum_degree = polynomial_vector
271        .iter()
272        .filter_map(degree)
273        .max()
274        .unwrap_or(0);
275    let mut result = vec![Number::Integer(0.into()); matrix_value.rows];
276    let mut matrix_power = RationalMatrix::identity(matrix_value.rows);
277    for exponent in 0..=maximum_degree {
278        let coefficient_vector = polynomial_vector
279            .iter()
280            .map(|value| {
281                value
282                    .terms
283                    .iter()
284                    .find(|(degree_value, _)| *degree_value == exponent)
285                    .and_then(|(_, coefficient)| coefficient_to_number(coefficient))
286                    .unwrap_or_else(|| Number::Integer(0.into()))
287            })
288            .collect::<Vec<_>>();
289        let contribution = rational_matrix_vector_multiply(&matrix_power, &coefficient_vector);
290        for index in 0..result.len() {
291            result[index] = number_arithmetic::add(&result[index], &contribution[index]);
292        }
293        matrix_power = rational_multiply(&matrix_power, matrix_value)
294            .ok_or(MatrixError::RationalCanonicalVerificationFailed)?;
295    }
296    Ok(result)
297}

299fn similarity_witness(
300    context: &Context,
301    matrix_value: &RationalMatrix,
302    certificate: &PolynomialSmithCertificate,
303    factors: &[(usize, Polynomial)],
304) -> MatrixResult<RationalMatrix> {
305    let route = decomposition_route(
306        context,
307        factors.len(),
308        matrix_value.rows.saturating_mul(256),
309    );
310    let factor_columns = ordered_map(route, factors, |(diagonal_index, factor)| {
311        similarity_columns_for_factor(matrix_value, certificate, *diagonal_index, factor)
312    });
313    let mut columns = Vec::with_capacity(matrix_value.rows);
314    for factor_column in factor_columns {
315        columns.extend(factor_column?);
316    }
317    if columns.len() != matrix_value.rows {
318        return Err(MatrixError::RationalCanonicalVerificationFailed);
319    }
320    let mut result = RationalMatrix::zero(matrix_value.rows, matrix_value.rows);
321    for (column, vector) in columns.into_iter().enumerate() {
322        for row in 0..matrix_value.rows {
323            *result.entry_mut(row, column) = vector[row].clone();
324        }
325    }
326    Ok(result)
327}

329fn evaluate_polynomial_at_matrix(
330    value: &Polynomial,
331    matrix_value: &RationalMatrix,
332) -> MatrixResult<RationalMatrix> {
333    let coefficients =
334        polynomial_coefficients(value).ok_or(MatrixError::RationalCanonicalVerificationFailed)?;
335    let mut result = RationalMatrix::zero(matrix_value.rows, matrix_value.columns);
336    for coefficient in coefficients.iter().rev() {
337        result = rational_multiply(&result, matrix_value)
338            .ok_or(MatrixError::RationalCanonicalVerificationFailed)?;
339        for index in 0..matrix_value.rows {
340            let updated = number_arithmetic::add(result.entry(index, index), coefficient);
341            *result.entry_mut(index, index) = updated;
342        }
343    }
344    Ok(result)
345}

358fn polynomial_expression(
359    context: &Context,
360    value: &Polynomial,
361    variable: SymbolIdentifier,
362) -> ExpressionReference {
363    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
364    let mut terms = Vec::with_capacity(value.terms.len());
365    for (exponent, coefficient) in value.terms.iter().rev() {
366        let numeric = coefficient_to_number(coefficient).unwrap();
367        let coefficient_expression = number_expression(context, &numeric);
368        let term = if *exponent == 0 {
369            coefficient_expression
370        } else {
371            let exponent_expression = builders::integer_value(context, *exponent as i64);
372            let power = builders::power(context, variable_expression.clone(), exponent_expression);
373            builders::product(context, vec![coefficient_expression, power])
374        };
375        terms.push(term);
376    }
377    builders::sum(context, terms)
378}

392fn compute(
393    context: &Context,
394    matrix_value: &Matrix,
395    variable: SymbolIdentifier,
396    maximum_coefficient_bits: Option<u64>,
397) -> MatrixResult<MatrixRationalCanonicalForm> {
398    let rational_value = extract_rational_matrix(matrix_value)?;
399    let source = polynomial_source(variable, &rational_value);
400    let certificate = crate::matrix::polynomial_smith::smith_form_with_route(
401        &source,
402        maximum_coefficient_bits,
403        decomposition_route(
404            context,
405            rational_value.rows.saturating_mul(rational_value.columns),
406            rational_value
407                .rows
408                .saturating_mul(rational_value.columns)
409                .saturating_mul(256),
410        ),
411    )?;
412    let all_factors = diagonal_factors(&certificate);
413    let indexed_factors = all_factors
414        .into_iter()
415        .enumerate()
416        .filter(|(_, value)| degree(value).is_some_and(|factor_degree| factor_degree >= 1))
417        .collect::<Vec<_>>();
418    if indexed_factors.is_empty() {
419        return Err(MatrixError::PolynomialSmithVerificationFailed);
420    }
421    let factors = indexed_factors
422        .iter()
423        .map(|(_, value)| value.clone())
424        .collect::<Vec<_>>();
425    let characteristic = product_of_polynomials(context, variable, &factors);
426    if characteristic != berkowitz_polynomial(context, matrix_value, variable)? {
427        return Err(MatrixError::PolynomialSmithVerificationFailed);
428    }
429    let canonical = companion_matrix(&factors)?;
430    let similarity = similarity_witness(context, &rational_value, &certificate, &indexed_factors)?;
431    let minimal = factors.last().unwrap().clone();
432    let (similarity_rank, source_times_similarity) = join(
433        decomposition_route(context, 2, rational_value.rows.saturating_mul(512)),
434        || rational_rank(&similarity),
435        || rational_multiply(&rational_value, &similarity),
436    );
437    let similarity_times_canonical = rational_multiply(&similarity, &canonical);
438    let minimal_value = evaluate_polynomial_at_matrix(&minimal, &rational_value)?;
439    let full_rank = similarity_rank == rational_value.rows;
440    let intertwining = source_times_similarity == similarity_times_canonical;
441    let annihilation =
442        minimal_value == RationalMatrix::zero(rational_value.rows, rational_value.columns);
443    if !full_rank || !intertwining || !annihilation {
444        return Err(MatrixError::RationalCanonicalVerificationFailed);
445    }
446    let invariant_factors = factors
447        .iter()
448        .map(|value| polynomial_expression(context, value, variable))
449        .collect();
450    let minimal_polynomial = polynomial_expression(context, &minimal, variable);
451    let characteristic_polynomial = polynomial_expression(context, &characteristic, variable);
452    Ok(MatrixRationalCanonicalForm {
453        canonical_form: expression_matrix(context, &canonical),
454        similarity: expression_matrix(context, &similarity),
455        invariant_factors,
456        minimal_polynomial,
457        characteristic_polynomial,
458    })
459}

461fn similarity_columns_for_factor(
462    matrix_value: &RationalMatrix,
463    certificate: &PolynomialSmithCertificate,
464    diagonal_index: usize,
465    factor: &Polynomial,
466) -> MatrixResult<Vec<Vec<Number>>> {
467    let polynomial_vector = (0..matrix_value.rows)
468        .map(|row| {
469            certificate
470                .inverse_row_transformation
471                .entry(row, diagonal_index)
472                .clone()
473        })
474        .collect::<Vec<_>>();
475    let mut columns = Vec::with_capacity(degree(factor).unwrap_or(0));
476    let mut vector = evaluate_polynomial_vector_at_matrix(matrix_value, &polynomial_vector)?;
477    for _ in 0..degree(factor).unwrap_or(0) {
478        columns.push(vector.clone());
479        vector = rational_matrix_vector_multiply(matrix_value, &vector);
480    }
481    Ok(columns)
482}

484fn decomposition_route(
485    context: &Context,
486    independent_unit_count: usize,
487    domain_work_units: usize,
488) -> ExecutionRoute {
489    if !context
490        .operation()
491        .may_fork_work_class("matrix_decomposition")
492    {
493        return ExecutionRoute::Serial;
494    }
495    let worker_count = crate::parallelism::maximum_operation_concurrency();
496    let requested_permits = independent_unit_count.min(worker_count.max(1));
497    context
498        .operation()
499        .execution_decision(
500            &WorkEstimate::new(
501                domain_work_units as u64,
502                independent_unit_count,
503                domain_work_units.saturating_mul(8),
504                0,
505                requested_permits,
506                "matrix_decomposition",
507            ),
508            requested_permits,
509            crate::parallelism::runtime_memory_capacity(domain_work_units.saturating_mul(8)),
510        )
511        .route
512}
