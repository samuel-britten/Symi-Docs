26/// Performs the advanced native `expand_complex` operation.
27pub fn expand_complex(context: &Context, target: &ExpressionReference) -> ExpressionReference {
28    let rewritten = rewrite_directed_moduli(context, target, DIRECTED_MODULUS_RECURSION_BUDGET);
29    if !Arc::ptr_eq(&rewritten, target)
30        && crate::assumptions::range_evidence::proves_real(&context.symbol_table, target)
31    {
32        return rewritten;
33    }
34    let (real_component, imaginary_component) = expanded_components(context, &rewritten);
35    let imaginary_unit = context
36        .interner
37        .intern(ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit));
38    let imaginary_term = simplify_product(context, vec![imaginary_unit, imaginary_component]);
39    simplify_sum(context, vec![real_component, imaginary_term])
40}

42pub(crate) fn exact_cartesian_modulus(
43    context: &Context,
44    argument: &ExpressionReference,
45) -> Option<ExpressionReference> {
46    if node_count(argument) > WRAPPER_FREE_COMPONENT_NODE_BUDGET {
47        return None;
48    }
49    let (real_component, imaginary_component) = wrapper_free_components(context, argument)?;
50    let two = integer(context, 2);
51    let real_square = simplify_power(context, real_component, Arc::clone(&two));
52    let imaginary_square = simplify_power(context, imaginary_component, two);
53    let squared_norm = simplify_sum(context, vec![real_square, imaginary_square]);
54    if node_count(&squared_norm) > WRAPPER_FREE_COMPONENT_NODE_BUDGET {
55        return None;
56    }
57    let expanded_norm = expand(context, &squared_norm);
58    let reduced_norm = crate::simplify::rational::cancel(context, &expanded_norm);
59    if node_count(&reduced_norm) > WRAPPER_FREE_COMPONENT_NODE_BUDGET {
60        return None;
61    }
62    let half = crate::expression::builders::rational(context, Integer::from(1), Integer::from(2));
63    let result = simplify_power(context, reduced_norm, half);
64    (node_count(&result) <= WRAPPER_FREE_COMPONENT_NODE_BUDGET).then_some(result)
65}

67fn rewrite_directed_moduli(
68    context: &Context,
69    target: &ExpressionReference,
70    remaining_depth: usize,
71) -> ExpressionReference {
72    if remaining_depth == 0 || node_count(target) > WRAPPER_FREE_COMPONENT_NODE_BUDGET {
73        return Arc::clone(target);
74    }
75    match &target.kind {
76        ExpressionKind::Sum(children) => {
77            let rewritten = children
78                .iter()
79                .map(|child| rewrite_directed_moduli(context, child, remaining_depth - 1))
80                .collect();
81            simplify_sum(context, rewritten)
82        }
83        ExpressionKind::Product(children) => {
84            let rewritten = children
85                .iter()
86                .map(|child| rewrite_directed_moduli(context, child, remaining_depth - 1))
87                .collect();
88            simplify_product(context, rewritten)
89        }
90        ExpressionKind::Power(base, exponent) => {
91            let rewritten_base = rewrite_directed_moduli(context, base, remaining_depth - 1);
92            let rewritten_exponent =
93                rewrite_directed_moduli(context, exponent, remaining_depth - 1);
94            simplify_power(context, rewritten_base, rewritten_exponent)
95        }
96        ExpressionKind::Factorial(argument) => {
97            let rewritten = rewrite_directed_moduli(context, argument, remaining_depth - 1);
98            crate::expression::builders::factorial(context, rewritten)
99        }
100        ExpressionKind::FunctionCall(function, arguments) => {
101            let rewritten_arguments = arguments
102                .iter()
103                .map(|argument| rewrite_directed_moduli(context, argument, remaining_depth - 1))
104                .collect::<Vec<_>>();
105            if *function == FunctionIdentifier::Builtin(BuiltinFunction::Abs)
106                && rewritten_arguments.len() == 1
107            {
108                let argument = &rewritten_arguments[0];
109                if !crate::assumptions::range_evidence::proves_real(&context.symbol_table, argument)
110                {
111                    if let ExpressionKind::Power(base, exponent) = &argument.kind {
112                        if matches!(exponent.kind, ExpressionKind::Integer(_)) {
113                            if let Some(base_modulus) = exact_cartesian_modulus(context, base) {
114                                return simplify_power(context, base_modulus, Arc::clone(exponent));
115                            }
116                        }
117                    }
118                    if let Some(modulus) = exact_cartesian_modulus(context, argument) {
119                        return modulus;
120                    }
121                }
122            }
123            simplify_function_call(context, *function, rewritten_arguments)
124        }
125        _ => Arc::clone(target),
126    }
127}

139fn expanded_components(
140    context: &Context,
141    target: &ExpressionReference,
142) -> (ExpressionReference, ExpressionReference) {
143    let (real_component, imaginary_component) = decompose(context, target);
144    let real_component = expand_wrapper_free_component(context, real_component);
145    let imaginary_component = expand_wrapper_free_component(context, imaginary_component);
146    (real_component, imaginary_component)
147}

149fn expand_wrapper_free_component(
150    context: &Context,
151    component: ExpressionReference,
152) -> ExpressionReference {
153    if contains_component_wrapper(&component) {
154        component
155    } else {
156        expand(context, &component)
157    }
158}

160fn contains_component_wrapper(target: &ExpressionReference) -> bool {
161    match &target.kind {
162        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(function), arguments) => {
163            matches!(
164                function,
165                BuiltinFunction::RealPart | BuiltinFunction::ImaginaryPart
166            ) || arguments.iter().any(contains_component_wrapper)
167        }
168        ExpressionKind::FunctionCall(_, arguments)
169        | ExpressionKind::Derivative(_, _, arguments) => {
170            arguments.iter().any(contains_component_wrapper)
171        }
172        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
173            children.iter().any(contains_component_wrapper)
174        }
175        ExpressionKind::Power(base, exponent) => {
176            contains_component_wrapper(base) || contains_component_wrapper(exponent)
177        }
178        ExpressionKind::Factorial(argument) => contains_component_wrapper(argument),
179        _ => false,
180    }
181}

183pub(crate) fn decompose(
184    context: &Context,
185    target: &ExpressionReference,
186) -> (ExpressionReference, ExpressionReference) {
187    if component_is_provably_real(context, target) {
188        return (target.clone(), integer(context, 0));
189    }
190    match &target.kind {
191        ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit) => {
192            (integer(context, 0), integer(context, 1))
193        }
194        ExpressionKind::Sum(terms) => decompose_sum(context, terms),
195        ExpressionKind::Product(factors) => decompose_product(context, factors),
196        ExpressionKind::Power(base, exponent) => decompose_power(context, target, base, exponent),
197        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(function), arguments)
198            if arguments.len() == 1 =>
199        {
200            decompose_function(context, target, *function, &arguments[0])
201        }
202        _ => symbolic_components(context, target),
203    }
204}

206pub(crate) fn wrapper_free_components(
207    context: &Context,
208    target: &ExpressionReference,
209) -> Option<(ExpressionReference, ExpressionReference)> {
210    if node_count(target) > WRAPPER_FREE_COMPONENT_NODE_BUDGET {
211        return None;
212    }
213    let (real_component, imaginary_component) = decompose_wrapper_free(context, target);
214    if contains_component_wrapper(&real_component)
215        || contains_component_wrapper(&imaginary_component)
216        || !component_is_provably_real(context, &real_component)
217        || !component_is_provably_real(context, &imaginary_component)
218        || component_size(&(real_component.clone(), imaginary_component.clone()))
219            > WRAPPER_FREE_COMPONENT_NODE_BUDGET
220    {
221        return None;
222    }
223    Some((real_component, imaginary_component))
224}

226fn decompose_wrapper_free(
227    context: &Context,
228    target: &ExpressionReference,
229) -> (ExpressionReference, ExpressionReference) {
230    if component_is_provably_real(context, target) {
231        return (Arc::clone(target), integer(context, 0));
232    }
233    match &target.kind {
234        ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit) => {
235            (integer(context, 0), integer(context, 1))
236        }
237        ExpressionKind::Sum(terms) => {
238            let mut real_terms = Vec::with_capacity(terms.len());
239            let mut imaginary_terms = Vec::with_capacity(terms.len());
240            for term in terms {
241                let (real_component, imaginary_component) = decompose_wrapper_free(context, term);
242                real_terms.push(real_component);
243                imaginary_terms.push(imaginary_component);
244            }
245            (
246                simplify_sum(context, real_terms),
247                simplify_sum(context, imaginary_terms),
248            )
249        }
250        ExpressionKind::Product(factors) => {
251            let mut result = (integer(context, 1), integer(context, 0));
252            for factor in factors {
253                let factor_components = decompose_wrapper_free(context, factor);
254                result = multiply_components(context, result, factor_components);
255            }
256            result
257        }
258        ExpressionKind::Power(_, _) => {
259            let (real_component, imaginary_component) = decompose(context, target);
260            if contains_component_wrapper(&real_component)
261                || contains_component_wrapper(&imaginary_component)
262            {
263                symbolic_components(context, target)
264            } else {
265                (real_component, imaginary_component)
266            }
267        }
268        _ => decompose(context, target),
269    }
270}

272fn component_is_provably_real(context: &Context, target: &ExpressionReference) -> bool {
273    crate::assumptions::range_evidence::proves_real(&context.symbol_table, target)
274        || is_nonnegative(target, context) == TruthValue::Yes
275        || matches!(
276            &target.kind,
277            ExpressionKind::Power(base, exponent)
278                if is_nonnegative(base, context) == TruthValue::Yes
279                    && crate::assumptions::range_evidence::proves_real(
280                        &context.symbol_table,
281                        exponent,
282                    )
283        )
284        || crate::exact_real_order::sign_of_real_expression(target, &context.symbol_table)
285            != SignDecision::Unknown
286}

288fn decompose_sum(
289    context: &Context,
290    terms: &[ExpressionReference],
291) -> (ExpressionReference, ExpressionReference) {
292    let mut real_terms = Vec::with_capacity(terms.len());
293    let mut imaginary_terms = Vec::with_capacity(terms.len());
294    for term in terms {
295        let (real_component, imaginary_component) = decompose(context, term);
296        real_terms.push(real_component);
297        imaginary_terms.push(imaginary_component);
298    }
299    (
300        simplify_sum(context, real_terms),
301        simplify_sum(context, imaginary_terms),
302    )
303}

305fn decompose_product(
306    context: &Context,
307    factors: &[ExpressionReference],
308) -> (ExpressionReference, ExpressionReference) {
309    let mut result = (integer(context, 1), integer(context, 0));
310    for factor in factors {
311        let factor_components = decompose(context, factor);
312        result = multiply_components(context, result, factor_components);
313    }
314    result
315}

317fn decompose_power(
318    context: &Context,
319    target: &ExpressionReference,
320    base: &ExpressionReference,
321    exponent: &ExpressionReference,
322) -> (ExpressionReference, ExpressionReference) {
323    let ExpressionKind::Integer(exponent_value) = &exponent.kind else {
324        return symbolic_components(context, target);
325    };
326    let Some(signed_exponent) = integer_to_i64(exponent_value) else {
327        return symbolic_components(context, target);
328    };
329    let Some(magnitude) = signed_exponent.checked_abs().map(|value| value as u64) else {
330        return symbolic_components(context, target);
331    };
332    let base_components = decompose(context, base);
333    let Some(mut powered) = power_components(context, base_components, magnitude) else {
334        return symbolic_components(context, target);
335    };
336    if signed_exponent < 0 {
337        let two = integer(context, 2);
338        let real_square = simplify_power(context, powered.0.clone(), two.clone());
339        let imaginary_square = simplify_power(context, powered.1.clone(), two);
340        let modulus_square = simplify_sum(context, vec![real_square, imaginary_square]);
341        if is_nonzero(&modulus_square, context) != TruthValue::Yes {
342            return symbolic_components(context, target);
343        }
344        let negative_one = integer(context, -1);
345        let reciprocal = simplify_power(context, modulus_square, negative_one.clone());
346        let negative_imaginary = simplify_product(context, vec![negative_one, powered.1.clone()]);
347        powered = (
348            simplify_product(context, vec![powered.0, reciprocal.clone()]),
349            simplify_product(context, vec![negative_imaginary, reciprocal]),
350        );
351    }
352    powered
353}

355fn power_components(
356    context: &Context,
357    mut base: (ExpressionReference, ExpressionReference),
358    mut exponent: u64,
359) -> Option<(ExpressionReference, ExpressionReference)> {
360    let mut result = (integer(context, 1), integer(context, 0));
361    while exponent > 0 {
362        if exponent & 1 == 1 {
363            result = multiply_components(context, result, base.clone());
364            if component_size(&result) > COMPONENT_NODE_BUDGET {
365                return None;
366            }
367        }
368        exponent >>= 1;
369        if exponent > 0 {
370            base = multiply_components(context, base.clone(), base);
371            if component_size(&base) > COMPONENT_NODE_BUDGET {
372                return None;
373            }
374        }
375    }
376    Some(result)
377}

379fn multiply_components(
380    context: &Context,
381    left: (ExpressionReference, ExpressionReference),
382    right: (ExpressionReference, ExpressionReference),
383) -> (ExpressionReference, ExpressionReference) {
384    let real_positive = simplify_product(context, vec![left.0.clone(), right.0.clone()]);
385    let negative_one = integer(context, -1);
386    let real_negative =
387        simplify_product(context, vec![negative_one, left.1.clone(), right.1.clone()]);
388    let imaginary_left = simplify_product(context, vec![left.0, right.1]);
389    let imaginary_right = simplify_product(context, vec![left.1, right.0]);
390    (
391        simplify_sum(context, vec![real_positive, real_negative]),
392        simplify_sum(context, vec![imaginary_left, imaginary_right]),
393    )
394}

396fn decompose_function(
397    context: &Context,
398    target: &ExpressionReference,
399    function: BuiltinFunction,
400    argument: &ExpressionReference,
401) -> (ExpressionReference, ExpressionReference) {
402    if function == BuiltinFunction::RealPart || function == BuiltinFunction::ImaginaryPart {
403        return (target.clone(), integer(context, 0));
404    }
405    if function == BuiltinFunction::Abs {
406        return (target.clone(), integer(context, 0));
407    }
408    let (real_argument, imaginary_argument) = decompose(context, argument);
409    let call = |context: &Context, function, argument| {
410        simplify_function_call(
411            context,
412            FunctionIdentifier::Builtin(function),
413            vec![argument],
414        )
415    };
416    match function {
417        BuiltinFunction::Exp => {
418            let exponential = call(context, BuiltinFunction::Exp, real_argument);
419            let cosine = call(context, BuiltinFunction::Cos, imaginary_argument.clone());
420            let sine = call(context, BuiltinFunction::Sin, imaginary_argument);
421            (
422                simplify_product(context, vec![exponential.clone(), cosine]),
423                simplify_product(context, vec![exponential, sine]),
424            )
425        }
426        BuiltinFunction::Sin => {
427            let sine = call(context, BuiltinFunction::Sin, real_argument.clone());
428            let cosine = call(context, BuiltinFunction::Cos, real_argument);
429            let hyperbolic_cosine =
430                call(context, BuiltinFunction::Cosh, imaginary_argument.clone());
431            let hyperbolic_sine = call(context, BuiltinFunction::Sinh, imaginary_argument);
432            (
433                simplify_product(context, vec![sine, hyperbolic_cosine]),
434                simplify_product(context, vec![cosine, hyperbolic_sine]),
435            )
436        }
437        BuiltinFunction::Cos => {
438            let sine = call(context, BuiltinFunction::Sin, real_argument.clone());
439            let cosine = call(context, BuiltinFunction::Cos, real_argument);
440            let hyperbolic_cosine =
441                call(context, BuiltinFunction::Cosh, imaginary_argument.clone());
442            let hyperbolic_sine = call(context, BuiltinFunction::Sinh, imaginary_argument);
443            let negative_one = integer(context, -1);
444            let negative_sine = simplify_product(context, vec![negative_one, sine]);
445            (
446                simplify_product(context, vec![cosine, hyperbolic_cosine]),
447                simplify_product(context, vec![negative_sine, hyperbolic_sine]),
448            )
449        }
450        BuiltinFunction::Sinh => {
451            let hyperbolic_sine = call(context, BuiltinFunction::Sinh, real_argument.clone());
452            let hyperbolic_cosine = call(context, BuiltinFunction::Cosh, real_argument);
453            let cosine = call(context, BuiltinFunction::Cos, imaginary_argument.clone());
454            let sine = call(context, BuiltinFunction::Sin, imaginary_argument);
455            (
456                simplify_product(context, vec![hyperbolic_sine, cosine]),
457                simplify_product(context, vec![hyperbolic_cosine, sine]),
458            )
459        }
460        BuiltinFunction::Cosh => {
461            let hyperbolic_sine = call(context, BuiltinFunction::Sinh, real_argument.clone());
462            let hyperbolic_cosine = call(context, BuiltinFunction::Cosh, real_argument);
463            let cosine = call(context, BuiltinFunction::Cos, imaginary_argument.clone());
464            let sine = call(context, BuiltinFunction::Sin, imaginary_argument);
465            (
466                simplify_product(context, vec![hyperbolic_cosine, cosine]),
467                simplify_product(context, vec![hyperbolic_sine, sine]),
468            )
469        }
470        _ => symbolic_components(context, target),
471    }
472}

474fn symbolic_components(
475    context: &Context,
476    target: &ExpressionReference,
477) -> (ExpressionReference, ExpressionReference) {
478    let real_component = context.interner.intern(ExpressionKind::FunctionCall(
479        FunctionIdentifier::Builtin(BuiltinFunction::RealPart),
480        vec![target.clone()],
481    ));
482    let imaginary_component = context.interner.intern(ExpressionKind::FunctionCall(
483        FunctionIdentifier::Builtin(BuiltinFunction::ImaginaryPart),
484        vec![target.clone()],
485    ));
486    (real_component, imaginary_component)
487}

489fn component_size(components: &(ExpressionReference, ExpressionReference)) -> usize {
490    node_count(&components.0).saturating_add(node_count(&components.1))
491}
