1// The DistributionIdentifier enum names every first-tier
2// univariate distribution currently shipped and the per-identifier parameter-arity
3// table used by simplify_distribution to validate parameter counts at AST
4// construction time. The parameter ordering pinned here is the contract that
5// the descriptor table (statistics/descriptor.rs), the interactive parser, and
6// the test corpus all consult; any future-round distribution addition is an
7// additive enum variant accompanied by a new parameter-arity entry.
8
9#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
10/// Public native enumeration named `DistributionIdentifier`.
11pub enum DistributionIdentifier {
12    // Continuous first tier
13    /// Represents the `Normal` case.
14    Normal,
15    /// Represents the `Exponential` case.
16    Exponential,
17    /// Represents the `Uniform` case.
18    Uniform,
19    /// Represents the `Gamma` case.
20    Gamma,
21    /// Represents the `Beta` case.
22    Beta,
23    /// Represents the `Lognormal` case.
24    Lognormal,
25    /// Represents the `StudentT` case.
26    StudentT,
27    /// Represents the `Cauchy` case.
28    Cauchy,
29    // Named Gamma special cases
30    /// Represents the `ChiSquared` case.
31    ChiSquared,
32    /// Represents the `Erlang` case.
33    Erlang,
34    // Discrete first tier
35    /// Represents the `Bernoulli` case.
36    Bernoulli,
37    /// Represents the `Binomial` case.
38    Binomial,
39    /// Represents the `Geometric` case.
40    Geometric,
41    /// Represents the `Poisson` case.
42    Poisson,
43    /// Represents the `NegativeBinomial` case.
44    NegativeBinomial,
45}

47/// Performs the advanced native `parameter_arity` operation.
48pub fn parameter_arity(identifier: DistributionIdentifier) -> usize {
49    // Return the number of positional parameters carried by the given
50    // distribution. Consulted by simplify_distribution to reject malformed AST
51    // construction at the earliest possible point; consulted by the parser to
52    // validate REPL-form distribution() calls before they reach the simplifier.
53    use DistributionIdentifier::*;
54    match identifier {
55        Normal => 2,
56        Exponential => 1,
57        Uniform => 2,
58        Gamma => 2,
59        Beta => 2,
60        Lognormal => 2,
61        StudentT => 1,
62        Cauchy => 2,
63        ChiSquared => 1,
64        Erlang => 2,
65        Bernoulli => 1,
66        Binomial => 2,
67        Geometric => 1,
68        Poisson => 1,
69        NegativeBinomial => 2,
70    }
71}

73/// Performs the advanced native `parameter_names` operation.
74pub fn parameter_names(identifier: DistributionIdentifier) -> &'static [&'static str] {
75    // Return the human-readable parameter names for the given distribution in
76    // canonical order. Consulted by display.rs and latex.rs only; the
77    // implementations of the descriptor table use positional access into the
78    // distribution(...) node so renaming a parameter here is a display-only
79    // change.
80    use DistributionIdentifier::*;
81    match identifier {
82        Normal => &["mean", "standard_deviation"],
83        Exponential => &["rate"],
84        Uniform => &["lower_bound", "upper_bound"],
85        Gamma => &["shape", "rate"],
86        Beta => &["shape_alpha", "shape_beta"],
87        Lognormal => &["location", "scale"],
88        StudentT => &["degrees_of_freedom"],
89        Cauchy => &["location", "scale"],
90        ChiSquared => &["degrees_of_freedom"],
91        Erlang => &["shape", "rate"],
92        Bernoulli => &["success_probability"],
93        Binomial => &["trials", "success_probability"],
94        Geometric => &["success_probability"],
95        Poisson => &["rate"],
96        NegativeBinomial => &["failure_count", "success_probability"],
97    }
98}

100/// Performs the advanced native `distribution_name` operation.
101pub fn distribution_name(identifier: DistributionIdentifier) -> &'static str {
102    // The textual name used by display.rs (REPL output, structural-equality
103    // identifiers) and consumed by serialize/parse so a round-trip from
104    // serialise() back to the AST recovers the same identifier. Snake_case
105    // matches the project's "complete English words" identifier rule.
106    use DistributionIdentifier::*;
107    match identifier {
108        Normal => "normal",
109        Exponential => "exponential",
110        Uniform => "uniform",
111        Gamma => "gamma_distribution",
112        Beta => "beta_distribution",
113        Lognormal => "lognormal",
114        StudentT => "student_t",
115        Cauchy => "cauchy",
116        ChiSquared => "chi_squared",
117        Erlang => "erlang",
118        Bernoulli => "bernoulli",
119        Binomial => "binomial",
120        Geometric => "geometric",
121        Poisson => "poisson",

126/// Performs the advanced native `distribution_identifier_from_name` operation.
127pub fn distribution_identifier_from_name(name: &str) -> Option<DistributionIdentifier> {
128    // Parser-side inverse of distribution_name. Returns None if the supplied
129    // text does not match any first-tier distribution. The parser uses this to
130    // route a call-form like `normal(0, 1)` through the distribution builder
131    // rather than the user-function arm.
132    use DistributionIdentifier::*;
133    match name {
134        "normal" => Some(Normal),
135        "exponential" => Some(Exponential),
136        "uniform" => Some(Uniform),
137        "gamma_distribution" => Some(Gamma),
138        "beta_distribution" => Some(Beta),
139        "lognormal" => Some(Lognormal),
140        "student_t" => Some(StudentT),
141        "cauchy" => Some(Cauchy),
142        "chi_squared" => Some(ChiSquared),
143        "erlang" => Some(Erlang),
144        "bernoulli" => Some(Bernoulli),
145        "binomial" => Some(Binomial),
146        "geometric" => Some(Geometric),
147        "poisson" => Some(Poisson),
148        "negative_binomial" => Some(NegativeBinomial),
149        _ => None,
150    }
151}

153/// Performs the advanced native `is_continuous` operation.
154pub fn is_continuous(identifier: DistributionIdentifier) -> bool {
155    // Predicate consulted by the expectation engine and the support dispatcher
156    // to choose integration over summation. The discrete identifiers carry
157    // probability mass functions and are summed; the continuous identifiers
158    // carry densities and are integrated.
159    use DistributionIdentifier::*;
160    match identifier {
161        Normal | Exponential | Uniform | Gamma | Beta | Lognormal | StudentT | Cauchy
162        | ChiSquared | Erlang => true,
163        Bernoulli | Binomial | Geometric | Poisson | NegativeBinomial => false,
164    }
165}
