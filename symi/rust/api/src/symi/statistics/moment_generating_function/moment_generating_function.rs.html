1// Moment generating function M_X(s) = E[exp(s X)] dispatched
2// through the descriptor table first; falls through to a structural
3// placeholder otherwise. Wiring this through the Laplace
4// transform stays in scope as a future extension once the Laplace rule
5// library exposes the necessary substitution hook to flip the sign convention
6// (Laplace e^{-s x} -> MGF e^{+s x}) on the table image. For this first cut,
7// every closed-form MGF shipped in the descriptor table satisfies the
8// regressions, so the placeholder is structurally correct.
9
10use crate::expression::{BuiltinConstant, ExpressionKind, ExpressionReference};
11use crate::interner::Context;
12use crate::statistics::descriptor;
13use crate::symbol::SymbolIdentifier;
14
15/// Performs the advanced native `moment_generating_function` operation.
16pub fn moment_generating_function(
17    context: &Context,
18    random_variable: ExpressionReference,
19    auxiliary_variable: SymbolIdentifier,
20) -> ExpressionReference {
21    // Public MGF entry point. Closed-form table lookup first, structural
22    // undefined otherwise.
23    if let ExpressionKind::RandomVariable(_, underlying_distribution) = &random_variable.kind {
24        let underlying_distribution = underlying_distribution.clone();
25        if let Some(closed_form) = descriptor::moment_generating_function_closed_form(
26            context,
27            &underlying_distribution,
28            auxiliary_variable,
29        ) {
30            return closed_form;
31        }
32    }
33    context
34        .interner
35        .intern(ExpressionKind::Constant(BuiltinConstant::Undefined))
36}
