1// Symbolic probability distributions (univariate first tier).
2// The module is split as follows:
3//   - identifier.rs        DistributionIdentifier enum + per-identifier
4//                          parameter-arity / parameter-name tables.
5//   - descriptor.rs        closed-form dispatchers (PDF / PMF / CDF /
6//                          support / mean / variance / raw_moment / MGF / CF)
7//                          dispatched centrally over DistributionIdentifier.
8//   - expectation.rs       random-variable algebra (expectation, variance,
9//                          covariance, probability, moment, cumulant).
10//   - moment_generating_function.rs   Laplace transform wire-up.
11//   - characteristic_function.rs      Fourier transform wire-up.
12//   - terms.rs             ordered execution of the independent additive
13//                          terms and peer moment-identity expectations.
14// State of the art: Casella, Berger, *Statistical Inference* 2nd ed., Chs. 3-4
15// for the canonical tables; Rocklin & Terrel (2012), *Symbolic Statistics with
16// SymPy*, for the architectural pattern.

54/// Performs the advanced native `density` operation.
55pub fn density(
56    context: &Context,
57    random_variable: ExpressionReference,
58    point: ExpressionReference,
59) -> ExpressionReference {
60    // Public entry point that returns the probability density (continuous) or
61    // probability mass (discrete) of the supplied random variable evaluated at
62    // the supplied point. Dispatches on the underlying DistributionIdentifier
63    // through the descriptor table; falls back to the unevaluated
64    // structural form when no closed-form entry exists.
65    descriptor::density(context, random_variable, point)
66}
