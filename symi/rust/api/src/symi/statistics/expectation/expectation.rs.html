1// Random-variable algebra over independent operands. The public
2// entry points (expectation, variance, covariance, probability, moment,
3// cumulant) all route through the same closed-form stack: structural
4// recogniser -> linearity decomposition -> integration / summation fallback
5// -> structural placeholder. Independence is the default assumption;
6// a future joint_distribution AST would lift that.

20/// Performs the advanced native `expectation` operation.
21pub fn expectation(
22    context: &Context,
23    target: ExpressionReference,
24    random_variable_list: Vec<ExpressionReference>,
25) -> ExpressionReference {
26    // Public expectation dispatcher. Walks structural recognisers first, then
27    // linearity, then the integration/summation fallback, then closes with a
28    // structural placeholder. The random_variable_list pins which leaves are
29    // treated as random; every other free variable is a constant parameter.
30    let random_variable_set = collect_random_variable_names(&random_variable_list);
31    expectation_stack(
32        context,
33        &target,
34        &random_variable_list,
35        &random_variable_set,
36    )
37}

39/// Performs the advanced native `variance` operation.
40pub fn variance(
41    context: &Context,
42    target: ExpressionReference,
43    random_variable_list: Vec<ExpressionReference>,
44) -> ExpressionReference {
45    // Var(X) = E[X^2] - E[X]^2. The recursion through expectation handles
46    // linearity and constants. The identity's two expectations are over
47    // different targets and read nothing of each other, so they are offered to
48    // the moment-identity coordinator as peers in identity order.
49    let two = integer_value(context, 2);
50    let target_squared = power(context, Arc::clone(&target), two);
51    let random_variable_set = collect_random_variable_names(&random_variable_list);
52    let [expectation_of_squared, expectation_of_target] =
53        crate::statistics::terms::moment_identity_expectations(
54            context,
55            [target_squared, Arc::clone(&target)],
56            |peer| expectation_stack(context, peer, &random_variable_list, &random_variable_set),
57        );
58    let two_ = integer_value(context, 2);
59    let expectation_squared = power(context, expectation_of_target, two_);
60    subtract(context, expectation_of_squared, expectation_squared)
61}

63/// Performs the advanced native `covariance` operation.
64pub fn covariance(
65    context: &Context,
66    left: ExpressionReference,
67    right: ExpressionReference,
68    random_variable_list: Vec<ExpressionReference>,
69) -> ExpressionReference {
70    // Cov(X, Y) = E[XY] - E[X] E[Y]. Independence of X and Y collapses Cov to
71    // zero through the expectation linearity arm (the product of two
72    // independent random variables splits as the product of expectations).
73    // The identity's three expectations are over different targets and read
74    // nothing of each other, so they are offered to the moment-identity
75    // coordinator as peers in identity order.
76    let left_right = product(context, vec![Arc::clone(&left), Arc::clone(&right)]);
77    let random_variable_set = collect_random_variable_names(&random_variable_list);
78    let [
79        expectation_of_product,
80        expectation_of_left,
81        expectation_of_right,
82    ] = crate::statistics::terms::moment_identity_expectations(
83        context,
84        [left_right, Arc::clone(&left), Arc::clone(&right)],
85        |peer| expectation_stack(context, peer, &random_variable_list, &random_variable_set),
86    );
87    let product_of_expectations = product(context, vec![expectation_of_left, expectation_of_right]);
88    subtract(context, expectation_of_product, product_of_expectations)
89}

91/// Performs the advanced native `probability` operation.
92pub fn probability(
93    context: &Context,
94    predicate: ExpressionReference,
95    random_variable_list: Vec<ExpressionReference>,
96) -> ExpressionReference {
97    // Probability of a predicate. This recognises the comparison shapes
98    // X < a, X <= a, X > a, X >= a, and a <= X <= b, all in a single random
99    // variable. Predicates outside this shape close to the structural
100    // placeholder.
101    let _ = random_variable_list;
102    if let Some((random_variable, threshold, upper_tail)) = recognise_tail_comparison(&predicate) {
103        let cumulative = descriptor::cumulative_distribution(context, random_variable, threshold);
104        if upper_tail {
105            let one = integer_value(context, 1);
106            let raw = subtract(context, one, cumulative);
107            // Pass through expand so the survival form 1 - (1 - tail(...)) =
108            // tail(...) collapses to its closed form. The CDF dispatcher
109            // returns the canonical "1 - exp(-lambda x)" / "1 - ..." shapes
110            // for the half-line distributions, and the survival form is the
111            // textbook result a user expects from P(X > a).
112            return crate::expand::expand(context, &raw);
113        }
114        return cumulative;
115    }
116    constant_undefined(context)
117}

119/// Performs the advanced native `moment` operation.
120pub fn moment(
121    context: &Context,
122    random_variable: ExpressionReference,
123    order: ExpressionReference,
124) -> ExpressionReference {
125    // Routes through the descriptor table's raw_moment when the table is
126    // populated for the supplied identifier; otherwise falls back to
127    // expectation(X^order, [X]).
128    if let Some(underlying_distribution) = unpack_random_variable_distribution(&random_variable) {
129        let raw = descriptor::raw_moment(context, &underlying_distribution, &order);
130        if !is_undefined(&raw) {
131            return raw;
132        }
133    }
134    let target = power(context, Arc::clone(&random_variable), order);
135    expectation(context, target, vec![random_variable])
136}

138/// Performs the advanced native `cumulant` operation.
139pub fn cumulant(
140    context: &Context,
141    random_variable: ExpressionReference,
142    order: ExpressionReference,
143) -> ExpressionReference {
144    // The k-th cumulant is the coefficient of s^k / k! in the series
145    // expansion of log(M_X(s)) about s = 0. This ships closed-form cumulants
146    // for the distributions whose MGF has a textbook log-expansion: Normal,
147    // Poisson, Exponential, Gamma. Other identifiers close to the structural
148    // placeholder pending the future-round series-coefficient hook.
149    let order_integer = order_to_nonnegative_integer(&order);
150    if let Some(k) = order_integer {
151        if let Some(underlying_distribution) = unpack_random_variable_distribution(&random_variable)
152        {
153            if let Some(cumulant_value) = closed_form_cumulant(context, &underlying_distribution, k)
154            {
155                return cumulant_value;
156            }
157        }
158    }
159    constant_undefined(context)
160}

162fn expectation_stack(
163    context: &Context,
164    target: &ExpressionReference,
165    random_variable_list: &[ExpressionReference],
166    random_variable_set: &HashSet<SymbolIdentifier>,
167) -> ExpressionReference {
168    // The shared work-horse the public algebra entry points consume. The stack
169    // is ordered cheapest-first so the structural recognisers fire before any
170    // integration step is attempted; the integration path closes the
171    // remainder for inputs whose closed form is not table-resident.
172    if is_constant_in_random_variables(target, random_variable_set) {
173        return Arc::clone(target);
174    }
175    if let Some(closed_form) =
176        expectation_via_linearity(context, target, random_variable_list, random_variable_set)
177    {
178        return closed_form;
179    }
180    if let Some(closed_form) =
181        expectation_via_random_variable_atom(context, target, random_variable_set)
182    {
183        return closed_form;
184    }
185    constant_undefined(context)
186}

188fn expectation_via_linearity(
189    context: &Context,
190    target: &ExpressionReference,
191    random_variable_list: &[ExpressionReference],
192    random_variable_set: &HashSet<SymbolIdentifier>,
193) -> Option<ExpressionReference> {
194    // Linearity of expectation: E[a + b] = E[a] + E[b]; E[c f(X)] = c E[f(X)]
195    // when c is constant in every random variable. The product-arm decomposes
196    // products into a constant prefix and a residual factor; the residual is
197    // either a single random_variable leaf (whose expectation is mean) or a
198    // structural product of random variables that are independent (in which
199    // case the expectation splits multiplicatively).
200    match &target.kind {
201        ExpressionKind::Sum(children) => {
202            let accumulator =
203                crate::statistics::terms::expectation_terms(context, children, |child| {
204                    expectation_stack(context, child, random_variable_list, random_variable_set)
205                });
206            Some(sum(context, accumulator))
207        }
208        ExpressionKind::Product(children) => {
209            let mut constants: Vec<ExpressionReference> = Vec::new();
210            let mut random_factors: Vec<ExpressionReference> = Vec::new();
211            for child in children {
212                if is_constant_in_random_variables(child, random_variable_set) {
213                    constants.push(Arc::clone(child));
214                } else {
215                    random_factors.push(Arc::clone(child));
216                }
217            }
218            if random_factors.is_empty() {
219                return Some(product(context, constants));
220            }
221            let factor_expectation = if random_factors.len() == 1 {
222                let only = random_factors.pop().unwrap();
223                expectation_via_random_variable_atom(context, &only, random_variable_set)?
224            } else {
225                let mut accumulator: Vec<ExpressionReference> =
226                    Vec::with_capacity(random_factors.len());
227                for factor in &random_factors {
228                    let single_random_set = random_variables_free_in(factor, random_variable_set);
229                    if single_random_set.len() != 1 {
230                        return None;
231                    }
232                    let factor_expectation =
233                        expectation_via_random_variable_atom(context, factor, random_variable_set)?;
234                    accumulator.push(factor_expectation);
235                }
236                product(context, accumulator)
237            };
238            if constants.is_empty() {
239                return Some(factor_expectation);
240            }
241            constants.push(factor_expectation);
242            Some(product(context, constants))
243        }
244        _ => None,
245    }
246}

248fn expectation_via_random_variable_atom(
249    context: &Context,
250    target: &ExpressionReference,
251    random_variable_set: &HashSet<SymbolIdentifier>,
252) -> Option<ExpressionReference> {
253    // Closed-form expectation of a single-random-variable atom. The target is
254    // either a bare random_variable leaf (closed form: mean of the underlying
255    // distribution) or a power of one (closed form: raw_moment of the
256    // underlying distribution). Anything outside that shape declines so the
257    // caller can fall through.
258    match &target.kind {
259        ExpressionKind::RandomVariable(name, underlying_distribution) => {
260            if !random_variable_set.contains(name) {
261                return None;
262            }
263            Some(descriptor::mean(context, underlying_distribution))
264        }
265        ExpressionKind::Power(base, exponent) => {
266            if let ExpressionKind::RandomVariable(name, underlying_distribution) = &base.kind {
267                if random_variable_set.contains(name) {
268                    let raw = descriptor::raw_moment(context, underlying_distribution, exponent);
269                    if !is_undefined(&raw) {
270                        return Some(raw);
271                    }
272                }
273            }
274            None
275        }
276        _ => None,
277    }
278}

280fn closed_form_cumulant(
281    context: &Context,
282    distribution: &ExpressionReference,
283    order: u64,
284) -> Option<ExpressionReference> {
285    // First-tier closed-form cumulants: every cumulant of a Poisson(lambda)
286    // is lambda; the first cumulant of a Normal is mu, the second is sigma^2,
287    // and the higher cumulants are zero; the k-th cumulant of an Exponential
288    // is (k - 1)! / lambda^k; the k-th cumulant of a Gamma is shape * (k-1)! /
289    // rate^k. Everything else returns None and the caller closes with the
290    // structural placeholder.
291    use crate::statistics::identifier::DistributionIdentifier::*;
292    let (identifier, parameters) = match &distribution.kind {
293        ExpressionKind::Distribution(identifier, parameters) => (*identifier, parameters.clone()),
294        _ => return None,
295    };
296    match identifier {
297        Poisson => Some(Arc::clone(&parameters[0])),
298        Normal => match order {
299            0 => None,
300            1 => Some(Arc::clone(&parameters[0])),
301            2 => {
302                let two = integer_value(context, 2);
303                Some(power(context, Arc::clone(&parameters[1]), two))
304            }
305            _ => Some(integer_value(context, 0)),
306        },
307        Exponential => {
308            // k-th cumulant of Exponential(lambda) is (k-1)! / lambda^k.
309            if order == 0 {
310                return None;
311            }
312            let factorial = concrete_factorial(context, order - 1);
313            let exponent = integer_value(context, order as i64);
314            let denominator = power(context, Arc::clone(&parameters[0]), exponent);
315            let negative_one = integer_value(context, -1);
316            let inverse_denominator = power(context, denominator, negative_one);
317            Some(product(context, vec![factorial, inverse_denominator]))
318        }
319        Gamma | Erlang => {
320            // k-th cumulant of Gamma(shape, rate) is shape * (k-1)! / rate^k.
321            if order == 0 {
322                return None;
323            }
324            let factorial = concrete_factorial(context, order - 1);
325            let exponent = integer_value(context, order as i64);
326            let denominator = power(context, Arc::clone(&parameters[1]), exponent);
327            let negative_one = integer_value(context, -1);
328            let inverse_denominator = power(context, denominator, negative_one);
329            Some(product(
330                context,
331                vec![Arc::clone(&parameters[0]), factorial, inverse_denominator],
332            ))
333        }
334        _ => None,
335    }
336}

338fn collect_random_variable_names(
339    random_variable_list: &[ExpressionReference],
340) -> HashSet<SymbolIdentifier> {
341    // Project the random_variable_list passed by the caller into the set of
342    // symbol_identifier names. Names absent from the list are treated as
343    // constants by the linearity rule.
344    let mut accumulator: HashSet<SymbolIdentifier> = HashSet::new();
345    for entry in random_variable_list {
346        if let ExpressionKind::RandomVariable(name, _) = &entry.kind {
347            accumulator.insert(*name);
348        }
349    }
350    accumulator
351}

353fn is_constant_in_random_variables(
354    target: &ExpressionReference,
355    random_variable_set: &HashSet<SymbolIdentifier>,
356) -> bool {
357    // A target is constant in the supplied random variables when none of those
358    // variable names occurs free inside it. Random variables are leaves and
359    // contribute their names as "free" through the random_variable arm of the
360    // free_variables walker.
361    let footprint = free_variables(target);
362    footprint
363        .iter()
364        .all(|identifier| !random_variable_set.contains(identifier))
365}

367fn random_variables_free_in(
368    target: &ExpressionReference,
369    random_variable_set: &HashSet<SymbolIdentifier>,
370) -> HashSet<SymbolIdentifier> {
371    // Intersection of the target's free-variable footprint with the supplied
372    // set of random-variable names. Used by the product-arm of the linearity
373    // rule to detect whether a factor mentions exactly one random variable.
374    let mut accumulator: HashSet<SymbolIdentifier> = HashSet::new();
375    let footprint = free_variables(target);
376    for identifier in footprint {
377        if random_variable_set.contains(&identifier) {
378            accumulator.insert(identifier);
379        }
380    }
381    accumulator
382}

384fn unpack_random_variable_distribution(
385    candidate: &ExpressionReference,
386) -> Option<ExpressionReference> {
387    // Return the distribution(...) subexpression of a random_variable(name,
388    // distribution) leaf.
389    if let ExpressionKind::RandomVariable(_, underlying_distribution) = &candidate.kind {
390        Some(Arc::clone(underlying_distribution))
391    } else {
392        None
393    }
394}

396fn recognise_tail_comparison(
397    predicate: &ExpressionReference,
398) -> Option<(ExpressionReference, ExpressionReference, bool)> {
399    // Recognise predicates of the shape X < a, X <= a, X > a, X >= a. Returns
400    // (random_variable, threshold, upper_tail), where upper_tail is true when
401    // the comparison is X > a or X >= a (so the caller emits 1 - CDF). Other
402    // predicate shapes (range, set-membership) close to the structural
403    // placeholder currently.
404    use crate::function::{BuiltinFunction, FunctionIdentifier};
405    if let ExpressionKind::FunctionCall(function, arguments) = &predicate.kind {
406        if let FunctionIdentifier::Builtin(builtin) = function {
407            if arguments.len() != 2 {
408                return None;
409            }
410            let left = &arguments[0];
411            let right = &arguments[1];
412            let upper_tail = matches!(
413                builtin,
414                BuiltinFunction::GreaterThan | BuiltinFunction::GreaterThanOrEqual
415            );
416            let lower_tail = matches!(
417                builtin,
418                BuiltinFunction::LessThan | BuiltinFunction::LessThanOrEqual
419            );
420            if !(upper_tail || lower_tail) {
421                return None;
422            }
423            if matches!(left.kind, ExpressionKind::RandomVariable(_, _)) {
424                return Some((Arc::clone(left), Arc::clone(right), upper_tail));
425            }
426            if matches!(right.kind, ExpressionKind::RandomVariable(_, _)) {
427                return Some((Arc::clone(right), Arc::clone(left), !upper_tail));
428            }
429        }
430    }
431    None
432}

434fn constant_undefined(context: &Context) -> ExpressionReference {
435    // Convenience wrapper around the undefined sentinel; mirrors the helper
436    // in statistics/descriptor.rs.
437    context
438        .interner
439        .intern(ExpressionKind::Constant(BuiltinConstant::Undefined))
440}

442fn is_undefined(candidate: &ExpressionReference) -> bool {
443    // Recognise the structural undefined sentinel produced by the descriptor
444    // table when a closed form is not available. Used by the moment / cumulant
445    // resolvers to decide whether to fall through to the integration path.
446    matches!(
447        &candidate.kind,
448        ExpressionKind::Constant(BuiltinConstant::Undefined)
449    )
450}

452fn concrete_factorial(context: &Context, value: u64) -> ExpressionReference {
453    // Fold k! to a concrete integer leaf for small k; fall back to the
454    // symbolic factorial(k) builder otherwise. Mirrors the descriptor-table
455    // helper of the same shape; centralised here because the closed-form
456    // cumulant formulas need numeric coefficients to combine like terms in
457    // downstream variance / cumulant expressions.
458    let mut product_accumulator: i128 = 1;
459    for factor in 1..=value {
460        match product_accumulator.checked_mul(factor as i128) {
461            Some(next) if next <= i64::MAX as i128 => product_accumulator = next,
462            _ => {
463                let argument = integer_value(context, value as i64);
464                return crate::expression::builders::factorial(context, argument);
465            }
466        }
467    }
468    integer_value(context, product_accumulator as i64)
469}

471fn order_to_nonnegative_integer(order: &ExpressionReference) -> Option<u64> {
472    // Interpret a moment / cumulant order expression as a concrete non-negative
473    // integer. Mirrors the descriptor-table helper so the two can be unified
474    // once the closed-form moment library grows further.
475    use crate::exact_arithmetic::Integer;
476    match &order.kind {
477        ExpressionKind::Integer(value) => {
478            if *value < Integer::from(0) {
479                return None;
480            }
481            let mut bytes = value.to_string().into_bytes();
482            bytes.retain(|byte| *byte != b'-');
483            std::str::from_utf8(&bytes).ok()?.parse::<u64>().ok()
484        }
485        _ => None,
486    }
487}

489// Silence-the-unused warnings for builders that the public surface re-exports
490// but that the implementation does not consume directly in the closed-form
491// path. negate and the symbol builder remain available for the future-round
492// integration / summation fallback.
493#[allow(dead_code)]
494fn _unused_builder_keepalive(context: &Context) -> ExpressionReference {
495    let zero = integer_value(context, 0);
496    let one = integer_value(context, 1);
497    let negated = negate(context, one);
498    let symbol_identifier_local = context.symbol_table.intern_symbol("uu_keepalive");
499    let _ = symbol_builder(context, "uu_keepalive");
500    let _ = symbol_identifier_local;
501    sum(context, vec![zero, negated])
