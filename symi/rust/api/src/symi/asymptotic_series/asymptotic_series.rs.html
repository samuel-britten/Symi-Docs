326/// Performs the advanced native `normalize_scale` operation.
327pub fn normalize_scale(scale: AsymptoticScale) -> AsymptoticScale {
328    let mut logarithmic_powers = scale.logarithmic_powers;
329    logarithmic_powers.sort_by_key(|value| value.depth);
330    let mut combined_logarithmic_powers: Vec<LogarithmicPower> = Vec::new();
331    for logarithmic_power in logarithmic_powers {
332        if logarithmic_power.exponent == Rational::ZERO {
333            continue;
334        }
335        if let Some(last) = combined_logarithmic_powers.last_mut() {
336            if last.depth == logarithmic_power.depth {
337                last.exponent += logarithmic_power.exponent;
338                if last.exponent == Rational::ZERO {
339                    combined_logarithmic_powers.pop();
340                }
341                continue;
342            }
343        }
344        combined_logarithmic_powers.push(logarithmic_power);
345    }
346    AsymptoticScale {
347        exponential_argument: scale.exponential_argument,
348        local_power: scale.local_power,
349        logarithmic_powers: combined_logarithmic_powers,
350        algebraic_branch_identifier: scale.algebraic_branch_identifier,
351    }
352}

394/// Performs the advanced native `multiply_scales` operation.
395pub fn multiply_scales(
396    context: &Context,
397    left: &AsymptoticScale,
398    right: &AsymptoticScale,
399) -> Option<AsymptoticScale> {
400    let exponential_argument = match (&left.exponential_argument, &right.exponential_argument) {
401        (Some(left_argument), Some(right_argument)) => Some(simplify_sum(
402            context,
403            vec![Arc::clone(left_argument), Arc::clone(right_argument)],
404        )),
405        (Some(argument), None) | (None, Some(argument)) => Some(Arc::clone(argument)),
406        (None, None) => None,
407    };
408    let mut logarithmic_powers = left.logarithmic_powers.clone();
409    logarithmic_powers.extend(right.logarithmic_powers.clone());
410    let algebraic_branch_identifier = match (
411        left.algebraic_branch_identifier,
412        right.algebraic_branch_identifier,
413    ) {
414        (Some(left_identifier), Some(right_identifier)) if left_identifier != right_identifier => {
415            return None;
416        }
417        (Some(identifier), _) | (_, Some(identifier)) => Some(identifier),
418        _ => None,
419    };
420    Some(normalize_scale(AsymptoticScale {
421        exponential_argument,
422        local_power: left.local_power.clone() + right.local_power.clone(),
423        logarithmic_powers,
424        algebraic_branch_identifier,
425    }))
426}

438/// Performs the advanced native `invert_scale` operation.
439pub fn invert_scale(context: &Context, scale: &AsymptoticScale) -> Option<AsymptoticScale> {
440    let exponential_argument = scale
441        .exponential_argument
442        .as_ref()
443        .map(|argument| builders::negate(context, Arc::clone(argument)));
444    let logarithmic_powers = scale
445        .logarithmic_powers
446        .iter()
447        .map(|value| LogarithmicPower {
448            depth: value.depth,
449            exponent: -value.exponent.clone(),
450        })
451        .collect();
452    Some(normalize_scale(AsymptoticScale {
453        exponential_argument,
454        local_power: -scale.local_power.clone(),
455        logarithmic_powers,
456        algebraic_branch_identifier: scale.algebraic_branch_identifier,
457    }))
458}

513/// Performs the advanced native `laurent_asymptotic_series` operation.
514pub fn laurent_asymptotic_series(
515    context: &Context,
516    target: &ExpressionReference,
517    variable: SymbolIdentifier,
518    center: &ExpressionReference,
519    order: usize,
520    direction: AsymptoticDirection,
521) -> AsymptoticSeriesResult {
522    let pole_order_value = match pole_order(context, target, variable, center) {
523        Some(value) => value,
524        None => {
525            return AsymptoticSeriesResult::Incomplete {
526                verified_expansion: None,
527                reason: AsymptoticIncompleteReason::UnsupportedExpression,
528            };
529        }
530    };
531    let approach = AsymptoticApproach::Finite {
532        point: Arc::clone(center),
533        direction,
534    };
535    if pole_order_value <= 0 {
536        return taylor_asymptotic_series(context, target, variable, center, order, direction);
537    }
538    let shifted_target = shift_and_multiply(context, target, variable, center, pole_order_value);
539    let first_power = -pole_order_value;
540    let term_count = order + pole_order_value as usize;
541    let terms = taylor_terms(
542        context,
543        &shifted_target,
544        variable,
545        center,
546        first_power,
547        term_count,
548    );
549    let bound_scale = power_scale(Rational::from(Integer::from(order)));
550    build_adapter_expansion(
551        context,
552        variable,
553        approach,
554        terms,
555        bound_scale,
556        AdapterSource::Laurent,
557        Rational::from(Integer::from(order)),
558    )
559}

561/// Performs the advanced native `puiseux_asymptotic_series` operation.
562pub fn puiseux_asymptotic_series(
563    context: &Context,
564    branch: &PuiseuxBranch,
565    branch_identifier: usize,
566) -> AsymptoticSeriesResult {
567    if !branch.residual_certificate.verified {
568        return AsymptoticSeriesResult::Incomplete {
569            verified_expansion: None,
570            reason: AsymptoticIncompleteReason::ResidualVerificationFailed,
571        };
572    }
573    let approach = match &branch.series.expansion_point {
574        LocalExpansionPoint::Finite(point) => AsymptoticApproach::Finite {
575            point: Arc::clone(point),
576            direction: AsymptoticDirection::FromRight,
577        },
578        LocalExpansionPoint::Infinity => AsymptoticApproach::PositiveInfinity,
579    };
580    let mut terms = Vec::new();
581    for term in &branch.series.terms {
582        let coefficient = puiseux_coefficient_to_expression(context, &term.coefficient);
583        let scale = normalize_scale(AsymptoticScale {
584            exponential_argument: None,
585            local_power: Rational::from_integers(
586                term.exponent_numerator.clone(),
587                Integer::from(branch.series.ramification_index),
588            ),
589            logarithmic_powers: Vec::new(),
590            algebraic_branch_identifier: Some(branch_identifier),
591        });
592        terms.push(AsymptoticTerm { coefficient, scale });
593    }
594    let omitted_order = Rational::from_integers(
595        branch.series.first_omitted_order_numerator.clone(),
596        Integer::from(branch.series.ramification_index),
597    );
598    let bound_scale = normalize_scale(AsymptoticScale {
599        exponential_argument: None,
600        local_power: omitted_order.clone(),
601        logarithmic_powers: Vec::new(),
602        algebraic_branch_identifier: Some(branch_identifier),
603    });
604    build_adapter_expansion(
605        context,
606        branch.series.local_variable,
607        approach,
608        terms,
609        bound_scale,
610        AdapterSource::Puiseux,
611        certificate_order(&branch.residual_certificate, omitted_order),
612    )
613}

651/// Performs the advanced native `add_expansions` operation.
652pub fn add_expansions(
653    context: &Context,
654    left: &AsymptoticExpansion,
655    right: &AsymptoticExpansion,
656) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
657    if left.variable != right.variable || left.approach != right.approach {
658        return Err(AsymptoticIncompleteReason::UnsupportedApproach);
659    }
660    let mut terms = left.terms.clone();
661    terms.extend(right.terms.clone());
662    let bound = combine_bounds(context, &left.bound, &right.bound, &left.approach)?;
663    normalize_expansion(
664        context,
665        left.variable,
666        left.approach.clone(),
667        terms,
668        bound.clone(),
669        AsymptoticCertificate::Closure {
670            operation: AsymptoticOperation::Addition,
671            inputs: vec![left.certificate.clone(), right.certificate.clone()],
672            proof: ResidualProof::ScaleBound {
673                residual_scale: bound_scale_or_unit(&bound),
674                bound_scale: bound_scale_or_unit(&bound),
675            },
676        },
677    )
678}

680/// Performs the advanced native `subtract_expansions` operation.
681pub fn subtract_expansions(
682    context: &Context,
683    left: &AsymptoticExpansion,
684    right: &AsymptoticExpansion,
685) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
686    let mut negated_right = right.clone();
687    for term in &mut negated_right.terms {
688        term.coefficient = builders::negate(context, Arc::clone(&term.coefficient));
689    }
690    let mut result = add_expansions(context, left, &negated_right)?;
691    result.certificate = AsymptoticCertificate::Closure {
692        operation: AsymptoticOperation::Subtraction,
693        inputs: vec![left.certificate.clone(), right.certificate.clone()],
694        proof: ResidualProof::ScaleBound {
695            residual_scale: bound_scale_or_unit(&result.bound),
696            bound_scale: bound_scale_or_unit(&result.bound),
697        },
698    };
699    Ok(result)
700}

702/// Performs the advanced native `multiply_expansions` operation.
703pub fn multiply_expansions(
704    context: &Context,
705    left: &AsymptoticExpansion,
706    right: &AsymptoticExpansion,
707) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
708    if left.variable != right.variable || left.approach != right.approach {
709        return Err(AsymptoticIncompleteReason::UnsupportedApproach);
710    }
711    let mut terms = Vec::new();
712    for left_term in &left.terms {
713        for right_term in &right.terms {
714            let coefficient = simplify_product(
715                context,
716                vec![
717                    Arc::clone(&left_term.coefficient),
718                    Arc::clone(&right_term.coefficient),
719                ],
720            );
721            let scale = multiply_scales(context, &left_term.scale, &right_term.scale)
722                .ok_or(AsymptoticIncompleteReason::IncomparableScale)?;
723            terms.push(AsymptoticTerm { coefficient, scale });
724        }
725    }
726    let bound = product_bound(context, left, right)?;
727    normalize_expansion(
728        context,
729        left.variable,
730        left.approach.clone(),
731        terms,
732        bound.clone(),
733        AsymptoticCertificate::Closure {
734            operation: AsymptoticOperation::Multiplication,
735            inputs: vec![left.certificate.clone(), right.certificate.clone()],
736            proof: ResidualProof::ScaleBound {
737                residual_scale: bound_scale_or_unit(&bound),
738                bound_scale: bound_scale_or_unit(&bound),
739            },
740        },
741    )
742}

744/// Performs the advanced native `reify_asymptotic_truncation` operation.
745pub fn reify_asymptotic_truncation(
746    context: &Context,
747    expansion: &AsymptoticExpansion,
748) -> Option<ExpressionReference> {
749    let mut terms = Vec::new();
750    for term in &expansion.terms {
751        let scale_expression = reify_scale(
752            context,
753            expansion.variable,
754            &expansion.approach,
755            &term.scale,
756        )?;
757        terms.push(simplify_product(
758            context,
759            vec![Arc::clone(&term.coefficient), scale_expression],
760        ));
761    }
762    Some(simplify_sum(context, terms))
763}

765/// Performs the advanced native `scale_strictly_dominates` operation.
766pub fn scale_strictly_dominates(
767    context: &Context,
768    left: &AsymptoticScale,
769    right: &AsymptoticScale,
770    approach: &AsymptoticApproach,
771) -> bool {
772    // Strict dominance query: true only when the comparison is decided and says the
773    // left scale grows strictly faster than the right one. Equivalent, incomparable
774    // and undecidable comparisons all answer false, so a caller that needs a leading
775    // term to outgrow its own residual cannot be satisfied by a same-order bound.
776    matches!(
777        compare_scales(context, left, right, approach),
778        DominanceComparison::LeftDominates
779    )
780}

782/// Performs the advanced native `leading_term_strictly_dominates_residual` operation.
783pub fn leading_term_strictly_dominates_residual(
784    context: &Context,
785    expansion: &AsymptoticExpansion,
786) -> bool {
787    // True when the expansion has a leading term that strictly outgrows every later
788    // term and the certified residual bound. This is the premise that lets a caller
789    // read the sign of a germ off the leading coefficient alone; an expansion whose
790    // bound sits at the same scale as its leading term proves nothing about the sign.
791    let Some(leading) = expansion.terms.first() else {
792        return false;
793    };
794    let approach = expansion.approach.clone();
795    for term in expansion.terms.iter().skip(1) {
796        let term_scale = term.scale.clone();
797        if !scale_strictly_dominates(context, &leading.scale, &term_scale, &approach) {
798            return false;
799        }
800    }
801    match &expansion.bound {
802        AsymptoticBound::ExactZero => true,
803        AsymptoticBound::BigO(bound_scale) => {
804            let bound_scale = bound_scale.clone();
805            scale_strictly_dominates(context, &leading.scale, &bound_scale, &approach)
806        }
807    }
808}

810/// Performs the advanced native `verify_expansion_residual` operation.
811pub fn verify_expansion_residual(
812    context: &Context,
813    source: &ExpressionReference,
814    expansion: &AsymptoticExpansion,
815) -> bool {
816    let truncation = match reify_asymptotic_truncation(context, expansion) {
817        Some(value) => value,
818        None => return false,
819    };
820    if expression_references_equal(source, &truncation) {
821        return true;
822    }
823    let residual = builders::subtract(context, Arc::clone(source), truncation);
824    if is_zero_expression(&residual) {
825        return true;
826    }
827    match &expansion.bound {
828        AsymptoticBound::ExactZero => false,
829        AsymptoticBound::BigO(scale) => residual_is_single_scale_multiple(
830            context,
831            &residual,
832            expansion.variable,
833            &expansion.approach,
834            scale,
835        ),
836    }
837}

839fn asymptotic_series_inner(
840    context: &Context,
841    target: &ExpressionReference,
842    variable: SymbolIdentifier,
843    approach: &AsymptoticApproach,
844    requested_order: usize,
845) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
846    if let Some(expanded) =
847        crate::special_functions::orthogonal_polynomials::series::finite_taylor_input(
848            context, target, variable,
849        )
850    {
851        return asymptotic_series_inner(context, &expanded, variable, approach, requested_order);
852    }
853    if is_zero_expression(target) {
854        return Ok(exact_zero_expansion(variable, approach.clone()));
855    }
856    if let Some(expansion) = constant_expansion(context, target, variable, approach)? {
857        return Ok(expansion);
858    }
859    match &target.kind {
860        ExpressionKind::Symbol(identifier) if *identifier == variable => {
861            symbol_expansion(context, variable, approach)
862        }
863        ExpressionKind::Sum(terms) => {
864            let mut accumulator = exact_zero_expansion(variable, approach.clone());
865            for term in terms {
866                let term_expansion =
867                    asymptotic_series_inner(context, term, variable, approach, requested_order)?;
868                accumulator = add_expansions(context, &accumulator, &term_expansion)?;
869            }
870            Ok(truncate_to_requested_order(
871                context,
872                accumulator,
873                requested_order,
874            )?)
875        }
876        ExpressionKind::Product(factors) => {
877            let mut accumulator = constant_one_expansion(context, variable, approach.clone());
878            for factor in factors {
879                let factor_expansion =
880                    asymptotic_series_inner(context, factor, variable, approach, requested_order)?;
881                accumulator = multiply_expansions(context, &accumulator, &factor_expansion)?;
882            }
883            Ok(truncate_to_requested_order(
884                context,
885                accumulator,
886                requested_order,
887            )?)
888        }
889        ExpressionKind::Power(base, exponent) => {
890            power_expansion(context, base, exponent, variable, approach, requested_order)
891        }
892        ExpressionKind::FunctionCall(
893            FunctionIdentifier::Builtin(BuiltinFunction::Zeta),
894            arguments,
895        ) if arguments.len() == 1 => zeta_local_laurent_expansion(
896            context,
897            &arguments[0],
898            variable,
899            approach,
900            requested_order,
901        ),
902        ExpressionKind::FunctionCall(
903            FunctionIdentifier::Builtin(BuiltinFunction::Exp),
904            arguments,
905        ) if arguments.len() == 1 => {
906            exponential_expansion(context, &arguments[0], variable, approach, requested_order)
907        }
908        ExpressionKind::FunctionCall(
909            FunctionIdentifier::Builtin(BuiltinFunction::Log),
910            arguments,
911        ) if arguments.len() == 1 => {
912            logarithm_expansion(context, &arguments[0], variable, approach, requested_order)
913        }
914        ExpressionKind::FunctionCall(
915            FunctionIdentifier::Builtin(BuiltinFunction::Sqrt),
916            arguments,
917        ) if arguments.len() == 1 => {
918            square_root_expansion(context, &arguments[0], variable, approach, requested_order)
919        }
920        ExpressionKind::FunctionCall(
921            FunctionIdentifier::Builtin(BuiltinFunction::Sin),
922            arguments,
923        )
924        | ExpressionKind::FunctionCall(
925            FunctionIdentifier::Builtin(BuiltinFunction::Cos),
926            arguments,
927        ) if arguments.len() == 1 => {
928            if matches!(
929                classify_infinity_growth(&arguments[0], variable, approach),
930                InfinityGrowth::PositiveInfinity | InfinityGrowth::NegativeInfinity
931            ) {
932                Err(AsymptoticIncompleteReason::OscillatoryExpression)
933            } else {
934                Err(AsymptoticIncompleteReason::UnsupportedExpression)
935            }
936        }
937        _ => Err(AsymptoticIncompleteReason::UnsupportedExpression),
938    }
939}

941fn zeta_local_laurent_expansion(
942    context: &Context,
943    argument: &ExpressionReference,
944    variable: SymbolIdentifier,
945    approach: &AsymptoticApproach,
946    requested_order: usize,
947) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
948    let AsymptoticApproach::Finite { point, .. } = approach else {
949        return Err(AsymptoticIncompleteReason::UnsupportedApproach);
950    };
951    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
952    let mut substitution = std::collections::HashMap::new();
953    substitution.insert(variable_expression, Arc::clone(point));
954    let argument_at_point = substitute(context, argument, &substitution);
955    if !is_one_expression(&argument_at_point) {
956        return Err(AsymptoticIncompleteReason::UnsupportedExpression);
957    }
958    let slope = differentiate(context, argument, variable);
959    if depends_on(&slope, variable)
960        || range_evidence::real_verdict(&context.symbol_table, &slope) != TruthValue::Yes
961        || range_evidence::nonzero_verdict(&context.symbol_table, &slope) != TruthValue::Yes
962    {
963        return Err(AsymptoticIncompleteReason::UnsupportedExpression);
964    }
965    let inverse_slope =
966        simplify_power(context, slope.clone(), builders::integer_value(context, -1));
967    let mut terms = vec![AsymptoticTerm {
968        coefficient: inverse_slope,
969        scale: power_scale(-Rational::ONE),
970    }];
971    if requested_order > 1 {
972        terms.push(AsymptoticTerm {
973            coefficient: builders::constant(
974                context,
975                crate::expression::BuiltinConstant::EulerGamma,
976            ),
977            scale: unit_scale(),
978        });
979    }
980    let bound = AsymptoticBound::BigO(power_scale(Rational::ONE));
981    normalize_expansion(
982        context,
983        variable,
984        approach.clone(),
985        terms,
986        bound.clone(),
987        theorem_certificate(
988            AsymptoticTheorem::ZetaLocalLaurent,
989            vec![AsymptoticPremiseCertificate::ExactlyNonzeroConstant { expression: slope }],
990            TheoremRemainderCertificate {
991                reference_identifier: "DLMF 25.2.4",
992                argument: Arc::clone(argument),
993                absolute_remainder_bound: reify_scale(
994                    context,
995                    variable,
996                    approach,
997                    &bound_scale_or_unit(&bound),
998                )
999                .ok_or(AsymptoticIncompleteReason::UnsupportedExpression)?,
1000            },
1001        ),
1002    )
1003}

1005fn power_expansion(
1006    context: &Context,
1007    base: &ExpressionReference,
1008    exponent: &ExpressionReference,
1009    variable: SymbolIdentifier,
1010    approach: &AsymptoticApproach,
1011    requested_order: usize,
1012) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1013    if let Some(integer_exponent) = integer_expression_value(exponent) {
1014        if integer_exponent == 0 {
1015            return Ok(constant_one_expansion(context, variable, approach.clone()));
1016        }
1017        if integer_exponent > 0 {
1018            let base_expansion =
1019                asymptotic_series_inner(context, base, variable, approach, requested_order)?;
1020            let mut result = constant_one_expansion(context, variable, approach.clone());
1021            for _ in 0..usize::try_from(integer_exponent).unwrap_or(usize::MAX) {
1022                result = multiply_expansions(context, &result, &base_expansion)?;
1023                result = truncate_to_requested_order(context, result, requested_order)?;
1024            }
1025            return Ok(result);
1026        }
1027        if integer_exponent < 0 {
1028            if integer_exponent == -1 {
1029                if let Ok(expansion) =
1030                    invert_expression_expansion(context, base, variable, approach, requested_order)
1031                {
1032                    return Ok(expansion);
1033                }
1034            }
1035            return negative_integer_power_expansion(
1036                context,
1037                base,
1038                integer_exponent,
1039                variable,
1040                approach,
1041                requested_order,
1042            );
1043        }
1044    }
1045    if rational_expression_value(exponent)
1046        == Some(Rational::from_integers(Integer::ONE, Integer::from(2)))
1047    {
1048        return square_root_expansion(context, base, variable, approach, requested_order);
1049    }
1050    Err(AsymptoticIncompleteReason::BranchAmbiguous)
1051}

1053fn exponential_expansion(
1054    context: &Context,
1055    argument: &ExpressionReference,
1056    variable: SymbolIdentifier,
1057    approach: &AsymptoticApproach,
1058    requested_order: usize,
1059) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1060    if let Ok(argument_expansion) =
1061        asymptotic_series_inner(context, argument, variable, approach, requested_order)
1062    {
1063        if expansion_tends_to_zero(context, &argument_expansion)? {
1064            return exponential_of_small_expansion(context, &argument_expansion, requested_order);
1065        }
1066        if let Some((constant_part, small_part)) =
1067            split_constant_plus_small_expansion(context, &argument_expansion)?
1068        {
1069            let small_exponential =
1070                exponential_of_small_expansion(context, &small_part, requested_order)?;
1071            let constant_exponential = builders::function_call(
1072                context,
1073                FunctionIdentifier::Builtin(BuiltinFunction::Exp),
1074                vec![constant_part],
1075            );
1076            let constant_expansion = single_term_exact_expansion(
1077                context,
1078                variable,
1079                approach.clone(),
1080                constant_exponential,
1081                unit_scale(),
1082            );
1083            return multiply_expansions(context, &constant_expansion, &small_exponential).and_then(
1084                |expansion| truncate_to_requested_order(context, expansion, requested_order),
1085            );
1086        }
1087    }
1088    match classify_infinity_growth(argument, variable, approach) {
1089        InfinityGrowth::PositiveInfinity
1090        | InfinityGrowth::NegativeInfinity
1091        | InfinityGrowth::Finite => {
1092            if depends_on(argument, variable) {
1093                let scale = exponential_scale(Arc::clone(argument));
1094                let term = AsymptoticTerm {
1095                    coefficient: builders::one(context),
1096                    scale,
1097                };
1098                normalize_expansion(
1099                    context,
1100                    variable,
1101                    approach.clone(),
1102                    vec![term],
1103                    AsymptoticBound::ExactZero,
1104                    elementary_certificate(AsymptoticOperation::Exponential),
1105                )
1106            } else {
1107                let exponential = builders::function_call(
1108                    context,
1109                    FunctionIdentifier::Builtin(BuiltinFunction::Exp),
1110                    vec![Arc::clone(argument)],
1111                );
1112                Ok(single_term_exact_expansion(
1113                    context,
1114                    variable,
1115                    approach.clone(),
1116                    exponential,
1117                    unit_scale(),
1118                ))
1119            }
1120        }
1121        InfinityGrowth::Oscillatory => Err(AsymptoticIncompleteReason::OscillatoryExpression),
1122        InfinityGrowth::Unknown => Err(AsymptoticIncompleteReason::UndecidableScaleComparison),
1123    }
1124    .and_then(|expansion| truncate_to_requested_order(context, expansion, requested_order))
1125}

1127fn logarithm_expansion(
1128    context: &Context,
1129    argument: &ExpressionReference,
1130    variable: SymbolIdentifier,
1131    approach: &AsymptoticApproach,
1132    requested_order: usize,
1133) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1134    let one = builders::one(context);
1135    let small_argument = builders::subtract(context, Arc::clone(argument), one);
1136    if let Ok(small_expansion) = asymptotic_series_inner(
1137        context,
1138        &small_argument,
1139        variable,
1140        approach,
1141        requested_order,
1142    ) {
1143        if expansion_tends_to_zero(context, &small_expansion)? {
1144            return logarithm_of_one_plus_small_expansion(
1145                context,
1146                &small_expansion,
1147                requested_order,
1148            );
1149        }
1150    }
1151    if matches!(approach, AsymptoticApproach::PositiveInfinity)
1152        && is_negative_symbol_product(argument, variable)
1153    {
1154        return Err(AsymptoticIncompleteReason::BranchAmbiguous);
1155    }
1156    if let Some(constant) = constant_shift_of_variable(argument, variable) {
1157        if matches!(
1158            approach,
1159            AsymptoticApproach::PositiveInfinity | AsymptoticApproach::NegativeInfinity
1160        ) {
1161            let mut terms = vec![AsymptoticTerm {
1162                coefficient: builders::one(context),
1163                scale: logarithmic_scale(0, Rational::ONE),
1164            }];
1165            for k in 1..requested_order {
1166                let coefficient_rational = alternating_log_coefficient(&constant, k);
1167                let coefficient = builders::rational_literal(context, coefficient_rational);
1168                terms.push(AsymptoticTerm {
1169                    coefficient,
1170                    scale: power_scale(-Rational::from(Integer::from(k))),
1171                });
1172            }
1173            return normalize_expansion(
1174                context,
1175                variable,
1176                approach.clone(),
1177                terms,
1178                AsymptoticBound::BigO(power_scale(-Rational::from(Integer::from(requested_order)))),
1179                elementary_certificate(AsymptoticOperation::Logarithm),
1180            );
1181        }
1182    }
1183    Err(AsymptoticIncompleteReason::UnsupportedExpression)
1184}

1186fn square_root_expansion(
1187    context: &Context,
1188    argument: &ExpressionReference,
1189    variable: SymbolIdentifier,
1190    approach: &AsymptoticApproach,
1191    requested_order: usize,
1192) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1193    if let Some(constant) = x_squared_plus_constant_x(argument, variable) {
1194        if matches!(
1195            approach,
1196            AsymptoticApproach::PositiveInfinity | AsymptoticApproach::NegativeInfinity
1197        ) {
1198            let leading_coefficient = builders::one(context);
1199            let adjusted_constant = match approach {
1200                AsymptoticApproach::NegativeInfinity => -constant.clone(),
1201                _ => constant.clone(),
1202            };
1203            let mut terms = vec![AsymptoticTerm {
1204                coefficient: leading_coefficient,
1205                scale: power_scale(Rational::ONE),
1206            }];
1207            for k in 1..requested_order {
1208                let coefficient_rational =
1209                    binomial_coefficient(
1210                        Rational::from_integers(Integer::ONE, Integer::from(2)),
1211                        k,
1212                    ) * crate::exact_arithmetic::rational_power(&adjusted_constant, k as u64);
1213                let coefficient = builders::rational_literal(context, coefficient_rational);
1214                terms.push(AsymptoticTerm {
1215                    coefficient,
1216                    scale: power_scale(Rational::ONE - Rational::from(Integer::from(k))),
1217                });
1218            }
1219            return normalize_expansion(
1220                context,
1221                variable,
1222                approach.clone(),
1223                terms,
1224                AsymptoticBound::BigO(power_scale(
1225                    Rational::ONE - Rational::from(Integer::from(requested_order)),
1226                )),
1227                elementary_certificate(AsymptoticOperation::RationalPower),
1228            );
1229        }
1230    }
1231    Err(AsymptoticIncompleteReason::BranchAmbiguous)
1232}

1234fn negative_integer_power_expansion(
1235    context: &Context,
1236    base: &ExpressionReference,
1237    exponent: i64,
1238    variable: SymbolIdentifier,
1239    approach: &AsymptoticApproach,
1240    requested_order: usize,
1241) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1242    let base_expansion =
1243        asymptotic_series_inner(context, base, variable, approach, requested_order)?;
1244    let leading_term = match base_expansion.terms.first() {
1245        Some(value) => value,
1246        None => return Err(AsymptoticIncompleteReason::UnsupportedExpression),
1247    };
1248    if is_zero_expression(&leading_term.coefficient) {
1249        return Err(AsymptoticIncompleteReason::UnsupportedExpression);
1250    }
1251    let mut scale = unit_scale();
1252    let inverse_scale = invert_scale(context, &leading_term.scale)
1253        .ok_or(AsymptoticIncompleteReason::IncomparableScale)?;
1254    for _ in 0..usize::try_from(-exponent).unwrap_or(usize::MAX) {
1255        scale = multiply_scales(context, &scale, &inverse_scale)
1256            .ok_or(AsymptoticIncompleteReason::IncomparableScale)?;
1257    }
1258    let exponent_expression = builders::integer_value(context, exponent);
1259    let coefficient = builders::power(
1260        context,
1261        Arc::clone(&leading_term.coefficient),
1262        exponent_expression,
1263    );
1264    normalize_expansion(
1265        context,
1266        variable,
1267        approach.clone(),
1268        vec![AsymptoticTerm { coefficient, scale }],
1269        AsymptoticBound::ExactZero,
1270        AsymptoticCertificate::Closure {
1271            operation: AsymptoticOperation::IntegerPower,
1272            inputs: vec![base_expansion.certificate],
1273            proof: ResidualProof::ScaleBound {
1274                residual_scale: bound_scale_or_unit(&base_expansion.bound),
1275                bound_scale: bound_scale_or_unit(&base_expansion.bound),
1276            },
1277        },
1278    )
1279}

1281fn expansion_tends_to_zero(
1282    context: &Context,
1283    expansion: &AsymptoticExpansion,
1284) -> Result<bool, AsymptoticIncompleteReason> {
1285    let unit = unit_scale();
1286    for term in &expansion.terms {
1287        match compare_scales(context, &term.scale, &unit, &expansion.approach) {
1288            DominanceComparison::RightDominates => {}
1289            DominanceComparison::Equivalent | DominanceComparison::LeftDominates => {
1290                return Ok(false);
1291            }
1292            DominanceComparison::Incomparable => {
1293                return Err(AsymptoticIncompleteReason::IncomparableScale);
1294            }
1295            DominanceComparison::Undecidable => {
1296                return Err(AsymptoticIncompleteReason::UndecidableScaleComparison);
1297            }
1298        }
1299    }
1300    Ok(true)
1301}

1303fn split_constant_plus_small_expansion(
1304    context: &Context,
1305    expansion: &AsymptoticExpansion,
1306) -> Result<Option<(ExpressionReference, AsymptoticExpansion)>, AsymptoticIncompleteReason> {
1307    let unit = unit_scale();
1308    let mut constant_part: Option<ExpressionReference> = None;
1309    let mut small_terms = Vec::new();
1310    for term in &expansion.terms {
1311        match compare_scales(context, &term.scale, &unit, &expansion.approach) {
1312            DominanceComparison::Equivalent => {
1313                if constant_part.is_some() {
1314                    return Ok(None);
1315                }
1316                constant_part = Some(Arc::clone(&term.coefficient));
1317            }
1318            DominanceComparison::RightDominates => small_terms.push(term.clone()),
1319            DominanceComparison::LeftDominates => return Ok(None),
1320            DominanceComparison::Incomparable => {
1321                return Err(AsymptoticIncompleteReason::IncomparableScale);
1322            }
1323            DominanceComparison::Undecidable => {
1324                return Err(AsymptoticIncompleteReason::UndecidableScaleComparison);
1325            }
1326        }
1327    }
1328    let Some(constant) = constant_part else {
1329        return Ok(None);
1330    };
1331    let small_expansion = normalize_expansion(
1332        context,
1333        expansion.variable,
1334        expansion.approach.clone(),
1335        small_terms,
1336        expansion.bound.clone(),
1337        expansion.certificate.clone(),
1338    )?;
1339    if expansion_tends_to_zero(context, &small_expansion)? {
1340        Ok(Some((constant, small_expansion)))
1341    } else {
1342        Ok(None)

1346fn exponential_of_small_expansion(
1347    context: &Context,
1348    small_expansion: &AsymptoticExpansion,
1349    requested_order: usize,
1350) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1351    let mut result = constant_one_expansion(
1352        context,
1353        small_expansion.variable,
1354        small_expansion.approach.clone(),
1355    );
1356    let mut power_expansion = constant_one_expansion(
1357        context,
1358        small_expansion.variable,
1359        small_expansion.approach.clone(),
1360    );
1361    let mut factorial_value = Integer::ONE;
1362    for power in 1..requested_order {
1363        power_expansion = multiply_expansions(context, &power_expansion, small_expansion)?;
1364        power_expansion = truncate_to_requested_order(context, power_expansion, requested_order)?;
1365        factorial_value *= Integer::from(power);
1366        let coefficient = builders::rational(context, Integer::ONE, factorial_value.clone());
1367        let coefficient_expansion = single_term_exact_expansion(
1368            context,
1369            small_expansion.variable,
1370            small_expansion.approach.clone(),
1371            coefficient,
1372            unit_scale(),
1373        );
1374        let term_expansion =
1375            multiply_expansions(context, &coefficient_expansion, &power_expansion)?;
1376        result = add_expansions(context, &result, &term_expansion)?;
1377        result = truncate_to_requested_order(context, result, requested_order)?;
1378    }
1379    Ok(result)
1380}

1385    requested_order: usize,

1410            multiply_expansions(context, &coefficient_expansion, &power_expansion)?;

1417fn invert_expression_expansion(
1418    context: &Context,
1419    base: &ExpressionReference,
1420    variable: SymbolIdentifier,
1421    approach: &AsymptoticApproach,
1422    requested_order: usize,
1423) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1424    if let Some(constant) = constant_shift_of_variable(base, variable) {
1425        if matches!(
1426            approach,
1427            AsymptoticApproach::PositiveInfinity | AsymptoticApproach::NegativeInfinity
1428        ) {
1429            let mut terms = Vec::new();
1430            for k in 0..requested_order {
1431                let coefficient_rational = if k % 2 == 0 {
1432                    crate::exact_arithmetic::rational_power(&constant, k as u64)
1433                } else {
1434                    -crate::exact_arithmetic::rational_power(&constant, k as u64)
1435                };
1436                let coefficient = builders::rational_literal(context, coefficient_rational);
1437                terms.push(AsymptoticTerm {

1443                context,

1448                    requested_order + 1,

1452        }

1455}

1457fn normalize_expansion(
1458    context: &Context,
1459    variable: SymbolIdentifier,
1460    approach: AsymptoticApproach,
1461    terms: Vec<AsymptoticTerm>,
1462    bound: AsymptoticBound,
1463    certificate: AsymptoticCertificate,
1464) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
1465    let mut merged: Vec<AsymptoticTerm> = Vec::new();
1466    for mut term in terms {
1467        if is_zero_expression(&term.coefficient) {
1468            continue;
1469        }
1470        if depends_on(&term.coefficient, variable) {
1471            return Err(AsymptoticIncompleteReason::CoefficientDependsOnVariable);
1472        }
1473        term.scale = normalize_scale(term.scale);
1474        if let Some(existing) = merged
1475            .iter_mut()
1476            .find(|existing_term| existing_term.scale == term.scale)
1477        {
1478            existing.coefficient = simplify_sum(
1479                context,
1480                vec![
1481                    Arc::clone(&existing.coefficient),
1482                    Arc::clone(&term.coefficient),
1483                ],
1484            );
1485        } else {
1486            merged.push(term);
1487        }
1488    }
1489    merged.retain(|term| !is_zero_expression(&term.coefficient));
1490    let sorted = sort_terms(context, merged, &approach)?;
1491    let filtered = filter_terms_by_bound(context, sorted, &bound, &approach)?;
1492    Ok(AsymptoticExpansion {
1493        variable,
1494        approach,
1495        terms: filtered,
1496        bound,
1497        certificate,
1498    })
1499}

1501fn sort_terms(
1502    context: &Context,
1503    terms: Vec<AsymptoticTerm>,
1504    approach: &AsymptoticApproach,
1505) -> Result<Vec<AsymptoticTerm>, AsymptoticIncompleteReason> {
1506    let mut sorted: Vec<AsymptoticTerm> = Vec::new();
1507    for term in terms {
1508        let mut insertion_index = sorted.len();
1509        for (index, existing) in sorted.iter().enumerate() {
1510            match compare_scales(context, &term.scale, &existing.scale, approach) {
1511                DominanceComparison::LeftDominates => {
1512                    insertion_index = index;
1513                    break;
1514                }
1515                DominanceComparison::RightDominates => {}
1516                DominanceComparison::Equivalent => {}
1517                DominanceComparison::Incomparable => {
1518                    return Err(AsymptoticIncompleteReason::IncomparableScale);
1519                }
1520                DominanceComparison::Undecidable => {
1521                    return Err(AsymptoticIncompleteReason::UndecidableScaleComparison);
1522                }
1523            }
1524        }
1525        sorted.insert(insertion_index, term);
1526    }
1527    Ok(sorted)
1528}

1530fn filter_terms_by_bound(
1531    context: &Context,
1532    terms: Vec<AsymptoticTerm>,
1533    bound: &AsymptoticBound,
1534    approach: &AsymptoticApproach,
1535) -> Result<Vec<AsymptoticTerm>, AsymptoticIncompleteReason> {
1536    match bound {
1537        AsymptoticBound::ExactZero => Ok(terms),
1538        AsymptoticBound::BigO(bound_scale) => {
1539            let mut filtered = Vec::new();
1540            for term in terms {
1541                match compare_scales(context, &term.scale, bound_scale, approach) {
1542                    DominanceComparison::LeftDominates => filtered.push(term),
1543                    DominanceComparison::Equivalent | DominanceComparison::RightDominates => {}
1544                    DominanceComparison::Incomparable => {
1545                        return Err(AsymptoticIncompleteReason::IncomparableScale);
1546                    }
1547                    DominanceComparison::Undecidable => {
1548                        return Err(AsymptoticIncompleteReason::UndecidableScaleComparison);
1549                    }
1550                }
1551            }
1552            Ok(filtered)
1553        }
1554    }
1555}

1585fn taylor_terms(
1586    context: &Context,
1587    target: &ExpressionReference,
1588    variable: SymbolIdentifier,
1589    expansion_point: &ExpressionReference,
1590    first_power: i64,
1591    term_count: usize,
1592) -> Vec<AsymptoticTerm> {
1593    if term_count == 0 {
1594        return Vec::new();
1595    }
1596    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
1597    let mut substitution: SubstitutionMap = std::collections::HashMap::new();
1598    substitution.insert(variable_expression, Arc::clone(expansion_point));
1599    let mut current_derivative = Arc::clone(target);
1600    let mut factorial_value: Integer = Integer::ONE;
1601    let mut terms = Vec::with_capacity(term_count);
1602    for k in 0..term_count {
1603        let coefficient_at_point = substitute(context, &current_derivative, &substitution);
1604        let coefficient = if k == 0 {
1605            coefficient_at_point
1606        } else {
1607            let factorial_expression = builders::integer(context, factorial_value.clone());
1608            let minus_one = builders::integer_value(context, -1);
1609            let inverse_factorial = simplify_power(context, factorial_expression, minus_one);
1610            simplify_product(context, vec![coefficient_at_point, inverse_factorial])
1611        };
1612        let power = first_power + k as i64;
1613        terms.push(AsymptoticTerm {
1614            coefficient,
1615            scale: power_scale(Rational::from(Integer::from(power))),
1616        });
1617        if k + 1 < term_count {
1618            current_derivative = differentiate(context, &current_derivative, variable);
1619            factorial_value *= Integer::from(k + 1);
1620        }
1621    }
1622    terms
1623}

1625fn compare_exponential_parts(
1626    context: &Context,
1627    left: &AsymptoticScale,
1628    right: &AsymptoticScale,
1629    approach: &AsymptoticApproach,
1630) -> DominanceComparison {
1631    match (&left.exponential_argument, &right.exponential_argument) {
1632        (Some(left_argument), Some(right_argument)) => {
1633            if expression_references_equal(left_argument, right_argument) {
1634                return DominanceComparison::Equivalent;
1635            }
1636            let difference = builders::subtract(
1637                context,
1638                Arc::clone(left_argument),
1639                Arc::clone(right_argument),
1640            );
1641            match classify_infinity_growth(
1642                &difference,
1643                symbol_from_approach_expression(&difference),
1644                approach,
1645            ) {
1646                InfinityGrowth::PositiveInfinity => DominanceComparison::LeftDominates,
1647                InfinityGrowth::NegativeInfinity => DominanceComparison::RightDominates,
1648                InfinityGrowth::Finite => DominanceComparison::Equivalent,
1649                InfinityGrowth::Oscillatory => DominanceComparison::Incomparable,
1650                InfinityGrowth::Unknown => DominanceComparison::Undecidable,
1651            }
1652        }
1653        (Some(argument), None) => match classify_scale_exponential_argument(argument, approach) {
1654            InfinityGrowth::PositiveInfinity => DominanceComparison::LeftDominates,
1655            InfinityGrowth::NegativeInfinity => DominanceComparison::RightDominates,
1656            InfinityGrowth::Finite => DominanceComparison::Equivalent,
1657            InfinityGrowth::Oscillatory => DominanceComparison::Incomparable,
1658            InfinityGrowth::Unknown => DominanceComparison::Undecidable,
1659        },
1660        (None, Some(argument)) => match classify_scale_exponential_argument(argument, approach) {
1661            InfinityGrowth::PositiveInfinity => DominanceComparison::RightDominates,
1662            InfinityGrowth::NegativeInfinity => DominanceComparison::LeftDominates,
1663            InfinityGrowth::Finite => DominanceComparison::Equivalent,
1664            InfinityGrowth::Oscillatory => DominanceComparison::Incomparable,
1665            InfinityGrowth::Unknown => DominanceComparison::Undecidable,
1666        },
1667        (None, None) => DominanceComparison::Equivalent,
1668    }
1669}

1671fn compare_local_powers(
1672    left: &Rational,
1673    right: &Rational,
1674    approach: &AsymptoticApproach,
1675) -> DominanceComparison {
1676    let ordering = left.cmp(right);
1677    match approach {
1678        AsymptoticApproach::Finite { .. } => match ordering {
1679            Ordering::Less => DominanceComparison::LeftDominates,
1680            Ordering::Greater => DominanceComparison::RightDominates,
1681            Ordering::Equal => DominanceComparison::Equivalent,
1682        },
1683        AsymptoticApproach::PositiveInfinity | AsymptoticApproach::NegativeInfinity => {
1684            match ordering {
1685                Ordering::Greater => DominanceComparison::LeftDominates,
1686                Ordering::Less => DominanceComparison::RightDominates,
1687                Ordering::Equal => DominanceComparison::Equivalent,
1688            }
1689        }
1690    }
1691}

1693fn compare_logarithmic_powers(
1694    left: &AsymptoticScale,
1695    right: &AsymptoticScale,
1696    approach: &AsymptoticApproach,
1697) -> DominanceComparison {
1698    let maximum_depth = left
1699        .logarithmic_powers
1700        .iter()
1701        .chain(right.logarithmic_powers.iter())
1702        .map(|value| value.depth)
1703        .max()
1704        .unwrap_or(0);
1705    for depth in 0..=maximum_depth {
1706        let left_exponent = logarithmic_exponent_at_depth(left, depth);
1707        let right_exponent = logarithmic_exponent_at_depth(right, depth);
1708        if left_exponent != right_exponent {
1709            return compare_local_powers(&left_exponent, &right_exponent, approach);
1710        }
1711    }
1712    DominanceComparison::Equivalent
1713}

1732fn classify_infinity_growth(
1733    expression: &ExpressionReference,
1734    variable: SymbolIdentifier,
1735    approach: &AsymptoticApproach,
1736) -> InfinityGrowth {
1737    if !depends_on(expression, variable) {
1738        return InfinityGrowth::Finite;
1739    }
1740    if contains_symbol_other_than(expression, variable) {
1741        return InfinityGrowth::Unknown;
1742    }
1743    if contains_undamped_oscillation(expression, variable) {
1744        return InfinityGrowth::Oscillatory;
1745    }
1746    let coefficient = match linear_coefficient(expression, variable) {
1747        Some(value) => value,
1748        None => return InfinityGrowth::Unknown,
1749    };
1750    if coefficient == Rational::ZERO {
1751        return InfinityGrowth::Finite;
1752    }
1753    let sign = coefficient.sign_comparison();
1754    match approach {
1755        AsymptoticApproach::PositiveInfinity => match sign {
1756            Ordering::Greater => InfinityGrowth::PositiveInfinity,
1757            Ordering::Less => InfinityGrowth::NegativeInfinity,
1758            Ordering::Equal => InfinityGrowth::Finite,
1759        },
1760        AsymptoticApproach::NegativeInfinity => match sign {
1761            Ordering::Greater => InfinityGrowth::NegativeInfinity,
1762            Ordering::Less => InfinityGrowth::PositiveInfinity,
1763            Ordering::Equal => InfinityGrowth::Finite,
1764        },
1765        AsymptoticApproach::Finite { .. } => InfinityGrowth::Unknown,
1766    }
1767}

1769fn linear_coefficient(
1770    expression: &ExpressionReference,
1771    variable: SymbolIdentifier,
1772) -> Option<Rational> {
1773    match &expression.kind {
1774        ExpressionKind::Symbol(identifier) if *identifier == variable => Some(Rational::ONE),
1775        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) | ExpressionKind::Constant(_) => {
1776            Some(Rational::ZERO)
1777        }
1778        ExpressionKind::Sum(terms) => {
1779            let mut total = Rational::ZERO;
1780            for term in terms {
1781                total += linear_coefficient(term, variable)?;
1782            }
1783            Some(total)
1784        }
1785        ExpressionKind::Product(factors) => {
1786            let mut numeric = Rational::ONE;
1787            let mut variable_factor_count = 0usize;
1788            for factor in factors {
1789                if let Some(value) = rational_expression_value(factor) {
1790                    numeric *= value;
1791                } else if matches!(&factor.kind, ExpressionKind::Symbol(identifier) if *identifier == variable)
1792                {
1793                    variable_factor_count += 1;
1794                } else {
1795                    return None;
1796                }
1797            }
1798            match variable_factor_count {
1799                0 => Some(Rational::ZERO),
1800                1 => Some(numeric),
1801                _ => None,
1802            }
1803        }
1804        _ => None,
1805    }
1806}

1808fn contains_symbol_other_than(
1809    expression: &ExpressionReference,
1810    variable: SymbolIdentifier,
1811) -> bool {
1812    // A growth classification is made with respect to one variable. A second symbol is
1813    // a parameter whose sign and size are unknown here, so its presence leaves the
1814    // classification undecided rather than letting it be read as a numeric factor.
1815    match &expression.kind {
1816        ExpressionKind::Symbol(identifier) => *identifier != variable,
1817        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => children
1818            .iter()
1819            .any(|child| contains_symbol_other_than(child, variable)),
1820        ExpressionKind::Power(base, exponent) => {
1821            contains_symbol_other_than(base, variable)
1822                || contains_symbol_other_than(exponent, variable)
1823        }
1824        ExpressionKind::FunctionCall(_, arguments) => arguments
1825            .iter()
1826            .any(|argument| contains_symbol_other_than(argument, variable)),
1827        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) | ExpressionKind::Constant(_) => {
1828            false
1829        }
1830        _ => true,
1831    }
1832}

1975fn combine_bounds(
1976    context: &Context,
1977    left: &AsymptoticBound,
1978    right: &AsymptoticBound,
1979    approach: &AsymptoticApproach,
1980) -> Result<AsymptoticBound, AsymptoticIncompleteReason> {
1981    match (left, right) {
1982        (AsymptoticBound::ExactZero, AsymptoticBound::ExactZero) => Ok(AsymptoticBound::ExactZero),
1983        (AsymptoticBound::ExactZero, AsymptoticBound::BigO(scale))
1984        | (AsymptoticBound::BigO(scale), AsymptoticBound::ExactZero) => {
1985            Ok(AsymptoticBound::BigO(scale.clone()))
1986        }
1987        (AsymptoticBound::BigO(left_scale), AsymptoticBound::BigO(right_scale)) => {
1988            match compare_scales(context, left_scale, right_scale, approach) {
1989                DominanceComparison::LeftDominates | DominanceComparison::Equivalent => {
1990                    Ok(AsymptoticBound::BigO(left_scale.clone()))
1991                }
1992                DominanceComparison::RightDominates => {
1993                    Ok(AsymptoticBound::BigO(right_scale.clone()))
1994                }
1995                DominanceComparison::Incomparable => {
1996                    Err(AsymptoticIncompleteReason::IncomparableScale)
1997                }
1998                DominanceComparison::Undecidable => {
1999                    Err(AsymptoticIncompleteReason::UndecidableScaleComparison)
2000                }
2001            }
2002        }
2003    }
2004}

2006fn product_bound(
2007    context: &Context,
2008    left: &AsymptoticExpansion,
2009    right: &AsymptoticExpansion,
2010) -> Result<AsymptoticBound, AsymptoticIncompleteReason> {
2011    match (&left.bound, &right.bound) {
2012        (AsymptoticBound::ExactZero, AsymptoticBound::ExactZero) => Ok(AsymptoticBound::ExactZero),
2013        (AsymptoticBound::BigO(left_scale), AsymptoticBound::ExactZero) => {
2014            if let Some(leading) = right.terms.first() {
2015                Ok(AsymptoticBound::BigO(
2016                    multiply_scales(context, left_scale, &leading.scale)
2017                        .ok_or(AsymptoticIncompleteReason::IncomparableScale)?,
2018                ))
2019            } else {
2020                Ok(AsymptoticBound::ExactZero)
2021            }
2022        }
2023        (AsymptoticBound::ExactZero, AsymptoticBound::BigO(right_scale)) => {
2024            if let Some(leading) = left.terms.first() {
2025                Ok(AsymptoticBound::BigO(
2026                    multiply_scales(context, right_scale, &leading.scale)
2027                        .ok_or(AsymptoticIncompleteReason::IncomparableScale)?,
2028                ))
2029            } else {
2030                Ok(AsymptoticBound::ExactZero)
2031            }
2032        }
2033        (AsymptoticBound::BigO(left_scale), AsymptoticBound::BigO(right_scale)) => {
2034            // (A + O(p)) * (B + O(q)) = A*B + O(p*B) + O(q*A) + O(p*q): the residual is
2035            // the largest of the three cross products, where an expansion with no
2036            // retained term is bounded by its own residual.
2037            let left_order = left
2038                .terms
2039                .first()
2040                .map_or(left_scale, |leading| &leading.scale);
2041            let right_order = right
2042                .terms
2043                .first()
2044                .map_or(right_scale, |leading| &leading.scale);
2045            let mut bound = AsymptoticBound::ExactZero;
2046            for (first, second) in [
2047                (left_scale, right_order),
2048                (right_scale, left_order),
2049                (left_scale, right_scale),
2050            ] {
2051                let cross = multiply_scales(context, first, second)
2052                    .ok_or(AsymptoticIncompleteReason::IncomparableScale)?;
2053                bound = combine_bounds(
2054                    context,
2055                    &bound,
2056                    &AsymptoticBound::BigO(cross),
2057                    &left.approach,
2058                )?;
2059            }
2060            Ok(bound)
2061        }
2062    }
2063}

2065fn truncate_to_requested_order(
2066    context: &Context,
2067    expansion: AsymptoticExpansion,
2068    requested_order: usize,
2069) -> Result<AsymptoticExpansion, AsymptoticIncompleteReason> {
2070    let bound_scale = match expansion.approach {
2071        AsymptoticApproach::PositiveInfinity | AsymptoticApproach::NegativeInfinity => {
2072            power_scale(-Rational::from(Integer::from(requested_order)))
2073        }
2074        AsymptoticApproach::Finite { .. } => {
2075            power_scale(Rational::from(Integer::from(requested_order)))
2076        }
2077    };
2078    // Truncation can only weaken a residual: the new bound is the larger of the
2079    // requested order and the residual the expansion already carries.
2080    let bound = combine_bounds(
2081        context,
2082        &expansion.bound,
2083        &AsymptoticBound::BigO(bound_scale),
2084        &expansion.approach,
2085    )?;
2086    normalize_expansion(
2087        context,
2088        expansion.variable,
2089        expansion.approach,
2090        expansion.terms,
2091        bound,
2092        expansion.certificate,
2093    )
2094}

2096fn reify_scale(
2097    context: &Context,
2098    variable: SymbolIdentifier,
2099    approach: &AsymptoticApproach,
2100    scale: &AsymptoticScale,
2101) -> Option<ExpressionReference> {
2102    let mut factors = Vec::new();
2103    if let Some(argument) = &scale.exponential_argument {
2104        factors.push(builders::function_call(
2105            context,
2106            FunctionIdentifier::Builtin(BuiltinFunction::Exp),
2107            vec![Arc::clone(argument)],
2108        ));
2109    }
2110    let local_parameter = local_parameter_expression(context, variable, approach)?;
2111    if scale.local_power != Rational::ZERO {
2112        let exponent = builders::rational_literal(context, scale.local_power.clone());
2113        factors.push(builders::power(
2114            context,
2115            Arc::clone(&local_parameter),
2116            exponent,
2117        ));
2118    }
2119    for logarithmic_power in &scale.logarithmic_powers {
2120        let mut logarithm_argument = Arc::clone(&local_parameter);
2121        for _ in 0..=logarithmic_power.depth {
2122            logarithm_argument = builders::function_call(
2123                context,
2124                FunctionIdentifier::Builtin(BuiltinFunction::Log),
2125                vec![logarithm_argument],
2126            );
2127        }
2128        let exponent = builders::rational_literal(context, logarithmic_power.exponent.clone());
2129        factors.push(builders::power(context, logarithm_argument, exponent));
2130    }
2131    if factors.is_empty() {
2132        Some(builders::one(context))
2133    } else {
2134        Some(simplify_product(context, factors))
2135    }
2136}
