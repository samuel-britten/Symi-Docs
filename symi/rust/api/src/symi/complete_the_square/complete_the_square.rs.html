14/// Performs the advanced native `complete_the_square` operation.
15pub fn complete_the_square(
16    context: &Context,
17    target: &ExpressionReference,
18    variable: &ExpressionReference,
19) -> ExpressionReference {
20    // Rewrite an expression that is quadratic in `variable` into the exact vertex
21    // form a·(variable − h)² + k, where a, b and c are the degree-2/1/0
22    // coefficients of the expression collected in `variable` and h = −b/(2a),
23    // k = c − b²/(4a). The rewrite is exact and equality-preserving by
24    // construction — the vertex form expands back to a·variable² + b·variable + c
25    // identically as rational-function arithmetic in the coefficients — and it
26    // uses no numeric approximation and makes no branch choice. Every shape
27    // outside that model returns the input unchanged: a non-symbol generator, a
28    // collected exponent set other than {2}, {2, 0}, {2, 1} or {2, 1, 0}, a
29    // coefficient still mentioning the variable, a leading coefficient not
30    // provably nonzero (dividing by it would make the emitted form invalid on its
31    // zero locus, and Symi never emits an assumption-unbacked result), an input
32    // already in vertex form (b = 0), and a completed form whose expansion
33    // disagrees with the expansion of the input.
34    let Some((quadratic, linear, constant)) = quadratic_coefficients(context, target, variable)
35    else {
36        return target.clone();
37    };
38    if !leading_coefficient_is_nonzero(&quadratic, &context.symbol_table) {
39        return target.clone();
40    }
41    if matches!(&linear.kind, ExpressionKind::Integer(value) if *value == integer_zero()) {
42        return target.clone();
43    }
44    let completed = vertex_form(context, &quadratic, &linear, &constant, variable);
45    if !expansions_agree(context, &completed, target) {
46        return target.clone();
47    }
48    completed
49}

51fn expansions_agree(
52    context: &Context,
53    completed: &ExpressionReference,
54    target: &ExpressionReference,
55) -> bool {
56    // The self-verification: the completed form and its source must expand to
57    // the same polynomial. Structural equality of the two expansions decides the
58    // common case; coefficients that are rational functions expand to equal
59    // rational functions carrying different denominators, so each expansion is
60    // also compared in rational normal form before the operation declines. This
61    // is a safety net only — a mismatch surviving both comparisons would
62    // indicate an implementation bug.
63    let expanded_completed = expand(context, completed);
64    let expanded_target = expand(context, target);
65    if expanded_completed == expanded_target {
66        return true;
67    }
68    let normalized_completed = cancel(context, &expanded_completed);
69    let normalized_target = cancel(context, &expanded_target);
70    normalized_completed == normalized_target
71}

73fn quadratic_coefficients(
74    context: &Context,
75    target: &ExpressionReference,
76    variable: &ExpressionReference,
77) -> Option<(
78    ExpressionReference,
79    ExpressionReference,
80    ExpressionReference,
81)> {
82    // Read the degree-2, degree-1 and degree-0 coefficients of the expression
83    // collected in `variable`, or None when the expression is not a quadratic
84    // polynomial slice in it. The accepted exponent sets are exactly {2}, {2, 0},
85    // {2, 1} and {2, 1, 0}, with every coefficient free of the variable: a
86    // coefficient that still mentions it — from sin(variable)·variable², from a
87    // symbolic power variable^n, or from any other kernel the power decomposition
88    // cannot reach — means there is no quadratic slice. The coefficients may
89    // contain other symbols freely. This is the natural seam for any consumer
90    // needing the a/b/c of a quadratic. A degree's terms are accumulated rather
91    // than slotted: automatic simplification flattens a collected group whose
92    // generator power is one back into the surrounding sum, so a coefficient can
93    // arrive spread over several terms.
94    let ExpressionKind::Symbol(identifier) = &variable.kind else {
95        return None;
96    };
97    let identifier = *identifier;
98    let collected = collect(context, target, variable);
99    let mut coefficients: [Vec<ExpressionReference>; 3] = [Vec::new(), Vec::new(), Vec::new()];
100    for term in sum_terms(&collected) {
101        let (exponent, coefficient) = split_term_by_generator(context, &term, variable);
102        let degree = nonnegative_integer_degree(&exponent)?;
103        if degree > 2 || depends_on(&coefficient, identifier) {
104            return None;
105        }
106        coefficients[degree].push(coefficient);
107    }
108    let [constant, linear, quadratic] = coefficients;
109    if quadratic.is_empty() {
110        return None;
111    }
112    Some((
113        gathered_coefficient(context, quadratic),
114        gathered_coefficient(context, linear),
115        gathered_coefficient(context, constant),
116    ))
117}

119fn gathered_coefficient(context: &Context, terms: Vec<ExpressionReference>) -> ExpressionReference {
120    // The coefficient a degree's collected terms add up to, which is zero when
121    // that degree is absent from the expression entirely.
122    if terms.is_empty() {
123        return builders::zero(context);
124    }
125    simplify_sum(context, terms)
126}

128fn nonnegative_integer_degree(exponent: &ExpressionReference) -> Option<usize> {
129    // The nonnegative integer degree a collected generator exponent denotes. A
130    // rational, negative or otherwise non-integer exponent has no place in a
131    // quadratic polynomial slice and yields None.
132    let ExpressionKind::Integer(value) = &exponent.kind else {
133        return None;
134    };
135    usize::try_from(integer_to_i64(value)?).ok()
136}

138fn leading_coefficient_is_nonzero(
139    leading_coefficient: &ExpressionReference,
140    table: &SymbolTable,
141) -> bool {
142    // Decide whether dividing by the leading coefficient is unconditionally valid.
143    // A numeric literal is decided by exact arithmetic; every other coefficient
144    // must be proven nonzero by the assumption context.
145    match &leading_coefficient.kind {
146        ExpressionKind::Integer(value) => *value != integer_zero(),
147        ExpressionKind::Rational(_) => true,
148        _ => matches!(is_nonzero(leading_coefficient, table), TruthValue::Yes),
149    }
150}

152fn vertex_form(
153    context: &Context,
154    quadratic: &ExpressionReference,
155    linear: &ExpressionReference,
156    constant: &ExpressionReference,
157    variable: &ExpressionReference,
158) -> ExpressionReference {
159    // Assemble a·(variable − h)² + k from the quadratic, linear and constant
160    // coefficients. The coefficient arithmetic runs through the ordinary
161    // simplifier seams, so rational and radical coefficients come out in
162    // canonical form, and the square is built as a power node over the shifted
163    // variable, which automatic simplification never distributes.
164    let two = builders::integer_value(context, 2);
165    let four = builders::integer_value(context, 4);
166    let double_leading = simplify_product(context, vec![two.clone(), quadratic.clone()]);
167    let quadruple_leading = simplify_product(context, vec![four, quadratic.clone()]);
168    let ratio = builders::divide(context, linear.clone(), double_leading);
169    let negated_ratio = builders::negate(context, ratio);
170    let vertex_abscissa = cancel(context, &negated_ratio);
171    let shifted_variable = builders::subtract(context, variable.clone(), vertex_abscissa);
172    let square = simplify_power(context, shifted_variable, two.clone());
173    let scaled_square = simplify_product(context, vec![quadratic.clone(), square]);
174    let squared_linear = simplify_power(context, linear.clone(), two);
175    let correction = builders::divide(context, squared_linear, quadruple_leading);
176    let negated_correction = builders::negate(context, correction);
177    let ordinate_sum = simplify_sum(context, vec![constant.clone(), negated_correction]);
178    let vertex_ordinate = cancel(context, &ordinate_sum);
179    simplify_sum(context, vec![scaled_square, vertex_ordinate])
180}
