17/// A dense univariate polynomial over `Q(i)`, stored low-to-high and always
18/// trimmed so the last entry is the true leading coefficient. The empty vector
19/// is the zero polynomial.
20pub(crate) type ExactPolynomial = Vec<GaussianRational>;

22/// Which of the two admitted exact coefficient fields an analysis is working in.
23/// `Q` is preferred whenever every coefficient is real so a real problem never
24/// acquires a complex closed form gratuitously.
25#[derive(Clone, Copy, Debug, PartialEq, Eq)]
26pub(crate) enum CoefficientField {
27    RationalField,
28    GaussianField,
29}

39pub(crate) fn polynomial_trim(target: &mut ExactPolynomial) {
40    while target.last().is_some_and(gaussian_is_zero) {
41        target.pop();
42    }

62    // The degree, or None for the zero polynomial which has no degree.

74pub(crate) fn polynomial_add(left: &ExactPolynomial, right: &ExactPolynomial) -> ExactPolynomial {
75    let mut result = vec![gaussian_zero(); left.len().max(right.len())];
76    for (position, value) in left.iter().enumerate() {
77        result[position] = gaussian_add(&result[position], value);
78    }
79    for (position, value) in right.iter().enumerate() {
80        result[position] = gaussian_add(&result[position], value);
81    }
82    polynomial_trim(&mut result);
83    result
84}

86pub(crate) fn polynomial_subtract(
87    left: &ExactPolynomial,
88    right: &ExactPolynomial,
89) -> ExactPolynomial {
90    let mut result = vec![gaussian_zero(); left.len().max(right.len())];
91    for (position, value) in left.iter().enumerate() {
92        result[position] = gaussian_add(&result[position], value);
93    }
94    for (position, value) in right.iter().enumerate() {
95        result[position] = gaussian_subtract(&result[position], value);
96    }
97    polynomial_trim(&mut result);
98    result
99}

101pub(crate) fn polynomial_multiply(
102    left: &ExactPolynomial,
103    right: &ExactPolynomial,
104) -> ExactPolynomial {
105    if left.is_empty() || right.is_empty() {
106        return polynomial_zero();
107    }
108    let mut result = gaussian_dense_multiply(left, right);
109    polynomial_trim(&mut result);
110    result

113pub(crate) fn polynomial_scale(
114    target: &ExactPolynomial,
115    factor: &GaussianRational,
116) -> ExactPolynomial {
117    if gaussian_is_zero(factor) {
118        return polynomial_zero();
119    }
120    let mut result: ExactPolynomial = target
121        .iter()
122        .map(|value| gaussian_multiply(value, factor))
123        .collect();
124    polynomial_trim(&mut result);
125    result
126}

128pub(crate) fn polynomial_divide_with_remainder(
129    dividend: &ExactPolynomial,
130    divisor: &ExactPolynomial,
131) -> Option<(ExactPolynomial, ExactPolynomial)> {
132    // Exact Euclidean division over the field. None only for a zero divisor.
133    if divisor.is_empty() {
134        return None;
135    }
136    let divisor_degree = divisor.len() - 1;
137    let divisor_leading = polynomial_leading(divisor);
138    let mut remainder = dividend.clone();
139    polynomial_trim(&mut remainder);
140    let mut quotient = polynomial_zero();
141    while remainder.len() > divisor_degree && !remainder.is_empty() {
142        let shift = remainder.len() - 1 - divisor_degree;
143        let factor = gaussian_divide(&polynomial_leading(&remainder), &divisor_leading)?;
144        let mut term = vec![gaussian_zero(); shift + 1];
145        term[shift] = factor;
146        quotient = polynomial_add(&quotient, &term);
147        let subtracted = polynomial_multiply(&term, divisor);
148        remainder = polynomial_subtract(&remainder, &subtracted);
149    }
150    Some((quotient, remainder))
151}

153pub(crate) fn polynomial_exact_divide(
154    dividend: &ExactPolynomial,
155    divisor: &ExactPolynomial,
156) -> Option<ExactPolynomial> {
157    // Division that must be exact; a nonzero remainder declines so a caller can
158    // never silently drop a factor.
159    let (quotient, remainder) = polynomial_divide_with_remainder(dividend, divisor)?;
160    if !polynomial_is_zero(&remainder) {
161        return None;
162    }
163    Some(quotient)
164}

166pub(crate) fn polynomial_monic(target: &ExactPolynomial) -> Option<ExactPolynomial> {
167    // Divide through by the leading coefficient. None for the zero polynomial.
168    if target.is_empty() {
169        return None;
170    }
171    let leading = polynomial_leading(target);
172    let inverse = gaussian_divide(&gaussian_one(), &leading)?;
173    Some(polynomial_scale(target, &inverse))
174}

176pub(crate) fn polynomial_gcd(left: &ExactPolynomial, right: &ExactPolynomial) -> ExactPolynomial {
177    // Monic greatest common divisor by the Euclidean algorithm over the exact
178    // field. The gcd of two zero polynomials is zero; otherwise the result is
179    // monic so orbit representatives are canonical.
180    let mut first = left.clone();
181    let mut second = right.clone();
182    polynomial_trim(&mut first);
183    polynomial_trim(&mut second);
184    while !polynomial_is_zero(&second) {
185        let remainder = match polynomial_divide_with_remainder(&first, &second) {
186            Some((_, remainder)) => remainder,
187            None => break,
188        };
189        first = second;
190        second = remainder;
191    }
192    if polynomial_is_zero(&first) {
193        return polynomial_zero();
194    }
195    polynomial_monic(&first).unwrap_or_else(polynomial_zero)
196}

198pub(crate) fn polynomial_shift(
199    target: &ExactPolynomial,
200    offset: &GaussianRational,
201) -> ExactPolynomial {
202    // The composed polynomial `target(x + offset)`. Expanding `target` in powers
203    // of `(x - offset)` gives coefficients `c_t` with
204    // `target(x) = sum(c_t * (x - offset)^t)`, so substituting `x -> x + offset`
205    // makes the same `c_t` the dense coefficients of the shifted polynomial.
206    if target.is_empty() {
207        return polynomial_zero();
208    }
209    let mut result = crate::exact_arithmetic::gaussian_rational::gaussian_taylor_coefficients(
210        target,
211        offset,
212        target.len() - 1,
213    );
214    polynomial_trim(&mut result);
215    result
216}

225pub(crate) fn polynomial_evaluate(
226    target: &ExactPolynomial,
227    point: &GaussianRational,
228) -> GaussianRational {
229    let mut total = gaussian_zero();
230    for coefficient in target.iter().rev() {
231        total = gaussian_add(&gaussian_multiply(&total, point), coefficient);
232    }

253pub(crate) fn cauchy_root_bound(target: &ExactPolynomial) -> Option<Rational> {
254    // The Cauchy bound `1 + max |c_j| / |c_n|` on the modulus of every root,
255    // computed with the exact rational magnitude bound so the result stays sound
256    // without leaving exact arithmetic. None for the zero polynomial.
257    let degree = polynomial_degree(target)?;
258    if degree == 0 {
259        return Some(Rational::ZERO);
260    }
261    let leading = polynomial_leading(target);
262    let leading_squared =
263        crate::exact_arithmetic::gaussian_rational::gaussian_squared_modulus(&leading);
264    if leading_squared == Rational::ZERO {
265        return None;
266    }
267    let mut largest = Rational::ZERO;
268    for coefficient in &target[..degree] {
269        let bound = gaussian_magnitude_bound(coefficient);
270        if bound > largest {
271            largest = bound;
272        }
273    }
274    // |leading| >= sqrt(leading_squared); use the exact lower bound
275    // leading_squared / magnitude_bound(leading), which never overestimates.
276    let leading_bound = gaussian_magnitude_bound(&leading);
277    if leading_bound == Rational::ZERO {
278        return None;
279    }
280    let leading_lower = &leading_squared / &leading_bound;
281    Some(Rational::ONE + largest / leading_lower)
282}

284pub(crate) fn coefficient_bits_within_budget(target: &ExactPolynomial) -> bool {
285    // Reject a polynomial whose exact coefficients have grown past the pinned
286    // bit budget, so a pathological input returns `resource_exhausted` instead of
287    // consuming unbounded time in the Euclidean algorithm.
288    target.iter().all(|coefficient| {
289        component_bits(&coefficient.real) <= budgets::MAXIMUM_COEFFICIENT_BITS
290            && component_bits(&coefficient.imaginary) <= budgets::MAXIMUM_COEFFICIENT_BITS
291    })
292}

294fn component_bits(value: &Rational) -> u64 {
295    let (numerator, denominator) = rational_signed_components(value);
296    integer_bits(&numerator).max(integer_bits(&denominator))
297}

299fn integer_bits(value: &Integer) -> u64 {
300    let mut magnitude = value.clone();
301    if magnitude.sign_comparison() == std::cmp::Ordering::Less {
302        magnitude = -magnitude;
303    }
304    let mut bits = 0u64;
305    let two = Integer::from(2);
306    while magnitude > Integer::ZERO {
307        magnitude /= two.clone();
308        bits += 1;
309        if bits > budgets::MAXIMUM_COEFFICIENT_BITS * 4 {
310            break;
311        }
312    }
313    bits
314}

316pub(crate) fn reduce_rational_function(
317    target: &ExactRationalFunction,
318) -> Option<ExactRationalFunction> {
319    // Cancel the exact gcd of numerator and denominator. Cancellation here is
320    // safe because it happens before any range reasoning: the pole ledger that
321    // decides whether a cancelled factor was ever zero on the range is applied
322    // separately by the callers that need it.
323    if polynomial_is_zero(&target.denominator) {
324        return None;
325    }
326    if polynomial_is_zero(&target.numerator) {
327        return Some(ExactRationalFunction {
328            numerator: polynomial_zero(),
329            denominator: polynomial_one(),
330        });
331    }
332    let common = polynomial_gcd(&target.numerator, &target.denominator);
333    let numerator = polynomial_exact_divide(&target.numerator, &common)?;
334    let denominator = polynomial_exact_divide(&target.denominator, &common)?;
335    Some(ExactRationalFunction {
336        numerator,
337        denominator,
338    })
339}

341pub(crate) fn rational_functions_equal(
342    left: &ExactRationalFunction,
343    right: &ExactRationalFunction,
344) -> bool {
345    // Exact equality of two rational functions by cross-multiplication, the
346    // comparison every certificate replay uses.
347    let left_side = polynomial_multiply(&left.numerator, &right.denominator);
348    let right_side = polynomial_multiply(&right.numerator, &left.denominator);
349    left_side == right_side
350}

376pub(crate) fn expression_to_exact_rational_function(
377    expression: &ExpressionReference,
378    index: SymbolIdentifier,
379) -> Option<ExactRationalFunction> {
380    // Convert an expression into an exact `Q(i)` rational function of the index.
381    // Anything outside sums, products, integer powers, exact rational literals,
382    // and the imaginary unit declines, so a symbolic parameter or a transcendental
383    // constant can never be silently absorbed into a coefficient.
384    let pair = rational_function_from_node(expression, index)?;
385    if polynomial_is_zero(&pair.denominator) {
386        return None;
387    }
388    if polynomial_degree(&pair.numerator).unwrap_or(0) > budgets::MAXIMUM_POLYNOMIAL_DEGREE
389        || polynomial_degree(&pair.denominator).unwrap_or(0) > budgets::MAXIMUM_POLYNOMIAL_DEGREE
390    {
391        return None;
392    }
393    if !coefficient_bits_within_budget(&pair.numerator)
394        || !coefficient_bits_within_budget(&pair.denominator)
395    {
396        return None;
397    }
398    reduce_rational_function(&pair)
399}

401fn rational_function_from_node(
402    expression: &ExpressionReference,
403    index: SymbolIdentifier,
404) -> Option<ExactRationalFunction> {
405    use ExpressionKind::*;
406    match &expression.kind {
407        Integer(value) => Some(constant_rational_function(gaussian_from_rational(
408            crate::exact_arithmetic::Rational::from(value.clone()),
409        ))),
410        Rational(value) => Some(constant_rational_function(gaussian_from_rational(
411            value.clone(),
412        ))),
413        Constant(BuiltinConstant::ImaginaryUnit) => {
414            Some(constant_rational_function(gaussian_imaginary_unit()))
415        }
416        Symbol(identifier) if *identifier == index => Some(ExactRationalFunction {
417            numerator: vec![gaussian_zero(), gaussian_one()],
418            denominator: polynomial_one(),
419        }),
420        Sum(children) => {
421            let mut total = ExactRationalFunction {
422                numerator: polynomial_zero(),
423                denominator: polynomial_one(),
424            };
425            for child in children {
426                let piece = rational_function_from_node(child, index)?;
427                let numerator = polynomial_add(
428                    &polynomial_multiply(&total.numerator, &piece.denominator),
429                    &polynomial_multiply(&piece.numerator, &total.denominator),
430                );
431                let denominator = polynomial_multiply(&total.denominator, &piece.denominator);
432                total = ExactRationalFunction {
433                    numerator,
434                    denominator,
435                };
436                if polynomial_degree(&total.denominator).unwrap_or(0)
437                    > budgets::MAXIMUM_POLYNOMIAL_DEGREE
438                {
439                    return None;
440                }
441            }
442            Some(total)
443        }
444        Product(children) => {
445            let mut total = ExactRationalFunction {
446                numerator: polynomial_one(),
447                denominator: polynomial_one(),
448            };
449            for child in children {
450                let piece = rational_function_from_node(child, index)?;
451                total = rational_function_multiply(&total, &piece);
452                if polynomial_degree(&total.numerator).unwrap_or(0)
453                    > budgets::MAXIMUM_POLYNOMIAL_DEGREE
454                    || polynomial_degree(&total.denominator).unwrap_or(0)
455                        > budgets::MAXIMUM_POLYNOMIAL_DEGREE
456                {
457                    return None;
458                }
459            }
460            Some(total)
461        }
462        Power(base, exponent) => {
463            let exponent_value = match &exponent.kind {
464                Integer(value) => i64::try_from(value).ok()?,
465                _ => return None,
466            };
467            let base_function = rational_function_from_node(base, index)?;
468            let magnitude = exponent_value.unsigned_abs() as usize;
469            if magnitude > budgets::MAXIMUM_POLYNOMIAL_DEGREE {
470                return None;
471            }
472            let mut accumulated = ExactRationalFunction {
473                numerator: polynomial_one(),
474                denominator: polynomial_one(),
475            };
476            for _ in 0..magnitude {
477                accumulated = rational_function_multiply(&accumulated, &base_function);
478            }
479            if exponent_value < 0 {
480                if polynomial_is_zero(&accumulated.numerator) {
481                    return None;
482                }
483                accumulated = ExactRationalFunction {
484                    numerator: accumulated.denominator,
485                    denominator: accumulated.numerator,
486                };
487            }
488            Some(accumulated)
489        }
490        _ => None,
491    }
492}

503pub(crate) fn gaussian_expression(
504    context: &Context,
505    value: &GaussianRational,
506) -> ExpressionReference {
507    // Reify an exact Gaussian rational as an expression, keeping the canonical
508    // `real + imaginary * i` shape.
509    let real_part = rational_expression(context, &value.real);
510    if value.imaginary == Rational::ZERO {
511        return real_part;
512    }
513    let imaginary_part = rational_expression(context, &value.imaginary);
514    let unit = builders::constant(context, BuiltinConstant::ImaginaryUnit);
515    let imaginary_term = builders::product(context, vec![imaginary_part, unit]);
516    if value.real == Rational::ZERO {
517        return imaginary_term;
518    }
519    builders::sum(context, vec![real_part, imaginary_term])
520}

527pub(crate) fn polynomial_expression(
528    context: &Context,
529    target: &ExactPolynomial,
530    index_symbol: &ExpressionReference,
531) -> ExpressionReference {
532    // Reify a dense polynomial as a sum of monomials in the index.
533    if target.is_empty() {
534        return builders::integer_value(context, 0);
535    }
536    let mut terms: Vec<ExpressionReference> = Vec::new();
537    for (exponent, coefficient) in target.iter().enumerate() {
538        if gaussian_is_zero(coefficient) {
539            continue;
540        }
541        let coefficient_expression = gaussian_expression(context, coefficient);
542        if exponent == 0 {
543            terms.push(coefficient_expression);
544            continue;
545        }
546        let power_expression = if exponent == 1 {
547            Arc::clone(index_symbol)
548        } else {
549            let exponent_expression = builders::integer_value(context, exponent as i64);
550            builders::power(context, Arc::clone(index_symbol), exponent_expression)
551        };
552        if gaussian_is_one(coefficient) {
553            terms.push(power_expression);
554        } else {
555            terms.push(builders::product(
556                context,
557                vec![coefficient_expression, power_expression],
558            ));
559        }
560    }
561    if terms.is_empty() {
562        return builders::integer_value(context, 0);
563    }
564    builders::sum(context, terms)
565}

567pub(crate) fn rational_function_expression(
568    context: &Context,
569    target: &ExactRationalFunction,
570    index_symbol: &ExpressionReference,
571) -> ExpressionReference {
572    let numerator = polynomial_expression(context, &target.numerator, index_symbol);
573    if polynomial_is_one(&target.denominator) {
574        return numerator;
575    }
576    let denominator = polynomial_expression(context, &target.denominator, index_symbol);
577    let minus_one = builders::integer_value(context, -1);
578    let inverse = builders::power(context, denominator, minus_one);
579    builders::product(context, vec![numerator, inverse])
580}

582pub(crate) fn integer_roots(target: &ExactPolynomial) -> Option<Vec<i64>> {
583    // Every integer root of the polynomial, found by evaluating at each integer
584    // inside the exact Cauchy root bound. The enumeration is complete, so an
585    // empty result is a proof that no integer root exists — that proof is what
586    // the pole ledger consumes. None means the bound exceeded the pinned search
587    // budget, which the caller reports as `resource_exhausted`.
588    if polynomial_is_zero(target) {
589        return None;
590    }
591    let bound = cauchy_root_bound(target)?;
592    let ceiling = rational_ceiling(&bound)?;
593    if ceiling > budgets::MAXIMUM_SHIFT_SEARCH {
594        return None;
595    }
596    let mut roots = Vec::new();
597    for candidate in -ceiling..=ceiling {
598        let point = gaussian_from_integer(candidate);
599        if gaussian_is_zero(&polynomial_evaluate(target, &point)) {
600            roots.push(candidate);
601        }
602    }
603    Some(roots)
604}

606pub(crate) fn rational_ceiling(value: &Rational) -> Option<i64> {
607    // Smallest integer at least `value`, computed exactly.
608    let (numerator, denominator) = rational_signed_components(value);
609    let quotient = numerator.clone() / denominator.clone();
610    let remainder = numerator - quotient.clone() * denominator;
611    let mut result = i64::try_from(&quotient).ok()?;
612    if remainder.sign_comparison() == std::cmp::Ordering::Greater {
613        result += 1;
614    }
615    Some(result)
616}

618pub(crate) fn is_pole_free_on_integer_range(
619    target: &ExactRationalFunction,
620    lower_bound: i64,
621    upper_bound: Option<i64>,
622) -> Option<bool> {
623    // The pole and zero ledger: the telescoping identity
624    // `product(P(k + 1)/P(k), k = lower..m) = P(m + 1)/P(lower)` is valid only
625    // while `P` is finite and nonzero at every integer of the closed range, so
626    // both the numerator's and the denominator's integer roots are excluded. An
627    // open upper end (an infinite or symbolic bound) excludes every integer root
628    // at or above the lower bound.
629    let numerator_roots = integer_roots(&target.numerator)?;
630    let denominator_roots = integer_roots(&target.denominator)?;
631    let in_range = |candidate: i64| match upper_bound {
632        Some(upper) => candidate >= lower_bound && candidate <= upper,
633        None => candidate >= lower_bound,
634    };
635    Some(
636        !numerator_roots.iter().copied().any(in_range)
637            && !denominator_roots.iter().copied().any(in_range),
638    )
639}
