19/// One removal along a shift orbit: the monic orbit representative `g`, the
20/// integer shift `j >= 1` separating the paired factors, how many times the pair
21/// was removed, and which side of the quotient carried the *leading* member.
22#[derive(Clone, Debug, PartialEq)]
23pub(crate) struct ShiftOrbitStep {
24    pub(crate) representative: ExactPolynomial,
25    pub(crate) shift: i64,
26    pub(crate) multiplicity: usize,
27    pub(crate) numerator_leads: bool,
28}

30/// Proof data for the decomposition
31/// `source(k) = constant * rational_part(k + 1)/rational_part(k) * residual(k)`
32/// discovered by shift-orbit reduction. Replay rebuilds the rational part from
33/// the recorded orbit steps and re-derives the identity over the exact field, so
34/// a corrupted factor, shift, multiplicity, or coefficient is rejected.
35#[derive(Clone, Debug)]
36pub(crate) struct ShiftQuotientCertificate {
37    pub(crate) source: ExactRationalFunction,
38    pub(crate) constant: GaussianRational,
39    pub(crate) rational_part: ExactRationalFunction,
40    pub(crate) residual: ExactRationalFunction,
41    pub(crate) orbit_steps: Vec<ShiftOrbitStep>,
42    pub(crate) field: CoefficientField,
43}

45pub(crate) fn reduce_shift_orbits(
46    source: &ExactRationalFunction,
47) -> Option<ShiftQuotientCertificate> {
48    // Sigma-factorisation of the shift-quotient problem `sigma(g)/g = f` in the
49    // rational difference field with `sigma(k) = k + 1`. Numerator and denominator
50    // factors that are integer shifts of one another are removed in pairs; each
51    // pair `g(k + j)/g(k)` is exactly the telescoping quotient of
52    // `prod(g(k + s), s = 0..j-1)`, so the removals accumulate into a rational
53    // part. What survives is shift-free by construction and is returned as the
54    // residual. None means a pinned budget was exceeded or the exact field
55    // arithmetic could not proceed; it never means "no reduction exists", because
56    // an empty orbit list is a valid certificate.
57    if polynomial_is_zero(&source.numerator) || polynomial_is_zero(&source.denominator) {
58        return None;
59    }
60    if !coefficient_bits_within_budget(&source.numerator)
61        || !coefficient_bits_within_budget(&source.denominator)
62    {
63        return None;
64    }
65    let numerator_degree = polynomial_degree(&source.numerator)?;
66    let denominator_degree = polynomial_degree(&source.denominator)?;
67    if numerator_degree > budgets::MAXIMUM_POLYNOMIAL_DEGREE
68        || denominator_degree > budgets::MAXIMUM_POLYNOMIAL_DEGREE
69    {
70        return None;
71    }
72    let field = combined_field(
73        polynomial_field(&source.numerator),
74        polynomial_field(&source.denominator),
75    );
76    let constant = gaussian_divide(
77        &polynomial_leading(&source.numerator),
78        &polynomial_leading(&source.denominator),
79    )?;
80    let mut numerator = polynomial_monic(&source.numerator)?;
81    let mut denominator = polynomial_monic(&source.denominator)?;
82    let search_bound = integer_shift_search_bound(&numerator, &denominator)?;
83    let mut orbit_steps: Vec<ShiftOrbitStep> = Vec::new();
84    let mut leading_part = polynomial_one();
85    let mut trailing_part = polynomial_one();
86    let mut gcd_operations = 0usize;
87    loop {
88        let mut progressed = false;
89        for shift in 1..=search_bound {
90            gcd_operations += 2;
91            if gcd_operations > budgets::MAXIMUM_GCD_OPERATIONS {
92                return None;
93            }
94            let shifted_numerator = polynomial_shift_by_integer(&numerator, -shift);
95            let common = polynomial_gcd(&shifted_numerator, &denominator);
96            if polynomial_degree(&common).unwrap_or(0) >= 1 {
97                let leading_member = polynomial_shift_by_integer(&common, shift);
98                numerator = polynomial_exact_divide(&numerator, &leading_member)?;
99                denominator = polynomial_exact_divide(&denominator, &common)?;
100                leading_part = polynomial_multiply(&leading_part, &orbit_window(&common, shift)?);
101                record_orbit_step(&mut orbit_steps, &common, shift, true);
102                progressed = true;
103                break;
104            }
105            let shifted_denominator = polynomial_shift_by_integer(&denominator, -shift);
106            let common = polynomial_gcd(&shifted_denominator, &numerator);
107            if polynomial_degree(&common).unwrap_or(0) >= 1 {
108                let leading_member = polynomial_shift_by_integer(&common, shift);
109                denominator = polynomial_exact_divide(&denominator, &leading_member)?;
110                numerator = polynomial_exact_divide(&numerator, &common)?;
111                trailing_part = polynomial_multiply(&trailing_part, &orbit_window(&common, shift)?);
112                record_orbit_step(&mut orbit_steps, &common, shift, false);
113                progressed = true;
114                break;
115            }
116        }
117        if !progressed {
118            break;
119        }
120    }
121    let certificate = ShiftQuotientCertificate {
122        source: source.clone(),
123        constant,
124        rational_part: ExactRationalFunction {
125            numerator: leading_part,
126            denominator: trailing_part,
127        },
128        residual: ExactRationalFunction {
129            numerator,
130            denominator,
131        },
132        orbit_steps,
133        field,
134    };
135    if !replay_shift_quotient(&certificate) {
136        return None;
137    }
138    Some(certificate)
139}

141fn record_orbit_step(
142    steps: &mut Vec<ShiftOrbitStep>,
143    representative: &ExactPolynomial,
144    shift: i64,
145    numerator_leads: bool,
146) {
147    // Merge a repeated removal of the same orbit representative at the same shift
148    // and side into one step's multiplicity rather than appending a duplicate.
149    for step in steps.iter_mut() {
150        if step.shift == shift
151            && step.numerator_leads == numerator_leads
152            && step.representative == *representative
153        {
154            step.multiplicity += 1;
155            return;
156        }
157    }
158    steps.push(ShiftOrbitStep {
159        representative: representative.clone(),
160        shift,
161        multiplicity: 1,
162        numerator_leads,
163    });
164}

166fn orbit_window(representative: &ExactPolynomial, shift: i64) -> Option<ExactPolynomial> {
167    // `prod(g(k + s), s = 0..shift-1)`, the anti-quotient of `g(k + shift)/g(k)`.
168    if shift < 1 || shift > budgets::MAXIMUM_SHIFT_SEARCH {
169        return None;
170    }
171    let mut window = polynomial_one();
172    for step in 0..shift {
173        window = polynomial_multiply(&window, &polynomial_shift_by_integer(representative, step));
174        if polynomial_degree(&window).unwrap_or(0) > budgets::MAXIMUM_POLYNOMIAL_DEGREE {
175            return None;
176        }
177    }
178    Some(window)
179}

181fn integer_shift_search_bound(
182    numerator: &ExactPolynomial,
183    denominator: &ExactPolynomial,
184) -> Option<i64> {
185    // Every integer shift `j` that can pair a numerator factor with a denominator
186    // factor is the difference of a root of one and a root of the other, so it is
187    // dominated by the sum of the two Cauchy root bounds. The search is therefore
188    // complete, not heuristic; when the bound exceeds the pinned cap the caller
189    // declines instead of silently truncating.
190    let numerator_bound = cauchy_root_bound(numerator)?;
191    let denominator_bound = cauchy_root_bound(denominator)?;
192    let total = rational_absolute(&numerator_bound) + rational_absolute(&denominator_bound);
193    let ceiling = rational_ceiling(&total)?;
194    if ceiling > budgets::MAXIMUM_SHIFT_SEARCH {
195        return None;
196    }
197    Some(ceiling.max(1))
198}

200pub(crate) fn replay_shift_quotient(certificate: &ShiftQuotientCertificate) -> bool {
201    // Independent re-derivation. The rational part is rebuilt from the recorded
202    // orbit representatives, shifts, and multiplicities alone; the reconstructed
203    // identity `constant * P(k + 1)/P(k) * residual = source` is then checked by
204    // exact cross-multiplication over the coefficient field. Any corrupted datum
205    // breaks one of the two checks.
206    let mut leading_part = polynomial_one();
207    let mut trailing_part = polynomial_one();
208    for step in &certificate.orbit_steps {
209        if step.shift < 1 || step.multiplicity == 0 {
210            return false;
211        }
212        let window = match orbit_window(&step.representative, step.shift) {
213            Some(window) => window,
214            None => return false,
215        };
216        for _ in 0..step.multiplicity {
217            if step.numerator_leads {
218                leading_part = polynomial_multiply(&leading_part, &window);
219            } else {
220                trailing_part = polynomial_multiply(&trailing_part, &window);
221            }
222        }
223    }
224    let rebuilt = ExactRationalFunction {
225        numerator: leading_part,
226        denominator: trailing_part,
227    };
228    if !rational_functions_equal(&rebuilt, &certificate.rational_part) {
229        return false;
230    }
231    let shifted = rational_function_shift(&certificate.rational_part, 1);
232    let quotient = ExactRationalFunction {
233        numerator: polynomial_multiply(&shifted.numerator, &certificate.rational_part.denominator),
234        denominator: polynomial_multiply(
235            &shifted.denominator,
236            &certificate.rational_part.numerator,
237        ),
238    };
239    if polynomial_is_zero(&quotient.denominator) {
240        return false;
241    }
242    let scaled = ExactRationalFunction {
243        numerator: polynomial_scale(&quotient.numerator, &certificate.constant),
244        denominator: quotient.denominator,
245    };
246    let reconstructed = rational_function_multiply(&scaled, &certificate.residual);
247    rational_functions_equal(&reconstructed, &certificate.source)
248}

250pub(crate) fn is_complete_antiquotient(certificate: &ShiftQuotientCertificate) -> bool {
251    // The rational part is a complete anti-quotient exactly when no constant
252    // content and no residual survive the reduction.
253    gaussian_is_one(&certificate.constant)
254        && polynomial_is_one(&certificate.residual.numerator)
255        && polynomial_is_one(&certificate.residual.denominator)
256}

309        let certificate = reduce_shift_orbits(&cube_ratio()).expect("reduction succeeds");
