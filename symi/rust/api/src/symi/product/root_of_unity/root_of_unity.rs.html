18/// Descriptor for the homogeneous quadratic conjugate-root family
19/// `a^2 - 2*a*b*cos(k*pi/n) + b^2`, extracted as the three index-free quantities
20/// the theorem actually needs: `first_square = a^2`, `second_square = b^2`, and
21/// `mixed = a*b`. Working with the squares and the mixed product avoids
22/// introducing square roots, and the premise `mixed^2 = first_square*second_square`
23/// is what ties them back to a genuine conjugate-root pair.
24#[derive(Clone, Debug)]
25pub(crate) struct RootOfUnityDescriptor {
26    pub(crate) first_square: ExpressionReference,
27    pub(crate) second_square: ExpressionReference,
28    pub(crate) mixed: ExpressionReference,
29    pub(crate) degree: ExpressionReference,
30    pub(crate) angle: ExpressionReference,
31}

33/// Proof data for the root-of-unity factorisation. The ledger records that the
34/// paired Euler factors run over the complete set of `2*n`-th roots of unity with
35/// exactly the roots `+1` and `-1` omitted, and that the exceptional value on the
36/// stratum `a^2 = b^2` was obtained from the finite geometric identity.
37#[derive(Clone, Debug)]
38pub(crate) struct RootOfUnityCertificate {
39    pub(crate) descriptor: RootOfUnityDescriptor,
40    pub(crate) omitted_roots: Vec<i64>,
41    pub(crate) generic_value: ExpressionReference,
42    pub(crate) exceptional_value: ExpressionReference,
43    pub(crate) theorems: Vec<TheoremIdentifier>,
44}

46pub(crate) fn root_of_unity_product(
47    context: &Context,
48    multiplicand: &ExpressionReference,
49    index: SymbolIdentifier,
50    lower_bound: &ExpressionReference,
51    upper_bound: &ExpressionReference,
52) -> Result<ExpressionReference, ProductDecline> {
53    // Parametric root-of-unity product. Pairing the Euler factors
54    // `(a - b*exp(i*k*pi/n))` and `(a - b*exp(-i*k*pi/n))` over `k = 1..n-1` sweeps
55    // every `2*n`-th root of unity except `+1` and `-1`, so the product is the
56    // complete factorisation of `a^(2 n) - b^(2 n)` with those two roots divided
57    // out. The quotient presentation has a removable value on `a^2 = b^2`, and the
58    // returned piecewise keeps it, because returning only the quotient would be a
59    // wrong answer on that stratum.
60    let descriptor = extract_descriptor(context, multiplicand, index, lower_bound, upper_bound)?;
61    if !proved_positive_integer(context, &descriptor.degree) {
62        return Err(ProductDecline::InsufficientAssumptions);
63    }
64    if !reconstructs_exactly(context, multiplicand, &descriptor) {
65        return Err(ProductDecline::CertificateFailed);
66    }
67    if !conjugate_premise_holds(context, &descriptor) {
68        return Err(ProductDecline::NotApplicable);
69    }
70    let generic_value = generic_quotient(context, &descriptor);
71    let exceptional_value = exceptional_stratum_value(context, &descriptor);
72    let certificate = RootOfUnityCertificate {
73        descriptor: descriptor.clone(),
74        omitted_roots: vec![1, -1],
75        generic_value: Arc::clone(&generic_value),
76        exceptional_value: Arc::clone(&exceptional_value),
77        theorems: vec![TheoremIdentifier::RootOfUnityFactorisation],
78    };
79    if !replay_root_of_unity(context, &certificate) {
80        return Err(ProductDecline::CertificateFailed);
81    }
82    let distinct = builders::not_equal(
83        context,
84        Arc::clone(&descriptor.first_square),
85        Arc::clone(&descriptor.second_square),
86    );
87    let coincident = builders::equal(
88        context,
89        Arc::clone(&descriptor.first_square),
90        Arc::clone(&descriptor.second_square),
91    );
92    let value = builders::piecewise(
93        context,
94        vec![(distinct, generic_value), (coincident, exceptional_value)],
95    );
96    if expression_node_count(&value) > budgets::MAXIMUM_EXPRESSION_NODES {
97        return Err(ProductDecline::ResourceExhausted);
98    }
99    Ok(value)
100}

102pub(crate) fn replay_root_of_unity(
103    context: &Context,
104    certificate: &RootOfUnityCertificate,
105) -> bool {
106    // Replay the theorem instance. The omitted roots must be exactly `+1` and
107    // `-1`; the conjugate premise must hold exactly; and the exceptional value
108    // must follow from the finite geometric identity
109    // `(A^n - B^n)/(A - B) = sum(A^j * B^(n - 1 - j), j = 0..n-1)`, whose value at
110    // `B = A` is `n * A^(n - 1)` — derived here through the summation engine
111    // rather than by differentiating the quotient at its pole.
112    if certificate.omitted_roots != vec![1, -1] {
113        return false;
114    }
115    if !conjugate_premise_holds(context, &certificate.descriptor) {
116        return false;
117    }
118    removable_branch_follows_from_geometric_identity(context, &certificate.descriptor)
119}

121fn extract_descriptor(
122    context: &Context,
123    multiplicand: &ExpressionReference,
124    index: SymbolIdentifier,
125    lower_bound: &ExpressionReference,
126    upper_bound: &ExpressionReference,
127) -> Result<RootOfUnityDescriptor, ProductDecline> {
128    // The bounds must be exactly `1..n-1`, which fixes the degree as
129    // `upper + 1`; the multiplicand must then be a three-term sum whose only
130    // index-dependent term is an index-free multiple of `cos(k*pi/n)`.
131    if !matches!(&lower_bound.kind, ExpressionKind::Integer(value) if *value == Integer::from(1)) {
132        return Err(ProductDecline::NotApplicable);
133    }
134    let one = builders::integer_value(context, 1);
135    let degree = builders::sum(context, vec![Arc::clone(upper_bound), one]);
136    if depends_on(&degree, index) {
137        return Err(ProductDecline::NotApplicable);
138    }
139    let terms = match &multiplicand.kind {
140        ExpressionKind::Sum(children) if children.len() == 3 => children.clone(),
141        _ => return Err(ProductDecline::NotApplicable),
142    };
143    let mut cosine_term: Option<ExpressionReference> = None;
144    let mut squares: Vec<ExpressionReference> = Vec::new();
145    for term in terms {
146        if depends_on(&term, index) {
147            if cosine_term.is_some() {
148                return Err(ProductDecline::NotApplicable);
149            }
150            cosine_term = Some(term);
151        } else {
152            squares.push(term);
153        }
154    }
155    let cosine_term = match cosine_term {
156        Some(term) => term,
157        None => return Err(ProductDecline::NotApplicable),
158    };
159    if squares.len() != 2 {
160        return Err(ProductDecline::NotApplicable);
161    }
162    let (angle, coefficient) = match split_cosine_factor(context, &cosine_term, index) {
163        Some(split) => split,
164        None => return Err(ProductDecline::NotApplicable),
165    };
166    if !angle_is_index_times_pi_over_degree(context, &angle, index, &degree) {
167        return Err(ProductDecline::NotApplicable);
168    }
169    let minus_half = builders::rational_value(context, -1, 2);
170    let mixed = builders::product(context, vec![minus_half, coefficient]);
171    Ok(RootOfUnityDescriptor {
172        first_square: Arc::clone(&squares[0]),
173        second_square: Arc::clone(&squares[1]),
174        mixed,
175        degree,
176        angle,
177    })
178}

180fn split_cosine_factor(
181    context: &Context,
182    term: &ExpressionReference,
183    index: SymbolIdentifier,
184) -> Option<(ExpressionReference, ExpressionReference)> {
185    // Split an index-dependent term into its single cosine call and the
186    // index-free coefficient multiplying it.
187    match &term.kind {
188        ExpressionKind::FunctionCall(identifier, arguments)
189            if *identifier == FunctionIdentifier::Builtin(BuiltinFunction::Cos)
190                && arguments.len() == 1 =>
191        {
192            let one = builders::integer_value(context, 1);
193            Some((Arc::clone(&arguments[0]), one))
194        }
195        ExpressionKind::Product(children) => {
196            let mut angle: Option<ExpressionReference> = None;
197            let mut remaining: Vec<ExpressionReference> = Vec::new();
198            for child in children {
199                if let ExpressionKind::FunctionCall(identifier, arguments) = &child.kind {
200                    if *identifier == FunctionIdentifier::Builtin(BuiltinFunction::Cos)
201                        && arguments.len() == 1
202                        && angle.is_none()
203                    {
204                        angle = Some(Arc::clone(&arguments[0]));
205                        continue;
206                    }
207                }
208                if depends_on(child, index) {
209                    return None;
210                }
211                remaining.push(Arc::clone(child));
212            }
213            let angle = angle?;
214            let coefficient = if remaining.is_empty() {
215                builders::integer_value(context, 1)
216            } else {
217                builders::product(context, remaining)
218            };
219            Some((angle, coefficient))
220        }
221        _ => None,
222    }
223}

225fn angle_is_index_times_pi_over_degree(
226    context: &Context,
227    angle: &ExpressionReference,
228    index: SymbolIdentifier,
229    degree: &ExpressionReference,
230) -> bool {
231    // The cosine phase must be exactly `k*pi/n`. The check is exact: the
232    // difference against the constructed phase must simplify to zero, so a phase
233    // that merely evaluates to the same values at sampled integers is refused.
234    let index_symbol = context.interner.intern(ExpressionKind::Symbol(index));
235    let pi = builders::constant(context, BuiltinConstant::Pi);
236    let minus_one = builders::integer_value(context, -1);
237    let reciprocal_degree = builders::power(context, Arc::clone(degree), minus_one);
238    let expected = builders::product(context, vec![index_symbol, pi, reciprocal_degree]);
239    let negated = builders::negate(context, expected);
240    let difference = builders::sum(context, vec![Arc::clone(angle), negated]);
241    let simplified = crate::simplify::simplify(context, difference);
242    matches!(&simplified.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
243}

245fn conjugate_premise_holds(context: &Context, descriptor: &RootOfUnityDescriptor) -> bool {
246    // `mixed^2 = first_square * second_square` is exactly the statement that the
247    // two paired factors are `a - b*exp(i*theta)` and `a - b*exp(-i*theta)` for a
248    // genuine `a`, `b`. Checked exactly, never by sampling.
249    let two = builders::integer_value(context, 2);
250    let squared = builders::power(context, Arc::clone(&descriptor.mixed), two);
251    let expected = builders::product(
252        context,
253        vec![
254            Arc::clone(&descriptor.first_square),
255            Arc::clone(&descriptor.second_square),
256        ],
257    );
258    let negated = builders::negate(context, expected);
259    let difference = builders::sum(context, vec![squared, negated]);
260    let expanded = crate::expand::expand(context, &difference);
261    let simplified = crate::simplify::simplify(context, expanded);
262    matches!(&simplified.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
263}

265fn reconstructs_exactly(
266    context: &Context,
267    multiplicand: &ExpressionReference,
268    descriptor: &RootOfUnityDescriptor,
269) -> bool {
270    // The multiplicand must reconstruct exactly from the extracted descriptor as
271    // `first_square + second_square - 2*mixed*cos(angle)`.
272    let cosine = builders::function_call(
273        context,
274        FunctionIdentifier::Builtin(BuiltinFunction::Cos),
275        vec![Arc::clone(&descriptor.angle)],
276    );
277    let minus_two = builders::integer_value(context, -2);
278    let cross = builders::product(
279        context,
280        vec![minus_two, Arc::clone(&descriptor.mixed), cosine],
281    );
282    let rebuilt = builders::sum(
283        context,
284        vec![
285            Arc::clone(&descriptor.first_square),
286            Arc::clone(&descriptor.second_square),
287            cross,
288        ],
289    );
290    let negated = builders::negate(context, rebuilt);
291    let difference = builders::sum(context, vec![Arc::clone(multiplicand), negated]);
292    let simplified = crate::simplify::simplify(context, difference);
293    matches!(&simplified.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
294}

296fn generic_quotient(context: &Context, descriptor: &RootOfUnityDescriptor) -> ExpressionReference {
297    // `(A^n - B^n)/(A - B)` with `A = a^2` and `B = b^2`, which is the homogeneous
298    // difference of powers `(a^(2 n) - b^(2 n))/(a^2 - b^2)`.
299    let first_power = builders::power(
300        context,
301        Arc::clone(&descriptor.first_square),
302        Arc::clone(&descriptor.degree),
303    );
304    let second_power = builders::power(
305        context,
306        Arc::clone(&descriptor.second_square),
307        Arc::clone(&descriptor.degree),
308    );
309    let negated_second = builders::negate(context, second_power);
310    let numerator = builders::sum(context, vec![first_power, negated_second]);
311    let negated_square = builders::negate(context, Arc::clone(&descriptor.second_square));
312    let denominator = builders::sum(
313        context,
314        vec![Arc::clone(&descriptor.first_square), negated_square],
315    );
316    builders::divide(context, numerator, denominator)
317}

319fn exceptional_stratum_value(
320    context: &Context,
321    descriptor: &RootOfUnityDescriptor,
322) -> ExpressionReference {
323    // The removable value `n * A^(n - 1)` on the stratum `A = B`.
324    let minus_one = builders::integer_value(context, -1);
325    let reduced_degree = builders::sum(context, vec![Arc::clone(&descriptor.degree), minus_one]);
326    let power = builders::power(
327        context,
328        Arc::clone(&descriptor.first_square),
329        reduced_degree,
330    );
331    builders::product(context, vec![Arc::clone(&descriptor.degree), power])
332}

334fn removable_branch_follows_from_geometric_identity(
335    context: &Context,
336    descriptor: &RootOfUnityDescriptor,
337) -> bool {
338    // `(A^n - B^n)/(A - B) = sum(A^j * B^(n - 1 - j), j = 0..n-1)` is the finite
339    // geometric identity. On the stratum `B = A` every summand is `A^(n - 1)`, so
340    // the sum is `n * A^(n - 1)`. The derivation is carried out here through the
341    // summation engine on fresh generic symbols, so the removable value is proved
342    // rather than asserted.
343    let base = builders::symbol(context, "root_of_unity_generic_base");
344    let summation_index = context
345        .symbol_table
346        .intern_symbol("root_of_unity_geometric_index");
347    let minus_one = builders::integer_value(context, -1);
348    let reduced_degree = builders::sum(context, vec![Arc::clone(&descriptor.degree), minus_one]);
349    let summand = builders::power(context, Arc::clone(&base), Arc::clone(&reduced_degree));
350    let zero = builders::integer_value(context, 0);
351    let summation = builders::summation(
352        context,
353        summand,
354        summation_index,
355        Some(zero),
356        Some(Arc::clone(&reduced_degree)),
357    );
358    let evaluated = crate::execute::execute(context, &summation);
359    let expected_power = builders::power(context, Arc::clone(&base), reduced_degree);
360    let expected = builders::product(
361        context,
362        vec![Arc::clone(&descriptor.degree), expected_power],
363    );
364    let negated = builders::negate(context, expected);
365    let difference = builders::sum(context, vec![evaluated, negated]);
366    let simplified = crate::simplify::simplify(context, difference);
367    matches!(&simplified.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
368}

375        // The Wester shape `x^2 - 2 x cos(k pi/n) + 1`, whose descriptor is
376        // first_square = x^2, second_square = 1, mixed = x.

380        let second_square = builders::integer_value(context, 1);
381        let degree = builders::symbol(context, "n");
382        let index = context.symbol_table.intern_symbol("k");
383        let index_symbol = context.interner.intern(ExpressionKind::Symbol(index));
384        let pi = builders::constant(context, BuiltinConstant::Pi);
385        let minus_one = builders::integer_value(context, -1);
386        let reciprocal_degree = builders::power(context, Arc::clone(&degree), minus_one);
387        let angle = builders::product(context, vec![index_symbol, pi, reciprocal_degree]);
388        RootOfUnityDescriptor {
389            first_square,
390            second_square,

392            degree,
