46/// A name resolved either from a fixed builtin spelling or from the symbol registry.
47///
48/// A builtin spelling is a compile-time constant and a registry name is a shared handle,
49/// so resolving a name costs no allocation in either case. The type exists because a
50/// registry name cannot be returned as a borrow: the registry is shared and the name
51/// must outlive the lock that read it.
52#[derive(Clone, Debug)]
53pub enum ResolvedName {
54    /// A fixed spelling belonging to a builtin function or constant.
55    Builtin(&'static str),
56    /// A name published in the symbol registry.
57    Published(Arc<str>),
58}
59
60impl std::ops::Deref for ResolvedName {
61    type Target = str;
62
63    fn deref(&self) -> &str {
64        match self {
65            ResolvedName::Builtin(name) => name,
66            ResolvedName::Published(name) => name,
67        }
68    }
69}

103/// How many times a durable mutation rebuilds itself against a base that replaced its own.
104///
105/// A rebuild is needed only when another mutation of the same table was published between
106/// this one's capture and its publication, and each rebuild is preceded by a publication that
107/// succeeded, so the bound is a starvation limit rather than an expected cost.
108const DURABLE_PUBLICATION_ATTEMPTS: usize = 64;
109
110#[derive(Debug)]
111/// Public native data structure named `symbol_table`.
112pub struct SymbolTable {
113    // Names are append-only and shared: a scoped table derived for a temporary
114    // assumption overlay publishes into the same registry as the table it came from, so a
115    // symbol a scoped operation mints is already resolvable when the result carrying it
116    // is returned. Facts are per-table and never shared, which is what keeps a scope's
117    // assumptions out of the durable state it was derived from.
118    registry: Arc<SymbolRegistry>,
119    // Durable assertions and the two indexes derived from them are one immutable value,
120    // republished as a whole. A query captures it and reads it outside every lock, so it
121    // sees the assertions and the indexes of one revision; a mutation builds its
122    // replacement against the value it captured and publishes only while that value is
123    // still current. Declaring an assumption is therefore a change to the table's state
124    // rather than to the table, and a query needs no exclusive access to consult it.
125    //
126    // The store is held by handle rather than by value so that the tables of the
127    // concurrent operations of one session can share one durable state. A scoped table,
128    // by contrast, is given a store of its own: a temporary overlay must not be visible
129    // to anything but the scope that declared it.
130    durable_assumptions: Arc<AssumptionSnapshotStore<DurableAssumptions>>,
131    // Facts about symbols the engine itself minted. The index is shared by handle for the
132    // same reason the registry is: a binder minted while a set is being built is named by
133    // the expression that leaves with it, so the operation that later reads that
134    // expression must still find the binder's domain. Nothing else can observe the fact,
135    // because no other operation can mention a symbol it did not mint.
136    binder_facts: Arc<RwLock<AssumptionFactIndex>>,
137    assumption_execution_mode: AtomicU8,
138}

161    /// Derive a table that shares this table's durable state rather than copying it.
162    ///
163    /// This is the table one operation of a session runs against. It shares the name
164    /// registry, the durable assumption store, and the engine-minted binder facts, so a
165    /// declaration the operation makes is a declaration of the session and a query it
166    /// asks sees everything the session has declared. What it does not share is the
167    /// ambient state of the context it belongs to, which is what makes two operations of
168    /// one session independent of one another.
169    pub(crate) fn sharing_durable_state(&self) -> SymbolTable {
170        SymbolTable {
171            registry: Arc::clone(&self.registry),
172            durable_assumptions: Arc::clone(&self.durable_assumptions),
173            binder_facts: Arc::clone(&self.binder_facts),
174            assumption_execution_mode: AtomicU8::new(
175                self.assumption_execution_mode.load(Ordering::Acquire),
176            ),
177        }
178    }

239    /// Return a stable identifier for the complete assumption snapshot used by a query.
240    ///
241    /// Every node of every memoised traversal asks this question, so it answers from state the
242    /// publication already computed rather than rebuilding a report and hashing its aggregate
243    /// formula per node. The published assertion flag settles it outright when it is clear: a
244    /// state with no assertion answers every query the way any other assertion-free state does,
245    /// and they therefore share one identity, so nothing has to be captured to distinguish them.
246    pub(crate) fn assumption_snapshot_identifier(&self) -> u64 {
247        if !self.durable_assumptions.may_carry_an_assertion() {
248            return ASSERTION_FREE_SNAPSHOT_IDENTIFIER;
249        }
250        self.durable_assumptions.capture().snapshot_identifier()
251    }

253    /// Return whether any durable assertion could bear on this scalar subject.
254    pub(crate) fn assumptions_bear_on(
255        &self,
256        subject: &crate::expression::ExpressionReference,
257    ) -> bool {
258        // The gate in front of the query session. It is a property of the subject
259        // rather than of the context: a declaration about some other symbol cannot
260        // move this subject's verdict, and asking anyway is what made building an
261        // expression cost a proof search once any symbol anywhere was declared.
262        if !self.durable_assumptions.may_carry_an_assertion() {
263            return false;
264        }
265        let durable = self.durable_assumptions.capture();
266        !durable.is_empty() && durable.relevance().bears_on(subject)
267    }

269    /// Apply a durable assumption mutation by optimistic publication.
270    ///
271    /// The replacement is built and validated against a captured base with no lock of any kind
272    /// held, so the model searches a declaration performs never block a query and a validation
273    /// that itself declares something cannot deadlock against the mutation it runs inside. The
274    /// replacement becomes visible only while its captured base is still the published state; a
275    /// mutation that lost that race is built again against the state that replaced it. A build
276    /// that leaves the revision where it found it changed nothing and is not published.
277    fn publish_durable_mutation<BuildType>(
278        &self,
279        build: BuildType,
280    ) -> Result<(), AssumptionMutationError>
281    where
282        BuildType:
283            Fn(&DurableAssumptions) -> Result<AssumptionKnowledgeBase, AssumptionMutationError>,
284    {
285        for _ in 0..DURABLE_PUBLICATION_ATTEMPTS {
286            let base = self.durable_assumptions.capture();
287            let mutated = build(&base)?;
288            if mutated.revision() == base.knowledge_base().revision() {
289                return Ok(());
290            }
291            match self
292                .durable_assumptions
293                .publish(&base, DurableAssumptions::derived_from(mutated))
294            {
295                Ok(_) => return Ok(()),
296                Err(AssumptionSnapshotPublicationError::Conflict) => continue,
297                Err(AssumptionSnapshotPublicationError::RevisionExhausted) => {
298                    return Err(AssumptionMutationError::ResourceExhausted);
299                }
300            }
301        }
302        Err(AssumptionMutationError::ResourceExhausted)
303    }

306    pub fn declared_assumptions_of(&self, identifier: SymbolIdentifier) -> Vec<AssumptionFormula> {
307        // The lossless replacement for the removed five-field projection. Every

313    /// Declare a canonical formula as one symbol's declaration, rejecting conflicts.
314    pub fn declare_assumptions(
315        &self,
316        identifier: SymbolIdentifier,
317        formula: AssumptionFormula,
318    ) -> Result<(), AssumptionMutationError> {
319        // The validated declaration path used by the symbol constructors. The
320        // consistency check is scoped to this symbol's own declaration rather than to
321        // the whole knowledge base: declaring a symbol is a statement about that
322        // symbol, and probing the entire base would make every `symbol(...)` call pay
323        // for a model search over every fact in the session. A declaration that
324        // contradicts a fact about some *other* subject is still caught when the two
325        // are queried together.
326        self.publish_durable_mutation(|base| {
327            let scoped = self.query_handler_snapshot();
328            for existing in declared_assumptions_in(base, identifier) {
329                scoped.add_formula_with_provenance(
330                    existing,
331                    AssumptionProvenance::SymbolDeclaration,
332                )?;
333            }
334            scoped.add_formula_with_provenance(
335                formula.clone(),
336                AssumptionProvenance::SymbolDeclaration,
337            )?;
338            let mut knowledge_base = base.knowledge_base_to_mutate();
339            knowledge_base.replace_symbol_declaration(formula.clone());
340            Ok(knowledge_base)
341        })
342    }

344    /// Replace one symbol's declaration with a list of scalar predicates.
345    pub fn declare_symbol_predicates(
346        &self,
347        identifier: SymbolIdentifier,
348        predicates: &[CurrentScalarPredicate],
349    ) {
350        // The canonical one-call replacement for reading the old five-field record,
351        // mutating a field, and writing it back. An empty list clears the symbol's
352        // declaration, which is what the old default record meant.
353        let subject = Arc::new(make_expression(ExpressionKind::Symbol(identifier)));
354        self.replace_declared_assumptions(
355            identifier,
356            crate::assumptions::scalar_declaration(subject, predicates),
357        );

360    /// Replace one symbol's declaration with a canonical formula, without validation.
361    pub fn replace_declared_assumptions(
362        &self,
363        identifier: SymbolIdentifier,
364        formula: AssumptionFormula,
365    ) {
366        // Replacement rather than accumulation, for the internal probes that force a
367        // reserved or freshly interned symbol into a domain and must not stack a
368        // contradiction onto whatever that symbol carried before. Callers that need
369        // the conflict check use `declare_assumptions`.
370        //
371        // Replacement validates nothing, so the publication can only decline once the
372        // durable revision of this session has run out of room, which takes more mutations
373        // than a session can perform. The state is then left exactly as it was rather than
374        // published against a base no query will observe.
375        let _ = self.publish_durable_mutation(|base| {
376            let mut knowledge_base = base.knowledge_base_to_mutate();
377            knowledge_base.clear_symbol_facts(identifier);
378            if formula != AssumptionFormula::LogicalTrue {
379                knowledge_base.replace_symbol_declaration(formula.clone());
380            }
381            Ok(knowledge_base)
382        });
383    }

393    fn add_formula_with_provenance(
394        &self,
395        formula: AssumptionFormula,
396        provenance: AssumptionProvenance,
397    ) -> Result<(), AssumptionMutationError> {
398        self.publish_durable_mutation(|base| {
399            let base_formula = base.knowledge_base().report().formula;
400            let entailment_probe = formula_with_handler_facts(
401                self,
402                AssumptionFormula::and([
403                    base_formula.clone(),
404                    AssumptionFormula::not(formula.clone()),
405                ]),
406            );
407            let validation_formula = formula_with_handler_facts(
408                self,
409                AssumptionFormula::and([base_formula, formula.clone()]),
410            );
411            let mut knowledge_base = base.knowledge_base_to_mutate();
412            knowledge_base.add_with_validation_formulas(
413                formula.clone(),
414                entailment_probe,
415                validation_formula,
416                provenance,
417            )?;
418            Ok(knowledge_base)
419        })
420    }

450    /// Record an internally minted symbol's own domain, outside the durable base.
451    pub(crate) fn declare_binder_predicates(
452        &self,
453        identifier: SymbolIdentifier,
454        predicates: &[CurrentScalarPredicate],
455    ) {
456        // A binder minted for an image_set, or a reserved placeholder an internal probe
457        // substitutes in and immediately substitutes away, is scaffolding of the operation
458        // that minted it — not a declaration the caller made about their context.
459        // Recording its domain as a durable assertion would make every later query work
460        // against a larger base, would move the durable revision and so discard the
461        // compiled query session, and would leave the operation's scaffolding behind in
462        // the context it ran in. The domain is recorded here instead — at the binding
463        // site, where the structural predicate handlers read it and where it costs nothing
464        // that grows. The subject is a symbol the engine mints and owns, so the fact
465        // cannot contradict a declaration the caller made.
466        let subject = Arc::new(make_expression(ExpressionKind::Symbol(identifier)));
467        let mut binder_facts = self
468            .binder_facts
469            .write()
470            .unwrap_or_else(|poisoned| poisoned.into_inner());
471        for predicate in predicates {
472            binder_facts.add_scalar_fact_for_subject(
473                PredicateIdentifier::CurrentScalar(*predicate),
474                Arc::clone(&subject),
475            );
476        }
477    }

479    pub(crate) fn direct_scalar_fact(
480        &self,
481        identifier: SymbolIdentifier,
482        predicate: CurrentScalarPredicate,
483    ) -> TruthValue {
484        let subject = Arc::new(make_expression(ExpressionKind::Symbol(identifier)));
485        let formula =
486            AssumptionFormula::atom(PredicateIdentifier::CurrentScalar(predicate), subject);
487        self.durable_assumptions
488            .capture()
489            .fast_facts()
490            .direct_verdict(&formula, crate::assumptions::AssumptionQueryKind::Ask)
491            .or_else(|| {
492                self.binder_facts
493                    .read()
494                    .unwrap_or_else(|poisoned| poisoned.into_inner())
495                    .direct_verdict(&formula, crate::assumptions::AssumptionQueryKind::Ask)
496            })
497            .unwrap_or(TruthValue::Unknown)
498    }

570    /// Allocate a fresh context-local symbol name.
571    pub fn fresh(&self, prefix: &str) -> SymbolIdentifier {
572        // The probe and the publication are one linearizable registry transition, so two
573        // callers minting from the same prefix at the same time can never be handed the
574        // same name and no candidate is observable before it is published.
575        record_counter!(fresh_symbol_mints);
576        let allocation = self.registry.fresh(prefix);
577        record_counter!(fresh_symbol_name_probes, allocation.probe_count);
578        allocation.identifier
579    }

587/// Return every assertion in one captured durable state whose subject mentions a symbol.
588fn declared_assumptions_in(
589    durable: &DurableAssumptions,
590    identifier: SymbolIdentifier,
591) -> Vec<AssumptionFormula> {
592    durable
593        .knowledge_base()
594        .explicit_assertions()
595        .iter()
596        .filter(|assertion| formula_mentions_symbol(&assertion.formula, identifier))
597        .map(|assertion| assertion.formula.clone())
598        .collect()
599}

632    fn a_declaration_published_through_a_shared_reference_is_visible_to_a_query() {

636        let table = SymbolTable::new();
