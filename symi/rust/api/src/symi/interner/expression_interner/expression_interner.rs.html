8const NUMBER_OF_SHARDS: usize = 32;
9const SHARD_INDEX_BITS: u32 = NUMBER_OF_SHARDS.trailing_zeros();

11/// The number of dead entries the whole store tolerates before a sweep can be triggered at all.
12///
13/// Reclamation is amortized by the factor-of-two growth trigger beside it; this floor only keeps
14/// a table that is still small from being rescanned on every insertion. It is a property of the
15/// store rather than of one shard, and it is divided by the shard count below for exactly that
16/// reason: a floor applied per shard multiplies the dead entries the store tolerates by the shard
17/// count, and every probe then walks a table whose slots are mostly weak references to
18/// expressions that no longer exist. That costs a probe far more than the sweeps it avoids,
19/// because probing is what every expression construction in a session pays for.
20const AGGREGATE_SWEEP_SIZE_FLOOR: usize = 1024;
21const SWEEP_SIZE_FLOOR: usize = AGGREGATE_SWEEP_SIZE_FLOOR / NUMBER_OF_SHARDS;
22
23const _: () = assert!(
24    NUMBER_OF_SHARDS.is_power_of_two(),
25    "the shard index is taken from the high hash bits, which requires a power-of-two shard count"
26);
27
28const _: () = assert!(
29    SWEEP_SIZE_FLOOR >= 1,
30    "a shard whose sweep floor rounds to zero would rescan its table on every insertion"
31);

38/// A yield injected between the steps of one interning transition.
39///
40/// Interning probes, upgrades a weak reference or inserts a strong one, and sweeps, all inside one
41/// short critical section. Windows that narrow are exactly the windows a test cannot reach by
42/// luck, so a stress test may ask for a yield at each step and force the interleavings that a
43/// barrier alone leaves to the scheduler. The hook is off everywhere else and costs one branch on
44/// a copied flag.
45#[derive(Clone, Copy, Default)]
46struct TransitionHook {
47    yields_between_steps: bool,
48}
49
50impl TransitionHook {
51    fn reached(self) {
52        if self.yields_between_steps {
53            std::thread::yield_now();
54        }
55    }
56}

58/// The store work one interning call performed, accumulated locally and recorded once.
59///
60/// Every field is a count a probe would otherwise report one transition at a time, which would
61/// make the instrumented measurement of the store dominated by the instrumentation. Accumulating
62/// in a local and recording once at the end of the call keeps the counts exact while leaving the
63/// uninstrumented build with arithmetic on locals that never escape.
64#[derive(Default)]
65struct ProbeCost {
66    hits: usize,
67    contentions: usize,
68    read_lock_acquisitions: usize,
69    write_lock_acquisitions: usize,
70    shared_probes: usize,
71    duplicate_probes: usize,
72    hash_comparisons: usize,
73    kind_comparisons: usize,
74    weak_upgrades: usize,
75    dead_entry_visits: usize,
76    publications: usize,
77    reused_expired_entries: usize,
78    expired_slot_comparisons: usize,
79    sweeps: usize,
80    swept_entries: usize,
81}
82
83impl ProbeCost {
84    fn record(&self) {
85        if self.shared_probes == 0 && self.write_lock_acquisitions == 0 {
86            return;
87        }
88        crate::instrumentation::update(|counters| {
89            counters.expression_interner_hits += self.hits;
90            counters.expression_interner_contentions += self.contentions;
91            counters.expression_interner_read_lock_acquisitions += self.read_lock_acquisitions;
92            counters.expression_interner_write_lock_acquisitions += self.write_lock_acquisitions;
93            counters.expression_interner_shared_probes += self.shared_probes;
94            counters.expression_interner_duplicate_probes += self.duplicate_probes;
95            counters.expression_interner_hash_comparisons += self.hash_comparisons;
96            counters.expression_interner_kind_comparisons += self.kind_comparisons;
97            counters.expression_interner_weak_upgrades += self.weak_upgrades;
98            counters.expression_interner_dead_entry_visits += self.dead_entry_visits;
99            counters.expression_interner_publications += self.publications;
100            counters.expression_interner_reused_expired_entries += self.reused_expired_entries;
101            counters.expression_interner_expired_slot_comparisons += self.expired_slot_comparisons;
102            counters.expression_interner_sweeps += self.sweeps;
103            counters.expression_interner_swept_entries += self.swept_entries;
104        });
105    }
106}

108struct InternerShard {
109    table: HashTable<InternerEntry>,
110    live_count_at_last_sweep: usize,
111}
112
113impl InternerShard {
114    fn new() -> Self {
115        InternerShard {
116            table: HashTable::new(),
117            live_count_at_last_sweep: 0,
118        }
119    }
120
121    fn probe(&self, candidate: &Expression, cost: &mut ProbeCost) -> Option<ExpressionReference> {
122        // The shared-borrow half of interning: find a live canonical allocation for a
123        // candidate without the authority to insert one. Almost every construction in a
124        // session rebuilds a shape the session already holds, so this is the path that
125        // decides what building an expression costs, and it must neither take an exclusive
126        // lock nor upgrade a weak reference more than once. The upgraded value is carried
127        // out of the probe rather than the entry, because upgrading again after the search
128        // would pay a second reference-count pair for an answer already in hand. The cost
129        // record accumulates in a local so that a probe pays one instrumentation transition
130        // for a whole search rather than one for every slot it walks.
131        let candidate_hash = candidate.hash;
132        let mut found = None;
133        self.table.find(candidate_hash, |entry| {
134            cost.hash_comparisons += 1;
135            if entry.hash != candidate_hash {
136                return false;
137            }
138            cost.weak_upgrades += 1;
139            let Some(existing) = entry.expression.upgrade() else {
140                cost.dead_entry_visits += 1;
141                return false;
142            };
143            cost.kind_comparisons += 1;
144            if existing.kind != candidate.kind {
145                return false;
146            }
147            found = Some(existing);
148            true
149        });
150        found
151    }
152
153    fn intern(
154        &mut self,
155        candidate: Expression,
156        hook: TransitionHook,
157        cost: &mut ProbeCost,
158    ) -> ExpressionReference {
159        // The exclusive half. A probe under the shared lock may have missed, so this
160        // probes once more under exclusive access before inserting: between releasing the
161        // shared lock and taking this one another thread may have published the same
162        // shape, and two allocations for one shape would break the identity the store
163        // exists to provide.
164        cost.duplicate_probes += 1;
165        let existing = self.probe(&candidate, cost);
166        hook.reached();
167        if let Some(existing) = existing {
168            return existing;
169        }
170        hook.reached();
171        let candidate_hash = candidate.hash;
172        let reference = Arc::new(candidate);
173        let published = Arc::downgrade(&reference);
174        // Publishing into an expired slot of the same hash rather than beside it is what keeps a
175        // probe's cost proportional to the live store. A shape a session builds, drops and builds
176        // again — every intermediate of a rewrite, every candidate a search rejects — leaves an
177        // entry whose weak reference can never be upgraded again, and inserting unconditionally
178        // gives its successor a second entry under the same hash. The two are indistinguishable
179        // to a probe, which must reach and fail to upgrade every one of them before it can answer,
180        // so the shapes a session rebuilds most often are exactly the ones whose lookup degrades
181        // fastest. An expired slot is unobservable — a strong count of zero is terminal, so no
182        // caller can ever hold that allocation again — which makes overwriting it sound, and it
183        // bounds the entries sharing one hash by the number of live expressions that share it.
184        let expired_slot = self.table.find_mut(candidate_hash, |entry| {
185            cost.expired_slot_comparisons += 1;
186            entry.hash == candidate_hash && entry.expression.strong_count() == 0
187        });
188        match expired_slot {
189            Some(entry) => {
190                entry.expression = published;
191                cost.reused_expired_entries += 1;
192            }
193            None => {
194                self.table.insert_unique(
195                    candidate_hash,
196                    InternerEntry {
197                        hash: candidate_hash,
198                        expression: published,
199                    },
200                    |entry: &InternerEntry| entry.hash,
201                );
202            }
203        }
204        hook.reached();
205        cost.publications += 1;
206        self.sweep_if_grown(cost);
207        reference
208    }
209
210    fn sweep_if_grown(&mut self, cost: &mut ProbeCost) {
211        if self.table.len() <= 2 * self.live_count_at_last_sweep + SWEEP_SIZE_FLOOR {
212            return;
213        }
214        let length_before_sweep = self.table.len();
215        self.table
216            .retain(|entry| entry.expression.strong_count() > 0);
217        self.live_count_at_last_sweep = self.table.len();
218        cost.sweeps += 1;
219        cost.swept_entries += length_before_sweep - self.table.len();
220    }
221
222    fn length(&self) -> usize {
223        self.table
224            .iter()
225            .filter(|entry| entry.expression.strong_count() > 0)
226            .count()
227    }
228
229    #[cfg(test)]
230    fn retained_entry_count(&self) -> usize {
231        self.table.len()
232    }
233}

236    shard_index: usize,
237    candidate: Expression,

240/// The smallest and largest integers the store keeps a canonical strong reference to.
241///
242/// A census of the interner traffic of the public workload set attributes 55.7 per cent of every
243/// construction in it to an integer in this range, led by minus one, zero, one and two at 16.3,
244/// 14.8, 11.7 and 8.9 per cent: they are the coefficients of a negation, the additive identity a
245/// cancellation produces, the exponent every unexponentiated factor carries, and the exponent of
246/// every square. Each of those constructions hashed a value, took a shard lock and walked a table
247/// to be handed back an allocation the session was certain to already hold. The range stops where
248/// the census does — beyond sixty-four no single value contributes as much as a twentieth of one
249/// per cent — and each slot is filled only when a session first asks for it, so the retention is
250/// paid for by the sessions that use it.
251const RETAINED_INTEGER_LOWER_BOUND: i64 = -16;
252const RETAINED_INTEGER_UPPER_BOUND: i64 = 64;
253const RETAINED_INTEGER_COUNT: usize =
254    (RETAINED_INTEGER_UPPER_BOUND - RETAINED_INTEGER_LOWER_BOUND + 1) as usize;
255
256/// A weak, sharded expression hash-cons store.
257pub struct ExpressionInterner {
258    shards: Box<[RwLock<InternerShard>]>,
259    transitions: TransitionHook,
260    retained_integers: Box<[OnceLock<ExpressionReference>]>,
261}
262
263impl Default for ExpressionInterner {
264    fn default() -> Self {
265        Self::new()
266    }
267}

269impl ExpressionInterner {
270    /// Construct an empty concurrent expression store.
271    pub fn new() -> Self {
272        let shards = (0..NUMBER_OF_SHARDS)
273            .map(|_| RwLock::new(InternerShard::new()))
274            .collect::<Vec<_>>()
275            .into_boxed_slice();
276        let retained_integers = (0..RETAINED_INTEGER_COUNT)
277            .map(|_| OnceLock::new())
278            .collect::<Vec<_>>()
279            .into_boxed_slice();
280        ExpressionInterner {
281            shards,
282            transitions: TransitionHook::default(),
283            retained_integers,
284        }
285    }

287    /// Yield between the steps of every interning transition, for concurrency stress tests.
288    ///
289    /// This changes only the timing of the store's internal steps. It never changes which
290    /// reference is canonical, so a test that turns it on is testing the same interner.
291    pub fn yield_between_transitions(&mut self) {
292        self.transitions = TransitionHook {
293            yields_between_steps: true,
294        };
295    }
296
297    /// Return the canonical additive identity.
298    ///
299    /// Zero and one are the two expressions a session builds far more often than any other: every
300    /// additive identity test, every empty sum, every default exponent and every unit coefficient
301    /// asks for one of them. They are the leading entries of the retained small-integer table and
302    /// are answered from it, so this is a reference-count clone rather than a shard lock, a hash
303    /// and a probe for an allocation the store already held.
304    pub fn integer_zero(&self) -> ExpressionReference {
305        self.retained_integer(0)
306    }
307
308    /// Return the canonical multiplicative identity.
309    ///
310    /// The multiplicative half of [`ExpressionInterner::integer_zero`], answered the same way.
311    pub fn integer_one(&self) -> ExpressionReference {
312        self.retained_integer(1)
313    }
314
315    /// Return the retained canonical allocation of one small integer, publishing it if needed.
316    ///
317    /// Retaining a strong reference is sound because the store's own reference keeps the weak
318    /// table entry alive: a caller interning the same literal by any other route still probes,
319    /// upgrades that same allocation, and receives the identical canonical reference. Identity is
320    /// therefore decided in exactly one place whether or not a value is retained.
321    fn retained_integer(&self, value: i64) -> ExpressionReference {
322        record_counter!(expression_interner_retained_leaf_clones);
323        let slot_index = (value - RETAINED_INTEGER_LOWER_BOUND) as usize;
324        Arc::clone(self.retained_integers[slot_index].get_or_init(|| {
325            let mut cost = ProbeCost::default();
326            let reference = self.publish(
327                make_expression(ExpressionKind::Integer(
328                    crate::exact_arithmetic::Integer::from(value),
329                )),
330                &mut cost,
331            );
332            cost.record();
333            reference
334        }))
335    }
336
337    /// Return the retained allocation of a candidate integer literal, when the store holds one.
338    fn retained_integer_candidate(&self, kind: &ExpressionKind) -> Option<ExpressionReference> {
339        let ExpressionKind::Integer(value) = kind else {
340            return None;
341        };
342        let value = crate::exact_arithmetic::integer_to_i64(value)?;
343        if !(RETAINED_INTEGER_LOWER_BOUND..=RETAINED_INTEGER_UPPER_BOUND).contains(&value) {
344            return None;
345        }
346        Some(self.retained_integer(value))
347    }
348
349    /// Publish a candidate through the sharded store, recovering a poisoned shard.
350    fn publish(&self, candidate: Expression, cost: &mut ProbeCost) -> ExpressionReference {
351        match self.intern_reporting_poisoning(candidate, cost) {
352            Ok(reference) => reference,
353            Err(PoisonedShard {
354                shard_index,
355                candidate,
356            }) => {
357                let mut shard = self.shards[shard_index]
358                    .write()
359                    .unwrap_or_else(|poisoned| poisoned.into_inner());
360                cost.write_lock_acquisitions += 1;
361                shard.intern(candidate, self.transitions, cost)
362            }
363        }
364    }
365
366    /// Intern an immutable expression and return the canonical allocation.
367    ///
368    /// A poisoned shard is recovered rather than reported. The recovery is sound because the
369    /// shard's only invariant is the hash table's own, and every operation inside the critical
370    /// section either completes or leaves the table exactly as it found it: a panic there can
371    /// only come from allocation, which aborts. Recovering keeps one panicking expression from
372    /// making every later construction in the process fail. A caller that must observe the
373    /// poisoning instead uses [`ExpressionInterner::try_intern`].
374    pub fn intern(&self, kind: ExpressionKind) -> ExpressionReference {
375        if let Some(retained) = self.retained_integer_candidate(&kind) {
376            return retained;
377        }
378        let mut cost = ProbeCost::default();
379        let reference = self.publish(make_expression(kind), &mut cost);
380        cost.record();
381        reference
382    }
383
384    /// Intern an immutable expression while reporting lock poisoning.
385    pub fn try_intern(
386        &self,
387        kind: ExpressionKind,
388    ) -> Result<ExpressionReference, crate::interner::CanonicalStoreError> {
389        if let Some(retained) = self.retained_integer_candidate(&kind) {
390            return Ok(retained);
391        }
392        let mut cost = ProbeCost::default();
393        let outcome = self
394            .intern_reporting_poisoning(make_expression(kind), &mut cost)
395            .map_err(|_| crate::interner::CanonicalStoreError::Poisoned("expression interner"));
396        cost.record();
397        outcome
398    }
399
400    fn intern_reporting_poisoning(
401        &self,
402        candidate: Expression,
403        cost: &mut ProbeCost,
404    ) -> Result<ExpressionReference, PoisonedShard> {
405        // Interning is overwhelmingly a lookup rather than a publication: a session
406        // rebuilds shapes it already holds far more often than it mints new ones, and every
407        // symbol, small integer, and builtin constant an algorithm reaches for is already
408        // canonical. Probing under the shared lock first is therefore what decides both what
409        // one construction costs and whether simultaneous construction scales, because an
410        // exclusive lock taken for a lookup serializes every thread building against the
411        // same shard whether or not any of them has anything to publish.
412        //
413        // The exclusive path re-probes, so a shape published between the two acquisitions is
414        // still found rather than allocated twice. Identity is thus decided by the exclusive
415        // path alone and the shared path can only answer earlier, never differently.
416        let shard_index = self.shard_index(candidate.hash);
417        let shard_lock = &self.shards[shard_index];
418        match shard_lock.read() {
419            Ok(shard) => {
420                cost.read_lock_acquisitions += 1;
421                cost.shared_probes += 1;
422                if let Some(existing) = shard.probe(&candidate, cost) {
423                    cost.hits += 1;
424                    return Ok(existing);
425                }
426            }
427            Err(_) => {
428                return Err(PoisonedShard {
429                    shard_index,
430                    candidate,
431                });
432            }
433        }
434        self.transitions.reached();
435        let mut shard = match shard_lock.try_write() {
436            Ok(guard) => guard,
437            Err(std::sync::TryLockError::WouldBlock) => {
438                cost.contentions += 1;
439                match shard_lock.write() {
440                    Ok(guard) => guard,
441                    Err(_) => {
442                        return Err(PoisonedShard {
443                            shard_index,
444                            candidate,
445                        });
446                    }
447                }
448            }
449            Err(std::sync::TryLockError::Poisoned(_)) => {
450                return Err(PoisonedShard {
451                    shard_index,
452                    candidate,
453                });
454            }
455        };
456        cost.write_lock_acquisitions += 1;
457        Ok(shard.intern(candidate, self.transitions, &mut *cost))
458    }

460    /// Return the shard selected by a stable structural hash.
461    pub(crate) fn shard_index(&self, hash: u64) -> usize {
462        // The high bits choose the shard and the low bits choose the bucket inside it, so a
463        // structural hash that clusters in one shard still spreads across that shard's table.
464        // The shift follows the shard count rather than being written out, because tuning the
465        // count is expected and a stale shift would silently strand half the shards.
466        (hash >> (u64::BITS - SHARD_INDEX_BITS)) as usize & (NUMBER_OF_SHARDS - 1)
467    }
468
469    /// Return the number of live canonical expressions.
470    pub fn length(&self) -> usize {
471        self.shards
472            .iter()
473            .map(|shard| match shard.read() {
474                Ok(shard) => shard.length(),
475                Err(poisoned) => poisoned.into_inner().length(),
476            })
477            .sum()
478    }
479
480    /// Return whether the store has no live canonical expressions.
481    pub fn is_empty(&self) -> bool {
482        self.length() == 0
483    }
484
485    /// Return the number of entries the store retains, live and dead together.
486    ///
487    /// This is what a probe walks, and it is therefore the quantity reclamation exists to bound.
488    /// The live count alone cannot see a store whose tables have filled with weak references to
489    /// expressions that no longer exist, which is exactly the state that makes every construction
490    /// in a session slower without changing a single answer.
491    #[cfg(test)]
492    pub(crate) fn retained_entry_count(&self) -> usize {
493        self.shards
494            .iter()
495            .map(|shard| match shard.read() {
496                Ok(shard) => shard.retained_entry_count(),
497                Err(poisoned) => poisoned.into_inner().retained_entry_count(),
498            })
499            .sum()
500    }
