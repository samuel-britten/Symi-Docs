7/// Performs the advanced native `is_elementary` operation.
8pub fn is_elementary(expression: &ExpressionReference) -> TruthValue {
9    // Structural classification of whether an expression lies in some finite
10    // elementary tower over Q(x). The class is built from rational operations
11    // together with the algebraic, exponential, and logarithmic extensions, plus
12    // the standard analytic trigonometric and hyperbolic functions which are
13    // elementary over the complexes via their exp-form definitions. Functions
14    // outside this class (gamma, digamma, generalised hypergeometric, distri-
15    // butional sentinels) collapse to no; user-defined functions and unevaluated
16    // derivatives produce unknown so callers can refuse to commit either way.
17    use ExpressionKind::*;
18    match &expression.kind {
19        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) => TruthValue::Yes,
20        Constant(value) => is_constant_elementary(value),
21        Sum(children) | Product(children) => combine_elementary(children),
22        Power(base, exponent) => kleene_and(is_elementary(base), is_elementary(exponent)),
23        Factorial(argument) => is_factorial_elementary(argument),
24        FunctionCall(function, arguments) => is_function_call_elementary(*function, arguments),
25        Derivative(_, _, _) => TruthValue::Unknown,
26        Integral(_, _, _, _) => TruthValue::Unknown,
27        Summation(_, _, _, _) => TruthValue::Unknown,
28        IndexedProduct(_, _, _, _) => TruthValue::Unknown,
29        LaplaceTransform(_, _, _) => TruthValue::No,
30        InverseLaplaceTransform(_, _, _) => TruthValue::No,
31        FourierTransform(_, _, _) => TruthValue::No,
32        InverseFourierTransform(_, _, _) => TruthValue::No,
33        Piecewise(_) => TruthValue::Unknown,
34        OrdinaryDifferentialEquation(_, _, _)
35        | RecurrenceEquation(_, _, _)
36        | Interval(_, _, _, _)
37        | FiniteSet(_)
38        | SetUnion(_)
39        | SetIntersection(_)
40        | SetComplement(_, _)
41        | ImageSet(_, _, _)
42        | ConditionSet(_, _, _) => TruthValue::Unknown,
43        Distribution(_, _) | RandomVariable(_, _) | SpecialFunctionParameterVector(_) => {
44            TruthValue::No
45        }
46    }
47}

49fn is_constant_elementary(value: &BuiltinConstant) -> TruthValue {
50    // Classify each builtin constant against the elementary class. Algebraic
51    // and named transcendentals such as pi and e enter as constants of the
52    // field of constants; the infinity sentinels are not field elements at all
53    // and the undefined sentinel propagates as a refusal-to-classify.
54    match value {
55        BuiltinConstant::Pi | BuiltinConstant::E | BuiltinConstant::ImaginaryUnit => {
56            TruthValue::Yes
57        }
58        BuiltinConstant::Infinity | BuiltinConstant::NegativeInfinity => TruthValue::No,
59        BuiltinConstant::Undefined => TruthValue::Unknown,
60        _ => TruthValue::Unknown,
61    }
62}

64fn combine_elementary(children: &[ExpressionReference]) -> TruthValue {
65    // Reduce the elementary classifications of an n-ary node via three-valued
66    // conjunction. A single no shortcircuits to no; otherwise unknown
67    // dominates yes so the surrounding caller never falsely upgrades partial
68    // knowledge to a certain affirmative.
69    let mut result = TruthValue::Yes;
70    for child in children {
71        result = kleene_and(result, is_elementary(child));
72        if result == TruthValue::No {
73            return TruthValue::No;
74        }
75    }
76    result
77}

79fn is_factorial_elementary(argument: &ExpressionReference) -> TruthValue {
80    // Factorial is elementary only on integer constants — the analytic
81    // continuation factorial(x) = gamma(x+1) is not elementary. A symbolic
82    // argument therefore falls outside the class, while a literal
83    // non-negative integer reduces to a constant and is elementary.
84    if let ExpressionKind::Integer(value) = &argument.kind {
85        if *value >= Integer::ZERO {
86            return TruthValue::Yes;
87        }
88    }
89    TruthValue::No
90}

92fn is_function_call_elementary(
93    function: FunctionIdentifier,
94    arguments: &[ExpressionReference],
95) -> TruthValue {
96    // Builtin calls are elementary iff the builtin itself sits in the
97    // elementary class and every argument is also elementary. User-defined
98    // functions inject an unknown classification because we cannot inspect
99    // their definitions through this lattice without extending the
100    // assumptions machinery.
101    match function {
102        FunctionIdentifier::Builtin(builtin) => {
103            let head_classification = is_builtin_elementary(builtin);
104            if head_classification == TruthValue::No {
105                return TruthValue::No;
106            }
107            kleene_and(head_classification, combine_elementary(arguments))
108        }
109        FunctionIdentifier::User(_) => TruthValue::Unknown,
110    }
111}

113fn is_builtin_elementary(value: BuiltinFunction) -> TruthValue {
114    // Static lookup table for builtin elementarity. Trigonometric,
115    // hyperbolic, exponential, logarithmic, and square root sit inside the
116    // elementary class because their definitions over C reduce to exp/log on
117    // complex arguments. Inverse trigonometric functions inherit elementarity
118    // through their integral representation 1/(1 + x^2) and 1/sqrt(1 - x^2),
119    // both of which are elementary. abs and signum are not analytic; their
120    // piecewise definitions cannot be expressed by a finite tower of analytic
121    // extensions, so they fall outside the class. digamma and the
122    // special functions (error_function, error_function_complementary,
123    // exponential_integral, sine_integral, cosine_integral,
124    // logarithmic_integral, lambert_w) are non-elementary by Liouville — the
125    // special-function antiderivative-table layer is what bridges between the elementary
126    // pipeline and these targets. A generalized hypergeometric or Meijer-G call
127    // is non-elementary for generic parameters but elementary for particular
128    // ones — 0F0 is the exponential and 2F1(1, 1; 2; z) is a logarithm — so the
129    // class cannot be decided from the head alone and answers unknown.
130    match value {
131        BuiltinFunction::Sin
132        | BuiltinFunction::Cos
133        | BuiltinFunction::Tan
134        | BuiltinFunction::Sec
135        | BuiltinFunction::Csc
136        | BuiltinFunction::Cot
137        | BuiltinFunction::Arcsin
138        | BuiltinFunction::Arccos
139        | BuiltinFunction::Arctan
140        | BuiltinFunction::Sinh
141        | BuiltinFunction::Cosh
142        | BuiltinFunction::Tanh
143        | BuiltinFunction::Sech
144        | BuiltinFunction::Csch
145        | BuiltinFunction::Coth
146        | BuiltinFunction::Arcsinh
147        | BuiltinFunction::Arccosh
148        | BuiltinFunction::Arctanh
149        | BuiltinFunction::Arcsec
150        | BuiltinFunction::Arccsc
151        | BuiltinFunction::Arccot
152        | BuiltinFunction::Arcsech
153        | BuiltinFunction::Arccsch
154        | BuiltinFunction::Arccoth
155        | BuiltinFunction::Exp
156        | BuiltinFunction::Log
157        | BuiltinFunction::Sqrt => TruthValue::Yes,
158        BuiltinFunction::Abs

161        | BuiltinFunction::Gamma
162        | BuiltinFunction::LogGamma
163        | BuiltinFunction::Polygamma
164        | BuiltinFunction::Beta
165        | BuiltinFunction::Zeta
166        | BuiltinFunction::HurwitzZeta
167        | BuiltinFunction::Polylogarithm
168        | BuiltinFunction::LerchPhi
169        | BuiltinFunction::ErrorFunction
170        | BuiltinFunction::ErrorFunctionComplementary
171        | BuiltinFunction::ExponentialIntegral
172        | BuiltinFunction::SineIntegral
173        | BuiltinFunction::CosineIntegral
174        | BuiltinFunction::LogarithmicIntegral
175        | BuiltinFunction::LambertW => TruthValue::No,
176        BuiltinFunction::BesselJ
177        | BuiltinFunction::BesselY
178        | BuiltinFunction::BesselI
179        | BuiltinFunction::BesselK
