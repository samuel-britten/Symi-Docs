17/// Public native data structure named `tower_polynomial`.
18pub struct TowerPolynomial {
19    // An element of k[t] where t is the top monomial and the coefficient ring k
20    // is the field of the `lower` sub-tower. Coefficients are field elements of
21    // k, each a rational_function whose main variable is top_variable(lower).
22    // The coefficient list is kept canonical: sorted by strictly descending
23    // degree with every stored coefficient non-zero, so the zero polynomial is
24    // the empty list and `degree`/`valuation` read off the ends.
25    /// Stores the `monomial` value.
26    pub monomial: SymbolIdentifier,
27    /// Stores the `lower` value.
28    pub lower: DifferentialField,
29    /// Stores the `coefficients` value.
30    pub coefficients: Vec<(usize, RationalFunction)>,
31}
32
33fn lower_main(lower: &DifferentialField) -> SymbolIdentifier {
34    // The main variable used for every coefficient (k-element) of a
35    // tower_polynomial over this lower sub-tower.
36    top_variable(lower)
37}

39fn k_zero(lower: &DifferentialField) -> RationalFunction {
40    rational_function_zero(lower_main(lower))
41}

43fn k_one(lower: &DifferentialField) -> RationalFunction {
44    rational_function_one(lower_main(lower))
45}

47fn k_is_zero(value: &RationalFunction) -> bool {
48    rational_function_is_zero(value)

51fn k_normalize(lower: &DifferentialField, value: RationalFunction) -> RationalFunction {
52    // Coerce a k-element to the canonical main variable top_variable(lower) so
53    // the coefficient-arithmetic invariant holds even when the value was
54    // produced against a lower variable (e.g. a derivative of a constant).
55    lift_rational_function_to_main(lower_main(lower), value)
56}

126/// Performs the advanced native `coefficient_at` operation.
127pub fn coefficient_at(value: &TowerPolynomial, exponent: usize) -> RationalFunction {
128    for (stored_exponent, coefficient) in value.coefficients.iter() {

136fn from_sparse(
137    monomial: SymbolIdentifier,
138    lower: DifferentialField,
139    mut terms: Vec<(usize, RationalFunction)>,
140) -> TowerPolynomial {
141    // Build a canonical tower_polynomial from an arbitrary term list: combine
142    // duplicate degrees, drop zero coefficients, and sort descending.
143    let mut combined: Vec<(usize, RationalFunction)> = Vec::new();
144    terms.sort_by(|left, right| right.0.cmp(&left.0));
145    for (exponent, coefficient) in terms {
146        let coefficient = k_normalize(&lower, coefficient);
147        if k_is_zero(&coefficient) {
148            continue;
149        }
150        if let Some(last) = combined.last_mut() {
151            if last.0 == exponent {
152                last.1 = add_rational_functions(&last.1, &coefficient);
153                if k_is_zero(&last.1) {
154                    combined.pop();
155                }
156                continue;
157            }
158        }
159        combined.push((exponent, coefficient));
160    }
161    TowerPolynomial {
162        monomial,
163        lower,
164        coefficients: combined,
165    }
166}

193/// Performs the advanced native `scale` operation.
194pub fn scale(value: &TowerPolynomial, factor: &RationalFunction) -> TowerPolynomial {
195    // Multiply by a single k-element.
196    if k_is_zero(factor) {
197        return zero(value.monomial, value.lower.clone());
198    }
199    let factor = k_normalize(&value.lower, factor.clone());
200    let terms: Vec<(usize, RationalFunction)> = value
201        .coefficients
202        .iter()
203        .map(|(exponent, coefficient)| {
204            (*exponent, multiply_rational_functions(coefficient, &factor))
205        })
206        .collect();
207    from_sparse(value.monomial, value.lower.clone(), terms)
208}

210/// Performs the advanced native `multiply` operation.
211pub fn multiply(left: &TowerPolynomial, right: &TowerPolynomial) -> TowerPolynomial {
212    let mut terms: Vec<(usize, RationalFunction)> = Vec::new();
213    for (left_exponent, left_coefficient) in left.coefficients.iter() {
214        for (right_exponent, right_coefficient) in right.coefficients.iter() {
215            let product = multiply_rational_functions(left_coefficient, right_coefficient);
216            terms.push((left_exponent + right_exponent, product));
217        }
218    }
219    from_sparse(left.monomial, left.lower.clone(), terms)
220}

247/// Performs the advanced native `divide_by_t_power` operation.
248pub fn divide_by_t_power(value: &TowerPolynomial, exponent: usize) -> Option<TowerPolynomial> {
249    // Exact division by t^exponent: succeeds only when the t-adic valuation is
250    // at least `exponent`.
251    if exponent == 0 {
252        return Some(value.clone());
253    }
254    if is_zero(value) {
255        return Some(value.clone());
256    }
257    if valuation(value)? < exponent {
258        return None;
259    }
260    Some(TowerPolynomial {
261        monomial: value.monomial,
262        lower: value.lower.clone(),
263        coefficients: value
264            .coefficients
265            .iter()
266            .map(|(stored, coefficient)| (stored - exponent, coefficient.clone()))
267            .collect(),
268    })
269}

271/// Performs the advanced native `divide_with_remainder` operation.
272pub fn divide_with_remainder(
273    dividend: &TowerPolynomial,
274    divisor: &TowerPolynomial,
275) -> Option<(TowerPolynomial, TowerPolynomial)> {
276    // Euclidean division in k[t]. Because k is a field the quotient and
277    // remainder are total whenever the divisor is non-zero; the returned
278    // remainder has degree strictly less than the divisor.
279    if is_zero(divisor) {
280        return None;
281    }
282    let monomial = dividend.monomial;
283    let lower = dividend.lower.clone();
284    let divisor_degree = degree(divisor)?;
285    let divisor_leading = leading_coefficient(divisor)?;
286    let mut remainder = dividend.clone();
287    let mut quotient = zero(monomial, lower.clone());
288    while !is_zero(&remainder) {
289        let remainder_degree = degree(&remainder)?;
290        if remainder_degree < divisor_degree {
291            break;
292        }
293        let remainder_leading = leading_coefficient(&remainder)?;
294        let factor = divide_rational_functions(&remainder_leading, &divisor_leading)?;
295        let shift = remainder_degree - divisor_degree;
296        let quotient_term = monomial_term(monomial, lower.clone(), shift, factor.clone());
297        quotient = add(&quotient, &quotient_term);
298        let subtraction = multiply(&quotient_term, divisor);
299        remainder = subtract(&remainder, &subtraction);
300    }
301    Some((quotient, remainder))
302}

304/// Performs the advanced native `make_monic` operation.
305pub fn make_monic(value: &TowerPolynomial) -> Option<TowerPolynomial> {
306    // Scale so the leading coefficient becomes one; None for the zero
307    // polynomial.
308    if is_zero(value) {
309        return None;
310    }
311    let leading = leading_coefficient(value)?;
312    let inverse = divide_rational_functions(&k_one(&value.lower), &leading)?;
313    Some(scale(value, &inverse))
314}

316/// Performs the advanced native `gcd` operation.
317pub fn gcd(left: &TowerPolynomial, right: &TowerPolynomial) -> TowerPolynomial {
318    // Monic Euclidean gcd over the field k.
319    let mut a = left.clone();
320    let mut b = right.clone();
321    while !is_zero(&b) {
322        let (_, remainder) = match divide_with_remainder(&a, &b) {
323            Some(value) => value,
324            None => break,
325        };
326        a = b;
327        b = remainder;
328    }
329    if is_zero(&a) {
330        return a;
331    }
332    make_monic(&a).unwrap_or(a)
333}

335/// Performs the advanced native `extended_gcd` operation.
336pub fn extended_gcd(
337    left: &TowerPolynomial,
338    right: &TowerPolynomial,
339) -> (TowerPolynomial, TowerPolynomial, TowerPolynomial) {
340    // Return (g, s, t) with left*s + right*t = g and g the monic gcd. Standard
341    // extended Euclidean recursion over the field k.
342    let monomial = left.monomial;
343    let lower = left.lower.clone();
344    let mut old_remainder = left.clone();
345    let mut remainder = right.clone();
346    let mut old_s = constant(monomial, lower.clone(), k_one(&lower));
347    let mut s = zero(monomial, lower.clone());
348    let mut old_t = zero(monomial, lower.clone());
349    let mut t = constant(monomial, lower.clone(), k_one(&lower));
350    while !is_zero(&remainder) {
351        let (quotient, next_remainder) = match divide_with_remainder(&old_remainder, &remainder) {
352            Some(value) => value,
353            None => break,
354        };
355        old_remainder = remainder;
356        remainder = next_remainder;
357        let next_s = subtract(&old_s, &multiply(&quotient, &s));
358        old_s = s;
359        s = next_s;
360        let next_t = subtract(&old_t, &multiply(&quotient, &t));
361        old_t = t;
362        t = next_t;
363    }
364    if is_zero(&old_remainder) {
365        return (old_remainder, old_s, old_t);
366    }
367    let leading = leading_coefficient(&old_remainder).unwrap();
368    let inverse = divide_rational_functions(&k_one(&lower), &leading).unwrap();
369    (
370        scale(&old_remainder, &inverse),
371        scale(&old_s, &inverse),
372        scale(&old_t, &inverse),
373    )
374}

376/// Performs the advanced native `gcdex_diophantine` operation.
377pub fn gcdex_diophantine(
378    b: &TowerPolynomial,
379    a: &TowerPolynomial,
380    c: &TowerPolynomial,
381) -> Option<(TowerPolynomial, TowerPolynomial)> {
382    // Solve b*u + a*v = c for (u, v) with deg(u) < deg(a). Returns None when
383    // gcd(a, b) does not divide c.
384    let (g, s, _) = extended_gcd(b, a);
385    let (multiplier, remainder) = divide_with_remainder(c, &g)?;
386    if !is_zero(&remainder) {
387        return None;
388    }
389    let u_unreduced = multiply(&s, &multiplier);
390    let (_, u) = divide_with_remainder(&u_unreduced, a)?;
391    let b_times_u = multiply(b, &u);
392    let numerator = subtract(c, &b_times_u);
393    let (v, v_remainder) = divide_with_remainder(&numerator, a)?;
394    if !is_zero(&v_remainder) {
395        return None;
396    }
397    Some((u, v))
398}

400fn monomial_derivation(
401    field: &DifferentialField,
402    monomial: SymbolIdentifier,
403    lower: &DifferentialField,
404) -> Option<TowerPolynomial> {
405    // The derivation of the monomial t itself, expressed as an element of k[t]:
406    // D(t) = u'*t for t = exp(u) (hyperexponential) and D(t) = u'/u in k for
407    // t = log(u) (primitive). Returns None for an algebraic top generator,
408    // which the RDE recursion never reaches (it defers algebraic towers first).
409    let extension = find_extension(field, monomial)?;
410    match &extension.kind {
411        ExtensionKind::Exponential(argument) => {
412            let argument_derivative = k_normalize(lower, derive_rational_function(lower, argument));
413            Some(monomial_term(
414                monomial,
415                lower.clone(),
416                1,
417                argument_derivative,
418            ))
419        }
420        ExtensionKind::Logarithmic(argument) => {
421            let argument_derivative = derive_rational_function(lower, argument);
422            let eta = divide_rational_functions(&argument_derivative, argument)?;
423            let eta = k_normalize(lower, eta);
424            Some(constant(monomial, lower.clone(), eta))
425        }
426        ExtensionKind::Algebraic(_) => None,
427    }
428}

430/// Performs the advanced native `derivation` operation.
431pub fn derivation(field: &DifferentialField, value: &TowerPolynomial) -> Option<TowerPolynomial> {
432    // Apply the tower derivation to a tower_polynomial:
433    //   D(sum a_i t^i) = sum D(a_i) t^i + (sum i a_i t^{i-1}) * D(t).
434    // The first sum is the coefficientwise derivation in k; the second folds in
435    // the monomial rule D(t). Returns None for an algebraic top generator.
436    let monomial = value.monomial;
437    let lower = &value.lower;
438    let derivative_of_monomial = monomial_derivation(field, monomial, lower)?;
439    let mut coefficientwise: Vec<(usize, RationalFunction)> = Vec::new();
440    let mut formal: Vec<(usize, RationalFunction)> = Vec::new();
441    for (exponent, coefficient) in value.coefficients.iter() {
442        let coefficient_derivative =
443            k_normalize(lower, derive_rational_function(lower, coefficient));
444        if !k_is_zero(&coefficient_derivative) {
445            coefficientwise.push((*exponent, coefficient_derivative));
446        }
447        if *exponent > 0 {
448            let scaled = multiply_rational_functions(
449                coefficient,
450                &from_integer(lower_main(lower), *exponent as i64),
451            );
452            formal.push((*exponent - 1, scaled));
453        }
454    }
455    let coefficientwise_polynomial = from_sparse(monomial, lower.clone(), coefficientwise);
456    let formal_polynomial = from_sparse(monomial, lower.clone(), formal);
457    let second_term = multiply(&formal_polynomial, &derivative_of_monomial);
458    Some(add(&coefficientwise_polynomial, &second_term))
459}

461/// Performs the advanced native `from_polynomial_in_t` operation.
462pub fn from_polynomial_in_t(
463    value: &Polynomial,
464    monomial: SymbolIdentifier,
465    lower: &DifferentialField,
466) -> TowerPolynomial {
467    // Convert a tower polynomial expressed in the nested `polynomial`
468    // representation (main variable = t, coefficients over the lower tower)
469    // into a tower_polynomial with k-element coefficients.
470    if polynomial_is_zero(value) {
471        return zero(monomial, lower.clone());
472    }
473    if value.main_variable != monomial {
474        let lifted = lift_polynomial_to_main(lower_main(lower), value.clone());
475        return constant(monomial, lower.clone(), from_polynomial(lifted));
476    }
477    let mut terms: Vec<(usize, RationalFunction)> = Vec::new();
478    for (exponent, coefficient) in value.terms.iter() {
479        let coefficient_polynomial =
480            coefficient_as_polynomial_in_main(lower_main(lower), coefficient);
481        terms.push((*exponent, from_polynomial(coefficient_polynomial)));
482    }
483    from_sparse(monomial, lower.clone(), terms)
484}

486/// Performs the advanced native `to_rational_function` operation.
487pub fn to_rational_function(value: &TowerPolynomial) -> RationalFunction {
488    // Re-express the k[t] element as a rational_function over the full tower
489    // (main variable = t). Clears each coefficient's k-denominator through the
490    // shared rational-function arithmetic; the zero polynomial maps to zero.
491    let monomial = value.monomial;
492    if is_zero(value) {
493        return rational_function_zero(monomial);
494    }
495    let mut result = rational_function_zero(monomial);
496    for (exponent, coefficient) in value.coefficients.iter() {
497        let coefficient_lifted = lift_rational_function_to_main(monomial, coefficient.clone());
498        let monomial_rational = from_polynomial(monomial_polynomial(monomial, *exponent));
499        let term = multiply_rational_functions(&coefficient_lifted, &monomial_rational);
500        result = add_rational_functions(&result, &term);
501    }
502    result
503}

505/// Performs the advanced native `squarefree_factorization` operation.
506pub fn squarefree_factorization(value: &TowerPolynomial) -> Vec<(usize, TowerPolynomial)> {
507    // Yun's squarefree factorization over the field k using the formal
508    // t-derivative. Returns (multiplicity, factor) pairs with monic factors of
509    // strictly increasing multiplicity. Constant or zero inputs yield an empty
510    // list.
511    if is_zero(value) || degree(value).unwrap_or(0) == 0 {
512        return Vec::new();
513    }
514    let monic = make_monic(value).unwrap_or_else(|| value.clone());
515    let derivative = formal_t_derivative(&monic);
516    let a = gcd(&monic, &derivative);
517    let mut b = divide_with_remainder(&monic, &a)
518        .map(|pair| pair.0)
519        .unwrap();
520    let mut c = divide_with_remainder(&derivative, &a)
521        .map(|pair| pair.0)
522        .unwrap();
523    let mut d = subtract(&c, &formal_t_derivative(&b));
524    let mut factors: Vec<(usize, TowerPolynomial)> = Vec::new();
525    let mut multiplicity = 1usize;
526    while degree(&b).unwrap_or(0) > 0 {
527        let factor = gcd(&b, &d);
528        if degree(&factor).unwrap_or(0) > 0 {
529            factors.push((multiplicity, factor.clone()));
530        }
531        b = divide_with_remainder(&b, &factor)
532            .map(|pair| pair.0)
533            .unwrap();
534        c = divide_with_remainder(&d, &factor)
535            .map(|pair| pair.0)
536            .unwrap();
537        d = subtract(&c, &formal_t_derivative(&b));
538        multiplicity += 1;
539    }
540    factors
541}

543/// Performs the advanced native `formal_t_derivative` operation.
544pub fn formal_t_derivative(value: &TowerPolynomial) -> TowerPolynomial {
545    // The formal derivative with respect to t only (coefficients treated as
546    // constants), used by Yun's squarefree factorization.
547    let mut terms: Vec<(usize, RationalFunction)> = Vec::new();
548    for (exponent, coefficient) in value.coefficients.iter() {
549        if *exponent > 0 {
550            let scaled = multiply_rational_functions(
551                coefficient,
552                &from_integer(lower_main(&value.lower), *exponent as i64),
553            );
554            terms.push((*exponent - 1, scaled));
555        }
556    }
557    from_sparse(value.monomial, value.lower.clone(), terms)
558}

560/// Performs the advanced native `resultant` operation.
561pub fn resultant(left: &TowerPolynomial, right: &TowerPolynomial) -> RationalFunction {
562    // Resultant over the field k via the Euclidean remainder recurrence
563    //   res(A, B) = (-1)^{deg A · deg B} · lc(B)^{deg A − deg R} · res(B, R),
564    // with R = A mod B and the base case res(A, b0) = b0^{deg A} for a constant
565    // divisor. A shared non-constant factor produces a zero remainder and hence
566    // a zero resultant.
567    let lower = left.lower.clone();
568    if is_zero(left) || is_zero(right) {
569        return k_zero(&lower);
570    }
571    let degree_left = degree(left).unwrap();
572    let degree_right = degree(right).unwrap();
573    if degree_left < degree_right {
574        let swapped = resultant(right, left);
575        if (degree_left * degree_right) % 2 == 1 {
576            return negate_rational_function(&swapped);
577        }
578        return swapped;
579    }
580    if degree_right == 0 {
581        return k_power(&leading_coefficient(right).unwrap(), degree_left, &lower);
582    }
583    let (_, remainder) = match divide_with_remainder(left, right) {
584        Some(value) => value,
585        None => return k_zero(&lower),
586    };
587    if is_zero(&remainder) {
588        return k_zero(&lower);
589    }
590    let degree_remainder = degree(&remainder).unwrap();
591    let power = k_power(
592        &leading_coefficient(right).unwrap(),
593        degree_left - degree_remainder,
594        &lower,
595    );
596    let mut result = multiply_rational_functions(&power, &resultant(right, &remainder));
597    if (degree_left * degree_right) % 2 == 1 {
598        result = negate_rational_function(&result);
599    }
600    result
601}

603fn k_power(
604    base: &RationalFunction,
605    exponent: usize,
606    lower: &DifferentialField,
607) -> RationalFunction {
608    let mut result = k_one(lower);
609    for _ in 0..exponent {
610        result = multiply_rational_functions(&result, base);
611    }
612    result
613}

615/// Performs the advanced native `k_element_from_number` operation.
616pub fn k_element_from_number(lower: &DifferentialField, value: Number) -> RationalFunction {
617    from_polynomial(crate::polynomial::constant_polynomial(
618        lower_main(lower),
619        value,
620    ))
621}
622
623/// Performs the advanced native `k_element_is_integer` operation.
624pub fn k_element_is_integer(value: &RationalFunction) -> Option<Integer> {
625    // If the k-element is an integer constant, return it. Used by the base and
626    // degree-bound routines that compare a leading-coefficient ratio to an
627    // integer.
628    if !polynomial_is_zero(&value.denominator) {
629        let denominator_number = polynomial_constant_number(&value.denominator)?;
630        if !crate::number::is_one(&denominator_number) {
631            return None;
632        }
633    }
634    let numerator_number = if polynomial_is_zero(&value.numerator) {
635        Number::Integer(Integer::from(0))
636    } else {
637        polynomial_constant_number(&value.numerator)?
638    };
639    match numerator_number {
640        Number::Integer(value) => Some(value),
641        Number::Rational(_) => None,
642    }
643}
644
645fn polynomial_constant_number(value: &Polynomial) -> Option<Number> {
646    if polynomial_is_zero(value) {
647        return Some(Number::Integer(Integer::from(0)));
648    }
649    if value.terms.len() != 1 || value.terms[0].0 != 0 {
650        return None;
651    }
652    crate::polynomial::coefficient_to_number(&value.terms[0].1)
653}

667    fn exponential_tower(
668        context: &Context,
669    ) -> (DifferentialField, DifferentialField, SymbolIdentifier) {
670        // Q(x)(theta), theta = exp(x). Returns (full tower, lower = Q(x), theta).
671        let x = context.symbol_table.intern_symbol("x");
672        let theta = context.symbol_table.intern_symbol("theta");
673        let lower = base_field(x);
674        let mut full = base_field(x);
675        let argument = from_polynomial(univariate_from_dense(
676            x,
677            vec![integer_number(0), integer_number(1)],
678        ));
679        extend_exponential(&mut full, theta, argument);
680        (full, lower, theta)
681    }

687    #[test]
688    fn multiply_then_divide_recovers_quotient() {
689        let mut context = Context::new();
690        let (_full, lower, theta) = exponential_tower(&mut context);
691        let x = lower.base_variable;
692        // p = t, q = t + 1
693        let p = monomial_term(theta, lower.clone(), 1, constant_over_x(x, 1));
694        let q = add(
695            &monomial_term(theta, lower.clone(), 1, constant_over_x(x, 1)),
696            &constant(theta, lower.clone(), constant_over_x(x, 1)),
697        );
698        let product = multiply(&p, &q);
699        // product = t^2 + t
700        assert_eq!(degree(&product), Some(2));
701        let (quotient, remainder) = divide_with_remainder(&product, &q).unwrap();
702        assert!(is_zero(&remainder));
703        // quotient should equal p = t
704        assert_eq!(degree(&quotient), Some(1));
705        let difference = subtract(&quotient, &p);
706        assert!(is_zero(&difference));
707    }
