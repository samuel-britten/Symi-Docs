27/// Public native enumeration named `geometry_error`.
28pub enum GeometryError {
29    /// Represents the `insufficient_vertices` case.
30    InsufficientVertices {
31        #[doc = "Stores the `got` value."]
32        got: usize,
33    },
34    /// Represents the `degenerate_object` case.
35    DegenerateObject,
36}
37
38impl std::fmt::Display for GeometryError {
39    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        // Render a geometry error in a compact diagnostic form for tests and
41        // interactive consumers, mirroring the MatrixError surface so the two
42        // structural-object modules report failures the same way.
43        match self {
44            GeometryError::InsufficientVertices { got } => write!(
45                formatter,
46                "insufficient vertices: a polygon needs at least three, got {}",
47                got,
48            ),
49            GeometryError::DegenerateObject => {
50                write!(formatter, "degenerate geometric object")
51            }
52        }
53    }
54}

62/// Public native data structure named `point_2d`.
63pub struct Point2d {
64    /// Stores the `x` value.
65    pub x: ExpressionReference,
66    /// Stores the `y` value.
67    pub y: ExpressionReference,
68}
69
70/// Performs the advanced native `point_2d` operation.
71pub fn point_2d(_context: &Context, x: ExpressionReference, y: ExpressionReference) -> Point2d {
72    // Capture a pair of coordinate expressions as a point. The coordinates
73    // arrive already in ASAE form from the auto-simplifying builders, so the
74    // constructor only needs to store them. The context is accepted for
75    // signature uniformity with the other constructors in this module.
76    Point2d { x, y }
77}
78
79impl Point2d {
80    /// Provides the `coordinates` operation on this native type.
81    pub fn coordinates(&self, _context: &Context) -> (ExpressionReference, ExpressionReference) {
82        // Return the (x, y) coordinate pair as freshly cloned references. The
83        // clones are reference-count bumps on the shared hash-cons tree, not
84        // deep copies.
85        (self.x.clone(), self.y.clone())
86    }
87
88    /// Provides the `distance_to` operation on this native type.
89    pub fn distance_to(&self, context: &Context, other: &Point2d) -> ExpressionReference {
90        // Compute the Euclidean distance sqrt((x2 - x1)^2 + (y2 - y1)^2) through
91        // the auto-simplifying builders so the result is canonical: a perfect
92        // square under the radical collapses to a rational, otherwise the
93        // half-power node is retained symbolically.
94        let difference_x = builders::subtract(context, other.x.clone(), self.x.clone());
95        let difference_y = builders::subtract(context, other.y.clone(), self.y.clone());
96        vector_magnitude(context, &difference_x, &difference_y)
97    }
98}

101/// Public native data structure named `segment_2d`.
102pub struct Segment2d {
103    /// Stores the `start` value.
104    pub start: Point2d,
105    /// Stores the `end` value.
106    pub end: Point2d,
107}
108
109/// Performs the advanced native `segment_2d` operation.
110pub fn segment_2d(_context: &Context, start: Point2d, end: Point2d) -> Segment2d {
111    // Capture an ordered pair of endpoints as a segment. No computation happens
112    // at construction; the endpoints are already canonical.
113    Segment2d { start, end }
114}
115
116impl Segment2d {
117    /// Provides the `length` operation on this native type.
118    pub fn length(&self, context: &Context) -> ExpressionReference {
119        // The segment length is the distance between its endpoints.
120        self.start.distance_to(context, &self.end)
121    }
122
123    /// Provides the `midpoint` operation on this native type.
124    pub fn midpoint(&self, context: &Context) -> Point2d {
125        // The midpoint is the coordinate-wise average ((x1 + x2)/2, (y1 + y2)/2),
126        // formed with the rational(1, 2) builder so the average folds to a clean
127        // rational whenever the endpoints are numeric.
128        let half = builders::rational_value(context, 1, 2);
129        let sum_x = builders::sum(context, vec![self.start.x.clone(), self.end.x.clone()]);
130        let midpoint_x = builders::product(context, vec![half.clone(), sum_x]);
131        let sum_y = builders::sum(context, vec![self.start.y.clone(), self.end.y.clone()]);
132        let midpoint_y = builders::product(context, vec![half, sum_y]);
133        Point2d {
134            x: midpoint_x,
135            y: midpoint_y,
136        }
137    }
138
139    /// Provides the `perpendicular_bisector_equation` operation on this native type.
140    pub fn perpendicular_bisector_equation(
141        &self,
142        context: &Context,
143        variable_x: SymbolIdentifier,
144        variable_y: SymbolIdentifier,
145    ) -> ExpressionReference {
146        // The perpendicular bisector is the locus of points equidistant from
147        // both endpoints: (x - midx)*(x2 - x1) + (y - midy)*(y2 - y1) = 0, where
148        // (x2 - x1, y2 - y1) is the segment direction vector. The left-hand side
149        // is returned in expanded polynomial form, matching SymPy's
150        // perpendicular_bisector().equation() shape.
151        let midpoint = self.midpoint(context);
152        let symbol_x = symbol_expression(context, variable_x);
153        let symbol_y = symbol_expression(context, variable_y);
154        let direction_x = builders::subtract(context, self.end.x.clone(), self.start.x.clone());
155        let direction_y = builders::subtract(context, self.end.y.clone(), self.start.y.clone());
156        let offset_x = builders::subtract(context, symbol_x, midpoint.x);
157        let offset_y = builders::subtract(context, symbol_y, midpoint.y);
158        let term_x = builders::product(context, vec![offset_x, direction_x]);
159        let term_y = builders::product(context, vec![offset_y, direction_y]);
160        let equation = builders::sum(context, vec![term_x, term_y]);
161        expand(context, &equation)
162    }
163}

166/// Public native data structure named `polygon_2d`.
167pub struct Polygon2d {
168    /// Stores the `vertices` value.
169    pub vertices: Vec<Point2d>,
170}
171
172/// Performs the advanced native `polygon_2d` operation.
173pub fn polygon_2d(_context: &Context, vertices: Vec<Point2d>) -> GeometryResult<Polygon2d> {
174    // Build a polygon from its ordered vertex list. Fewer than three vertices
175    // cannot bound a region, so the constructor rejects them; the degenerate
176    // (self-intersecting or collinear) case is left to the type-level triangle
177    // constructor since a general polygon has no single collinearity test.
178    if vertices.len() < 3 {
179        return Err(GeometryError::InsufficientVertices {
180            got: vertices.len(),
181        });
182    }
183    Ok(Polygon2d { vertices })
184}
185
186impl Polygon2d {
187    /// Provides the `area` operation on this native type.
188    pub fn area(&self, context: &Context) -> ExpressionReference {
189        // The shoelace formula gives the area as
190        //   (1/2) * |Σ_i (x_i * y_{i+1} - x_{i+1} * y_i)|
191        // with the index wrapping around so the last vertex closes back to the
192        // first. The absolute value keeps the result orientation-independent;
193        // for numeric vertices the inner sum folds to a rational and the abs
194        // canonicalises away.
195        let count = self.vertices.len();
196        let mut cross_terms = Vec::with_capacity(count);
197        for index in 0..count {
198            let current = &self.vertices[index];
199            let next = &self.vertices[(index + 1) % count];
200            let forward = builders::product(context, vec![current.x.clone(), next.y.clone()]);
201            let backward = builders::product(context, vec![next.x.clone(), current.y.clone()]);
202            cross_terms.push(builders::subtract(context, forward, backward));
203        }
204        let cross_sum = builders::sum(context, cross_terms);
205        let magnitude = builders::function_call(
206            context,
207            FunctionIdentifier::Builtin(BuiltinFunction::Abs),
208            vec![cross_sum],
209        );
210        let half = builders::rational_value(context, 1, 2);
211        builders::product(context, vec![half, magnitude])
212    }
213
214    /// Provides the `perimeter` operation on this native type.
215    pub fn perimeter(&self, context: &Context) -> ExpressionReference {
216        // The perimeter is the sum of the side lengths around the boundary, the
217        // index wrapping so the final side closes the polygon.
218        let count = self.vertices.len();
219        let mut side_lengths = Vec::with_capacity(count);
220        for index in 0..count {
221            let current = &self.vertices[index];
222            let next = &self.vertices[(index + 1) % count];
223            side_lengths.push(current.distance_to(context, next));
224        }
225        builders::sum(context, side_lengths)
226    }
227
228    /// Provides the `centroid` operation on this native type.
229    pub fn centroid(&self, context: &Context) -> Point2d {
230        // The vertex centroid is the coordinate-wise average of every vertex.
231        // This is the centroid of the vertex set (not the area centroid); it
232        // matches SymPy's Polygon.centroid only for the triangle, where the two
233        // coincide, but is the conventional and least-surprising choice for the
234        // general polygon vertex average.
235        let count = self.vertices.len();
236        let coordinates_x: Vec<ExpressionReference> = self
237            .vertices
238            .iter()
239            .map(|vertex| vertex.x.clone())
240            .collect();
241        let coordinates_y: Vec<ExpressionReference> = self
242            .vertices
243            .iter()
244            .map(|vertex| vertex.y.clone())
245            .collect();
246        let sum_x = builders::sum(context, coordinates_x);
247        let sum_y = builders::sum(context, coordinates_y);
248        let count_expression = builders::integer_value(context, count as i64);
249        let centroid_x = builders::divide(context, sum_x, count_expression.clone());
250        let centroid_y = builders::divide(context, sum_y, count_expression);
251        Point2d {
252            x: centroid_x,
253            y: centroid_y,
254        }
255    }
256
257    /// Provides the `interior_angles` operation on this native type.
258    pub fn interior_angles(&self, context: &Context) -> Vec<(Point2d, ExpressionReference)> {
259        // At each vertex B the interior angle between the incoming edge BA and
260        // the outgoing edge BC is arccos((BA·BC) / (|BA|·|BC|)), where BA and BC
261        // are the edge vectors emanating from B toward its previous and next
262        // neighbours. The result pairs each vertex with its angle in vertex
263        // order.
264        let count = self.vertices.len();
265        let mut angles = Vec::with_capacity(count);
266        for index in 0..count {
267            let previous = &self.vertices[(index + count - 1) % count];
268            let vertex = &self.vertices[index];
269            let next = &self.vertices[(index + 1) % count];
270            let incoming_x = builders::subtract(context, previous.x.clone(), vertex.x.clone());
271            let incoming_y = builders::subtract(context, previous.y.clone(), vertex.y.clone());
272            let outgoing_x = builders::subtract(context, next.x.clone(), vertex.x.clone());
273            let outgoing_y = builders::subtract(context, next.y.clone(), vertex.y.clone());
274            let dot_x = builders::product(context, vec![incoming_x.clone(), outgoing_x.clone()]);
275            let dot_y = builders::product(context, vec![incoming_y.clone(), outgoing_y.clone()]);
276            let dot_product = builders::sum(context, vec![dot_x, dot_y]);
277            let incoming_magnitude = vector_magnitude(context, &incoming_x, &incoming_y);
278            let outgoing_magnitude = vector_magnitude(context, &outgoing_x, &outgoing_y);
279            let denominator =
280                builders::product(context, vec![incoming_magnitude, outgoing_magnitude]);
281            let cosine = builders::divide(context, dot_product, denominator);
282            let angle = builders::function_call(
283                context,
284                FunctionIdentifier::Builtin(BuiltinFunction::Arccos),
285                vec![cosine],
286            );
287            angles.push((vertex.clone(), angle));
288        }
289        angles
290    }
291}

296    polygon: Polygon2d,
297}

299/// Performs the advanced native `triangle_2d` operation.
300pub fn triangle_2d(
301    context: &Context,
302    first: Point2d,
303    second: Point2d,
304    third: Point2d,
305) -> GeometryResult<Triangle2d> {
306    // Build a triangle, enforcing the three-vertex invariant at the type level.
307    // The signed area (twice the oriented area) is the cross product of the two
308    // edge vectors at the first vertex; when it is provably zero the three
309    // points are collinear and the triangle is degenerate. The degeneracy test
310    // only rejects a triangle when collinearity can be proven, so symbolic
311    // coordinates that cannot be decided are accepted and the property methods
312    // remain total.
313    let edge_one_x = builders::subtract(context, second.x.clone(), first.x.clone());
314    let edge_two_y = builders::subtract(context, third.y.clone(), first.y.clone());
315    let edge_two_x = builders::subtract(context, third.x.clone(), first.x.clone());
316    let edge_one_y = builders::subtract(context, second.y.clone(), first.y.clone());
317    let forward = builders::product(context, vec![edge_one_x, edge_two_y]);
318    let backward = builders::product(context, vec![edge_two_x, edge_one_y]);
319    let signed_area = builders::subtract(context, forward, backward);
320    if is_nonzero(&signed_area, &context.symbol_table) == TruthValue::No {
321        return Err(GeometryError::DegenerateObject);
322    }
323    Ok(Triangle2d {
324        polygon: Polygon2d {
325            vertices: vec![first, second, third],
326        },
327    })
328}

330impl Deref for Triangle2d {
331    type Target = Polygon2d;

333    fn deref(&self) -> &Polygon2d {
334        // A triangle is a polygon with a guaranteed three vertices; every
335        // polygon property method applies unchanged through this coercion.
336        &self.polygon
337    }

341/// Public native data structure named `circle_2d`.
342pub struct Circle2d {
343    /// Stores the `center` value.
344    pub center: Point2d,
345    /// Stores the `radius` value.
346    pub radius: ExpressionReference,
347}
348
349/// Performs the advanced native `circle_2d` operation.
350pub fn circle_2d(
351    context: &Context,
352    center: Point2d,
353    radius: ExpressionReference,
354) -> GeometryResult<Circle2d> {
355    // Build a circle from its centre and radius. A provably-zero radius collapses
356    // the circle to a point and is rejected as degenerate; an undecidable
357    // symbolic radius is accepted so the property methods stay total.
358    if is_nonzero(&radius, &context.symbol_table) == TruthValue::No {
359        return Err(GeometryError::DegenerateObject);
360    }
361    Ok(Circle2d { center, radius })
362}
363
364impl Circle2d {
365    /// Provides the `area` operation on this native type.
366    pub fn area(&self, context: &Context) -> ExpressionReference {
367        // The enclosed area is pi * r^2.
368        let pi = builders::constant(context, BuiltinConstant::Pi);
369        let two = builders::integer_value(context, 2);
370        let radius_squared = builders::power(context, self.radius.clone(), two);
371        builders::product(context, vec![pi, radius_squared])
372    }
373
374    /// Provides the `circumference` operation on this native type.
375    pub fn circumference(&self, context: &Context) -> ExpressionReference {
376        // The circumference is 2 * pi * r.
377        let two = builders::integer_value(context, 2);
378        let pi = builders::constant(context, BuiltinConstant::Pi);
379        builders::product(context, vec![two, pi, self.radius.clone()])
380    }
381
382    /// Provides the `equation` operation on this native type.
383    pub fn equation(
384        &self,
385        context: &Context,
386        variable_x: SymbolIdentifier,
387        variable_y: SymbolIdentifier,
388    ) -> ExpressionReference {
389        // The implicit circle equation (x - cx)^2 + (y - cy)^2 - r^2 = 0 is
390        // returned in expanded form, matching SymPy's Circle.equation()
391        // convention of an expression that is zero on the circle.
392        let symbol_x = symbol_expression(context, variable_x);
393        let symbol_y = symbol_expression(context, variable_y);
394        let two = builders::integer_value(context, 2);
395        let offset_x = builders::subtract(context, symbol_x, self.center.x.clone());
396        let offset_y = builders::subtract(context, symbol_y, self.center.y.clone());
397        let term_x = builders::power(context, offset_x, two.clone());
398        let term_y = builders::power(context, offset_y, two.clone());
399        let radius_squared = builders::power(context, self.radius.clone(), two);
400        let negated_radius_squared = builders::negate(context, radius_squared);
401        let equation = builders::sum(context, vec![term_x, term_y, negated_radius_squared]);
402        expand(context, &equation)
403    }
404}

406fn vector_magnitude(
407    context: &Context,
408    component_x: &ExpressionReference,
409    component_y: &ExpressionReference,
410) -> ExpressionReference {
411    // Shared Euclidean-norm helper: sqrt(cx^2 + cy^2) through the
412    // auto-simplifying builders. Used by the distance, perimeter, and
413    // interior-angle computations so the radical shape is identical everywhere.
414    let two = builders::integer_value(context, 2);
415    let x_squared = builders::power(context, component_x.clone(), two.clone());
416    let y_squared = builders::power(context, component_y.clone(), two);
417    let sum_of_squares = builders::sum(context, vec![x_squared, y_squared]);
418    let half = builders::rational_value(context, 1, 2);
419    builders::power(context, sum_of_squares, half)
420}

422fn symbol_expression(context: &Context, identifier: SymbolIdentifier) -> ExpressionReference {
423    // Lift a symbol identifier to its interned symbol-leaf expression so the
424    // equation builders can place free variables alongside coordinate data.
425    context.interner.intern(ExpressionKind::Symbol(identifier))
426}

428/// Performs the advanced native `point_2d_to_latex` operation.

449/// Performs the advanced native `polygon_2d_to_latex` operation.
450pub fn polygon_2d_to_latex(polygon: &Polygon2d, table: &SymbolTable) -> String {
451    // Render a polygon as the comma-separated sequence of its vertices under a
452    // Polygon label. Triangles reach this helper through the Deref coercion.
453    let vertices: Vec<String> = polygon
454        .vertices
455        .iter()
456        .map(|vertex| point_2d_to_latex(vertex, table))
457        .collect();
458    format!("\\text{{Polygon}}\\left({}\\right)", vertices.join(", "))
459}
