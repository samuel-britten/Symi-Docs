12/// Public native constant `MAXIMUM_FOLDED_BRANCH_COUNT`.
13pub const MAXIMUM_FOLDED_BRANCH_COUNT: usize = 16;
14/// Public native constant `MAXIMUM_FOLDED_NODE_COUNT`.
15pub const MAXIMUM_FOLDED_NODE_COUNT: usize = 128;
16/// Public native constant `MAXIMUM_FOLDED_NODE_MULTIPLIER`.
17pub const MAXIMUM_FOLDED_NODE_MULTIPLIER: usize = 8;

19#[derive(Clone)]
20enum ParentRebuilder {
21    Sum,
22    Product,
23    Power,
24    Factorial,
25    FunctionCall(FunctionIdentifier),
26}

28/// Performs the advanced native `piecewise_fold` operation.
29pub fn piecewise_fold(context: &Context, target: &ExpressionReference) -> ExpressionReference {
30    fold_recursive(context, target)
31}

33fn fold_recursive(context: &Context, target: &ExpressionReference) -> ExpressionReference {
34    match &target.kind {
35        ExpressionKind::Sum(terms) => fold_combiner(context, target, ParentRebuilder::Sum, terms),
36        ExpressionKind::Product(factors) => {
37            fold_combiner(context, target, ParentRebuilder::Product, factors)
38        }
39        ExpressionKind::Power(base, exponent) => fold_combiner(
40            context,
41            target,
42            ParentRebuilder::Power,
43            &[Arc::clone(base), Arc::clone(exponent)],
44        ),
45        ExpressionKind::Factorial(argument) => fold_combiner(
46            context,
47            target,
48            ParentRebuilder::Factorial,
49            &[Arc::clone(argument)],
50        ),
51        ExpressionKind::FunctionCall(function, arguments) => {
52            fold_function_call(context, target, *function, arguments)
53        }
54        ExpressionKind::Piecewise(branches) => fold_piecewise(context, target, branches),
55        ExpressionKind::Integer(_)
56        | ExpressionKind::Rational(_)
57        | ExpressionKind::PolynomialRoot(_)
58        | ExpressionKind::Symbol(_)
59        | ExpressionKind::Constant(_)
60        | ExpressionKind::Derivative(_, _, _)
61        | ExpressionKind::Integral(_, _, _, _)
62        | ExpressionKind::Summation(_, _, _, _)
63        | ExpressionKind::IndexedProduct(_, _, _, _)
64        | ExpressionKind::LaplaceTransform(_, _, _)
65        | ExpressionKind::InverseLaplaceTransform(_, _, _)
66        | ExpressionKind::FourierTransform(_, _, _)
67        | ExpressionKind::InverseFourierTransform(_, _, _)
68        | ExpressionKind::OrdinaryDifferentialEquation(_, _, _)
69        | ExpressionKind::RecurrenceEquation(_, _, _)
70        | ExpressionKind::Interval(_, _, _, _)
71        | ExpressionKind::FiniteSet(_)
72        | ExpressionKind::SetUnion(_)
73        | ExpressionKind::SetIntersection(_)
74        | ExpressionKind::SetComplement(_, _)
75        | ExpressionKind::ImageSet(_, _, _)
76        | ExpressionKind::ConditionSet(_, _, _)
77        | ExpressionKind::Distribution(_, _)
78        | ExpressionKind::RandomVariable(_, _)
79        | ExpressionKind::SpecialFunctionParameterVector(_) => Arc::clone(target),
80    }
81}

83fn fold_function_call(
84    context: &Context,
85    target: &ExpressionReference,
86    function: FunctionIdentifier,
87    arguments: &[ExpressionReference],
88) -> ExpressionReference {
89    let folded_arguments: Vec<_> = arguments
90        .iter()
91        .map(|argument| fold_recursive(context, argument))
92        .collect();
93    let rebuilt = fold_processed_children(
94        context,
95        target,
96        ParentRebuilder::FunctionCall(function),
97        folded_arguments,
98    );
99    let Some(builtin) = definition_builtin(function) else {
100        return rebuilt;
101    };
102    match &rebuilt.kind {
103        ExpressionKind::FunctionCall(_, rebuilt_arguments) if rebuilt_arguments.len() == 1 => {
104            expand_definition(context, builtin, Arc::clone(&rebuilt_arguments[0]))
105        }
106        ExpressionKind::Piecewise(branches) => {
107            let expanded_branches: Vec<_> = branches
108                .iter()
109                .map(|(condition, value)| {
110                    let expanded_value = match &value.kind {
111                        ExpressionKind::FunctionCall(value_function, value_arguments)
112                            if *value_function == function && value_arguments.len() == 1 =>
113                        {
114                            expand_definition(context, builtin, Arc::clone(&value_arguments[0]))
115                        }
116                        _ => Arc::clone(value),
117                    };
118                    (Arc::clone(condition), expanded_value)
119                })
120                .collect();
121            build_flattened_piecewise(context, target, expanded_branches).unwrap_or(rebuilt)
122        }
123        _ => rebuilt,
124    }
125}

127fn definition_builtin(function: FunctionIdentifier) -> Option<BuiltinFunction> {
128    match function {
129        FunctionIdentifier::Builtin(
130            builtin @ (BuiltinFunction::Abs
131            | BuiltinFunction::Signum
132            | BuiltinFunction::HeavisideStep),
133        ) => Some(builtin),
134        _ => None,
135    }
136}

138fn expand_definition(
139    context: &Context,
140    builtin: BuiltinFunction,
141    argument: ExpressionReference,
142) -> ExpressionReference {
143    match builtin {
144        BuiltinFunction::Abs => piecewise_from_abs(context, argument),
145        BuiltinFunction::Signum => piecewise_from_signum(context, argument),
146        BuiltinFunction::HeavisideStep => piecewise_from_heaviside(context, argument),
147        _ => argument,
148    }

151fn fold_combiner(
152    context: &Context,
153    target: &ExpressionReference,
154    rebuilder: ParentRebuilder,
155    children: &[ExpressionReference],
156) -> ExpressionReference {
157    let processed_children: Vec<_> = children
158        .iter()
159        .map(|child| fold_recursive(context, child))
160        .collect();
161    fold_processed_children(context, target, rebuilder, processed_children)
162}

164fn fold_processed_children(
165    context: &Context,
166    target: &ExpressionReference,
167    rebuilder: ParentRebuilder,
168    children: Vec<ExpressionReference>,
169) -> ExpressionReference {
170    if !children
171        .iter()
172        .any(|child| matches!(child.kind, ExpressionKind::Piecewise(_)))
173    {
174        return rebuild(context, &rebuilder, children);
175    }
176    fold_node(context, target, &rebuilder, &children)
177        .unwrap_or_else(|| rebuild(context, &rebuilder, children))
178}

180fn fold_node(
181    context: &Context,
182    target: &ExpressionReference,
183    rebuilder: &ParentRebuilder,
184    children: &[ExpressionReference],
185) -> Option<ExpressionReference> {
186    let mut branch_count = 1usize;
187    for child in children {
188        if let ExpressionKind::Piecewise(branches) = &child.kind {
189            branch_count = branch_count.checked_mul(branches.len())?;
190            if branch_count > MAXIMUM_FOLDED_BRANCH_COUNT {
191                return None;
192            }
193        }
194    }
195    let mut combinations = vec![(Vec::with_capacity(children.len()), Vec::new())];
196    for child in children {
197        let choices: Vec<_> = match &child.kind {
198            ExpressionKind::Piecewise(branches) => branches
199                .iter()
200                .map(|(condition, value)| (Some(Arc::clone(condition)), Arc::clone(value)))
201                .collect(),
202            _ => vec![(None, Arc::clone(child))],
203        };
204        let mut next = Vec::with_capacity(combinations.len().saturating_mul(choices.len()));
205        for (values, conditions) in combinations {
206            for (condition, value) in &choices {
207                let mut next_values = values.clone();
208                next_values.push(Arc::clone(value));
209                let mut next_conditions = conditions.clone();
210                if let Some(condition) = condition {
211                    next_conditions.push(Arc::clone(condition));
212                }
213                next.push((next_values, next_conditions));
214            }
215        }
216        combinations = next;
217    }
218    let branches = combinations
219        .into_iter()
220        .map(|(values, conditions)| {
221            let value = rebuild(context, rebuilder, values);
222            let condition = conjunction(context, conditions);
223            (condition, value)
224        })
225        .collect();
226    build_flattened_piecewise(context, target, branches)
227}

229fn fold_piecewise(
230    context: &Context,
231    target: &ExpressionReference,
232    branches: &[(ExpressionReference, ExpressionReference)],
233) -> ExpressionReference {
234    let processed_branches = branches
235        .iter()
236        .map(|(condition, value)| (Arc::clone(condition), fold_recursive(context, value)))
237        .collect();
238    build_flattened_piecewise(context, target, processed_branches)
239        .unwrap_or_else(|| Arc::clone(target))
240}

242fn build_flattened_piecewise(
243    context: &Context,
244    budget_source: &ExpressionReference,
245    branches: Vec<(ExpressionReference, ExpressionReference)>,
246) -> Option<ExpressionReference> {
247    let flattened = splice_nested_branches(context, branches)?;
248    if flattened.len() > MAXIMUM_FOLDED_BRANCH_COUNT {
249        return None;
250    }
251    let candidate = piecewise(context, flattened);
252    let relative_limit = node_count(budget_source)
253        .saturating_mul(MAXIMUM_FOLDED_NODE_MULTIPLIER)
254        .max(1);
255    if node_count(&candidate) > relative_limit.min(MAXIMUM_FOLDED_NODE_COUNT) {
256        return None;
257    }
258    Some(candidate)
259}

261fn splice_nested_branches(
262    context: &Context,
263    branches: Vec<(ExpressionReference, ExpressionReference)>,
264) -> Option<Vec<(ExpressionReference, ExpressionReference)>> {
265    let resulting_count = branches.iter().try_fold(0usize, |count, (_, value)| {
266        let increment = match &value.kind {
267            ExpressionKind::Piecewise(inner) => inner.len(),
268            _ => 1,
269        };
270        count.checked_add(increment)
271    })?;
272    if resulting_count > MAXIMUM_FOLDED_BRANCH_COUNT {
273        return None;
274    }
275    let mut flattened = Vec::with_capacity(resulting_count);
276    for (outer_condition, value) in branches {
277        match &value.kind {
278            ExpressionKind::Piecewise(inner) => {
279                for (inner_condition, inner_value) in inner {
280                    let condition = conjunction(
281                        context,
282                        vec![Arc::clone(&outer_condition), Arc::clone(inner_condition)],
283                    );
284                    flattened.push((condition, Arc::clone(inner_value)));
285                }
286            }
287            _ => flattened.push((outer_condition, value)),
288        }
289    }
290    Some(flattened)
291}

293fn conjunction(context: &Context, conditions: Vec<ExpressionReference>) -> ExpressionReference {
294    if conditions.len() == 1 {
295        return Arc::clone(&conditions[0]);
296    }
297    function_call(
298        context,
299        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
300        conditions,
301    )
302}

304fn rebuild(
305    context: &Context,
306    rebuilder: &ParentRebuilder,
307    children: Vec<ExpressionReference>,
308) -> ExpressionReference {
309    match rebuilder {
310        ParentRebuilder::Sum => sum(context, children),
311        ParentRebuilder::Product => product(context, children),
312        ParentRebuilder::Power => {
313            let mut iterator = children.into_iter();
314            power(context, iterator.next().unwrap(), iterator.next().unwrap())
315        }
316        ParentRebuilder::Factorial => factorial(context, children.into_iter().next().unwrap()),
317        ParentRebuilder::FunctionCall(function) => function_call(context, *function, children),
318    }
319}
