14/// Public native data structure named `real_algebraic_number`.
15pub struct RealAlgebraicNumber {
16    // A real algebraic number in the standard isolating-interval representation
17    // (Emiris–Mourrain–Tsigaridas; Cohen §4.2): a squarefree defining polynomial
18    // over ℚ together with a rational isolating interval that brackets exactly
19    // one of its real roots. A collapsed interval (lower == upper) marks an exact
20    // rational value. The defining polynomial need not be irreducible — squarefree
21    // suffices to make the bracket a one-root indicator (the minimal polynomial is
22    // recovered on demand by minimal_polynomial_of). Equality is decided exactly
23    // by compare.rs, never by the stored interval alone, so there is no float in
24    // the type.
25    /// Stores the `defining_polynomial` value.
26    pub defining_polynomial: Polynomial,
27    /// Stores the `isolating_interval` value.
28    pub isolating_interval: IsolatingInterval,
29}
30
31/// Performs the advanced native `real_root_of` operation.
32pub fn real_root_of(value: &Polynomial, root_index: usize) -> Option<RealAlgebraicNumber> {
33    // The root_index-th smallest real root of a univariate ℚ-polynomial, as a
34    // real_algebraic_number whose defining polynomial is the squarefree part of
35    // value. None when value is not univariate over ℚ, is zero, or has fewer than
36    // root_index + 1 distinct real roots.
37    let partition = solve_sign_partition(value)?;
38    let interval = partition.roots.get(root_index)?.clone();
39    Some(RealAlgebraicNumber {
40        defining_polynomial: partition.squarefree,
41        isolating_interval: interval,
42    })
43}

45/// Performs the advanced native `from_rational` operation.
46pub fn from_rational(
47    main_variable: SymbolIdentifier,
48    value: &exact_rational,
49) -> RealAlgebraicNumber {
50    // Lift a rational scalar to a real_algebraic_number: the defining polynomial
51    // is the monic linear x − value and the isolating interval is the collapsed
52    // point [value, value].
53    let defining = dense::from_dense(main_variable, &[-value.clone(), rational_one()]);
54    RealAlgebraicNumber {
55        defining_polynomial: defining,
56        isolating_interval: dense::exact_interval(value.clone()),
57    }
58}

73/// Performs the advanced native `is_rational` operation.
74pub fn is_rational(number: &RealAlgebraicNumber) -> bool {
75    // True when the number is an exact rational, marked by a collapsed interval.
76    number.isolating_interval.is_exact()
77}

79/// Performs the advanced native `as_rational` operation.
80pub fn as_rational(number: &RealAlgebraicNumber) -> Option<exact_rational> {
81    // The exact rational value when the number is rational; None otherwise.
82    if number.isolating_interval.is_exact() {
83        Some(number.isolating_interval.lower.clone())
84    } else {
85        None
86    }

89/// Performs the advanced native `refine` operation.
90pub fn refine(number: &mut RealAlgebraicNumber, max_width: &exact_rational) {
91    // Narrow the isolating interval to width at most max_width by
92    // sign-guided bisection of the defining polynomial. A no-op on a rational.
93    let refined = refine_isolating_interval(
94        &number.defining_polynomial,
95        &number.isolating_interval,
96        max_width,
97    );
98    number.isolating_interval = refined;
99}

101/// Performs the advanced native `evaluate_to_interval` operation.
102pub fn evaluate_to_interval(
103    number: &RealAlgebraicNumber,
104    precision_bits: u64,
105) -> (exact_rational, exact_rational) {
106    // Refine the isolating interval to width below 2^(-precision_bits) and return
107    // its rational endpoints — the exact bracket the numeric evaluator
108    // consumes. The exact value lies strictly inside (or equals both endpoints
109    // for a rational).
110    let mut working = number.clone();
111    let width = rational_from_integers(
112        integer_one(),
113        integer_power(&exact_integer::from(2), precision_bits + 1),
114    )
115    .expect("nonzero denominator");
116    refine(&mut working, &width);
117    (
118        working.isolating_interval.lower.clone(),
119        working.isolating_interval.upper.clone(),
120    )
121}

123/// Performs the advanced native `minimal_polynomial_of` operation.
124pub fn minimal_polynomial_of(number: &RealAlgebraicNumber) -> Polynomial {
125    // The minimal (irreducible, over ℚ) polynomial of the number: the unique
126    // irreducible factor of the squarefree defining polynomial whose real root
127    // lies in the isolating interval. Found by factoring over ℤ and selecting the
128    // factor that changes sign across the open interval (or vanishes at the exact
129    // point). Falls back to the defining polynomial when factoring cannot isolate
130    // a single factor (e.g. a degree-one input, already irreducible).
131    let factors = factor_univariate_over_integers(&number.defining_polynomial);
132    if factors.len() == 1 {
133        return number.defining_polynomial.clone();
134    }
135    let interval =
136        tighten_off_endpoint_roots(&number.defining_polynomial, &number.isolating_interval);
137    for (_, factor) in factors.iter() {
138        if factor_has_root_in_interval(factor, &interval) {
139            return factor.clone();
140        }
141    }
142    number.defining_polynomial.clone()
143}

145fn tighten_off_endpoint_roots(
146    defining: &Polynomial,
147    interval: &IsolatingInterval,
148) -> IsolatingInterval {
149    // Refine an isolating interval until neither endpoint is a root of the defining
150    // polynomial. Real-root isolation can leave an endpoint coincident with a
151    // neighbouring root (a different irreducible factor's root), which would make
152    // factor_has_root_in_interval match that wrong factor through its endpoint-zero
153    // rule. The contained root is strictly interior, so tightening moves both
154    // endpoints onto non-roots and the loop terminates.
155    if interval.is_exact() {
156        return interval.clone();
157    }
158    let dense_defining = match dense::to_dense(defining) {
159        Some(dense) if !dense.is_empty() => dense,
160        _ => return interval.clone(),
161    };
162    let mut working = interval.clone();
163    loop {
164        let lower_is_root =
165            dense::rational_sign(&dense::evaluate(&dense_defining, &working.lower)) == 0;
166        let upper_is_root =
167            dense::rational_sign(&dense::evaluate(&dense_defining, &working.upper)) == 0;
168        if !lower_is_root && !upper_is_root {
169            return working;
170        }
171        let half = (&working.upper - &working.lower) / exact_rational::from(2);
172        working = refine_isolating_interval(defining, &working, &half);
173        if working.is_exact() {
174            return working;
175        }
176    }
177}

179fn factor_has_root_in_interval(factor: &Polynomial, interval: &IsolatingInterval) -> bool {
180    // Whether an irreducible factor has its root inside the bracket. The
181    // defining polynomial is squarefree, so each of its roots belongs to exactly
182    // one factor; the factor whose sign changes across the open interval (or
183    // which vanishes at a collapsed point) is the one carrying this root.
184    if degree(factor).unwrap_or(0) == 0 {
185        return false;
186    }
187    let factor_dense = match dense::to_dense(factor) {
188        Some(dense) if !dense.is_empty() => dense,
189        _ => return false,
190    };
191    if interval.is_exact() {
192        return dense::rational_sign(&dense::evaluate(&factor_dense, &interval.lower)) == 0;
193    }
194    let lower_sign = dense::rational_sign(&dense::evaluate(&factor_dense, &interval.lower));
195    let upper_sign = dense::rational_sign(&dense::evaluate(&factor_dense, &interval.upper));
196    if lower_sign == 0 || upper_sign == 0 {
197        return true;
198    }
199    lower_sign != upper_sign
200}
