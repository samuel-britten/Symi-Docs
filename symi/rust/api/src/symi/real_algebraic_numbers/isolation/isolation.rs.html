25/// Performs the advanced native `isolate_real_roots_over_field` operation.
26pub fn isolate_real_roots_over_field(
27    context: &Context,
28    embedding: &RealEmbeddedNumberField,
29    coefficients: &[AlgebraicElement],
30) -> Vec<RealAlgebraicNumber> {
31    // Real-root isolation of a univariate polynomial f(z) = Σ cᵢ zⁱ whose
32    // coefficients are elements of a real-embedded Q(γ) — the CAD lifting
33    // primitive. Method (norm-to-ℚ + sign sieve): make f squarefree over Q(γ) so
34    // its real roots are simple, form the ℚ-resultant Res_γ(F(z,γ), m_γ(γ)) whose
35    // real roots superset f's real roots at the embedding, isolate those over ℚ
36    // with real-root isolation, and keep each candidate at which f changes sign through the
37    // embedding (sieving out the extraneous conjugate-norm roots). The survivors
38    // are returned in increasing order as real algebraic numbers over ℚ.
39    let field = &embedding.field;
40    let trimmed = trim_high_zeros(coefficients);
41    if trimmed.len() <= 1 {
42        return Vec::new();
43    }
44    let squarefree_coefficients =
45        square_free_over_field(context, &trimmed, field).unwrap_or(trimmed);
46    let norm = match norm_polynomial(context, &squarefree_coefficients, field) {
47        Some(norm) if degree(&norm).unwrap_or(0) >= 1 => norm,
48        _ => return Vec::new(),
49    };
50    let norm_squarefree_dense = dense::squarefree_part(&dense::to_dense(&norm).unwrap_or_default());
51    let norm_variable = norm.main_variable;
52    let norm_squarefree = dense::from_dense(norm_variable, &norm_squarefree_dense);
53    let candidates = match isolate_real_roots(&norm_squarefree) {
54        Some(candidates) => candidates,
55        None => return Vec::new(),
56    };
57    let mut roots: Vec<RealAlgebraicNumber> = Vec::new();
58    for candidate in candidates.into_iter() {
59        if let Some(interval) = sieve_candidate(
60            embedding,
61            &squarefree_coefficients,
62            &norm_squarefree,
63            candidate,
64        ) {
65            roots.push(from_squarefree_and_interval(
66                norm_squarefree.clone(),
67                interval,
68            ));
69        }
70    }
71    roots
72}

74fn sieve_candidate(
75    embedding: &RealEmbeddedNumberField,
76    coefficients: &[AlgebraicElement],
77    norm_squarefree: &Polynomial,
78    candidate: IsolatingInterval,
79) -> Option<IsolatingInterval> {
80    // Decide whether the norm root isolated by candidate is a genuine real root
81    // of f at the embedding, returning its bracket when so. An exact (rational)
82    // candidate is a root iff f vanishes there; an open candidate is a root iff f
83    // changes sign across it — exact because f was made squarefree over Q(γ), so
84    // every real root is simple and produces a sign change. The candidate is
85    // refined until both endpoints carry a definite non-zero sign.
86    if candidate.is_exact() {
87        let value = sign_of_field_polynomial_at_rational(embedding, coefficients, &candidate.lower);
88        if value == 0 {
89            return Some(candidate);
90        }
91        return None;
92    }
93    let mut working = candidate;
94    loop {
95        let lower_sign =
96            sign_of_field_polynomial_at_rational(embedding, coefficients, &working.lower);
97        let upper_sign =
98            sign_of_field_polynomial_at_rational(embedding, coefficients, &working.upper);
99        if lower_sign != 0 && upper_sign != 0 {
100            if lower_sign != upper_sign {
101                return Some(working);
102            }
103            return None;
104        }
105        let half = dense::interval_width(&working) / exact_rational::from(2);
106        working = refine_isolating_interval(norm_squarefree, &working, &half);
107        if working.is_exact() {
108            let value =
109                sign_of_field_polynomial_at_rational(embedding, coefficients, &working.lower);
110            if value == 0 {
111                return Some(working);
112            }
113            return None;
114        }
115    }
116}

118fn sign_of_field_polynomial_at_rational(
119    embedding: &RealEmbeddedNumberField,
120    coefficients: &[AlgebraicElement],
121    point: &exact_rational,
122) -> i32 {
123    // The sign of f(point) for a rational point: f(point) = Σ cᵢ pointⁱ is an
124    // element of Q(γ), whose sign is read through the real embedding.
125    let element = evaluate_field_polynomial_at_rational(&embedding.field, coefficients, point);
126    sign_of_element(embedding, &element)
127}

129fn evaluate_field_polynomial_at_rational(
130    field: &Arc<NumberField>,
131    coefficients: &[AlgebraicElement],
132    point: &exact_rational,
133) -> AlgebraicElement {
134    // Horner-free evaluation of Σ cᵢ pointⁱ over Q(γ), accumulating rational
135    // powers of the point as it goes.
136    let mut result = zero_element(field);
137    let mut power = rational_one();
138    for coefficient in coefficients.iter() {
139        let term = scale_by_rational(coefficient, &power);
140        result = element_add(&result, &term);
141        power *= point.clone();
142    }
143    result
144}

146fn norm_polynomial(
147    context: &Context,
148    coefficients: &[AlgebraicElement],
149    field: &Arc<NumberField>,
150) -> Option<Polynomial> {
151    // The norm Res_γ(F(z,γ), m_γ(γ)) as a polynomial in a fresh variable z. F is
152    // the bivariate lift of f over the power basis of γ: the coefficient of γʲ is
153    // Σᵢ repᵢ[j] zⁱ. The resultant against the field's defining polynomial removes
154    // γ, leaving a ℚ-polynomial whose real roots superset f's.
155    let generator = field.generator;
156    let z_variable = context.symbol_table.fresh("__lift_z");
157    let mut gamma_terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
158    for power_of_generator in 0..field.degree {
159        let mut z_dense: Vec<exact_rational> = Vec::new();
160        for (degree_in_z, coefficient) in coefficients.iter().enumerate() {
161            let entry = coefficient
162                .representative
163                .get(power_of_generator)
164                .cloned()
165                .unwrap_or(rational_zero());
166            if entry != rational_zero() {
167                if z_dense.len() <= degree_in_z {
168                    z_dense.resize(degree_in_z + 1, rational_zero());
169                }
170                z_dense[degree_in_z] = entry;
171            }
172        }
173        dense::normalize(&mut z_dense);
174        if z_dense.is_empty() {
175            continue;
176        }
177        let inner = dense::from_dense(z_variable, &z_dense);
178        gamma_terms.push((
179            power_of_generator,
180            PolynomialCoefficient::Nested(Box::new(inner)),
181        ));
182    }
183    if gamma_terms.is_empty() {
184        return None;
185    }
186    gamma_terms.sort_by(|left, right| right.0.cmp(&left.0));
187    let lifted = Polynomial {
188        main_variable: generator,
189        terms: gamma_terms,
190    };
191    let resultant_value = sylvester_resultant(&lifted, &field.defining_polynomial);
192    match resultant_value {
193        PolynomialCoefficient::Nested(boxed) => {
194            if boxed.main_variable == z_variable {
195                Some(*boxed)
196            } else {
197                None
198            }
199        }
200        PolynomialCoefficient::Numeric(numeric_value) => {
201            if crate::number::is_zero(&numeric_value) {
202                None
203            } else {
204                Some(crate::polynomial::constant_polynomial(
205                    z_variable,
206                    numeric_value,
207                ))
208            }
209        }
210        PolynomialCoefficient::Algebraic(_) => None,
211    }
212}

214fn square_free_over_field(
215    context: &Context,
216    coefficients: &[AlgebraicElement],
217    field: &Arc<NumberField>,
218) -> Option<Vec<AlgebraicElement>> {
219    // The squarefree part of f over Q(γ): f / gcd(f, f'). Guarantees simple real
220    // roots so the sign-change sieve is complete. None (falling back to the input)
221    // when any step cannot be carried out.
222    let z_variable = context.symbol_table.fresh("__square_free_z");
223    let f_polynomial = coefficients_to_polynomial(z_variable, coefficients);
224    let derivative = derivative_coefficients(coefficients);
225    if derivative.is_empty() {
226        return None;
227    }
228    let derivative_polynomial = coefficients_to_polynomial(z_variable, &derivative);
229    let common = euclidean_gcd(&f_polynomial, &derivative_polynomial, field)?;
230    if degree(&common).unwrap_or(0) == 0 {
231        return None;
232    }
233    let (quotient, _) = divide_with_remainder(&f_polynomial, &common, field)?;
234    polynomial_to_coefficients(&quotient, field)
235}

237fn coefficients_to_polynomial(
238    z_variable: SymbolIdentifier,
239    coefficients: &[AlgebraicElement],
240) -> Polynomial {
241    // Build the polynomial Σ cᵢ zⁱ over Q(γ) from an ascending coefficient list.
242    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
243    for (degree_in_z, coefficient) in coefficients.iter().enumerate() {
244        if !element_is_zero(coefficient) {
245            terms.push((
246                degree_in_z,
247                PolynomialCoefficient::Algebraic(coefficient.clone()),
248            ));
249        }
250    }
251    terms.sort_by(|left, right| right.0.cmp(&left.0));
252    Polynomial {
253        main_variable: z_variable,
254        terms,
255    }
256}

258fn derivative_coefficients(coefficients: &[AlgebraicElement]) -> Vec<AlgebraicElement> {
259    // The coefficients of f'(z) = Σ i·cᵢ z^{i-1}, ascending.
260    if coefficients.len() <= 1 {
261        return Vec::new();
262    }
263    let mut result: Vec<AlgebraicElement> = Vec::with_capacity(coefficients.len() - 1);
264    for (degree_in_z, coefficient) in coefficients.iter().enumerate().skip(1) {
265        let factor = exact_rational::from(exact_integer::from(degree_in_z as i64));
266        result.push(scale_by_rational(coefficient, &factor));
267    }
268    result
269}

271fn polynomial_to_coefficients(
272    value: &Polynomial,
273    field: &Arc<NumberField>,
274) -> Option<Vec<AlgebraicElement>> {
275    // Extract the ascending coefficient list of a polynomial over Q(γ).
276    let polynomial_degree = degree(value)?;
277    let mut result = vec![zero_element(field); polynomial_degree + 1];
278    for (exponent, coefficient) in value.terms.iter() {
279        result[*exponent] = coefficient_to_element(coefficient, field)?;
280    }
281    Some(result)
282}

284fn trim_high_zeros(coefficients: &[AlgebraicElement]) -> Vec<AlgebraicElement> {
285    // Drop trailing (high-degree) zero coefficients so the leading coefficient is
286    // non-zero.
287    let mut trimmed = coefficients.to_vec();
288    while trimmed.last().map(element_is_zero).unwrap_or(false) {
289        trimmed.pop();
290    }
291    trimmed
292}
