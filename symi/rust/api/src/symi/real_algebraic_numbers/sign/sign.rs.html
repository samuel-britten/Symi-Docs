16/// Public native data structure named `real_algebraic_point`.
17pub struct RealAlgebraicPoint {
18    // A point of ℝⁿ whose coordinates are real algebraic numbers co-residing in
19    // one real-embedded extension Q(γ) — the CAD sample-point representation. The
20    // i-th coordinate is the value of variables[i], given as an algebraic_element
21    // of embedding.field.
22    /// Stores the `embedding` value.
23    pub embedding: RealEmbeddedNumberField,
24    /// Stores the `coordinates` value.
25    pub coordinates: Vec<AlgebraicElement>,
26    /// Stores the `variables` value.
27    pub variables: Vec<SymbolIdentifier>,
28}

30/// Performs the advanced native `sign_of_polynomial_at_real_point` operation.
31pub fn sign_of_polynomial_at_real_point(
32    value: &ExpressionReference,
33    point: &RealAlgebraicPoint,
34) -> Option<i32> {
35    // The sign of a polynomial, specialised at a real algebraic point — the
36    // sign-invariant cell test the CAD lifting reads. The expression is converted
37    // to a polynomial in the point's variables (declining outside ℚ[x₁,…,xₙ]),
38    // evaluated to a single algebraic_element of Q(γ) by algebraic-number arithmetic, and its
39    // sign read through the real embedding. None outside ℚ[·].
40    let polynomial_value = expression_to_polynomial(value, &point.variables)?;
41    let element = evaluate_polynomial_at_point(&polynomial_value, point)?;
42    Some(sign_of_element(&point.embedding, &element))
43}

45/// Performs the advanced native `evaluate_polynomial_at_point` operation.
46pub fn evaluate_polynomial_at_point(
47    polynomial_value: &Polynomial,
48    point: &RealAlgebraicPoint,
49) -> Option<AlgebraicElement> {
50    // Evaluate a (possibly nested multivariate) polynomial at the point's
51    // coordinates, returning an element of the embedded field. Each variable is
52    // replaced by its coordinate value; coefficients are evaluated recursively.
53    let field = &point.embedding.field;
54    let variable_value = coordinate_for(polynomial_value.main_variable, point)?;
55    let mut result = zero_element(field);
56    for (exponent, coefficient) in polynomial_value.terms.iter() {
57        let coefficient_element = evaluate_coefficient(coefficient, point)?;
58        let power = power_integer(&variable_value, &exact_integer::from(*exponent as i64))?;
59        let term = element_multiply(&coefficient_element, &power);
60        result = element_add(&result, &term);
61    }
62    Some(result)
63}

65fn evaluate_coefficient(
66    coefficient: &PolynomialCoefficient,
67    point: &RealAlgebraicPoint,
68) -> Option<AlgebraicElement> {
69    // Lift a polynomial coefficient into the embedded field: a numeric scalar
70    // becomes a rational element, a nested polynomial is evaluated recursively at
71    // the point, and an algebraic coefficient is accepted only when it already
72    // lives in the embedded field.
73    let field = &point.embedding.field;
74    match coefficient {
75        PolynomialCoefficient::Numeric(numeric_value) => Some(rational_element(
76            field,
77            dense::number_to_rational(numeric_value),
78        )),
79        PolynomialCoefficient::Nested(nested_polynomial) => {
80            evaluate_polynomial_at_point(nested_polynomial, point)
81        }
82        PolynomialCoefficient::Algebraic(element) => {
83            if Arc::ptr_eq(&element.field, field) {
84                Some(element.clone())
85            } else {
86                None
87            }
88        }
89    }
90}

92fn coordinate_for(
93    variable: SymbolIdentifier,
94    point: &RealAlgebraicPoint,
95) -> Option<AlgebraicElement> {
96    // The coordinate value of a variable at the point, or None when the variable
97    // is not among the point's named coordinates (a genuinely free variable the
98    // sign of which is undetermined).
99    point
100        .variables
101        .iter()
102        .position(|candidate| *candidate == variable)
103        .map(|index| point.coordinates[index].clone())
104}

106/// Performs the advanced native `single_variable_point` operation.
107pub fn single_variable_point(
108    embedding: RealEmbeddedNumberField,
109    variable: SymbolIdentifier,
110    value: AlgebraicElement,
111) -> RealAlgebraicPoint {
112    // Convenience constructor for a one-dimensional sample point.
113    RealAlgebraicPoint {
114        embedding,
115        coordinates: vec![value],
116        variables: vec![variable],
