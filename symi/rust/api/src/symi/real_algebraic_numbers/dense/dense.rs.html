12// Dense ascending-degree rational-vector view of a univariate ℚ-polynomial,
13// shared by every real_algebraic_numbers submodule. The empty vector is the
14// unique zero polynomial; the leading (highest-degree) coefficient is always
15// non-zero after normalize. These mirror the private helpers in
16// polynomial/real_roots.rs so this module can manipulate defining polynomials without
17// duplicating its isolation internals.
18
19/// Performs the advanced native `to_dense` operation.
20pub fn to_dense(value: &Polynomial) -> Option<Vec<exact_rational>> {
21    // Project a univariate polynomial whose coefficients are all numeric onto a
22    // dense ascending-degree rational vector. None for any nested or algebraic
23    // coefficient (a multivariate or extension input).
24    let degree = match crate::polynomial::degree(value) {
25        Some(degree) => degree,
26        None => return Some(Vec::new()),
27    };
28    let mut dense = vec![rational_zero(); degree + 1];
29    for (exponent, coefficient) in value.terms.iter() {
30        match coefficient {
31            PolynomialCoefficient::Numeric(numeric_value) => {
32                dense[*exponent] = number_to_rational(numeric_value);
33            }
34            _ => return None,
35        }
36    }
37    normalize(&mut dense);
38    Some(dense)
39}

41/// Performs the advanced native `from_dense` operation.
42pub fn from_dense(main_variable: SymbolIdentifier, dense: &[exact_rational]) -> Polynomial {
43    // Rebuild a univariate polynomial from a dense rational coefficient vector
44    // under the supplied main variable.
45    let coefficients: Vec<Number> = dense.iter().map(rational_to_number).collect();
46    univariate_from_dense(main_variable, coefficients)
47}

49/// Performs the advanced native `evaluate` operation.
50pub fn evaluate(dense: &[exact_rational], point: &exact_rational) -> exact_rational {
51    // Horner evaluation at a rational point.
52    let mut accumulator = rational_zero();
53    for coefficient in dense.iter().rev() {
54        accumulator = accumulator * point + coefficient;
55    }
56    accumulator
57}

59/// Performs the advanced native `rational_sign` operation.
60pub fn rational_sign(value: &exact_rational) -> i32 {
61    // The sign of a rational as -1, 0, or 1.
62    match crate::exact_arithmetic::rational_sign(value) {
63        Ordering::Less => -1,
64        Ordering::Equal => 0,
65        Ordering::Greater => 1,
66    }
67}

69/// Performs the advanced native `reflect` operation.
70pub fn reflect(dense: &[exact_rational]) -> Vec<exact_rational> {
71    // The polynomial p(-x): negate every odd-degree coefficient. The roots of
72    // the result are the negations of the roots of p.
73    dense
74        .iter()
75        .enumerate()
76        .map(|(exponent, coefficient)| {
77            if exponent % 2 == 0 {
78                coefficient.clone()
79            } else {
80                -coefficient.clone()
81            }
82        })
83        .collect()
84}

86/// Performs the advanced native `reverse` operation.
87pub fn reverse(dense: &[exact_rational]) -> Vec<exact_rational> {
88    // The reversal x^deg · p(1/x): reverse the coefficient order. The non-zero
89    // roots of the result are the reciprocals of the non-zero roots of p.
90    let mut reversed = dense.to_vec();
91    reversed.reverse();
92    normalize(&mut reversed);
93    reversed
94}

96/// Performs the advanced native `squarefree_part` operation.
97pub fn squarefree_part(dense: &[exact_rational]) -> Vec<exact_rational> {
98    // The squarefree part p / gcd(p, p'), returned monic. Every root of the
99    // result is simple, which makes an isolating interval a one-root indicator.
100    let mut working = dense.to_vec();
101    normalize(&mut working);
102    if degree_of(&working) == 0 {
103        make_monic(&mut working);
104        return working;
105    }
106    let derivative_polynomial = derivative(&working);
107    let common = monic_gcd(&working, &derivative_polynomial);
108    if degree_of(&common) == 0 {
109        make_monic(&mut working);
110        return working;
111    }
112    let (mut quotient, _) = divide(&working, &common);
113    make_monic(&mut quotient);
114    quotient
115}

117/// Performs the advanced native `monic_gcd` operation.
118pub fn monic_gcd(left: &[exact_rational], right: &[exact_rational]) -> Vec<exact_rational> {
119    // Monic greatest common divisor over the rationals by the Euclidean
120    // algorithm.
121    let mut first = left.to_vec();
122    normalize(&mut first);
123    let mut second = right.to_vec();
124    normalize(&mut second);
125    while !second.is_empty() {
126        let (_, remainder) = divide(&first, &second);
127        first = second;
128        second = remainder;
129    }
130    make_monic(&mut first);
131    first
132}

134/// Performs the advanced native `derivative` operation.
135pub fn derivative(dense: &[exact_rational]) -> Vec<exact_rational> {
136    // The formal derivative in the dense representation.
137    if dense.len() <= 1 {
138        return Vec::new();
139    }
140    let mut result: Vec<exact_rational> = Vec::with_capacity(dense.len() - 1);
141    for (exponent, coefficient) in dense.iter().enumerate().skip(1) {
142        result.push(coefficient * rational_from_integer(exact_integer::from(exponent as i64)));
143    }
144    normalize(&mut result);
145    result
146}

148/// Performs the advanced native `divide` operation.
149pub fn divide(
150    dividend: &[exact_rational],
151    divisor: &[exact_rational],
152) -> (Vec<exact_rational>, Vec<exact_rational>) {
153    // Euclidean division over the rationals, returning (quotient, remainder).
154    let divisor_degree = degree_of(divisor);
155    let mut remainder = dividend.to_vec();
156    normalize(&mut remainder);
157    if remainder.len() < divisor.len() {
158        return (Vec::new(), remainder);
159    }
160    let divisor_leading = divisor[divisor_degree].clone();
161    let mut quotient = vec![rational_zero(); remainder.len() - divisor_degree];
162    while !remainder.is_empty() && degree_of(&remainder) >= divisor_degree {
163        let remainder_degree = degree_of(&remainder);
164        let factor = &remainder[remainder_degree] / &divisor_leading;
165        let shift = remainder_degree - divisor_degree;
166        quotient[shift] = factor.clone();
167        for index in 0..divisor.len() {
168            let term = &factor * &divisor[index];
169            remainder[shift + index] = &remainder[shift + index] - &term;
170        }
171        normalize(&mut remainder);
172    }
173    normalize(&mut quotient);
174    (quotient, remainder)
175}

177/// Performs the advanced native `make_monic` operation.
178pub fn make_monic(dense: &mut Vec<exact_rational>) {
179    // Scale so the leading coefficient is one; a no-op on the zero polynomial.
180    normalize(dense);
181    if dense.is_empty() {
182        return;
183    }
184    let leading = dense.last().unwrap().clone();
185    if leading == rational_one() {
186        return;
187    }
188    for coefficient in dense.iter_mut() {
189        *coefficient /= &leading;
190    }
191}

193/// Performs the advanced native `normalize` operation.
194pub fn normalize(dense: &mut Vec<exact_rational>) {
195    // Drop trailing (high-degree) zero coefficients so the leading coefficient is
196    // always non-zero and the empty vector is the unique zero polynomial.
197    while let Some(last) = dense.last() {
198        if crate::exact_arithmetic::rational_sign(last) == Ordering::Equal {
199            dense.pop();
200        } else {
201            break;
202        }
203    }
204}

206/// Performs the advanced native `degree_of` operation.
207pub fn degree_of(dense: &[exact_rational]) -> usize {
208    // Degree of a normalized dense polynomial; the zero and constant polynomials
209    // both report zero.
210    if dense.is_empty() { 0 } else { dense.len() - 1 }
211}

213/// Performs the advanced native `interval_width` operation.
214pub fn interval_width(interval: &IsolatingInterval) -> exact_rational {
215    // The width of an isolating interval.
216    &interval.upper - &interval.lower
217}

219/// Performs the advanced native `exact_interval` operation.
220pub fn exact_interval(value: exact_rational) -> IsolatingInterval {
221    // A collapsed interval marking an exact rational value.
222    IsolatingInterval {
223        lower: value.clone(),
224        upper: value,
225    }

228/// Performs the advanced native `number_to_rational` operation.
229pub fn number_to_rational(value: &Number) -> exact_rational {
230    // View a number as a rational.
231    match value {
232        Number::Integer(integer_value) => exact_rational::from(integer_value.clone()),
233        Number::Rational(rational_value) => rational_value.clone(),
234    }
235}

237/// Performs the advanced native `rational_to_number` operation.
238pub fn rational_to_number(value: &exact_rational) -> Number {
239    // Convert a rational to a number, collapsing an integral value to the
240    // integer variant to match the invariant number::from_integers maintains.
241    if rational_is_integer(value) {
242        Number::Integer(rational_to_integer(value).expect("integral rational converts"))
243    } else {
244        Number::Rational(value.clone())
245    }
