1// Abramov polynomial-search closure. Given a linear recurrence
2// with polynomial coefficients sum_k q_k(n) * c(n + k) = 0, computes
3// nonzero polynomial solutions c(n) in Q[n] via the degree bound from
4// Abramov-Bronstein-Petkovsek (1995) followed by an undetermined-
5// coefficient ansatz and a small Gaussian-elimination linear solve
6// over Q.
7//
8// The implementation deliberately stays inside the polynomial /
9// rational-number layer: the linear system is at most (d+1)x(d+1)
10// where d is the Abramov degree bound, which for this module's test corpus
11// stays under 6.
12

26/// Performs the advanced native `polynomial_search` operation.
27pub fn polynomial_search(
28    transformed: &[Polynomial],
29    index_variable: SymbolIdentifier,
30) -> Vec<Polynomial> {
31    // Public entry. Returns every nonzero polynomial solution c(n) of
32    // the recurrence sum_k transformed[k](n) * c(n + k) = 0, walking
33    // the null space of the linear system that falls out of the
34    // Abramov degree-bound + ansatz closure.
35    if transformed.is_empty() {
36        return Vec::new();
37    }
38    if transformed.iter().all(super::super::polynomial::is_zero) {
39        return Vec::new();
40    }
41    let degree_bound = abramov_degree_bound(transformed);
42    if degree_bound < 0 {
43        return Vec::new();
44    }
45    let unknown_count = (degree_bound as usize) + 1;
46    let (rows, columns) = build_linear_system(transformed, unknown_count);
47    let null_basis = nullspace_basis(&rows, columns);
48    null_basis
49        .into_iter()
50        .filter(|vector| vector.iter().any(|entry| *entry != rational_zero()))
51        .map(|vector| polynomial_from_rational_vector(&vector, index_variable))
52        .collect()
53}

55fn abramov_degree_bound(transformed: &[Polynomial]) -> i64 {
56    // Compute Abramov-Bronstein-Petkovsek's degree bound for polynomial
57    // solutions of a homogeneous linear recurrence with polynomial
58    // coefficients. The published formula (Abramov-Bronstein-Petkovsek
59    // 1995; HandWiki, "Polynomial solutions of P-recursive equations")
60    // uses the transformed coefficients
61    //   q_i(n) = Σ_{k ≥ i} C(k, i) · p_k(n)
62    // (the Newton-style forward-difference basis) rather than the
63    // original p_k directly. With f = 0 (homogeneous case) the bound
64    // collapses to
65    //   d ≤ max{ -b - 1, d_α }
66    // where
67    //   b      = max_{i} (deg(q_i) - i),
68    //   α(n)   = Σ_{deg(q_i)-i == b} lc(q_i) * (n)_{i↓}
69    //              (with (n)_{i↓} the i-th falling factorial of n),
70    //   d_α    = max{ m ∈ N : α(m) = 0 } ∪ {-1}.
71    let newton_basis: Vec<Vec<exact_rational>> = newton_basis_of(transformed);
72    let mut max_shift: Option<i64> = None;
73    for (i, polynomial_coefficients) in newton_basis.iter().enumerate() {
74        let polynomial_degree = leading_dense_degree(polynomial_coefficients);
75        if polynomial_degree.is_none() {
76            continue;
77        }
78        let candidate = polynomial_degree.unwrap() - (i as i64);
79        if max_shift.is_none_or(|current| candidate > current) {
80            max_shift = Some(candidate);
81        }
82    }
83    let b = match max_shift {
84        Some(value) => value,
85        None => return -1,
86    };
87    let alpha = build_alpha_polynomial_from_newton(&newton_basis, b);
88    let cauchy_bound = cauchy_root_bound(&alpha);
89    let mut d_alpha: i64 = -1;
90    for candidate in 0..=cauchy_bound {
91        if evaluate_rational_polynomial_at_integer(&alpha, candidate) == rational_zero() {
92            if candidate as i64 > d_alpha {
93                d_alpha = candidate as i64;
94            }
95        }
96    }
97    let from_b = -b - 1;
98    if d_alpha > from_b { d_alpha } else { from_b }
99}

101fn newton_basis_of(transformed: &[Polynomial]) -> Vec<Vec<exact_rational>> {
102    // Compute the Newton (forward-difference) basis q_i(n) = Σ_{k ≥ i}
103    // C(k, i) · p_k(n). Each row is a dense rational-coefficient
104    // polynomial indexed by power of n. The transformation lifts the
105    // original recurrence into the basis where Abramov's degree bound
106    // is naturally stated.
107    let order = transformed.len();
108    let mut basis: Vec<Vec<exact_rational>> = vec![Vec::new(); order];
109    for k in 0..order {
110        if super::super::polynomial::is_zero(&transformed[k]) {
111            continue;
112        }
113        let p_k_dense = polynomial_to_dense_rationals(&transformed[k]);
114        if p_k_dense.is_empty() {
115            continue;
116        }
117        for i in 0..=k {
118            let binomial_value = exact_rational::from(integer_binomial(k as u64, i as u64));
119            while basis[i].len() < p_k_dense.len() {
120                basis[i].push(rational_zero());
121            }
122            for (power, coefficient) in p_k_dense.iter().enumerate() {
123                basis[i][power] = &basis[i][power] + &(&binomial_value * coefficient);
124            }
125        }
126    }
127    basis
128}

130fn leading_dense_degree(coefficients: &[exact_rational]) -> Option<i64> {
131    // Return the highest power of n with a nonzero coefficient in the
132    // dense rational representation. Returns None for the zero
133    // polynomial (encoded as an empty or all-zero vector).
134    for (index, coefficient) in coefficients.iter().enumerate().rev() {
135        if *coefficient != rational_zero() {
136            return Some(index as i64);
137        }
138    }
139    None
140}

142fn build_alpha_polynomial_from_newton(
143    newton_basis: &[Vec<exact_rational>],
144    b: i64,
145) -> Vec<exact_rational> {
146    // α(n) = Σ_{deg(q_i) - i == b} lc(q_i) · (n)_{i↓}. Each
147    // qualifying q_i contributes its leading rational coefficient
148    // times the i-th falling factorial expansion in n. The dense
149    // accumulator below is widened on demand to fit the largest
150    // contributing falling factorial.
151    let mut accumulator: Vec<exact_rational> = Vec::new();
152    for (i, polynomial_coefficients) in newton_basis.iter().enumerate() {
153        let polynomial_degree = match leading_dense_degree(polynomial_coefficients) {
154            Some(value) => value,
155            None => continue,
156        };
157        if polynomial_degree - (i as i64) != b {
158            continue;
159        }
160        let leading_coefficient_value = polynomial_coefficients[polynomial_degree as usize].clone();
161        if leading_coefficient_value == rational_zero() {
162            continue;
163        }
164        let falling_factorial_coefficients = falling_factorial_dense(i);
165        for (power, falling_coefficient) in falling_factorial_coefficients.iter().enumerate() {
166            let scaled = &leading_coefficient_value * falling_coefficient;
167            while accumulator.len() <= power {
168                accumulator.push(rational_zero());
169            }
170            accumulator[power] = &accumulator[power] + &scaled;
171        }
172    }
173    accumulator
174}

176#[allow(dead_code)]
177fn build_alpha_polynomial(transformed: &[Polynomial], b: i64) -> Vec<exact_rational> {
178    // Legacy helper retained for the test surface; the production
179    // path now uses build_alpha_polynomial_from_newton against the
180    // forward-difference basis q_i instead of the raw p_k.
181    let mut accumulator: Vec<exact_rational> = Vec::new();
182    for (i, polynomial_value) in transformed.iter().enumerate() {
183        if super::super::polynomial::is_zero(polynomial_value) {
184            continue;
185        }
186        let polynomial_degree = degree(polynomial_value).unwrap_or(0) as i64;
187        if polynomial_degree - (i as i64) != b {
188            continue;
189        }
190        let leading_coefficient_value = match leading_coefficient(polynomial_value)
191            .and_then(|coefficient| coefficient_to_number(&coefficient))
192        {
193            Some(value) => number_to_rational(value),
194            None => continue,
195        };
196        let falling_factorial_coefficients = falling_factorial_dense(i);
197        for (power, falling_coefficient) in falling_factorial_coefficients.iter().enumerate() {
198            let scaled = &leading_coefficient_value * falling_coefficient;
199            while accumulator.len() <= power {
200                accumulator.push(rational_zero());
201            }
202            accumulator[power] = &accumulator[power] + &scaled;
203        }
204    }
205    accumulator
206}

208fn falling_factorial_dense(order: usize) -> Vec<exact_rational> {
209    // Expand (n)_{order↓} = n · (n - 1) · ... · (n - order + 1) into a
210    // dense rational coefficient vector indexed by power of n. The
211    // recursion (n)_{(k+1)↓} = (n)_{k↓} · (n - k) lets us build the
212    // expansion iteratively. Returns [1] for order = 0.
213    let mut current: Vec<exact_rational> = vec![rational_one()];
214    for shift in 0..order {
215        let mut next: Vec<exact_rational> = vec![rational_zero(); current.len() + 1];
216        let shift_rational = exact_rational::from(shift as i64);
217        for (power, coefficient) in current.iter().enumerate() {
218            next[power + 1] = &next[power + 1] + coefficient;
219            next[power] = &next[power] - &(coefficient * &shift_rational);
220        }
221        current = next;
222    }
223    current
224}

226fn cauchy_root_bound(polynomial_coefficients: &[exact_rational]) -> u64 {
227    // Cauchy bound on positive roots: if α(n) = Σ a_i · n^i with a_d
228    // the leading coefficient, then every real root satisfies
229    //   |n| ≤ 1 + max_i |a_i / a_d|.
230    // Returns a u64 ceiling on this quantity; falls back to a wide
231    // default (256) when the polynomial is zero so we never silently
232    // drop a root.
233    if polynomial_coefficients.is_empty() {
234        return 0;
235    }
236    let leading = polynomial_coefficients
237        .iter()
238        .rposition(|coefficient| *coefficient != rational_zero());
239    let leading_index = match leading {
240        None => return 0,
241        Some(index) => index,
242    };
243    let leading_value = &polynomial_coefficients[leading_index];
244    let mut max_ratio = rational_zero();
245    for coefficient in &polynomial_coefficients[..leading_index] {
246        let ratio = abs_rational(&(coefficient / leading_value));
247        if ratio > max_ratio {
248            max_ratio = ratio;
249        }
250    }
251    let bound = &max_ratio + &rational_one();
252    let bound_ceiling = rational_ceiling_to_u64(&bound);
253    bound_ceiling.max(1)
254}

256fn evaluate_rational_polynomial_at_integer(
257    polynomial_coefficients: &[exact_rational],
258    n: u64,
259) -> exact_rational {
260    // Horner-style evaluation of a dense rational polynomial at a
261    // non-negative integer point. Used by the falling-factorial root
262    // search to detect d_α.
263    let n_rational = exact_rational::from(n);
264    let mut accumulator = rational_zero();
265    for coefficient in polynomial_coefficients.iter().rev() {
266        accumulator = &accumulator * &n_rational + coefficient;
267    }
268    accumulator
269}

271fn build_linear_system(
272    transformed: &[Polynomial],
273    unknown_count: usize,
274) -> (Vec<Vec<exact_rational>>, usize) {
275    // For the ansatz c(n) = Σ_{j=0}^{d} u_j · n^j, expand the
276    // recurrence sum_k transformed[k](n) · c(n + k) into a polynomial
277    // in n whose coefficients are linear in (u_0, ..., u_d). Each
278    // coefficient of n^m must equal zero, so the rows of the linear
279    // system are indexed by m and the columns by j. The (m, j) entry
280    // sums Σ_k Σ_{i+l=m} C(j, i) · k^{j-i} · [n^l] q_k(n).
281    let mut rows: Vec<Vec<exact_rational>> = Vec::new();
282    for (k, polynomial_value) in transformed.iter().enumerate() {
283        if super::super::polynomial::is_zero(polynomial_value) {
284            continue;
285        }
286        let dense_coefficients = polynomial_to_dense_rationals(polynomial_value);
287        for j in 0..unknown_count {
288            for i in 0..=j {
289                let binomial_coefficient =
290                    exact_rational::from(integer_binomial(j as u64, i as u64));
291                let k_power = if i == j {
292                    rational_one()
293                } else {
294                    let mut accumulator = rational_one();
295                    let base = exact_rational::from(k as i64);
296                    for _ in 0..(j - i) {
297                        accumulator = &accumulator * &base;
298                    }
299                    accumulator
300                };
301                let prefactor = &binomial_coefficient * &k_power;
302                for (l, polynomial_coefficient_value) in dense_coefficients.iter().enumerate() {
303                    if *polynomial_coefficient_value == rational_zero() {
304                        continue;
305                    }
306                    let row_index = i + l;
307                    while rows.len() <= row_index {
308                        rows.push(vec![rational_zero(); unknown_count]);
309                    }
310                    let entry = &prefactor * polynomial_coefficient_value;
311                    rows[row_index][j] = &rows[row_index][j] + &entry;
312                }
313            }
314        }
315    }
316    (rows, unknown_count)
317}

319fn polynomial_to_dense_rationals(polynomial_value: &Polynomial) -> Vec<exact_rational> {
320    // Convert a sparse polynomial with rational/integer coefficients
321    // to a dense rational-coefficient vector indexed by power. Slots
322    // not present in the sparse form are zero. Bails on the first
323    // nested coefficient because this closure's input is a univariate polynomial
324    // in n with concrete numeric coefficients.
325    let polynomial_degree = match degree(polynomial_value) {
326        Some(value) => value,
327        None => return Vec::new(),
328    };
329    let mut dense: Vec<exact_rational> = vec![rational_zero(); polynomial_degree + 1];
330    for (exponent, coefficient) in &polynomial_value.terms {
331        let numeric_value = match coefficient_to_number(coefficient) {
332            Some(value) => value,
333            None => return Vec::new(),
334        };
335        dense[*exponent] = number_to_rational(numeric_value);
336    }
337    dense
338}

340fn integer_binomial(top: u64, bottom: u64) -> exact_integer {
341    // Closed-form binomial coefficient over the integers. Mirrors the
342    // helper in summation.rs but local to this module so we do not
343    // introduce a cross-module dependency on summation just for one
344    // utility.
345    if bottom > top {
346        return integer_zero();
347    }
348    let effective_bottom = bottom.min(top - bottom);
349    let mut result = integer_one();
350    for step in 0..effective_bottom {
351        result = &result * &exact_integer::from(top - step) / &exact_integer::from(step + 1);
352    }
353    result
354}

356fn nullspace_basis(rows: &[Vec<exact_rational>], columns: usize) -> Vec<Vec<exact_rational>> {
357    // Compute a basis for the null space of the matrix encoded by
358    // `rows` (m × columns) over Q via Gauss-Jordan elimination with
359    // partial pivoting on the first nonzero coefficient. Free
360    // variables yield basis vectors with the free coordinate set to
361    // one and dependent coordinates back-substituted from the reduced
362    // form. The empty-input case returns the identity-basis spanning
363    // every coordinate axis.
364    if rows.is_empty() {
365        return (0..columns)
366            .map(|free| {
367                let mut vector = vec![rational_zero(); columns];
368                vector[free] = rational_one();
369                vector
370            })
371            .collect();
372    }
373    let mut matrix: Vec<Vec<exact_rational>> = rows.iter().map(|row| row.clone()).collect();
374    let row_count = matrix.len();
375    let mut pivot_columns: Vec<Option<usize>> = Vec::new();
376    let mut row_cursor: usize = 0;
377    for column in 0..columns {
378        if row_cursor >= row_count {
379            break;
380        }
381        let mut pivot_row: Option<usize> = None;
382        for candidate in row_cursor..row_count {
383            if matrix[candidate][column] != rational_zero() {
384                pivot_row = Some(candidate);
385                break;
386            }
387        }
388        let pivot_row = match pivot_row {
389            None => continue,
390            Some(value) => value,
391        };
392        matrix.swap(row_cursor, pivot_row);
393        let pivot_value = matrix[row_cursor][column].clone();
394        for entry in &mut matrix[row_cursor] {
395            *entry = &*entry / &pivot_value;
396        }
397        for row in 0..row_count {
398            if row == row_cursor {
399                continue;
400            }
401            let factor = matrix[row][column].clone();
402            if factor == rational_zero() {
403                continue;
404            }
405            for column_index in 0..columns {
406                let pivot_entry = matrix[row_cursor][column_index].clone();
407                matrix[row][column_index] = &matrix[row][column_index] - &(&factor * &pivot_entry);
408            }
409        }
410        pivot_columns.push(Some(column));
411        row_cursor += 1;
412    }
413    let mut column_to_pivot_row: Vec<Option<usize>> = vec![None; columns];
414    for (row, pivot_column) in pivot_columns.iter().enumerate() {
415        if let Some(column) = pivot_column {
416            column_to_pivot_row[*column] = Some(row);
417        }
418    }
419    let mut basis: Vec<Vec<exact_rational>> = Vec::new();
420    for free_column in 0..columns {
421        if column_to_pivot_row[free_column].is_some() {
422            continue;
423        }
424        let mut vector = vec![rational_zero(); columns];
425        vector[free_column] = rational_one();
426        for column in 0..columns {
427            if let Some(row) = column_to_pivot_row[column] {
428                vector[column] = -matrix[row][free_column].clone();
429            }
430        }
431        basis.push(vector);
432    }
433    basis
434}

436fn polynomial_from_rational_vector(
437    coefficients: &[exact_rational],
438    index_variable: SymbolIdentifier,
439) -> Polynomial {
440    // Reify a dense rational coefficient vector back into a univariate
441    // polynomial over the named index variable. Used to lift each null-
442    // space basis vector back to the polynomial layer for the caller.
443    if coefficients.is_empty() {
444        return zero_polynomial(index_variable);
445    }
446    if coefficients
447        .iter()
448        .all(|coefficient| *coefficient == rational_zero())
449    {
450        return zero_polynomial(index_variable);
451    }
452    let dense_numbers: Vec<Number> = coefficients.iter().map(rational_to_number).collect();
453    univariate_from_dense(index_variable, dense_numbers)
454}

456fn rational_to_number(value: &exact_rational) -> Number {
457    // Convert an exact rational to the polynomial-layer number
458    // representation. The collapse rule for denominators equal to one
459    // is applied through the existing `number_rational` helper which
460    // re-routes through the integer variant.
461    let parts = rational_parts(value);
462    crate::number::from_integers(parts.numerator, natural_to_integer(parts.denominator))
463        .unwrap_or_else(|| Number::Integer(integer_zero()))
464}

466fn number_to_rational(value: Number) -> exact_rational {
467    // Convert a polynomial-layer number into an exact rational.
468    // Used by the linear-system construction so every coefficient
469    // lives in the same arithmetic domain regardless of whether it
470    // originated as an integer or a rational.
471    match value {
472        Number::Integer(integer_value) => exact_rational::from(integer_value),
473        Number::Rational(rational_value) => rational_value,
474    }
475}

477fn abs_rational(value: &exact_rational) -> exact_rational {
478    // Absolute value of a rational. Used by the Cauchy root bound to
479    // ensure the search range is symmetric around zero.
480    if *value < rational_zero() {
481        -value.clone()
482    } else {
483        value.clone()
484    }
485}

487fn rational_ceiling_to_u64(value: &exact_rational) -> u64 {
488    // Ceiling of a non-negative rational, returned as u64 with a hard
489    // cap at 256 to bound the falling-factorial root scan. The cap is
490    // generous given the typical recurrence degrees in this module's test corpus and keeps
491    // the polynomial-search cost bounded.
492    if *value < rational_zero() {
493        return 0;
494    }
495    let candidate = integer_to_u64(&rational_ceiling(value)).unwrap_or(256);
496    candidate.min(256)
497}
