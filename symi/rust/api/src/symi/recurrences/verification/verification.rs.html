1// Independent exact verification of linear recurrence solutions.
2//
3// A closed form is a solution of a recurrence when three separate things hold, and checking only
4// the first is the standard way to accept a wrong answer:
5//
6//  1. The recurrence identity itself, proved symbolically in the index rather than at a handful
7//     of sampled integers. A finite sample can refute a candidate; it cannot establish an
8//     identity over an infinite index domain.
9//  2. The complete initial window, evaluated exactly at each declared index. A candidate that
10//     satisfies the recurrence and misses the first value solves a different problem.
11//  3. Uniqueness, which needs the coefficient of the highest shift to be nonvanishing on the
12//     declared range. Without it the recurrence does not determine its solution from the window,
13//     and the first two facts together still do not identify the sequence.
14//
15// The report keeps the three apart, so a caller can tell an unproved identity from a refuted one
16// and from a proved identity whose uniqueness argument is missing.
17

39/// Public native data structure named `linear_recurrence_index_range`.
40///
41/// The two index ranges a recurrence request carries, which are not the same range and cannot be
42/// represented by one number. `solution_index_lower_bound` is the smallest index at which the
43/// closed form is claimed, and it is where the initial window lives. `equation_index_lower_bound`
44/// is the smallest index at which the recurrence identity itself is asserted, and it is larger
45/// whenever the residual reads backward: `r(n) - r(n-1) - r(n-2) = 0` says nothing at `n = 1`,
46/// because `r(-1)` is outside the sequence. Collapsing the two forces the identity to be replayed
47/// at indices where the recurrence is not valid, which is a wrong refutation rather than a proof.
48#[derive(Debug, Clone, Default, PartialEq, Eq)]
49pub struct LinearRecurrenceIndexRange {
50    /// Stores the smallest index at which the closed form is claimed.
51    pub solution_index_lower_bound: Option<ExpressionReference>,
52    /// Stores the smallest index at which the recurrence identity is asserted.
53    pub equation_index_lower_bound: Option<ExpressionReference>,
54}
55
56impl LinearRecurrenceIndexRange {
57    /// Builds an index range from its two independent lower bounds.
58    pub fn new(
59        solution_index_lower_bound: Option<ExpressionReference>,
60        equation_index_lower_bound: Option<ExpressionReference>,
61    ) -> Self {
62        Self {
63            solution_index_lower_bound,
64            equation_index_lower_bound,
65        }
66    }
67}

69/// Public native enumeration named `linear_recurrence_problem_error`.
70#[derive(Debug, Clone, PartialEq, Eq)]
71pub enum LinearRecurrenceProblemError {
72    /// The unknown sequence is a builtin rather than a user function.
73    SequenceFunctionMustBeUserFunction,
74    /// The residual never mentions the unknown sequence.
75    NoSequenceOccurrence,
76    /// An occurrence of the unknown sequence is not a one-argument call.
77    SequenceArityMismatch,
78    /// An occurrence of the unknown sequence is a derivative rather than a shifted call.
79    SequenceOccurrenceIsADerivative,
80    /// A shift offset is not an exact integer displacement of the index.
81    ShiftIsNotAnIntegerDisplacement,
82    /// Every occurrence sits at the same shift, so the request is not a recurrence.
83    NoShiftedOccurrence,
84    /// A datum is stated at an index that still mentions the index variable.
85    ConditionIndexContainsIndexVariable,
86    /// Two data constrain the same index.
87    DuplicateCondition,
88    /// A datum is stated below the smallest index at which the closed form is claimed.
89    ConditionIndexBelowSolutionRange,
90    /// Asserting the recurrence from its declared index would read the closed form below the
91    /// smallest index at which it is claimed.
92    EquationRangeReadsBelowSolutionRange,
93}

95/// Public native data structure named `linear_recurrence_problem`.
96#[derive(Debug, Clone)]
97pub struct LinearRecurrenceProblem {
98    equation: ExpressionReference,
99    sequence_function: FunctionIdentifier,
100    index_variable: SymbolIdentifier,
101    conditions: Vec<LinearRecurrenceCondition>,
102    index_range: LinearRecurrenceIndexRange,
103    maximum_shift: i64,
104    minimum_shift: i64,
105}
106
107impl LinearRecurrenceProblem {
108    /// Constructs a validated recurrence request with its declared window and index range.
109    pub fn new(
110        context: &Context,
111        equation: ExpressionReference,
112        sequence_function: FunctionIdentifier,
113        index_variable: SymbolIdentifier,
114        conditions: Vec<LinearRecurrenceCondition>,
115        index_range: LinearRecurrenceIndexRange,
116    ) -> Result<Self, LinearRecurrenceProblemError> {
117        if !matches!(sequence_function, FunctionIdentifier::User(_)) {
118            return Err(LinearRecurrenceProblemError::SequenceFunctionMustBeUserFunction);
119        }
120        let occurrences = collect_sequence_occurrences(&equation, sequence_function);
121        if occurrences.is_empty() {
122            return Err(LinearRecurrenceProblemError::NoSequenceOccurrence);
123        }
124        let mut shifts = Vec::with_capacity(occurrences.len());
125        for occurrence in &occurrences {
126            match &occurrence.kind {
127                ExpressionKind::Derivative(_, _, _) => {
128                    return Err(LinearRecurrenceProblemError::SequenceOccurrenceIsADerivative);
129                }
130                ExpressionKind::FunctionCall(_, arguments) => {
131                    if arguments.len() != 1 {
132                        return Err(LinearRecurrenceProblemError::SequenceArityMismatch);
133                    }
134                    let Some(shift) = integer_displacement(context, &arguments[0], index_variable)
135                    else {
136                        return Err(LinearRecurrenceProblemError::ShiftIsNotAnIntegerDisplacement);
137                    };
138                    shifts.push(shift);
139                }
140                _ => continue,
141            }
142        }
143        let maximum_shift = *shifts
144            .iter()
145            .max()
146            .ok_or(LinearRecurrenceProblemError::NoSequenceOccurrence)?;
147        let minimum_shift = *shifts
148            .iter()
149            .min()
150            .ok_or(LinearRecurrenceProblemError::NoSequenceOccurrence)?;
151        if maximum_shift == minimum_shift {
152            return Err(LinearRecurrenceProblemError::NoShiftedOccurrence);
153        }
154        let mut seen: Vec<ExpressionReference> = Vec::new();
155        for condition in &conditions {
156            if crate::differential_equations::problem::contains_symbol(
157                &condition.index,
158                index_variable,
159            ) {
160                return Err(LinearRecurrenceProblemError::ConditionIndexContainsIndexVariable);
161            }
162            if seen.contains(&condition.index) {
163                return Err(LinearRecurrenceProblemError::DuplicateCondition);
164            }
165            seen.push(condition.index.clone());
166        }
167        let solution_lower = index_range
168            .solution_index_lower_bound
169            .as_ref()
170            .and_then(exact_integer_value);
171        if let Some(solution_lower) = solution_lower {
172            for condition in &conditions {
173                if exact_integer_value(&condition.index).is_some_and(|index| index < solution_lower)
174                {
175                    return Err(LinearRecurrenceProblemError::ConditionIndexBelowSolutionRange);
176                }
177            }
178            let equation_lower = index_range
179                .equation_index_lower_bound
180                .as_ref()
181                .and_then(exact_integer_value);
182            if let Some(equation_lower) = equation_lower
183                && equation_lower.saturating_add(minimum_shift) < solution_lower
184            {
185                return Err(LinearRecurrenceProblemError::EquationRangeReadsBelowSolutionRange);
186            }
187        }
188        Ok(Self {
189            equation,
190            sequence_function,
191            index_variable,
192            conditions,
193            index_range,
194            maximum_shift,
195            minimum_shift,
196        })
197    }
198
199    /// Returns the governing residual interpreted as equal to zero.
200    pub fn equation(&self) -> &ExpressionReference {
201        &self.equation
202    }
203
204    /// Returns the unknown sequence identifier.
205    pub fn sequence_function(&self) -> FunctionIdentifier {
206        self.sequence_function
207    }
208
209    /// Returns the index variable.
210    pub fn index_variable(&self) -> SymbolIdentifier {
211        self.index_variable
212    }
213
214    /// Returns the declared initial window in insertion order.
215    pub fn conditions(&self) -> &[LinearRecurrenceCondition] {
216        &self.conditions
217    }
218
219    /// Returns the two declared index ranges of the request.
220    pub fn index_range(&self) -> &LinearRecurrenceIndexRange {
221        &self.index_range
222    }
223
224    /// Returns the smallest index at which the closed form is claimed, when one is declared.
225    pub fn solution_index_lower_bound(&self) -> Option<&ExpressionReference> {
226        self.index_range.solution_index_lower_bound.as_ref()
227    }
228
229    /// Returns the smallest index at which the recurrence is asserted, when one is declared.
230    pub fn equation_index_lower_bound(&self) -> Option<&ExpressionReference> {
231        self.index_range.equation_index_lower_bound.as_ref()
232    }
233
234    /// Returns the smallest index at which the recurrence may be asserted without reading the
235    /// closed form below the range it is claimed on.
236    pub fn least_valid_equation_index(&self) -> Option<i64> {
237        let solution_lower = self
238            .index_range
239            .solution_index_lower_bound
240            .as_ref()
241            .and_then(exact_integer_value)?;
242        Some(solution_lower.saturating_sub(self.minimum_shift))
243    }
244
245    /// Returns the highest shift offset appearing in the residual.
246    pub fn maximum_shift(&self) -> i64 {
247        self.maximum_shift
248    }
249
250    /// Returns the lowest shift offset appearing in the residual.
251    pub fn minimum_shift(&self) -> i64 {
252        self.minimum_shift
253    }
254
255    /// Returns the order of the recurrence, which is the span of its shift offsets.
256    pub fn order(&self) -> usize {
257        (self.maximum_shift - self.minimum_shift) as usize
258    }

298/// Verifies a closed form against a validated recurrence request by exact substitution.
299pub fn verify_linear_recurrence_solution(
300    context: &Context,
301    problem: &LinearRecurrenceProblem,
302    candidate: &ExpressionReference,
303) -> LinearRecurrenceVerificationReport {
304    let index_expression = context
305        .interner
306        .intern(ExpressionKind::Symbol(problem.index_variable()));
307    let occurrences = collect_sequence_occurrences(problem.equation(), problem.sequence_function());
308    let mut mapping = SubstitutionMap::new();
309    for occurrence in &occurrences {
310        let ExpressionKind::FunctionCall(_, arguments) = &occurrence.kind else {
311            continue;
312        };
313        let Some(argument) = arguments.first() else {
314            continue;
315        };
316        let mut shift_mapping = SubstitutionMap::new();
317        shift_mapping.insert(index_expression.clone(), argument.clone());
318        mapping.insert(
319            occurrence.clone(),
320            substitute(context, candidate, &shift_mapping),
321        );
322    }
323    let reduced = substitute(context, problem.equation(), &mapping);
324    let normalized = split_numeric_exponent_offsets(context, &reduced);
325    let mut recurrence_residual = decide(context, normalized);
326    if recurrence_residual.verdict == LinearRecurrenceVerificationVerdict::Inconclusive
327        && (constant_coefficient_mode_certificate(context, problem, candidate)
328            || mixed_difference_field_certificate(context, problem, candidate))
329    {
330        recurrence_residual.verdict = LinearRecurrenceVerificationVerdict::Verified;
331    }
332    let candidate_branches: Vec<(Option<ExpressionReference>, ExpressionReference)> =
333        match &candidate.kind {
334            ExpressionKind::Piecewise(branches) => branches
335                .iter()
336                .map(|(guard, value)| (Some(guard.clone()), value.clone()))
337                .collect(),
338            _ => vec![(None, candidate.clone())],
339        };
340    let condition_residuals: Vec<_> = problem
341        .conditions()
342        .iter()
343        .map(|condition| {
344            let mut condition_mapping = SubstitutionMap::new();
345            condition_mapping.insert(index_expression.clone(), condition.index.clone());
346            decide_across_branches(context, &candidate_branches, &condition_mapping, condition)
347        })
348        .collect();
349    let leading_coefficient = leading_shift_coefficient(context, problem);
350    let leading_coefficient_is_nonvanishing = match &leading_coefficient {
351        Some(coefficient) => is_nonzero(coefficient, &context.symbol_table),
352        None => TruthValue::Unknown,
353    };
354    let all = std::iter::once(&recurrence_residual).chain(condition_residuals.iter());
355    let verdict = if all
356        .clone()
357        .any(|residual| residual.verdict == LinearRecurrenceVerificationVerdict::Rejected)
358    {
359        LinearRecurrenceVerificationVerdict::Rejected
360    } else if all
361        .clone()
362        .any(|residual| residual.verdict == LinearRecurrenceVerificationVerdict::Inconclusive)
363    {
364        LinearRecurrenceVerificationVerdict::Inconclusive
365    } else {
366        LinearRecurrenceVerificationVerdict::Verified
367    };
368    LinearRecurrenceVerificationReport {
369        verdict,
370        recurrence_residual,
371        condition_residuals,
372        leading_coefficient,
373        leading_coefficient_is_nonvanishing,
374        window_is_complete: problem.conditions().len() >= problem.order(),
375    }
376}

378fn decide_across_branches(
379    context: &Context,
380    branches: &[(Option<ExpressionReference>, ExpressionReference)],
381    condition_mapping: &SubstitutionMap,
382    condition: &LinearRecurrenceCondition,
383) -> LinearRecurrenceVerificationResidual {
384    // Replay one datum against a candidate that may be stated as a guarded family. Every branch has
385    // to meet the datum: the datum is confirmed only when it holds on the whole family, and a
386    // single branch that provably misses it rejects the candidate. A guard that pins a parameter to
387    // a value, though, says that its branch is the answer only on that stratum, and the datum is
388    // then the datum on that stratum too — a datum written in the parameter takes its own value
389    // there. Replaying such a branch against the unspecialised datum would compare the answer at
390    // one parameter value against the request at another, so the pinning is applied to both sides.
391    let mut residuals: Vec<LinearRecurrenceVerificationResidual> = branches
392        .iter()
393        .map(|(guard, branch)| {
394            let pinning = guard
395                .as_ref()
396                .map(|guard| parameter_pinnings(guard))
397                .unwrap_or_default();
398            let value = substitute(context, branch, condition_mapping);
399            let value = substitute(context, &value, &pinning);
400            let datum = substitute(context, &condition.value, &pinning);
401            let residual = builders::subtract(context, value, datum);
402            decide(context, residual)
403        })
404        .collect();
405    if let Some(rejected) = residuals
406        .iter()
407        .position(|residual| residual.verdict == LinearRecurrenceVerificationVerdict::Rejected)
408    {
409        return residuals.swap_remove(rejected);
410    }
411    if let Some(inconclusive) = residuals
412        .iter()
413        .position(|residual| residual.verdict == LinearRecurrenceVerificationVerdict::Inconclusive)
414    {
415        return residuals.swap_remove(inconclusive);
416    }
417    residuals
418        .into_iter()
419        .next()
420        .unwrap_or(LinearRecurrenceVerificationResidual {
421            residual: builders::zero(context),
422            verdict: LinearRecurrenceVerificationVerdict::Inconclusive,
423        })
424}

426fn parameter_pinnings(guard: &ExpressionReference) -> SubstitutionMap {
427    // The substitution a branch guard forces on its own stratum: every conjunct of the form
428    // `symbol = value`, where the value names no symbol, sends that symbol to that value. Any other
429    // guard contributes nothing, so an unguarded or inequality-guarded branch is replayed exactly
430    // as before.
431    let mut pinnings = SubstitutionMap::new();
432    collect_parameter_pinnings(guard, &mut pinnings);
433    pinnings
434}
435
436fn collect_parameter_pinnings(guard: &ExpressionReference, pinnings: &mut SubstitutionMap) {
437    let ExpressionKind::FunctionCall(called, arguments) = &guard.kind else {
438        return;
439    };
440    match called {
441        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd) => {
442            for argument in arguments {
443                collect_parameter_pinnings(argument, pinnings);
444            }
445        }
446        FunctionIdentifier::Builtin(BuiltinFunction::Equal) if arguments.len() == 2 => {
447            for (left, right) in [(0, 1), (1, 0)] {
448                if matches!(&arguments[left].kind, ExpressionKind::Symbol(_))
449                    && crate::expression::free_variables::free_variables(&arguments[right])
450                        .is_empty()
451                {
452                    pinnings.insert(arguments[left].clone(), arguments[right].clone());
453                }
454            }
455        }
456        _ => {}
457    }
458}

460fn mixed_difference_field_certificate(
461    context: &Context,
462    problem: &LinearRecurrenceProblem,
463    candidate: &ExpressionReference,
464) -> bool {
465    // Exact residual proof in the multiplicative difference field the equation itself generates.
466    // It is applied to the candidate under verification, so it is evidence about that expression
467    // and not agreement with the solver that produced it.
468    let condition_indices: Vec<ExpressionReference> = problem
469        .conditions()
470        .iter()
471        .map(|condition| condition.index.clone())
472        .collect();
473    crate::recurrences::mixed_geometric_residual_is_zero(
474        context,
475        problem.equation(),
476        problem.sequence_function(),
477        problem.index_variable(),
478        &condition_indices,
479        candidate,
480    )
481}

483fn constant_coefficient_mode_certificate(
484    context: &Context,
485    problem: &LinearRecurrenceProblem,
486    candidate: &ExpressionReference,
487) -> bool {
488    // A finite combination of geometric modes satisfies a constant-coefficient
489    // recurrence when each base is a characteristic root. This exact certificate
490    // is needed for algebraic roots whose principal-power spelling cannot be
491    // simplified as an identity in a symbolic integer index.
492    let Some(extracted) = crate::recurrences::shape::extract_recurrence_coefficients(
493        context,
494        problem.equation(),
495        problem.sequence_function(),
496        problem.index_variable(),
497    ) else {
498        return false;
499    };
500    let mut constants = Vec::with_capacity(extracted.coefficients.len());
501    for coefficient in &extracted.coefficients {
502        if crate::polynomial::is_zero(coefficient) {
503            constants.push(crate::exact_arithmetic::rational_zero());
504            continue;
505        }
506        if crate::polynomial::degree(coefficient) != Some(0) {
507            return false;
508        }
509        let Some(number) = crate::polynomial::coefficient_to_number(&coefficient.terms[0].1) else {
510            return false;
511        };
512        constants.push(match number {
513            crate::number::Number::Integer(value) => {
514                crate::exact_arithmetic::rational_from_integer(value)
515            }
516            crate::number::Number::Rational(value) => value,
517        });
518    }
519    let index = context
520        .interner
521        .intern(ExpressionKind::Symbol(problem.index_variable()));
522    let terms: Vec<_> = match &candidate.kind {
523        ExpressionKind::Sum(terms) => terms.iter().collect(),
524        _ => vec![candidate],
525    };
526    !terms.is_empty()
527        && terms.into_iter().all(|term| {
528            let Some(base) = geometric_mode_base(term, &index) else {
529                return false;
530            };
531            let characteristic = builders::sum(
532                context,
533                constants
534                    .iter()
535                    .enumerate()
536                    .map(|(degree, coefficient)| {
537                        builders::product(
538                            context,
539                            vec![
540                                builders::rational_literal(context, coefficient.clone()),
541                                builders::power(
542                                    context,
543                                    base.clone(),
544                                    builders::integer_value(context, degree as i64),
545                                ),
546                            ],
547                        )
548                    })
549                    .collect(),
550            );
551            prove_residual_zero(context, &characteristic) == ZeroEquivalenceVerdict::TrueValue
552        })
553}

555fn geometric_mode_base(
556    term: &ExpressionReference,
557    index: &ExpressionReference,
558) -> Option<ExpressionReference> {
559    let factors: Vec<_> = match &term.kind {
560        ExpressionKind::Product(factors) => factors.iter().collect(),
561        _ => vec![term],
562    };
563    let mut base = None;
564    for factor in factors {
565        match &factor.kind {
566            ExpressionKind::Power(candidate_base, exponent)
567                if candidate_base != index && exponent == index =>
568            {
569                if base.replace(candidate_base.clone()).is_some() {
570                    return None;
571                }
572            }
573            _ if crate::expression::queries::depends_on(
574                factor,
575                match &index.kind {
576                    ExpressionKind::Symbol(identifier) => *identifier,
577                    _ => return None,
578                },
579            ) =>
580            {
581                return None;
582            }
583            _ => {}
584        }
585    }
586    base
587}

589fn leading_shift_coefficient(
590    context: &Context,
591    problem: &LinearRecurrenceProblem,
592) -> Option<ExpressionReference> {
593    // The coefficient of the highest shift, read off without assuming a polynomial-coefficient
594    // shape: the highest-shift call becomes a working symbol and the residual is differentiated
595    // with respect to it. A residual that is genuinely linear in that call loses the symbol,
596    // which is exactly the condition under which the derivative is the coefficient; a residual
597    // that keeps it is nonlinear there and supports no uniqueness argument.
598    let index_expression = context
599        .interner
600        .intern(ExpressionKind::Symbol(problem.index_variable()));
601    let shift = builders::integer_value(context, problem.maximum_shift());
602    let shifted_index = builders::sum(context, vec![index_expression, shift]);
603    let leading_call =
604        builders::function_call(context, problem.sequence_function(), vec![shifted_index]);
605    let working = context.symbol_table.fresh("recurrence_leading_value");
606    let working_expression = context.interner.intern(ExpressionKind::Symbol(working));
607    let mut mapping = SubstitutionMap::new();
608    mapping.insert(leading_call, working_expression);
609    let replaced = substitute(context, problem.equation(), &mapping);
610    let coefficient = differentiate(context, &replaced, working);
611    let simplified = crate::simplify::simplify(context, coefficient);
612    if crate::differential_equations::problem::contains_symbol(&simplified, working) {
613        return None;
614    }
615    Some(simplified)
616}

618fn decide(
619    context: &Context,
620    residual: ExpressionReference,
621) -> LinearRecurrenceVerificationResidual {
622    let verdict = match prove_residual_zero(context, &residual) {
623        ZeroEquivalenceVerdict::TrueValue => LinearRecurrenceVerificationVerdict::Verified,
624        ZeroEquivalenceVerdict::FalseValue => LinearRecurrenceVerificationVerdict::Rejected,
625        ZeroEquivalenceVerdict::Unknown => LinearRecurrenceVerificationVerdict::Inconclusive,
626    };
627    LinearRecurrenceVerificationResidual { residual, verdict }
628}

630fn collect_sequence_occurrences(
631    expression: &ExpressionReference,
632    sequence_function: FunctionIdentifier,
633) -> Vec<ExpressionReference> {
634    let mut occurrences = Vec::new();
635    collect_sequence_occurrences_into(expression, sequence_function, &mut occurrences);

639fn collect_sequence_occurrences_into(
640    expression: &ExpressionReference,
641    sequence_function: FunctionIdentifier,
642    occurrences: &mut Vec<ExpressionReference>,
643) {
644    if matches!(
645        &expression.kind,
646        ExpressionKind::FunctionCall(function, _) | ExpressionKind::Derivative(function, _, _)
647            if *function == sequence_function
648    ) {
649        occurrences.push(expression.clone());
650    }
651    crate::expression::dag::for_each_child(expression, |child| {
652        collect_sequence_occurrences_into(child, sequence_function, occurrences);
653    });
654}

656fn exact_integer_value(expression: &ExpressionReference) -> Option<i64> {
657    match &expression.kind {
658        ExpressionKind::Integer(value) => i64::try_from(value.clone()).ok(),
659        _ => None,
660    }
661}
662
663fn integer_displacement(
664    context: &Context,
665    argument: &ExpressionReference,
666    index_variable: SymbolIdentifier,
667) -> Option<i64> {
668    let index_expression = context
669        .interner
670        .intern(ExpressionKind::Symbol(index_variable));
671    let difference = builders::subtract(context, argument.clone(), index_expression);
672    let simplified = crate::simplify::simplify(context, difference);
673    match &simplified.kind {
674        ExpressionKind::Integer(value) => i64::try_from(value.clone()).ok(),
675        ExpressionKind::Constant(BuiltinConstant::Undefined) => None,
676        _ => None,
677    }
678}
