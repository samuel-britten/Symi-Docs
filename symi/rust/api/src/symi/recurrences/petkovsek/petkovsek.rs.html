1// Petkovsek (1992) `Hyper` algorithm for hypergeometric term
2// solutions of a linear recurrence
3//   Σ_{k=0}^{N} P_k(n) * y(n + k) = 0,    P_0 · P_N ≢ 0,
4// over Q[n]. The algorithm decomposes the consecutive-term ratio
5//   y(n + 1) / y(n) = z * (a(n) / b(n)) * (c(n + 1) / c(n))
6// (the Gosper-Petkovsek normal form) and enumerates the finite set of
7// monic squarefree (a, b) pairs subject to the coprimality condition
8// gcd(a(n), b(n + h)) = 1 for all h ∈ N_{≥0}. For each pair the
9// leading-coefficient (in n) of the transformed recurrence yields a
10// polynomial equation in z whose rational roots are candidate
11// hypergeometric ratios; for each surviving (a, b, z) triple the
12// polynomial-search closure lifts c(n) ∈ Q[n].
13

53/// Performs the advanced native `hyper` operation.
54pub fn hyper(
55    context: &Context,
56    coefficients: &[Polynomial],
57    index_variable: SymbolIdentifier,
58) -> Option<Vec<HypergeometricSolution>> {
59    // Public entry. Returns None when the input fails the basic LRPC
60    // shape check (zero leading or trailing coefficient, fewer than two
61    // total coefficients); returns Some(vec) on success, where the
62    // vector may be empty if Petkovsek's enumeration exhausts every
63    // candidate without finding a hypergeometric solution. The
64    // dispatcher uses an empty vec to fall back to the unsolved
65    // placeholder — no heuristic guess is fabricated.
66    if coefficients.len() < 2 {
67        return None;
68    }
69    let order = coefficients.len() - 1;
70    if polynomial_is_zero(&coefficients[0]) || polynomial_is_zero(&coefficients[order]) {
71        return None;
72    }
73    let trailing = monic_normalised(&coefficients[0]);
74    let leading_shifted =
75        shift_polynomial(&coefficients[order], -(order as i64) + 1, index_variable);
76    let leading = monic_normalised(&leading_shifted);
77    let trailing_divisors =
78        monic_squarefree_divisors(&trailing, index_variable, context.operation());
79    let leading_divisors = monic_squarefree_divisors(&leading, index_variable, context.operation());
80    let pair_count = trailing_divisors
81        .len()
82        .saturating_mul(leading_divisors.len());
83    let execution_route = if context
84        .operation()
85        .may_fork_work_class("recurrence_candidate")
86    {
87        let work_estimate = WorkEstimate::new(
88            (pair_count as u64)
89                .saturating_mul(coefficients.len() as u64)
90                .saturating_mul(16),
91            pair_count,
92            pair_count.saturating_mul(std::mem::size_of::<HypergeometricSolution>()),
93            0,
94            pair_count,
95            "recurrence_candidate",
96        );
97        context
98            .operation()
99            .execution_decision_silent(&work_estimate, pair_count, None)
100            .route
101    } else {
102        ExecutionRoute::Serial
103    };
104    // The divisor pairs in canonical branch order, which is the trailing index major and the
105    // leading index minor: they are the coordinator's own mathematical enumeration, and the peer
106    // group reads them where they are rather than wrapping each one in a task record.
107    let divisor_pairs = (0..trailing_divisors.len())
108        .flat_map(|trailing_index| {
109            (0..leading_divisors.len()).map(move |leading_index| (trailing_index, leading_index))
110        })
111        .collect::<Vec<_>>();
112    let outcomes = execute_peer_group(
113        execution_route,
114        1,
115        &divisor_pairs,
116        |branch_index, _| {
117            SearchTaskIdentifier::new("recurrence", 0, 1, branch_index, "factor_pair")
118        },
119        &context.operation().cancellation_token(),
120        |(trailing_index, leading_index), identifier| {
121            if context.operation().is_cancelled() {
122                return SearchTaskOutcome::<Vec<HypergeometricSolution>, (), String>::Cancelled;
123            }
124            let trailing_divisor = &trailing_divisors[*trailing_index];
125            let leading_divisor = &leading_divisors[*leading_index];
126            if !gosper_petkovsek_coprime(
127                trailing_divisor,
128                leading_divisor,
129                index_variable,
130                context.operation(),
131            ) {
132                return SearchTaskOutcome::ProvedEmpty {
133                    evidence: (),
134                    conditions: SearchConditionLedger::empty(),
135                };
136            }
137            let leading_z_polynomial =
138                build_leading_z_polynomial(coefficients, trailing_divisor, leading_divisor);
139            let mut solutions = Vec::new();
140            for z in rational_roots(&leading_z_polynomial) {
141                let transformed = build_transformed_recurrence(
142                    coefficients,
143                    trailing_divisor,
144                    leading_divisor,
145                    &z,
146                    index_variable,
147                );
148                for c in polynomial_search(&transformed, index_variable) {
149                    let ratio = build_ratio_expression(
150                        context,
151                        trailing_divisor,
152                        leading_divisor,
153                        &c,
154                        &z,
155                        index_variable,
156                    );
157                    let closed_form = build_closed_form(
158                        context,
159                        trailing_divisor,
160                        leading_divisor,
161                        &c,
162                        &z,
163                        index_variable,
164                    );
165                    solutions.push(HypergeometricSolution { ratio, closed_form });
166                }
167            }
168            if solutions.is_empty() {
169                SearchTaskOutcome::ProvedEmpty {
170                    evidence: (),
171                    conditions: SearchConditionLedger::empty(),
172                }
173            } else {
174                SearchTaskOutcome::Candidate {
175                    value: solutions,
176                    evidence: (),
177                    conditions: SearchConditionLedger::empty(),
178                    completeness_contribution: SearchCompletenessContribution::new(
179                        "Petkovsek factor pair",
180                        true,
181                        true,
182                    ),
183                    quality_key: vec![identifier.branch_index as u64],
184                }
185            }
186        },
187    );
188    let mut solutions: Vec<HypergeometricSolution> = Vec::new();
189    for indexed_outcome in outcomes {
190        match indexed_outcome.outcome {
191            SearchTaskOutcome::Candidate { mut value, .. } => solutions.append(&mut value),
192            SearchTaskOutcome::ProvedEmpty { .. } => {}
193            SearchTaskOutcome::Declined { .. }
194            | SearchTaskOutcome::ResourceExhausted
195            | SearchTaskOutcome::Cancelled => return None,
196        }
197    }
198    Some(solutions)
199}

201fn monic_normalised(value: &Polynomial) -> Polynomial {
202    // Scale `value` by the inverse of its leading coefficient so the
203    // result is monic. The polynomial-layer arithmetic preserves
204    // rational coefficients; the squarefree-divisor enumeration that
205    // consumes this routine relies on monic inputs to avoid spurious
206    // unit-factor enumeration.
207    if polynomial_is_zero(value) {
208        return value.clone();
209    }
210    let leading_number = match leading_coefficient(value)
211        .and_then(|coefficient| coefficient_to_number(&coefficient))
212    {
213        Some(value) => value,
214        None => return value.clone(),
215    };
216    if crate::number::is_one(&leading_number) {
217        return value.clone();
218    }
219    let inverse =
220        match crate::number::divide(&number_value::Integer(integer_one()), &leading_number) {
221            Some(value) => value,
222            None => return value.clone(),
223        };
224    let scaled: Vec<(usize, PolynomialCoefficient)> = value
225        .terms
226        .iter()
227        .map(|(exponent, coefficient)| {
228            let coefficient_number =
229                coefficient_to_number(coefficient).unwrap_or(number_value::Integer(integer_zero()));
230            let new_coefficient = crate::number::multiply(&coefficient_number, &inverse);
231            (*exponent, PolynomialCoefficient::Numeric(new_coefficient))
232        })
233        .collect();
234    Polynomial {
235        main_variable: value.main_variable,
236        terms: scaled,
237    }
238}

240fn shift_polynomial(
241    value: &Polynomial,
242    shift: i64,
243    index_variable: SymbolIdentifier,
244) -> Polynomial {
245    // Substitute n → n + shift into the polynomial. Implemented via
246    // binomial expansion on each term: a coefficient * n^e maps to
247    // coefficient * Σ_{i=0..e} C(e, i) * shift^{e-i} * n^i. The shift
248    // can be negative, in which case shift^{e-i} carries the sign.
249    let mut accumulator: Polynomial = zero_polynomial(index_variable);
250    let shift_integer = exact_integer::from(shift);
251    for (exponent, coefficient) in &value.terms {
252        let coefficient_number = match coefficient_to_number(coefficient) {
253            Some(value) => value,
254            None => return value.clone(),
255        };
256        for i in 0..=*exponent {
257            let binomial_value = integer_binomial(*exponent as u64, i as u64);
258            let shift_power = integer_pow(&shift_integer, (*exponent - i) as u64);
259            let mixed_integer = &binomial_value * &shift_power;
260            if mixed_integer == integer_zero() {
261                continue;
262            }
263            let mixed_number = crate::number::multiply(
264                &coefficient_number,
265                &number_value::Integer(mixed_integer.clone()),
266            );
267            let mut dense = vec![number_value::Integer(integer_zero()); i + 1];
268            dense[i] = mixed_number;
269            let term_polynomial = univariate_from_dense(index_variable, dense);
270            accumulator = polynomial_add(&accumulator, &term_polynomial);
271        }
272    }
273    accumulator
274}

276fn integer_pow(base: &exact_integer, exponent: u64) -> exact_integer {
277    // Repeated-squaring power on exact integer. Used by the shift
278    // expansion to compute shift^{e-i}.
279    if exponent == 0 {
280        return integer_one();
281    }
282    let mut accumulator = integer_one();
283    let mut base_power = base.clone();
284    let mut e = exponent;
285    while e > 0 {
286        if e & 1 == 1 {
287            accumulator = &accumulator * &base_power;
288        }
289        e >>= 1;
290        if e > 0 {
291            base_power = &base_power * &base_power;
292        }
293    }
294    accumulator
295}

297fn integer_binomial(top: u64, bottom: u64) -> exact_integer {
298    // Closed-form binomial coefficient. Local helper duplicated from
299    // polynomial_search.rs / summation.rs to keep dependency surface
300    // tight inside the recurrences module.
301    if bottom > top {
302        return integer_zero();
303    }
304    let effective_bottom = bottom.min(top - bottom);
305    let mut result = integer_one();
306    for step in 0..effective_bottom {
307        result = &result * &exact_integer::from(top - step) / &exact_integer::from(step + 1);
308    }
309    result
310}

312fn monic_squarefree_divisors(
313    value: &Polynomial,
314    index_variable: SymbolIdentifier,
315    operation: &OperationContext,
316) -> Vec<Polynomial> {
317    // Enumerate every monic squarefree divisor of `value`. The
318    // Gosper-Petkovsek decomposition admits only squarefree (a, b)
319    // factors (Petkovsek 1992 §2), so the enumeration walks all
320    // 2^k subsets of the irreducible-factor set rather than the full
321    // 2^{Σ m_i} divisor lattice.
322    let irreducibles: Vec<Polynomial> =
323        factor_univariate_over_integers_with_operation(value, operation)
324            .into_iter()
325            .map(|(_, factor)| monic_normalised(&factor))
326            .collect();
327    let mut divisors: Vec<Polynomial> = vec![constant_polynomial(
328        index_variable,
329        number_value::Integer(integer_one()),
330    )];
331    for irreducible in &irreducibles {
332        let monic_irreducible = monic_normalised(irreducible);
333        let mut next_layer: Vec<Polynomial> = Vec::with_capacity(divisors.len() * 2);
334        for divisor in &divisors {
335            next_layer.push(divisor.clone());
336            next_layer.push(polynomial_multiply(divisor, &monic_irreducible));
337        }
338        divisors = next_layer;
339    }
340    divisors
341}

343fn gosper_petkovsek_coprime(
344    a: &Polynomial,
345    b: &Polynomial,
346    index_variable: SymbolIdentifier,
347    operation: &OperationContext,
348) -> bool {
349    // Gosper-Petkovsek coprimality predicate: gcd(a(n), b(n + h)) is
350    // the constant polynomial for every h ∈ N_{≥0}. The check is
351    // finite because the dispersion of (a, b) is bounded by deg(a) +
352    // deg(b); the constant 32 below caps the practical search and is
353    // generous for this module's test corpus.
354    let deg_a = degree(a).unwrap_or(0);
355    let deg_b = degree(b).unwrap_or(0);
356    let bound = (deg_a + deg_b + 4).min(32) as i64;
357    for h in 0..=bound {
358        let shifted_b = shift_polynomial(b, h, index_variable);
359        let gcd_value = polynomial_gcd_with_operation(a, &shifted_b, operation);
360        if degree(&gcd_value).unwrap_or(0) > 0 {
361            return false;
362        }
363    }
364    true
365}

367fn build_leading_z_polynomial(
368    coefficients: &[Polynomial],
369    a: &Polynomial,
370    b: &Polynomial,
371) -> Vec<exact_rational> {
372    // Compute L(z) = Σ_{k achieving max degree} lc_n(P_k) * z^k, where
373    // the maximum is over deg(q_k) = deg(P_k) + k*deg(a) + (N-k)*deg(b).
374    // Because a and b are monic, lc(q_k) = lc(P_k) * z^k, so the
375    // leading-n coefficient of the transformed recurrence is
376    // u_d * L(z) * n^{D + d}. For non-trivial u_d the rational roots
377    // of L(z) are the candidate hypergeometric-ratio constants.
378    let order = coefficients.len() - 1;
379    let deg_a = degree(a).unwrap_or(0) as i64;
380    let deg_b = degree(b).unwrap_or(0) as i64;
381    let mut max_degree: Option<i64> = None;
382    let mut per_k_degrees: Vec<Option<i64>> = Vec::with_capacity(coefficients.len());
383    for (k, polynomial_value) in coefficients.iter().enumerate() {
384        if polynomial_is_zero(polynomial_value) {
385            per_k_degrees.push(None);
386            continue;
387        }
388        let deg_p_k = degree(polynomial_value).unwrap_or(0) as i64;
389        let total = deg_p_k + (k as i64) * deg_a + ((order - k) as i64) * deg_b;
390        per_k_degrees.push(Some(total));
391        if max_degree.is_none_or(|current| total > current) {
392            max_degree = Some(total);
393        }
394    }
395    let max_degree = match max_degree {
396        Some(value) => value,
397        None => return Vec::new(),
398    };
399    let mut z_coefficients: Vec<exact_rational> = vec![rational_zero(); order + 1];
400    for (k, maybe_degree) in per_k_degrees.iter().enumerate() {
401        if maybe_degree.is_none_or(|value| value != max_degree) {
402            continue;
403        }
404        let leading_p_k = match leading_coefficient(&coefficients[k])
405            .and_then(|coefficient| coefficient_to_number(&coefficient))
406        {
407            Some(value) => number_to_rational(value),
408            None => return Vec::new(),
409        };
410        z_coefficients[k] = &z_coefficients[k] + &leading_p_k;
411    }
412    z_coefficients
413}

415fn rational_roots(z_polynomial: &[exact_rational]) -> Vec<exact_rational> {
416    // exact_rational-root enumeration. Given a polynomial Σ a_i · z^i with
417    // rational coefficients, clears denominators to integers and uses
418    // the rational-root theorem: any rational root p/q satisfies
419    // p | a_0 and q | a_d (after the clearance). Enumerates every
420    // ±p/q combination and tests each. Trivial cases (degree 0, all
421    // zero) return an empty list.
422    if z_polynomial.is_empty() {
423        return Vec::new();
424    }
425    let mut leading_index: Option<usize> = None;
426    for (index, coefficient) in z_polynomial.iter().enumerate().rev() {
427        if *coefficient != rational_zero() {
428            leading_index = Some(index);
429            break;
430        }
431    }
432    let leading_index = match leading_index {
433        Some(value) => value,
434        None => return Vec::new(),
435    };
436    if leading_index == 0 {
437        return Vec::new();
438    }
439    let trailing_index = z_polynomial
440        .iter()
441        .position(|coefficient| *coefficient != rational_zero())
442        .unwrap();
443    if trailing_index == leading_index && trailing_index > 0 {
444        return vec![rational_zero()];
445    }
446    let denominator_lcm = z_polynomial
447        .iter()
448        .fold(integer_one(), |accumulator, coefficient| {
449            let denominator_integer = natural_to_integer(rational_parts(coefficient).denominator);
450            integer_lcm(&accumulator, &denominator_integer)
451        });
452    let integer_coefficients: Vec<exact_integer> = z_polynomial
453        .iter()
454        .map(|coefficient| rational_to_integer_scaled(coefficient, &denominator_lcm))
455        .collect();
456    let mut leading_value = integer_coefficients[leading_index].clone();
457    let mut trailing_value = integer_coefficients[trailing_index].clone();
458    if leading_value < integer_zero() {
459        leading_value = -leading_value;
460    }
461    if trailing_value < integer_zero() {
462        trailing_value = -trailing_value;
463    }
464    let trailing_divisors = positive_divisors(&trailing_value);
465    let leading_divisors = positive_divisors(&leading_value);
466    let mut roots: Vec<exact_rational> = Vec::new();
467    if trailing_index > 0 {
468        roots.push(rational_zero());
469    }
470    for numerator_unsigned in &trailing_divisors {
471        for denominator_unsigned in &leading_divisors {
472            for sign in [1i64, -1] {
473                let signed_numerator = if sign > 0 {
474                    exact_integer::from(numerator_unsigned.clone())
475                } else {
476                    -exact_integer::from(numerator_unsigned.clone())
477                };
478                let denominator_integer = exact_integer::from(denominator_unsigned.clone());
479                let candidate =
480                    match crate::number::from_integers(signed_numerator, denominator_integer) {
481                        Some(value) => value,
482                        None => continue,
483                    };
484                let candidate_rational = number_to_rational(candidate);
485                if evaluate_at(z_polynomial, &candidate_rational) == rational_zero() {
486                    if !roots.iter().any(|existing| *existing == candidate_rational) {
487                        roots.push(candidate_rational);
488                    }
489                }
490            }
491        }
492    }
493    roots
494}

496fn evaluate_at(z_polynomial: &[exact_rational], point: &exact_rational) -> exact_rational {
497    // Horner evaluation of a dense rational polynomial at a rational
498    // point. Used by the rational-root test.
499    let mut accumulator = rational_zero();
500    for coefficient in z_polynomial.iter().rev() {
501        accumulator = &accumulator * point + coefficient;
502    }
503    accumulator
504}

506fn rational_to_integer_scaled(
507    value: &exact_rational,
508    denominator_lcm: &exact_integer,
509) -> exact_integer {
510    // Multiply a rational by the supplied common denominator to land
511    // an integer. The caller guarantees the denominator divides
512    // `denominator_lcm` so the result is exact.
513    let parts = rational_parts(value);
514    let numerator_integer = parts.numerator;
515    let denominator_integer = natural_to_integer(parts.denominator);
516    if denominator_integer == integer_zero() {
517        return integer_zero();
518    }
519    let lift = denominator_lcm / &denominator_integer;
520    &numerator_integer * &lift
521}

523fn integer_lcm(left: &exact_integer, right: &exact_integer) -> exact_integer {
524    // Least common multiple via gcd. Used to clear denominators when
525    // converting a rational-coefficient polynomial in z to an
526    // integer-coefficient polynomial for the rational-root scan.
527    if *left == integer_zero() || *right == integer_zero() {
528        return integer_zero();
529    }
530    let gcd = crate::polynomial::gcd::integer_gcd(left, right);
531    let product = left * right;
532    let absolute = if product < integer_zero() {
533        -product
534    } else {
535        product
536    };
537    &absolute / &gcd
538}

540fn positive_divisors(value: &exact_integer) -> Vec<exact_integer> {
541    // Enumerate every positive divisor of |value| in ascending order.
542    // The walk uses trial division up to sqrt(|value|); each divisor
543    // d yields the partner |value|/d. Capped at a generous practical
544    // bound to avoid pathological inputs.
545    if *value == integer_zero() {
546        return vec![integer_one()];
547    }
548    let absolute = if *value < integer_zero() {
549        -value.clone()
550    } else {
551        value.clone()
552    };
553    let absolute_u64 = integer_to_u64(&absolute).unwrap_or(u64::MAX);
554    if absolute_u64 == u64::MAX {
555        return vec![integer_one(), absolute];
556    }
557    let mut small: Vec<exact_integer> = Vec::new();
558    let mut large: Vec<exact_integer> = Vec::new();
559    let mut divisor: u64 = 1;
560    while divisor * divisor <= absolute_u64 {
561        if absolute_u64 % divisor == 0 {
562            small.push(exact_integer::from(divisor));
563            let partner = absolute_u64 / divisor;
564            if partner != divisor {
565                large.push(exact_integer::from(partner));
566            }
567        }
568        divisor += 1;
569    }
570    small.extend(large.into_iter().rev());
571    small
572}

574fn build_transformed_recurrence(
575    coefficients: &[Polynomial],
576    a: &Polynomial,
577    b: &Polynomial,
578    z: &exact_rational,
579    index_variable: SymbolIdentifier,
580) -> Vec<Polynomial> {
581    // Construct the transformed recurrence sum_k q_k(n) * c(n + k) = 0
582    // with
583    //   q_k(n) = P_k(n) * z^k * Π_{j=0}^{k-1} a(n + j) * Π_{j=k}^{N-1} b(n + j).
584    // Each q_k is a polynomial in n over Q; the polynomial-search
585    // closure then walks the null space of the linear system that
586    // falls out of an undetermined-coefficient ansatz for c(n).
587    let order = coefficients.len() - 1;
588    let mut transformed: Vec<Polynomial> = Vec::with_capacity(coefficients.len());
589    for k in 0..=order {
590        let mut accumulator = scale_polynomial_by_rational(&coefficients[k], z, k as u32);
591        if polynomial_is_zero(&accumulator) {
592            transformed.push(accumulator);
593            continue;
594        }
595        for j in 0..k {
596            let shifted_a = shift_polynomial(a, j as i64, index_variable);
597            accumulator = polynomial_multiply(&accumulator, &shifted_a);
598        }
599        for j in k..order {
600            let shifted_b = shift_polynomial(b, j as i64, index_variable);
601            accumulator = polynomial_multiply(&accumulator, &shifted_b);
602        }
603        transformed.push(accumulator);
604    }
605    transformed
606}

608fn scale_polynomial_by_rational(
609    value: &Polynomial,
610    z: &exact_rational,
611    z_power: u32,
612) -> Polynomial {
613    // Scale a polynomial by z^{z_power}. Combines the rational power
614    // computation with the existing scale-polynomial helper so the
615    // transformed-recurrence loop stays terse.
616    if polynomial_is_zero(value) {
617        return value.clone();
618    }
619    let mut z_accumulator = rational_one();
620    for _ in 0..z_power {
621        z_accumulator = &z_accumulator * z;
622    }
623    if z_accumulator == rational_zero() {
624        return zero_polynomial(value.main_variable);
625    }
626    let z_number = match crate::number::from_integers(
627        rational_signed_numerator(&z_accumulator),
628        rational_denominator_integer(&z_accumulator),
629    ) {
630        Some(value) => value,
631        None => return value.clone(),
632    };
633    crate::polynomial::scale_polynomial(value, &z_number)
634}
635
636fn rational_signed_numerator(value: &exact_rational) -> exact_integer {
637    rational_parts(value).numerator
638}
639
640fn rational_denominator_integer(value: &exact_rational) -> exact_integer {
641    natural_to_integer(rational_parts(value).denominator)
642}

644fn build_ratio_expression(
645    context: &Context,
646    a: &Polynomial,
647    b: &Polynomial,
648    c: &Polynomial,
649    z: &exact_rational,
650    index_variable: SymbolIdentifier,
651) -> ExpressionReference {
652    // Build the consecutive-term ratio expression
653    //   r(n) = z * (a(n) / b(n)) * (c(n + 1) / c(n))
654    // routed through the expression-layer builders so the result is
655    // canonicalised. The closed-form representative is assembled
656    // separately by build_closed_form so the two surface forms stay
657    // available downstream (the ratio matters for downstream
658    // consistency checks; the closed form matters for the user-facing
659    // result).
660    let z_expression = rational_to_expression(context, z);
661    let a_expression = polynomial_to_expression(context, a);
662    let b_expression = polynomial_to_expression(context, b);
663    let c_n_expression = polynomial_to_expression(context, c);
664    let c_n_plus_one = polynomial_to_expression(context, &shift_polynomial(c, 1, index_variable));
665    let minus_one_inverse_b = builders::integer_value(context, -1);
666    let inverse_b = builders::power(context, b_expression, minus_one_inverse_b);
667    let minus_one_inverse_c_n = builders::integer_value(context, -1);
668    let inverse_c_n = builders::power(context, c_n_expression, minus_one_inverse_c_n);
669    builders::product(
670        context,
671        vec![
672            z_expression,
673            a_expression,
674            inverse_b,
675            c_n_plus_one,
676            inverse_c_n,
677        ],
678    )
679}

681fn build_closed_form(
682    context: &Context,
683    a: &Polynomial,
684    b: &Polynomial,
685    c: &Polynomial,
686    z: &exact_rational,
687    index_variable: SymbolIdentifier,
688) -> ExpressionReference {
689    // Assemble a closed-form representative for y(n) satisfying the
690    // recurrence with ratio z · (a(n)/b(n)) · (c(n+1)/c(n)).
691    //   y(n) = z^n · (Π_{j=0}^{n-1} a(j) / Π_{j=0}^{n-1} b(j)) · c(n)
692    // We normalise c so c(0) = 1 by dividing through; the resulting
693    // c(n) factor folds into the rising-factorial product. Each linear
694    // factor (n + α_i) of a contributes rising_factorial(α_i, n);
695    // each linear factor of b contributes 1/rising_factorial(α_i, n).
696    // Non-linear or irrational-root factors fall back to an
697    // unevaluated product representation: a multiplicative factor of
698    // the polynomial evaluated at n divided by its constant term,
699    // which is a defensible safe form even when the closed form lacks
700    // a rising-factorial reduction.
701    let index_expression = context
702        .interner
703        .intern(ExpressionKind::Symbol(index_variable));
704    let z_power_term = build_z_power(context, z, &index_expression);
705    let a_product = build_linear_root_product(context, a, &index_expression, false);
706    let b_product = build_linear_root_product(context, b, &index_expression, true);
707    let c_factor = build_c_factor(context, c, &index_expression, index_variable);
708    builders::product(context, vec![z_power_term, a_product, b_product, c_factor])
709}
710
711fn build_z_power(
712    context: &Context,
713    z: &exact_rational,
714    index_expression: &ExpressionReference,
715) -> ExpressionReference {
716    // Build the closed-form z^n term. The trivial z = 1 case collapses
717    // to the integer one so the surrounding product fold stays clean;
718    // every other rational z reifies through the rational builder and
719    // raises to the power of the index symbol.
720    if *z == rational_one() {
721        return builders::one(context);
722    }
723    let z_expression = rational_to_expression(context, z);
724    builders::power(context, z_expression, Arc::clone(index_expression))
725}
726
727fn build_linear_root_product(
728    context: &Context,
729    polynomial_value: &Polynomial,
730    index_expression: &ExpressionReference,
731    is_denominator: bool,
732) -> ExpressionReference {
733    // Decompose `polynomial_value` into its rational-linear factors and
734    // contribute a rising_factorial for each: Π_{j=0}^{n-1} (j + α) =
735    // rising_factorial(α, n). Non-linear or non-rational factors fall
736    // back to evaluating the polynomial at n and dividing by its value
737    // at 0 (the constant-term scaling). The is_denominator flag flips
738    // the result to its reciprocal so the same helper covers both the
739    // a(n) numerator product and the b(n) denominator product.
740    let polynomial_degree = degree(polynomial_value).unwrap_or(0);
741    if polynomial_degree == 0 {
742        let constant_value = match leading_coefficient(polynomial_value)
743            .and_then(|coefficient| coefficient_to_number(&coefficient))
744        {
745            Some(value) => value,
746            None => return builders::one(context),
747        };
748        if crate::number::is_one(&constant_value) {
749            return builders::one(context);
750        }
751        let constant_expression = number_to_expression(context, &constant_value);
752        let constant_power =
753            builders::power(context, constant_expression, Arc::clone(index_expression));
754        return if is_denominator {
755            let minus_one = builders::integer_value(context, -1);
756            builders::power(context, constant_power, minus_one)
757        } else {
758            constant_power
759        };
760    }
761    let factorisation =
762        factor_univariate_over_integers_with_operation(polynomial_value, context.operation());
763    let mut product_terms: Vec<ExpressionReference> = Vec::new();
764    for (multiplicity, factor) in factorisation {
765        let factor_monic = monic_normalised(&factor);
766        let factor_degree = degree(&factor_monic).unwrap_or(0);
767        for _ in 0..multiplicity {
768            if factor_degree == 1 {
769                if let Some(rising_factorial_term) =
770                    build_rising_factorial_for_linear(context, &factor_monic, index_expression)
771                {
772                    let term = if is_denominator {
773                        let minus_one = builders::integer_value(context, -1);
774                        builders::power(context, rising_factorial_term, minus_one)
775                    } else {
776                        rising_factorial_term
777                    };
778                    product_terms.push(term);
779                    continue;
780                }
781            }
782            // Fallback: emit the polynomial evaluated symbolically at
783            // the index, scaled so the value at n = 0 is one. This is
784            // a safe representative for any factor whose explicit
785            // closed-form reduction is not yet implemented.
786            let polynomial_at_n = polynomial_to_expression(context, &factor_monic);
787            let polynomial_at_zero_value = constant_term_of(&factor_monic);
788            let scaling = match polynomial_at_zero_value {
789                Some(value) if !crate::number::is_zero(&value) => {
790                    let value_expression = number_to_expression(context, &value);
791                    let minus_one = builders::integer_value(context, -1);
792                    builders::power(context, value_expression, minus_one)
793                }
794                _ => builders::one(context),
795            };
796            let term = builders::product(context, vec![polynomial_at_n, scaling]);
797            let oriented = if is_denominator {
798                let minus_one = builders::integer_value(context, -1);
799                builders::power(context, term, minus_one)
800            } else {
801                term
802            };
803            product_terms.push(oriented);
804        }
805    }
806    if product_terms.is_empty() {
807        return builders::one(context);
808    }
809    if product_terms.len() == 1 {
810        return product_terms.into_iter().next().unwrap();
811    }
812    builders::product(context, product_terms)
813}
814
815fn build_rising_factorial_for_linear(
816    context: &Context,
817    monic_linear: &Polynomial,
818    index_expression: &ExpressionReference,
819) -> Option<ExpressionReference> {
820    // For a monic linear polynomial f(n) = n + α with α ∈ Q, build
821    //   rising_factorial(α, n) = α(α+1)(α+2)...(α+n-1) = Π_{j=0}^{n-1} (j+α).
822    // Returns None when the linear factor's constant term is not a
823    // rational so the caller can fall back to the safe representation.
824    if degree(monic_linear).unwrap_or(0) != 1 {
825        return None;
826    }
827    let constant_term = constant_term_of(monic_linear)?;
828    let alpha_expression = number_to_expression(context, &constant_term);
829    Some(builders::function_call(
830        context,
831        FunctionIdentifier::Builtin(BuiltinFunction::RisingFactorial),
832        vec![alpha_expression, Arc::clone(index_expression)],
833    ))
834}

836fn build_c_factor(
837    context: &Context,
838    c: &Polynomial,
839    index_expression: &ExpressionReference,
840    index_variable: SymbolIdentifier,
841) -> ExpressionReference {
842    // Emit c(n) / c(0). The polynomial c comes from the polynomial-search closure; in the most
843    // common case c is constant (the leading u_0 term in the ansatz),
844    // in which case the ratio collapses to one. When c is non-constant,
845    // we materialise the polynomial expression and divide by its value
846    // at zero, which is the polynomial's constant term.
847    if polynomial_is_zero(c) {
848        return builders::one(context);
849    }
850    let polynomial_degree = degree(c).unwrap_or(0);
851    if polynomial_degree == 0 {
852        return builders::one(context);
853    }
854    let _ = index_variable;
855    let polynomial_expression = polynomial_to_expression(context, c);
856    let polynomial_at_zero = constant_term_of(c);
857    let denominator = match polynomial_at_zero {
858        Some(value) if !crate::number::is_zero(&value) => number_to_expression(context, &value),
859        _ => return polynomial_expression,
860    };
861    let _ = Arc::clone(index_expression);
862    let minus_one = builders::integer_value(context, -1);
863    let denominator_inverse = builders::power(context, denominator, minus_one);
864    builders::product(context, vec![polynomial_expression, denominator_inverse])
865}
866
867fn constant_term_of(value: &Polynomial) -> Option<number_value> {
868    // Extract the constant term of a polynomial (coefficient of n^0).
869    // Returns None when the constant term is nested (non-numeric)
870    // because the closed-form scaling helper cannot use a non-numeric
871    // value as a divisor.
872    for (exponent, coefficient) in &value.terms {
873        if *exponent == 0 {
874            return coefficient_to_number(coefficient);
875        }
876    }
877    Some(number_value::Integer(integer_zero()))
878}
879
880fn number_to_expression(context: &Context, value: &number_value) -> ExpressionReference {
881    // Reify a polynomial-layer number as an expression leaf via the
882    // existing builders. Mirrors the helper in summation.rs but is
883    // duplicated here so the recurrences module does not pull in the
884    // summation surface.
885    match value {
886        number_value::Integer(integer_value) => builders::integer(context, integer_value.clone()),
887        number_value::Rational(rational_value) => {
888            builders::rational_literal(context, rational_value.clone())
889        }
890    }
891}
892
893fn rational_to_expression(context: &Context, value: &exact_rational) -> ExpressionReference {
894    // Convert a exact rational to an interned expression. Used at
895    // the boundary between the rational-coefficient algorithm and the
896    // expression-layer closed form.
897    if *value == rational_zero() {
898        return builders::zero(context);
899    }
900    if *value == rational_one() {
901        return builders::one(context);
902    }
903    let signed_numerator = rational_signed_numerator(value);
904    let denominator_integer = rational_denominator_integer(value);
905    builders::rational(context, signed_numerator, denominator_integer)
906}
907
908fn number_to_rational(value: number_value) -> exact_rational {
909    // Convert a polynomial-layer number into a exact rational.
910    // Used by the leading-coefficient routine and the rational-root
911    // walker.
912    match value {
913        number_value::Integer(integer_value) => exact_rational::from(integer_value),
914        number_value::Rational(rational_value) => rational_value,
915    }
916}
