1// Linear recurrence equation solver with polynomial
2// coefficients. The public entry point routes through Petkovsek's
3// (1992) `Hyper` algorithm for hypergeometric term solutions; inputs
4// outside the recognised shape fall back to the unsolved
5// recurrence_equation AST placeholder
6// (no heuristic guess leaks into the user-facing result).
7//
8// The module sibling-of-differential_equations layout reflects the
9// type-level distinction between sequences (y: N -> K) and continuous
10// functions (y: R -> K); the algorithm family (monic-divisor
11// enumeration + Abramov polynomial search) likewise has no overlap
12// with the ODE classifier ladder.

51/// Performs the advanced native `solve_linear_recurrence` operation.
52pub fn solve_linear_recurrence(
53    context: &Context,
54    equation: &ExpressionReference,
55    sequence_variable_call: &ExpressionReference,
56    index_variable: SymbolIdentifier,
57    initial_conditions: Option<&[InitialCondition]>,
58) -> ExpressionReference {
59    // Public entry for the linear-recurrence dispatcher. Walks the
60    // shape recogniser to extract the polynomial-coefficient
61    // vector P_0, ..., P_N and the forcing from the LHS - RHS = 0
62    // canonical form; routes homogeneous equations through
63    // petkovsek::hyper, constant-coefficient equations with
64    // polynomial-exponential forcing through the characteristic
65    // route, and coefficients rational in the index and a geometric
66    // atom through the mixed-field rational tier. Initial-condition
67    // fitting replays every supplied datum. Inputs outside every
68    // recognised shape fall back to the unsolved recurrence_equation
69    // AST placeholder.
70    if let Some(closed_form) = attempt_solve(
71        context,
72        equation,
73        sequence_variable_call,
74        index_variable,
75        initial_conditions,
76    ) {
77        return closed_form;
78    }
79    builders::recurrence_equation(
80        context,
81        Arc::clone(equation),
82        Arc::clone(sequence_variable_call),
83        index_variable,
84    )
85}

87fn attempt_solve(
88    context: &Context,
89    equation: &ExpressionReference,
90    sequence_variable_call: &ExpressionReference,
91    index_variable: SymbolIdentifier,
92    initial_conditions: Option<&[InitialCondition]>,
93) -> Option<ExpressionReference> {
94    // Inner dispatcher returning Option so the placeholder fallback is
95    // a single rebuild in the public entry. Walks the shape extraction,
96    // classification, and closure stages in order,
97    // and short-circuits to None on any rejection along the chain.
98    //
99    // The C-finite closure adds one arm. Petkovsek's `hyper` enumerates hypergeometric solutions over the
100    // rational function field, so a homogeneous constant-coefficient recurrence whose
101    // characteristic polynomial has no rational root — Fibonacci is the standard example — leaves
102    // it with an empty basis even though the recurrence is completely solvable in closed form.
103    // When the coefficients are all constants the characteristic-polynomial route below solves it
104    // through the certified all-roots pipeline, with multiplicity factors n^j and exact
105    // multi-constant fitting. Petkovsek keeps priority whenever it returns a *complete* basis, so
106    // the simpler spellings already shipped are unchanged.
107    let function = shape::sequence_function_identifier(sequence_variable_call)?;
108    let extracted =
109        match shape::extract_recurrence_coefficients(context, equation, function, index_variable) {
110            Some(extracted) => extracted,
111            None => {
112                return mixed_geometric_particular_solution(
113                    context,
114                    equation,
115                    function,
116                    index_variable,
117                    initial_conditions,
118                );
119            }
120        };
121    let order = extracted.coefficients.len() - 1;
122    let forcing_is_zero = is_simplified_zero(&crate::simplify::simplify(
123        context,
124        Arc::clone(&extracted.forcing),
125    ));
126    let hypergeometric = forcing_is_zero
127        .then(|| petkovsek::hyper(context, &extracted.coefficients, index_variable))
128        .flatten();
129    if let Some(solutions) = &hypergeometric {
130        if solutions.len() == order && !solutions.is_empty() {
131            let basis: Vec<ExpressionReference> = solutions
132                .iter()
133                .map(|solution| Arc::clone(&solution.closed_form))
134                .collect();
135            return assemble_general_solution(
136                context,
137                &basis,
138                sequence_variable_call,
139                index_variable,
140                initial_conditions,
141            );
142        }
143    }
144    if let Some(closed_form) = constant_coefficient_solution(
145        context,
146        &extracted.coefficients,
147        &extracted.forcing,
148        sequence_variable_call,
149        index_variable,
150        initial_conditions,
151    ) {
152        return Some(closed_form);
153    }
154    let solutions = hypergeometric?;
155    if solutions.is_empty() {
156        return None;
157    }
158    let basis: Vec<ExpressionReference> = solutions
159        .iter()
160        .map(|solution| Arc::clone(&solution.closed_form))
161        .collect();
162    assemble_general_solution(
163        context,
164        &basis,
165        sequence_variable_call,
166        index_variable,
167        initial_conditions,
168    )
169}

171fn constant_coefficient_solution(
172    context: &Context,
173    coefficients: &[crate::polynomial::Polynomial],
174    forcing: &ExpressionReference,
175    sequence_variable_call: &ExpressionReference,
176    index_variable: SymbolIdentifier,
177    initial_conditions: Option<&[InitialCondition]>,
178) -> Option<ExpressionReference> {
179    // Architectural decision 6: complete closure for homogeneous constant-coefficient
180    // recurrences. Every coefficient must be a plain rational constant; the characteristic
181    // polynomial Σ_j c_j z^j is then factored over Q into irreducible factors with exact Yun
182    // multiplicities, and each irreducible factor is solved through the shared radical solver so
183    // the roots arrive in the radical vocabulary rather than as opaque algebraic atoms. Basis
184    // element (j, root) is n^j * root^n, which satisfies the recurrence by the standard theorem
185    // once `root` is a root of multiplicity greater than j — so the basis is constructed, not
186    // searched. The construction declines unless the roots and multiplicities account for the whole
187    // order, because an incomplete basis is not a general solution.
188    let constants = constant_rational_coefficients(coefficients)?;
189    let order = constants.len() - 1;
190    if order == 0 {
191        return None;
192    }
193    let root_variable = context.symbol_table.fresh("characteristic_root");
194    let mut characteristic = crate::polynomial::zero_polynomial(root_variable);
195    for (degree, constant) in constants.iter().enumerate() {
196        if crate::exact_arithmetic::rational_is_zero(constant) {
197            continue;
198        }
199        let term = crate::polynomial::Polynomial {
200            main_variable: root_variable,
201            terms: vec![(
202                degree,
203                crate::polynomial::PolynomialCoefficient::Numeric(crate::number::Number::Rational(
204                    constant.clone(),
205                )),
206            )],
207        };
208        characteristic = crate::polynomial::arithmetic::add(&characteristic, &term);
209    }
210    let factors = crate::polynomial::factorization::factor_univariate_over_integers_with_operation(
211        &characteristic,
212        context.operation(),
213    );
214    let mut roots: Vec<(ExpressionReference, usize)> = Vec::new();
215    let mut total = 0usize;
216    for (multiplicity, factor) in &factors {
217        let factor_degree = crate::polynomial::degree(factor).unwrap_or(0);
218        if factor_degree == 0 {
219            continue;
220        }
221        if factor_degree > RADICAL_FITTING_FACTOR_DEGREE_CEILING {
222            // Exact multi-constant fitting happens by symbolic elimination over the root
223            // expressions. Square roots from a quadratic factor cancel reliably there; the nested
224            // Cardano and Ferrari radicals of a cubic or quartic factor do not, and eliminating
225            // over them costs more than any answer is worth. Doing that properly needs algebraic
226            // *number field* arithmetic rather than expression rewriting, which is
227            // deferred, so the route declines here and the caller keeps the unevaluated form
228            // instead of returning something unfitted or spending unbounded time.
229            return None;
230        }
231        let factor_expression =
232            crate::polynomial::conversion::polynomial_to_expression(context, factor);
233        let radical_roots = crate::solve::solve(context, &factor_expression, root_variable);
234        let factor_roots = if radical_roots.len() == factor_degree {
235            radical_roots
236                .into_iter()
237                .map(|root| (root, 1usize))
238                .collect()
239        } else {
240            crate::solve::roots_with_multiplicities(context, &factor_expression, root_variable)
241        };
242        if factor_roots
243            .iter()
244            .map(|(_, multiplicity)| *multiplicity)
245            .sum::<usize>()
246            != factor_degree
247        {
248            return None;
249        }
250        for (root, root_multiplicity) in factor_roots {
251            if crate::expression::queries::depends_on(&root, root_variable) {
252                return None;
253            }
254            let overall_multiplicity = multiplicity.checked_mul(root_multiplicity)?;
255            roots.push((root, overall_multiplicity));
256            total += overall_multiplicity;
257        }
258    }
259    if total != order {
260        return None;
261    }
262    let index_expression = context
263        .interner
264        .intern(crate::expression::ExpressionKind::Symbol(index_variable));
265    let mut basis: Vec<ExpressionReference> = Vec::with_capacity(order);
266    for (root, multiplicity) in &roots {
267        let exponential = builders::power(context, Arc::clone(root), Arc::clone(&index_expression));
268        for factor_degree in 0..*multiplicity {
269            let element = if factor_degree == 0 {
270                Arc::clone(&exponential)
271            } else {
272                let exponent = builders::integer_value(context, factor_degree as i64);
273                let polynomial_factor =
274                    builders::power(context, Arc::clone(&index_expression), exponent);
275                builders::product(context, vec![polynomial_factor, Arc::clone(&exponential)])
276            };
277            basis.push(element);
278        }
279    }
280    let particular_solution =
281        polynomial_particular_solution(context, &constants, forcing, index_variable)?;
282    assemble_general_solution_with_particular(
283        context,
284        &basis,
285        particular_solution,
286        sequence_variable_call,
287        index_variable,
288        initial_conditions,
289    )
290}

292fn constant_rational_coefficients(
293    coefficients: &[crate::polynomial::Polynomial],
294) -> Option<Vec<crate::exact_arithmetic::Rational>> {
295    // Project the extracted coefficient polynomials onto plain rationals, declining as soon as one
296    // of them actually depends on the index variable. That decline is what keeps the
297    // characteristic-polynomial route out of the polynomial-coefficient territory Petkovsek owns.
298    let mut constants = Vec::with_capacity(coefficients.len());
299    for coefficient in coefficients {
300        if crate::polynomial::is_zero(coefficient) {
301            constants.push(crate::exact_arithmetic::rational_zero());
302            continue;
303        }
304        if crate::polynomial::degree(coefficient) != Some(0) {
305            return None;
306        }
307        let value = crate::polynomial::coefficient_to_number(&coefficient.terms[0].1)?;
308        constants.push(match value {
309            crate::number::Number::Integer(integer_value) => {
310                crate::exact_arithmetic::rational_from_integer(integer_value)
311            }
312            crate::number::Number::Rational(rational_value) => rational_value,
313        });
314    }
315    Some(constants)
316}

318fn polynomial_particular_solution(
319    context: &Context,
320    coefficients: &[crate::exact_arithmetic::Rational],
321    forcing: &ExpressionReference,
322    index_variable: SymbolIdentifier,
323) -> Option<ExpressionReference> {
324    // A constant-coefficient shift operator maps polynomials to polynomials.
325    // If one is a characteristic root of multiplicity m, its polynomial
326    // particular solution starts at n^m; this makes the resulting coefficient
327    // system square and excludes homogeneous terms deterministically.
328    let forcing = crate::simplify::simplify(context, Arc::clone(forcing));
329    let forcing_polynomial =
330        crate::polynomial::conversion::expression_to_univariate(&forcing, index_variable)?;
331    if crate::polynomial::is_zero(&forcing_polynomial) {
332        return Some(builders::integer_value(context, 0));
333    }
334    let forcing_degree = crate::polynomial::degree(&forcing_polynomial)?;
335    let zero_multiplicity = (0..coefficients.len()).find(|degree| {
336        !crate::polynomial::is_zero(&constant_operator_on_power(
337            context,
338            coefficients,
339            index_variable,
340            *degree,
341        ))
342    })?;
343    let mut matrix = Vec::with_capacity(forcing_degree + 1);
344    for output_degree in 0..=forcing_degree {
345        let mut row = Vec::with_capacity(forcing_degree + 1);
346        for input_offset in 0..=forcing_degree {
347            let image = constant_operator_on_power(
348                context,
349                coefficients,
350                index_variable,
351                zero_multiplicity + input_offset,
352            );
353            row.push(polynomial_coefficient_expression(
354                context,
355                &image,
356                output_degree,
357            ));
358        }
359        matrix.push(row);
360    }
361    let right_hand_side = (0..=forcing_degree)
362        .map(|degree| {
363            builders::product(
364                context,
365                vec![
366                    builders::integer_value(context, -1),
367                    polynomial_coefficient_expression(context, &forcing_polynomial, degree),
368                ],
369            )
370        })
371        .collect();
372    let values = solve_symbolic_linear_system(context, matrix, right_hand_side)?;
373    let index = context
374        .interner
375        .intern(crate::expression::ExpressionKind::Symbol(index_variable));
376    let terms = values
377        .into_iter()
378        .enumerate()
379        .map(|(offset, value)| {
380            let degree = zero_multiplicity + offset;
381            let power = if degree == 0 {
382                builders::integer_value(context, 1)
383            } else {
384                builders::power(
385                    context,
386                    index.clone(),
387                    builders::integer_value(context, degree as i64),
388                )
389            };
390            builders::product(context, vec![value, power])
391        })
392        .collect();
393    Some(crate::simplify::simplify(
394        context,
395        builders::sum(context, terms),
396    ))
397}

399fn constant_operator_on_power(
400    context: &Context,
401    coefficients: &[crate::exact_arithmetic::Rational],
402    index_variable: SymbolIdentifier,
403    degree: usize,
404) -> crate::polynomial::Polynomial {
405    let _ = context;
406    let mut accumulated: std::collections::BTreeMap<usize, crate::exact_arithmetic::Rational> =
407        std::collections::BTreeMap::new();
408    for (shift, coefficient) in coefficients.iter().enumerate() {
409        for (output_degree, binomial_coefficient) in
410            crate::summation::coefficient_field::binomial_shift_contributions(degree, shift as i64)
411        {
412            let contribution = coefficient * &binomial_coefficient;
413            let current = accumulated
414                .entry(output_degree)
415                .or_insert_with(crate::exact_arithmetic::rational_zero);
416            *current = &*current + &contribution;
417        }
418    }
419    crate::polynomial::Polynomial {
420        main_variable: index_variable,
421        terms: accumulated
422            .into_iter()
423            .filter(|(_, coefficient)| !crate::exact_arithmetic::rational_is_zero(coefficient))
424            .map(|(exponent, coefficient)| {
425                (
426                    exponent,
427                    crate::polynomial::PolynomialCoefficient::Numeric(
428                        crate::number::Number::Rational(coefficient),
429                    ),
430                )
431            })
432            .collect(),
433    }
434}

436fn polynomial_coefficient_expression(
437    context: &Context,
438    polynomial: &crate::polynomial::Polynomial,
439    degree: usize,
440) -> ExpressionReference {
441    polynomial
442        .terms
443        .iter()
444        .find(|(exponent, _)| *exponent == degree)
445        .map(|(_, coefficient)| {

451pub(crate) fn mixed_geometric_particular_solution(
452    context: &Context,
453    equation: &ExpressionReference,
454    function: crate::function::FunctionIdentifier,
455    index_variable: SymbolIdentifier,
456    initial_conditions: Option<&[InitialCondition]>,
457) -> Option<ExpressionReference> {
458    let conditions = initial_conditions?;
459    let raw = shape::extract_linear_recurrence(context, equation, function, index_variable)?;
460    let base = mixed_geometric_base(equation, index_variable)?;
461    let crate::expression::ExpressionKind::Symbol(parameter) = &base.kind else {
462        return None;
463    };
464    let geometric_identifier = context.symbol_table.fresh("geometric_shift_orbit");
465    let field = CoefficientField::with_operation(
466        vec![*parameter, geometric_identifier],
467        context.operation().clone(),
468    );
469    let geometric_position = field.position_of(geometric_identifier)?;
470    let coefficients: Vec<_> = raw
471        .coefficients
472        .iter()
473        .map(|coefficient| {
474            difference_rational_from_expression(
475                coefficient,
476                &field,
477                index_variable,
478                *parameter,
479                geometric_identifier,
480            )
481        })
482        .collect::<Option<Vec<_>>>()?;
483    let order = coefficients.len().checked_sub(1)?;
484    if order == 0 || conditions.len() != order {
485        return None;
486    }
487    let forcing = difference_rational_from_expression(
488        &raw.forcing,
489        &field,
490        index_variable,
491        *parameter,
492        geometric_identifier,
493    )?;
494    let minimum_offset = minimum_geometric_exponent_offset(equation, *parameter, index_variable)?;
495    let scaling_power = minimum_offset.checked_neg()?;
496    let parameter_position = field.position_of(*parameter)?;
497    let scaling = field.power(&field.parameter(parameter_position), scaling_power)?;
498    let coefficients: Vec<_> = coefficients
499        .iter()
500        .map(|coefficient| scale_rational_function(&field, coefficient, &scaling))
501        .collect();
502    let forcing = scale_rational_function(&field, &forcing, &scaling);
503    let operator = cleared_difference_operator(&field, &coefficients)?;
504    let denominator =
505        derived_universal_denominator(&field, &operator, parameter_position, geometric_position)?;
506    let bounds = derived_numerator_degree_bounds(
507        &field,
508        &operator,
509        &denominator,
510        &forcing,
511        parameter_position,
512        geometric_position,
513    )?;
514    let basis = mixed_rational_basis(&field, &denominator, &bounds, geometric_position);
515    let residuals: Vec<_> = basis
516        .iter()
517        .map(|candidate| {
518            difference_operator_apply(&field, &coefficients, candidate, geometric_position)
519        })
520        .collect::<Option<Vec<_>>>()?;
521    let parameter_field =
522        CoefficientField::with_operation(vec![*parameter], context.operation().clone());
523    let (rows, right_hand_side) = rational_operator_linear_system(
524        &field,
525        &parameter_field,
526        &residuals,
527        &forcing,
528        geometric_position,
529    )?;
530    let particular =
531        fraction_free_particular_solution(&parameter_field, &rows, &right_hand_side, basis.len())?;
532    let homogeneous = parameter_field.null_space(&rows, basis.len());
533    // The ansatz spans every numerator the derived bounds admit over the derived universal
534    // denominator, so this null space is the complete space of rational solutions rather than
535    // whatever a fixed span happened to reach. A space smaller than the operator's order means the
536    // remaining solutions are not rational, and this tier does not claim a general solution it
537    // cannot exhibit.
538    if homogeneous.len() != order {
539        return None;
540    }
541    let particular_solution = combine_rational_basis(
542        &field,
543        &basis,
544        &lift_parameter_values(context, &field, &parameter_field, &particular)?,
545    );
546    let homogeneous_solutions: Vec<_> = homogeneous
547        .iter()
548        .map(|coefficients| {
549            Some(combine_rational_basis(
550                &field,
551                &basis,
552                &lift_parameter_values(context, &field, &parameter_field, coefficients)?,
553            ))
554        })
555        .collect::<Option<Vec<_>>>()?;
556    let (condition_rows, condition_values) = initial_value_linear_system(
557        context,
558        &field,
559        &parameter_field,
560        &homogeneous_solutions,
561        &particular_solution,
562        conditions,
563        index_variable,
564        *parameter,
565        geometric_identifier,
566    )?;
567    let values = parameter_field.solve_linear_system(&condition_rows, &condition_values, order)?;
568    let mut result = particular_solution;
569    for (solution, value) in homogeneous_solutions.iter().zip(values.iter()) {
570        let lifted = field.from_expression(&parameter_field.to_expression(context, value))?;
571        result = rational_function_add(
572            &field,
573            &result,
574            &scale_rational_function(&field, solution, &lifted),
575        );
576    }
577    let residual = difference_operator_apply(&field, &coefficients, &result, geometric_position)?;
578    if !rational_function_add(&field, &residual, &forcing)
579        .numerator
580        .is_zero()
581    {
582        return None;
583    }
584    let result = canonical_difference_rational_function(&field, &result)?;
585    let reduced = rational_function_reduce(&field, &result);
586    let expression = rational_function_to_expression(&field, context, &reduced, index_variable);
587    let denominator = defining_denominator(&field, context, &reduced, index_variable);
588    let geometric = builders::power(
589        context,
590        base.clone(),
591        context
592            .interner
593            .intern(crate::expression::ExpressionKind::Symbol(index_variable)),
594    );
595    let geometric_symbol = context
596        .interner
597        .intern(crate::expression::ExpressionKind::Symbol(
598            geometric_identifier,
599        ));
600    let mut mapping = crate::expression::substitute::SubstitutionMap::new();
601    mapping.insert(geometric_symbol, geometric);
602    let generic_solution =
603        crate::expression::substitute::substitute(context, &expression, &mapping);
604    let generic_denominator =
605        crate::expression::substitute::substitute(context, &denominator, &mapping);
606    // The family is claimed exactly where it denotes something. The derivation scales by a negative
607    // power of the geometric base, so the base has to be invertible; and the fitted rational
608    // function has to have a nonvanishing denominator at the index it is read at, which is the
609    // condition that fails on the strata where a power of the base meets one. Both are carried as
610    // explicit conditions rather than left for the reader to infer.
611    let zero = builders::zero(context);
612    let mut branches = vec![(
613        builders::function_call(
614            context,
615            crate::function::FunctionIdentifier::Builtin(
616                crate::function::BuiltinFunction::LogicalAnd,
617            ),
618            vec![
619                builders::not_equal(context, base.clone(), Arc::clone(&zero)),
620                builders::not_equal(context, generic_denominator, Arc::clone(&zero)),
621            ],
622        ),
623        generic_solution,
624    )];
625    // The degenerate base is a different request rather than a value of this one: the geometric
626    // powers collapse there and the equation has to be specialized and solved in its own right.
627    // When that specialization solves, its answer is carried as its own branch; when it does not,
628    // the stratum is left unclaimed instead of being guessed at.
629    if let Some(degenerate) = degenerate_base_solution(
630        context,
631        equation,
632        function,
633        index_variable,
634        &base,
635        conditions,
636    ) {
637        branches.push((
638            builders::equal(context, base, Arc::clone(&zero)),
639            degenerate,
640        ));
641    }
642    Some(builders::piecewise(context, branches))
643}

645fn defining_denominator(
646    field: &CoefficientField,
647    context: &Context,
648    value: &IndexRationalFunction,
649    index_variable: SymbolIdentifier,
650) -> ExpressionReference {
651    // A polynomial whose nonvanishing is exactly what the fitted rational function needs in order
652    // to denote a value: the denominator in the index, together with the ground-field denominator
653    // of every coefficient the index polynomials carry. Those ground denominators are where the
654    // fitting divided by expressions in the parameter, so this is the record of which parameter
655    // values the derivation quietly assumed away.
656    let mut factors = vec![rational_function_to_expression(
657        field,
658        context,
659        &IndexRationalFunction {
660            numerator: value.denominator.clone(),
661            denominator: field.polynomial_one(),
662        },
663        index_variable,
664    )];
665    let mut seen: Vec<ParameterPolynomial> = Vec::new();
666    for polynomial in [&value.numerator, &value.denominator] {
667        for coefficient in polynomial.terms.values() {
668            let denominator = field.denominator_of(coefficient);
669            if denominator.as_rational().is_some() || seen.contains(&denominator) {
670                continue;
671            }
672            seen.push(denominator.clone());
673            factors.push(field.parameter_to_expression(context, &denominator));
674        }
675    }
676    builders::product(context, factors)
677}

679fn degenerate_base_solution(
680    context: &Context,
681    equation: &ExpressionReference,
682    function: crate::function::FunctionIdentifier,
683    index_variable: SymbolIdentifier,
684    base: &ExpressionReference,
685    conditions: &[InitialCondition],
686) -> Option<ExpressionReference> {
687    // Solve the request on the stratum where the multiplicative base vanishes. There `c^(m n + o)`
688    // is zero at every index the equation is actually used at, so the coefficients lose their
689    // multiplicative structure entirely and the specialized equation is an ordinary recurrence the
690    // existing ladder solves. The answer is obtained from that equation and its own specialized
691    // data; nothing is read off the generic family, whose derivation divides by the base.
692    let specialized = degenerate_base_equation(
693        context,
694        equation,
695        function,
696        index_variable,
697        base,
698        conditions.iter().map(|(index, _)| index),
699    )?;
700    let mut mapping = crate::expression::substitute::SubstitutionMap::new();
701    mapping.insert(Arc::clone(base), builders::zero(context));
702    let specialized_conditions: Vec<InitialCondition> = conditions
703        .iter()
704        .map(|(index, value)| {
705            (
706                Arc::clone(index),
707                crate::simplify::simplify(
708                    context,
709                    crate::expression::substitute::substitute(context, value, &mapping),
710                ),
711            )
712        })
713        .collect();
714    let index_expression = context
715        .interner
716        .intern(crate::expression::ExpressionKind::Symbol(index_variable));
717    let sequence_call = builders::function_call(context, function, vec![index_expression]);
718    attempt_solve(
719        context,
720        &specialized,
721        &sequence_call,
722        index_variable,
723        Some(&specialized_conditions),
724    )
725}

727fn degenerate_base_equation<'condition>(
728    context: &Context,
729    equation: &ExpressionReference,
730    function: crate::function::FunctionIdentifier,
731    index_variable: SymbolIdentifier,
732    base: &ExpressionReference,
733    condition_indices: impl Iterator<Item = &'condition ExpressionReference>,
734) -> Option<ExpressionReference> {
735    // Specialize the equation at a vanishing multiplicative base. Every power of the base that
736    // carries the index is replaced by zero, but only once the exponent is shown to be positive
737    // over the whole range the equation is used on: the equation is an identity from the index at
738    // which its lowest sequence reference reaches the earliest supplied datum, so the exponent
739    // `m n + o` is bounded below there. An exponent that cannot be shown positive would make the
740    // replacement a guess about `0^0` or a negative power, and the specialization declines instead.
741    let earliest_index = condition_indices
742        .map(|index| match &index.kind {
743            crate::expression::ExpressionKind::Integer(value) => {
744                crate::exact_arithmetic::integer_to_i64(value)
745            }
746            _ => None,
747        })
748        .min()??;
749    let lowest_shift = minimum_sequence_shift(equation, function, index_variable)?;
750    let lowest_index = earliest_index.checked_sub(lowest_shift)?;
751    let mut powers = Vec::new();
752    collect_geometric_powers(equation, base, index_variable, &mut powers);
753    let mut mapping = crate::expression::substitute::SubstitutionMap::new();
754    let zero = builders::zero(context);
755    for (power, multiplier, offset) in powers {
756        let lowest_exponent = multiplier.checked_mul(lowest_index)?.checked_add(offset)?;
757        if lowest_exponent < 1 {
758            return None;
759        }
760        mapping.insert(power, Arc::clone(&zero));
761    }
762    mapping.insert(Arc::clone(base), Arc::clone(&zero));
763    Some(crate::simplify::simplify(
764        context,
765        crate::expression::substitute::substitute(context, equation, &mapping),
766    ))
767}

769fn collect_geometric_powers(
770    expression: &ExpressionReference,
771    base: &ExpressionReference,
772    index_variable: SymbolIdentifier,
773    powers: &mut Vec<(ExpressionReference, i64, i64)>,
774) {
775    if let crate::expression::ExpressionKind::Power(power_base, exponent) = &expression.kind
776        && power_base == base
777        && let Some((multiplier, offset)) = geometric_exponent_shape(exponent, index_variable)
778        && multiplier > 0
779    {
780        powers.push((Arc::clone(expression), multiplier, offset));
781    }
782    crate::expression::dag::for_each_child(expression, |child| {
783        collect_geometric_powers(child, base, index_variable, powers);
784    });
785}

818pub(crate) fn mixed_geometric_residual_is_zero(
819    context: &Context,
820    equation: &ExpressionReference,
821    function: crate::function::FunctionIdentifier,
822    index_variable: SymbolIdentifier,
823    condition_indices: &[ExpressionReference],
824    candidate: &ExpressionReference,
825) -> bool {
826    // Exact residual certificate for a candidate closed form over the multiplicative difference
827    // field `K(n, z)` with `sigma(n) = n + 1` and `sigma(z) = c * z`. The candidate is mapped into
828    // the field on its own terms and the equation's own difference operator is applied to it; a
829    // vanishing numerator is then a proof that the recurrence identity holds identically, which the
830    // general expression decider cannot reach because a principal-branch power with a symbolic
831    // integer index is not a rational expression it can normalise. The certificate reads only the
832    // equation and the candidate, so it proves the candidate rather than agreeing with whatever the
833    // solver happened to build.
834    //
835    // A branch that pins the multiplicative base to zero is a statement about the degenerate
836    // stratum, where the field this certificate is built in does not exist. Such a branch is
837    // proved against the equation specialized at that base instead, so every branch is checked on
838    // the equation it actually claims something about.
839    let branches: Vec<(Option<ExpressionReference>, ExpressionReference)> = match &candidate.kind {
840        crate::expression::ExpressionKind::Piecewise(branches) => branches
841            .iter()
842            .map(|(condition, value)| (Some(Arc::clone(condition)), Arc::clone(value)))
843            .collect(),
844        _ => vec![(None, Arc::clone(candidate))],
845    };
846    if branches.is_empty() {
847        return false;
848    }
849    let base = mixed_geometric_base(equation, index_variable);
850    branches.iter().all(|(condition, value)| {
851        let degenerate = match (condition, &base) {
852            (Some(condition), Some(base)) => condition_pins_base_to_zero(condition, base),
853            _ => false,
854        };
855        if degenerate {
856            let Some(base) = &base else {
857                return false;
858            };
859            degenerate_branch_residual_is_zero(
860                context,
861                equation,
862                function,
863                index_variable,
864                base,
865                condition_indices,
866                value,
867            )
868            .unwrap_or(false)
869        } else {
870            mixed_geometric_branch_residual_is_zero(
871                context,
872                equation,
873                function,
874                index_variable,
875                value,
876            )
877            .unwrap_or(false)
878        }
879    })
880}

882fn condition_pins_base_to_zero(
883    condition: &ExpressionReference,
884    base: &ExpressionReference,
885) -> bool {
886    // Whether a branch condition is exactly the equality that sends the multiplicative base to
887    // zero. Anything else is read as a generic-stratum branch, which is the conservative reading:
888    // it demands the stronger difference-field certificate.
889    let crate::expression::ExpressionKind::FunctionCall(called, arguments) = &condition.kind else {
890        return false;
891    };
892    if *called
893        != crate::function::FunctionIdentifier::Builtin(crate::function::BuiltinFunction::Equal)
894        || arguments.len() != 2
895    {
896        return false;
897    }
898    let is_zero = |value: &ExpressionReference| {
899        matches!(&value.kind, crate::expression::ExpressionKind::Integer(integer)
900            if crate::exact_arithmetic::integer_is_zero(integer))
901    };
902    (arguments[0] == *base && is_zero(&arguments[1]))
903        || (arguments[1] == *base && is_zero(&arguments[0]))
904}

906fn degenerate_branch_residual_is_zero(
907    context: &Context,
908    equation: &ExpressionReference,
909    function: crate::function::FunctionIdentifier,
910    index_variable: SymbolIdentifier,
911    base: &ExpressionReference,
912    condition_indices: &[ExpressionReference],
913    candidate: &ExpressionReference,
914) -> Option<bool> {
915    // Replay the candidate against the equation specialized at a vanishing base: substitute the
916    // candidate, shifted by each of the operator's own shifts, into the extracted coefficients and
917    // ask whether the residual simplifies to zero. This reads the equation and the candidate only.
918    let specialized = degenerate_base_equation(
919        context,
920        equation,
921        function,
922        index_variable,
923        base,
924        condition_indices.iter(),
925    )?;
926    let raw = shape::extract_linear_recurrence(context, &specialized, function, index_variable)?;
927    let index_expression = context
928        .interner
929        .intern(crate::expression::ExpressionKind::Symbol(index_variable));
930    let mut summands = vec![Arc::clone(&raw.forcing)];
931    for (shift, coefficient) in raw.coefficients.iter().enumerate() {
932        let displacement = builders::integer_value(context, i64::try_from(shift).ok()?);
933        let shifted_index =
934            builders::sum(context, vec![Arc::clone(&index_expression), displacement]);
935        let mut mapping = crate::expression::substitute::SubstitutionMap::new();
936        mapping.insert(Arc::clone(&index_expression), shifted_index);
937        let shifted_candidate =
938            crate::expression::substitute::substitute(context, candidate, &mapping);
939        summands.push(builders::product(
940            context,
941            vec![Arc::clone(coefficient), shifted_candidate],
942        ));
943    }
944    let residual = crate::simplify::simplify(context, builders::sum(context, summands));
945    Some(is_simplified_zero(&residual))
946}

948fn mixed_geometric_branch_residual_is_zero(
949    context: &Context,
950    equation: &ExpressionReference,
951    function: crate::function::FunctionIdentifier,
952    index_variable: SymbolIdentifier,
953    candidate: &ExpressionReference,
954) -> Option<bool> {
955    // Build the difference field from the equation, carry the coefficients, the forcing and the
956    // candidate into it, and apply the operator. Every step declines rather than approximates when
957    // a piece falls outside the field.
958    let raw = shape::extract_linear_recurrence(context, equation, function, index_variable)?;
959    let base = mixed_geometric_base(equation, index_variable)?;
960    let crate::expression::ExpressionKind::Symbol(parameter) = &base.kind else {
961        return None;
962    };
963    let geometric_identifier = context.symbol_table.fresh("geometric_shift_orbit");
964    let field = CoefficientField::with_operation(
965        vec![*parameter, geometric_identifier],
966        context.operation().clone(),
967    );
968    let geometric_position = field.position_of(geometric_identifier)?;
969    let coefficients: Vec<_> = raw
970        .coefficients
971        .iter()
972        .map(|coefficient| {
973            difference_rational_from_expression(
974                coefficient,
975                &field,
976                index_variable,
977                *parameter,
978                geometric_identifier,
979            )
980        })
981        .collect::<Option<Vec<_>>>()?;
982    let forcing = difference_rational_from_expression(
983        &raw.forcing,
984        &field,
985        index_variable,
986        *parameter,
987        geometric_identifier,
988    )?;
989    let value = difference_rational_from_expression(
990        candidate,
991        &field,
992        index_variable,
993        *parameter,
994        geometric_identifier,
995    )?;
996    let residual = difference_operator_apply(&field, &coefficients, &value, geometric_position)?;
997    let total = rational_function_add(&field, &residual, &forcing);
998    Some(total.numerator.is_zero())
999}

1056fn difference_rational_from_expression(
1057    expression: &ExpressionReference,
1058    field: &CoefficientField,
1059    index_variable: SymbolIdentifier,
1060    parameter: SymbolIdentifier,
1061    geometric_identifier: SymbolIdentifier,
1062) -> Option<IndexRationalFunction> {
1063    use crate::expression::ExpressionKind;
1064    match &expression.kind {
1065        ExpressionKind::Power(base, exponent) if matches!(&base.kind, ExpressionKind::Symbol(identifier) if *identifier == parameter) =>
1066        {
1067            let (multiplier, offset) = geometric_exponent_shape(exponent, index_variable)?;
1068            let geometric_position = field.position_of(geometric_identifier)?;
1069            let parameter_position = field.position_of(parameter)?;
1070            let geometric_power = field.power(&field.parameter(geometric_position), multiplier)?;
1071            let parameter_power = field.power(&field.parameter(parameter_position), offset)?;
1072            let value = field.multiply(&geometric_power, &parameter_power);
1073            Some(rational_function_from_polynomial(
1074                field,
1075                &field.polynomial_constant(&value),
1076            ))
1077        }
1078        ExpressionKind::Symbol(identifier) if *identifier == index_variable => Some(
1079            rational_function_from_polynomial(field, &field.polynomial_index()),
1080        ),
1081        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) | ExpressionKind::Symbol(_) => {
1082            Some(rational_function_from_polynomial(
1083                field,
1084                &field.polynomial_constant(&field.from_expression(expression)?),
1085            ))
1086        }
1087        ExpressionKind::Sum(children) => {
1088            children
1089                .iter()
1090                .try_fold(rational_function_zero(field), |total, child| {
1091                    Some(rational_function_add(
1092                        field,
1093                        &total,
1094                        &difference_rational_from_expression(
1095                            child,
1096                            field,
1097                            index_variable,
1098                            parameter,
1099                            geometric_identifier,
1100                        )?,
1101                    ))
1102                })
1103        }
1104        ExpressionKind::Product(children) => children.iter().try_fold(
1105            rational_function_from_polynomial(field, &field.polynomial_one()),
1106            |total, child| {
1107                Some(rational_function_multiply(
1108                    field,
1109                    &total,
1110                    &difference_rational_from_expression(
1111                        child,
1112                        field,
1113                        index_variable,
1114                        parameter,
1115                        geometric_identifier,
1116                    )?,
1117                ))
1118            },
1119        ),
1120        ExpressionKind::Power(base, exponent) => {
1121            let exponent = match &exponent.kind {
1122                ExpressionKind::Integer(value) => crate::exact_arithmetic::integer_to_i64(value)?,
1123                _ => return None,
1124            };
1125            let value = difference_rational_from_expression(
1126                base,
1127                field,
1128                index_variable,
1129                parameter,
1130                geometric_identifier,
1131            )?;
1132            crate::summation::hypergeometric_term::rational_function_power(field, &value, exponent)
1133        }
1134        _ => None,
1135    }
1136}

1138fn geometric_exponent_shape(
1139    expression: &ExpressionReference,
1140    index_variable: SymbolIdentifier,
1141) -> Option<(i64, i64)> {
1142    // Read an exponent as the affine form `multiplier * index + offset`, returning the pair. Both
1143    // halves matter: `c^(2 n + 2)` is `z^2 c^2`, not `z c^2`, and a constant exponent is a power of
1144    // the base alone with no `z` at all. Collapsing either case into a bare offset would silently
1145    // rewrite the equation, so a nonaffine exponent declines instead.
1146    use crate::expression::ExpressionKind;
1147    match &expression.kind {
1148        ExpressionKind::Symbol(identifier) if *identifier == index_variable => Some((1, 0)),
1149        ExpressionKind::Sum(children) => children.iter().try_fold((0i64, 0i64), |total, child| {
1150            let (multiplier, offset) = geometric_exponent_shape(child, index_variable)?;
1151            Some((
1152                total.0.checked_add(multiplier)?,
1153                total.1.checked_add(offset)?,
1154            ))
1155        }),
1156        ExpressionKind::Integer(value) => {
1157            Some((0, crate::exact_arithmetic::integer_to_i64(value)?))
1158        }
1159        ExpressionKind::Product(children) if children.len() == 2 => {
1160            let integer = children.iter().find_map(|child| match &child.kind {
1161                ExpressionKind::Integer(value) => crate::exact_arithmetic::integer_to_i64(value),
1162                _ => None,
1163            })?;
1164            let other = children
1165                .iter()
1166                .find(|child| !matches!(&child.kind, ExpressionKind::Integer(_)))?;
1167            let (multiplier, offset) = geometric_exponent_shape(other, index_variable)?;
1168            Some((
1169                integer.checked_mul(multiplier)?,
1170                integer.checked_mul(offset)?,
1171            ))
1172        }
1173        _ => None,
1174    }
1175}

1206// The difference operator with polynomial coefficients, in the form both the denominator and the
1207// degree derivations read it. `coefficients` are `A_0, ..., A_N` after the common denominator
1208// `clearing` has been multiplied through and the content that introduced has been divided out, so
1209// that `content * A_j` is `clearing` times the operator's original `j`-th coefficient.
1210struct ClearedDifferenceOperator {
1211    coefficients: Vec<ParameterPolynomial>,
1212    clearing: ParameterPolynomial,
1213    content: ParameterPolynomial,
1214}
1215
1216fn cleared_difference_operator(
1217    field: &CoefficientField,
1218    coefficients: &[IndexRationalFunction],
1219) -> Option<ClearedDifferenceOperator> {
1220    let elements: Vec<FieldElement> = coefficients
1221        .iter()
1222        .map(|coefficient| index_free_element(field, coefficient))
1223        .collect::<Option<Vec<_>>>()?;
1224    let mut clearing = field.parameter_constant(&crate::exact_arithmetic::rational_one());
1225    for element in &elements {
1226        clearing = field.parameter_multiply(&clearing, &field.denominator_of(element));
1227    }
1228    let mut cleared: Vec<ParameterPolynomial> = elements
1229        .iter()
1230        .map(|element| {
1231            let complement =
1232                field.parameter_divide_exact(&clearing, &field.denominator_of(element))?;
1233            Some(field.parameter_multiply(&element.numerator, &complement))
1234        })
1235        .collect::<Option<Vec<_>>>()?;
1236    // Divide out the content the common clearing multiple introduced. A larger denominator would
1237    // still be sound, but it would widen the ansatz for no mathematical reason.
1238    let mut content = cleared.first()?.clone();
1239    for value in &cleared[1..] {
1240        content = field.parameter_greatest_common_divisor(&content, value);
1241    }
1242    if content.total_degree() > 0 {
1243        for value in &mut cleared {
1244            *value = field.parameter_divide_exact(value, &content)?;
1245        }
1246    } else {
1247        content = field.parameter_constant(&crate::exact_arithmetic::rational_one());
1248    }
1249    Some(ClearedDifferenceOperator {
1250        coefficients: cleared,
1251        clearing,
1252        content,
1253    })
1254}

1256fn derived_universal_denominator(
1257    field: &CoefficientField,
1258    operator: &ClearedDifferenceOperator,
1259    parameter_position: usize,
1260    geometric_position: usize,
1261) -> Option<FieldPolynomial> {
1262    // Abramov's universal denominator, carried over to the multiplicative difference field
1263    // `Q(c)(z)` with `sigma(z) = c * z`. Clear the operator to polynomial coefficients
1264    // `A_0, ..., A_N`. If `y = p / q` in lowest terms solves the equation, every irreducible factor
1265    // of `q` lies in a `sigma`-orbit segment whose bottom divides `A_0` and whose top, raised by
1266    // `N + height`, divides `A_N`. Collecting
1267    // `G_height = sigma^-(N + height) gcd(sigma^(N + height) A_0, A_N)` over the dispersion range the
1268    // operator's own multiplicative degrees bound, and multiplying each orbit segment
1269    // `sigma^0 .. sigma^height`, yields a polynomial every rational solution's denominator divides.
1270    // The whole construction reads the operator; no denominator belonging to a particular equation's
1271    // answer is supplied from outside.
1272    let order = operator.coefficients.len().checked_sub(1)?;
1273    let trailing = operator.coefficients.first()?.clone();
1274    let leading = operator.coefficients.last()?.clone();
1275    if trailing.is_zero() || leading.is_zero() {
1276        return None;
1277    }
1278    // Every orbit segment that can contribute is shorter than the combined multiplicative degree of
1279    // the two extreme coefficients, because each step of the segment consumes one root of `A_N`.
1280    let dispersion_bound =
1281        trailing.degree_in(geometric_position) + leading.degree_in(geometric_position);
1282    let mut denominator = field.parameter_constant(&crate::exact_arithmetic::rational_one());
1283    for height in 0..=dispersion_bound {
1284        let steps = i64::try_from(order + height).ok()?;
1285        let raised = geometric_shift_substitution(
1286            field,
1287            &trailing,
1288            parameter_position,
1289            geometric_position,
1290            steps,
1291        )?;
1292        let common_factor = field.parameter_greatest_common_divisor(&raised, &leading);
1293        if common_factor.degree_in(geometric_position) == 0 {
1294            continue;
1295        }
1296        let segment_bottom = geometric_shift_substitution(
1297            field,
1298            &common_factor,
1299            parameter_position,
1300            geometric_position,
1301            -steps,
1302        )?;
1303        for step in 0..=height {
1304            let orbit_member = geometric_shift_substitution(
1305                field,
1306                &segment_bottom,
1307                parameter_position,
1308                geometric_position,
1309                i64::try_from(step).ok()?,
1310            )?;
1311            denominator = field.parameter_multiply(&denominator, &orbit_member);
1312        }
1313    }
1314    if denominator.degree_in(geometric_position) == 0 {
1315        return None;
1316    }
1317    Some(field.polynomial_constant(&field.from_parameter_polynomial(&denominator)))
1318}

1320/// The rectangle of numerator monomials the rational tier searches: every `z^a * n^b` with
1321/// `a <= geometric_degree` and `b <= index_degree`, over the derived universal denominator.
1322struct NumeratorDegreeBounds {
1323    geometric_degree: usize,
1324    index_degree: usize,
1325}
1326
1327/// How far the derived ansatz is allowed to reach before the tier reports a resource limit instead
1328/// of building the system. Exceeding it is not a statement that the equation has no rational
1329/// solution: it is a refusal to spend unbounded work on one.
1330const RATIONAL_ANSATZ_DEGREE_CEILING: usize = 12;
1331
1332/// How far a candidate resonant exponent may lie before it is treated as out of reach. The
1333/// candidates come from degree differences in the parameter, so a large one means the indicial
1334/// polynomial's monomial roots are spread far apart, and the ansatz it would demand is far past the
1335/// ceiling above in any case.
1336const RESONANT_EXPONENT_SEARCH_CEILING: i64 = 1024;
1337
1338fn derived_numerator_degree_bounds(
1339    field: &CoefficientField,
1340    operator: &ClearedDifferenceOperator,
1341    denominator: &FieldPolynomial,
1342    forcing: &IndexRationalFunction,
1343    parameter_position: usize,
1344    geometric_position: usize,
1345) -> Option<NumeratorDegreeBounds> {
1346    // Degree bounds for the numerator `P` of a solution `y = P / Q`, where `Q` is the universal
1347    // denominator derived above, so that the ansatz is read off the operator instead of being
1348    // fixed in advance. Write `N` for the order, `q` for the multiplicative degree of `Q`, and
1349    // substitute `y = P / Q` into `sum_j A_j sigma^j y = f`; multiplying by `prod_k sigma^k Q`
1350    // turns it into the polynomial equation `sum_j B_j sigma^j P = R` with
1351    // `B_j = A_j prod_{k != j} sigma^k Q` and `R = f prod_k sigma^k Q`.
1352    //
1353    // Multiplicative degree. Let `K` be the largest multiplicative degree among the `B_j` and
1354    // `b_j` the coefficient of `z^K` in `B_j`. For `P` of multiplicative degree `d` whose `z^d`
1355    // coefficient is a polynomial in the index of degree `m`, the coefficient of `z^(d + K) n^m` in
1356    // `sum_j B_j sigma^j P` is the leading index coefficient times `chi(c^d)`, where
1357    // `chi(X) = sum_j b_j X^j`: applying `sigma^j` multiplies `z^d` by `c^(j d)` and shifting the
1358    // index does not change a leading coefficient. So either `chi(c^d) = 0`, or `z^(d + K)` really
1359    // occurs and `d = deg R - K`. Since `sigma^k Q` has degree `q` with leading coefficient
1360    // `c^(k q) lc(Q)`, the common factor `lc(Q)^N c^(q N (N + 1) / 2)` cancels out of `chi` and
1361    // leaves `chi(X) = sum_j lc_z(A_j) c^(q (N - j)) X^j`, read directly off the cleared operator;
1362    // and `deg R - K` collapses to `deg_z f + q - deg_z A`, with `deg_z A` the largest
1363    // multiplicative degree among the cleared coefficients.
1364    //
1365    // The exponents `d` with `chi(c^d) = 0` are found exactly rather than searched. For symbolic
1366    // `c` the sum `sum_j b_j c^(j d)` vanishes only if its highest power of `c` cancels, so two
1367    // indices `j < k` must satisfy `deg_c b_j + j d = deg_c b_k + k d`; every candidate is
1368    // therefore `(deg_c b_j - deg_c b_k) / (k - j)`, and each candidate is confirmed or discarded
1369    // by evaluating `chi(c^d)` exactly. This is the point at which the field's constants are taken
1370    // to have infinite multiplicative orbits: the candidate test compares powers of a symbolic `c`,
1371    // and a root of unity would identify powers this argument keeps apart, which is why the family
1372    // is returned under a domain condition that excludes those specializations.
1373    //
1374    // Index degree. The `B_j` do not contain the index, so the operator commutes with the pure
1375    // index shift and hence with the difference `Delta P = P(n + 1, z) - P(n, z)`, which lowers the
1376    // index degree by exactly one. If `deg_n R = rho` then `Delta^(rho + 1) P` solves the
1377    // homogeneous equation; a homogeneous solution of index degree `m` yields the `m + 1`
1378    // independent homogeneous solutions `Delta^i P`, and the solution space of an order-`N`
1379    // operator has dimension at most `N`, so `m <= N - 1`. Therefore `deg_n P <= rho + N`, and
1380    // `deg_n P <= N - 1` when the equation is homogeneous.
1381    let order = operator.coefficients.len().checked_sub(1)?;
1382    let universal = index_free_polynomial_element(field, denominator)?;
1383    if field
1384        .denominator_of(&universal)
1385        .degree_in(geometric_position)
1386        != 0
1387    {
1388        return None;
1389    }
1390    let universal_degree = universal.numerator.degree_in(geometric_position);
1391    let greatest_coefficient_degree = operator
1392        .coefficients
1393        .iter()
1394        .filter(|coefficient| !coefficient.is_zero())
1395        .map(|coefficient| coefficient.degree_in(geometric_position))
1396        .max()?;
1397    let parameter = field.parameter_monomial(parameter_position);
1398    let indicial: Vec<ParameterPolynomial> = operator
1399        .coefficients
1400        .iter()
1401        .enumerate()
1402        .map(|(shift, coefficient)| {
1403            let leading = geometric_coefficient(
1404                field,
1405                coefficient,
1406                geometric_position,
1407                greatest_coefficient_degree,
1408            );
1409            let compensation =
1410                field.parameter_power(&parameter, universal_degree * (order - shift));
1411            field.parameter_multiply(&leading, &compensation)
1412        })
1413        .collect();
1414    let mut candidates: Vec<i64> = Vec::new();
1415    for (lower, lower_value) in indicial.iter().enumerate() {
1416        if lower_value.is_zero() {
1417            continue;
1418        }
1419        for (upper, upper_value) in indicial.iter().enumerate().skip(lower + 1) {
1420            if upper_value.is_zero() {
1421                continue;
1422            }
1423            let difference = i64::try_from(lower_value.degree_in(parameter_position)).ok()?
1424                - i64::try_from(upper_value.degree_in(parameter_position)).ok()?;
1425            let gap = i64::try_from(upper - lower).ok()?;
1426            if difference % gap != 0 {
1427                continue;
1428            }
1429            let candidate = difference / gap;
1430            if candidate >= 0 && !candidates.contains(&candidate) {
1431                candidates.push(candidate);
1432            }
1433        }
1434    }
1435    let mut resonant: Vec<i64> = Vec::new();
1436    for candidate in candidates {
1437        if candidate > RESONANT_EXPONENT_SEARCH_CEILING {
1438            return None;
1439        }
1440        let mut total = field.polynomial_zero_parameter();
1441        for (shift, value) in indicial.iter().enumerate() {
1442            let exponent = usize::try_from(candidate).ok()?.checked_mul(shift)?;
1443            let power = field.parameter_power(&parameter, exponent);
1444            total = field.parameter_add(&total, &field.parameter_multiply(value, &power));
1445        }
1446        if total.is_zero() {
1447            resonant.push(candidate);
1448        }
1449    }
1450    let forcing_degrees = cleared_forcing_degrees(field, operator, forcing, geometric_position)?;
1451    let mut geometric_bound = resonant.into_iter().max();
1452    if let Some((geometric_degree, _)) = forcing_degrees {
1453        let from_forcing = i64::try_from(geometric_degree + universal_degree).ok()?
1454            - i64::try_from(greatest_coefficient_degree).ok()?;
1455        geometric_bound =
1456            Some(geometric_bound.map_or(from_forcing, |bound| bound.max(from_forcing)));
1457    }
1458    let geometric_degree = usize::try_from(geometric_bound?).ok()?;
1459    let index_degree = match forcing_degrees {
1460        Some((_, index_degree)) => index_degree.checked_add(order)?,
1461        None => order.checked_sub(1)?,
1462    };
1463    (geometric_degree <= RATIONAL_ANSATZ_DEGREE_CEILING
1464        && index_degree <= RATIONAL_ANSATZ_DEGREE_CEILING)
1465        .then_some(NumeratorDegreeBounds {
1466            geometric_degree,
1467            index_degree,
1468        })
1469}

1471fn cleared_forcing_degrees(
1472    field: &CoefficientField,
1473    operator: &ClearedDifferenceOperator,
1474    forcing: &IndexRationalFunction,
1475    geometric_position: usize,
1476) -> Option<Option<(usize, usize)>> {
1477    // The multiplicative and index degrees of the forcing once the operator's clearing multiple has
1478    // been applied to it, or `None` for a homogeneous equation. A forcing that is not polynomial in
1479    // the multiplicative variable after clearing leaves the class the universal denominator was
1480    // derived for, so the whole derivation declines rather than bounding a class it does not cover.
1481    let forcing_denominator = index_free_polynomial_element(field, &forcing.denominator)?;
1482    if forcing_denominator.is_zero() {
1483        return None;
1484    }
1485    let scale = field.divide(
1486        &field.from_parameter_polynomial(&operator.clearing),
1487        &field.multiply(
1488            &field.from_parameter_polynomial(&operator.content),
1489            &forcing_denominator,
1490        ),
1491    )?;
1492    let mut degrees: Option<(usize, usize)> = None;
1493    for (index_degree, coefficient) in &forcing.numerator.terms {
1494        let scaled = field.multiply(coefficient, &scale);
1495        if scaled.is_zero() {
1496            continue;
1497        }
1498        if field.denominator_of(&scaled).degree_in(geometric_position) != 0 {
1499            return None;
1500        }
1501        let geometric_degree = scaled.numerator.degree_in(geometric_position);
1502        degrees = Some(match degrees {
1503            None => (geometric_degree, *index_degree),
1504            Some((geometric, index)) => (geometric.max(geometric_degree), index.max(*index_degree)),
1505        });
1506    }
1507    Some(degrees)
1508}

1510fn index_free_polynomial_element(
1511    field: &CoefficientField,
1512    value: &FieldPolynomial,
1513) -> Option<FieldElement> {
1514    // Read a polynomial in the index as a ground-field element, declining when the index occurs.
1515    match value.degree() {
1516        None => Some(field.zero()),
1517        Some(0) => value.coefficient(0).cloned(),
1518        Some(_) => None,
1519    }
1520}

1522fn index_free_element(
1523    field: &CoefficientField,
1524    value: &IndexRationalFunction,
1525) -> Option<FieldElement> {
1526    // Read a rational function of the index as a ground-field element, declining when the index
1527    // actually occurs. The denominator derivation below is stated for index-free coefficients; an
1528    // index-dependent operator leaves this tier rather than being approximated inside it.
1529    let numerator = match value.numerator.degree() {
1530        None => return Some(field.zero()),
1531        Some(0) => value.numerator.coefficient(0)?.clone(),
1532        Some(_) => return None,
1533    };
1534    let denominator = match value.denominator.degree() {
1535        Some(0) => value.denominator.coefficient(0)?.clone(),
1536        _ => return None,
1537    };
1538    field.divide(&numerator, &denominator)
1539}

1541fn geometric_shift_substitution(
1542    field: &CoefficientField,
1543    value: &ParameterPolynomial,
1544    parameter_position: usize,
1545    geometric_position: usize,
1546    steps: i64,
1547) -> Option<ParameterPolynomial> {
1548    // Apply `sigma^steps` to a polynomial in the multiplicative variable, where `sigma(z) = c * z`.
1549    // A negative step sends `z` to `z / c^|steps|`, which pairs each stored power of `z` with a
1550    // negative power of `c`; clearing them multiplies the polynomial by `c^(|steps| * degree)`. That
1551    // factor is a unit of the difference field for every nonzero `c`, so it changes no divisibility
1552    // claim, only which representative of the associate class is returned.
1553    if steps == 0 {
1554        return Some(value.clone());
1555    }
1556    let geometric = field.parameter_monomial(geometric_position);
1557    let parameter = field.parameter_monomial(parameter_position);
1558    let magnitude = usize::try_from(steps.unsigned_abs()).ok()?;
1559    if steps > 0 {
1560        let scale = field.parameter_power(&parameter, magnitude);
1561        let replacement = field.parameter_multiply(&geometric, &scale);
1562        return Some(field.parameter_substitute(value, geometric_position, &replacement));
1563    }
1564    let degree = value.degree_in(geometric_position);
1565    let mut result = field.polynomial_zero_parameter();
1566    for exponent in 0..=degree {
1567        let coefficient = geometric_coefficient(field, value, geometric_position, exponent);
1568        if coefficient.is_zero() {
1569            continue;
1570        }
1571        let compensation = field.parameter_power(&parameter, (degree - exponent) * magnitude);
1572        let geometric_power = field.parameter_power(&geometric, exponent);
1573        let scaled = field.parameter_multiply(&coefficient, &compensation);
1574        let term = field.parameter_multiply(&scaled, &geometric_power);
1575        result = field.parameter_add(&result, &term);
1576    }
1577    Some(result)
1578}

1580fn geometric_coefficient(
1581    field: &CoefficientField,
1582    value: &ParameterPolynomial,
1583    geometric_position: usize,
1584    exponent: usize,
1585) -> ParameterPolynomial {
1586    // Read the coefficient of `z^exponent`, itself a polynomial in the remaining parameters.
1587    let mut result = field.polynomial_zero_parameter();
1588    for (exponents, coefficient) in &value.terms {
1589        if exponents[geometric_position] as usize != exponent {
1590            continue;
1591        }
1592        let mut remaining = exponents.clone();
1593        remaining[geometric_position] = 0;
1594        let mut term = field.polynomial_zero_parameter();
1595        term.terms.insert(remaining, coefficient.clone());
1596        result = field.parameter_add(&result, &term);
1597    }
1598    result
1599}

1620                numerator: field.polynomial_monomial(index_degree, &geometric_power),

1623        }
1624    }

1628fn difference_sigma(
1629    field: &CoefficientField,
1630    value: &IndexRationalFunction,
1631    geometric_position: usize,
1632    offset: i64,
1633) -> Option<IndexRationalFunction> {
1634    let shifted = rational_function_shift_index(field, value, offset);
1635    let parameter_position =
1636        field
1637            .parameters()
1638            .iter()
1639            .enumerate()
1640            .find_map(|(position, identifier)| {
1641                (*identifier != field.parameters()[geometric_position]).then_some(position)
1642            })?;
1643    let parameter = field.parameter(parameter_position).numerator;
1644    let geometric = field.parameter(geometric_position).numerator;
1645    let factor = field.parameter_power(&parameter, offset.unsigned_abs() as usize);
1646    let replacement = if offset >= 0 {
1647        field.parameter_multiply(&geometric, &factor)
1648    } else {
1649        return None;
1650    };
1651    Some(IndexRationalFunction {
1652        numerator: field.polynomial_substitute_parameter(
1653            &shifted.numerator,
1654            geometric_position,
1655            &replacement,
1656        )?,
1657        denominator: field.polynomial_substitute_parameter(
1658            &shifted.denominator,
1659            geometric_position,
1660            &replacement,
1661        )?,
1662    })
1663}

1665fn difference_operator_apply(
1666    field: &CoefficientField,
1667    coefficients: &[IndexRationalFunction],
1668    candidate: &IndexRationalFunction,
1669    geometric_position: usize,
1670) -> Option<IndexRationalFunction> {
1671    let terms = coefficients
1672        .iter()
1673        .enumerate()
1674        .map(|(shift, coefficient)| {
1675            let shifted = difference_sigma(
1676                field,
1677                candidate,
1678                geometric_position,
1679                i64::try_from(shift).ok()?,
1680            )?;
1681            Some(IndexRationalFunction {
1682                numerator: field
1683                    .polynomial_multiply_uncancelled(&coefficient.numerator, &shifted.numerator)?,
1684                denominator: field
1685                    .polynomial_multiply(&coefficient.denominator, &shifted.denominator),
1686            })
1687        })
1688        .collect::<Option<Vec<_>>>()?;
1689    let mut distinct_denominators = Vec::new();
1690    for term in &terms {
1691        if !distinct_denominators.contains(&term.denominator) {
1692            distinct_denominators.push(term.denominator.clone());
1693        }
1694    }
1695    let common_denominator = distinct_denominators
1696        .iter()
1697        .fold(field.polynomial_one(), |product, denominator| {
1698            field.polynomial_multiply(&product, denominator)
1699        });
1700    let scaled_numerators = terms
1701        .iter()
1702        .map(|term| {
1703            let multiplier =
1704                field.polynomial_divide_exact(&common_denominator, &term.denominator)?;
1705            field.polynomial_multiply_uncancelled(&term.numerator, &multiplier)
1706        })
1707        .collect::<Option<Vec<_>>>()?;
1708    Some(IndexRationalFunction {
1709        numerator: field_polynomial_sum(field, &scaled_numerators)?,
1710        denominator: common_denominator,
1711    })
1712}

1714fn field_polynomial_sum(
1715    field: &CoefficientField,
1716    polynomials: &[FieldPolynomial],
1717) -> Option<FieldPolynomial> {
1718    let mut terms: BTreeMap<usize, Vec<FieldElement>> = BTreeMap::new();
1719    for polynomial in polynomials {
1720        for (degree, coefficient) in &polynomial.terms {
1721            terms.entry(*degree).or_default().push(coefficient.clone());
1722        }
1723    }
1724    let terms = terms
1725        .into_iter()
1726        .map(|(degree, coefficients)| Some((degree, field_element_sum(field, &coefficients)?)))
1727        .collect::<Option<BTreeMap<_, _>>>()?;
1728    Some(FieldPolynomial { terms })
1729}

1731fn field_element_sum(field: &CoefficientField, values: &[FieldElement]) -> Option<FieldElement> {
1732    let mut distinct_denominators = Vec::new();
1733    for value in values {
1734        if !distinct_denominators.contains(&value.denominator) {
1735            distinct_denominators.push(value.denominator.clone());
1736        }
1737    }
1738    let denominator = distinct_denominators.iter().fold(
1739        field.parameter_constant(&crate::exact_arithmetic::Rational::ONE),
1740        |product, factor| field.parameter_multiply(&product, factor),
1741    );
1742    let numerator = values
1743        .iter()
1744        .try_fold(field.polynomial_zero_parameter(), |total, value| {
1745            let factor = field.parameter_divide_exact(&denominator, &value.denominator)?;
1746            Some(field.parameter_add(&total, &field.parameter_multiply(&value.numerator, &factor)))
1747        })?;
1748    field.make_element(numerator, denominator)
1749}

1751fn rational_operator_linear_system(
1752    field: &CoefficientField,
1753    parameter_field: &CoefficientField,
1754    residuals: &[IndexRationalFunction],
1755    forcing: &IndexRationalFunction,
1756    geometric_position: usize,
1757) -> Option<(Vec<Vec<FieldElement>>, Vec<FieldElement>)> {
1758    let mut distinct_denominators = vec![forcing.denominator.clone()];
1759    for residual in residuals {
1760        if !distinct_denominators.contains(&residual.denominator) {
1761            distinct_denominators.push(residual.denominator.clone());
1762        }
1763    }
1764    let common_denominator = distinct_denominators
1765        .iter()
1766        .fold(field.polynomial_one(), |product, denominator| {
1767            field.polynomial_multiply(&product, denominator)
1768        });
1769    let scaled_numerators: Vec<_> = residuals
1770        .iter()
1771        .map(|residual| {
1772            let multiplier =
1773                field.polynomial_divide_exact(&common_denominator, &residual.denominator)?;
1774            Some(field.polynomial_multiply(&residual.numerator, &multiplier))
1775        })
1776        .collect::<Option<Vec<_>>>()?;
1777    let forcing_numerator = field.polynomial_multiply(
1778        &forcing.numerator,
1779        &field.polynomial_divide_exact(&common_denominator, &forcing.denominator)?,
1780    );
1781    let greatest_degree = scaled_numerators
1782        .iter()
1783        .chain(std::iter::once(&forcing_numerator))
1784        .filter_map(FieldPolynomial::degree)
1785        .max()
1786        .unwrap_or(0);
1787    let mut rows = Vec::new();
1788    let mut right_hand_side = Vec::new();
1789    for degree in 0..=greatest_degree {
1790        let entries: Vec<_> = scaled_numerators
1791            .iter()
1792            .map(|numerator| {
1793                numerator
1794                    .coefficient(degree)
1795                    .cloned()
1796                    .unwrap_or_else(|| field.zero())
1797            })
1798            .collect();
1799        let value = field.negate(
1800            &forcing_numerator
1801                .coefficient(degree)
1802                .cloned()
1803                .unwrap_or_else(|| field.zero()),
1804        );
1805        let mut denominator = value.denominator.clone();
1806        for entry in &entries {
1807            denominator = field.parameter_multiply(&denominator, &entry.denominator);
1808        }
1809        let scaled_entries: Vec<_> = entries
1810            .iter()
1811            .map(|entry| {
1812                let factor = field.parameter_divide_exact(&denominator, &entry.denominator)?;
1813                Some(field.parameter_multiply(&entry.numerator, &factor))
1814            })
1815            .collect::<Option<Vec<_>>>()?;
1816        let value_factor = field.parameter_divide_exact(&denominator, &value.denominator)?;
1817        let scaled_value = field.parameter_multiply(&value.numerator, &value_factor);
1818        let greatest_geometric_degree = scaled_entries
1819            .iter()
1820            .chain(std::iter::once(&scaled_value))
1821            .flat_map(|polynomial| {
1822                polynomial
1823                    .terms
1824                    .keys()
1825                    .map(|exponents| exponents[geometric_position])
1826            })
1827            .max()
1828            .unwrap_or(0);
1829        for geometric_degree in 0..=greatest_geometric_degree {
1830            rows.push(
1831                scaled_entries
1832                    .iter()
1833                    .map(|polynomial| {
1834                        parameter_polynomial_coefficient(
1835                            field,
1836                            parameter_field,
1837                            polynomial,
1838                            geometric_position,
1839                            geometric_degree,
1840                        )
1841                    })
1842                    .collect::<Option<Vec<_>>>()?,
1843            );
1844            right_hand_side.push(parameter_polynomial_coefficient(
1845                field,
1846                parameter_field,
1847                &scaled_value,
1848                geometric_position,
1849                geometric_degree,
1850            )?);
1851        }
1852    }
1853    Some((rows, right_hand_side))
1854}

1997fn initial_value_in_parameter_field(
1998    context: &Context,
1999    field: &CoefficientField,
2000    parameter_field: &CoefficientField,
2001    value: &IndexRationalFunction,
2002    index: &ExpressionReference,
2003    _index_variable: SymbolIdentifier,
2004    parameter: SymbolIdentifier,
2005    geometric_identifier: SymbolIdentifier,
2006) -> Option<FieldElement> {
2007    let index_value = match &index.kind {
2008        crate::expression::ExpressionKind::Integer(value) => {
2009            crate::exact_arithmetic::integer_to_i64(value)?
2010        }
2011        _ => return None,
2012    };
2013    let shifted = difference_sigma(
2014        field,
2015        value,
2016        field.position_of(geometric_identifier)?,
2017        index_value,
2018    )?;
2019    let zero = field.zero();
2020    let numerator = field.polynomial_evaluate(&shifted.numerator, &zero);
2021    let denominator = field.polynomial_evaluate(&shifted.denominator, &zero);
2022    let evaluated = field.divide(&numerator, &denominator)?;
2023    let expression = field.to_expression(context, &evaluated);
2024    let geometric_symbol = context
2025        .interner
2026        .intern(crate::expression::ExpressionKind::Symbol(
2027            geometric_identifier,
2028        ));
2029    let parameter_symbol = context
2030        .interner
2031        .intern(crate::expression::ExpressionKind::Symbol(parameter));
2032    let _ = parameter_symbol;
2033    let replacement = builders::integer_value(context, 1);
2034    let mut mapping = crate::expression::substitute::SubstitutionMap::new();
2035    mapping.insert(geometric_symbol, replacement);
2036    let specialised = crate::simplify::simplify(
2037        context,
2038        crate::expression::substitute::substitute(context, &expression, &mapping),
2039    );
2040    parameter_field.from_expression(&specialised)
2041}

2060fn assemble_general_solution_with_particular(
2061    context: &Context,
2062    basis: &[ExpressionReference],
2063    particular_solution: ExpressionReference,
2064    sequence_variable_call: &ExpressionReference,
2065    index_variable: SymbolIdentifier,
2066    initial_conditions: Option<&[InitialCondition]>,
2067) -> Option<ExpressionReference> {
2068    // Glue the hypergeometric basis into a linear combination with one
2069    // fresh symbolic constant per basis element. When initial
2070    // conditions are supplied the constants are determined by solving
2071    // the resulting linear system; otherwise the symbolic-constant
2072    // form is returned so the user can fit them downstream. Data that
2073    // contradict the recurrence have no solution at all, and that is
2074    // reported as a decline: answering such a request with the
2075    // unconstrained family would silently drop the window the caller
2076    // asked about.
2077    let mut basis_terms: Vec<ExpressionReference> = Vec::with_capacity(basis.len());
2078    let mut constants: Vec<ExpressionReference> = Vec::with_capacity(basis.len());
2079    for (index, element) in basis.iter().enumerate() {
2080        let constant_identifier = context
2081            .symbol_table
2082            .fresh(&format!("constant_recurrence_{}", index));
2083        let constant_symbol = context
2084            .interner
2085            .intern(crate::expression::ExpressionKind::Symbol(
2086                constant_identifier,
2087            ));
2088        constants.push(Arc::clone(&constant_symbol));
2089        let term = builders::product(context, vec![constant_symbol, Arc::clone(element)]);
2090        basis_terms.push(term);
2091    }
2092    let homogeneous_combination = if basis_terms.len() == 1 {
2093        basis_terms.into_iter().next().unwrap()
2094    } else {
2095        builders::sum(context, basis_terms)
2096    };
2097    let combination = builders::sum(
2098        context,
2099        vec![particular_solution.clone(), homogeneous_combination],
2100    );
2101    match initial_conditions {
2102        None => Some(combination),
2103        Some(conditions) => match fit_initial_conditions(
2104            context,
2105            &combination,
2106            &constants,
2107            basis,
2108            &particular_solution,
2109            sequence_variable_call,
2110            index_variable,
2111            conditions,
2112        ) {
2113            InitialWindowFit::Fitted(fitted) => Some(fitted),
2114            InitialWindowFit::Undetermined => Some(combination),
2115            InitialWindowFit::NoSolution => None,
2116        },
2117    }
2118}

2120// The three outcomes of fitting a declared initial window, which a single `Option` cannot tell
2121// apart. `Undetermined` means the window leaves the family constrained but not pinned, or the fit
2122// could not be certified, and the symbolic-constant form is still a correct answer. `NoSolution`
2123// means a datum is provably incompatible with the rest of the request, so no sequence answers it.
2124enum InitialWindowFit {
2125    Fitted(ExpressionReference),
2126    Undetermined,
2127    NoSolution,
2128}

2130fn fit_initial_conditions(
2131    context: &Context,
2132    combination: &ExpressionReference,
2133    constants: &[ExpressionReference],
2134    basis: &[ExpressionReference],
2135    particular_solution: &ExpressionReference,
2136    sequence_variable_call: &ExpressionReference,
2137    index_variable: SymbolIdentifier,
2138    conditions: &[InitialCondition],
2139) -> InitialWindowFit {
2140    // Determine every symbolic constant from a complete initial window. The general solution
2141    // is linear in the constants, so the window gives the square system
2142    //   Σ_j basis_j(index_i) * constant_j = value_i,
2143    // whose matrix entries are obtained by substituting each condition's index into each basis
2144    // element — no differentiation and no pattern matching. The system is solved by exact
2145    // symbolic Gauss–Jordan elimination, which is what lets an algebraic basis such as Binet's
2146    // golden-ratio pair be fitted without ever leaving exact arithmetic.
2147    //
2148    // A window may carry more data than there are constants. The extra data are consistency
2149    // constraints rather than degrees of freedom, so the square subsystem determines the constants
2150    // and every remaining datum is a condition the fitted form has to meet.
2151    //
2152    // The fitted form is then *replayed* against every supplied condition before it is returned. A
2153    // residual that is a provably nonzero number refutes the request outright: the data contradict
2154    // the recurrence and no sequence satisfies both. A residual that merely cannot be proved zero
2155    // is not a refutation, and there the symbolic-constant form is still a correct general
2156    // solution. Returning an unverified fit would be a rigour violation.
2157    let _ = sequence_variable_call;
2158    if constants.len() != basis.len() || basis.is_empty() || conditions.len() < constants.len() {
2159        return InitialWindowFit::Undetermined;
2160    }
2161    let index_expression = context
2162        .interner
2163        .intern(crate::expression::ExpressionKind::Symbol(index_variable));
2164    let fitting_conditions = &conditions[..constants.len()];
2165    let mut matrix: Vec<Vec<ExpressionReference>> = Vec::with_capacity(fitting_conditions.len());
2166    let mut right_hand_side: Vec<ExpressionReference> =
2167        Vec::with_capacity(fitting_conditions.len());
2168    for (index_value, sequence_value) in fitting_conditions {
2169        let mut substitution: crate::expression::substitute::SubstitutionMap =
2170            std::collections::HashMap::new();
2171        substitution.insert(Arc::clone(&index_expression), Arc::clone(index_value));
2172        let mut row: Vec<ExpressionReference> = Vec::with_capacity(basis.len());
2173        for element in basis {
2174            let entry = crate::expression::substitute::substitute(context, element, &substitution);
2175            row.push(crate::simplify::simplify(context, entry));
2176        }
2177        matrix.push(row);
2178        let particular_value =
2179            crate::expression::substitute::substitute(context, particular_solution, &substitution);
2180        right_hand_side.push(crate::simplify::simplify(
2181            context,
2182            builders::subtract(context, Arc::clone(sequence_value), particular_value),
2183        ));
2184    }
2185    let Some(fitted_values) = solve_symbolic_linear_system(context, matrix, right_hand_side) else {
2186        return InitialWindowFit::Undetermined;
2187    };
2188    let mut final_substitution: crate::expression::substitute::SubstitutionMap =
2189        std::collections::HashMap::new();
2190    for (constant, value) in constants.iter().zip(fitted_values.iter()) {
2191        final_substitution.insert(Arc::clone(constant), Arc::clone(value));
2192    }
2193    let fitted =
2194        crate::expression::substitute::substitute(context, combination, &final_substitution);
2195    let fitted = crate::simplify::simplify(context, fitted);
2196    for (index_value, sequence_value) in conditions {
2197        let mut substitution: crate::expression::substitute::SubstitutionMap =
2198            std::collections::HashMap::new();
2199        substitution.insert(Arc::clone(&index_expression), Arc::clone(index_value));
2200        let replayed = crate::expression::substitute::substitute(context, &fitted, &substitution);
2201        let residual = builders::subtract(context, replayed, Arc::clone(sequence_value));
2202        let residual = crate::simplify::simplify(context, residual);
2203        if matches!(
2204            &residual.kind,
2205            crate::expression::ExpressionKind::Integer(value) if crate::exact_arithmetic::integer_is_zero(value)
2206        ) {
2207            continue;
2208        }
2209        if matches!(
2210            &residual.kind,
2211            crate::expression::ExpressionKind::Integer(_)
2212                | crate::expression::ExpressionKind::Rational(_)
2213        ) {
2214            return InitialWindowFit::NoSolution;
2215        }
2216        return InitialWindowFit::Undetermined;
2217    }
2218    InitialWindowFit::Fitted(fitted)
2219}

2221fn solve_symbolic_linear_system(
2222    context: &Context,
2223    mut matrix: Vec<Vec<ExpressionReference>>,
2224    mut right_hand_side: Vec<ExpressionReference>,
2225) -> Option<Vec<ExpressionReference>> {
2226    // Exact Gauss–Jordan elimination over interned expressions. A pivot is any entry the
2227    // simplifier does not reduce to the zero literal, and the whole elimination routes every
2228    // arithmetic result back through `simplify`, so the algebraic cancellations an algebraic basis
2229    // needs (φ - ψ = √5, and the reciprocals that follow) happen inside the elimination rather
2230    // than being deferred to the caller. A singular or inconsistent system declines, which the
2231    // caller reads as "this window does not determine the constants".
2232    let size = matrix.len();
2233    if size == 0 || matrix.iter().any(|row| row.len() != size) {
2234        return None;
2235    }
2236    for pivot in 0..size {
2237        let candidate = (pivot..size).find(|row| !is_simplified_zero(&matrix[*row][pivot]))?;
2238        matrix.swap(pivot, candidate);
2239        right_hand_side.swap(pivot, candidate);
2240        let pivot_value = Arc::clone(&matrix[pivot][pivot]);
2241        for column in pivot..size {
2242            let scaled = builders::divide(
2243                context,
2244                Arc::clone(&matrix[pivot][column]),
2245                Arc::clone(&pivot_value),
2246            );
2247            matrix[pivot][column] = crate::simplify::simplify(context, scaled);
2248        }
2249        let scaled_value = builders::divide(
2250            context,
2251            Arc::clone(&right_hand_side[pivot]),
2252            Arc::clone(&pivot_value),
2253        );
2254        right_hand_side[pivot] = crate::simplify::simplify(context, scaled_value);
2255        for row in 0..size {
2256            if row == pivot || is_simplified_zero(&matrix[row][pivot]) {
2257                continue;
2258            }
2259            let factor = Arc::clone(&matrix[row][pivot]);
2260            for column in pivot..size {
2261                let adjustment = builders::product(
2262                    context,
2263                    vec![Arc::clone(&factor), Arc::clone(&matrix[pivot][column])],
2264                );
2265                let updated =
2266                    builders::subtract(context, Arc::clone(&matrix[row][column]), adjustment);
2267                matrix[row][column] = crate::simplify::simplify(context, updated);
2268            }
2269            let adjustment =
2270                builders::product(context, vec![factor, Arc::clone(&right_hand_side[pivot])]);
2271            let updated =
2272                builders::subtract(context, Arc::clone(&right_hand_side[row]), adjustment);
2273            right_hand_side[row] = crate::simplify::simplify(context, updated);
2274        }
2275    }
2276    Some(right_hand_side)
2277}

2280    // Whether an already-simplified entry is the zero literal. Used as the pivot test; anything the
2281    // simplifier leaves non-zero is treated as a usable pivot, which is sound because a spurious
2282    // pivot can only make the elimination decline later, never produce a wrong fit — and the fit
2283    // is replayed against every condition regardless.
2284    matches!(
2285        &value.kind,
2286        crate::expression::ExpressionKind::Integer(integer_value)
2287            if crate::exact_arithmetic::integer_is_zero(integer_value)
2288    )
