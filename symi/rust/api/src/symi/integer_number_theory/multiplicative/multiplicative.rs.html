11/// Performs the advanced native `euler_totient` operation.
12pub fn euler_totient(value: &Number) -> Option<Number> {
13    // Euler's totient function phi(n) = |{k in [1, n] : gcd(k, n) = 1}|. Uses
14    // the multiplicative-from-prime-factorisation formula
15    //     phi(n) = n * product_{p | n} (1 - 1/p)
16    // which yields the exact rational rearrangement
17    //     phi(n) = product_{p^k || n} (p - 1) * p^(k - 1).
18    // Returns None for rational inputs or non-positive integers; phi(1) is 1
19    // by convention.
20    let integer_value = positive_integer_or_none(value)?;
21    if integer_value == integer_one() {
22        return Some(Number::Integer(integer_one()));
23    }
24    let factorisation = factor_integer_internal(&integer_value)?;
25    let mut accumulator = integer_one();
26    for (prime_factor, multiplicity) in factorisation.iter() {
27        let multiplicity_minus_one = (*multiplicity as i64) - 1;
28        let prime_power = integer_power(prime_factor, multiplicity_minus_one as u64);
29        accumulator *= (prime_factor.clone() - integer_one()) * prime_power;
30    }
31    Some(Number::Integer(accumulator))
32}

34/// Performs the advanced native `mobius` operation.
35pub fn mobius(value: &Number) -> Option<Number> {
36    // Möbius function mu(n). Returns 1 when n is a squarefree positive integer
37    // with an even number of prime factors, -1 when squarefree with an odd
38    // count, and 0 when n has a squared prime factor. mu(1) = 1.
39    let integer_value = positive_integer_or_none(value)?;
40    if integer_value == integer_one() {
41        return Some(Number::Integer(integer_one()));
42    }
43    let factorisation = factor_integer_internal(&integer_value)?;
44    let mut squarefree_factor_count: i64 = 0;
45    for (_, multiplicity) in factorisation.iter() {
46        if *multiplicity > 1 {
47            return Some(Number::Integer(integer_zero()));
48        }
49        squarefree_factor_count += 1;
50    }
51    if squarefree_factor_count % 2 == 0 {
52        Some(Number::Integer(integer_one()))
53    } else {
54        Some(Number::Integer(-integer_one()))
55    }
56}

58/// Performs the advanced native `divisor_count` operation.
59pub fn divisor_count(value: &Number) -> Option<Number> {
60    // Number of positive divisors tau(n). Computed from the factorisation as
61    // product_{p^k || n} (k + 1). Returns None for rationals or non-positive
62    // inputs. tau(1) = 1.
63    let integer_value = positive_integer_or_none(value)?;
64    if integer_value == integer_one() {
65        return Some(Number::Integer(integer_one()));
66    }
67    let factorisation = factor_integer_internal(&integer_value)?;
68    let mut accumulator = integer_one();
69    for (_, multiplicity) in factorisation.iter() {
70        accumulator *= exact_integer::from((*multiplicity as i64) + 1);
71    }
72    Some(Number::Integer(accumulator))
73}

75/// Performs the advanced native `divisor_sigma` operation.
76pub fn divisor_sigma(value: &Number, power: u32) -> Option<Number> {
77    // Divisor power sum sigma_k(n) = sum_{d | n} d^k. The k = 0 case is the
78    // divisor count tau(n); k = 1 is the standard sum of divisors. Computed
79    // from the factorisation using the geometric-series-per-prime identity
80    //     sigma_k(p^a) = sum_{j = 0..a} p^(j*k) = (p^(k*(a+1)) - 1) / (p^k - 1)
81    // when k >= 1, and (a + 1) when k = 0. Returns None for rational or
82    // non-positive inputs.
83    let integer_value = positive_integer_or_none(value)?;
84    if power == 0 {
85        return divisor_count(&Number::Integer(integer_value));
86    }
87    if integer_value == integer_one() {
88        return Some(Number::Integer(integer_one()));
89    }
90    let factorisation = factor_integer_internal(&integer_value)?;
91    let mut accumulator = integer_one();
92    for (prime_factor, multiplicity) in factorisation.iter() {
93        let prime_power_unit = integer_power(prime_factor, power as u64);
94        let mut term_accumulator = integer_zero();
95        let mut current_power = integer_one();
96        for _ in 0..=*multiplicity {
97            term_accumulator += current_power.clone();
98            current_power *= prime_power_unit.clone();
99        }
100        accumulator *= term_accumulator;
101    }
102    Some(Number::Integer(accumulator))
103}

105/// Performs the advanced native `jacobi_symbol` operation.
106pub fn jacobi_symbol(numerator: &Number, denominator: &Number) -> Option<i8> {
107    // Jacobi symbol (a/n). Defined for any integer a and positive odd integer
108    // n; returns None for even or non-positive denominators and for rational
109    // inputs. The exact-arithmetic helper validates the odd-positive-denominator
110    // precondition and never panics on out-of-contract input.
111    let numerator_integer = integer_or_none(numerator)?;
112    let denominator_integer = integer_or_none(denominator)?;
113    integer_jacobi_symbol(numerator_integer, denominator_integer)
114}

116/// Performs the advanced native `legendre_symbol` operation.
117pub fn legendre_symbol(numerator: &Number, prime: &Number) -> Option<i8> {
118    // Legendre symbol (a/p) for odd prime p. The trait implementation matches
119    // Jacobi numerically; this wrapper documents the precondition that the
120    // second argument is an odd prime and validates only the easy structural
121    // constraints (odd, > 1). Primality is the caller's responsibility — the
122    // `is_prime` routine is the appropriate check.
123    let numerator_integer = integer_or_none(numerator)?;
124    let prime_integer = integer_or_none(prime)?;
125    integer_legendre_symbol(numerator_integer, prime_integer)
126}

128/// Performs the advanced native `kronecker_symbol` operation.
129pub fn kronecker_symbol(numerator: &Number, denominator: &Number) -> Option<i8> {
130    // Kronecker symbol (a/n). Defined for any pair of integers; returns None
131    // for rational inputs only. The Kronecker symbol extends Jacobi to even
132    // and negative denominators using the standard table of values, which the
133    // exact-arithmetic helper computes directly.
134    let numerator_integer = integer_or_none(numerator)?;
135    let denominator_integer = integer_or_none(denominator)?;
136    Some(integer_kronecker_symbol(
137        numerator_integer,
138        denominator_integer,
139    ))
140}
141
142fn integer_or_none(value: &Number) -> Option<&exact_integer> {
143    match value {
144        Number::Integer(integer_value) => Some(integer_value),
145        Number::Rational(_) => None,
146    }
147}
148
149fn positive_integer_or_none(value: &Number) -> Option<exact_integer> {
150    match value {
151        Number::Integer(integer_value) if integer_sign(integer_value) == Ordering::Greater => {
152            Some(integer_value.clone())
153        }
154        _ => None,
155    }
156}
