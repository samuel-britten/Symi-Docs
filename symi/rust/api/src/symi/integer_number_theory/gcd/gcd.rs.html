11/// Performs the advanced native `integer_gcd` operation.
12pub fn integer_gcd(left: &Number, right: &Number) -> Option<Number> {
13    // Greatest common divisor of two integer-valued numbers. Returns None if
14    // either operand is the rational variant, which is the rigour-required
15    // signal that gcd is only defined on the integer ring. The result is
16    // non-negative, with gcd(0, 0) = 0 by the standard convention.
17    let left_integer = integer_or_none(left)?;
18    let right_integer = integer_or_none(right)?;
19    Some(Number::Integer(integer_gcd_internal(
20        left_integer,
21        right_integer,
22    )))
23}

25/// Performs the advanced native `integer_lcm` operation.
26pub fn integer_lcm(left: &Number, right: &Number) -> Option<Number> {
27    // Least common multiple of two integer-valued numbers. Returns None if
28    // either operand is the rational variant. lcm(a, 0) = 0 by the standard
29    // convention. The result is non-negative.
30    let left_integer = integer_or_none(left)?;
31    let right_integer = integer_or_none(right)?;
32    Some(Number::Integer(integer_lcm_internal(
33        left_integer,
34        right_integer,
35    )))
36}

38/// Performs the advanced native `extended_euclidean` operation.
39pub fn extended_euclidean(left: &Number, right: &Number) -> Option<(Number, Number, Number)> {
40    // Extended Euclidean algorithm. Returns (gcd, multiplier_left,
41    // multiplier_right) such that
42    //     gcd = multiplier_left * left + multiplier_right * right
43    // with gcd non-negative. Returns None when either input is rational, since
44    // the Bezout identity is meaningful only over the integers. The extended
45    // GCD helper gives the canonical (non-negative gcd, cofactors bounded by
46    // |right|/g and |left|/g) pair documented in Bach & Shallit §4.3.
47    let left_integer = integer_or_none(left)?;
48    let right_integer = integer_or_none(right)?;
49    let (gcd_value, cofactor_left, cofactor_right) =
50        integer_extended_greatest_common_divisor(left_integer, right_integer);
51    Some((
52        Number::Integer(natural_to_integer(gcd_value)),
53        Number::Integer(cofactor_left),
54        Number::Integer(cofactor_right),
55    ))
56}

58/// Performs the advanced native `bezout_coefficients` operation.
59pub fn bezout_coefficients(
60    left: &exact_integer,
61    right: &exact_integer,
62) -> (exact_integer, exact_integer, exact_integer) {
63    // Internal helper returning the raw exact_integer-form Bezout triple used by
64    // higher-level routines that want to stay inside the exact-arithmetic integer
65    // type. Mirrors extended_euclidean but skips the number wrapping. gcd(0, 0)
66    // returns (0, 0, 0) to match the public contract.
67    if integer_is_zero(left) && integer_is_zero(right) {
68        return (integer_zero(), integer_zero(), integer_zero());
69    }
70    let (gcd_value, cofactor_left, cofactor_right) =
71        integer_extended_greatest_common_divisor(left, right);
72    (natural_to_integer(gcd_value), cofactor_left, cofactor_right)
73}
74
75fn integer_or_none(value: &Number) -> Option<&exact_integer> {
76    match value {
77        Number::Integer(integer_value) => Some(integer_value),
78        Number::Rational(_) => None,
79    }
80}

139    #[test]
140    fn extended_euclidean_returns_bezout_pair() {
141        let (gcd_value, multiplier_left, multiplier_right) =
142            extended_euclidean(&integer_number(240), &integer_number(46)).unwrap();
143        assert_eq!(gcd_value, integer_number(2));
144        // 240 * multiplier_left + 46 * multiplier_right == 2
145        let lifted_left = match multiplier_left {
146            Number::Integer(value) => value,
147            _ => panic!("multiplier must be integer"),
148        };
149        let lifted_right = match multiplier_right {
150            Number::Integer(value) => value,
151            _ => panic!("multiplier must be integer"),
152        };
153        let combination =
154            exact_integer::from(240) * lifted_left + exact_integer::from(46) * lifted_right;
155        assert_eq!(combination, exact_integer::from(2));
156    }

161            extended_euclidean(&integer_number(0), &integer_number(0)).unwrap();
162        assert_eq!(gcd_value, integer_number(0));
163        // (0, 0) yields any cofactor pair; the exact-arithmetic contract returns (0, 0).
164        match (multiplier_left, multiplier_right) {
165            (Number::Integer(left_value), Number::Integer(right_value)) => {
166                assert_eq!(left_value, integer_zero());
167                assert_eq!(right_value, integer_zero());
168            }
169            _ => panic!("multipliers must be integer"),
170        }
