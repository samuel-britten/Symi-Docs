15/// Performs the advanced native `factor_integer` operation.
16pub fn factor_integer(value: &Number) -> Option<Vec<(Number, u32)>> {
17    // Prime factorisation of a positive integer. The return is a list of
18    // (prime, multiplicity) pairs in ascending order of the prime. Returns
19    // None for rational inputs and for integers ≤ 0; factor_integer(1) returns
20    // an empty list per the empty-product convention. The pipeline is
21    //   trial division → Brent's variant of Pollard rho → ECM (Lenstra 1987)
22    // Primality at every step is checked with the
23    // primality routine so the output is rigorous: every
24    // returned factor is verified prime.
25    let integer_value = match value {
26        Number::Integer(integer_value) => integer_value,
27        Number::Rational(_) => return None,
28    };
29    if integer_sign(integer_value) != Ordering::Greater {
30        return None;
31    }
32    let factors = factor_integer_internal(integer_value)?;
33    Some(
34        factors
35            .into_iter()
36            .map(|(prime_factor, multiplicity)| (Number::Integer(prime_factor), multiplicity))
37            .collect(),
38    )
39}

41pub(crate) fn factor_integer_internal(value: &exact_integer) -> Option<Vec<(exact_integer, u32)>> {
42    // Internal exact_integer-form factorisation used by the multiplicative function
43    // tier. Mirrors the public surface but returns exact-arithmetic integer values
44    // directly and assumes the input is a positive integer.
45    if integer_sign(value) != Ordering::Greater {
46        return None;
47    }
48    if *value == integer_one() {
49        return Some(Vec::new());
50    }
51    let mut working_value = value.clone();
52    let mut factor_accumulator: Vec<(exact_integer, u32)> = Vec::new();
53    working_value = strip_trial_divisors(working_value, &mut factor_accumulator);
54    if working_value == integer_one() {
55        merge_and_sort_factors(&mut factor_accumulator);
56        return Some(factor_accumulator);
57    }
58    let mut to_split: Vec<exact_integer> = vec![working_value];
59    while let Some(current) = to_split.pop() {
60        if current == integer_one() {
61            continue;
62        }
63        if integer_is_prime(&current) {
64            push_or_combine(&mut factor_accumulator, current, 1);
65            continue;
66        }
67        let factor = match split_composite(&current) {
68            Some(found) => found,
69            None => return None,
70        };
71        let cofactor = current / factor.clone();
72        to_split.push(factor);
73        to_split.push(cofactor);
74    }
75    merge_and_sort_factors(&mut factor_accumulator);
76    Some(factor_accumulator)
77}

79fn strip_trial_divisors(
80    mut value: exact_integer,
81    factor_accumulator: &mut Vec<(exact_integer, u32)>,
82) -> exact_integer {
83    // Trial-divide value by every prime up to TRIAL_DIVISION_LIMIT, recording
84    // multiplicities into the accumulator. Returns the cofactor that remains
85    // after every small prime has been stripped. The loop covers 2 separately
86    // so the remaining sweep can step in twos.
87    let mut trial_candidate: u64 = 2;
88    while trial_candidate <= TRIAL_DIVISION_LIMIT {
89        let candidate_integer = exact_integer::from(trial_candidate);
90        if (&candidate_integer * &candidate_integer) > value {
91            break;
92        }
93        let mut multiplicity: u32 = 0;
94        while value.clone() % candidate_integer.clone() == integer_zero() {
95            value /= candidate_integer.clone();
96            multiplicity += 1;
97        }
98        if multiplicity > 0 {
99            push_or_combine(factor_accumulator, candidate_integer, multiplicity);
100        }
101        trial_candidate = if trial_candidate == 2 {
102            3
103        } else {
104            trial_candidate + 2
105        };
106    }
107    value
108}

110fn split_composite(value: &exact_integer) -> Option<exact_integer> {
111    // Find a non-trivial factor of a composite value. First tries Brent's
112    // variant of Pollard rho with a small parameter sweep, then falls back to
113    // a tiny Lenstra ECM pass for values that resist rho (typically those with
114    // very close prime factors or square structure rho's cycle detection
115    // mishandles).
116    if integer_is_perfect_square(value) {
117        let root = crate::integer_number_theory::roots::integer_square_root(&Number::Integer(
118            value.clone(),
119        ))?;
120        match root {
121            Number::Integer(integer_root) => return Some(integer_root),
122            Number::Rational(_) => return None,
123        }
124    }
125    for seed in 1..=12u64 {
126        if let Some(factor) = pollard_rho_brent(value, seed) {
127            return Some(factor);
128        }
129    }
130    lenstra_elliptic_curve_method(value)
131}

133fn pollard_rho_brent(value: &exact_integer, seed: u64) -> Option<exact_integer> {
134    // Brent (1980) improvement of Pollard rho. The iteration
135    //   x_{k+1} = x_k^2 + c (mod n)
136    // is followed via Floyd cycle detection with batched gcd accumulation. A
137    // distinct seed shifts both the starting value and the constant offset c,
138    // so the parameter sweep recovers from the rare cycle that doesn't expose
139    // a factor under one choice.
140    let constant_offset = exact_integer::from((seed as i64) % 100 + 1);
141    let mut tortoise = exact_integer::from((seed as i64 % 7) + 2);
142    let mut hare = tortoise.clone();
143    let mut accumulated_difference = integer_one();
144    let mut iteration_count: u64 = 0;
145    let iteration_cap = 1_000_000_u64;
146    let batch_size: u64 = 128;
147    loop {
148        if iteration_count >= iteration_cap {
149            return None;
150        }
151        for _ in 0..batch_size {
152            tortoise = modulo_canonical(&(&tortoise * &tortoise + &constant_offset), value);
153            hare = modulo_canonical(&(&hare * &hare + &constant_offset), value);
154            hare = modulo_canonical(&(&hare * &hare + &constant_offset), value);
155            let difference = (integer_absolute_value(&(&tortoise - &hare)) % value.clone()).clone();
156            let reduced_difference = if integer_sign(&difference) == Ordering::Less {
157                difference + value.clone()
158            } else {
159                difference
160            };
161            if reduced_difference != integer_zero() {
162                accumulated_difference =
163                    modulo_canonical(&(&accumulated_difference * &reduced_difference), value);
164            }
165        }
166        iteration_count += batch_size;
167        let candidate_factor = euclidean_gcd(&accumulated_difference, value);
168        if candidate_factor > integer_one() && &candidate_factor < value {
169            return Some(candidate_factor);
170        }
171        if candidate_factor == *value {
172            return None;
173        }
174    }
175}

177fn lenstra_elliptic_curve_method(value: &exact_integer) -> Option<exact_integer> {
178    // Single-stage ECM (Lenstra 1987). Repeatedly draws curve parameters
179    // (a, x, y) and computes B! * P for an increasing smoothness bound B,
180    // following the addition formula in projective coordinates. A non-trivial
181    // factor surfaces whenever a partial denominator inversion fails because
182    // the candidate divisor shares a factor with n.
183    let smoothness_bounds: [u64; 4] = [2_000, 11_000, 50_000, 250_000];
184    for (curve_index, smoothness_bound) in smoothness_bounds.iter().enumerate() {
185        for trial_index in 1..=8u64 {
186            let parameter_a = exact_integer::from(
187                ((curve_index as i64) * 7 + (trial_index as i64) * 11 + 6) % 997,
188            );
189            let starting_x = exact_integer::from(((trial_index as i64) * 13 + 5) % 991);
190            let starting_y = exact_integer::from(((trial_index as i64) * 17 + 11) % 983);
191            let parameter_b =
192                compute_curve_parameter_b(&parameter_a, &starting_x, &starting_y, value);
193            if let Some(factor) = run_one_ecm_curve(
194                value,
195                &parameter_a,
196                &parameter_b,
197                &starting_x,
198                &starting_y,
199                *smoothness_bound,
200            ) {
201                if factor > integer_one() && &factor < value {
202                    return Some(factor);
203                }
204            }
205        }
206    }
207    None
208}

210fn compute_curve_parameter_b(
211    parameter_a: &exact_integer,
212    starting_x: &exact_integer,
213    starting_y: &exact_integer,
214    modulus: &exact_integer,
215) -> exact_integer {
216    // The starting point (x_0, y_0) determines b via the Weierstrass curve
217    // equation y^2 = x^3 + a*x + b (mod n), which the helper rearranges to
218    //     b = y_0^2 - x_0^3 - a*x_0 (mod n).
219    let y_squared = modulo_canonical(&(starting_y * starting_y), modulus);
220    let x_cubed = modulo_canonical(&(starting_x * starting_x * starting_x), modulus);
221    let a_times_x = modulo_canonical(&(parameter_a * starting_x), modulus);
222    let combined = y_squared - x_cubed - a_times_x;
223    modulo_canonical(&combined, modulus)
224}

226fn run_one_ecm_curve(
227    modulus: &exact_integer,
228    parameter_a: &exact_integer,
229    _parameter_b: &exact_integer,
230    starting_x: &exact_integer,
231    starting_y: &exact_integer,
232    smoothness_bound: u64,
233) -> Option<exact_integer> {
234    // Compute B! * P on the curve y^2 = x^3 + a*x + b (mod n) using affine
235    // addition. A "non-invertible denominator" event surfaces a non-trivial
236    // gcd with the modulus, which is exactly the factor we seek.
237    let mut current_point = Some((starting_x.clone(), starting_y.clone()));
238    for multiplier in 2..=smoothness_bound {
239        let multiplier_integer = exact_integer::from(multiplier as i64);
240        match scalar_multiply(
241            current_point.clone(),
242            &multiplier_integer,
243            parameter_a,
244            modulus,
245        ) {
246            Ok(new_point) => current_point = new_point,
247            Err(factor) => return Some(factor),
248        }
249    }
250    None
251}

253type ECMPoint = Option<(exact_integer, exact_integer)>;

255fn scalar_multiply(
256    point: ECMPoint,
257    scalar: &exact_integer,
258    parameter_a: &exact_integer,
259    modulus: &exact_integer,
260) -> Result<ECMPoint, exact_integer> {
261    // Double-and-add scalar multiplication. Returns Err(factor) on a
262    // non-invertible denominator; that factor is a non-trivial gcd of the
263    // intended inverse's denominator with the modulus.
264    let mut accumulator: ECMPoint = None;
265    let mut current = point;
266    let mut remaining = scalar.clone();
267    while remaining > integer_zero() {
268        if &remaining % exact_integer::from(2) == integer_one() {
269            accumulator = add_points(accumulator, current.clone(), parameter_a, modulus)?;
270        }
271        remaining /= exact_integer::from(2);
272        if remaining > integer_zero() {
273            current = add_points(current.clone(), current, parameter_a, modulus)?;
274        }
275    }
276    Ok(accumulator)
277}

279fn add_points(
280    left_point: ECMPoint,
281    right_point: ECMPoint,
282    parameter_a: &exact_integer,
283    modulus: &exact_integer,
284) -> Result<ECMPoint, exact_integer> {
285    // Affine addition with side-channel reporting of any non-invertible
286    // denominator encountered during the slope calculation. The "infinity"
287    // point is represented as None.
288    let left_unpacked = match left_point {
289        Some(value) => value,
290        None => return Ok(right_point),
291    };
292    let right_unpacked = match right_point {
293        Some(value) => value,
294        None => return Ok(Some(left_unpacked)),
295    };
296    if left_unpacked.0 == right_unpacked.0 {
297        if modulo_canonical(&(&left_unpacked.1 + &right_unpacked.1), modulus) == integer_zero() {
298            return Ok(None);
299        }
300        return double_point(left_unpacked, parameter_a, modulus);
301    }
302    let denominator = modulo_canonical(&(&right_unpacked.0 - &left_unpacked.0), modulus);
303    let denominator_inverse =
304        match crate::polynomial::finite_field::integer_mod_inverse(&denominator, modulus) {
305            Some(inverse) => inverse,
306            None => {
307                let factor = euclidean_gcd(&denominator, modulus);
308                if factor > integer_one() && &factor < modulus {
309                    return Err(factor);
310                }
311                return Ok(None);
312            }
313        };
314    let numerator = modulo_canonical(&(&right_unpacked.1 - &left_unpacked.1), modulus);
315    let slope = modulo_canonical(&(&numerator * &denominator_inverse), modulus);
316    let new_x_raw = &slope * &slope - &left_unpacked.0 - &right_unpacked.0;
317    let new_x = modulo_canonical(&new_x_raw, modulus);
318    let new_y_raw = &slope * (&left_unpacked.0 - &new_x) - &left_unpacked.1;
319    let new_y = modulo_canonical(&new_y_raw, modulus);
320    Ok(Some((new_x, new_y)))
321}

323fn double_point(
324    point: (exact_integer, exact_integer),
325    parameter_a: &exact_integer,
326    modulus: &exact_integer,
327) -> Result<ECMPoint, exact_integer> {
328    // Special case of point addition where the two summands are equal. Uses
329    // the tangent-line slope (3x^2 + a) / (2y) (mod modulus).
330    let two_y = modulo_canonical(&(&point.1 * exact_integer::from(2)), modulus);
331    if two_y == integer_zero() {
332        return Ok(None);
333    }
334    let two_y_inverse = match crate::polynomial::finite_field::integer_mod_inverse(&two_y, modulus)
335    {
336        Some(inverse) => inverse,
337        None => {
338            let factor = euclidean_gcd(&two_y, modulus);
339            if factor > integer_one() && &factor < modulus {
340                return Err(factor);
341            }
342            return Ok(None);
343        }
344    };
345    let numerator = modulo_canonical(
346        &(&point.0 * &point.0 * exact_integer::from(3) + parameter_a),
347        modulus,
348    );
349    let slope = modulo_canonical(&(&numerator * &two_y_inverse), modulus);
350    let new_x_raw = &slope * &slope - exact_integer::from(2) * &point.0;
351    let new_x = modulo_canonical(&new_x_raw, modulus);
352    let new_y_raw = &slope * (&point.0 - &new_x) - &point.1;
353    let new_y = modulo_canonical(&new_y_raw, modulus);
354    Ok(Some((new_x, new_y)))
355}

357fn euclidean_gcd(left: &exact_integer, right: &exact_integer) -> exact_integer {
358    // Local Euclidean gcd. Free-standing to avoid the dependency on
359    // polynomial::gcd from inside the integer_number_theory module.
360    let mut a = integer_absolute_value(left);
361    let mut b = integer_absolute_value(right);
362    while b != integer_zero() {
363        let raw = a % b.clone();
364        a = b;
365        b = if integer_sign(&raw) == Ordering::Less {
366            raw + a.clone()
367        } else {
368            raw
369        };
370    }
371    a
372}

374fn push_or_combine(
375    accumulator: &mut Vec<(exact_integer, u32)>,
376    prime_factor: exact_integer,
377    multiplicity_to_add: u32,
378) {
379    // Append a prime factor to the accumulator, combining with any existing
380    // entry of the same prime by summing multiplicities. Keeps the
381    // multiplicity invariant before the final sort step canonicalises order.
382    for entry in accumulator.iter_mut() {
383        if entry.0 == prime_factor {
384            entry.1 += multiplicity_to_add;
385            return;
386        }
387    }
388    accumulator.push((prime_factor, multiplicity_to_add));
389}
390
391fn merge_and_sort_factors(accumulator: &mut Vec<(exact_integer, u32)>) {
392    // Sort the factor list in ascending order of the prime. Multiplicities
393    // were already combined by push_or_combine.
394    accumulator.sort_by(|left, right| left.0.cmp(&right.0));
395}

397#[allow(dead_code)]
398fn assert_factor_round_trips(value: &exact_integer, factors: &[(exact_integer, u32)]) {
399    // Sanity helper for use in tests: reconstruct the input by multiplying
400    // every prime^multiplicity together and assert structural equality. Not
401    // called from production code; kept around for the test surface.
402    let mut product = integer_one();
403    for (prime_factor, multiplicity) in factors.iter() {
404        for _ in 0..*multiplicity {
405            product *= prime_factor.clone();
406        }
407    }
408    assert_eq!(&product, value);
409}
