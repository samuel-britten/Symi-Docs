17/// How much of the assumption engine one definedness judgment may consult.
18#[derive(Debug, Clone, Copy, PartialEq, Eq)]
19pub(crate) enum DefinednessMode {
20    /// Structural facts only. The bounded model search calls the predicate handlers
21    /// once per rule-reachable atom, so a handler that reaches back into the
22    /// knowledge base would both re-enter it and multiply every query's cost.
23    Structural,
24    /// The full query surface. Reachable only from the public `is_defined` entry,
25    /// which is never called from inside a model search.
26    Deep,
27}

40fn declared_defined_verdict(
41    mode: DefinednessMode,
42    reference: &ExpressionReference,
43    table: &SymbolTable,
44) -> TruthValue {
45    // Definedness read from the assumption scope rather than from the
46    // structure. Every scalar predicate entails `defined`, so a subject
47    // carrying any admitted fact answers here even when its universe is

49    // it would re-enter is this one.

52        DefinednessMode::Deep => queries::is_defined(reference, table),
53    }

189/// Decide pointwise definedness and retain the derivation that established it.
190pub(crate) fn judge_definedness(
191    reference: &ExpressionReference,
192    table: &SymbolTable,
193    mode: DefinednessMode,
194    remaining_depth: usize,
195) -> AnalyticJudgment {
196    // The deterministic ladder for one pointwise goal: exact leaves, then
197    // arithmetic shapes with their extended-value obligations, then registered
198    // built-in point domains. Exhausting the ceiling yields unknown rather than
199    // a guessed verdict, and every non-scalar node kind stays unknown so generic
200    // finite-complex closure is never applied to a set, relation, distribution,
201    // or deferred operation.
202    if remaining_depth == 0 {
203        return AnalyticJudgment::opaque(TruthValue::Unknown);
204    }
205    use ExpressionKind::*;
206    match &reference.kind {
207        Integer(_) | Rational(_) | PolynomialRoot(_) => AnalyticJudgment::pointwise(
208            TruthValue::Yes,
209            AnalyticEvidence::ExactLeafDefinedness {
210                state: PointwiseValueState::DefinedFinite,
211            },
212        ),
213        Constant(value) => constant_judgment(value),
214        Symbol(_) => AnalyticJudgment::opaque(TruthValue::Unknown),
215        Sum(children) => additive_judgment(children, table, mode, remaining_depth),
216        Product(children) => multiplicative_judgment(children, table, mode, remaining_depth),
217        Power(base, exponent) => power_judgment(base, exponent, table, mode, remaining_depth),
218        Factorial(argument) => factorial_judgment(argument, table, mode, remaining_depth),
219        FunctionCall(FunctionIdentifier::Builtin(function), arguments) => {
220            function_call_judgment(*function, arguments, table, mode, remaining_depth)
221        }
222        FunctionCall(FunctionIdentifier::User(_), _) => {
223            AnalyticJudgment::opaque(TruthValue::Unknown)
224        }
225        Derivative(_, _, _)
226        | Integral(_, _, _, _)
227        | Summation(_, _, _, _)
228        | IndexedProduct(_, _, _, _)
229        | LaplaceTransform(_, _, _)
230        | InverseLaplaceTransform(_, _, _)
231        | FourierTransform(_, _, _)
232        | InverseFourierTransform(_, _, _)
233        | Piecewise(_)
234        | OrdinaryDifferentialEquation(_, _, _)
235        | RecurrenceEquation(_, _, _)
236        | Interval(_, _, _, _)
237        | FiniteSet(_)
238        | SetUnion(_)
239        | SetIntersection(_)
240        | SetComplement(_, _)
241        | ImageSet(_, _, _)
242        | ConditionSet(_, _, _)
243        | Distribution(_, _)
244        | RandomVariable(_, _)
245        | SpecialFunctionParameterVector(_) => AnalyticJudgment::opaque(TruthValue::Unknown),
246    }
247}

249fn constant_judgment(value: &BuiltinConstant) -> AnalyticJudgment {
250    // The exact leaf table of plan section 8.1. The signed infinities are defined
251    // extended-real values but not finite ones; the undefined sentinel is the
252    // proved negation; every non-scalar constant declines instead of being closed
253    // under scalar rules.
254    let state = match value {
255        BuiltinConstant::Pi
256        | BuiltinConstant::E
257        | BuiltinConstant::EulerGamma
258        | BuiltinConstant::ImaginaryUnit => PointwiseValueState::DefinedFinite,
259        BuiltinConstant::Infinity => PointwiseValueState::DefinedPositiveInfinity,
260        BuiltinConstant::NegativeInfinity => PointwiseValueState::DefinedNegativeInfinity,
261        BuiltinConstant::Undefined => PointwiseValueState::Undefined,
262        BuiltinConstant::EmptySet
263        | BuiltinConstant::RealLine
264        | BuiltinConstant::ComplexPlane
265        | BuiltinConstant::IntegerSet
266        | BuiltinConstant::LogicalTrue
267        | BuiltinConstant::LogicalFalse => PointwiseValueState::Unknown,
268    };
269    match state {
270        PointwiseValueState::Unknown => AnalyticJudgment::opaque(TruthValue::Unknown),
271        PointwiseValueState::Undefined => AnalyticJudgment::pointwise(
272            TruthValue::No,
273            AnalyticEvidence::ExactLeafDefinedness { state },
274        ),
275        state => AnalyticJudgment::pointwise(
276            TruthValue::Yes,
277            AnalyticEvidence::ExactLeafDefinedness { state },
278        ),
279    }
280}

282/// How one operand of an arithmetic node sits in the extended-real universe.
283#[derive(Debug, Clone, Copy, PartialEq, Eq)]
284enum OperandExtent {
285    /// The operand is proved finite.
286    Finite,
287    /// The operand is proved to be positive infinity.
288    PositiveInfinity,
289    /// The operand is proved to be negative infinity.
290    NegativeInfinity,
291    /// The operand's extent is not proved.
292    Unproved,
293}
294
295fn operand_extent_of(
296    operand: &ExpressionReference,
297    table: &SymbolTable,
298    mode: DefinednessMode,
299) -> OperandExtent {
300    // Place one operand on the extended-real line. An infinity must carry a
301    // proved sign, because the indeterminate-form tables are stated in terms of
302    // signs and an unsigned infinity has no place in this module's universe.
303    if matches!(finite_verdict(mode, operand, table), TruthValue::Yes) {
304        return OperandExtent::Finite;
305    }
306    match signed_infinity(operand) {
307        Some(InfinitySign::Positive) => return OperandExtent::PositiveInfinity,
308        Some(InfinitySign::Negative) => return OperandExtent::NegativeInfinity,
309        None => {}
310    }
311    if !matches!(infinite_verdict(mode, operand, table), TruthValue::Yes) {
312        return OperandExtent::Unproved;
313    }
314    if matches!(positive_verdict(mode, operand, table), TruthValue::Yes) {
315        return OperandExtent::PositiveInfinity;
316    }
317    if matches!(negative_verdict(mode, operand, table), TruthValue::Yes) {
318        return OperandExtent::NegativeInfinity;
319    }
320    OperandExtent::Unproved
321}

323fn additive_judgment(
324    children: &[ExpressionReference],
325    table: &SymbolTable,
326    mode: DefinednessMode,
327    remaining_depth: usize,
328) -> AnalyticJudgment {
329    // A sum denotes a value when every operand does, every operand sits on the
330    // extended-real line, and the infinite operands do not mix signs. Mixed signs
331    // are the sole additive indeterminate form.
332    let Some(parts) = combination_parts(children, table, mode, remaining_depth) else {
333        return AnalyticJudgment::opaque(TruthValue::Unknown);
334    };
335    let CombinationParts {
336        derivations,
337        obligations,
338        extents,
339        undefined_operand,
340    } = parts;
341    if undefined_operand {
342        return undefined_combination();
343    }
344    let saw_positive = extents.contains(&OperandExtent::PositiveInfinity);
345    let saw_negative = extents.contains(&OperandExtent::NegativeInfinity);
346    if saw_positive && saw_negative {
347        return AnalyticJudgment::pointwise(
348            TruthValue::No,
349            AnalyticEvidence::ExceptionalPoint { function: None },
350        );
351    }
352    AnalyticJudgment::pointwise(
353        TruthValue::Yes,
354        AnalyticEvidence::ArithmeticDefinedness {
355            children: derivations,
356            side_conditions: obligations,
357        },
358    )
359}

361fn multiplicative_judgment(
362    children: &[ExpressionReference],
363    table: &SymbolTable,
364    mode: DefinednessMode,
365    remaining_depth: usize,
366) -> AnalyticJudgment {
367    // A product denotes a value when every operand does, every operand sits on
368    // the extended-real line, and no zero is paired with an infinity. Zero times
369    // infinity is the sole multiplicative indeterminate form.
370    let Some(parts) = combination_parts(children, table, mode, remaining_depth) else {
371        return AnalyticJudgment::opaque(TruthValue::Unknown);
372    };
373    let CombinationParts {
374        derivations,
375        mut obligations,
376        extents,
377        undefined_operand,
378    } = parts;
379    if undefined_operand {
380        return undefined_combination();
381    }
382    if !extents.iter().any(|extent| {
383        matches!(
384            extent,
385            OperandExtent::PositiveInfinity | OperandExtent::NegativeInfinity
386        )
387    }) {
388        return AnalyticJudgment::pointwise(
389            TruthValue::Yes,
390            AnalyticEvidence::ArithmeticDefinedness {
391                children: derivations,
392                side_conditions: obligations,
393            },
394        );
395    }
396    for (child, extent) in children.iter().zip(extents.iter()) {
397        if !matches!(extent, OperandExtent::Finite) {
398            continue;
399        }
400        if matches!(zero_verdict(mode, child, table), TruthValue::Yes) {
401            return AnalyticJudgment::pointwise(
402                TruthValue::No,
403                AnalyticEvidence::ExceptionalPoint { function: None },
404            );
405        }
406        if !matches!(nonzero_verdict(mode, child, table), TruthValue::Yes) {
407            return AnalyticJudgment::opaque(TruthValue::Unknown);
408        }
409        obligations.push(SideCondition::Nonzero(child.clone()));
410    }
411    AnalyticJudgment::pointwise(
412        TruthValue::Yes,
413        AnalyticEvidence::ArithmeticDefinedness {
414            children: derivations,
415            side_conditions: obligations,
416        },
417    )
418}

420struct CombinationParts {
421    derivations: Vec<AnalyticEvidence>,
422    obligations: Vec<SideCondition>,
423    extents: Vec<OperandExtent>,
424    undefined_operand: bool,
425}
426
427fn combination_parts(
428    children: &[ExpressionReference],
429    table: &SymbolTable,
430    mode: DefinednessMode,
431    remaining_depth: usize,
432) -> Option<CombinationParts> {
433    // Judge every operand once and record its derivation, its definedness
434    // obligation and its extended-real extent. Returns None as soon as any
435    // operand is unproved, because an arithmetic node cannot denote a value
436    // through an operand that may not.
437    let mut derivations = Vec::with_capacity(children.len());
438    let mut obligations = Vec::with_capacity(children.len());
439    let mut extents = Vec::with_capacity(children.len());
440    for child in children {
441        let child_judgment = judge_definedness(child, table, mode, remaining_depth - 1);
442        match child_judgment.verdict {
443            TruthValue::No => {
444                return Some(CombinationParts {
445                    derivations: Vec::new(),
446                    obligations: Vec::new(),
447                    extents: Vec::new(),
448                    undefined_operand: true,
449                });
450            }
451            TruthValue::Unknown => return None,
452            TruthValue::Yes => {}
453        }
454        let extent = operand_extent_of(child, table, mode);
455        if matches!(extent, OperandExtent::Unproved) {
456            return None;
457        }
458        extents.push(extent);
459        obligations.push(SideCondition::Defined(child.clone()));
460        if let Some(evidence) = child_judgment.evidence {
461            derivations.push(evidence);
462        }
463    }
464    Some(CombinationParts {
465        derivations,
466        obligations,
467        extents,
468        undefined_operand: false,
469    })
470}

482fn power_judgment(
483    base: &ExpressionReference,
484    exponent: &ExpressionReference,
485    table: &SymbolTable,
486    mode: DefinednessMode,
487    remaining_depth: usize,
488) -> AnalyticJudgment {
489    // The power obligations of plan section 4.2, kept separate so an integer
490    // theorem is never replayed as a principal-power theorem. A negative integer
491    // exponent needs a nonzero finite base; a nonnegative integer exponent needs
492    // only a finite base under the project's exact zero-to-the-zero policy; a
493    // general principal power follows exp(exponent * log(base)) and therefore
494    // needs a nonzero finite base and a finite exponent.
495    let base_judgment = judge_definedness(base, table, mode, remaining_depth - 1);
496    let exponent_judgment = judge_definedness(exponent, table, mode, remaining_depth - 1);
497    if matches!(base_judgment.verdict, TruthValue::No)
498        || matches!(exponent_judgment.verdict, TruthValue::No)
499    {
500        return AnalyticJudgment::pointwise(
501            TruthValue::No,
502            AnalyticEvidence::ArithmeticDefinedness {
503                children: Vec::new(),
504                side_conditions: Vec::new(),
505            },
506        );
507    }
508    if let Some(negative_exponent) = literal_exponent_sign(exponent) {
509        if literal_exponent_is_zero(exponent) {
510            // The zero exponent is total only away from a zero and away from an
511            // infinite base; both exclusions live in the exceptional table below.
512            return zero_exponent_judgment(base, table, mode);
513        }
514        let mut obligations = vec![SideCondition::Defined(base.clone())];
515        if negative_exponent {
516            if matches!(zero_verdict(mode, base, table), TruthValue::Yes) {
517                return AnalyticJudgment::pointwise(
518                    TruthValue::No,
519                    AnalyticEvidence::ExceptionalPoint { function: None },
520                );
521            }
522            obligations.push(SideCondition::Nonzero(base.clone()));
523        }
524        // An integer power needs a defined base, not a finite one: a signed
525        // infinity raised to a positive integer stays a defined extended value,
526        // and its reciprocal is the defined value zero. Finiteness of the
527        // result is a separate question that `finite` answers on its own.
528        let base_is_defined = matches!(base_judgment.verdict, TruthValue::Yes)
529            || matches!(finite_verdict(mode, base, table), TruthValue::Yes)
530            || matches!(declared_defined_verdict(mode, base, table), TruthValue::Yes);
531        let proved = base_is_defined
532            && (!negative_exponent
533                || matches!(nonzero_verdict(mode, base, table), TruthValue::Yes));
534        if !proved {
535            return AnalyticJudgment::opaque(TruthValue::Unknown);
536        }
537        // A positive exact exponent keeps a zero base defined, so no nonzero
538        // obligation is recorded for it.
539        let children = base_judgment.evidence.into_iter().collect();
540        return AnalyticJudgment::pointwise(
541            TruthValue::Yes,
542            AnalyticEvidence::ArithmeticDefinedness {
543                children,
544                side_conditions: obligations,
545            },
546        );
547    }
548    if matches!(zero_verdict(mode, base, table), TruthValue::Yes) {
549        return zero_base_principal_power_judgment(exponent, table, mode);
550    }
551    let proved = matches!(finite_verdict(mode, base, table), TruthValue::Yes)
552        && matches!(nonzero_verdict(mode, base, table), TruthValue::Yes)
553        && matches!(finite_verdict(mode, exponent, table), TruthValue::Yes);
554    if !proved {
555        return AnalyticJudgment::opaque(TruthValue::Unknown);
556    }
557    let children = base_judgment
558        .evidence
559        .into_iter()
560        .chain(exponent_judgment.evidence)
561        .collect();
562    AnalyticJudgment::pointwise(
563        TruthValue::Yes,
564        AnalyticEvidence::ArithmeticDefinedness {
565            children,
566            side_conditions: vec![
567                SideCondition::Finite(base.clone()),
568                SideCondition::Nonzero(base.clone()),
569                SideCondition::Finite(exponent.clone()),
570            ],
571        },
572    )
573}

575fn literal_exponent_is_zero(exponent: &ExpressionReference) -> bool {
576    // Only an exact integer literal can be the zero exponent; a rational literal
577    // is normalized away from zero by construction.
578    matches!(&exponent.kind, ExpressionKind::Integer(value) if integer_is_zero(value))
579}

581fn literal_exponent_sign(exponent: &ExpressionReference) -> Option<bool> {
582    // Report whether an exact numeric exponent is negative. Integer and rational
583    // literals share one obligation table: a positive real exponent keeps a zero
584    // base defined because the principal power of zero is zero, while a negative
585    // one does not. Every other exponent shape falls through to the general
586    // principal-power rule.
587    match &exponent.kind {
588        ExpressionKind::Integer(value) => Some(integer_is_negative(value)),
589        ExpressionKind::Rational(value) => {
590            Some(crate::exact_arithmetic::rational_is_negative(value))
591        }
592        _ => None,
593    }
594}

596fn zero_exponent_judgment(
597    base: &ExpressionReference,
598    table: &SymbolTable,
599    mode: DefinednessMode,
600) -> AnalyticJudgment {
601    // Zero to the zero and infinity to the zero are both indeterminate; every
602    // other defined base gives one.
603    if matches!(zero_verdict(mode, base, table), TruthValue::Yes) || signed_infinity(base).is_some()
604    {
605        return AnalyticJudgment::pointwise(
606            TruthValue::No,
607            AnalyticEvidence::ExceptionalPoint { function: None },
608        );
609    }
610    if !matches!(finite_verdict(mode, base, table), TruthValue::Yes) {
611        return AnalyticJudgment::opaque(TruthValue::Unknown);
612    }
613    AnalyticJudgment::pointwise(
614        TruthValue::Yes,
615        AnalyticEvidence::ArithmeticDefinedness {
616            children: Vec::new(),
617            side_conditions: vec![SideCondition::Finite(base.clone())],
618        },
619    )
620}

622fn zero_base_principal_power_judgment(
623    exponent: &ExpressionReference,
624    table: &SymbolTable,
625    mode: DefinednessMode,
626) -> AnalyticJudgment {
627    // Zero raised to a strictly positive real power is zero; to a zero or
628    // negative real power it is undefined, and never a signed infinity. A complex
629    // or unproved exponent stays unknown.
630    if matches!(positive_verdict(mode, exponent, table), TruthValue::Yes) {
631        return AnalyticJudgment::pointwise(
632            TruthValue::Yes,
633            AnalyticEvidence::ArithmeticDefinedness {
634                children: Vec::new(),
635                side_conditions: vec![SideCondition::Positive(exponent.clone())],
636            },
637        );
638    }
639    if matches!(negative_verdict(mode, exponent, table), TruthValue::Yes)
640        || matches!(zero_verdict(mode, exponent, table), TruthValue::Yes)
641    {
642        return AnalyticJudgment::pointwise(
643            TruthValue::No,
644            AnalyticEvidence::ExceptionalPoint { function: None },
645        );
646    }
647    AnalyticJudgment::opaque(TruthValue::Unknown)
648}

650fn factorial_judgment(
651    argument: &ExpressionReference,
652    table: &SymbolTable,
653    mode: DefinednessMode,
654    remaining_depth: usize,
655) -> AnalyticJudgment {
656    // Factorial semantics stay owned elsewhere. This module adds only the pointwise
657    // reading: a nonnegative integer argument denotes a value, a negative integer
658    // argument denotes a pole of the gamma continuation, and everything else is
659    // left to B.
660    let argument_judgment = judge_definedness(argument, table, mode, remaining_depth - 1);
661    if matches!(argument_judgment.verdict, TruthValue::No) {
662        return AnalyticJudgment::pointwise(
663            TruthValue::No,
664            AnalyticEvidence::ArithmeticDefinedness {
665                children: Vec::new(),
666                side_conditions: Vec::new(),
667            },
668        );
669    }
670    if matches!(integer_verdict(mode, argument, table), TruthValue::Yes) {
671        if matches!(negative_verdict(mode, argument, table), TruthValue::Yes) {
672            return AnalyticJudgment::pointwise(
673                TruthValue::No,
674                AnalyticEvidence::ExceptionalPoint { function: None },
675            );
676        }
677        if matches!(positive_verdict(mode, argument, table), TruthValue::Yes)
678            || matches!(zero_verdict(mode, argument, table), TruthValue::Yes)
679        {
680            return AnalyticJudgment::pointwise(
681                TruthValue::Yes,
682                AnalyticEvidence::ArithmeticDefinedness {
683                    children: Vec::new(),
684                    side_conditions: vec![
685                        SideCondition::Integer(argument.clone()),
686                        SideCondition::Finite(argument.clone()),
687                    ],
688                },
689            );
690        }
691    }
692    AnalyticJudgment::opaque(TruthValue::Unknown)
693}

695fn function_call_judgment(
696    function: BuiltinFunction,
697    arguments: &[ExpressionReference],
698    table: &SymbolTable,
699    mode: DefinednessMode,
700    remaining_depth: usize,
701) -> AnalyticJudgment {
702    // A call denotes a value only through an exact exceptional rule, a total-domain
703    // theorem with defined arguments, or a domain formula proved for its arguments.
704    // Availability of a numeric evaluator is never a domain proof, and a built-in
705    // absent from the authoritative specification has no admitted rule at all.
706    let Some(specification) = specification_of(function) else {
707        return AnalyticJudgment::opaque(TruthValue::Unknown);
708    };
709    if arguments.len() != specification.arity {
710        return AnalyticJudgment::opaque(TruthValue::Unknown);
711    }
712    if specification.call_shape != CallShape::UnaryAnalyticArgument {
713        return parameterized_call_judgment(
714            function,
715            specification.call_shape,
716            arguments,
717            table,
718            mode,
719            remaining_depth,
720        );
721    }
722    let argument = &arguments[specification.call_shape.analytic_argument_index()];
723    let argument_judgment = judge_definedness(argument, table, mode, remaining_depth - 1);
724    if matches!(argument_judgment.verdict, TruthValue::No) {
725        return AnalyticJudgment::pointwise(
726            TruthValue::No,
727            AnalyticEvidence::FunctionDomain {
728                function,
729                arguments: Vec::new(),
730                domain_obligations: Vec::new(),
731            },
732        );
733    }
734    if argument_is_a_registered_pole(specification.poles, argument, table, mode) {
735        return AnalyticJudgment::pointwise(
736            TruthValue::No,
737            AnalyticEvidence::ExceptionalPoint {
738                function: Some(function),
739            },
740        );
741    }
742    let mut obligations = vec![SideCondition::Finite(argument.clone())];
743    let mut proved = matches!(finite_verdict(mode, argument, table), TruthValue::Yes);
744    match specification.point_domain {
745        PointDomainRule::TotalOnFiniteComplex => {}
746        PointDomainRule::NonzeroFiniteComplex => {
747            obligations.push(SideCondition::Nonzero(argument.clone()));
748            proved = proved && matches!(nonzero_verdict(mode, argument, table), TruthValue::Yes);
749        }
750        PointDomainRule::TotalOnFiniteReal => {
751            obligations.push(SideCondition::Defined(argument.clone()));
752            proved = proved && matches!(real_verdict(mode, argument, table), TruthValue::Yes);
753        }
754        PointDomainRule::FiniteComplexOffPoles => {
755            // Positive definedness needs the whole pole set excluded, which the
756            // regional layer owns. Only the negative verdict above is
757            // available here.
758            return AnalyticJudgment::opaque(TruthValue::Unknown);
759        }
760        PointDomainRule::Unsupported => return AnalyticJudgment::opaque(TruthValue::Unknown),
761    }
762    if !proved {
763        return AnalyticJudgment::opaque(TruthValue::Unknown);
764    }
765    AnalyticJudgment::pointwise(
766        TruthValue::Yes,
767        AnalyticEvidence::FunctionDomain {
768            function,
769            arguments: argument_judgment.evidence.into_iter().collect(),
770            domain_obligations: obligations,
771        },
772    )
773}

775fn parameterized_call_judgment(
776    function: BuiltinFunction,
777    shape: CallShape,
778    arguments: &[ExpressionReference],
779    table: &SymbolTable,
780    mode: DefinednessMode,
781    remaining_depth: usize,
782) -> AnalyticJudgment {
783    // The generic full-call definedness path for a built-in whose analytic
784    // variable sits behind ordered parameter groups. Nothing here is specific
785    // to one family: the shape says which leading slots hold groups and which
786    // scalar carries the analytic variable, and the judgment is assembled from
787    // the whole call rather than from argument zero.
788    //
789    // A call is undefined as soon as any parameter or the argument is itself
790    // undefined, which is the one negative verdict available from structure
791    // alone. The positive direction needs the family's parameter conditions —
792    // a lower Pochhammer denominator that never vanishes, a Meijer-G contour
793    // that exists — and those are not settled from definedness of the parts,
794    // so a call whose pieces are all defined answers unknown rather than yes.
795    if matches!(
796        crate::assumptions::function_specification::full_call_domain_classification(
797            function, arguments,
798        ),
799        crate::assumptions::function_specification::FullCallDomainClassification::Undefined
800    ) {
801        return AnalyticJudgment::pointwise(
802            TruthValue::No,
803            AnalyticEvidence::ExceptionalPoint {
804                function: Some(function),
805            },
806        );
807    }
808    let group_count = shape.parameter_group_count();
809    for slot in 0..group_count {
810        let ExpressionKind::SpecialFunctionParameterVector(parameters) = &arguments[slot].kind
811        else {
812            return AnalyticJudgment::opaque(TruthValue::Unknown);
813        };
814        for parameter in parameters {
815            let judgment = judge_definedness(parameter, table, mode, remaining_depth - 1);
816            if matches!(judgment.verdict, TruthValue::No) {
817                return AnalyticJudgment::pointwise(
818                    TruthValue::No,
819                    AnalyticEvidence::FunctionDomain {
820                        function,
821                        arguments: Vec::new(),
822                        domain_obligations: Vec::new(),
823                    },
824                );
825            }
826        }
827    }
828    // Scalar-parameterized heads still have several scalar slots. Every slot
829    // participates in a call, so an undefined characteristic or parameter
830    // makes the whole call undefined just as an undefined analytic argument
831    // does. The former code examined only the declared analytic slot.
832    let mut scalar_argument_judgments = Vec::with_capacity(arguments.len());
833    for argument in arguments {
834        let judgment = judge_definedness(argument, table, mode, remaining_depth - 1);
835        if matches!(judgment.verdict, TruthValue::No) {
836            return AnalyticJudgment::pointwise(
837                TruthValue::No,
838                AnalyticEvidence::FunctionDomain {
839                    function,
840                    arguments: Vec::new(),
841                    domain_obligations: Vec::new(),
842                },
843            );
844        }
845        scalar_argument_judgments.push(judgment);
846    }
847    let argument = &arguments[shape.analytic_argument_index()];
848    // The positive direction needs the family's parameter conditions, and the
849    // registry obtains them through the declarative full-call classifier rather
850    // than by naming a family here. A call classified entire is defined wherever
851    // its argument is finite: there is no parameter configuration left that
852    // could put a Pochhammer denominator on zero, and no region of the finite
853    // plane where the value fails to exist. Any other class leaves the question
854    // open and answers unknown, never yes.
855    if !matches!(
856        crate::assumptions::function_specification::full_call_analytic_class(
857            function, shape, arguments, table
858        ),
859        crate::assumptions::function_specification::AnalyticClass::Entire
860    ) {
861        return AnalyticJudgment::opaque(TruthValue::Unknown);
862    }
863    if !matches!(finite_verdict(mode, argument, table), TruthValue::Yes) {
864        return AnalyticJudgment::opaque(TruthValue::Unknown);
865    }
866    AnalyticJudgment::pointwise(
867        TruthValue::Yes,
868        AnalyticEvidence::FunctionDomain {
869            function,
870            arguments: scalar_argument_judgments
871                .into_iter()
872                .filter_map(|judgment| judgment.evidence)
873                .collect(),
874            domain_obligations: vec![SideCondition::Finite(argument.clone())],
875        },
876    )
877}

879fn argument_is_a_registered_pole(
880    source: PoleSource,
881    argument: &ExpressionReference,
882    table: &SymbolTable,
883    mode: DefinednessMode,
884) -> bool {
885    // Decide whether the argument provably lands on the registered pole set.
886    // Only exactly decided membership counts; an argument merely not proved to
887    // avoid the set is never classified as a pole.
888    match source {
889        PoleSource::ZeroArgument => {
890            matches!(zero_verdict(mode, argument, table), TruthValue::Yes)
891        }
892        PoleSource::NonpositiveIntegerArgument => {
893            matches!(integer_verdict(mode, argument, table), TruthValue::Yes)
894                && !matches!(positive_verdict(mode, argument, table), TruthValue::Yes)
895                && (matches!(negative_verdict(mode, argument, table), TruthValue::Yes)
896                    || matches!(zero_verdict(mode, argument, table), TruthValue::Yes))
897        }
898        PoleSource::ArgumentEqualToOne => {
899            matches!(&argument.kind, ExpressionKind::Integer(value) if *value == 1.into())
900        }
901        PoleSource::None
902        | PoleSource::CosineZeros
903        | PoleSource::SineZeros
904        | PoleSource::HyperbolicCosineZeros
905        | PoleSource::HyperbolicSineZeros
906        | PoleSource::Unsupported => false,
907    }
908}
