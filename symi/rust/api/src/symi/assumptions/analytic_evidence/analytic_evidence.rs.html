11/// The scope a piece of analytic evidence was established in.
12#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
13pub(crate) enum EvidenceKind {
14    /// The evidence holds at the active point only.
15    Pointwise,
16    /// The evidence holds throughout a real interval.
17    RealInterval,
18    /// The evidence holds on a complex neighborhood.
19    ComplexNeighborhood,
20    /// The evidence holds along a path.
21    Path,
22    /// The evidence holds eventually along one directed germ.
23    EventualGerm,
24}

26/// The pointwise semantic value state of a scalar expression.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
28pub(crate) enum PointwiseValueState {
29    /// The expression denotes a finite complex value.
30    DefinedFinite,
31    /// The expression denotes positive infinity in the extended-real universe.
32    DefinedPositiveInfinity,

34    DefinedNegativeInfinity,

36    Undefined,

38    Unknown,

58/// A checked derivation of a pointwise definedness verdict.
59#[derive(Debug, Clone, PartialEq, Eq)]
60pub(crate) enum AnalyticEvidence {
61    /// The subject is an exact leaf whose value state is fixed by the expression model.
62    ExactLeafDefinedness {
63        /// The proved value state.
64        state: PointwiseValueState,
65    },
66    /// An explicit assumption or its universe closure supplies the verdict.
67    ExplicitDefinedFact,
68    /// An arithmetic node inherits definedness from its children under side conditions.
69    ArithmeticDefinedness {
70        /// The child derivations, in operand order.
71        children: Vec<AnalyticEvidence>,
72        /// The obligations the arithmetic rule additionally requires.
73        side_conditions: Vec<SideCondition>,
74    },
75    /// A built-in call is discharged by its point-domain rule.
76    FunctionDomain {
77        /// The built-in whose point domain was used.
78        function: BuiltinFunction,
79        /// The argument derivations, in argument order.
80        arguments: Vec<AnalyticEvidence>,
81        /// The obligations the point-domain rule requires.
82        domain_obligations: Vec<SideCondition>,
83    },
84    /// The subject is proved undefined by an exact exceptional point.
85    ExceptionalPoint {
86        /// The built-in owning the exceptional point, when the subject is a call.
87        function: Option<BuiltinFunction>,
88    },
89    /// A checked exact range certificate.
90    RangeInference {
91        /// The range evidence.
92        evidence: crate::assumptions::range_evidence::ScalarRangeEvidence,
93    },
94    /// A composition of regional continuity certificates.
95    ContinuityComposition {
96        /// The child certificates.
97        children: Vec<AnalyticEvidence>,
98    },
99    /// A composition of complex holomorphy certificates.
100    HolomorphicComposition {
101        /// The child certificates.
102        children: Vec<AnalyticEvidence>,
103    },
104    /// A composition of meromorphy certificates.
105    MeromorphicComposition {
106        /// The child certificates.
107        children: Vec<AnalyticEvidence>,
108    },
109    /// A checked pole exclusion for one regional goal.
110    PoleExclusion {
111        /// The specified region.
112        region: crate::assumptions::analytic_goal::AnalyticRegion,
113    },
114    /// A branch-image implication with a fixed branch descriptor.
115    BranchRegionImplication {
116        /// The admitted branch region.
117        region: crate::assumptions::branch_region::BranchRegion,
118    },
119    /// A proposition proved on one directed germ.
120    EventualGerm {
121        /// The proposition established on the germ.
122        proposition: crate::assumptions::formula::AssumptionFormula,
123        /// The variable carried by the germ.
124        variable: crate::symbol::SymbolIdentifier,
125        /// The directed approach.
126        approach: crate::asymptotic_series::AsymptoticApproach,
127    },
128}

130/// A verdict published together with the derivation that established it.
131#[derive(Debug, Clone, PartialEq, Eq)]
132pub(crate) struct AnalyticJudgment {
133    /// The scope the derivation establishes.
134    pub kind: EvidenceKind,
135    /// The three-valued verdict.
136    pub verdict: TruthValue,
137    /// The derivation, present only for a definite verdict.
138    pub evidence: Option<AnalyticEvidence>,
139}
140
141impl AnalyticJudgment {
142    /// Publish a pointwise verdict without a retained derivation.
143    pub(crate) fn opaque(verdict: TruthValue) -> Self {
144        AnalyticJudgment {
145            kind: EvidenceKind::Pointwise,
146            verdict,
147            evidence: None,
148        }
149    }
150
151    /// Publish a pointwise verdict together with its derivation.
152    pub(crate) fn pointwise(verdict: TruthValue, evidence: AnalyticEvidence) -> Self {
153        AnalyticJudgment {
154            kind: EvidenceKind::Pointwise,
155            verdict,
156            evidence: Some(evidence),
157        }
158    }
159
160    /// Report whether this judgment may discharge a goal of the requested scope.
161    pub(crate) fn discharges(&self, requested: EvidenceKind) -> bool {
162        // A derivation discharges exactly its own scope. Promoting a pointwise
163        // proof to an interval, neighborhood, path, or germ goal is the central
164        // unsoundness this module exists to prevent, so the comparison is
165        // equality rather than an ordering.
166        self.kind == requested
167    }
168}

170impl AnalyticEvidence {
171    /// Return the number of derivation nodes retained by this certificate.
172    pub(crate) fn node_count(&self) -> usize {
173        match self {
174            AnalyticEvidence::ExactLeafDefinedness { .. }
175            | AnalyticEvidence::ExplicitDefinedFact
176            | AnalyticEvidence::ExceptionalPoint { .. }
177            | AnalyticEvidence::RangeInference { .. }
178            | AnalyticEvidence::PoleExclusion { .. }
179            | AnalyticEvidence::BranchRegionImplication { .. }
180            | AnalyticEvidence::EventualGerm { .. } => 1,
181            AnalyticEvidence::ArithmeticDefinedness { children, .. } => {
182                1 + children
183                    .iter()
184                    .map(AnalyticEvidence::node_count)
185                    .sum::<usize>()
186            }
187            AnalyticEvidence::FunctionDomain { arguments, .. } => {
188                1 + arguments
189                    .iter()
190                    .map(AnalyticEvidence::node_count)
191                    .sum::<usize>()
192            }
193            AnalyticEvidence::ContinuityComposition { children }
194            | AnalyticEvidence::HolomorphicComposition { children }
195            | AnalyticEvidence::MeromorphicComposition { children } => {
196                1 + children
197                    .iter()
198                    .map(AnalyticEvidence::node_count)
199                    .sum::<usize>()
200            }
201        }
202    }
203
204    /// Return the obligations this derivation depends on, including its children's.
205    pub(crate) fn obligations(&self) -> Vec<SideCondition> {
206        let mut collected = Vec::new();
207        self.collect_obligations(&mut collected);
208        collected
209    }
210
211    fn collect_obligations(&self, collected: &mut Vec<SideCondition>) {
212        match self {
213            AnalyticEvidence::ExactLeafDefinedness { .. }
214            | AnalyticEvidence::ExplicitDefinedFact
215            | AnalyticEvidence::ExceptionalPoint { .. }
216            | AnalyticEvidence::RangeInference { .. }
217            | AnalyticEvidence::PoleExclusion { .. }
218            | AnalyticEvidence::BranchRegionImplication { .. }
219            | AnalyticEvidence::EventualGerm { .. } => {}
220            AnalyticEvidence::ArithmeticDefinedness {
221                children,
222                side_conditions,
223            } => {
224                collected.extend(side_conditions.iter().cloned());
225                for child in children {
226                    child.collect_obligations(collected);
227                }
228            }
229            AnalyticEvidence::FunctionDomain {
230                arguments,
231                domain_obligations,
232                ..
233            } => {
234                collected.extend(domain_obligations.iter().cloned());
235                for argument in arguments {
236                    argument.collect_obligations(collected);
237                }
238            }
239            AnalyticEvidence::ContinuityComposition { children }
240            | AnalyticEvidence::HolomorphicComposition { children }
241            | AnalyticEvidence::MeromorphicComposition { children } => {
242                for child in children {
243                    child.collect_obligations(collected);
244                }
245            }
246        }
247    }
248}

250/// Replay an analytic certificate without relying on its original search path.
251pub(crate) fn replay_analytic_evidence(evidence: &AnalyticEvidence, table: &SymbolTable) -> bool {
252    if !replay_obligations(evidence, table) {
253        return false;
254    }
255    match evidence {
256        AnalyticEvidence::RangeInference { evidence } => {
257            crate::assumptions::range_evidence::replay(evidence)
258        }
259        AnalyticEvidence::ContinuityComposition { children }
260        | AnalyticEvidence::HolomorphicComposition { children }
261        | AnalyticEvidence::MeromorphicComposition { children } => children
262            .iter()
263            .all(|child| replay_analytic_evidence(child, table)),
264        AnalyticEvidence::EventualGerm { proposition, .. } => !matches!(
265            proposition,
266            crate::assumptions::formula::AssumptionFormula::LogicalFalse
267        ),
268        AnalyticEvidence::ExactLeafDefinedness { .. }
269        | AnalyticEvidence::ExplicitDefinedFact
270        | AnalyticEvidence::ArithmeticDefinedness { .. }
271        | AnalyticEvidence::FunctionDomain { .. }
272        | AnalyticEvidence::ExceptionalPoint { .. }
273        | AnalyticEvidence::PoleExclusion { .. } => true,
274        AnalyticEvidence::BranchRegionImplication { region } => region.is_well_formed(),
275    }
276}

278/// Recheck every side condition a derivation depends on, independently of the search path.
279pub(crate) fn replay_obligations(evidence: &AnalyticEvidence, table: &SymbolTable) -> bool {
280    // Replay is deliberately independent of the handler that produced the
281    // derivation: it re-asks each recorded obligation through the ordinary
282    // query surface and requires every one of them to still answer yes.
283    evidence
284        .obligations()
285        .iter()
286        .all(|obligation| obligation_holds(obligation, table))
287}
288
289fn obligation_holds(obligation: &SideCondition, table: &SymbolTable) -> bool {
290    // One obligation class maps to exactly one checked query; an obligation that
291    // no longer answers yes invalidates the whole derivation.
292    use crate::assumptions::queries;
293    let verdict = match obligation {
294        SideCondition::Finite(subject) => queries::is_finite(subject, table),
295        SideCondition::Nonzero(subject) => queries::is_nonzero(subject, table),
296        SideCondition::Zero(subject) => queries::is_zero(subject, table),
297        SideCondition::Positive(subject) => queries::is_positive(subject, table),
298        SideCondition::Integer(subject) => queries::is_integer(subject, table),
299        SideCondition::Defined(subject) => queries::is_defined(subject, table),
300    };
301    matches!(verdict, TruthValue::Yes)
302}
