51/// Lazily compiled relational theory for one immutable assumption base.
52///
53/// Branch literals, classification state, order closure, and exact QE outcomes
54/// belong to the query session. Query-local literals are compiled against this
55/// base instead of rebuilding the base theory for every relation prerequisite.
56pub(crate) struct RelationTheory {
57    base: AssumptionFormula,
58    base_branches: Option<Vec<CompiledRelationBranch>>,
59    qe_results: HashMap<AssumptionFormula, Option<BranchSatisfiability>>,
60}

71    pub(crate) fn ask(
72        &mut self,
73        proposition: &AssumptionFormula,
74        table: Option<&SymbolTable>,
75    ) -> TruthValue {
76        let Some(base_branches) = self.base_branches.as_ref() else {
77            let base = self.base.clone();
78            return self
79                .semialgebraic_entailment(&base, proposition)
80                .unwrap_or(TruthValue::Unknown);
81        };
82        let mut saw_satisfiable_branch = false;
83        let mut every_satisfiable_branch_entails = true;
84        let mut every_satisfiable_branch_refutes = true;
85        for branch in base_branches {
86            match branch.classification {
87                BranchSatisfiability::Inconsistent => {}
88                BranchSatisfiability::Unknown => {
89                    let base = self.base.clone();
90                    return self
91                        .semialgebraic_entailment(&base, proposition)
92                        .unwrap_or(TruthValue::Unknown);
93                }
94                BranchSatisfiability::Satisfiable => {
95                    saw_satisfiable_branch = true;
96                    match branch
97                        .state
98                        .entails_with_graph(proposition, &branch.graph, table)
99                    {
100                        TruthValue::Yes => every_satisfiable_branch_refutes = false,
101                        TruthValue::No => every_satisfiable_branch_entails = false,
102                        TruthValue::Unknown => {
103                            every_satisfiable_branch_entails = false;
104                            every_satisfiable_branch_refutes = false;
105                        }
106                    }
107                }
108            }
109        }
110        if !saw_satisfiable_branch {
111            TruthValue::Unknown
112        } else if every_satisfiable_branch_entails {
113            TruthValue::Yes
114        } else if every_satisfiable_branch_refutes {
115            TruthValue::No
116        } else {
117            TruthValue::Unknown
118        }
119    }

134    pub(crate) fn satisfiability(
135        &mut self,
136        formula: &AssumptionFormula,
137        table: Option<&SymbolTable>,
138    ) -> BranchSatisfiability {
139        if formula == &self.base {
140            let classification = self
141                .base_branches
142                .as_ref()
143                .map(|branches| classify_compiled_branches(branches))
144                .unwrap_or_else(|| {
145                    self.semialgebraic_satisfiability(formula)
146                        .unwrap_or(BranchSatisfiability::Unknown)
147                });
148            return if classification == BranchSatisfiability::Unknown {
149                self.semialgebraic_satisfiability(formula)
150                    .unwrap_or(BranchSatisfiability::Unknown)
151            } else {
152                classification
153            };
154        }
155        let Some(compiled_branches) = compile_branches(formula, table) else {
156            return self
157                .semialgebraic_satisfiability(formula)
158                .unwrap_or(BranchSatisfiability::Unknown);
159        };
160        let classification = classify_compiled_branches(&compiled_branches);
161        if classification == BranchSatisfiability::Unknown {
162            self.semialgebraic_satisfiability(formula)
163                .unwrap_or(BranchSatisfiability::Unknown)
164        } else {
165            classification
166        }
167    }

169    fn semialgebraic_satisfiability(
170        &mut self,
171        formula: &AssumptionFormula,
172    ) -> Option<BranchSatisfiability> {
173        if let Some(result) = self.qe_results.get(formula) {
174            return *result;
175        }
176        let result = semialgebraic_satisfiability_uncached(formula);
177        self.qe_results.insert(formula.clone(), result);
178        result
179    }

181    fn semialgebraic_entailment(
182        &mut self,
183        base: &AssumptionFormula,
184        proposition: &AssumptionFormula,
185    ) -> Option<TruthValue> {
186        let proposition_possible = self.semialgebraic_satisfiability(&AssumptionFormula::and([
187            base.clone(),
188            proposition.clone(),
189        ]))?;
190        let negation_possible = self.semialgebraic_satisfiability(&AssumptionFormula::and([
191            base.clone(),
192            AssumptionFormula::not(proposition.clone()),
193        ]))?;
194        match (proposition_possible, negation_possible) {
195            (BranchSatisfiability::Inconsistent, _) => Some(TruthValue::No),
196            (_, BranchSatisfiability::Inconsistent) => Some(TruthValue::Yes),
197            (BranchSatisfiability::Satisfiable, BranchSatisfiability::Satisfiable) => {
198                Some(TruthValue::Unknown)
199            }
200            _ => None,
201        }
202    }

205fn compile_branches(
206    formula: &AssumptionFormula,
207    table: Option<&SymbolTable>,
208) -> Option<Vec<CompiledRelationBranch>> {
209    let formula_branches = branches(formula)?;
210    crate::assumptions::knowledge_base::record_dnf_branches(formula_branches.len());
211    let route = relation_branch_execution_route(formula_branches.len(), table);
212    let outcomes: Vec<
213        crate::parallelism::IndexedSearchTaskOutcome<CompiledRelationBranch, (), String>,
214    > = execute_peer_group(
215        route,
216        0,
217        &formula_branches,
218        |branch_index, _| {
219            SearchTaskIdentifier::new("assumption", 0, 0, branch_index, "relation_branch")
220        },
221        &crate::parallelism::CancellationToken::new(),
222        |branch, _identifier| {
223            let mut state = RelationState::default();
224            let literals_are_supported = branch
225                .literals
226                .iter()
227                .all(|literal| state.add_literal(literal));
228            let (classification, graph) = if !literals_are_supported {
229                (BranchSatisfiability::Unknown, OrderGraph::default())
230            } else {
231                state.classify_with_graph(table)
232            };
233            SearchTaskOutcome::Candidate {
234                value: CompiledRelationBranch {
235                    state,
236                    classification,
237                    graph,
238                },
239                evidence: (),
240                conditions: SearchConditionLedger::empty(),
241                completeness_contribution: SearchCompletenessContribution::new(
242                    "relation branch",
243                    true,
244                    true,
245                ),
246                quality_key: vec![0],
247            }
248        },
249    );
250    // Every disjunct of the normal form has to be present for the classification below to mean
251    // anything: a satisfiable branch that went missing would let an inconsistency verdict be
252    // read off a formula that has a model. A branch that did not produce its compiled state
253    // therefore declines the whole compilation rather than shortening the disjunction.
254    outcomes
255        .into_iter()
256        .map(|indexed_outcome| match indexed_outcome.outcome {
257            SearchTaskOutcome::Candidate { value, .. } => Some(value),
258            SearchTaskOutcome::ProvedEmpty { .. }
259            | SearchTaskOutcome::Declined { .. }
260            | SearchTaskOutcome::ResourceExhausted
261            | SearchTaskOutcome::Cancelled => None,
262        })
263        .collect()
264}

266fn relation_branch_execution_route(
267    branch_count: usize,
268    table: Option<&SymbolTable>,
269) -> ExecutionRoute {
270    // Compiling one disjunct reads only that disjunct's literals and the immutable symbol table,
271    // so the branches are independent. The decision still runs through the operation's cost model
272    // because most formulas carry a handful of small branches that a serial pass finishes sooner
273    // than a batch can be handed out.
274    if branch_count < 2 {
275        return ExecutionRoute::Serial;
276    }
277    let execution_mode = table
278        .map(SymbolTable::assumption_execution_mode)
279        .unwrap_or(ExecutionMode::Sequential);
280    if !crate::parallelism::work_class_may_fork(
281        execution_mode,
282        &crate::parallelism::CancellationToken::new(),
283        "assumption_relation_branch",
284        crate::parallelism::NO_ADMITTED_SERIAL_ONLY_WORK_CLASSES,
285    ) {
286        return ExecutionRoute::Serial;
287    }
288    let requested_permits = branch_count.min(crate::parallelism::maximum_operation_concurrency());
289    crate::parallelism::execution_decision(
290        execution_mode,
291        crate::parallelism::runtime_availability(),
292        &WorkEstimate::new(
293            (branch_count as u64).saturating_mul(64),
294            branch_count,
295            branch_count.saturating_mul(std::mem::size_of::<CompiledRelationBranch>()),
296            0,
297            requested_permits,
298            "assumption_relation_branch",
299        ),
300        requested_permits,
301        crate::parallelism::runtime_memory_capacity(
302            branch_count.saturating_mul(std::mem::size_of::<CompiledRelationBranch>()),
303        ),
304        &crate::parallelism::CancellationToken::new(),
305        crate::parallelism::NO_ADMITTED_SERIAL_ONLY_WORK_CLASSES,
306    )
307    .route
308}

310fn classify_compiled_branches(branches: &[CompiledRelationBranch]) -> BranchSatisfiability {
311    let mut saw_unknown = false;
312    for branch in branches {
313        match branch.classification {
314            BranchSatisfiability::Satisfiable => return BranchSatisfiability::Satisfiable,
315            BranchSatisfiability::Unknown => saw_unknown = true,
316            BranchSatisfiability::Inconsistent => {}
317        }
318    }
319    if saw_unknown {
320        BranchSatisfiability::Unknown
321    } else {
322        BranchSatisfiability::Inconsistent
323    }

435fn semialgebraic_satisfiability_uncached(
436    formula: &AssumptionFormula,
437) -> Option<BranchSatisfiability> {
438    crate::assumptions::knowledge_base::record_qe_eligibility_check();
439    let variables = semialgebraic_variables(formula)
440        .into_iter()
441        .collect::<Vec<_>>();
442    if variables.is_empty() || variables.len() > MAXIMUM_SEMIALGEBRAIC_VARIABLES {
443        return None;
444    }
445    if !semialgebraic_variables_are_explicitly_real(formula, &variables) {
446        return None;
447    }
448    let lowered_formula = erase_semialgebraic_domain_guards(formula, &variables)?;
449    if semialgebraic_atom_count(&lowered_formula)? > MAXIMUM_SEMIALGEBRAIC_ATOMS {
450        return None;
451    }
452    let mut context = crate::interner::Context::new();
453    let greatest_identifier = variables.last()?.0 as usize;
454    for position in 0..=greatest_identifier {
455        context
456            .symbol_table
457            .intern_symbol(&format!("assumption_variable_{position}"));
458    }
459    if !semialgebraic_atoms_can_be_polynomials(&context, &lowered_formula, &variables) {
460        return None;
461    }
462    let logical_expression =
463        crate::assumptions::logical_expression_from_proposition(&mut context, &lowered_formula)?;
464    let quantified_formula = crate::real_quantifier_elimination::formula_from_expression(
465        &mut context,
466        &logical_expression,
467        vec![crate::real_quantifier_elimination::Quantifier::Existential; variables.len()],
468        &variables,
469    )?;
470    crate::assumptions::knowledge_base::record_qe_invocation();
471    let verdict = crate::real_quantifier_elimination::eliminate_quantifiers(
472        &mut context,
473        &quantified_formula,
474    )?;
475    match verdict.kind {
476        ExpressionKind::Constant(BuiltinConstant::LogicalTrue) => {
477            Some(BranchSatisfiability::Satisfiable)
478        }
479        ExpressionKind::Constant(BuiltinConstant::LogicalFalse) => {
480            Some(BranchSatisfiability::Inconsistent)
481        }
482        _ => None,
483    }
484}

486fn semialgebraic_atoms_can_be_polynomials(
487    context: &crate::interner::Context,
488    formula: &AssumptionFormula,
489    variables: &[crate::symbol::SymbolIdentifier],
490) -> bool {
491    // The eligibility question the Tarski-formula construction answers anyway, asked before
492    // the atoms are built rather than after. A relation atom reaches quantifier elimination
493    // only if its `left - right` lies in Q[variables]; an atom outside that ring — a
494    // transcendental constant, an irrational coefficient, a function call — makes the whole
495    // formula ineligible. Building the atoms first is what costs: interning a relation
496    // evaluates it, and evaluating `z > -pi/2` drives the exact-real order through the
497    // simplifier for a comparison that no free-variable atom can settle. Forming the
498    // difference alone is arithmetic on an interned sum and settles nothing.
499    //
500    // An atom whose difference is free of variables is admitted without the ring test,
501    // because such an atom is a ground comparison that construction decides to a truth
502    // constant, and the construction path never asks it for a polynomial.
503    match formula {
504        AssumptionFormula::LogicalTrue | AssumptionFormula::LogicalFalse => true,
505        AssumptionFormula::Predicate(predicate) => {
506            let PredicateSubject::Scalar(subject) = &predicate.subject else {
507                return true;
508            };
509            difference_can_be_a_polynomial(subject, variables)
510        }
511        AssumptionFormula::ScalarRelation(relation) => {
512            let difference = crate::expression::builders::subtract(
513                context,
514                Arc::clone(&relation.left),
515                Arc::clone(&relation.right),
516            );
517            difference_can_be_a_polynomial(&difference, variables)
518        }
519        AssumptionFormula::LogicalNot(value) => {
520            semialgebraic_atoms_can_be_polynomials(context, value, variables)
521        }
522        AssumptionFormula::LogicalAnd(values) | AssumptionFormula::LogicalOr(values) => values
523            .iter()
524            .all(|value| semialgebraic_atoms_can_be_polynomials(context, value, variables)),
525        AssumptionFormula::ScalarMembership(_) | AssumptionFormula::IntegerCongruence(_) => true,
526    }
527}

537fn semialgebraic_variables_are_explicitly_real(
538    formula: &AssumptionFormula,
539    variables: &[crate::symbol::SymbolIdentifier],
540) -> bool {
541    let Some(formula_branches) = branches(formula) else {
542        return false;
543    };
544    !formula_branches.is_empty()
545        && formula_branches.iter().all(|branch| {
546            variables.iter().all(|variable| {
547                branch.literals.iter().any(|literal| {
548                    matches!(
549                        literal,
550                        AssumptionFormula::Predicate(AppliedPredicate {
551                            identifier: PredicateIdentifier::CurrentScalar(
552                                CurrentScalarPredicate::Real
553                            ),
554                            subject: PredicateSubject::Scalar(subject),
555                            ..
556                        }) if matches!(subject.kind, ExpressionKind::Symbol(identifier) if identifier == *variable)
557                    )
558                })
559            })
560        })
561}

563fn erase_semialgebraic_domain_guards(
564    formula: &AssumptionFormula,
565    variables: &[crate::symbol::SymbolIdentifier],
566) -> Option<AssumptionFormula> {
567    match formula {
568        AssumptionFormula::Predicate(AppliedPredicate {
569            identifier: PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Real),
570            subject: PredicateSubject::Scalar(subject),
571            ..
572        }) if matches!(subject.kind, ExpressionKind::Symbol(identifier) if variables.contains(&identifier)) => {
573            Some(AssumptionFormula::LogicalTrue)
574        }
575        AssumptionFormula::LogicalNot(value) => Some(AssumptionFormula::not(
576            erase_semialgebraic_domain_guards(value, variables)?,
577        )),
578        AssumptionFormula::LogicalAnd(values) => Some(AssumptionFormula::and(
579            values
580                .iter()
581                .map(|value| erase_semialgebraic_domain_guards(value, variables))
582                .collect::<Option<Vec<_>>>()?,
583        )),
584        AssumptionFormula::LogicalOr(values) => Some(AssumptionFormula::or(
585            values
586                .iter()
587                .map(|value| erase_semialgebraic_domain_guards(value, variables))
588                .collect::<Option<Vec<_>>>()?,
589        )),
590        _ => Some(formula.clone()),
591    }
592}

594fn semialgebraic_variables(
595    formula: &AssumptionFormula,
596) -> BTreeSet<crate::symbol::SymbolIdentifier> {
597    let mut variables = BTreeSet::new();
598    match formula {
599        AssumptionFormula::Predicate(predicate) => {
600            let PredicateSubject::Scalar(subject) = &predicate.subject else {
601                return BTreeSet::new();
602            };
603            variables.extend(crate::expression::free_variables::free_variables(subject));
604        }
605        AssumptionFormula::ScalarRelation(relation) => {
606            variables.extend(crate::expression::free_variables::free_variables(
607                &relation.left,
608            ));
609            variables.extend(crate::expression::free_variables::free_variables(
610                &relation.right,
611            ));
612        }
613        AssumptionFormula::LogicalNot(value) => {
614            variables.extend(semialgebraic_variables(value));
615        }
616        AssumptionFormula::LogicalAnd(values) | AssumptionFormula::LogicalOr(values) => {
617            for value in values {
618                variables.extend(semialgebraic_variables(value));
619            }
620        }
621        AssumptionFormula::LogicalTrue
622        | AssumptionFormula::LogicalFalse
623        | AssumptionFormula::ScalarMembership(_)
624        | AssumptionFormula::IntegerCongruence(_) => {}
625    }
626    variables
627}

629fn semialgebraic_atom_count(formula: &AssumptionFormula) -> Option<usize> {
630    match formula {
631        AssumptionFormula::LogicalTrue | AssumptionFormula::LogicalFalse => Some(0),
632        AssumptionFormula::Predicate(predicate) => match predicate.identifier {
633            PredicateIdentifier::CurrentScalar(
634                CurrentScalarPredicate::Positive
635                | CurrentScalarPredicate::Negative
636                | CurrentScalarPredicate::Zero
637                | CurrentScalarPredicate::Nonzero,
638            ) => Some(1),
639            _ => None,
640        },
641        AssumptionFormula::ScalarRelation(_) => Some(1),
642        AssumptionFormula::LogicalNot(value) => semialgebraic_atom_count(value),
643        AssumptionFormula::LogicalAnd(values) | AssumptionFormula::LogicalOr(values) => {
644            values.iter().try_fold(0_usize, |total, value| {
645                total.checked_add(semialgebraic_atom_count(value)?)
646            })
647        }
648        AssumptionFormula::ScalarMembership(_) | AssumptionFormula::IntegerCongruence(_) => None,
649    }
650}
651
652fn branches(formula: &AssumptionFormula) -> Option<Vec<RelationBranch>> {
653    match formula {
654        AssumptionFormula::LogicalTrue => Some(vec![RelationBranch {
655            literals: Vec::new(),
656        }]),
657        AssumptionFormula::LogicalFalse => Some(Vec::new()),
658        AssumptionFormula::Predicate(_)
659        | AssumptionFormula::ScalarRelation(_)
660        | AssumptionFormula::ScalarMembership(_)
661        | AssumptionFormula::IntegerCongruence(_)
662        | AssumptionFormula::LogicalNot(_) => Some(vec![RelationBranch {
663            literals: vec![formula.clone()],
664        }]),
665        AssumptionFormula::LogicalOr(children) => {
666            let mut result = Vec::new();
667            for child in children {
668                result.extend(branches(child)?);
669                if result.len() > MAXIMUM_RELATION_BRANCHES {
670                    return None;
671                }
672            }
673            Some(result)
674        }
675        AssumptionFormula::LogicalAnd(children) => {
676            let mut result = vec![RelationBranch {
677                literals: Vec::new(),
678            }];
679            for child in children {
680                let child_branches = branches(child)?;
681                let mut next = Vec::new();
682                for left in &result {
683                    for right in &child_branches {
684                        let mut literals = left.literals.clone();
685                        literals.extend(right.literals.clone());
686                        next.push(RelationBranch { literals });
687                        if next.len() > MAXIMUM_RELATION_BRANCHES {
688                            return None;
689                        }
690                    }
691                }
692                result = next;
693            }
694            Some(result)
695        }
696    }
697}

699fn classify_branch(branch: &RelationBranch, table: Option<&SymbolTable>) -> BranchSatisfiability {
700    let mut state = RelationState::default();
701    for literal in &branch.literals {
702        if !state.add_literal(literal) {
703            return BranchSatisfiability::Unknown;
704        }
705    }
706    state.classify(table)
707}
708
709fn branch_entails(
710    branch: &RelationBranch,
711    proposition: &AssumptionFormula,
712    table: Option<&SymbolTable>,
713) -> TruthValue {
714    match proposition {
715        AssumptionFormula::LogicalTrue => TruthValue::Yes,
716        AssumptionFormula::LogicalFalse => TruthValue::No,
717        AssumptionFormula::LogicalAnd(children) => combine_conjunction(
718            children
719                .iter()
720                .map(|child| branch_entails(branch, child, table)),
721        ),
722        AssumptionFormula::LogicalOr(children) => combine_disjunction(
723            children
724                .iter()
725                .map(|child| branch_entails(branch, child, table)),
726        ),
727        _ => {
728            let mut state = RelationState::default();
729            if !branch
730                .literals
731                .iter()
732                .all(|literal| state.add_literal(literal))
733            {
734                return TruthValue::Unknown;
735            }
736            state.entails(proposition, table)
737        }
738    }
739}
740
741fn combine_conjunction(values: impl Iterator<Item = TruthValue>) -> TruthValue {
742    let mut result = TruthValue::Yes;
743    for value in values {
744        match value {
745            TruthValue::No => return TruthValue::No,
746            TruthValue::Unknown => result = TruthValue::Unknown,
747            TruthValue::Yes => {}
748        }
749    }
750    result
751}
752
753fn combine_disjunction(values: impl Iterator<Item = TruthValue>) -> TruthValue {
754    let mut result = TruthValue::No;
755    for value in values {
756        match value {
757            TruthValue::Yes => return TruthValue::Yes,
758            TruthValue::Unknown => result = TruthValue::Unknown,
759            TruthValue::No => {}
760        }
761    }
762    result
763}

819    fn classify_with_graph(
820        &self,
821        table: Option<&SymbolTable>,
822    ) -> (BranchSatisfiability, OrderGraph) {
823        if self.has_direct_complement() {
824            return (BranchSatisfiability::Inconsistent, OrderGraph::default());
825        }
826        match crate::assumptions::knowledge_base::predicate_literals_are_satisfiable(
827            &self.predicates,
828            table,
829        ) {
830            Some(true) => {}
831            Some(false) => return (BranchSatisfiability::Inconsistent, OrderGraph::default()),
832            None => return (BranchSatisfiability::Unknown, OrderGraph::default()),
833        }
834        if self
835            .congruences
836            .iter()
837            .any(|(congruence, _)| match exact_integer(&congruence.modulus) {
838                Some(_) => affine_congruence(congruence).is_none(),
839                None => !matches!(congruence.modulus.kind, ExpressionKind::Symbol(_)),
840            })
841            || self.relations.iter().any(|(relation, _)| {
842                exact_relation_verdict(relation, table).is_none()
843                    && (!affine_expression_is_supported(&relation.left)
844                        || !affine_expression_is_supported(&relation.right))
845            })
846        {
847            return (BranchSatisfiability::Unknown, OrderGraph::default());
848        }
849        crate::assumptions::knowledge_base::record_relation_graph_build();
850        let graph = OrderGraph::from_state(self, table);
851        let classification = if graph.exhausted {
852            BranchSatisfiability::Unknown
853        } else if graph.inconsistent
854            || self.closed_relation_conflict(table)
855            || self.relation_conflict(&graph, table)
856            || self.congruence_conflict(&graph)
857            || self.predicate_conflict(&graph, table)
858            || self.membership_conflict(&graph, table)
859        {
860            BranchSatisfiability::Inconsistent
861        } else {
862            BranchSatisfiability::Satisfiable
863        };
864        (classification, graph)
865    }

868        crate::assumptions::knowledge_base::record_relation_graph_build();
869        let graph = OrderGraph::from_state(self, table);

873    fn entails_with_graph(
874        &self,
875        proposition: &AssumptionFormula,
876        graph: &OrderGraph,
877        table: Option<&SymbolTable>,
878    ) -> TruthValue {
879        if graph.exhausted {
880            return TruthValue::Unknown;
881        }
882        match proposition {
883            AssumptionFormula::ScalarRelation(relation) => {
884                self.relation_verdict(relation, &graph, table)
885            }
886            AssumptionFormula::ScalarMembership(membership) => {
887                self.membership_verdict(membership, &graph, table)
888            }
889            AssumptionFormula::IntegerCongruence(congruence) => {
890                self.congruence_verdict(congruence, &graph)
891            }
892            AssumptionFormula::Predicate(predicate) => {
893                self.predicate_verdict(predicate, &graph, table)
894            }
895            AssumptionFormula::LogicalNot(value) => {
896                negate_truth(self.entails_with_graph(value, graph, table))
897            }
898            _ => TruthValue::Unknown,
899        }
900    }

990        &self,
991        predicate: &AppliedPredicate,
992        graph: &OrderGraph,
993        table: Option<&SymbolTable>,
994    ) -> TruthValue {
995        let PredicateSubject::Scalar(subject) = &predicate.subject else {
996            return TruthValue::Unknown;
997        };

999            let PredicateSubject::Scalar(candidate_subject) = &candidate.subject else {
1000                return None;
1001            };

1003                && candidate_subject.kind != subject.kind
1004                && graph.equal(&subject_key(candidate_subject), &subject_key(subject)))
1005            .then_some(*value)
1006        }) {
1007            return truth_from_boolean(value);
1008        }
1009        let relational_verdict = match predicate.identifier {
1010            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Positive) => graph
1011                .relation_verdict(&CanonicalScalarRelation {
1012                    operator: ScalarRelationOperator::Less,
1013                    left: synthetic_integer(0),
1014                    right: subject.clone(),
1015                }),
1016            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Negative) => graph
1017                .relation_verdict(&CanonicalScalarRelation {
1018                    operator: ScalarRelationOperator::Less,
1019                    left: subject.clone(),
1020                    right: synthetic_integer(0),
1021                }),
1022            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Zero) => graph
1023                .relation_verdict(&CanonicalScalarRelation {
1024                    operator: ScalarRelationOperator::Equal,
1025                    left: subject.clone(),
1026                    right: synthetic_integer(0),
1027                }),
1028            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Nonzero) => graph
1029                .relation_verdict(&CanonicalScalarRelation {
1030                    operator: ScalarRelationOperator::NotEqual,
1031                    left: subject.clone(),
1032                    right: synthetic_integer(0),
1033                }),
1034            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Even) => {
1035                self.parity_verdict(subject, 0, graph)
1036            }
1037            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Odd) => {
1038                self.parity_verdict(subject, 1, graph)
1039            }
1040            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Real) => {
1041                if self.memberships.iter().any(|(membership, value)| {
1042                    *value
1043                        && membership.element.kind == subject.kind
1044                        && matches!(
1045                            membership.set_expression.kind,
1046                            ExpressionKind::Constant(BuiltinConstant::RealLine)
1047                                | ExpressionKind::Interval(..)
1048                        )
1049                }) {
1050                    TruthValue::Yes
1051                } else {
1052                    TruthValue::Unknown
1053                }
1054            }
1055            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::ExtendedReal) => {
1056                if graph.contains_ordered_subject(subject)
1057                    || self.memberships.iter().any(|(membership, value)| {
1058                        *value
1059                            && membership.element.kind == subject.kind
1060                            && matches!(
1061                                membership.set_expression.kind,
1062                                ExpressionKind::Constant(BuiltinConstant::RealLine)
1063                                    | ExpressionKind::Interval(..)
1064                            )
1065                    })
1066                {
1067                    TruthValue::Yes
1068                } else {
1069                    TruthValue::Unknown
1070                }
1071            }
1072            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Integer) => {
1073                if self.congruences.iter().any(|(congruence, value)| {
1074                    *value
1075                        && ([&congruence.left, &congruence.right]
1076                            .into_iter()
1077                            .any(|operand| {
1078                                operand.kind == subject.kind
1079                                    || graph.equal(&subject_key(operand), &subject_key(subject))
1080                            }))
1081                }) || self.memberships.iter().any(|(membership, value)| {
1082                    *value
1083                        && membership.element.kind == subject.kind
1084                        && matches!(
1085                            membership.set_expression.kind,
1086                            ExpressionKind::Constant(BuiltinConstant::IntegerSet)
1087                        )
1088                }) {
1089                    TruthValue::Yes
1090                } else {
1091                    TruthValue::Unknown
1092                }
1093            }
1094            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::PositiveInteger) => {
1095                if self
1096                    .congruences
1097                    .iter()
1098                    .any(|(congruence, value)| *value && congruence.modulus.kind == subject.kind)
1099                {
1100                    TruthValue::Yes
1101                } else {
1102                    TruthValue::Unknown
1103                }
1104            }
1105            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Complex) => {
1106                if self.memberships.iter().any(|(membership, value)| {
1107                    *value
1108                        && membership.element.kind == subject.kind
1109                        && matches!(
1110                            membership.set_expression.kind,
1111                            ExpressionKind::Constant(BuiltinConstant::ComplexPlane)
1112                        )
1113                }) {
1114                    TruthValue::Yes
1115                } else {
1116                    TruthValue::Unknown
1117                }
1118            }
1119            PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Finite) => {
1120                if self.memberships.iter().any(|(membership, value)| {
1121                    *value
1122                        && membership.element.kind == subject.kind
1123                        && matches!(membership.set_expression.kind, ExpressionKind::Interval(..))
1124                }) {
1125                    TruthValue::Yes
1126                } else {
1127                    TruthValue::Unknown
1128                }
1129            }
1130            _ => TruthValue::Unknown,
1131        };
1132        if relational_verdict == TruthValue::Unknown {
1133            table
1134                .map(|table| crate::assumptions::knowledge_base::handler_verdict(predicate, table))
1135                .unwrap_or(TruthValue::Unknown)
1136        } else {
1137            relational_verdict
1138        }
1139    }

1251    fn congruence_conflict(&self, graph: &OrderGraph) -> bool {
1252        if self.congruences.iter().any(|(congruence, value)| {
1253            let Some(modulus) = exact_integer(&congruence.modulus) else {
1254                return false;
1255            };
1256            if modulus <= Integer::from(0) {
1257                return *value;
1258            }
1259            affine_congruence(congruence).is_some_and(|constraint| {
1260                let divisor =
1261                    integer_gcd(constraint.coefficient.clone(), constraint.modulus.clone());
1262                let equation_is_possible =
1263                    positive_remainder(-constraint.constant.clone(), divisor) == Integer::from(0);
1264                if *value {
1265                    !equation_is_possible
1266                } else {
1267                    constraint.coefficient == Integer::from(0) && equation_is_possible
1268                }
1269            })
1270        }) {
1271            return true;
1272        }
1273        let positive_constraints = self
1274            .congruences
1275            .iter()
1276            .filter(|(_, value)| *value)
1277            .filter_map(|(congruence, _)| affine_congruence(congruence))
1278            .filter_map(solve_affine_congruence)
1279            .collect::<Vec<_>>();
1280        if positive_constraints
1281            .iter()
1282            .enumerate()
1283            .any(|(left_index, left)| {
1284                positive_constraints
1285                    .iter()
1286                    .skip(left_index + 1)
1287                    .any(|right| {
1288                        (left.subject == right.subject
1289                            || graph.equal(&left.subject, &right.subject))
1290                            && (left.residue.clone() - right.residue.clone())
1291                                % integer_gcd(left.modulus.clone(), right.modulus.clone())
1292                                != Integer::from(0)
1293                    })
1294            })
1295        {
1296            return true;
1297        }
1298        self.congruences
1299            .iter()
1300            .filter(|(_, value)| !*value)
1301            .filter_map(|(congruence, _)| {
1302                affine_congruence(congruence).and_then(solve_affine_congruence)
1303            })
1304            .any(|negative| {
1305                combined_congruence_for_equivalent_subject(
1306                    &positive_constraints,
1307                    &negative.subject,
1308                    graph,
1309                )
1310                .is_some_and(|positive| congruence_entails(&positive, &negative))
1311            })
1312    }

1314    fn congruence_verdict(
1315        &self,
1316        requested: &CanonicalIntegerCongruence,
1317        graph: &OrderGraph,
1318    ) -> TruthValue {
1319        if let Some(value) = self
1320            .congruences
1321            .iter()
1322            .find_map(|(candidate, value)| (candidate == requested).then_some(*value))
1323        {
1324            return truth_from_boolean(value);
1325        }
1326        let Some(requested) = affine_congruence(requested).and_then(solve_affine_congruence) else {
1327            return TruthValue::Unknown;
1328        };
1329        let positive_constraints = self
1330            .congruences
1331            .iter()
1332            .filter(|(_, value)| *value)
1333            .filter_map(|(congruence, _)| {
1334                affine_congruence(congruence).and_then(solve_affine_congruence)
1335            })
1336            .collect::<Vec<_>>();
1337        if let Some(asserted) = combined_congruence_for_equivalent_subject(
1338            &positive_constraints,
1339            &requested.subject,
1340            graph,
1341        ) {
1342            if congruence_entails(&asserted, &requested) {
1343                return TruthValue::Yes;
1344            }
1345            let common_divisor = integer_gcd(asserted.modulus.clone(), requested.modulus.clone());
1346            if positive_remainder(
1347                asserted.residue.clone() - requested.residue.clone(),
1348                common_divisor,
1349            ) != Integer::from(0)
1350            {
1351                return TruthValue::No;
1352            }
1353        }
1354        TruthValue::Unknown
1355    }

1357    fn parity_verdict(
1358        &self,
1359        subject: &ExpressionReference,
1360        residue: i64,
1361        graph: &OrderGraph,
1362    ) -> TruthValue {
1363        let requested_subject = subject_key(subject);
1364        let positive_constraints = self
1365            .congruences
1366            .iter()
1367            .filter(|(_, value)| *value)
1368            .filter_map(|(congruence, _)| {
1369                affine_congruence(congruence).and_then(solve_affine_congruence)
1370            })
1371            .collect::<Vec<_>>();
1372        if let Some(constraint) = combined_congruence_for_equivalent_subject(
1373            &positive_constraints,
1374            &requested_subject,
1375            graph,
1376        ) {
1377            if constraint.modulus.clone() % Integer::from(2) == Integer::from(0) {
1378                return truth_from_boolean(
1379                    positive_remainder(constraint.residue, Integer::from(2))
1380                        == Integer::from(residue),
1381                );
1382            }
1383        }
1384        TruthValue::Unknown
1385    }

1406    fn from_state(state: &RelationState, table: Option<&SymbolTable>) -> Self {
1407        let mut graph = OrderGraph::default();
1408        for (relation, value) in &state.relations {
1409            if !value {
1410                continue;
1411            }
1412            graph.add_relation(relation);
1413        }
1414        for (predicate, value) in &state.predicates {
1415            if !value {
1416                continue;
1417            }
1418            let PredicateSubject::Scalar(subject) = &predicate.subject else {
1419                continue;
1420            };
1421            let relation = match predicate.identifier {
1422                PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Positive) => {
1423                    Some(CanonicalScalarRelation {
1424                        operator: ScalarRelationOperator::Less,
1425                        left: synthetic_integer(0),
1426                        right: subject.clone(),
1427                    })
1428                }
1429                PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Negative) => {
1430                    Some(CanonicalScalarRelation {
1431                        operator: ScalarRelationOperator::Less,
1432                        left: subject.clone(),
1433                        right: synthetic_integer(0),
1434                    })
1435                }
1436                PredicateIdentifier::CurrentScalar(CurrentScalarPredicate::Zero) => {
1437                    Some(CanonicalScalarRelation {
1438                        operator: ScalarRelationOperator::Equal,
1439                        left: subject.clone(),
1440                        right: synthetic_integer(0),
1441                    })
1442                }
1443                _ => None,
1444            };
1445            if let Some(relation) = relation {
1446                graph.add_relation(&relation);
1447            }
1448        }
1449        for (membership, value) in &state.memberships {
1450            if !value {
1451                continue;
1452            }
1453            if let ExpressionKind::Interval(lower, upper, lower_closed, upper_closed) =
1454                &membership.set_expression.kind
1455            {
1456                graph.add_relation(&CanonicalScalarRelation {
1457                    operator: if *lower_closed {
1458                        ScalarRelationOperator::LessOrEqual
1459                    } else {
1460                        ScalarRelationOperator::Less
1461                    },
1462                    left: lower.clone(),
1463                    right: membership.element.clone(),
1464                });
1465                graph.add_relation(&CanonicalScalarRelation {
1466                    operator: if *upper_closed {
1467                        ScalarRelationOperator::LessOrEqual
1468                    } else {
1469                        ScalarRelationOperator::Less
1470                    },
1471                    left: membership.element.clone(),
1472                    right: upper.clone(),
1473                });
1474            }
1475        }
1476        graph.add_exact_constant_relations(state, table);
1477        if graph.nodes.len() > MAXIMUM_RELATION_NODES {
1478            graph.exhausted = true;
1479            return graph;
1480        }
1481        graph.close();
1482        graph
1483    }

1485    fn add_relation(&mut self, relation: &CanonicalScalarRelation) {
1486        for expression in [&relation.left, &relation.right] {
1487            if let Some(value) = exact_rational(expression) {
1488                self.numeric_values.insert(rational_key(&value), value);
1489            }
1490        }
1491        if relation.operator == ScalarRelationOperator::NotEqual {
1492            insert_unordered_pair(
1493                &mut self.disequalities,
1494                subject_key(&relation.left),
1495                subject_key(&relation.right),
1496            );
1497            return;
1498        }
1499        let Some(normalized) = normalized_affine_relation(relation) else {
1500            if relation.operator == ScalarRelationOperator::Equal {
1501                let left = subject_key(&relation.left);
1502                let right = subject_key(&relation.right);
1503                self.add_edge(left.clone(), right.clone(), rational_zero(), false);
1504                self.add_edge(right, left, rational_zero(), false);
1505            } else {
1506                self.add_edge(
1507                    subject_key(&relation.left),
1508                    subject_key(&relation.right),
1509                    rational_zero(),
1510                    relation.operator == ScalarRelationOperator::Less,
1511                );
1512            }
1513            return;
1514        };
1515        match normalized {
1516            NormalizedRelation::Constant(_) => {}
1517            NormalizedRelation::Edge {
1518                left,
1519                right,
1520                offset,
1521                strict,
1522                equality,
1523            } => {
1524                if !equality {
1525                    self.ordered_subjects.insert(left.clone());
1526                    self.ordered_subjects.insert(right.clone());
1527                }
1528                self.add_edge(left.clone(), right.clone(), offset.clone(), strict);
1529                if equality {
1530                    self.add_edge(right, left, -offset, false);
1531                }
1532            }
1533        }
1534    }

1536    fn add_exact_constant_relations(&mut self, state: &RelationState, table: Option<&SymbolTable>) {
1537        for (relation, _) in &state.relations {
1538            for expression in [&relation.left, &relation.right] {
1539                if let Some(value) = exact_rational(expression) {
1540                    self.numeric_values.insert(rational_key(&value), value);
1541                } else if let Some(table) = table {
1542                    if crate::exact_real_order::compare_real_expressions(
1543                        expression, expression, table,
1544                    ) == Some(Ordering::Equal)
1545                        && !matches!(expression.kind, ExpressionKind::Symbol(_))
1546                    {
1547                        self.nodes.insert(subject_key(expression));
1548                    }
1549                }
1550            }
1551        }
1552        self.numeric_values
1553            .insert(rational_key(&rational_zero()), rational_zero());
1554        let values = self
1555            .numeric_values
1556            .iter()
1557            .map(|(key, value)| (key.clone(), value.clone()))
1558            .collect::<Vec<_>>();
1559        for (left_key, left_value) in &values {
1560            for (right_key, right_value) in &values {
1561                self.add_edge(
1562                    left_key.clone(),
1563                    right_key.clone(),
1564                    right_value.clone() - left_value.clone(),
1565                    false,
1566                );
1567            }
1568        }
1569    }

1572        &mut self,
1573        left: StructuralSubjectKey,
1574        right: StructuralSubjectKey,
1575        offset: Rational,
1576        strict: bool,

1592        let nodes = self.nodes.iter().cloned().collect::<Vec<_>>();
1593        for middle in &nodes {
1594            for left in &nodes {
1595                for right in &nodes {
1596                    let Some(left_edge) = self.edges.get(&(left.clone(), middle.clone())).cloned()
1597                    else {
1598                        continue;
1599                    };
1600                    let Some(right_edge) =

1602                    else {
1603                        continue;
1604                    };
1605                    self.add_edge(
1606                        left.clone(),

1618                    edge.offset > rational_zero() || (edge.offset == rational_zero() && edge.strict)
1619                })

1621            .disequalities
1622            .iter()

1626    fn relation_verdict(&self, relation: &CanonicalScalarRelation) -> TruthValue {
1627        if relation.operator == ScalarRelationOperator::NotEqual {
1628            let equality = CanonicalScalarRelation {
1629                operator: ScalarRelationOperator::Equal,
1630                left: relation.left.clone(),
1631                right: relation.right.clone(),
1632            };
1633            return match self.relation_verdict(&equality) {
1634                TruthValue::Yes => TruthValue::No,
1635                TruthValue::No => TruthValue::Yes,
1636                TruthValue::Unknown => {
1637                    if contains_unordered_pair(
1638                        &self.disequalities,
1639                        subject_key(&relation.left),
1640                        subject_key(&relation.right),
1641                    ) {
1642                        TruthValue::Yes
1643                    } else {
1644                        TruthValue::Unknown
1645                    }
1646                }
1647            };
1648        }
1649        let Some(normalized) = normalized_affine_relation(relation) else {
1650            return self.raw_relation_verdict(relation);
1651        };
1652        match normalized {
1653            NormalizedRelation::Constant(value) => truth_from_boolean(value),
1654            NormalizedRelation::Edge {
1655                left,
1656                right,
1657                offset,
1658                strict,
1659                equality,
1660            } => {
1661                if equality {
1662                    if self.edge_entails(&left, &right, &offset, false)
1663                        && self.edge_entails(&right, &left, &(-offset.clone()), false)
1664                    {
1665                        TruthValue::Yes
1666                    } else if self.edge_entails(&left, &right, &offset, true)
1667                        || self.edge_entails(&right, &left, &(-offset), true)
1668                    {
1669                        TruthValue::No
1670                    } else {
1671                        TruthValue::Unknown
1672                    }
1673                } else if self.edge_entails(&left, &right, &offset, strict) {
1674                    TruthValue::Yes
1675                } else if self.edge_entails(&right, &left, &(-offset), !strict) {
1676                    TruthValue::No
1677                } else {
1678                    TruthValue::Unknown
1679                }
1680            }
1681        }
1682    }

1684    fn raw_relation_verdict(&self, relation: &CanonicalScalarRelation) -> TruthValue {
1685        let left = subject_key(&relation.left);
1686        let right = subject_key(&relation.right);
1687        match relation.operator {
1688            ScalarRelationOperator::Equal => {
1689                if self.equal(&left, &right) {
1690                    TruthValue::Yes
1691                } else if contains_unordered_pair(&self.disequalities, left, right) {
1692                    TruthValue::No
1693                } else {
1694                    TruthValue::Unknown
1695                }
1696            }
1697            ScalarRelationOperator::Less => {
1698                if self.edge_entails(&left, &right, &rational_zero(), true) {
1699                    TruthValue::Yes
1700                } else if self.edge_entails(&right, &left, &rational_zero(), false) {
1701                    TruthValue::No
1702                } else {
1703                    TruthValue::Unknown
1704                }
1705            }
1706            ScalarRelationOperator::LessOrEqual => {
1707                if self.edge_entails(&left, &right, &rational_zero(), false) {
1708                    TruthValue::Yes
1709                } else if self.edge_entails(&right, &left, &rational_zero(), true) {
1710                    TruthValue::No
1711                } else {
1712                    TruthValue::Unknown
1713                }
1714            }
1715            ScalarRelationOperator::NotEqual => TruthValue::Unknown,
1716        }
1717    }

1719    fn edge_entails(
1720        &self,
1721        left: &StructuralSubjectKey,
1722        right: &StructuralSubjectKey,
1723        offset: &Rational,
1724        strict: bool,
1725    ) -> bool {
1726        let direct = self.edges.get(&(left.clone(), right.clone()));
1727        if direct.is_some_and(|edge| edge_entails_request(edge, offset, strict)) {
1728            return true;
1729        }
1730        if let Some(left_value) = rational_from_key(left) {
1731            for (numeric_key, numeric_value) in &self.numeric_values {
1732                if let Some(edge) = self.edges.get(&(numeric_key.clone(), right.clone())) {
1733                    let translated = OrderEdge {
1734                        offset: numeric_value.clone() - left_value.clone() + edge.offset.clone(),
1735                        strict: edge.strict,
1736                    };
1737                    if edge_entails_request(&translated, offset, strict) {
1738                        return true;
1739                    }
1740                }
1741            }
1742        }
1743        if let Some(right_value) = rational_from_key(right) {
1744            for (numeric_key, numeric_value) in &self.numeric_values {
1745                if let Some(edge) = self.edges.get(&(left.clone(), numeric_key.clone())) {
1746                    let translated = OrderEdge {
1747                        offset: edge.offset.clone() + right_value.clone() - numeric_value.clone(),
1748                        strict: edge.strict,
1749                    };
1750                    if edge_entails_request(&translated, offset, strict) {
1751                        return true;
1752                    }
1753                }
1754            }
1755        }
1756        false
1757    }

1761            && self.edge_entails(right, left, &rational_zero(), false)
1762    }

1789fn normalized_affine_relation(relation: &CanonicalScalarRelation) -> Option<NormalizedRelation> {
1790    let mut difference = affine_expression(&relation.left);
1791    difference.subtract(affine_expression(&relation.right));
1792    if difference.terms.is_empty() {
1793        let ordering = difference.constant.cmp(&rational_zero());
1794        return Some(NormalizedRelation::Constant(relation_truth(
1795            relation.operator,
1796            ordering,
1797        )));
1798    }
1799    let strict = relation.operator == ScalarRelationOperator::Less;
1800    let equality = relation.operator == ScalarRelationOperator::Equal;
1801    if relation.operator == ScalarRelationOperator::NotEqual {
1802        return None;
1803    }
1804    let terms = difference.terms.into_iter().collect::<Vec<_>>();
1805    match terms.as_slice() {
1806        [(subject, coefficient)] => {
1807            let boundary = -difference.constant / coefficient.clone();
1808            let boundary_key = rational_key(&boundary);
1809            if coefficient > &rational_zero() {
1810                Some(NormalizedRelation::Edge {
1811                    left: subject.clone(),
1812                    right: boundary_key,
1813                    offset: rational_zero(),
1814                    strict,
1815                    equality,
1816                })
1817            } else {
1818                Some(NormalizedRelation::Edge {
1819                    left: boundary_key,
1820                    right: subject.clone(),
1821                    offset: rational_zero(),
1822                    strict,
1823                    equality,
1824                })
1825            }
1826        }
1827        [
1828            (first_subject, first_coefficient),
1829            (second_subject, second_coefficient),
1830        ] if first_coefficient.clone() + second_coefficient.clone() == rational_zero() => {
1831            let offset = difference.constant / first_coefficient.clone();
1832            if first_coefficient > &rational_zero() {
1833                Some(NormalizedRelation::Edge {
1834                    left: first_subject.clone(),
1835                    right: second_subject.clone(),
1836                    offset,
1837                    strict,
1838                    equality,
1839                })
1840            } else {
1841                Some(NormalizedRelation::Edge {
1842                    left: second_subject.clone(),
1843                    right: first_subject.clone(),
1844                    offset: -offset,
1845                    strict,
1846                    equality,
1847                })
1848            }
1849        }
1850        _ => None,
1851    }
1852}

1857    constant: Rational,

1860impl AffineForm {
1861    fn subtract(&mut self, other: AffineForm) {
1862        self.constant -= other.constant;
1863        for (subject, coefficient) in other.terms {
1864            self.terms
1865                .entry(subject)
1866                .and_modify(|existing| *existing -= coefficient.clone())
1867                .or_insert(-coefficient);
1868        }
1869        self.terms
1870            .retain(|_, coefficient| *coefficient != rational_zero());
1871    }

1873    fn scale(&mut self, coefficient: Rational) {
1874        self.constant *= coefficient.clone();
1875        for value in self.terms.values_mut() {
1876            *value *= coefficient.clone();
1877        }
1878    }

1881        self.constant += other.constant;
1882        for (subject, coefficient) in other.terms {
1883            self.terms
1884                .entry(subject)
1885                .and_modify(|existing| *existing += coefficient.clone())
1886                .or_insert(coefficient);
1887        }
1888        self.terms
1889            .retain(|_, coefficient| *coefficient != rational_zero());
1890    }

1893fn affine_expression(expression: &ExpressionReference) -> AffineForm {
1894    if let Some(value) = exact_rational(expression) {
1895        return AffineForm {
1896            terms: HashMap::new(),
1897            constant: value,
1898        };
1899    }
1900    match &expression.kind {
1901        ExpressionKind::Sum(terms) => {
1902            let mut result = AffineForm::default();
1903            for term in terms {
1904                result.add(affine_expression(term));
1905            }
1906            result
1907        }
1908        ExpressionKind::Product(factors) => {
1909            let mut coefficient = Rational::from(Integer::from(1));
1910            let mut nonnumeric_factor = None;
1911            for factor in factors {
1912                if let Some(value) = exact_rational(factor) {
1913                    coefficient *= value;
1914                } else if nonnumeric_factor.is_none() {
1915                    nonnumeric_factor = Some(factor);
1916                } else {
1917                    return atomic_affine_expression(expression);
1918                }
1919            }
1920            let mut result =
1921                nonnumeric_factor
1922                    .map(affine_expression)
1923                    .unwrap_or_else(|| AffineForm {
1924                        terms: HashMap::new(),
1925                        constant: Rational::from(Integer::from(1)),
1926                    });
1927            result.scale(coefficient);
1928            result
1929        }
1930        _ => atomic_affine_expression(expression),
1931    }
1932}

1934fn affine_expression_is_supported(expression: &ExpressionReference) -> bool {
1935    if exact_rational(expression).is_some() {
1936        return true;
1937    }
1938    match &expression.kind {
1939        ExpressionKind::Symbol(_) => true,
1940        ExpressionKind::Sum(terms) => terms.iter().all(affine_expression_is_supported),
1941        ExpressionKind::Product(factors) => {
1942            let nonnumeric_factors = factors
1943                .iter()
1944                .filter(|factor| exact_rational(factor).is_none())
1945                .collect::<Vec<_>>();
1946            nonnumeric_factors.len() <= 1
1947                && nonnumeric_factors
1948                    .first()
1949                    .is_none_or(|factor| affine_expression_is_supported(factor))
1950        }
1951        _ => false,
1952    }
1953}

1958        constant: rational_zero(),

2066                AffineIntegerForm {
2067                    terms: HashMap::new(),

2069                }
2070            })
2071        }

2081    let divisor = integer_gcd(constraint.coefficient.clone(), constraint.modulus.clone());

2083        return None;
2084    }

2087            subject: constraint.subject,
2088            residue: Integer::from(0),
2089            modulus: Integer::from(1),

2093    let reduced_modulus = constraint.modulus / divisor.clone();

2096        positive_remainder(reduced_coefficient, reduced_modulus.clone()),
2097        reduced_modulus.clone(),

2142    left: SolvedCongruence,
2143    right: SolvedCongruence,
2144) -> Option<SolvedCongruence> {

2146        return None;
2147    }
2148    let divisor = integer_gcd(left.modulus.clone(), right.modulus.clone());

2151        return None;

2156        positive_remainder(left_reduced, right_reduced.clone()),
2157        right_reduced.clone(),
2158    )?;

2160        (residue_difference / divisor.clone()) * inverse,
2161        right_reduced,
2162    );

2164    Some(SolvedCongruence {
2165        subject: left.subject,
2166        residue: positive_remainder(left.residue + left.modulus * multiplier, modulus.clone()),
