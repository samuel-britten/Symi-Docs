4/// The matrix predicate vocabulary.
5pub enum MatrixPredicate {
6    /// A parameterized matrix shape proposition.
7    Shape,
8    /// The matrix is square.
9    Square,
10    /// Every entry is zero.
11    Zero,
12    /// The matrix is an identity matrix.
13    Identity,
14    /// Every off-diagonal entry is zero.
15    Diagonal,
16    /// Every entry below the main diagonal is zero.
17    UpperTriangular,
18    /// Every entry above the main diagonal is zero.
19    LowerTriangular,
20    /// The matrix equals its transpose.
21    Symmetric,
22    /// The matrix equals its conjugate transpose.
23    Hermitian,
24    /// The matrix is real and its transpose is its inverse.
25    Orthogonal,
26    /// The matrix is unitary over the finite complex field.
27    Unitary,
28    /// The matrix commutes with its conjugate transpose.
29    Normal,
30    /// The matrix has rank equal to the smaller dimension.
31    FullRank,
32    /// The square matrix has a two-sided inverse.
33    Invertible,
34    /// The square matrix is not invertible.
35    Singular,
36    /// Every entry is an integer.
37    IntegerEntries,
38    /// Every entry is finite real.
39    RealEntries,
40    /// Every entry is finite complex.
41    ComplexEntries,
42    /// The matrix is Hermitian positive definite.
43    PositiveDefinite,
44    /// The matrix is Hermitian positive semidefinite.
45    PositiveSemidefinite,
46}

49/// The typed argument signature of a matrix predicate.
50pub enum MatrixPredicateSignature {
51    /// A predicate over an immutable matrix value.
52    Matrix,
53    /// A predicate over a matrix value and two positive dimensions.
54    MatrixWithShape,
55}

67/// One declarative matrix-theory rule.
68pub enum MatrixRule {
69    /// An unconditional implication between predicates on one subject.
70    Implication {
71        /// The antecedent predicate.
72        antecedent: MatrixPredicate,
73        /// The consequent predicate.
74        consequent: MatrixPredicate,
75    },
76    /// Two predicates cannot both hold for one subject.
77    Exclusion {
78        /// The first excluded predicate.
79        left: MatrixPredicate,
80        /// The second excluded predicate.
81        right: MatrixPredicate,
82    },
83}

384/// A typed matrix predicate application over one immutable subject.
385#[derive(Debug, Clone, PartialEq, Eq, Hash)]
386pub struct MatrixPredicateApplication {
387    /// The registered predicate.
388    pub predicate: MatrixPredicate,
389    /// The immutable subject value.
390    pub subject: MatrixSubject,
391    /// Shape parameters, required only by `matrix_shape`.
392    pub shape: Option<(usize, usize)>,
393}
394
395impl MatrixPredicateApplication {
396    /// Construct a non-parameterized matrix predicate application.
397    pub fn new(predicate: MatrixPredicate, subject: MatrixSubject) -> Option<Self> {
398        if predicate.signature() != MatrixPredicateSignature::Matrix {
399            return None;
400        }
401        Some(MatrixPredicateApplication {
402            predicate,
403            subject,
404            shape: None,
405        })
406    }
407
408    /// Construct a shape predicate after validating its positive dimensions.
409    pub fn shape(subject: MatrixSubject, rows: usize, columns: usize) -> Option<Self> {
410        if rows == 0 || columns == 0 {
411            return None;
412        }
413        Some(MatrixPredicateApplication {
414            predicate: MatrixPredicate::Shape,
415            subject,
416            shape: Some((rows, columns)),
417        })
418    }
