221/// How a call's arguments are laid out for analytic classification.
222///
223/// The unary rule is the shape every built-in registered before Special
224/// Functions A uses: one scalar argument that is also the analytic variable.
225/// The parameterized rules describe calls whose analytic variable sits behind
226/// ordered parameter groups, so that the shared definedness path can locate the
227/// analytic argument and validate the non-analytic slots instead of assuming
228/// that argument zero is the one to classify.
229#[derive(Debug, Clone, Copy, PartialEq, Eq)]
230pub(crate) enum CallShape {
231    /// One scalar argument, which is also the analytic variable.
232    UnaryAnalyticArgument,
233    /// Two ordered parameter groups followed by the scalar analytic argument.
234    GeneralizedHypergeometric,
235    /// Four ordered parameter groups followed by the scalar analytic argument.
236    MeijerG,
237    /// Scalar parameter slots followed by one scalar analytic argument.
238    ScalarParameterizedAnalyticArgument {
239        /// The zero-based slot containing the analytic argument.
240        analytic_argument_index: usize,
241    },
242}

270            CallShape::GeneralizedHypergeometric => 2,
271            CallShape::MeijerG => 4,
272            CallShape::ScalarParameterizedAnalyticArgument { .. } => 0,
273        }
274    }

318const fn entry(
319    function: BuiltinFunction,
320    arity: usize,
321    point_domain: PointDomainRule,
322    real_domain: RealRegionSpecification,
323    real_range: RealRegionSpecification,
324    poles: PoleSource,
325    branch_cut: BranchCutSpecification,
326    analytic_class: AnalyticClass,
327    reference: &'static str,
328) -> FunctionSpecification {
329    // Continuity follows the analytic classification unless an entry overrides
330    // it: an entire or meromorphic built-in is continuous wherever it is
331    // defined, a principal branch is continuous off its chosen cut, and a
332    // real-valued-only entry defaults to continuous so that the three genuinely
333    // discontinuous built-ins have to name their exception explicitly.
334    let continuity_exception = match analytic_class {
335        AnalyticClass::Entire | AnalyticClass::Meromorphic | AnalyticClass::RealValuedOnly => {
336            ContinuityException::None
337        }
338        AnalyticClass::PrincipalBranch => ContinuityException::ChosenBranchCut,
339        AnalyticClass::Unsupported => ContinuityException::Unsupported,
340    };
341    FunctionSpecification {
342        function,
343        arity,
344        call_shape: CallShape::UnaryAnalyticArgument,
345        point_domain,
346        real_domain,
347        real_range,
348        poles,
349        branch_cut,
350        analytic_class,
351        continuity_exception,
352        reference,
353    }
354}

390/// The analytic class of one parameterized call, derived from the whole call.
391pub(crate) fn full_call_analytic_class(
392    function: BuiltinFunction,
393    shape: CallShape,
394    arguments: &[crate::expression::ExpressionReference],
395    table: &SymbolTable,
396) -> AnalyticClass {
397    // The parameter-sensitive companion of the table's `analytic_class` field.
398    //
399    // For a head whose class depends on its parameters, the table records
400    // `unsupported` and the class is derived here from the full call instead.
401    // The dispatch key is the declared `call_shape`, not the built-in's name, so
402    // every future parameterized family answers through the same seam rather
403    // than through a branch naming one function.
404    //
405    // The family module owns the mathematics and supplies the fact; this layer
406    // owns the judgment and the class vocabulary it is expressed in. Only a
407    // class that is proved for the whole finite plane is reported: an argument
408    // region where the claim would need a disk or a cut answers `unsupported`,
409    // which is an honest absence of a claim rather than a negative one.
410    match shape {
411        CallShape::UnaryAnalyticArgument => AnalyticClass::Unsupported,
412        CallShape::GeneralizedHypergeometric => {
413            let Some((upper_parameters, lower_parameters, _)) =
414                crate::special_functions::hypergeometric::parameter_analysis::split_call_arguments(
415                    arguments,
416                )
417            else {
418                return AnalyticClass::Unsupported;
419            };
420            if crate::special_functions::hypergeometric::is_admissible_and_entire(
421                upper_parameters,
422                lower_parameters,
423            ) {
424                AnalyticClass::Entire
425            } else {
426                AnalyticClass::Unsupported
427            }
428        }
429        // The first-tier Meijer-G contract states no regularity claim that holds
430        // for every call of the head. A configuration whose contour existence is
431        // unresolved has no analytic class at all, and one that reduces has the
432        // class of its reduction rather than of the G head.
433        CallShape::MeijerG => AnalyticClass::Unsupported,
434        CallShape::ScalarParameterizedAnalyticArgument { .. } => {
435            if crate::special_functions::orthogonal_polynomials::family_of(function).is_some() {
436                return orthogonal_polynomial_analytic_class(function, arguments, table);
437            }
438            if function == BuiltinFunction::Polylogarithm {
439                return polylogarithm_analytic_class(arguments);
440            }
441            if function == BuiltinFunction::LerchPhi {
442                return crate::special_functions::lerch_phi::classification::analytic_class(
443                    arguments,
444                );
445            }
446            if crate::special_functions::jacobi_theta::ThetaIndex::of_builtin(function).is_some() {
447                return theta_analytic_class(arguments, table);
448            }
449            if matches!(
450                function,
451                BuiltinFunction::JacobiSn
452                    | BuiltinFunction::JacobiCn
453                    | BuiltinFunction::JacobiDn
454                    | BuiltinFunction::JacobiAmplitude
455            ) {
456                return jacobi_analytic_class(function, arguments, table);
457            }
458            parameterized_bessel_analytic_class(function, arguments)
459        }
460    }
461}

463/// Classify exact exceptional full-call domains for scalar-parameterized heads.
464pub(crate) fn full_call_domain_classification(
465    function: BuiltinFunction,
466    arguments: &[crate::expression::ExpressionReference],
467) -> FullCallDomainClassification {
468    use BuiltinFunction::*;
469    let is_zero_at = |index: usize| {
470        arguments.get(index).is_some_and(|argument| {
471            matches!(&argument.kind, ExpressionKind::Integer(value) if *value == exact_integer::ZERO)
472        })
473    };
474    let is_one_at = |index: usize| {
475        arguments.get(index).is_some_and(|argument| {
476            matches!(&argument.kind, ExpressionKind::Integer(value) if *value == exact_integer::ONE)
477        })
478    };
479    let is_undefined = match function {
480        CarlsonRf => {
481            (0..arguments.len())
482                .filter(|index| is_zero_at(*index))
483                .count()
484                >= 2
485        }
486        CarlsonRc => is_zero_at(1),
487        CarlsonRd => is_zero_at(2),
488        CarlsonRj => is_zero_at(3),
489        EllipticK => is_one_at(0),
490        // The complete third kind has a nonintegrable endpoint pole whenever
491        // its characteristic is one; it is not restricted to the m = 0
492        // degeneration. Principal values are deliberately a separate API.
493        EllipticPi => is_one_at(0),
494        // The incomplete integral is regular at zero amplitude, even for n =
495        // 1. Its endpoint pole is represented only when the exact endpoint is
496        // available to this registry; symbolic amplitudes remain unresolved.
497        EllipticPiIncomplete => {
498            is_one_at(0)
499                && arguments.get(1).is_some_and(|argument| {
500                    matches!(
501                        &argument.kind,
502                        ExpressionKind::Constant(crate::expression::BuiltinConstant::Pi)
503                    )
504                })
505        }
506        // The three principal Jacobi functions share the simple poles of their period
507        // lattice. Only a structurally exact lattice point counts; the family module
508        // owns the recognition and never lets a small numeric denominator stand in for
509        // it.
510        JacobiSn | JacobiCn | JacobiDn => {
511            arguments.len() == 2
512                && crate::special_functions::jacobi_elliptic::classification::proves_period_lattice_pole(
513                    &arguments[0],
514                    &arguments[1],
515                )
516        }
517        _ => false,
518    };
519    if is_undefined {
520        FullCallDomainClassification::Undefined
521    } else {
522        FullCallDomainClassification::Unresolved
523    }
524}

539fn orthogonal_polynomial_analytic_class(
540    function: BuiltinFunction,
541    arguments: &[crate::expression::ExpressionReference],
542    table: &SymbolTable,
543) -> AnalyticClass {
544    // A classical orthogonal polynomial is a finite polynomial in its argument,
545    // and therefore entire in it, exactly when its degree is established as a
546    // non-negative integer. The coefficients of every family currently supported
547    // are polynomial in the continuous family parameters as well, so no
548    // parameter configuration can introduce a pole or a cut in the analytic
549    // slot. A degree that is not established leaves the head with no analytic
550    // claim at all, which is an absence of a claim rather than a negative one:
551    // the orthogonality hypotheses on the parameters are theorem conditions and
552    // are deliberately not consulted here.
553    let Some(family) = crate::special_functions::orthogonal_polynomials::family_of(function) else {
554        return AnalyticClass::Unsupported;
555    };
556    if arguments.len() != family.arity() {
557        return AnalyticClass::Unsupported;
558    }
559    if crate::special_functions::orthogonal_polynomials::classify_degree(&arguments[0], table)
560        .is_admissible()
561    {
562        AnalyticClass::Entire
563    } else {
564        AnalyticClass::Unsupported
565    }
566}

568fn theta_analytic_class(
569    arguments: &[crate::expression::ExpressionReference],
570    table: &SymbolTable,
571) -> AnalyticClass {
572    // For a half-period ratio in the open upper half plane the nome has modulus below
573    // one, the theta Fourier series converges uniformly on every compact set, and each
574    // theta function is entire in its argument. The family module owns the evidence
575    // that the ratio lies there; the class it earns is declared here. Outside that
576    // established region the head carries no analytic claim, which is an absence of a
577    // claim rather than a negative one.
578    if arguments.len() != 2 {
579        return AnalyticClass::Unsupported;
580    }
581    if crate::special_functions::jacobi_theta::classification::proves_upper_half_plane(
582        &arguments[1],
583        table,
584    ) {
585        AnalyticClass::Entire
586    } else {
587        AnalyticClass::Unsupported
588    }
589}

591fn jacobi_analytic_class(
592    function: BuiltinFunction,
593    arguments: &[crate::expression::ExpressionReference],
594    table: &SymbolTable,
595) -> AnalyticClass {
596    // For a fixed admissible parameter the three principal Jacobi functions are doubly
597    // periodic and holomorphic away from the period lattice, so their claim is
598    // meromorphy in the argument and the pole set is the lattice the family module
599    // recognises. The amplitude is a different function: it inverts an incomplete
600    // integral and is single-valued only on the sheet the convention pins, so it earns
601    // a branch claim on its proved real principal region and nothing at all elsewhere.
602    // It never inherits the meromorphic claim of the sine amplitude.
603    if arguments.len() != 2 {
604        return AnalyticClass::Unsupported;
605    }
606    if function == BuiltinFunction::JacobiAmplitude {
607        return if crate::special_functions::jacobi_elliptic::classification::proves_real_principal_amplitude_region(
608            &arguments[0],
609            &arguments[1],
610            table,
611        ) {
612            AnalyticClass::PrincipalBranch
613        } else {
614            AnalyticClass::Unsupported
615        };
616    }
617    if crate::special_functions::jacobi_elliptic::classification::proves_admissible_parameter(
618        &arguments[1],
619        table,
620    ) {
621        AnalyticClass::Meromorphic
622    } else {
623        AnalyticClass::Unsupported
624    }
625}

627fn parameterized_bessel_analytic_class(
628    function: BuiltinFunction,
629    arguments: &[crate::expression::ExpressionReference],
630) -> AnalyticClass {
631    if arguments.len() != 2 {
632        return AnalyticClass::Unsupported;
633    }
634    let Some(order) = arguments.first().and_then(literal_integer) else {
635        return match function {
636            BuiltinFunction::BesselJ
637            | BuiltinFunction::BesselY
638            | BuiltinFunction::BesselI
639            | BuiltinFunction::BesselK
640            | BuiltinFunction::HankelFirstKind
641            | BuiltinFunction::HankelSecondKind => AnalyticClass::PrincipalBranch,
642            _ => AnalyticClass::Unsupported,
643        };
644    };
645    match function {
646        BuiltinFunction::BesselJ | BuiltinFunction::BesselI => AnalyticClass::Entire,
647        BuiltinFunction::SphericalBesselJ if order >= exact_integer::ZERO => AnalyticClass::Entire,
648        BuiltinFunction::SphericalBesselY
649        | BuiltinFunction::SphericalHankelFirstKind
650        | BuiltinFunction::SphericalHankelSecondKind
651            if order >= exact_integer::ZERO =>
652        {
653            AnalyticClass::Meromorphic
654        }
655        BuiltinFunction::BesselY
656        | BuiltinFunction::BesselK
657        | BuiltinFunction::HankelFirstKind
658        | BuiltinFunction::HankelSecondKind => AnalyticClass::PrincipalBranch,
659        _ => AnalyticClass::Unsupported,
660    }
661}

1570        // A range is bounded exactly when it is one interval with two finite

1578        ) && !matches!(
1579            interval.upper,

1581        )
