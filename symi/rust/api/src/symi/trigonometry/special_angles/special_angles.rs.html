15/// Performs the advanced native `pi_multiple_of` operation.
16pub fn pi_multiple_of(argument: &ExpressionReference) -> Option<Rational> {
17    // GG_addendum_4.1 pi-multiple recogniser. Returns Some(r) when the
18    // argument equals the literal `r * pi` (or the bare pi constant, for
19    // which r = 1). Returns None for any other shape so the caller may fall
20    // through to its structural arms. The match is shape-strict: only the
21    // canonical ASAE forms `constant(pi)` and `product([numeric, constant(pi)])`
22    // are recognised. Inputs whose numeric coefficient is folded into a
23    // larger product (e.g. `product([2, pi, x])`) are deliberately rejected
24    // because they do not represent a pure pi-multiple — the residual
25    // factor would have to participate in the table lookup.
26    if is_pi_constant(argument) {
27        return Some(Rational::ONE);
28    }
29    let ExpressionKind::Product(factors) = &argument.kind else {
30        return None;
31    };
32    if factors.len() != 2 {
33        return None;
34    }
35    if !is_pi_constant(&factors[1]) {
36        return None;
37    }
38    let numeric = extract_number(&factors[0])?;
39    Some(match numeric {
40        Number::Integer(value) => Rational::from(value),
41        Number::Rational(value) => value,
42    })
43}

49fn mentions_pi(target: &ExpressionReference) -> bool {
50    let mut found = is_pi_constant(target);
51    crate::expression::dag::for_each_child(target, |child| {
52        found |= mentions_pi(child);
53    });
54    found
55}

57fn is_pi_constant(argument: &ExpressionReference) -> bool {
58    // True iff the argument is the canonical interned `constant(pi)` leaf.
59    // Centralised so the variant name appears in only one place.
60    matches!(
61        &argument.kind,
62        ExpressionKind::Constant(BuiltinConstant::Pi)
63    )
64}

66/// Performs the advanced native `simplify_pi_multiple_call` operation.
67pub fn simplify_pi_multiple_call(
68    context: &Context,
69    builtin: BuiltinFunction,
70    argument: &ExpressionReference,
71) -> Option<ExpressionReference> {
72    // GG_addendum_4.5 entry point for the auto-simplifier. Recognises the
73    // pi-multiple shape, applies the Fu TR3 reflection / parity / periodicity
74    // reductions to canonicalise into the first quadrant, then performs the
75    // Fu TR4 table lookup. Returns Some(value) when the table contains the
76    // reduced angle and None otherwise so the caller falls through to the
77    // structural simplification path. The call is restricted to the six
78    // direct trigonometric builtins; the inverse and hyperbolic families
79    // are deferred until the corresponding special-value tables ship.
80    if !is_table_supported(builtin) {
81        return None;
82    }
83    let multiplier = pi_multiple_of(argument)?;
84    evaluate_via_table(context, builtin, &multiplier)
85}

87/// Performs the advanced native `simplify_integer_shifted_pi_multiple_call` operation.
88pub fn simplify_integer_shifted_pi_multiple_call(
89    context: &Context,
90    builtin: BuiltinFunction,
91    argument: &ExpressionReference,
92) -> Option<ExpressionReference> {
93    // The table above reads a numeric multiple of pi. This reads a multiple that is numeric only
94    // after a part of it has been proved to be an integer: `sin(m pi) = 0` and
95    // `cos(m pi) = (-1)^m` for every integer `m`, so the angle-addition identities give
96    // `sin(m pi + r pi) = (-1)^m sin(r pi)` and `cos(m pi + r pi) = (-1)^m cos(r pi)`.
97    //
98    // That is the identity an indexed eigenvalue family stands on. The endpoint determinant of a
99    // Sturm-Liouville problem vanishes at its own eigenvalues only because the mode index is an
100    // integer, and with the index left as a free symbol that fact is invisible: the family can be
101    // sampled at one index after another without the general statement ever being proved. The
102    // reduction here is what makes it an identity in the index instead.
103    if !matches!(builtin, BuiltinFunction::Sin | BuiltinFunction::Cos) {
104        return None;
105    }
106    let (integer_parts, rational_multiplier) = split_pi_coefficient(context, argument)?;
107    if integer_parts.is_empty() {
108        return None;
109    }
110    let value = evaluate_via_table(context, builtin, &rational_multiplier)?;
111    let minus_one = builders::integer(context, Integer::from(-1));
112    let exponent = builders::sum(context, integer_parts);
113    let parity = builders::power(context, minus_one, exponent);
114    Some(builders::product(context, vec![parity, value]))
115}

117fn split_pi_coefficient(
118    context: &Context,
119    argument: &ExpressionReference,
120) -> Option<(Vec<ExpressionReference>, Rational)> {
121    // Splits the angle into the multiples of pi it is a sum of, separating the summands proved to
122    // be integer multiples from the numeric remainder. A summand that is neither declines the
123    // whole reduction, because a part of unknown integrality carries no periodicity statement at
124    // all. The reading is structural rather than a division by pi followed by a simplification:
125    // this runs inside the auto-simplifier's own function-call arm, and re-entering the
126    // simplifier from there is neither cheap nor safe.
127    let summands: Vec<ExpressionReference> = match &argument.kind {
128        ExpressionKind::Sum(children) => children.iter().map(Arc::clone).collect(),
129        _ => vec![Arc::clone(argument)],
130    };
131    let mut integer_parts = Vec::new();
132    let mut rational_multiplier = Rational::ZERO;
133    for summand in summands {
134        let coefficient = pi_coefficient_of_summand(context, &summand)?;
135        let parts: Vec<ExpressionReference> = match &coefficient.kind {
136            ExpressionKind::Sum(children) => children.iter().map(Arc::clone).collect(),
137            _ => vec![coefficient],
138        };
139        for part in parts {
140            if let Some(numeric) = extract_number(&part) {
141                rational_multiplier = rational_multiplier
142                    + match numeric {
143                        Number::Integer(value) => Rational::from(value),
144                        Number::Rational(value) => value,
145                    };
146                continue;
147            }
148            if is_integer(&part, &context.symbol_table) != TruthValue::Yes {
149                return None;
150            }
151            integer_parts.push(part);
152        }
153    }
154    Some((integer_parts, rational_multiplier))
155}

157fn pi_coefficient_of_summand(
158    context: &Context,
159    summand: &ExpressionReference,
160) -> Option<ExpressionReference> {
161    // The factor a single summand multiplies pi by, or `None` when the summand is not a multiple
162    // of pi at all. A summand carrying pi more than once is not a multiple of it either.
163    if is_pi_constant(summand) {
164        return Some(builders::one(context));
165    }
166    let ExpressionKind::Product(factors) = &summand.kind else {
167        return None;
168    };
169    let mut remaining = Vec::with_capacity(factors.len());
170    let mut pi_occurrences = 0usize;
171    for factor in factors {
172        if is_pi_constant(factor) {
173            pi_occurrences += 1;
174            continue;
175        }
176        if mentions_pi(factor) {
177            return None;
178        }
179        remaining.push(Arc::clone(factor));
180    }
181    if pi_occurrences != 1 {
182        return None;
183    }
184    Some(builders::product(context, remaining))
185}

187/// Performs the advanced native `simplify_inverse_special_value_call` operation.
188pub fn simplify_inverse_special_value_call(
189    context: &Context,
190    builtin: BuiltinFunction,
191    argument: &ExpressionReference,
192) -> Option<ExpressionReference> {
193    // Inverse counterpart to simplify_pi_multiple_call: collapse
194    // Arcsin / Arccos / Arctan of a special value into the principal-branch
195    // angle pi * rational. The lookup is performed by running the forward
196    // table over the small fixed set of candidate principal angles and keeping
197    // the one whose forward value is provably Equal to the argument (their
198    // difference auto-simplifies to the zero literal). Returning the angle only
199    // on an exact structural cancellation keeps the rewrite rigorous; a value
200    // that merely resembles a table entry numerically is left unevaluated.
201    // Arguments containing a free variable are skipped because a special value
202    // is necessarily constant.
203    let (forward_builtin, candidate_multipliers) = inverse_table_specification(builtin)?;
204    if !crate::expression::free_variables::free_variables(argument).is_empty() {
205        return None;
206    }
207    for (numerator, denominator) in candidate_multipliers {
208        let angle = build_pi_multiple(context, *numerator, *denominator);
209        let forward_value = match simplify_pi_multiple_call(context, forward_builtin, &angle) {
210            Some(value) => value,
211            None => continue,
212        };
213        let difference = builders::subtract(context, forward_value, Arc::clone(argument));
214        if matches!(
215            &difference.kind,
216            ExpressionKind::Integer(value) if *value == Integer::ZERO
217        ) {
218            return Some(angle);
219        }
220    }
221    None
222}

224fn inverse_table_specification(
225    builtin: BuiltinFunction,
226) -> Option<(BuiltinFunction, &'static [(i64, i64)])> {
227    // Map each supported inverse builtin to its forward partner and the list of
228    // principal-branch angle multipliers of pi to probe. The ranges follow the
229    // standard principal branches: Arcsin and Arctan over (-pi/2, pi/2) (Arcsin
230    // including the endpoints), Arccos over [0, pi]. Tan is undefined at the
231    // half-pi endpoints, so they are excluded from the Arctan list.
232    use BuiltinFunction::*;
233    let specification = match builtin {
234        Arcsin => (
235            Sin,
236            &[
237                (-1, 2),
238                (-1, 3),
239                (-1, 4),
240                (-1, 6),
241                (0, 1),
242                (1, 6),
243                (1, 4),
244                (1, 3),
245                (1, 2),
246            ][..],
247        ),
248        Arccos => (
249            Cos,
250            &[
251                (0, 1),
252                (1, 6),
253                (1, 4),
254                (1, 3),
255                (1, 2),
256                (2, 3),
257                (3, 4),
258                (5, 6),
259                (1, 1),
260            ][..],
261        ),
262        Arctan => (
263            Tan,
264            &[(-1, 3), (-1, 4), (-1, 6), (0, 1), (1, 6), (1, 4), (1, 3)][..],
265        ),
266        _ => return None,
267    };
268    Some(specification)
269}

271fn build_pi_multiple(context: &Context, numerator: i64, denominator: i64) -> ExpressionReference {
272    // Construct the angle expression (numerator / denominator) * pi in canonical
273    // ASAE form: the zero multiplier collapses to the integer zero, the unit
274    // multiplier to the bare pi constant, and every other multiplier to the
275    // product of its rational literal with pi.
276    if numerator == 0 {
277        return builders::zero(context);
278    }
279    let pi_expression = context
280        .interner
281        .intern(ExpressionKind::Constant(BuiltinConstant::Pi));
282    if numerator == denominator {
283        return pi_expression;
284    }
285    let coefficient = if denominator == 1 {
286        builders::integer_value(context, numerator)
287    } else {
288        builders::rational_value(context, numerator, denominator)
289    };
290    builders::product(context, vec![coefficient, pi_expression])
291}

293fn is_table_supported(builtin: BuiltinFunction) -> bool {
294    // Membership predicate over the six direct trigonometric builtins whose
295    // special-angle table ships with GG_addendum_4. Hyperbolic and inverse
296    // families are deferred — they appear in the broader Fu architecture
297    // only once a matching special-value table is added in a later round.
298    matches!(
299        builtin,
300        BuiltinFunction::Sin
301            | BuiltinFunction::Cos
302            | BuiltinFunction::Tan
303            | BuiltinFunction::Sec
304            | BuiltinFunction::Csc
305            | BuiltinFunction::Cot
306    )
307}

309fn evaluate_via_table(
310    context: &Context,
311    builtin: BuiltinFunction,
312    multiplier: &Rational,
313) -> Option<ExpressionReference> {
314    // Apply Fu TR3 to canonicalise the multiplier into the first quadrant,
315    // pick up the sign, and perform the TR4 table lookup at the reduced
316    // first-quadrant value. A table miss returns None so the caller leaves
317    // the call symbolic; a hit on a singularity row returns the undefined
318    // sentinel verbatim because the sign cannot be propagated through an
319    // unsigned limit. The two period classes are split — Sin / Cos / Sec /
320    // Csc share the 2*pi periodicity of Cos, and Tan / Cot share the pi
321    // periodicity of Tan.
322    let (sign, first_quadrant) = match builtin {
323        BuiltinFunction::Sin
324        | BuiltinFunction::Cos
325        | BuiltinFunction::Sec
326        | BuiltinFunction::Csc => reduce_for_two_pi_family(builtin, multiplier),
327        BuiltinFunction::Tan | BuiltinFunction::Cot => reduce_for_pi_family(builtin, multiplier),
328        _ => return None,
329    };
330    let raw_value = table_lookup(context, builtin, &first_quadrant)?;
331    if is_undefined(&raw_value) {
332        return Some(raw_value);
333    }
334    if sign < 0 {
335        Some(builders::negate(context, raw_value))
336    } else {
337        Some(raw_value)
338    }
339}

341fn reduce_for_two_pi_family(builtin: BuiltinFunction, multiplier: &Rational) -> (i32, Rational) {
342    // TR3 reduction for the 2*pi-periodic family (Sin / Cos / Sec / Csc).
343    // The multiplier r is reduced modulo 2 to bring it into [0, 2). The
344    // resulting r' is then split into the four quadrants of [0, 2*pi):
345    //   r' in [0, 1/2]   → first  quadrant, every builtin keeps its sign.
346    //   r' in (1/2, 1]   → second quadrant, cos/sec flip, sin/csc keep.
347    //   r' in (1, 3/2]   → third  quadrant, both pairs flip.
348    //   r' in (3/2, 2)   → fourth quadrant, sin/csc flip, cos/sec keep.
349    // The first-quadrant residual phi is computed so the table lookup at the
350    // bottom always sees a value in [0, 1/2]: r' itself in the first
351    // quadrant, 1 - r' in the second, r' - 1 in the third, 2 - r' in the
352    // fourth. The parity rules Sin(-x) = -Sin(x) and Cos(-x) = Cos(x) drop
353    // out of the same partition once the mod-2 reduction pulls negative
354    // inputs into [0, 2).
355    let two = Rational::from(Integer::from(2));
356    let reduced = reduce_mod(multiplier, &two);
357    let one = Rational::ONE;
358    let half = rational_one_over(2);
359    let three_halves = Rational::from_integers(Integer::from(3), Integer::from(2));
360    let (sign_sin_family, sign_cos_family, first_quadrant) = if reduced <= half {
361        (1, 1, reduced)
362    } else if reduced <= one {
363        let phi = &one - &reduced;
364        (1, -1, phi)
365    } else if reduced <= three_halves {
366        let phi = &reduced - &one;
367        (-1, -1, phi)
368    } else {
369        let phi = &two - &reduced;
370        (-1, 1, phi)
371    };
372    let sign = match builtin {
373        BuiltinFunction::Sin | BuiltinFunction::Csc => sign_sin_family,
374        BuiltinFunction::Cos | BuiltinFunction::Sec => sign_cos_family,
375        _ => 1,
376    };
377    (sign, first_quadrant)
378}

380fn reduce_for_pi_family(builtin: BuiltinFunction, multiplier: &Rational) -> (i32, Rational) {
381    // TR3 reduction for the pi-periodic family (Tan / Cot). The multiplier
382    // r is reduced modulo 1 to bring it into [0, 1). The resulting r' is
383    // then split into two halves:
384    //   r' in [0, 1/2]  → keep sign, table at r'.
385    //   r' in (1/2, 1)  → flip sign, table at 1 - r' (since
386    //                      Tan(pi - phi) = -Tan(phi) and Cot is its
387    //                      reciprocal so the same sign rule applies).
388    let one = Rational::ONE;
389    let reduced = reduce_mod(multiplier, &one);
390    let half = rational_one_over(2);
391    let (sign, first_quadrant) = if reduced <= half {
392        (1, reduced)
393    } else {
394        let phi = &one - &reduced;
395        (-1, phi)
396    };
397    let _ = builtin;
398    (sign, first_quadrant)
399}

401fn reduce_mod(value: &Rational, modulus: &Rational) -> Rational {
402    // Total reduction of a Rational into the half-open interval [0, modulus).
403    // The result equals value - Floor(value / modulus) * modulus so negative
404    // inputs land in the positive interval directly. Used by the TR3
405    // periodicity reductions to fold an arbitrary multiplier of pi into the
406    // canonical first-period representative.
407    let quotient_integer = (value / modulus).floor();
408    let quotient_rational = Rational::from(quotient_integer);
409    value - &(quotient_rational * modulus)
410}

412fn rational_one_over(denominator: i64) -> Rational {
413    // Convenience constructor for the canonical 1/n rational used by the TR3
414    // quadrant boundaries. Centralised so the exact-integer wrap appears
415    // in only one place.
416    Rational::from_integers(Integer::from(1), Integer::from(denominator))
417}

419fn table_lookup(
420    context: &Context,
421    builtin: BuiltinFunction,
422    first_quadrant: &Rational,
423) -> Option<ExpressionReference> {
424    // GG_addendum_4.2 special-angle value table. The five canonical
425    // first-quadrant rows cover {0, pi/6, pi/4, pi/3, pi/2}; any other
426    // reduced multiplier returns None so the call stays symbolic. Each
427    // returned value is built through the auto-simplifying builders so the
428    // reference is interned in ASAE form. The singularity rows return the
429    // undefined sentinel rather than a signed infinity because the
430    // one-sided limits disagree at every Tan / Sec / Csc / Cot pole.
431    let zero = Rational::ZERO;
432    let one_sixth = rational_one_over(6);
433    let one_quarter = rational_one_over(4);
434    let one_third = rational_one_over(3);
435    let one_half = rational_one_over(2);
436    let row = if *first_quadrant == zero {
437        SpecialAngleRow::Zero
438    } else if *first_quadrant == one_sixth {
439        SpecialAngleRow::OneSixth
440    } else if *first_quadrant == one_quarter {
441        SpecialAngleRow::OneQuarter
442    } else if *first_quadrant == one_third {
443        SpecialAngleRow::OneThird
444    } else if *first_quadrant == one_half {
445        SpecialAngleRow::OneHalf
446    } else {
447        return None;
448    };
449    Some(build_table_value(context, builtin, row))
450}

452#[derive(Clone, Copy)]
453enum SpecialAngleRow {
454    Zero,
455    OneSixth,
456    OneQuarter,
457    OneThird,
458    OneHalf,
459}

461fn build_table_value(
462    context: &Context,
463    builtin: BuiltinFunction,
464    row: SpecialAngleRow,
465) -> ExpressionReference {
466    // Materialise the canonical closed form for the supplied (builtin, row)
467    // pair. Each construction path routes through the auto-simplifying
468    // builders so the returned reference is interned in ASAE form and shares
469    // hash-cons identity with any Equivalent expression elsewhere in the
470    // session. The reciprocal-family entries (Sec / Csc / Cot) are
471    // hand-rolled rather than delegating to `power(primary_value, -1)` so
472    // that values such as `Cot(pi/6) = Sqrt(3)` land on the same canonical
473    // expression as `Tan(pi/3) = Sqrt(3)` — the auto-simplifier's product
474    // folder keeps the numeric coefficient separate from a `power(_, -k/2)`
475    // factor with the same integer base, which would otherwise produce
476    // structurally distinct representatives of the same value.
477    use SpecialAngleRow::*;
478    match (builtin, row) {
479        (BuiltinFunction::Sin, Zero) => builders::integer_value(context, 0),
480        (BuiltinFunction::Sin, OneSixth) => builders::rational_value(context, 1, 2),
481        (BuiltinFunction::Sin, OneQuarter) => build_half_times_square_root(context, 2),
482        (BuiltinFunction::Sin, OneThird) => build_half_times_square_root(context, 3),
483        (BuiltinFunction::Sin, OneHalf) => builders::integer_value(context, 1),
484        (BuiltinFunction::Cos, Zero) => builders::integer_value(context, 1),
485        (BuiltinFunction::Cos, OneSixth) => build_half_times_square_root(context, 3),
486        (BuiltinFunction::Cos, OneQuarter) => build_half_times_square_root(context, 2),
487        (BuiltinFunction::Cos, OneThird) => builders::rational_value(context, 1, 2),
488        (BuiltinFunction::Cos, OneHalf) => builders::integer_value(context, 0),
489        (BuiltinFunction::Tan, Zero) => builders::integer_value(context, 0),
490        (BuiltinFunction::Tan, OneSixth) => build_one_over_three_times_square_root_three(context),
491        (BuiltinFunction::Tan, OneQuarter) => builders::integer_value(context, 1),
492        (BuiltinFunction::Tan, OneThird) => build_square_root(context, 3),
493        (BuiltinFunction::Tan, OneHalf) => intern_undefined(context),
494        (BuiltinFunction::Sec, Zero) => builders::integer_value(context, 1),
495        (BuiltinFunction::Sec, OneSixth) => build_two_over_three_times_square_root_three(context),
496        (BuiltinFunction::Sec, OneQuarter) => build_square_root(context, 2),
497        (BuiltinFunction::Sec, OneThird) => builders::integer_value(context, 2),
498        (BuiltinFunction::Sec, OneHalf) => intern_undefined(context),
499        (BuiltinFunction::Csc, Zero) => intern_undefined(context),
500        (BuiltinFunction::Csc, OneSixth) => builders::integer_value(context, 2),
501        (BuiltinFunction::Csc, OneQuarter) => build_square_root(context, 2),
502        (BuiltinFunction::Csc, OneThird) => build_two_over_three_times_square_root_three(context),
503        (BuiltinFunction::Csc, OneHalf) => builders::integer_value(context, 1),
504        (BuiltinFunction::Cot, Zero) => intern_undefined(context),
505        (BuiltinFunction::Cot, OneSixth) => build_square_root(context, 3),
506        (BuiltinFunction::Cot, OneQuarter) => builders::integer_value(context, 1),
507        (BuiltinFunction::Cot, OneThird) => build_one_over_three_times_square_root_three(context),
508        (BuiltinFunction::Cot, OneHalf) => builders::integer_value(context, 0),
509        _ => intern_undefined(context),
510    }
511}

513fn build_half_times_square_root(context: &Context, radicand: i64) -> ExpressionReference {
514    // Build the canonical ASAE form `product([rational(1, 2), power(radicand, 1/2)])`
515    // for the recurring `Sqrt(radicand) / 2` entries in the special-angle
516    // table. The numeric coefficient stays as a rational leaf rather than
517    // being folded into the power exponent because the like-base combine
518    // rule only fires when both factors share a base — here the rational
519    // 1/2 and the radicand are different bases, so the auto-simplifier
520    // keeps them as a two-factor product.
521    let half = builders::rational_value(context, 1, 2);
522    let square_root_value = build_square_root(context, radicand);
523    builders::product(context, vec![half, square_root_value])
524}

526fn build_square_root(context: &Context, radicand: i64) -> ExpressionReference {
527    // Build `power(radicand, 1/2)`, the canonical ASAE shape produced by
528    // the `Sqrt` builtin's auto-simplifier arm.
529    let radicand_value = builders::integer_value(context, radicand);
530    let half = builders::rational_value(context, 1, 2);
531    builders::power(context, radicand_value, half)
532}

534fn build_one_over_three_times_square_root_three(context: &Context) -> ExpressionReference {
535    // Build the canonical ASAE form `product([rational(1, 3), power(3, 1/2)])`
536    // for the recurring `Sqrt(3) / 3` entries (Tan(pi/6) and Cot(pi/3)).
537    // Mirrors [build_half_times_square_root]; the auto-simplifier keeps the
538    // rational 1/3 and Sqrt(3) as separate factors because their bases
539    // differ.
540    let third = builders::rational_value(context, 1, 3);
541    let square_root_three = build_square_root(context, 3);
542    builders::product(context, vec![third, square_root_three])
543}

545fn build_two_over_three_times_square_root_three(context: &Context) -> ExpressionReference {
546    // Build `product([rational(2, 3), power(3, 1/2)])` — the canonical form
547    // of `2*Sqrt(3)/3` used by Sec(pi/6) and Csc(pi/3). The auto-simplifier
548    // keeps the rational coefficient separate from the radical for the same
549    // reason that [build_one_over_three_times_square_root_three] does.
550    let two_thirds = builders::rational_value(context, 2, 3);
551    let square_root_three = build_square_root(context, 3);
552    builders::product(context, vec![two_thirds, square_root_three])
553}
554
555fn intern_undefined(context: &Context) -> ExpressionReference {
556    // Centralised constructor for the undefined sentinel used at every
557    // singularity row of the table (Tan(pi/2), Sec(pi/2), Csc(0), Cot(0)).
558    context
559        .interner
560        .intern(ExpressionKind::Constant(BuiltinConstant::Undefined))
561}
562
563fn is_undefined(target: &ExpressionReference) -> bool {
564    // Predicate identifying the undefined sentinel. Used by [evaluate_via_table]
565    // to short-circuit the sign-application step on singularity rows because
566    // `negate(undefined)` is still undefined and the extra builder call would
567    // be wasted work.
568    matches!(
569        &target.kind,
570        ExpressionKind::Constant(BuiltinConstant::Undefined)
571    )
572}

574/// Performs the advanced native `walk_apply_special_angle_lookup` operation.
575pub fn walk_apply_special_angle_lookup(
576    context: &Context,
577    target: &ExpressionReference,
578) -> ExpressionReference {
579    // Fu candidate ladder entry that walks the expression bottom-up and
580    // applies the TR3 + TR4 pi-multiple recogniser to every trigonometric
581    // function call it encounters. The auto-simplifier in
582    // [simplify_function_call] runs the same recogniser at construction so
583    // a freshly-built expression already carries every fold this walker
584    // would produce; the walker is registered with the Fu greedy driver so
585    // any pi-multiple call that arrived through a code path that bypassed
586    // the auto-simplifier (e.g. a hand-constructed `function_call` AST
587    // node) still folds when the User explicitly calls
588    // [simplify_trigonometric].
589    use crate::simplify::{simplify_function_call, simplify_power, simplify_product, simplify_sum};
590    use ExpressionKind::*;
591    match &target.kind {
592        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => {
593            Arc::clone(target)
594        }
595        Sum(children) => {
596            let rewritten: Vec<ExpressionReference> = children
597                .iter()
598                .map(|child| walk_apply_special_angle_lookup(context, child))
599                .collect();
600            simplify_sum(context, rewritten)
601        }
602        Product(children) => {
603            let rewritten: Vec<ExpressionReference> = children
604                .iter()
605                .map(|child| walk_apply_special_angle_lookup(context, child))
606                .collect();
607            simplify_product(context, rewritten)
608        }
609        Power(base, exponent) => {
610            let new_base = walk_apply_special_angle_lookup(context, base);
611            let new_exponent = walk_apply_special_angle_lookup(context, exponent);
612            simplify_power(context, new_base, new_exponent)
613        }
614        Factorial(argument) => {
615            let inner = walk_apply_special_angle_lookup(context, argument);
616            builders::factorial(context, inner)
617        }
618        FunctionCall(function, arguments) => {
619            let rewritten: Vec<ExpressionReference> = arguments
620                .iter()
621                .map(|child| walk_apply_special_angle_lookup(context, child))
622                .collect();
623            if let FunctionIdentifier::Builtin(builtin) = function {
624                if rewritten.len() == 1 {
625                    if let Some(value) = simplify_pi_multiple_call(context, *builtin, &rewritten[0])
626                    {
627                        return value;
628                    }
629                }
630            }
631            simplify_function_call(context, *function, rewritten)
632        }
633        Derivative(function, orders, arguments) => {
634            let rewritten: Vec<ExpressionReference> = arguments
635                .iter()
636                .map(|child| walk_apply_special_angle_lookup(context, child))
637                .collect();
638            context.interner.intern(ExpressionKind::Derivative(
639                *function,
640                orders.clone(),
641                rewritten,
642            ))
643        }
644        Integral(integrand, variable, lower_bound, upper_bound) => {
645            let new_integrand = walk_apply_special_angle_lookup(context, integrand);
646            let new_lower = lower_bound
647                .as_ref()
648                .map(|bound| walk_apply_special_angle_lookup(context, bound));
649            let new_upper = upper_bound
650                .as_ref()
651                .map(|bound| walk_apply_special_angle_lookup(context, bound));
652            builders::integral(context, new_integrand, *variable, new_lower, new_upper)
653        }
654        Summation(summand, index, lower_bound, upper_bound) => {
655            let new_summand = walk_apply_special_angle_lookup(context, summand);
656            let new_lower = lower_bound
657                .as_ref()
658                .map(|bound| walk_apply_special_angle_lookup(context, bound));
659            let new_upper = upper_bound
660                .as_ref()
661                .map(|bound| walk_apply_special_angle_lookup(context, bound));
662            builders::summation(context, new_summand, *index, new_lower, new_upper)
663        }
664        LaplaceTransform(target_expression, time_variable, frequency_variable) => {
665            let new_target = walk_apply_special_angle_lookup(context, target_expression);
666            builders::laplace_transform_unevaluated(
667                context,
668                new_target,
669                *time_variable,
670                *frequency_variable,
671            )
672        }
673        InverseLaplaceTransform(target_expression, frequency_variable, time_variable) => {
674            let new_target = walk_apply_special_angle_lookup(context, target_expression);
675            builders::inverse_laplace_transform_unevaluated(
676                context,
677                new_target,
678                *frequency_variable,
679                *time_variable,
680            )
681        }
682        Piecewise(branches) => {
683            let rebuilt: Vec<(ExpressionReference, ExpressionReference)> = branches
684                .iter()
685                .map(|(condition, value)| {
686                    (
687                        Arc::clone(condition),
688                        walk_apply_special_angle_lookup(context, value),
689                    )
690                })
691                .collect();
692            crate::simplify::simplify_piecewise(context, rebuilt)
693        }
694        OrdinaryDifferentialEquation(equation, dependent_variable_call, independent_variable) => {
695            let new_equation = walk_apply_special_angle_lookup(context, equation);
696            let new_dependent_call =
697                walk_apply_special_angle_lookup(context, dependent_variable_call);
698            builders::ordinary_differential_equation(
699                context,
700                new_equation,
701                new_dependent_call,
702                *independent_variable,
703            )
704        }
705        RecurrenceEquation(equation, sequence_variable_call, index_variable) => {
706            let new_equation = walk_apply_special_angle_lookup(context, equation);
707            let new_sequence_call =
708                walk_apply_special_angle_lookup(context, sequence_variable_call);
709            builders::recurrence_equation(context, new_equation, new_sequence_call, *index_variable)
710        }
711        _ => std::sync::Arc::clone(target),
712    }
713}
