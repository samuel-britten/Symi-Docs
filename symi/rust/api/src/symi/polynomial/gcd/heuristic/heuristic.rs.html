12const MAX_HEURISTIC_RETRIES: u32 = 6;
13
14pub fn char_geddes_gonnet_gcd(
15    left: &Polynomial,
16    right: &Polynomial,
17    random_source: &mut dyn RandomSource,
18) -> Option<Polynomial> {
19    // Char–Geddes–Gonnet (1989) heuristic gcd for univariate inputs over Z.
20    // Evaluates both inputs at an integer point ξ chosen large relative to the
21    // coefficient magnitudes, takes the integer gcd of the evaluations, then
22    // reconstructs a polynomial candidate by interpreting the integer gcd as a
23    // symmetric base-ξ expansion. The candidate is accepted only when it
24    // divides both inputs exactly; otherwise ξ is doubled (with a small
25    // pseudo-random perturbation to dodge pathological inputs) and the cycle
26    // restarts. Returns None when MAX_HEURISTIC_RETRIES attempts fail.
27    let left_dense = univariate_integer_coefficients(left)?;
28    let right_dense = univariate_integer_coefficients(right)?;
29    if left_dense.is_empty() || right_dense.is_empty() {
30        return None;
31    }
32    let max_norm_left = infinity_norm(&left_dense);
33    let max_norm_right = infinity_norm(&right_dense);
34    let height_minimum = if max_norm_left < max_norm_right {
35        max_norm_left
36    } else {
37        max_norm_right
38    };
39    let base_xi: exact_integer = exact_integer::from(2) * height_minimum + exact_integer::from(2);
40    let mut xi = base_xi.clone();
41    for _ in 0..MAX_HEURISTIC_RETRIES {
42        if let Some(candidate) = attempt_reconstruction(left, right, &left_dense, &right_dense, &xi)
43        {
44            return Some(candidate);
45        }
46        let perturbation = (random_source.next_u64() % 8) as i64;
47        xi = (xi * exact_integer::from(2)) + exact_integer::from(perturbation);
48        if xi < exact_integer::from(4) {
49            xi = exact_integer::from(4);
50        }
51    }
52    None
53}

55fn attempt_reconstruction(
56    left: &Polynomial,
57    right: &Polynomial,
58    left_dense: &[exact_integer],
59    right_dense: &[exact_integer],
60    xi: &exact_integer,
61) -> Option<Polynomial> {
62    // One CGGH attempt. Evaluates both inputs at ξ, gcd's the results, lifts
63    // back to a polynomial via the symmetric base-ξ expansion, strips content,
64    // and trial-divides. Returns Some(gcd) on success or None when the lifted
65    // candidate fails the divisibility check at this ξ.
66    let left_value = horner_evaluate(left_dense, xi);
67    let right_value = horner_evaluate(right_dense, xi);
68    if left_value == integer_zero() || right_value == integer_zero() {
69        return None;
70    }
71    let integer_gcd_value = integer_gcd(&left_value, &right_value);
72    if integer_gcd_value == integer_zero() {
73        return None;
74    }
75    let coefficients = symmetric_base_digits(&integer_gcd_value, xi);
76    if coefficients.is_empty() {
77        return None;
78    }
79    let candidate = univariate_from_dense(
80        left.main_variable,
81        coefficients
82            .into_iter()
83            .map(|integer_value| Number::Integer(integer_value))
84            .collect(),
85    );
86    if is_zero(&candidate) {
87        return None;
88    }
89    let candidate_primitive = primitive_part(&candidate);
90    if is_zero(&candidate_primitive) {
91        return None;
92    }
93    if divide_exact(left, &candidate_primitive).is_none() {
94        return None;
95    }
96    if divide_exact(right, &candidate_primitive).is_none() {
97        return None;
98    }
99    Some(normalise_gcd(&candidate_primitive))
100}

102fn univariate_integer_coefficients(value: &Polynomial) -> Option<Vec<exact_integer>> {
103    // Project a polynomial into a dense ascending-degree integer vector.
104    // Returns None for any non-integer coefficient, which signals to the
105    // caller that the heuristic path is inapplicable (the dispatcher already
106    // calls this after primitive-part normalisation, so a None here means a
107    // genuinely multivariate or rational-coefficient input).
108    if is_zero(value) {
109        return Some(Vec::new());
110    }
111    let maximum_exponent = degree(value)?;
112    let mut dense: Vec<exact_integer> = vec![integer_zero(); maximum_exponent + 1];
113    for (exponent, coefficient) in value.terms.iter() {
114        match coefficient {
115            PolynomialCoefficient::Numeric(numeric_value) => match numeric_value {
116                Number::Integer(integer_value) => {
117                    dense[*exponent] = integer_value.clone();
118                }
119                Number::Rational(_) => return None,
120            },
121            PolynomialCoefficient::Nested(_) => return None,
122            PolynomialCoefficient::Algebraic(_) => return None,
123        }
124    }
125    Some(dense)
126}

128fn infinity_norm(coefficients: &[exact_integer]) -> exact_integer {
129    // Maximum |coefficient| in a dense coefficient vector. Used to size the
130    // starting evaluation point in CGGH so the symmetric base-ξ digits land
131    // in the (-ξ/2, ξ/2] band, which is the correctness precondition for the
132    // reconstruction step.
133    let mut maximum = integer_zero();
134    for coefficient in coefficients.iter() {
135        let absolute = integer_absolute_value(&coefficient);
136        if absolute > maximum {
137            maximum = absolute;
138        }
139    }
140    maximum
141}

143fn horner_evaluate(coefficients_ascending: &[exact_integer], xi: &exact_integer) -> exact_integer {
144    // Horner evaluation of a polynomial at an integer point. Walks the dense
145    // coefficient vector from highest to lowest exponent so the loop touches
146    // each coefficient exactly once and performs exactly one bignum multiply
147    // per intermediate degree.
148    let mut result = integer_zero();
149    for coefficient in coefficients_ascending.iter().rev() {
150        result = result * xi.clone() + coefficient.clone();
151    }
152    result
153}

155fn symmetric_base_digits(value: &exact_integer, base: &exact_integer) -> Vec<exact_integer> {
156    // Expand value in base `base` with each digit lying in (-base/2, base/2].
157    // The CGGH reconstruction interprets these digits as polynomial
158    // coefficients in ascending degree. The loop runs in O(log_base(value))
159    // iterations; the divisor is always positive so the standard backend
160    // % operator yields the canonical [0, base) representative.
161    if *base <= integer_one() {
162        return Vec::new();
163    }
164    let half = base.clone() / exact_integer::from(2);
165    let mut digits: Vec<exact_integer> = Vec::new();
166    let mut remaining = value.clone();
167    while remaining != integer_zero() {
168        let raw_modulo = (&remaining).clone() % base.clone();
169        let canonical_modulo = if raw_modulo < integer_zero() {
170            raw_modulo + base.clone()
171        } else {
172            raw_modulo
173        };
174        let signed_digit = if canonical_modulo > half {
175            canonical_modulo - base.clone()
176        } else {
177            canonical_modulo
178        };
179        remaining = (remaining - signed_digit.clone()) / base.clone();
180        digits.push(signed_digit);
181    }
182    digits
183}
