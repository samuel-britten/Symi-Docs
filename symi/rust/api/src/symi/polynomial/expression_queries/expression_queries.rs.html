1// User-facing polynomial queries that operate on expressions instead of the
2// internal recursive-sparse polynomial representation. Each entry point
3// converts through `expression_to_polynomial` (treating the named variable as
4// the main variable and every other free variable as a nested coefficient
5// variable), performs the polynomial-level operation, and reifies the result
6// back through `polynomial_to_expression` / `coefficient_to_expression`. The
7// wrappers exist so the symi-python and symi-wasm bindings share one
8// code path for the group-J surface of the full-bindings plan; they are
9// general-purpose and carry no binding-specific behavior.
10

28fn convert_with_variable_first(
29    target: &ExpressionReference,
30    variable: SymbolIdentifier,
31) -> Option<Polynomial> {
32    // Convert an expression to polynomial form with `variable` as the main
33    // variable and every other free variable, in sorted identifier order, as a
34    // nested coefficient variable. Returns None when the expression is not
35    // polynomial in the resulting variable list.
36    let mut order: Vec<SymbolIdentifier> = vec![variable];
37    let mut remaining: Vec<SymbolIdentifier> = free_variables(target)
38        .into_iter()
39        .filter(|free_variable| *free_variable != variable)
40        .collect();
41    remaining.sort();
42    order.extend(remaining);
43    expression_to_polynomial(target, &order)
44}
45
46fn convert_pair_with_variable_first(
47    left: &ExpressionReference,
48    right: &ExpressionReference,
49    variable: SymbolIdentifier,
50) -> Option<(Polynomial, Polynomial)> {
51    // Convert two expressions over a shared variable order so that binary
52    // polynomial operations (GCD, resultant) see compatible nested-coefficient
53    // chains for every free variable of either input.
54    let mut order: Vec<SymbolIdentifier> = vec![variable];
55    let mut remaining: Vec<SymbolIdentifier> = free_variables(left)
56        .into_iter()
57        .chain(free_variables(right))
58        .filter(|free_variable| *free_variable != variable)
59        .collect();
60    remaining.sort();
61    remaining.dedup();
62    order.extend(remaining);
63    let left_polynomial = expression_to_polynomial(left, &order)?;
64    let right_polynomial = expression_to_polynomial(right, &order)?;
65    Some((left_polynomial, right_polynomial))
66}

68/// Performs the advanced native `degree` operation.
69pub fn degree(target: &ExpressionReference, variable: SymbolIdentifier) -> Option<usize> {
70    // Degree of the expression viewed as a polynomial in `variable`. Returns
71    // None when the expression is not polynomial in `variable` or when it is
72    // the zero polynomial (whose degree is undefined under the module-wide
73    // convention).
74    let converted = convert_with_variable_first(target, variable)?;
75    crate::polynomial::degree(&converted)
76}

78/// Performs the advanced native `leading_coefficient` operation.
79pub fn leading_coefficient(
80    context: &Context,
81    target: &ExpressionReference,
82    variable: SymbolIdentifier,
83) -> Option<ExpressionReference> {
84    // Coefficient of the highest power of `variable`, reified back to an
85    // expression. None when the input is not polynomial in `variable` or is
86    // the zero polynomial.
87    let converted = convert_with_variable_first(target, variable)?;
88    let coefficient = crate::polynomial::leading_coefficient(&converted)?;
89    Some(coefficient_to_expression(context, &coefficient))
90}

92/// Performs the advanced native `coefficient` operation.
93pub fn coefficient(
94    context: &Context,
95    target: &ExpressionReference,
96    variable: SymbolIdentifier,
97    power: usize,
98) -> Option<ExpressionReference> {
99    // Coefficient of `variable^power` in the expression viewed as a polynomial
100    // in `variable`; integer zero when the term is absent. None when the
101    // expression is not polynomial in `variable`.
102    let converted = convert_with_variable_first(target, variable)?;
103    for (exponent, term_coefficient) in converted.terms.iter() {
104        if *exponent == power {
105            return Some(coefficient_to_expression(context, term_coefficient));
106        }
107    }
108    Some(crate::expression::builders::zero(context))
109}

111/// Performs the advanced native `polynomial_gcd` operation.
112pub fn polynomial_gcd(
113    context: &Context,
114    left: &ExpressionReference,
115    right: &ExpressionReference,
116    variable: SymbolIdentifier,
117) -> Option<ExpressionReference> {
118    // Greatest common divisor of two expressions viewed as polynomials in
119    // `variable` (other free variables ride along as nested coefficients).
120    // None when either input fails the polynomial conversion.
121    let (left_polynomial, right_polynomial) =
122        convert_pair_with_variable_first(left, right, variable)?;
123    let divisor = polynomial_gcd_internal(&left_polynomial, &right_polynomial, context.operation());
124    Some(polynomial_to_expression(context, &divisor))
125}

127/// Performs the advanced native `resultant` operation.
128pub fn resultant(
129    context: &Context,
130    left: &ExpressionReference,
131    right: &ExpressionReference,
132    variable: SymbolIdentifier,
133) -> Option<ExpressionReference> {
134    // Resultant of two expressions with respect to `variable`, reified back to
135    // an expression. None when either input fails the polynomial conversion.
136    let (left_polynomial, right_polynomial) =
137        convert_pair_with_variable_first(left, right, variable)?;
138    let value = resultant_internal(&left_polynomial, &right_polynomial, context.operation());
139    Some(coefficient_to_expression(context, &value))
140}

142/// Performs the advanced native `isolate_real_roots` operation.
143pub fn isolate_real_roots(
144    context: &Context,
145    target: &ExpressionReference,
146    variable: SymbolIdentifier,
147) -> Option<Vec<(ExpressionReference, ExpressionReference)>> {
148    // One (lower, upper) rational isolating interval per distinct real root of
149    // the expression viewed as a univariate polynomial over the rationals in
150    // `variable`, sorted by ascending position. A collapsed pair (r, r) marks
151    // an exact rational root. None when the expression is not a univariate
152    // rational-coefficient polynomial in `variable` or is the zero polynomial.
153    let converted = expression_to_univariate(target, variable)?;
154    let intervals = isolate_real_roots_internal(&converted, context.operation())?;
155    Some(
156        intervals
157            .into_iter()
158            .map(|interval| {
159                (
160                    rational_to_expression(context, interval.lower),
161                    rational_to_expression(context, interval.upper),
162                )
163            })
164            .collect(),
165    )
166}
167
168fn rational_to_expression(context: &Context, value: exact_rational) -> ExpressionReference {
169    // Reify an exact rational as a canonical numeric expression leaf,
170    // routing through the simplifying rational builder so integer-valued
171    // rationals collapse to integer leaves.
172    let parts = rational_into_parts(value);
173    crate::expression::builders::rational(
174        context,
175        parts.numerator,
176        natural_to_integer(parts.denominator),
177    )
178}

187    fn parse(context: &Context, text: &str) -> ExpressionReference {
188        let bindings: HashMap<String, ExpressionReference> = HashMap::new();
189        match parse_statement(text, context, &bindings).expect("parse failure") {
190            Statement::Expression(expression) => expression,
191            _ => panic!("expected an expression statement"),
192        }
193    }

195    fn formatted(context: &Context, target: &ExpressionReference) -> String {
196        format!("{}", format_expression(target, &context.symbol_table))
197    }
