15/// Attempts coefficient-aware polynomial addition without fabricating a result
16/// when a coefficient combination is unsupported.
17pub(crate) fn try_add(left: &Polynomial, right: &Polynomial) -> Option<Polynomial> {
18    // Sum two polynomials with the same main variable. Terms are merged by
19    // exponent: when both inputs contain a term at the same degree the
20    // coefficients are added and a zero result drops out of the canonical form.
21    debug_assert_eq!(left.main_variable, right.main_variable);
22    let mut left_iterator = left.terms.iter();
23    let mut right_iterator = right.terms.iter();
24    let mut left_current = left_iterator.next();
25    let mut right_current = right_iterator.next();
26    let mut combined: Vec<(usize, PolynomialCoefficient)> = Vec::new();
27    loop {
28        match (left_current, right_current) {
29            (Some(left_term), Some(right_term)) => match left_term.0.cmp(&right_term.0) {
30                Ordering::Greater => {
31                    combined.push((left_term.0, left_term.1.clone()));
32                    left_current = left_iterator.next();
33                }
34                Ordering::Less => {
35                    combined.push((right_term.0, right_term.1.clone()));
36                    right_current = right_iterator.next();
37                }
38                Ordering::Equal => {
39                    let summed = super::add_coefficients(&left_term.1, &right_term.1)?;
40                    if !coefficient_is_zero(&summed) {
41                        combined.push((left_term.0, summed));
42                    }
43                    left_current = left_iterator.next();
44                    right_current = right_iterator.next();
45                }
46            },
47            (Some(left_term), None) => {
48                combined.push((left_term.0, left_term.1.clone()));
49                left_current = left_iterator.next();
50            }
51            (None, Some(right_term)) => {
52                combined.push((right_term.0, right_term.1.clone()));
53                right_current = right_iterator.next();
54            }
55            (None, None) => break,
56        }
57    }
58    Some(Polynomial {
59        main_variable: left.main_variable,
60        terms: combined,
61    })
62}

69/// Performs the advanced native `subtract` operation.
70pub fn subtract(left: &Polynomial, right: &Polynomial) -> Polynomial {
71    // Subtraction implemented as addition of the negated right operand. The
72    // canonical-form merge in add eliminates any zero terms produced by exact
73    // cancellation.
74    let negated = super::negate(right);
75    add(left, &negated)
76}

78/// Attempts coefficient-aware polynomial multiplication without fabricating a
79/// result when a coefficient combination is unsupported.
80pub(crate) fn try_multiply(left: &Polynomial, right: &Polynomial) -> Option<Polynomial> {
81    // Schoolbook multiplication over the sparse representation. The cross
82    // product is accumulated through add_coefficients so any nested-coefficient
83    // multiplications recurse into the same arithmetic dispatch.
84    debug_assert_eq!(left.main_variable, right.main_variable);
85    if is_zero(left) || is_zero(right) {
86        return Some(zero_polynomial(left.main_variable));
87    }
88    let mut accumulator: Vec<(usize, PolynomialCoefficient)> = Vec::new();
89    for (left_exponent, left_coefficient) in left.terms.iter() {
90        for (right_exponent, right_coefficient) in right.terms.iter() {
91            let exponent = left_exponent + right_exponent;
92            let product_coefficient =
93                super::multiply_coefficients(left_coefficient, right_coefficient)?;
94            if coefficient_is_zero(&product_coefficient) {
95                continue;
96            }
97            try_insert_or_combine(&mut accumulator, exponent, product_coefficient)?;
98        }
99    }
100    Some(Polynomial {
101        main_variable: left.main_variable,
102        terms: accumulator,
103    })
104}

111/// Performs the advanced native `pseudo_divide` operation.
112pub fn pseudo_divide(
113    dividend: &Polynomial,
114    divisor: &Polynomial,
115) -> Option<(Polynomial, Polynomial)> {
116    // Compute polynomial pseudo-division with the leading-coefficient power
117    // trick described in Cohen ch. 6: multiply the dividend by the divisor's
118    // leading coefficient raised to (deg(dividend) - deg(divisor) + 1) so the
119    // remainder coefficients stay in the original ring without introducing
120    // fractions. Returns (quotient, remainder) such that
121    //   leading(divisor)^k * dividend == divisor * quotient + remainder
122    // with deg(remainder) < deg(divisor). Returns None when the divisor is
123    // zero or the polynomials disagree on main variable.
124    if is_zero(divisor) || dividend.main_variable != divisor.main_variable {
125        return None;
126    }
127    let dividend_degree = match degree(dividend) {
128        Some(value) => value,
129        None => {
130            return Some((
131                zero_polynomial(divisor.main_variable),
132                zero_polynomial(divisor.main_variable),
133            ));
134        }
135    };
136    let divisor_degree = degree(divisor).unwrap();
137    if dividend_degree < divisor_degree {
138        return Some((zero_polynomial(divisor.main_variable), dividend.clone()));
139    }
140    let leading_divisor = leading_coefficient(divisor).unwrap();
141    let degree_difference = dividend_degree - divisor_degree;
142    let scale_power = degree_difference + 1;
143    let scale_factor = pow_coefficient(&leading_divisor, scale_power);
144    let mut working_dividend = scale_dividend(dividend, &scale_factor);
145    let mut quotient = zero_polynomial(divisor.main_variable);
146    while !is_zero(&working_dividend) {
147        let working_degree = degree(&working_dividend).unwrap();
148        if working_degree < divisor_degree {
149            break;
150        }
151        let working_leading = leading_coefficient(&working_dividend).unwrap();
152        let coefficient_quotient = divide_coefficient_exact(&working_leading, &leading_divisor)?;
153        let exponent = working_degree - divisor_degree;
154        super::arithmetic::insert_or_combine(
155            &mut quotient.terms,
156            exponent,
157            coefficient_quotient.clone(),
158        );
159        let multiplier_polynomial = Polynomial {
160            main_variable: divisor.main_variable,
161            terms: vec![(exponent, coefficient_quotient.clone())],
162        };
163        let subtraction_term = multiply(&multiplier_polynomial, divisor);
164        working_dividend = subtract(&working_dividend, &subtraction_term);
165    }
166    Some((quotient, working_dividend))
167}

169/// Performs the advanced native `divide_exact` operation.
170pub fn divide_exact(dividend: &Polynomial, divisor: &Polynomial) -> Option<Polynomial> {
171    // Exact polynomial division: returns Some(quotient) when divisor divides
172    // dividend without remainder, or None otherwise. Performed by the same
173    // long-division loop as pseudo_divide but without the leading-coefficient
174    // pre-scale, so the coefficients are required to divide cleanly at every
175    // step. Used by the primitive-part extraction path in gcd.rs.
176    if is_zero(divisor) || dividend.main_variable != divisor.main_variable {
177        return None;
178    }
179    if is_zero(dividend) {
180        return Some(zero_polynomial(divisor.main_variable));
181    }
182    let divisor_degree = degree(divisor).unwrap();
183    let leading_divisor = leading_coefficient(divisor).unwrap();
184    let mut working_dividend = dividend.clone();
185    let mut quotient = zero_polynomial(divisor.main_variable);
186    while !is_zero(&working_dividend) {
187        let working_degree = degree(&working_dividend).unwrap();
188        if working_degree < divisor_degree {
189            return None;
190        }
191        let working_leading = leading_coefficient(&working_dividend).unwrap();
192        let coefficient_quotient = divide_coefficient_exact(&working_leading, &leading_divisor)?;
193        let exponent = working_degree - divisor_degree;
194        super::arithmetic::insert_or_combine(
195            &mut quotient.terms,
196            exponent,
197            coefficient_quotient.clone(),
198        );
199        let multiplier_polynomial = Polynomial {
200            main_variable: divisor.main_variable,
201            terms: vec![(exponent, coefficient_quotient.clone())],
202        };
203        let subtraction_term = multiply(&multiplier_polynomial, divisor);
204        working_dividend = subtract(&working_dividend, &subtraction_term);
205    }
206    Some(quotient)
207}

209/// Performs the advanced native `divide_polynomial_by_coefficient` operation.
210pub fn divide_polynomial_by_coefficient(
211    value: &Polynomial,
212    divisor: &PolynomialCoefficient,
213) -> Option<Polynomial> {
214    // Divide every coefficient of the polynomial by a constant coefficient.
215    // Returns None when any individual division is inexact, signalling that the
216    // caller's exactness assumption was wrong.
217    let mut new_terms: Vec<(usize, PolynomialCoefficient)> = Vec::with_capacity(value.terms.len());
218    for (exponent, coefficient) in value.terms.iter() {
219        let quotient = divide_coefficient_exact(coefficient, divisor)?;
220        if !coefficient_is_zero(&quotient) {
221            new_terms.push((*exponent, quotient));
222        }
223    }
224    Some(Polynomial {
225        main_variable: value.main_variable,
226        terms: new_terms,
227    })
228}

230/// Performs the advanced native `divide_coefficient_exact` operation.
231pub fn divide_coefficient_exact(
232    dividend: &PolynomialCoefficient,
233    divisor: &PolynomialCoefficient,
234) -> Option<PolynomialCoefficient> {
235    // Exact division for polynomial coefficients. The numeric/numeric case goes
236    // straight through the number module and never fails (rationals form a
237    // field). The nested cases route through the recursive divide_exact so the
238    // result is well-defined only when the divisor divides the dividend in the
239    // multivariate ring.
240    use PolynomialCoefficient::*;
241    if coefficient_is_zero(divisor) {
242        return None;
243    }
244    if coefficient_is_zero(dividend) {
245        return Some(coefficient_zero());
246    }
247    if coefficient_is_one(divisor) {
248        return Some(dividend.clone());
249    }
250    match (dividend, divisor) {
251        (Numeric(left_value), Numeric(right_value)) => {
252            crate::number::divide(left_value, right_value).map(Numeric)
253        }
254        (Nested(left_polynomial), Nested(right_polynomial)) => {
255            // Operands may disagree on main_variable when simplify_coefficient has
256            // collapsed an intermediate variable on one side (the same situation
257            // polynomial_gcd_subresultant promotes for). The deeper-variable
258            // operand is wrapped as a constant of the shallower variable so the
259            // long division sees a uniform structure; a shallower divisor that
260            // genuinely involves its variable then fails the division naturally.
261            if left_polynomial.main_variable != right_polynomial.main_variable {
262                if left_polynomial.main_variable.0 < right_polynomial.main_variable.0 {
263                    let promoted_divisor = Polynomial {
264                        main_variable: left_polynomial.main_variable,
265                        terms: vec![(0, Nested(right_polynomial.clone()))],
266                    };
267                    let quotient = divide_exact(left_polynomial, &promoted_divisor)?;
268                    return Some(simplify_coefficient(Nested(Box::new(quotient))));
269                }
270                let promoted_dividend = Polynomial {
271                    main_variable: right_polynomial.main_variable,
272                    terms: vec![(0, Nested(left_polynomial.clone()))],
273                };
274                let quotient = divide_exact(&promoted_dividend, right_polynomial)?;
275                return Some(simplify_coefficient(Nested(Box::new(quotient))));
276            }
277            let quotient = divide_exact(left_polynomial, right_polynomial)?;
278            Some(simplify_coefficient(Nested(Box::new(quotient))))
279        }
280        (Nested(left_polynomial), Numeric(right_value)) => {
281            let mut new_terms: Vec<(usize, PolynomialCoefficient)> =
282                Vec::with_capacity(left_polynomial.terms.len());
283            for (exponent, coefficient) in left_polynomial.terms.iter() {
284                let quotient = divide_coefficient_by_numeric(coefficient, right_value)?;
285                if !coefficient_is_zero(&quotient) {
286                    new_terms.push((*exponent, quotient));
287                }
288            }
289            Some(simplify_coefficient(Nested(Box::new(Polynomial {
290                main_variable: left_polynomial.main_variable,
291                terms: new_terms,
292            }))))
293        }
294        (Numeric(_), Nested(_)) => None,
295        (Algebraic(left_value), Algebraic(right_value)) => {
296            crate::algebraic_numbers::arithmetic::divide(left_value, right_value)
297                .map(PolynomialCoefficient::Algebraic)
298        }
299        (Algebraic(left_value), Numeric(right_value)) => {
300            let right_rational = match right_value {
301                Number::Integer(integer_value) => exact_rational::from(integer_value.clone()),
302                Number::Rational(rational_value) => rational_value.clone(),
303            };
304            if right_rational == rational_zero() {
305                return None;
306            }
307            let inverse_factor = rational_one() / right_rational;
308            Some(PolynomialCoefficient::Algebraic(
309                crate::algebraic_numbers::arithmetic::scale_by_rational(
310                    left_value,
311                    &inverse_factor,
312                ),
313            ))
314        }
315        (Numeric(left_value), Algebraic(right_value)) => {
316            let left_rational = match left_value {
317                Number::Integer(integer_value) => exact_rational::from(integer_value.clone()),
318                Number::Rational(rational_value) => rational_value.clone(),
319            };
320            let lifted = crate::algebraic_numbers::element::rational_element(
321                &right_value.field,
322                left_rational,
323            );
324            crate::algebraic_numbers::arithmetic::divide(&lifted, right_value)
325                .map(PolynomialCoefficient::Algebraic)
326        }
327        (Algebraic(_), Nested(_)) | (Nested(_), Algebraic(_)) => None,
328    }
329}

331fn divide_coefficient_by_numeric(
332    dividend: &PolynomialCoefficient,
333    divisor: &Number,
334) -> Option<PolynomialCoefficient> {
335    // Specialised fast path that divides any coefficient by a numeric divisor
336    // without traversing the polynomial cross-case. Returns None on a zero
337    // divisor.
338    if crate::number::is_zero(divisor) {
339        return None;
340    }
341    use PolynomialCoefficient::*;
342    match dividend {
343        Numeric(value) => crate::number::divide(value, divisor).map(Numeric),
344        Nested(polynomial_value) => {
345            let mut new_terms: Vec<(usize, PolynomialCoefficient)> =
346                Vec::with_capacity(polynomial_value.terms.len());
347            for (exponent, coefficient) in polynomial_value.terms.iter() {
348                let quotient = divide_coefficient_by_numeric(coefficient, divisor)?;
349                if !coefficient_is_zero(&quotient) {
350                    new_terms.push((*exponent, quotient));
351                }
352            }
353            Some(simplify_coefficient(Nested(Box::new(Polynomial {
354                main_variable: polynomial_value.main_variable,
355                terms: new_terms,
356            }))))
357        }
358        Algebraic(algebraic_value) => {
359            let divisor_rational = match divisor {
360                Number::Integer(integer_value) => exact_rational::from(integer_value.clone()),
361                Number::Rational(rational_value) => rational_value.clone(),
362            };
363            let inverse_factor = rational_one() / divisor_rational;
364            Some(Algebraic(
365                crate::algebraic_numbers::arithmetic::scale_by_rational(
366                    algebraic_value,
367                    &inverse_factor,
368                ),
369            ))
370        }
371    }
372}

374/// Performs the advanced native `pow_coefficient` operation.
375pub fn pow_coefficient(base: &PolynomialCoefficient, exponent: usize) -> PolynomialCoefficient {
376    // Raise a coefficient to a non-negative integer power by repeated squaring.
377    // Used by the pseudo-division pre-scale step.
378    if exponent == 0 {
379        return coefficient_one();
380    }
381    if exponent == 1 {
382        return base.clone();
383    }
384    let mut result = coefficient_one();
385    let mut current = base.clone();
386    let mut remaining = exponent;
387    while remaining > 0 {
388        if remaining & 1 == 1 {
389            result = multiply_coefficients_or_zero(&result, &current);
390        }
391        remaining >>= 1;
392        if remaining > 0 {
393            current = multiply_coefficients_or_zero(&current, &current);
394        }
395    }
396    result
397}

399fn scale_dividend(value: &Polynomial, factor: &PolynomialCoefficient) -> Polynomial {
400    // Scale every coefficient of a polynomial by a coefficient factor. The
401    // numeric fast path falls back through scale_polynomial; the nested case
402    // recurses through multiply_coefficients on each term.
403    if let Some(numeric_factor) = coefficient_to_number(factor) {
404        return scale_polynomial(value, &numeric_factor);
405    }
406    Polynomial {
407        main_variable: value.main_variable,
408        terms: value
409            .terms
410            .iter()
411            .map(|(exponent, coefficient)| {
412                (
413                    *exponent,
414                    multiply_coefficients_or_zero(coefficient, factor),
415                )
416            })
417            .collect(),
418    }
419}

421/// Performs the advanced native `pow` operation.
422pub fn pow(value: &Polynomial, exponent: usize) -> Polynomial {
423    // Raise a polynomial to a non-negative integer power by repeated squaring.
424    // The empty exponent collapses to the constant one polynomial in the same
425    // main variable.
426    if exponent == 0 {
427        return Polynomial {
428            main_variable: value.main_variable,
429            terms: vec![(
430                0,
431                PolynomialCoefficient::Numeric(Number::Integer(integer_one())),
432            )],
433        };
434    }
435    let mut result = Polynomial {
436        main_variable: value.main_variable,
437        terms: vec![(
438            0,
439            PolynomialCoefficient::Numeric(Number::Integer(integer_one())),
440        )],
441    };
442    let mut current = value.clone();
443    let mut remaining = exponent;
444    while remaining > 0 {
445        if remaining & 1 == 1 {
446            result = multiply(&result, &current);
447        }
448        remaining >>= 1;
449        if remaining > 0 {
450            current = multiply(&current, &current);
451        }
452    }
453    result
454}

456/// Performs the advanced native `shift_by_degree` operation.
457pub fn shift_by_degree(value: &Polynomial, additional_degree: usize) -> Polynomial {
458    // Multiply a polynomial by main_variable^additional_degree. Used by the
459    // long-division loop and by the resultant subresultant PRS to build the
460    // multiplier polynomial without going through full multiplication.
461    if is_zero(value) || additional_degree == 0 {
462        return value.clone();
463    }
464    Polynomial {
465        main_variable: value.main_variable,
466        terms: value
467            .terms
468            .iter()
469            .map(|(exponent, coefficient)| (*exponent + additional_degree, coefficient.clone()))
470            .collect(),
471    }
472}

474/// Attempts to insert or combine a coefficient without hiding an unsupported
475/// coefficient combination.
476pub(crate) fn try_insert_or_combine(
477    terms: &mut Vec<(usize, PolynomialCoefficient)>,
478    exponent: usize,
479    coefficient: PolynomialCoefficient,
480) -> Option<()> {
481    // Insert a coefficient into a sorted (descending exponent) term list,
482    // combining with any existing entry of the same degree. A combined result of
483    // zero is removed so the canonical-form invariant is preserved.
484    let position = terms
485        .iter()
486        .position(|(existing_exponent, _)| *existing_exponent <= exponent);
487    match position {
488        Some(index) if terms[index].0 == exponent => {
489            let combined = super::add_coefficients(&terms[index].1, &coefficient)?;
490            if coefficient_is_zero(&combined) {
491                terms.remove(index);
492            } else {
493                terms[index].1 = combined;
494            }
495        }
496        Some(index) => {
497            terms.insert(index, (exponent, coefficient));
498        }
499        None => {
500            terms.push((exponent, coefficient));
501        }
502    }
503    Some(())
504}

506/// Performs the advanced native `insert_or_combine` operation.
507pub fn insert_or_combine(
508    terms: &mut Vec<(usize, PolynomialCoefficient)>,
509    exponent: usize,
510    coefficient: PolynomialCoefficient,
511) {
512    let _ = try_insert_or_combine(terms, exponent, coefficient);
513}

515/// Performs the advanced native `ensure_canonical` operation.
516pub fn ensure_canonical(value: &mut Polynomial) {
517    // Reorder a freshly assembled polynomial into descending degree and drop any
518    // accidentally zero coefficients. Used by the test helpers that build
519    // polynomials term-by-term outside the constructor functions.
520    value
521        .terms
522        .retain(|(_, coefficient)| !coefficient_is_zero(coefficient));
523    value.terms.sort_by(|left, right| right.0.cmp(&left.0));
524}

526/// Performs the advanced native `coefficients_subtract_assign` operation.
527pub fn coefficients_subtract_assign(
528    target: &mut PolynomialCoefficient,
529    other: &PolynomialCoefficient,
530) {
531    // In-place subtraction used by inner loops that already own the target
532    // coefficient. The replacement is computed via subtract_coefficients so the
533    // numeric/nested dispatch is shared.
534    if let Some(updated) = subtract_coefficients(target, other) {
535        *target = updated;
536    }
537}
