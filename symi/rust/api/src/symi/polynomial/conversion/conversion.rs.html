17/// Performs the advanced native `expression_to_polynomial` operation.
18pub fn expression_to_polynomial(
19    expression: &ExpressionReference,
20    variable_order: &[SymbolIdentifier],
21) -> Option<Polynomial> {
22    // Convert an expression into a recursive sparse polynomial whose main
23    // variable is variable_order[0] and whose nested coefficients use the
24    // remaining variables in order. Returns None when the expression contains
25    // a node that is not polynomial in any of the listed variables (a
26    // transcendental call, a non-integer power, an unbound symbol outside the
27    // list, etc.).
28    if variable_order.is_empty() {
29        return None;
30    }
31    let main_variable = variable_order[0];
32    let remaining_variables = &variable_order[1..];
33    convert_node(expression, main_variable, remaining_variables)
34}

36fn convert_node(
37    expression: &ExpressionReference,
38    main_variable: SymbolIdentifier,
39    remaining_variables: &[SymbolIdentifier],
40) -> Option<Polynomial> {
41    // Recursive structural conversion. Every shape that polynomial arithmetic
42    // can absorb has its own branch; everything else returns None to abort the
43    // conversion. The remaining_variables list encodes the recursive
44    // representation: nested coefficient slots that contain remaining variables
45    // continue down the chain instead of returning to the numeric leaf case.
46    use ExpressionKind::*;
47    match &expression.kind {
48        Integer(value) => Some(constant_polynomial_with_chain(
49            main_variable,
50            remaining_variables,
51            PolynomialCoefficient::Numeric(Number::Integer(value.clone())),
52        )),
53        Rational(value) => Some(constant_polynomial_with_chain(
54            main_variable,
55            remaining_variables,
56            PolynomialCoefficient::Numeric(Number::Rational(value.clone())),
57        )),
58        Symbol(identifier) => {
59            if *identifier == main_variable {
60                let one = PolynomialCoefficient::Numeric(Number::Integer(integer_one()));
61                let coefficient = promote_coefficient_through_chain(remaining_variables, one);
62                Some(Polynomial {
63                    main_variable,
64                    terms: vec![(1, coefficient)],
65                })
66            } else if remaining_variables.contains(identifier) {
67                let nested = convert_node(
68                    expression,
69                    remaining_variables[0],
70                    &remaining_variables[1..],
71                )?;
72                let coefficient =
73                    simplify_coefficient(PolynomialCoefficient::Nested(Box::new(nested)));
74                Some(Polynomial {
75                    main_variable,
76                    terms: vec![(0, coefficient)],
77                })
78            } else {
79                None
80            }
81        }
82        Sum(children) => {
83            let mut total = zero_polynomial(main_variable);
84            for child in children.iter() {
85                let summand = convert_node(child, main_variable, remaining_variables)?;
86                total = arithmetic::add(&total, &summand);
87            }
88            Some(total)
89        }
90        Product(children) => {
91            let mut accumulated = constant_polynomial_with_chain(
92                main_variable,
93                remaining_variables,
94                PolynomialCoefficient::Numeric(Number::Integer(integer_one())),
95            );
96            for child in children.iter() {
97                let factor = convert_node(child, main_variable, remaining_variables)?;
98                accumulated = arithmetic::multiply(&accumulated, &factor);
99            }
100            Some(accumulated)
101        }
102        Power(base, exponent) => {
103            let exponent_value = match &exponent.kind {
104                ExpressionKind::Integer(value) => value.clone(),
105                _ => return None,
106            };
107            let signed_exponent = integer_to_i64(&exponent_value)?;
108            if signed_exponent < 0 {
109                return None;
110            }
111            let base_polynomial = convert_node(base, main_variable, remaining_variables)?;
112            Some(arithmetic::pow(&base_polynomial, signed_exponent as usize))
113        }
114        PolynomialRoot(_)
115        | Constant(_)
116        | Factorial(_)
117        | FunctionCall(_, _)
118        | Derivative(_, _, _)
119        | Integral(_, _, _, _)
120        | Summation(_, _, _, _)
121        | IndexedProduct(_, _, _, _)
122        | LaplaceTransform(_, _, _)
123        | InverseLaplaceTransform(_, _, _)
124        | FourierTransform(_, _, _)
125        | InverseFourierTransform(_, _, _)
126        | Piecewise(_)
127        | OrdinaryDifferentialEquation(_, _, _)
128        | RecurrenceEquation(_, _, _)
129        | Interval(_, _, _, _)
130        | FiniteSet(_)
131        | SetUnion(_)
132        | SetIntersection(_)
133        | SetComplement(_, _)
134        | ImageSet(_, _, _)
135        | ConditionSet(_, _, _)
136        | Distribution(_, _)
137        | RandomVariable(_, _)
138        | SpecialFunctionParameterVector(_) => None,
139    }
140}

142fn constant_polynomial_with_chain(
143    main_variable: SymbolIdentifier,
144    remaining_variables: &[SymbolIdentifier],
145    coefficient: PolynomialCoefficient,
146) -> Polynomial {
147    // Produce a polynomial whose only term is the supplied coefficient at
148    // degree zero, promoting the coefficient through the remaining-variable
149    // chain so the resulting nested structure matches the agreed canonical
150    // ordering of variables.
151    let promoted_coefficient = promote_coefficient_through_chain(remaining_variables, coefficient);
152    if super::coefficient_is_zero(&promoted_coefficient) {
153        return zero_polynomial(main_variable);
154    }
155    Polynomial {
156        main_variable,
157        terms: vec![(0, promoted_coefficient)],
158    }
159}

161fn promote_coefficient_through_chain(
162    remaining_variables: &[SymbolIdentifier],
163    coefficient: PolynomialCoefficient,
164) -> PolynomialCoefficient {
165    // Wrap a coefficient in the nested chain implied by remaining_variables. An
166    // empty chain leaves the coefficient unchanged. Each step builds a constant
167    // polynomial in the next variable that collapses back to the original
168    // numeric form when the polynomial degenerates to a single zero-degree
169    // term.
170    if remaining_variables.is_empty() {
171        return coefficient;
172    }
173    let inner = promote_coefficient_through_chain(&remaining_variables[1..], coefficient);
174    if super::coefficient_is_zero(&inner) {
175        return PolynomialCoefficient::Numeric(Number::Integer(integer_zero()));
176    }
177    simplify_coefficient(PolynomialCoefficient::Nested(Box::new(Polynomial {
178        main_variable: remaining_variables[0],
179        terms: vec![(0, inner)],
180    })))
181}

183/// Performs the advanced native `polynomial_to_expression` operation.
184pub fn polynomial_to_expression(context: &Context, value: &Polynomial) -> ExpressionReference {
185    // Reify a polynomial back into the expression layer, routing through the
186    // auto-simplifying builders so the resulting expression is in ASAE form.
187    // The empty term list returns integer zero; a constant returns its numeric
188    // coefficient directly; everything else assembles a sum of products of
189    // powers of the main variable.
190    if super::is_zero(value) {
191        return builders::integer(context, integer_zero());
192    }
193    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(value.terms.len());
194    for (exponent, coefficient) in value.terms.iter() {
195        let coefficient_expression = coefficient_to_expression(context, coefficient);
196        let term = if *exponent == 0 {
197            coefficient_expression
198        } else {
199            let base = builders::symbol(
200                context,
201                &context.symbol_table.name(value.main_variable).to_string(),
202            );
203            let exponent_expression = builders::integer_value(context, *exponent as i64);
204            let power_expression = builders::power(context, base, exponent_expression);
205            if is_coefficient_one(coefficient) {
206                power_expression
207            } else if is_coefficient_negative_one(coefficient) {
208                builders::negate(context, power_expression)
209            } else {
210                builders::product(context, vec![coefficient_expression, power_expression])
211            }
212        };
213        terms.push(term);
214    }
215    if terms.len() == 1 {
216        terms.into_iter().next().unwrap()
217    } else {
218        builders::sum(context, terms)
219    }
220}

222/// Performs the advanced native `coefficient_to_expression` operation.
223pub fn coefficient_to_expression(
224    context: &Context,
225    coefficient: &PolynomialCoefficient,
226) -> ExpressionReference {
227    // Convert a coefficient back to an expression, promoting nested polynomial
228    // coefficients through the same builder pipeline so multivariate
229    // polynomials round-trip correctly.
230    match coefficient {
231        PolynomialCoefficient::Numeric(numeric_value) => match numeric_value {
232            Number::Integer(value) => builders::integer(context, value.clone()),
233            Number::Rational(value) => builders::rational_literal(context, value.clone()),
234        },
235        PolynomialCoefficient::Nested(nested_polynomial) => {
236            polynomial_to_expression(context, nested_polynomial)
237        }
238        PolynomialCoefficient::Algebraic(algebraic_value) => {
239            // Reify an algebraic-number coefficient through the standalone
240            // element reifier: it emits the
241            // canonical sum c_0 + c_1·G + ... in the field's generator, with G
242            // expressed as a closed-form surd when the defining polynomial has
243            // pure-radical shape. This is the single seam every coefficient-ring
244            // consumer (factor / solve / partial fractions / resultant
245            // back-substitution / the optimization sign oracle) routes through,
246            // so it must never panic.
247            crate::algebraic_numbers::conversion::algebraic_element_to_expression(
248                context,
249                algebraic_value,
250            )
251        }
252    }
253}

255fn is_coefficient_one(coefficient: &PolynomialCoefficient) -> bool {
256    // Detect a coefficient equal to one so the surrounding term renderer can
257    // omit a redundant multiplication factor.
258    match coefficient_to_number(coefficient) {
259        Some(value) => crate::number::is_one(&value),
260        None => false,
261    }
262}
263
264fn is_coefficient_negative_one(coefficient: &PolynomialCoefficient) -> bool {
265    // Detect a coefficient equal to negative one so the surrounding term
266    // renderer can use a unary negate instead of a redundant multiplication.
267    if let PolynomialCoefficient::Numeric(numeric) = coefficient {
268        if let Number::Integer(value) = numeric {
269            return *value == exact_integer::from(-1);
270        }
271    }
272    false
273}

275/// Performs the advanced native `expression_to_univariate` operation.
276pub fn expression_to_univariate(
277    expression: &ExpressionReference,
278    variable: SymbolIdentifier,
279) -> Option<Polynomial> {
280    // Convenience wrapper that converts an expression into a univariate
281    // polynomial in the supplied variable. Equivalent to
282    // expression_to_polynomial with a single-element variable order.
283    expression_to_polynomial(expression, &[variable])
284}

286/// Performs the advanced native `extract_variables_in_order` operation.
287pub fn extract_variables_in_order(
288    expression: &ExpressionReference,
289    fixed_first: Option<SymbolIdentifier>,
290) -> Vec<SymbolIdentifier> {
291    // Walk the expression to collect every distinct symbol_identifier in the
292    // order they first appear. When fixed_first is Some, that identifier is
293    // placed at the head of the result regardless of position in the
294    // expression. Used by the public factor / partial-fractions helpers to
295    // fix the main variable while still capturing nested coefficient symbols.
296    let mut found: Vec<SymbolIdentifier> = Vec::new();
297    if let Some(identifier) = fixed_first {
298        found.push(identifier);
299    }
300    collect_variables(expression, &mut found);
301    found
302}
303
304fn collect_variables(expression: &ExpressionReference, found: &mut Vec<SymbolIdentifier>) {
305    // Recursive helper for extract_variables_in_order. Visits each child once
306    // and pushes any new symbol_identifier on first encounter so the resulting
307    // order reflects depth-first appearance.
308    use ExpressionKind::*;
309    match &expression.kind {
310        Integer(_) | Rational(_) | Constant(_) => {}
311        Symbol(identifier) => {
312            if !found.contains(identifier) {
313                found.push(*identifier);
314            }
315        }
316        Sum(children) | Product(children) => {
317            for child in children {
318                collect_variables(child, found);
319            }
320        }
321        Power(base, exponent) => {
322            collect_variables(base, found);
323            collect_variables(exponent, found);
324        }
325        Factorial(argument) => collect_variables(argument, found),
326        FunctionCall(_, arguments) | Derivative(_, _, arguments) => {
327            for argument in arguments {
328                collect_variables(argument, found);
329            }
330        }
331        Integral(integrand, _, lower_bound, upper_bound) => {
332            collect_variables(integrand, found);
333            if let Some(lower) = lower_bound {
334                collect_variables(lower, found);
335            }
336            if let Some(upper) = upper_bound {
337                collect_variables(upper, found);
338            }
339        }
340        Summation(summand, _, lower_bound, upper_bound)
341        | IndexedProduct(summand, _, lower_bound, upper_bound) => {
342            collect_variables(summand, found);
343            if let Some(lower) = lower_bound {
344                collect_variables(lower, found);
345            }
346            if let Some(upper) = upper_bound {
347                collect_variables(upper, found);
348            }
349        }
350        LaplaceTransform(target, _, _)
351        | InverseLaplaceTransform(target, _, _)
352        | FourierTransform(target, _, _)
353        | InverseFourierTransform(target, _, _) => {
354            collect_variables(target, found);
355        }
356        Piecewise(branches) => {
357            for (condition, value) in branches {
358                collect_variables(condition, found);
359                collect_variables(value, found);
360            }
361        }
362        OrdinaryDifferentialEquation(equation, dependent_variable_call, _)
363        | RecurrenceEquation(equation, dependent_variable_call, _) => {
364            collect_variables(equation, found);
365            collect_variables(dependent_variable_call, found);
366        }
367        _ => {}
368    }
369}

387    fn univariate_round_trip_through_polynomial() {
388        let mut context = Context::new();
389        let identifier = context.symbol_table.intern_symbol("x");
390        let x = symbol(&mut context, "x");
391        let two = integer_value(&mut context, 2);
392        let three = integer_value(&mut context, 3);
393        let x_squared = power(&mut context, Arc::clone(&x), two);
394        let three_x = product(&mut context, vec![three, Arc::clone(&x)]);
395        let one = integer_value(&mut context, 1);
396        let target = sum(&mut context, vec![x_squared, three_x, one]);
397        let polynomial_value = expression_to_univariate(&target, identifier).unwrap();
398        assert_eq!(degree(&polynomial_value), Some(2));
399        let recovered = polynomial_to_expression(&mut context, &polynomial_value);
400        assert!(Arc::ptr_eq(&recovered, &target));
