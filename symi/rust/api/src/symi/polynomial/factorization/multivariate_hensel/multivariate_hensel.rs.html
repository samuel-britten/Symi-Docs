22/// Performs the advanced native `multivariate_hensel_lift` operation.
23pub fn multivariate_hensel_lift(
24    target: &Polynomial,
25    univariate_factors: &[Polynomial],
26    leading_distribution: &[Polynomial],
27    evaluation_point: &[exact_integer],
28    variable_order: &[SymbolIdentifier],
29) -> Option<Vec<Polynomial>> {
30    // Wang/EEZ multivariate Hensel lift. Starting from univariate factors of
31    // target evaluated at the trailing-variable point, lifts one variable at
32    // a time until the full multivariate factorisation is recovered. Leading
33    // coefficients are tracked through leading_distribution so the lifted
34    // factors land with the predetermined multivariate leading coefficient
35    // exactly — without that constraint Wang's classical lift overcounts on
36    // bad-LC inputs.
37    if variable_order.is_empty() {
38        return None;
39    }
40    if univariate_factors.is_empty() {
41        return None;
42    }
43    if univariate_factors.len() != leading_distribution.len() {
44        return None;
45    }
46    if variable_order.len() == 1 {
47        if compute_factor_product(univariate_factors) == *target {
48            return Some(univariate_factors.to_vec());
49        }
50        return None;
51    }
52    let n = variable_order.len();
53    if evaluation_point.len() != n - 1 {
54        return None;
55    }
56    let mut current_factors: Vec<Polynomial> = univariate_factors.to_vec();
57    for k in 2..=n {
58        let restricted_target =
59            restrict_target_to_depth(target, variable_order, evaluation_point, k)?;
60        let restricted_lcs: Vec<Polynomial> = leading_distribution
61            .iter()
62            .map(|lc| {
63                restrict_polynomial_to_depth(lc, &variable_order[1..], evaluation_point, k - 1)
64            })
65            .collect();
66        current_factors = lift_one_variable(
67            &current_factors,
68            &restricted_target,
69            &restricted_lcs,
70            variable_order[k - 1],
71            &evaluation_point[k - 2],
72            &variable_order[..k],
73        )?;
74    }
75    if compute_factor_product(&current_factors) != *target {
76        return None;
77    }
78    Some(current_factors)
79}

81fn compute_factor_product(factors: &[Polynomial]) -> Polynomial {
82    // Multiply a non-empty slice of factors into a single polynomial.
83    // Caller-side helper that lets multivariate_hensel_lift assert the
84    // post-condition ∏ factors == target without rewriting the loop at the
85    // call site.
86    if factors.is_empty() {
87        panic!("compute_factor_product called with empty slice");
88    }
89    let mut accumulator = factors[0].clone();
90    for factor in factors.iter().skip(1) {
91        accumulator = multiply(&accumulator, factor);
92    }
93    accumulator
94}

96fn restrict_target_to_depth(
97    target: &Polynomial,
98    variable_order: &[SymbolIdentifier],
99    evaluation_point: &[exact_integer],
100    depth: usize,
101) -> Option<Polynomial> {
102    // Substitute x_{depth+1} = α_{depth+1}, …, x_n = α_n into target,
103    // producing a polynomial in (x_1, …, x_{depth}). Used at each Hensel
104    // lift step to project the full polynomial down to the depth the lift
105    // is currently working at.
106    if depth >= variable_order.len() {
107        return Some(target.clone());
108    }
109    let mut current = target.clone();
110    for index in depth..variable_order.len() {
111        let variable = variable_order[index];
112        let alpha = &evaluation_point[index - 1];
113        current = evaluate_polynomial_at_variable(&current, variable, alpha);
114    }
115    Some(current)
116}

118fn restrict_polynomial_to_depth(
119    value: &Polynomial,
120    variable_order: &[SymbolIdentifier],
121    evaluation_point: &[exact_integer],
122    depth: usize,
123) -> Polynomial {
124    // Sibling helper to restrict_target_to_depth used by the
125    // leading-coefficient restriction. The variable_order passed in here is
126    // already the tail (x_2, …, x_n) because the leading coefficient lives
127    // in the trailing variables; the same evaluation_point is reused
128    // because the indexing convention treats the head variable as absent.
129    if depth >= variable_order.len() {
130        return value.clone();
131    }
132    let mut current = value.clone();
133    for index in depth..variable_order.len() {
134        let variable = variable_order[index];
135        let alpha = &evaluation_point[index];
136        current = evaluate_polynomial_at_variable(&current, variable, alpha);
137    }
138    current
139}

141fn lift_one_variable(
142    factors: &[Polynomial],
143    target: &Polynomial,
144    leading_coefficients: &[Polynomial],
145    lift_variable: SymbolIdentifier,
146    alpha: &exact_integer,
147    variable_order_up_to_lift: &[SymbolIdentifier],
148) -> Option<Vec<Polynomial>> {
149    // Add x_k dependence to the current factors so the lifted product agrees
150    // with target. Strategy: convert target to its Taylor expansion in
151    // y = x_k - α_k, expand the factors so the leading coefficient matches
152    // the predetermined multivariate value, and run the standard
153    // coefficient-by-coefficient Hensel correction with a multivariate
154    // Diophantine solver. The lift is exact — target is recovered modulo
155    // y^{d+1} where d = deg_{x_k}(target), and the polynomial has degree at
156    // most d in x_k so the result holds without further refinement.
157    let lower_variables = &variable_order_up_to_lift[..variable_order_up_to_lift.len() - 1];
158    let head_variable = variable_order_up_to_lift[0];
159    let target_taylor = shift_to_taylor(target, lift_variable, alpha, head_variable);
160    let degree_in_lift_variable = degree_in_variable(target, lift_variable);
161    let mut factor_taylors: Vec<Vec<Polynomial>> = Vec::with_capacity(factors.len());
162    for index in 0..factors.len() {
163        let mut taylor: Vec<Polynomial> =
164            vec![zero_polynomial(head_variable); degree_in_lift_variable + 1];
165        taylor[0] = factors[index].clone();
166        adjust_factor_for_leading_coefficient(
167            &mut taylor,
168            &factors[index],
169            &leading_coefficients[index],
170            lift_variable,
171            alpha,
172            head_variable,
173            variable_order_up_to_lift,
174        )?;
175        factor_taylors.push(taylor);
176    }
177    let base_factors: Vec<Polynomial> = factors.to_vec();
178    let target_product_at_y0 = compute_factor_product(&base_factors);
179    if target_product_at_y0 != target_taylor[0] {
180        return None;
181    }
182    for j in 1..=degree_in_lift_variable {
183        let product_truncated = multiply_taylors_truncated(&factor_taylors, j + 1);
184        let residual = subtract(&target_taylor[j], &product_truncated[j]);
185        if is_zero(&residual) {
186            continue;
187        }
188        let corrections = multivariate_diophantine(
189            &base_factors,
190            &residual,
191            &evaluation_point_for_lower_depth(),
192            lower_variables,
193        )?;
194        for (factor_index, correction) in corrections.iter().enumerate() {
195            if !is_zero(correction) {
196                factor_taylors[factor_index][j] = add(&factor_taylors[factor_index][j], correction);
197            }
198        }
199    }
200    let new_factors: Vec<Polynomial> = factor_taylors
201        .into_iter()
202        .map(|taylor| shift_from_taylor(&taylor, lift_variable, alpha, variable_order_up_to_lift))
203        .collect();
204    Some(new_factors)
205}

207fn evaluation_point_for_lower_depth() -> Vec<exact_integer> {
208    // The lower-depth multivariate_diophantine recursion in this module
209    // never substitutes variables — it solves directly in the current ring.
210    // Passing an empty point makes that explicit: the base-case solver never
211    // dereferences it, and the depth-1 recursion below carries its own
212    // explicit substitutions when needed.
213    Vec::new()
214}

216fn adjust_factor_for_leading_coefficient(
217    taylor: &mut [Polynomial],
218    base_factor: &Polynomial,
219    target_leading_coefficient: &Polynomial,
220    lift_variable: SymbolIdentifier,
221    alpha: &exact_integer,
222    head_variable: SymbolIdentifier,
223    variable_order_up_to_lift: &[SymbolIdentifier],
224) -> Option<()> {
225    // Pre-load the Taylor entries y^1, y^2, … with the difference between
226    // the predetermined multivariate leading coefficient and the
227    // already-baked-in leading coefficient of base_factor (in x_1). This
228    // gives the factor the correct top-x_1 coefficient at every y-level so
229    // the subsequent Hensel correction only needs to fix the strictly
230    // lower-x_1-degree slots — matching the Wang/EEZ invariant
231    // deg_{x_1}(δ_i) < deg_{x_1}(u_i). The leading-coefficient comparison
232    // routes through subtract_coefficients (PolynomialCoefficient arithmetic)
233    // because the target lc lives in (x_2, …, x_k) while the base factor's
234    // own lc is a number — direct polynomial subtraction would fail the
235    // main_variable consistency check.
236    let factor_degree_in_head = degree(base_factor).unwrap_or(0);
237    if factor_degree_in_head == 0 {
238        return Some(());
239    }
240    let current_leading_coefficient =
241        leading_coefficient(base_factor).unwrap_or_else(coefficient_zero);
242    let target_leading_as_coefficient = if degree(target_leading_coefficient).unwrap_or(0) == 0 {
243        leading_coefficient(target_leading_coefficient).unwrap_or_else(coefficient_zero)
244    } else {
245        simplify_coefficient(PolynomialCoefficient::Nested(Box::new(
246            target_leading_coefficient.clone(),
247        )))
248    };
249    let leading_difference_coefficient =
250        subtract_coefficients(&target_leading_as_coefficient, &current_leading_coefficient)?;
251    if coefficient_is_zero(&leading_difference_coefficient) {
252        return Some(());
253    }
254    let leading_difference =
255        coefficient_as_polynomial(&leading_difference_coefficient, head_variable);
256    let leading_difference_taylor =
257        shift_to_taylor(&leading_difference, lift_variable, alpha, head_variable);
258    let head_polynomial_pow_head_variable = build_head_power(
259        head_variable,
260        factor_degree_in_head,
261        variable_order_up_to_lift,
262    );
263    for (j, leading_taylor_entry) in leading_difference_taylor.iter().enumerate() {
264        if j == 0 {
265            if !is_zero(leading_taylor_entry) {
266                return None;
267            }
268            continue;
269        }
270        if j >= taylor.len() {
271            if !is_zero(leading_taylor_entry) {
272                return None;
273            }
274            continue;
275        }
276        let addend = multiply(leading_taylor_entry, &head_polynomial_pow_head_variable);
277        taylor[j] = add(&taylor[j], &addend);
278    }
279    Some(())
280}

282fn build_head_power(
283    head_variable: SymbolIdentifier,
284    degree: usize,
285    variable_order_up_to_lift: &[SymbolIdentifier],
286) -> Polynomial {
287    // Build head_variable^degree as a polynomial in variable_order_up_to_lift.
288    // Used by adjust_factor_for_leading_coefficient to embed the leading-x_1
289    // monomial when applying the Taylor correction.
290    let _ = variable_order_up_to_lift;
291    if degree == 0 {
292        return constant_polynomial(head_variable, Number::Integer(integer_one()));
293    }
294    Polynomial {
295        main_variable: head_variable,
296        terms: vec![(
297            degree,
298            PolynomialCoefficient::Numeric(Number::Integer(integer_one())),
299        )],
300    }
301}

303/// Performs the advanced native `shift_to_taylor` operation.
304pub fn shift_to_taylor(
305    target: &Polynomial,
306    lift_variable: SymbolIdentifier,
307    alpha: &exact_integer,
308    head_variable: SymbolIdentifier,
309) -> Vec<Polynomial> {
310    // Compute the Taylor expansion of target around lift_variable = alpha.
311    // Returns Vec<polynomial>, with entry j giving the y^j coefficient as a
312    // polynomial in the variables that remain after lift_variable is
313    // removed. The expansion uses the direct binomial form
314    // c_j = Σ_{m ≥ j} a_m · C(m, j) · alpha^{m-j} so no division by j! is
315    // required — the result stays in Z whenever the input does.
316    let degree_in_lift = degree_in_variable(target, lift_variable);
317    let mut a_values: Vec<PolynomialCoefficient> = Vec::with_capacity(degree_in_lift + 1);
318    for m in 0..=degree_in_lift {
319        a_values.push(coefficient_in_variable(target, lift_variable, m));
320    }
321    let mut result: Vec<Polynomial> = Vec::with_capacity(degree_in_lift + 1);
322    for j in 0..=degree_in_lift {
323        let mut accumulator_coefficient = coefficient_zero();
324        for m in j..=degree_in_lift {
325            let binomial = binomial_coefficient(m, j);
326            let alpha_power = integer_power(alpha, (m - j) as u32);
327            let factor = binomial * alpha_power;
328            if factor != integer_zero() {
329                let scaled = multiply_coefficient_numeric(&a_values[m], &Number::Integer(factor));
330                accumulator_coefficient =
331                    add_coefficients_or_zero(&accumulator_coefficient, &scaled);
332            }
333        }
334        let coefficient_polynomial =
335            coefficient_as_polynomial(&accumulator_coefficient, head_variable);
336        result.push(coefficient_polynomial);
337    }
338    result
339}

341fn coefficient_as_polynomial(
342    coefficient: &PolynomialCoefficient,
343    head_variable: SymbolIdentifier,
344) -> Polynomial {
345    // Wrap a PolynomialCoefficient as a polynomial whose main variable is
346    // head_variable. Numeric values become constant polynomials; nested
347    // polynomials whose main variable already matches are returned
348    // unchanged, otherwise they sit as the constant coefficient.
349    match coefficient {
350        PolynomialCoefficient::Numeric(value) => {
351            if crate::number::is_zero(value) {
352                zero_polynomial(head_variable)
353            } else {
354                constant_polynomial(head_variable, value.clone())
355            }
356        }
357        PolynomialCoefficient::Nested(nested) => {
358            if nested.main_variable == head_variable {
359                (**nested).clone()
360            } else {
361                Polynomial {
362                    main_variable: head_variable,
363                    terms: vec![(0, PolynomialCoefficient::Nested(nested.clone()))],
364                }
365            }
366        }
367        PolynomialCoefficient::Algebraic(_) => Polynomial {
368            main_variable: head_variable,
369            terms: vec![(0, coefficient.clone())],
370        },
371    }
372}

374/// Performs the advanced native `shift_from_taylor` operation.
375pub fn shift_from_taylor(
376    taylor: &[Polynomial],
377    lift_variable: SymbolIdentifier,
378    alpha: &exact_integer,
379    variable_order_up_to_lift: &[SymbolIdentifier],
380) -> Polynomial {
381    // Reconstruct a polynomial in (x_1, …, x_k) from its y-Taylor expansion
382    // around lift_variable = alpha. Builds (lift_variable - alpha) once
383    // through build_variable_polynomial / subtract and multiplies it through
384    // the running power to produce y^j; the polynomial-arithmetic path keeps
385    // the result canonical without manual coefficient management.
386    let head_variable = variable_order_up_to_lift[0];
387    if taylor.is_empty() {
388        return zero_polynomial(head_variable);
389    }
390    let var_polynomial = build_variable_polynomial(lift_variable, variable_order_up_to_lift);
391    let alpha_polynomial = constant_polynomial(head_variable, Number::Integer(alpha.clone()));
392    let var_minus_alpha = subtract(&var_polynomial, &alpha_polynomial);
393    let one_polynomial = constant_polynomial(head_variable, Number::Integer(integer_one()));
394    let mut result = zero_polynomial(head_variable);
395    let mut current_power = one_polynomial;
396    for j in 0..taylor.len() {
397        if j > 0 {
398            current_power = multiply(&current_power, &var_minus_alpha);
399        }
400        if !is_zero(&taylor[j]) {
401            let term = multiply(&taylor[j], &current_power);
402            result = add(&result, &term);
403        }
404    }
405    result
406}

408fn multiply_taylors_truncated(
409    factors: &[Vec<Polynomial>],
410    truncation_degree: usize,
411) -> Vec<Polynomial> {
412    // Multiply a list of Taylor expansions and truncate the result so y^d
413    // entries with d ≥ truncation_degree are dropped. Used to compute the
414    // current product modulo y^(j+1) during the Hensel iteration without
415    // expanding to the full degree at every step.
416    if factors.is_empty() {
417        return Vec::new();
418    }
419    let head_variable = factors[0][0].main_variable;
420    let mut accumulator: Vec<Polynomial> = vec![zero_polynomial(head_variable); truncation_degree];
421    accumulator[0] = constant_polynomial(head_variable, Number::Integer(integer_one()));
422    for factor in factors.iter() {
423        let mut next: Vec<Polynomial> = vec![zero_polynomial(head_variable); truncation_degree];
424        for j in 0..accumulator.len() {
425            if is_zero(&accumulator[j]) {
426                continue;
427            }
428            for i in 0..factor.len() {
429                if i + j >= truncation_degree {
430                    break;
431                }
432                if is_zero(&factor[i]) {
433                    continue;
434                }
435                let term = multiply(&accumulator[j], &factor[i]);
436                next[i + j] = add(&next[i + j], &term);
437            }
438        }
439        accumulator = next;
440    }
441    accumulator
442}

444fn multivariate_diophantine(
445    factors: &[Polynomial],
446    target: &Polynomial,
447    evaluation_point: &[exact_integer],
448    variable_order: &[SymbolIdentifier],
449) -> Option<Vec<Polynomial>> {
450    // Solve Σ_i δ_i · ∏_{j ≠ i} factors[j] = target. δ_i live in the same
451    // multivariate ring as factors; the degree-in-head constraint
452    // deg_{x_1}(δ_i) < deg_{x_1}(factors[i]) keeps the system square.
453    // For higher-depth inputs we delegate the lift to the y-Taylor scheme
454    // built around evaluate-at-α and shift_to_taylor / shift_from_taylor;
455    // the depth-1 base case routes through
456    // solve_univariate_diophantine_over_rationals which uses Q[x_1]
457    // partial-fractions internally.
458    let _ = evaluation_point;
459    if factors.is_empty() {
460        return None;
461    }
462    if variable_order.len() == 1 {
463        return solve_univariate_diophantine_over_rationals(factors, target);
464    }
465    if is_zero(target) {
466        return Some(
467            factors
468                .iter()
469                .map(|f| zero_polynomial(f.main_variable))
470                .collect(),
471        );
472    }
473    let lift_variable = *variable_order.last().unwrap();
474    let lower_variables = &variable_order[..variable_order.len() - 1];
475    let alpha_index = lift_variable_index_in_factors(factors, lift_variable, lower_variables);
476    let alpha = alpha_index.unwrap_or(integer_zero());
477    let lower_factors: Vec<Polynomial> = factors
478        .iter()
479        .map(|f| evaluate_polynomial_at_variable(f, lift_variable, &alpha))
480        .collect();
481    let lower_target = evaluate_polynomial_at_variable(target, lift_variable, &alpha);
482    let lower_solutions = multivariate_diophantine(
483        &lower_factors,
484        &lower_target,
485        evaluation_point,
486        lower_variables,
487    )?;
488    let mut current_solutions: Vec<Vec<Polynomial>> = Vec::with_capacity(factors.len());
489    let head_variable = variable_order[0];
490    let degree_target = degree_in_variable(target, lift_variable);
491    for solution in lower_solutions.iter() {
492        let mut taylor: Vec<Polynomial> = vec![zero_polynomial(head_variable); degree_target + 1];
493        taylor[0] = solution.clone();
494        current_solutions.push(taylor);
495    }
496    let target_taylor = shift_to_taylor(target, lift_variable, &alpha, head_variable);
497    let factor_taylors: Vec<Vec<Polynomial>> = factors
498        .iter()
499        .map(|f| shift_to_taylor(f, lift_variable, &alpha, head_variable))
500        .collect();
501    for j in 1..=degree_target {
502        let solutions_at_level_j: Vec<Polynomial> =
503            current_solutions.iter().map(|t| t[j].clone()).collect();
504        let lhs_at_j = compute_diophantine_taylor_residual(&factor_taylors, &current_solutions, j);
505        let target_at_j = if j < target_taylor.len() {
506            target_taylor[j].clone()
507        } else {
508            zero_polynomial(head_variable)
509        };
510        let residual = subtract(&target_at_j, &lhs_at_j);
511        let _ = solutions_at_level_j;
512        if is_zero(&residual) {
513            continue;
514        }
515        let inner_solutions =
516            multivariate_diophantine(&lower_factors, &residual, evaluation_point, lower_variables)?;
517        for (index, value) in inner_solutions.iter().enumerate() {
518            current_solutions[index][j] = value.clone();
519        }
520    }
521    let result: Vec<Polynomial> = current_solutions
522        .iter()
523        .map(|taylor| shift_from_taylor(taylor, lift_variable, &alpha, variable_order))
524        .collect();
525    Some(result)
526}

528fn lift_variable_index_in_factors(
529    factors: &[Polynomial],
530    lift_variable: SymbolIdentifier,
531    lower_variables: &[SymbolIdentifier],
532) -> Option<exact_integer> {
533    // Always evaluate the recursive Diophantine at zero in the new
534    // variable. This matches Wang's evaluation point conventions for the
535    // Hensel-style lift inside the Diophantine recursion: the inner system
536    // is solved relative to the user's α only at the outermost level, and
537    // every deeper "expand around" step naturally happens around zero in
538    // its own variable.
539    let _ = (factors, lift_variable, lower_variables);
540    Some(integer_zero())
541}

543fn compute_diophantine_taylor_residual(
544    factor_taylors: &[Vec<Polynomial>],
545    solution_taylors: &[Vec<Polynomial>],
546    j: usize,
547) -> Polynomial {
548    // Compute the y^j coefficient of Σ_i solutions[i] · ∏_{k ≠ i} factors[k],
549    // each multi-Taylor expansion participating in the convolution at the
550    // appropriate degree. Used by the recursive multivariate_diophantine to
551    // produce the residual that the inner solver should annihilate.
552    let head_variable = factor_taylors[0][0].main_variable;
553    let mut total = zero_polynomial(head_variable);
554    for index in 0..factor_taylors.len() {
555        let cofactor_taylor = build_cofactor_taylor(factor_taylors, index, j + 1);
556        let solution_taylor = &solution_taylors[index];
557        let mut term_j = zero_polynomial(head_variable);
558        for s_idx in 0..=j {
559            if s_idx >= solution_taylor.len() {
560                break;
561            }
562            let c_idx = j - s_idx;
563            if c_idx >= cofactor_taylor.len() {
564                continue;
565            }
566            if is_zero(&solution_taylor[s_idx]) || is_zero(&cofactor_taylor[c_idx]) {
567                continue;
568            }
569            let product = multiply(&solution_taylor[s_idx], &cofactor_taylor[c_idx]);
570            term_j = add(&term_j, &product);
571        }
572        total = add(&total, &term_j);
573    }
574    total
575}

577fn build_cofactor_taylor(
578    factor_taylors: &[Vec<Polynomial>],
579    skip_index: usize,
580    truncation_degree: usize,
581) -> Vec<Polynomial> {
582    // Build the Taylor expansion of ∏_{k ≠ skip_index} factors[k], truncated
583    // to degree < truncation_degree. Wraps multiply_taylors_truncated with
584    // the obvious filtering.
585    let filtered: Vec<Vec<Polynomial>> = factor_taylors
586        .iter()
587        .enumerate()
588        .filter_map(|(index, taylor)| {
589            if index != skip_index {
590                Some(taylor.clone())
591            } else {
592                None
593            }
594        })
595        .collect();
596    multiply_taylors_truncated(&filtered, truncation_degree)
597}

599fn solve_univariate_diophantine_over_rationals(
600    factors: &[Polynomial],
601    target: &Polynomial,
602) -> Option<Vec<Polynomial>> {
603    // Base case of the multivariate Diophantine recursion. Solve
604    // Σ_i δ_i · U_i = target in Q[x] where U_i = ∏_{j ≠ i} factors[j],
605    // then verify each δ_i is integer-valued. We work in Q[x] via
606    // representative coefficient vectors of Rationals; the extended
607    // Euclidean / partial-fractions structure stays linear-algebra-friendly
608    // because Q[x] is a Euclidean domain.
609    if factors.is_empty() {
610        return None;
611    }
612    let head_variable = factors[0].main_variable;
613    if factors.len() == 1 {
614        let factor_rationals = polynomial_to_rational_vector(&factors[0])?;
615        let target_rationals = polynomial_to_rational_vector(target)?;
616        let (_, remainder) = rational_polynomial_divide(&target_rationals, &factor_rationals)?;
617        return Some(vec![rational_vector_to_polynomial(
618            &remainder,
619            head_variable,
620        )?]);
621    }
622    let factor_rationals: Vec<Vec<exact_rational>> = factors
623        .iter()
624        .map(|factor| polynomial_to_rational_vector(factor).unwrap_or_default())
625        .collect();
626    let target_rationals = polynomial_to_rational_vector(target)?;
627    let solutions = solve_diophantine_recursive(&factor_rationals, &target_rationals)?;
628    let mut result: Vec<Polynomial> = Vec::with_capacity(solutions.len());
629    for solution in solutions.iter() {
630        result.push(rational_vector_to_polynomial(solution, head_variable)?);
631    }
632    Some(result)
633}
634
635fn solve_diophantine_recursive(
636    factors: &[Vec<exact_rational>],
637    target: &[exact_rational],
638) -> Option<Vec<Vec<exact_rational>>> {
639    // Partial-fractions style recursion in Q[x]. Solve
640    //   s · U_first + t · F_first = target
641    // where U_first = ∏_{j ≠ first} factors[j] and F_first = factors[0],
642    // reduce s modulo F_first to enforce the degree constraint, fold the
643    // residue into the right-hand side for the inner recursion. Caller
644    // ensures factors are pairwise coprime in Q[x] — guaranteed by
645    // factor_squarefree_univariate having produced a clean factorisation
646    // of the univariate image.
647    if factors.is_empty() {
648        return None;
649    }
650    if factors.len() == 1 {
651        let (_, remainder) = rational_polynomial_divide(target, &factors[0])?;
652        return Some(vec![remainder]);
653    }
654    let f_first = factors[0].clone();
655    let mut tail_product: Vec<exact_rational> = vec![exact_rational::from(1)];
656    for factor in factors.iter().skip(1) {
657        tail_product = rational_polynomial_multiply(&tail_product, factor);
658    }
659    let (gcd_value, s_coefficient, t_coefficient) =
660        rational_polynomial_extended_gcd(&f_first, &tail_product)?;
661    if rational_polynomial_degree(&gcd_value) > 0 {
662        return None;
663    }
664    let leading = gcd_value.last().cloned().unwrap_or(exact_rational::from(1));
665    if leading == exact_rational::from(0) {
666        return None;
667    }
668    let s_scaled =
669        rational_polynomial_scale(&s_coefficient, &(exact_rational::from(1) / leading.clone()));
670    let t_scaled =
671        rational_polynomial_scale(&t_coefficient, &(exact_rational::from(1) / leading.clone()));
672    let s_times_target = rational_polynomial_multiply(&t_scaled, target);
673    let (quotient_first, s_first) = rational_polynomial_divide(&s_times_target, &f_first)?;
674    let t_times_target = rational_polynomial_multiply(&s_scaled, target);
675    let new_target = rational_polynomial_add(
676        &t_times_target,
677        &rational_polynomial_multiply(&quotient_first, &tail_product),
678    );
679    let inner_solutions = solve_diophantine_recursive(&factors[1..], &new_target)?;
680    let mut result = Vec::with_capacity(factors.len());
681    result.push(s_first);
682    result.extend(inner_solutions);
683    Some(result)
684}

686fn polynomial_to_rational_vector(value: &Polynomial) -> Option<Vec<exact_rational>> {
687    // Convert a univariate polynomial into a dense ascending-degree
688    // exact_rational coefficient vector. Returns None if the polynomial contains
689    // a non-numeric coefficient (i.e. is genuinely multivariate); the base
690    // Diophantine solver only operates on Q[x].
691    let degree_value = degree(value).unwrap_or(0);
692    let mut result = vec![exact_rational::from(0); degree_value + 1];
693    for (exponent, coefficient) in value.terms.iter() {
694        let numeric = coefficient_to_number(coefficient)?;
695        let rational = match numeric {
696            Number::Integer(integer_value) => rational_from_integer(integer_value.clone()),
697            Number::Rational(rational_value) => rational_value,
698        };
699        result[*exponent] = rational;
700    }
701    Some(result)
702}
703
704fn rational_vector_to_polynomial(
705    vector: &[exact_rational],
706    main_variable: SymbolIdentifier,
707) -> Option<Polynomial> {
708    // Inverse of polynomial_to_rational_vector. Returns None if any
709    // coefficient is not an integer — the base Diophantine solver enforces
710    // an integer result via the predetermined leading coefficients, so a
711    // non-integer δ indicates the input was outside the expected Wang/EEZ
712    // shape. The exact-arithmetic rational-to-integer conversion preserves the
713    // sign of negative rationals and declines when the value is not integral.
714    let mut new_terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
715    for (exponent, coefficient) in vector.iter().enumerate() {
716        if *coefficient == exact_rational::from(0) {
717            continue;
718        }
719        let integer_value = rational_to_integer(coefficient)?;
720        new_terms.push((
721            exponent,
722            PolynomialCoefficient::Numeric(Number::Integer(integer_value)),
723        ));
724    }
725    new_terms.sort_by(|left, right| right.0.cmp(&left.0));
726    Some(Polynomial {
727        main_variable,
728        terms: new_terms,
729    })
730}

732fn rational_polynomial_degree(value: &[exact_rational]) -> isize {
733    // Highest degree with a non-zero coefficient. Returns -1 for the zero
734    // polynomial so callers can compare degrees uniformly via signed
735    // arithmetic without special-casing.
736    for index in (0..value.len()).rev() {
737        if value[index] != exact_rational::from(0) {
738            return index as isize;
739        }
740    }
741    -1
742}

744fn rational_polynomial_add(
745    left: &[exact_rational],
746    right: &[exact_rational],
747) -> Vec<exact_rational> {
748    // Pad-and-add two dense ascending-degree exact_rational vectors.
749    let length = left.len().max(right.len());
750    let mut result = vec![exact_rational::from(0); length];
751    for (index, value) in left.iter().enumerate() {
752        result[index] = result[index].clone() + value.clone();
753    }
754    for (index, value) in right.iter().enumerate() {
755        result[index] = result[index].clone() + value.clone();
756    }
757    trim_trailing_zeros(&mut result);
758    result
759}
760
761fn rational_polynomial_subtract(
762    left: &[exact_rational],
763    right: &[exact_rational],
764) -> Vec<exact_rational> {
765    // Pad-and-subtract two dense ascending-degree exact_rational vectors.
766    let length = left.len().max(right.len());
767    let mut result = vec![exact_rational::from(0); length];
768    for (index, value) in left.iter().enumerate() {
769        result[index] = result[index].clone() + value.clone();
770    }
771    for (index, value) in right.iter().enumerate() {
772        result[index] = result[index].clone() - value.clone();
773    }
774    trim_trailing_zeros(&mut result);
775    result
776}
777
778fn rational_polynomial_multiply(
779    left: &[exact_rational],
780    right: &[exact_rational],
781) -> Vec<exact_rational> {
782    // Schoolbook multiplication of dense ascending-degree exact_rational vectors.
783    if left.is_empty() || right.is_empty() {
784        return Vec::new();
785    }
786    let mut result = vec![exact_rational::from(0); left.len() + right.len() - 1];
787    for (left_index, left_value) in left.iter().enumerate() {
788        if *left_value == exact_rational::from(0) {
789            continue;
790        }
791        for (right_index, right_value) in right.iter().enumerate() {
792            if *right_value == exact_rational::from(0) {
793                continue;
794            }
795            let product = left_value.clone() * right_value.clone();
796            result[left_index + right_index] = result[left_index + right_index].clone() + product;
797        }
798    }
799    trim_trailing_zeros(&mut result);
800    result
801}

803fn rational_polynomial_scale(
804    value: &[exact_rational],
805    factor: &exact_rational,
806) -> Vec<exact_rational> {
807    // Scale every coefficient of a exact_rational vector by a non-zero scalar.
808    let mut result: Vec<exact_rational> =
809        value.iter().map(|v| v.clone() * factor.clone()).collect();
810    trim_trailing_zeros(&mut result);
811    result
812}

814fn rational_polynomial_divide(
815    dividend: &[exact_rational],
816    divisor: &[exact_rational],
817) -> Option<(Vec<exact_rational>, Vec<exact_rational>)> {
818    // Long division in Q[x]. Returns (quotient, remainder) with
819    // deg(remainder) < deg(divisor); each step divides by the divisor's
820    // leading coefficient — a field operation that always succeeds in Q.
821    let divisor_degree = rational_polynomial_degree(divisor);
822    if divisor_degree < 0 {
823        return None;
824    }
825    let mut working = dividend.to_vec();
826    let mut quotient = vec![exact_rational::from(0); 1];
827    let divisor_leading = divisor[divisor_degree as usize].clone();
828    loop {
829        let working_degree = rational_polynomial_degree(&working);
830        if working_degree < divisor_degree {
831            break;
832        }
833        let degree_difference = (working_degree - divisor_degree) as usize;
834        let working_leading = working[working_degree as usize].clone();
835        let coefficient = working_leading / divisor_leading.clone();
836        while quotient.len() <= degree_difference {
837            quotient.push(exact_rational::from(0));
838        }
839        quotient[degree_difference] = quotient[degree_difference].clone() + coefficient.clone();
840        for (index, divisor_coefficient) in divisor.iter().enumerate() {
841            if *divisor_coefficient == exact_rational::from(0) {
842                continue;
843            }
844            let position = index + degree_difference;
845            while working.len() <= position {
846                working.push(exact_rational::from(0));
847            }
848            working[position] =
849                working[position].clone() - coefficient.clone() * divisor_coefficient.clone();
850        }
851        trim_trailing_zeros(&mut working);
852        if working.is_empty() {
853            break;
854        }
855    }
856    trim_trailing_zeros(&mut quotient);
857    Some((quotient, working))
858}

860fn rational_polynomial_extended_gcd(
861    left: &[exact_rational],
862    right: &[exact_rational],
863) -> Option<(
864    Vec<exact_rational>,
865    Vec<exact_rational>,
866    Vec<exact_rational>,
867)> {
868    // Extended Euclidean algorithm in Q[x]. Returns (g, s, t) with
869    // s · left + t · right = g. g is monic up to the final divide step;
870    // the caller handles unit normalisation if needed.
871    let mut old_remainder = left.to_vec();
872    let mut remainder = right.to_vec();
873    let mut old_s = vec![exact_rational::from(1)];
874    let mut s_value: Vec<exact_rational> = Vec::new();
875    let mut old_t: Vec<exact_rational> = Vec::new();
876    let mut t_value = vec![exact_rational::from(1)];
877    while !is_rational_polynomial_zero(&remainder) {
878        let (quotient, new_remainder) = rational_polynomial_divide(&old_remainder, &remainder)?;
879        let new_s = rational_polynomial_subtract(
880            &old_s,
881            &rational_polynomial_multiply(&quotient, &s_value),
882        );
883        let new_t = rational_polynomial_subtract(
884            &old_t,
885            &rational_polynomial_multiply(&quotient, &t_value),
886        );
887        old_remainder = remainder;
888        remainder = new_remainder;
889        old_s = s_value;
890        s_value = new_s;
891        old_t = t_value;
892        t_value = new_t;
893    }
894    Some((old_remainder, old_s, old_t))
895}
896
897fn is_rational_polynomial_zero(value: &[exact_rational]) -> bool {
898    // True iff every coefficient is exactly zero, irrespective of vector
899    // length. Used by the Euclidean loop to detect termination.
900    value
901        .iter()
902        .all(|coefficient| *coefficient == exact_rational::from(0))
903}
904
905fn trim_trailing_zeros(value: &mut Vec<exact_rational>) {
906    // Shrink a exact_rational vector by dropping the trailing zero coefficients.
907    // Keeps the canonical form predictable for the divide / extended-gcd
908    // routines.
909    while value
910        .last()
911        .map_or(false, |coefficient| *coefficient == exact_rational::from(0))
912    {
913        value.pop();
914    }
915}
