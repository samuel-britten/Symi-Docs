18/// Public native data structure named `isolating_interval`.
19pub struct IsolatingInterval {
20    /// Stores the `lower` value.
21    pub lower: exact_rational,
22    /// Stores the `upper` value.
23    pub upper: exact_rational,
24}
25
26impl IsolatingInterval {
27    /// Provides the `is_exact` operation on this native type.
28    pub fn is_exact(&self) -> bool {
29        // True when the interval has collapsed to a single rational point, which
30        // happens exactly when the isolated root is itself rational.
31        self.lower == self.upper
32    }
33}

35/// Performs the advanced native `isolate_real_roots` operation.
36pub fn isolate_real_roots(value: &Polynomial) -> Option<Vec<IsolatingInterval>> {
37    // Return one isolating interval per distinct real root of value, sorted by
38    // ascending position. Each interval contains exactly one real root of the
39    // squarefree part; a collapsed [r, r] interval marks an exact rational root.
40    // Returns None when value is not a univariate polynomial over the rationals
41    // or when it is the zero polynomial (which has every real as a root).
42    let dense = dense_from_polynomial(value)?;
43    if dense.is_empty() {
44        return None;
45    }
46    Some(isolate_real_roots_dense(&dense))
47}

49pub(crate) fn isolate_real_roots_with_operation(
50    value: &Polynomial,
51    operation: &OperationContext,
52) -> Option<Vec<IsolatingInterval>> {
53    // The isolation scheduled by the operation that asked for it. The decision itself is taken
54    // inside the recursion, where the two halves it would hand out are known.
55    let dense = dense_from_polynomial(value)?;
56    if dense.is_empty() {
57        return None;
58    }
59    Some(isolate_real_roots_dense_with_operation(
60        &dense,
61        Some(operation),
62    ))
63}

65/// Performs the advanced native `count_distinct_real_roots` operation.
66pub fn count_distinct_real_roots(value: &Polynomial) -> Option<usize> {
67    // The number of distinct real roots of value, computed as the Sturm sign
68    // variation difference V(-infinity) - V(+infinity), which is unambiguous
69    // because neither bound is ever a root. None for non-rational or zero input.
70    let dense = dense_from_polynomial(value)?;
71    if dense.is_empty() {
72        return None;
73    }
74    let squarefree = squarefree_part(&dense);
75    Some(sturm_count_total(&squarefree))
76}

78/// Performs the advanced native `real_root_count` operation.
79pub fn real_root_count(value: &Polynomial, lower: &Number, upper: &Number) -> Option<usize> {
80    // The number of distinct real roots of value in the half-open interval
81    // (lower, upper], computed from the Sturm chain as V(lower) - V(upper).
82    // Callers pass bounds that are not themselves roots. None for non-rational
83    // or zero input.
84    let dense = dense_from_polynomial(value)?;
85    if dense.is_empty() {
86        return None;
87    }
88    let lower_rational = number_to_rational(lower);
89    let upper_rational = number_to_rational(upper);
90    if lower_rational >= upper_rational {
91        return Some(0);
92    }
93    let squarefree = squarefree_part(&dense);
94    let chain = sturm_chain(&squarefree);
95    let lower_variations = sturm_variations_at(&chain, &lower_rational);
96    let upper_variations = sturm_variations_at(&chain, &upper_rational);
97    Some(lower_variations - upper_variations)
98}

100/// Performs the advanced native `refine_isolating_interval` operation.
101pub fn refine_isolating_interval(
102    value: &Polynomial,
103    interval: &IsolatingInterval,
104    max_width: &exact_rational,
105) -> IsolatingInterval {
106    // Bisect an isolating interval until its width is at most max_width or an
107    // exact rational root is found. Bisection is guided by the sign of the
108    // squarefree part at the midpoint, which equals the one-root Sturm count of
109    // the bracket. A non-rational value is returned unchanged.
110    let dense = match dense_from_polynomial(value) {
111        Some(dense) if !dense.is_empty() => dense,
112        _ => return interval.clone(),
113    };
114    let squarefree = squarefree_part(&dense);
115    refine_dense(&squarefree, interval.clone(), max_width)
116}

118/// Performs the advanced native `sign_at_real_root` operation.
119pub fn sign_at_real_root(
120    defining: &Polynomial,
121    interval: &IsolatingInterval,
122    test: &Polynomial,
123) -> Option<i32> {
124    // Sign of test evaluated at the unique real root alpha of the squarefree
125    // defining polynomial that is isolated by interval. Returns Some(-1), Some(0)
126    // or Some(1); None when either operand is not a rational univariate
127    // polynomial. The root is shared by test (so test(alpha) = 0) exactly when
128    // gcd(defining, test) has its root inside the interval; otherwise the
129    // interval is refined until test keeps a constant nonzero sign across it.
130    let defining_dense = dense_from_polynomial(defining)?;
131    let test_dense = dense_from_polynomial(test)?;
132    if defining_dense.is_empty() {
133        return None;
134    }
135    if test_dense.is_empty() {
136        return Some(0);
137    }
138    Some(sign_at_real_root_dense(
139        &defining_dense,
140        interval.clone(),
141        &test_dense,
142    ))
143}

156/// Performs the advanced native `solve_sign_partition` operation.
157pub fn solve_sign_partition(value: &Polynomial) -> Option<SignPartition> {
158    // Partition the real line by the distinct real roots of value and record the
159    // sign of value on each open region between and around them. The region signs
160    // are sampled by exact evaluation of value (not the squarefree part) at a
161    // rational interior point, so an even-multiplicity root — across which the
162    // sign does not change — is handled correctly without any sign-flip rule.
163    // The squarefree field carries the squarefree part as a polynomial so the
164    // caller can name irrational roots by their real-root index. Returns None for
165    // non-rational or zero input.
166    let operation = OperationContext::new(0);
167    solve_sign_partition_with_operation(value, &operation)
168}
169
170pub(crate) fn solve_sign_partition_with_operation(
171    value: &Polynomial,
172    operation: &OperationContext,
173) -> Option<SignPartition> {
174    // The sign partition computed against a caller-supplied operation, so the isolation it opens
175    // with is scheduled by the same execution decision as the call that asked for it. Only the
176    // isolation is scheduled: the region signs are read off the isolated roots in real order and
177    // stay on the coordinator, so the partition a caller receives does not depend on the route.
178    let dense = dense_from_polynomial(value)?;
179    if dense.is_empty() {
180        return None;
181    }
182    let roots = isolate_real_roots_dense_with_operation(&dense, Some(operation));
183    let squarefree_dense = squarefree_part(&dense);
184    let squarefree = dense_to_polynomial(value.main_variable, &squarefree_dense);
185    let count = roots.len();
186    let mut region_signs: Vec<i32> = Vec::with_capacity(count + 1);
187    if count == 0 {
188        region_signs.push(rational_sign(&evaluate(&dense, &rational_zero())));
189        return Some(SignPartition {
190            roots,
191            region_signs,
192            squarefree,
193        });
194    }
195    let below = &roots[0].lower - rational_one();
196    region_signs.push(rational_sign(&evaluate(&dense, &below)));
197    for index in 0..count - 1 {
198        let separator = separating_point(&squarefree_dense, &roots[index], &roots[index + 1]);
199        region_signs.push(rational_sign(&evaluate(&dense, &separator)));
200    }
201    let above = &roots[count - 1].upper + rational_one();
202    region_signs.push(rational_sign(&evaluate(&dense, &above)));
203    Some(SignPartition {
204        roots,
205        region_signs,
206        squarefree,
207    })
208}

210fn separating_point(
211    squarefree: &[exact_rational],
212    left: &IsolatingInterval,
213    right: &IsolatingInterval,
214) -> exact_rational {
215    // A rational strictly between the roots isolated by two adjacent intervals,
216    // obtained by refining whichever interval is wider until the brackets are
217    // disjoint and then taking the midpoint of the gap.
218    let mut left_interval = left.clone();
219    let mut right_interval = right.clone();
220    while left_interval.upper >= right_interval.lower {
221        let left_width = &left_interval.upper - &left_interval.lower;
222        let right_width = &right_interval.upper - &right_interval.lower;
223        let refine_right =
224            !right_interval.is_exact() && (left_interval.is_exact() || right_width >= left_width);
225        if refine_right {
226            let half = right_width / exact_rational::from(2);
227            right_interval = refine_dense(squarefree, right_interval, &half);
228        } else {
229            let half = left_width / exact_rational::from(2);
230            left_interval = refine_dense(squarefree, left_interval, &half);
231        }
232    }
233    (&left_interval.upper + &right_interval.lower) / exact_rational::from(2)
234}

256fn isolation_route(
257    operation: Option<&OperationContext>,
258    reduced: &[exact_rational],
259    reflected: &[exact_rational],
260) -> ExecutionRoute {
261    // The two halves of the isolation are independent, but a half whose Descartes bound is at
262    // most one root subdivides nothing: it brackets its single root and returns. Counting only
263    // the halves that can recurse is what separates a polynomial whose roots are spread over
264    // both halves, where the split pays, from a binomial of any degree, whose halves hold one
265    // root each and whose split costs more than it saves.
266    let Some(operation) = operation else {
267        return ExecutionRoute::Serial;
268    };
269    if !operation.may_fork_work_class("real_root_isolation") {
270        return ExecutionRoute::Serial;
271    }
272    let degree = degree_of(reduced) as u64;
273    let independent_unit_count = [reduced, reflected]
274        .into_iter()
275        .filter(|half| sign_variations(half) >= 2)
276        .count();
277    let predicted_task_local_bytes = reduced.len().saturating_mul(128);
278    operation
279        .execution_decision(
280            &WorkEstimate::new(
281                degree.saturating_mul(degree.max(1)),
282                independent_unit_count,
283                predicted_task_local_bytes,
284                0,
285                independent_unit_count,
286                "real_root_isolation",
287            ),
288            independent_unit_count,
289            crate::parallelism::runtime_memory_capacity(predicted_task_local_bytes),
290        )
291        .route
292}

294fn isolate_real_roots_dense_with_operation(
295    dense: &[exact_rational],
296    operation: Option<&OperationContext>,
297) -> Vec<IsolatingInterval> {
298    // Core isolation over the dense coefficient view. Splits off the zero and
299    // rational roots as exact points, isolates the remaining (irrational) real
300    // roots with the VAS continued-fraction recursion, then cross-checks the
301    // total against the Sturm count and falls back to Sturm bisection on the
302    // (defended-against) impossible mismatch.
303    let squarefree = squarefree_part(dense);
304    if degree_of(&squarefree) == 0 {
305        return Vec::new();
306    }
307    let sturm_total = sturm_count_total(&squarefree);
308    let mut roots: Vec<IsolatingInterval> = Vec::new();
309    let working = squarefree.clone();
310    // Every exact rational root the irrational VAS brackets must be refined away from to
311    // keep the isolating set mutually disjoint — the zero root (split off here) as well as
312    // the non-zero rational roots `extract_rational_roots` pulls out below. Omitting the
313    // zero root let an irrational bracket straddling 0 overlap it and hang separating_point.
314    let mut exact_rational_roots: Vec<exact_rational> = Vec::new();
315    let rational_extraction = crate::polynomial::rational_roots::extract_rational_roots(&working);
316    let (rational_roots, reduced) = match rational_extraction {
317        Some(extraction) => (extraction.roots, extraction.quotient),
318        None => (Vec::new(), working.clone()),
319    };
320    for root in rational_roots.iter() {
321        roots.push(exact_interval(root.clone()));
322        exact_rational_roots.push(root.clone());
323    }
324    if degree_of(&reduced) >= 1 {
325        let mut irrational_intervals: Vec<IsolatingInterval> = Vec::new();
326        let reflected = reflect_variable(&reduced);
327        let route = isolation_route(operation, &reduced, &reflected);
328        let (positive_intervals, negative_intervals) = join(
329            route,
330            || isolate_positive_roots(&reduced),
331            || isolate_positive_roots(&reflected),
332        );
333        for interval in positive_intervals {
334            irrational_intervals.push(interval);
335        }
336        for interval in negative_intervals {
337            irrational_intervals.push(IsolatingInterval {
338                lower: -interval.upper,
339                upper: -interval.lower,
340            });
341        }
342        // The VAS brackets isolate the roots of `reduced` (all irrational), but a loose
343        // bracket can still contain an already-extracted rational root, leaving the
344        // isolating set not mutually disjoint — which then makes `separating_point` (which
345        // refines against the multi-root squarefree) fail to converge. Refine each bracket
346        // against the rational-root-free `reduced` until it excludes every rational root,
347        // restoring disjointness.
348        for interval in irrational_intervals {
349            roots.push(refine_interval_off_rationals(
350                &reduced,
351                interval,
352                &exact_rational_roots,
353            ));
354        }
355    }
356    roots.sort_by(compare_intervals);
357    if roots.len() != sturm_total {
358        return isolate_by_sturm(&squarefree);
359    }
360    roots
361}

363fn refine_interval_off_rationals(
364    reduced: &[exact_rational],
365    interval: IsolatingInterval,
366    rationals: &[exact_rational],
367) -> IsolatingInterval {
368    // Refine an irrational-root isolating interval (against the rational-root-free
369    // `reduced` polynomial, so it keeps isolating its single root) until it contains none
370    // of the extracted rational roots in its closed bracket. The contained root of
371    // `reduced` is irrational, hence distinct from every rational, so halving the width
372    // strictly excludes each rational point in finitely many steps; an exact bracket (an
373    // impossibility for an irrational root) is a defensive stopping condition.
374    let mut working = interval;
375    loop {
376        let contains_rational = rationals
377            .iter()
378            .any(|rational| working.lower <= *rational && *rational <= working.upper);
379        if !contains_rational || working.is_exact() {
380            return working;
381        }
382        let half = (&working.upper - &working.lower) / exact_rational::from(2);
383        working = refine_dense(reduced, working, &half);
384    }
385}

387struct PositiveRootRegion {
388    path: Vec<usize>,
389    polynomial: Vec<exact_rational>,
390    transform: Mobius,
391}

393fn isolate_positive_roots(dense: &[exact_rational]) -> Vec<IsolatingInterval> {
394    // Vincent-Akritas-Strzebonski continued-fraction isolation of the positive
395    // real roots of a squarefree polynomial. Each stack entry pairs a working
396    // polynomial with the Mobius transformation M(x) = (a x + b) / (c x + d) that
397    // maps the positive axis onto the original interval under study. Descartes'
398    // sign-variation count drives the recursion: zero variations means no root,
399    // one variation isolates exactly one root (Vincent's theorem), and two or
400    // more variations are subdivided by a lower-bound shift followed by the split
401    // of the positive axis at one through the substitutions x -> x + 1 (roots
402    // above one) and x -> 1/(x + 1) (roots below one).
403    let mut results: Vec<IsolatingInterval> = Vec::new();
404    let mut stack = vec![PositiveRootRegion {
405        path: Vec::new(),
406        polynomial: dense.to_vec(),
407        transform: Mobius::identity(),
408    }];
409    while let Some(region) = stack.pop() {
410        let mut polynomial = region.polynomial;
411        let mut transform = region.transform;
412        let path = region.path;
413        normalize(&mut polynomial);
414        while !polynomial.is_empty() && rational_is_zero(&polynomial[0]) {
415            results.push(exact_interval(transform.image_of_zero()));
416            polynomial = polynomial.split_first().unwrap().1.to_vec();
417            normalize(&mut polynomial);
418        }
419        if polynomial.is_empty() {
420            continue;
421        }
422        let variations = sign_variations(&polynomial);
423        if variations == 0 {
424            continue;
425        }
426        if variations == 1 {
427            results.push(transform.image_interval(&positive_upper_bound(&polynomial)));
428            continue;
429        }
430        let shift = positive_lower_bound_floor(&polynomial);
431        if shift > integer_zero() {
432            let shift_rational = exact_rational::from(shift);
433            polynomial = taylor_shift(&polynomial, &shift_rational);
434            transform = transform.compose_shift(&shift_rational);
435            normalize(&mut polynomial);
436            while !polynomial.is_empty() && rational_is_zero(&polynomial[0]) {
437                results.push(exact_interval(transform.image_of_zero()));
438                polynomial = polynomial.split_first().unwrap().1.to_vec();
439                normalize(&mut polynomial);
440            }
441            if polynomial.is_empty() {
442                continue;
443            }
444        }
445        if rational_is_zero(&evaluate(&polynomial, &rational_one())) {
446            results.push(exact_interval(transform.image_of_one()));
447            polynomial = divide_by_linear(&polynomial, &rational_one());
448            normalize(&mut polynomial);
449            if polynomial.is_empty() {
450                continue;
451            }
452        }
453        let one = rational_one();
454        let above_one = taylor_shift(&polynomial, &one);
455        let above_transform = transform.compose_shift(&one);
456        let mut below_one = polynomial.clone();
457        below_one.reverse();
458        let below_one = taylor_shift(&below_one, &one);
459        let below_transform = transform.compose_reciprocal_shift();
460        let mut above_path = path.clone();
461        above_path.push(0);
462        let mut below_path = path;
463        below_path.push(1);
464        stack.push(PositiveRootRegion {
465            path: above_path,
466            polynomial: above_one,
467            transform: above_transform,
468        });
469        stack.push(PositiveRootRegion {
470            path: below_path,
471            polynomial: below_one,
472            transform: below_transform,
473        });
474    }
475    results
476}

478#[derive(Debug, Clone)]
479struct Mobius {
480    a: exact_rational,
481    b: exact_rational,
482    c: exact_rational,
483    d: exact_rational,
484}
485
486impl Mobius {
487    fn identity() -> Mobius {
488        Mobius {
489            a: rational_one(),
490            b: rational_zero(),
491            c: rational_zero(),
492            d: rational_one(),
493        }
494    }
495
496    fn image_of_zero(&self) -> exact_rational {
497        // M(0) = b / d.
498        &self.b / &self.d
499    }
500
501    fn image_of_one(&self) -> exact_rational {
502        // M(1) = (a + b) / (c + d).
503        (&self.a + &self.b) / (&self.c + &self.d)
504    }
505
506    fn compose_shift(&self, shift: &exact_rational) -> Mobius {
507        // Compose with x -> x + shift, i.e. the new transform is M(x + shift).
508        Mobius {
509            a: self.a.clone(),
510            b: &self.a * shift + &self.b,
511            c: self.c.clone(),
512            d: &self.c * shift + &self.d,
513        }
514    }
515
516    fn compose_reciprocal_shift(&self) -> Mobius {
517        // Compose with x -> 1 / (x + 1), the substitution that maps the positive
518        // axis onto the sub-interval of roots below one. Algebraically
519        // M(1/(x+1)) = (b x + (a + b)) / (d x + (c + d)).
520        Mobius {
521            a: self.b.clone(),
522            b: &self.a + &self.b,
523            c: self.d.clone(),
524            d: &self.c + &self.d,
525        }
526    }
527
528    fn image_interval(&self, working_upper_bound: &exact_rational) -> IsolatingInterval {
529        // The original-coordinate interval that the single isolated positive root
530        // occupies. When the transform sends the point at infinity to a finite
531        // value (c != 0) both endpoints are M(0) and M(infinity); otherwise the
532        // upper endpoint is taken from M applied to a finite upper bound on the
533        // current polynomial's positive roots.
534        let lower_image = self.image_of_zero();
535        let upper_image = if rational_is_zero(&self.c) {
536            (&self.a * working_upper_bound + &self.b) / &self.d
537        } else {
538            &self.a / &self.c
539        };
540        if lower_image <= upper_image {
541            IsolatingInterval {
542                lower: lower_image,
543                upper: upper_image,
544            }
545        } else {
546            IsolatingInterval {
547                lower: upper_image,
548                upper: lower_image,
549            }
550        }
551    }
552}

554fn isolate_by_sturm(squarefree: &[exact_rational]) -> Vec<IsolatingInterval> {
555    // Deterministic Sturm-count-guided bisection isolator, used only as the
556    // correctness fallback when the VAS interval count disagrees with the Sturm
557    // total. The working bounds are kept off any root by starting from the strict
558    // two-sided Cauchy bound; counts use the canonical half-open (a, b]
559    // convention V(a) - V(b).
560    let chain = sturm_chain(squarefree);
561    let total = sturm_count_total(squarefree);
562    if total == 0 {
563        return Vec::new();
564    }
565    let bound = two_sided_bound(squarefree);
566    let mut results: Vec<IsolatingInterval> = Vec::new();
567    let lower = -bound.clone();
568    let upper = bound;
569    let mut stack: Vec<(exact_rational, exact_rational, usize)> = Vec::new();
570    let count = sturm_variations_at(&chain, &lower) - sturm_variations_at(&chain, &upper);
571    stack.push((lower, upper, count));
572    while let Some((lower, upper, count)) = stack.pop() {
573        if count == 0 {
574            continue;
575        }
576        if count == 1 {
577            if rational_is_zero(&evaluate(squarefree, &upper)) {
578                results.push(exact_interval(upper));
579            } else {
580                results.push(IsolatingInterval { lower, upper });
581            }
582            continue;
583        }
584        let midpoint = (&lower + &upper) / exact_rational::from(2);
585        let lower_count =
586            sturm_variations_at(&chain, &lower) - sturm_variations_at(&chain, &midpoint);
587        let upper_count =
588            sturm_variations_at(&chain, &midpoint) - sturm_variations_at(&chain, &upper);
589        stack.push((lower, midpoint.clone(), lower_count));
590        stack.push((midpoint, upper, upper_count));
591    }
592    results.sort_by(compare_intervals);
593    results
594}

596fn refine_dense(
597    squarefree: &[exact_rational],
598    mut interval: IsolatingInterval,
599    max_width: &exact_rational,
600) -> IsolatingInterval {
601    // Sign-guided bisection of a one-root bracket of the squarefree polynomial.
602    if interval.is_exact() {
603        return interval;
604    }
605    let lower_sign = rational_sign(&evaluate(squarefree, &interval.lower));
606    while &interval.upper - &interval.lower > *max_width {
607        let midpoint = (&interval.lower + &interval.upper) / exact_rational::from(2);
608        let midpoint_sign = rational_sign(&evaluate(squarefree, &midpoint));
609        if midpoint_sign == 0 {
610            return exact_interval(midpoint);
611        }
612        if midpoint_sign == lower_sign {
613            interval.lower = midpoint;
614        } else {
615            interval.upper = midpoint;
616        }
617    }
618    interval
619}
620
621fn sign_at_real_root_dense(
622    defining: &[exact_rational],
623    mut interval: IsolatingInterval,
624    test: &[exact_rational],
625) -> i32 {
626    // Sign of test at the unique root of the squarefree defining polynomial in
627    // interval. See sign_at_real_root for the method.
628    if interval.is_exact() {
629        return rational_sign(&evaluate(test, &interval.lower));
630    }
631    let common = monic_gcd(defining, test);
632    if degree_of(&common) >= 1 {
633        let common_chain = sturm_chain(&common);
634        let roots_in_interval = sturm_variations_at(&common_chain, &interval.lower)
635            - sturm_variations_at(&common_chain, &interval.upper);
636        if roots_in_interval >= 1 {
637            return 0;
638        }
639    }
640    let defining_squarefree = squarefree_part(defining);
641    loop {
642        let lower_sign = rational_sign(&evaluate(test, &interval.lower));
643        let upper_sign = rational_sign(&evaluate(test, &interval.upper));
644        if lower_sign != 0 && lower_sign == upper_sign {
645            return lower_sign;
646        }
647        let width = &interval.upper - &interval.lower;
648        let half = width / exact_rational::from(2);
649        interval = refine_dense(&defining_squarefree, interval, &half);
650        if interval.is_exact() {
651            return rational_sign(&evaluate(test, &interval.lower));
652        }
653    }
654}
655
656fn squarefree_part(dense: &[exact_rational]) -> Vec<exact_rational> {
657    // The squarefree part p / gcd(p, p'), returned monic. Every root of the
658    // result is simple, which is what makes both the Descartes sign test and the
659    // Sturm chain exact root indicators.
660    let mut polynomial = dense.to_vec();
661    normalize(&mut polynomial);
662    if degree_of(&polynomial) == 0 {
663        make_monic(&mut polynomial);
664        return polynomial;
665    }
666    let derivative_polynomial = derivative(&polynomial);
667    let common = monic_gcd(&polynomial, &derivative_polynomial);
668    if degree_of(&common) == 0 {
669        make_monic(&mut polynomial);
670        return polynomial;
671    }
672    let (mut quotient, _) = divide(&polynomial, &common);
673    make_monic(&mut quotient);
674    quotient
675}
676
677fn sturm_chain(squarefree: &[exact_rational]) -> Vec<Vec<exact_rational>> {
678    // The canonical Sturm chain p0 = p, p1 = p', p_{k+1} = -rem(p_{k-1}, p_k).
679    let mut chain: Vec<Vec<exact_rational>> = vec![squarefree.to_vec()];
680    let derivative_polynomial = derivative(squarefree);
681    if derivative_polynomial.is_empty() {
682        return chain;
683    }
684    chain.push(derivative_polynomial);
685    loop {
686        let length = chain.len();
687        let (_, remainder) = divide(&chain[length - 2], &chain[length - 1]);
688        if remainder.is_empty() {
689            break;
690        }
691        chain.push(negate(&remainder));
692    }
693    chain
694}
695
696fn sturm_count_total(squarefree: &[exact_rational]) -> usize {
697    // Number of distinct real roots = V(-infinity) - V(+infinity).
698    if degree_of(squarefree) == 0 {
699        return 0;
700    }
701    let chain = sturm_chain(squarefree);
702    sturm_variations_at_infinity(&chain, false) - sturm_variations_at_infinity(&chain, true)
703}
704
705fn sturm_variations_at(chain: &[Vec<exact_rational>], point: &exact_rational) -> usize {
706    // Count sign changes in the Sturm chain evaluated at a finite point.
707    let signs: Vec<i32> = chain
708        .iter()
709        .map(|polynomial| rational_sign(&evaluate(polynomial, point)))
710        .collect();
711    count_sign_changes(&signs)
712}
713
714fn sturm_variations_at_infinity(chain: &[Vec<exact_rational>], positive: bool) -> usize {
715    // Count sign changes of the chain at plus or minus infinity, where the sign
716    // of each member is that of its leading coefficient, with an extra factor of
717    // (-1)^degree at minus infinity.
718    let signs: Vec<i32> = chain
719        .iter()
720        .map(|polynomial| leading_sign_at_infinity(polynomial, positive))
721        .collect();
722    count_sign_changes(&signs)
723}
724
725fn leading_sign_at_infinity(polynomial: &[exact_rational], positive: bool) -> i32 {
726    // Sign of the polynomial as the argument tends to plus or minus infinity.
727    if polynomial.is_empty() {
728        return 0;
729    }
730    let leading = rational_sign(polynomial.last().unwrap());
731    if positive {
732        return leading;
733    }
734    let degree = polynomial.len() - 1;
735    if degree % 2 == 0 { leading } else { -leading }
736}
737
738fn count_sign_changes(signs: &[i32]) -> usize {
739    // Number of sign changes in a sequence, ignoring zero entries.
740    let mut changes = 0;
741    let mut previous: i32 = 0;
742    for &sign in signs.iter() {
743        if sign == 0 {
744            continue;
745        }
746        if previous != 0 && sign != previous {
747            changes += 1;
748        }
749        previous = sign;
750    }
751    changes
752}
753
754fn sign_variations(dense: &[exact_rational]) -> usize {
755    // Descartes sign-variation count of the coefficient sequence.
756    let signs: Vec<i32> = dense.iter().map(rational_sign).collect();
757    count_sign_changes(&signs)
758}

760fn positive_upper_bound(dense: &[exact_rational]) -> exact_rational {
761    // A strict Cauchy upper bound on the positive real roots:
762    // 1 + max_i |a_i / a_n|.
763    let degree = degree_of(dense);
764    if degree == 0 {
765        return rational_one();
766    }
767    let leading = rational_absolute_value(&dense[degree]);
768    let mut maximum = rational_zero();
769    for coefficient in dense.iter().take(degree) {
770        let ratio = rational_absolute_value(coefficient) / &leading;
771        if ratio > maximum {
772            maximum = ratio;
773        }
774    }
775    maximum + rational_one()
776}
777
778fn two_sided_bound(dense: &[exact_rational]) -> exact_rational {
779    // A strict bound B with every real root in (-B, B); shares the Cauchy form
780    // with positive_upper_bound.
781    positive_upper_bound(dense)
782}
783
784fn positive_lower_bound_floor(dense: &[exact_rational]) -> exact_integer {
785    // Floor of a positive lower bound on the smallest positive root, used as the
786    // continued-fraction shift accelerator. The smallest positive root of p is
787    // the reciprocal of the largest positive root of the reversed polynomial, so
788    // a lower bound is 1 / (Cauchy upper bound of the reverse).
789    let mut reversed = dense.to_vec();
790    reversed.reverse();
791    normalize(&mut reversed);
792    if reversed.is_empty() {
793        return integer_zero();
794    }
795    let upper = positive_upper_bound(&reversed);
796    if rational_is_zero(&upper) {
797        return integer_zero();
798    }
799    let lower = rational_one() / upper;
800    rational_floor(&lower)
801}

803fn dense_from_polynomial(value: &Polynomial) -> Option<Vec<exact_rational>> {
804    // Project a univariate polynomial whose coefficients are all numeric onto a
805    // dense ascending-degree rational vector. None for any nested (multivariate)
806    // or algebraic coefficient.
807    let degree = match crate::polynomial::degree(value) {
808        Some(degree) => degree,
809        None => return Some(Vec::new()),
810    };
811    let mut dense = vec![rational_zero(); degree + 1];
812    for (exponent, coefficient) in value.terms.iter() {
813        match coefficient {
814            PolynomialCoefficient::Numeric(numeric_value) => {
815                dense[*exponent] = number_to_rational(numeric_value);
816            }
817            _ => return None,
818        }
819    }
820    normalize(&mut dense);
821    Some(dense)
822}

824fn reflect_variable(dense: &[exact_rational]) -> Vec<exact_rational> {
825    // The polynomial p(-x): negate every odd-degree coefficient.
826    dense
827        .iter()
828        .enumerate()
829        .map(|(exponent, coefficient)| {
830            if exponent % 2 == 0 {
831                coefficient.clone()
832            } else {
833                -coefficient.clone()
834            }
835        })
836        .collect()
837}
838
839fn derivative(dense: &[exact_rational]) -> Vec<exact_rational> {
840    // The formal derivative in the dense representation.
841    if dense.len() <= 1 {
842        return Vec::new();
843    }
844    let mut result: Vec<exact_rational> = Vec::with_capacity(dense.len() - 1);
845    for (exponent, coefficient) in dense.iter().enumerate().skip(1) {
846        result.push(coefficient * exact_rational::from(exact_integer::from(exponent as i64)));
847    }
848    normalize(&mut result);
849    result
850}
851
852fn evaluate(dense: &[exact_rational], point: &exact_rational) -> exact_rational {
853    // Horner evaluation at a rational point.
854    let mut accumulator = rational_zero();
855    for coefficient in dense.iter().rev() {
856        accumulator = accumulator * point + coefficient;
857    }
858    accumulator
859}

861fn taylor_shift(dense: &[exact_rational], shift: &exact_rational) -> Vec<exact_rational> {
862    // Coefficients of p(x + shift), computed by repeated synthetic division by
863    // (x - shift): each remainder is the next Taylor coefficient in ascending
864    // order.
865    if dense.is_empty() {
866        return Vec::new();
867    }
868    let mut descending: Vec<exact_rational> = dense.to_vec();
869    descending.reverse();
870    let mut shifted: Vec<exact_rational> = Vec::with_capacity(dense.len());
871    let mut current = descending;
872    while !current.is_empty() {
873        for index in 1..current.len() {
874            let increment = &current[index - 1] * shift;
875            current[index] = &current[index] + &increment;
876        }
877        let remainder = current.pop().unwrap();
878        shifted.push(remainder);
879    }
880    normalize(&mut shifted);
881    shifted
882}

884fn divide_by_linear(dense: &[exact_rational], root: &exact_rational) -> Vec<exact_rational> {
885    // Exact division by (x - root) via synthetic division; the caller guarantees
886    // root is an actual root so the remainder vanishes.
887    let degree = degree_of(dense);
888    if degree == 0 {
889        return Vec::new();
890    }
891    let mut quotient = vec![rational_zero(); degree];
892    let mut carry = dense[degree].clone();
893    quotient[degree - 1] = carry.clone();
894    for index in (1..degree).rev() {
895        carry = &dense[index] + &carry * root;
896        quotient[index - 1] = carry.clone();
897    }
898    normalize(&mut quotient);
899    quotient
900}

902fn divide(
903    dividend: &[exact_rational],
904    divisor: &[exact_rational],
905) -> (Vec<exact_rational>, Vec<exact_rational>) {
906    // Euclidean division over the rationals, returning (quotient, remainder).
907    let divisor_degree = degree_of(divisor);
908    let mut remainder = dividend.to_vec();
909    normalize(&mut remainder);
910    if remainder.len() < divisor.len() {
911        return (Vec::new(), remainder);
912    }
913    let divisor_leading = divisor[divisor_degree].clone();
914    let mut quotient = vec![rational_zero(); remainder.len() - divisor_degree];
915    while !remainder.is_empty() && degree_of(&remainder) >= divisor_degree {
916        let remainder_degree = degree_of(&remainder);
917        let factor = &remainder[remainder_degree] / &divisor_leading;
918        let shift = remainder_degree - divisor_degree;
919        quotient[shift] = factor.clone();
920        for index in 0..divisor.len() {
921            let term = &factor * &divisor[index];
922            remainder[shift + index] = &remainder[shift + index] - &term;
923        }
924        normalize(&mut remainder);
925    }
926    normalize(&mut quotient);
927    (quotient, remainder)
928}

930fn monic_gcd(left: &[exact_rational], right: &[exact_rational]) -> Vec<exact_rational> {
931    // Monic greatest common divisor over the rationals by the Euclidean
932    // algorithm.
933    let mut first = left.to_vec();
934    normalize(&mut first);
935    let mut second = right.to_vec();
936    normalize(&mut second);
937    while !second.is_empty() {
938        let (_, remainder) = divide(&first, &second);
939        first = second;
940        second = remainder;
941    }
942    make_monic(&mut first);
943    first
944}

946fn negate(dense: &[exact_rational]) -> Vec<exact_rational> {
947    // Coefficient-wise negation.
948    dense
949        .iter()
950        .map(|coefficient| -coefficient.clone())
951        .collect()
952}
953
954fn make_monic(dense: &mut Vec<exact_rational>) {
955    // Scale so the leading coefficient is one; a no-op on the zero polynomial.
956    normalize(dense);
957    if dense.is_empty() {
958        return;
959    }
960    let leading = dense.last().unwrap().clone();
961    if leading == rational_one() {
962        return;
963    }
964    for coefficient in dense.iter_mut() {
965        *coefficient /= &leading;
966    }
967}

969fn normalize(dense: &mut Vec<exact_rational>) {
970    // Drop trailing (high-degree) zero coefficients so the leading coefficient is
971    // always nonzero and the empty vector is the unique zero polynomial.
972    while let Some(last) = dense.last() {
973        if rational_is_zero(last) {
974            dense.pop();
975        } else {
976            break;
977        }
978    }
979}

981fn degree_of(dense: &[exact_rational]) -> usize {
982    // Degree of a normalized dense polynomial; the zero and constant polynomials
983    // both report zero.
984    if dense.is_empty() { 0 } else { dense.len() - 1 }
985}

987fn compare_intervals(left: &IsolatingInterval, right: &IsolatingInterval) -> Ordering {
988    // Order isolating intervals by ascending position.
989    left.lower
990        .cmp(&right.lower)
991        .then_with(|| left.upper.cmp(&right.upper))
992}

995    // A collapsed interval marking an exact rational root.
996    IsolatingInterval {
997        lower: value.clone(),
998        upper: value,
999    }

1003    // True when a rational is exactly zero.

1005}

1008    // The sign of a rational as -1, 0, or 1.
