8#[derive(Clone, Debug, Eq, Ord, PartialEq, PartialOrd)]
9pub(crate) struct ModularTaskIdentifier {
10    pub(crate) operation_family: &'static str,
11    pub(crate) batch_number: usize,
12    pub(crate) logical_image_index: usize,

16#[derive(Debug, Eq, PartialEq)]
17pub(crate) enum ModularImageOutcome<Signature, Payload> {

19        identifier: ModularTaskIdentifier,
20        signature: Signature,
21        payload: Payload,
22    },
23    Unlucky {
24        identifier: ModularTaskIdentifier,

49pub(crate) fn canonical_task_batches(
50    operation_family: &'static str,
51    prime_or_evaluation_identifiers: &[u64],
52    batch_width: usize,
53) -> Vec<ModularTaskBatch> {
54    let effective_batch_width = batch_width.max(1);
55    prime_or_evaluation_identifiers
56        .chunks(effective_batch_width)
57        .enumerate()
58        .map(|(batch_number, identifiers)| ModularTaskBatch {
59            batch_number,
60            tasks: identifiers
61                .iter()
62                .enumerate()
63                .map(|(offset, identifier)| ModularTaskIdentifier {
64                    operation_family,
65                    batch_number,
66                    logical_image_index: batch_number * effective_batch_width + offset,
67                    prime_or_evaluation_identifier: *identifier,
68                })
69                .collect(),
70        })
71        .collect()
72}

127pub(crate) fn run_modular_batches<
128    Signature,
129    Payload,
130    State,
131    Candidate,
132    ImageFunction,
133    StateFactory,
134    MergeFunction,
135    ReconstructFunction,
136    CertifyFunction,
137>(
138    route: ExecutionRoute,
139    batches: &[ModularTaskBatch],
140    required_evidence: usize,
141    cancellation: &CancellationToken,
142    memory_capacity: Option<usize>,
143    predicted_task_local_bytes: usize,
144    image_function: ImageFunction,
145    state_factory: StateFactory,
146    merge_function: MergeFunction,
147    reconstruct_function: ReconstructFunction,
148    certify_function: CertifyFunction,
149) -> ModularRunReport<Signature, Candidate>
150where
151    Signature: Clone + Ord + Send,
152    Payload: Send,
153    State: Send,
154    Candidate: Send,
155    ImageFunction:
156        Fn(&ModularTaskIdentifier) -> ModularImageOutcome<Signature, Payload> + Send + Sync,
157    StateFactory: Fn() -> State,
158    MergeFunction: Fn(&mut State, &ModularTaskIdentifier, Payload) -> bool,
159    ReconstructFunction: Fn(&mut State, usize) -> Option<Candidate>,
160    CertifyFunction: Fn(&Candidate) -> bool,
161{
162    // Drive the coordinator over a batch list that is already fully formed. The
163    // canonical batch order is the slice order, so the pulled source below reads the
164    // slice one batch at a time and stops when it is exhausted.
165    let mut next_batch_position = 0usize;
166    run_modular_batches_from_source(
167        route,
168        || {
169            let batch = batches.get(next_batch_position).cloned();
170            next_batch_position += 1;
171            batch
172        },
173        required_evidence,
174        cancellation,
175        memory_capacity,
176        predicted_task_local_bytes,
177        image_function,
178        state_factory,
179        merge_function,
180        reconstruct_function,
181        certify_function,
182    )
183}

185pub(crate) fn run_modular_batches_from_source<
186    Signature,
187    Payload,
188    State,
189    Candidate,
190    BatchSource,
191    ImageFunction,
192    StateFactory,
193    MergeFunction,
194    ReconstructFunction,
195    CertifyFunction,
196>(
197    route: ExecutionRoute,
198    mut batch_source: BatchSource,
199    required_evidence: usize,
200    cancellation: &CancellationToken,
201    memory_capacity: Option<usize>,
202    predicted_task_local_bytes: usize,
203    image_function: ImageFunction,
204    state_factory: StateFactory,
205    merge_function: MergeFunction,
206    reconstruct_function: ReconstructFunction,
207    certify_function: CertifyFunction,
208) -> ModularRunReport<Signature, Candidate>
209where
210    Signature: Clone + Ord + Send,
211    Payload: Send,
212    State: Send,
213    Candidate: Send,
214    BatchSource: FnMut() -> Option<ModularTaskBatch>,
215    ImageFunction:
216        Fn(&ModularTaskIdentifier) -> ModularImageOutcome<Signature, Payload> + Send + Sync,
217    StateFactory: Fn() -> State,
218    MergeFunction: Fn(&mut State, &ModularTaskIdentifier, Payload) -> bool,
219    ReconstructFunction: Fn(&mut State, usize) -> Option<Candidate>,
220    CertifyFunction: Fn(&Candidate) -> bool,
221{
222    // Form and consume one canonical batch at a time so a coordinator whose
223    // prime or evaluation stream is expensive to extend only pays for the images it
224    // actually runs. The source owns canonical stream generation; this loop owns
225    // grouping, merging, reconstruction, and certification.
226    let mut report = ModularRunReport {
227        candidate: None,
228        outcomes: Vec::new(),
229        used_identifiers: Vec::new(),
230        reconstructed_candidate_count: 0,
231    };
232    let mut selected_signature: Option<Signature> = None;
233    let mut state = state_factory();
234    let mut selected_image_count = 0usize;
235    let mut memory = memory_capacity.map(MemoryReservationCoordinator::new);
236    let required_evidence = required_evidence.max(1);
237
238    while let Some(batch) = batch_source() {
239        let batch = &batch;
240        let reservations = match memory.as_mut() {
241            Some(coordinator) => {
242                let predicted_bytes = vec![predicted_task_local_bytes; batch.tasks.len()];
243                match coordinator.reserve_batch(&predicted_bytes) {
244                    Some(reservations) => Some(reservations),
245                    None => {
246                        report.outcomes.extend(
247                            batch
248                                .tasks
249                                .iter()
250                                .map(|_| ModularImageOutcome::ResourceExhausted),
251                        );
252                        break;
253                    }
254                }
255            }
256            None => None,
257        };
258
259        let root_identifier = LogicalTaskIdentifier::root(batch.batch_number as u64);
260        let indexed_outcomes = ordered_try_map_with_identifiers(
261            route,
262            &batch.tasks,
263            &root_identifier,
264            cancellation,
265            |task, _| TaskOutcome::Value(image_function(task)),
266        );
267        let mut grouped_images =
268            BTreeMap::<Signature, Vec<(ModularTaskIdentifier, Payload)>>::new();
269        let mut cancellation_seen = false;
270        for (task_position, indexed_outcome) in indexed_outcomes.into_iter().enumerate() {
271            let task_identifier = batch.tasks[task_position].clone();
272            match indexed_outcome.outcome {
273                TaskOutcome::Value(outcome) => match outcome {
274                    ModularImageOutcome::Accepted {
275                        identifier,
276                        signature,
277                        payload,
278                    } => {
279                        report.outcomes.push(ModularImageOutcome::Accepted {
280                            identifier: identifier.clone(),
281                            signature: signature.clone(),
282                            payload: (),
283                        });
284                        grouped_images
285                            .entry(signature)
286                            .or_default()
287                            .push((identifier, payload));
288                    }
289                    ModularImageOutcome::Unlucky { identifier, reason } => {
290                        report
291                            .outcomes
292                            .push(ModularImageOutcome::Unlucky { identifier, reason });
293                    }
294                    ModularImageOutcome::Declined { identifier, reason } => {
295                        report
296                            .outcomes
297                            .push(ModularImageOutcome::Declined { identifier, reason });
298                    }
299                    ModularImageOutcome::Cancelled => {
300                        cancellation_seen = true;
301                        report.outcomes.push(ModularImageOutcome::Cancelled);
302                    }
303                    ModularImageOutcome::ResourceExhausted => {
304                        report.outcomes.push(ModularImageOutcome::ResourceExhausted);
305                    }
306                },
307                TaskOutcome::Cancelled => {
308                    cancellation_seen = true;
309                    report.outcomes.push(ModularImageOutcome::Cancelled);
310                }
311                TaskOutcome::Declined => {
312                    report.outcomes.push(ModularImageOutcome::Declined {
313                        identifier: task_identifier.clone(),
314                        reason: "the modular task declined",
315                    });
316                }
317                TaskOutcome::ResourceExhausted => {
318                    report.outcomes.push(ModularImageOutcome::ResourceExhausted);
319                }
320                TaskOutcome::Error(()) => {
321                    report.outcomes.push(ModularImageOutcome::Declined {
322                        identifier: task_identifier,
323                        reason: "the modular task failed",
324                    });
325                }
326            }
327        }
328
329        if let Some(coordinator) = memory.as_mut() {
330            if let Some(reservations) = reservations {
331                coordinator.release_batch(&reservations);
332            }
333        }
334
335        let mut groups = grouped_images.into_iter().collect::<Vec<_>>();
336        let evidence_for = |signature: &Signature, batch_count: usize| {
337            if selected_signature.as_ref() == Some(signature) {
338                batch_count + selected_image_count
339            } else {
340                batch_count
341            }
342        };
343        groups.sort_by(|left, right| {
344            evidence_for(&right.0, right.1.len())
345                .cmp(&evidence_for(&left.0, left.1.len()))
346                .then_with(|| left.0.cmp(&right.0))
347        });
348        let Some((batch_signature, mut images)) = groups.into_iter().next() else {
349            if cancellation_seen {
350                break;
351            }
352            continue;
353        };
354        images.sort_by(|left, right| left.0.cmp(&right.0));
355
356        if selected_signature.as_ref() != Some(&batch_signature) {
357            selected_signature = Some(batch_signature);
358            state = state_factory();
359            selected_image_count = 0;
360            report.used_identifiers.clear();
361        }
362        // Reconstruct after every merged image rather than once per batch. The prefix of
363        // the canonical image stream that first reconstructs and certifies is then the same
364        // whatever the batch width is, so the images a run consumes — and the primes,
365        // counters and diagnostics derived from them — are a function of the input alone and
366        // not of how many workers happened to be available. A wider batch still computes its
367        // images concurrently; it merely stops using them at the same canonical boundary a
368        // width-one run stops at.
369        let mut candidate_certified = false;
370        for (identifier, payload) in images {
371            if !merge_function(&mut state, &identifier, payload) {
372                selected_signature = None;
373                state = state_factory();
374                selected_image_count = 0;
375                report.used_identifiers.clear();
376                break;
377            }
378            selected_image_count += 1;
379            report.used_identifiers.push(identifier);
380            if selected_image_count < required_evidence {
381                continue;
382            }
383            if let Some(candidate) = reconstruct_function(&mut state, selected_image_count) {
384                report.reconstructed_candidate_count += 1;
385                if certify_function(&candidate) {
386                    report.candidate = Some(candidate);
387                    candidate_certified = true;
388                    break;
389                }
390            }
391        }
392
393        if candidate_certified || cancellation_seen {
394            break;
395        }
396    }
397    report
398}

484    #[test]
485    fn the_consumed_image_prefix_does_not_depend_on_the_batch_width() {
486        // The canonical stream is the same however it is grouped, so the prefix that first
487        // reconstructs and certifies must be the same too: a wider batch may compute more
488        // images concurrently, but it may not consume more of them.
489        let identifiers = [11u64, 13, 17, 19, 23, 29, 31, 37];
490        let run_at_width = |batch_width: usize| {
491            let batches = canonical_task_batches("test", &identifiers, batch_width);
492            let cancellation = CancellationToken::new();
493            run_serial_modular_batches(
494                &batches,
495                1,
496                &cancellation,
497                None,
498                0,
499                |identifier| accepted(identifier, 1),
500                Vec::<usize>::new,
501                |state, _, payload| {
502                    state.push(payload);
503                    true
504                },
505                |state, _| Some(state.clone()),
506                |candidate: &Vec<usize>| candidate.len() >= 3,
507            )
508        };
509
510        let reference = run_at_width(1);
511        assert_eq!(reference.candidate, Some(vec![11, 13, 17]));
512        for batch_width in [2, 3, 4, 8] {
513            let report = run_at_width(batch_width);
514            assert_eq!(report.candidate, reference.candidate);
515            assert_eq!(
516                report
517                    .used_identifiers
518                    .iter()
519                    .map(|identifier| identifier.prime_or_evaluation_identifier)
520                    .collect::<Vec<_>>(),
521                reference
522                    .used_identifiers
523                    .iter()
524                    .map(|identifier| identifier.prime_or_evaluation_identifier)
525                    .collect::<Vec<_>>(),
526                "batch width {batch_width} consumed a different image prefix"
527            );
528            assert_eq!(
529                report.reconstructed_candidate_count, reference.reconstructed_candidate_count,
530                "batch width {batch_width} attempted a different number of reconstructions"
531            );
532        }
533    }
