8/// Performs the advanced native `yun_squarefree_factorization` operation.
9pub fn yun_squarefree_factorization(value: &Polynomial) -> Vec<(usize, Polynomial)> {
10    // The entry for callers that hold no operation. Its default operation is sequential, so this
11    // decomposition runs its greatest common divisors on the serial backend; a caller inside an
12    // API operation reaches the entry below instead and its divisors follow that operation's
13    // execution mode and stop when it is cancelled.
14    yun_squarefree_factorization_with_operation(value, &OperationContext::new(0))
15}

17/// Decompose a polynomial into squarefree factors under the supplied operation.
18pub(crate) fn yun_squarefree_factorization_with_operation(
19    value: &Polynomial,
20    operation: &OperationContext,
21) -> Vec<(usize, Polynomial)> {
22    // Yun's algorithm for squarefree factorisation over a coefficient ring of
23    // characteristic zero (Q[x] in our case). Returns a list of (multiplicity,
24    // squarefree factor) pairs whose product, raising each factor to its
25    // multiplicity, equals the input up to a unit. Yun's algorithm is strictly
26    // preferred over Tobey-Horowitz because it performs fewer GCD computations
27    // and exhibits better coefficient growth on rational inputs.
28    if is_zero(value) || degree(value).unwrap_or(0) == 0 {
29        if is_zero(value) {
30            return Vec::new();
31        }
32        return vec![(1, value.clone())];
33    }
34    let derivative = formal_derivative(value);
35    let mut common = polynomial_gcd_with_operation(value, &derivative, operation);
36    let mut working = match divide_exact(value, &common) {
37        Some(quotient) => quotient,
38        None => return vec![(1, value.clone())],
39    };
40    let mut derivative_quotient = match divide_exact(&derivative, &common) {
41        Some(quotient) => quotient,
42        None => return vec![(1, value.clone())],
43    };
44    let mut factors: Vec<(usize, Polynomial)> = Vec::new();
45    let mut multiplicity: usize = 1;
46    loop {
47        let working_derivative = formal_derivative(&working);
48        let difference = super::arithmetic::subtract(&derivative_quotient, &working_derivative);
49        let next_factor = polynomial_gcd_with_operation(&working, &difference, operation);
50        let next_quotient = match divide_exact(&working, &next_factor) {
51            Some(quotient) => quotient,
52            None => break,
53        };
54        if degree(&next_factor).unwrap_or(0) > 0 {
55            factors.push((multiplicity, next_factor.clone()));
56        }
57        derivative_quotient = match divide_exact(&difference, &next_factor) {
58            Some(quotient) => quotient,
59            None => break,
60        };
61        working = next_quotient;
62        if degree(&working).unwrap_or(0) == 0 {
63            if degree(&working).is_some() && degree(&working).unwrap() == 0 {
64                let _ = working;
65            }
66            break;
67        }
68        multiplicity += 1;
69        let _ = &mut common;
70    }
71    if degree(&working).unwrap_or(0) > 0 {
72        factors.push((multiplicity, working));
73    }
74    factors
75}

96    #[test]
97    fn squarefree_polynomial_returns_single_multiplicity_one_factor() {
98        let mut context = Context::new();
99        let identifier = variable(&mut context, "x");
100        let value = univariate_from_dense(
101            identifier,
102            vec![integer_number(-1), integer_number(0), integer_number(1)],
103        );
104        let factors = yun_squarefree_factorization(&value);
105        assert_eq!(factors.len(), 1);
106        assert_eq!(factors[0].0, 1);
107    }
108
109    #[test]
110    fn squared_factor_is_detected_at_multiplicity_two() {
111        let mut context = Context::new();
112        let identifier = variable(&mut context, "x");
113        let factor = univariate_from_dense(identifier, vec![integer_number(-1), integer_number(1)]);
114        let factor_squared = multiply(&factor, &factor);
115        let factors = yun_squarefree_factorization(&factor_squared);
116        let total_multiplicity: usize = factors.iter().map(|(multiplicity, _)| *multiplicity).sum();
117        assert_eq!(total_multiplicity, 2);
118    }
119
120    #[test]
121    fn cubed_factor_is_detected_at_multiplicity_three() {
122        let mut context = Context::new();
123        let identifier = variable(&mut context, "x");
124        let factor = univariate_from_dense(identifier, vec![integer_number(-1), integer_number(1)]);
125        let cubed = multiply(&multiply(&factor, &factor), &factor);
126        let factors = yun_squarefree_factorization(&cubed);
127        let highest = factors
128            .iter()
129            .map(|(multiplicity, _)| *multiplicity)
130            .max()
131            .unwrap();
132        assert_eq!(highest, 3);
133    }
