27/// Public native data structure named `integer_matrix`.
28pub struct IntegerMatrix {
29    /// Stores the `rows` value.
30    pub rows: usize,
31    /// Stores the `columns` value.
32    pub columns: usize,
33    /// Stores the `entries` value.
34    pub entries: Vec<Integer>,
35}
36
37impl IntegerMatrix {
38    /// Provides the `new` operation on this native type.
39    pub fn new(rows: usize, columns: usize, entries: Vec<Integer>) -> Self {
40        assert_eq!(entries.len(), rows * columns);
41        Self {
42            rows,
43            columns,
44            entries,
45        }
46    }
47
48    /// Provides the `identity` operation on this native type.
49    pub fn identity(size: usize) -> Self {
50        let mut entries = vec![crate::exact_arithmetic::integer_zero(); size * size];
51        for index in 0..size {
52            entries[index * size + index] = crate::exact_arithmetic::integer_one();
53        }
54        Self::new(size, size, entries)
55    }
56
57    /// Provides the `entry` operation on this native type.
58    pub fn entry(&self, row: usize, column: usize) -> &Integer {
59        &self.entries[row * self.columns + column]
60    }
61
62    /// Provides the `transpose` operation on this native type.
63    pub fn transpose(&self) -> Self {
64        let mut entries = vec![crate::exact_arithmetic::integer_zero(); self.entries.len()];
65        for row in 0..self.rows {
66            for column in 0..self.columns {
67                entries[column * self.rows + row] = self.entry(row, column).clone();
68            }
69        }
70        Self::new(self.columns, self.rows, entries)
71    }
72}

112/// Public native enumeration named `integer_linear_algebra_error`.
113pub enum IntegerLinearAlgebraError {
114    /// Represents the `nonintegral_entry` case.
115    NonintegralEntry {
116        #[doc = "Stores the `row` value."]
117        row: usize,
118        #[doc = "Stores the `column` value."]
119        column: usize,
120    },
121    /// Represents the `size_budget_exceeded` case.
122    SizeBudgetExceeded {
123        #[doc = "Stores the `limit_bits` value."]
124        limit_bits: u64,
125    },
126    /// Represents the `verification_failed` case.
127    VerificationFailed {
128        #[doc = "Stores the `description` value."]
129        description: &'static str,
130    },
131}
132
133impl fmt::Display for IntegerLinearAlgebraError {
134    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
135        match self {
136            Self::NonintegralEntry { row, column } => {
137                write!(
138                    formatter,
139                    "matrix entry ({row}, {column}) is not an integer"
140                )
141            }
142            Self::SizeBudgetExceeded { limit_bits } => {
143                write!(
144                    formatter,
145                    "integer linear algebra exceeded the {limit_bits}-bit entry budget"
146                )
147            }
148            Self::VerificationFailed { description } => {
149                write!(
150                    formatter,
151                    "integer linear algebra verification failed: {description}"
152                )
153            }
154        }
155    }
156}
