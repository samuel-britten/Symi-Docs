13/// Public native enumeration named `zero_equivalence_verdict`.
14pub enum ZeroEquivalenceVerdict {
15    /// Represents the `true_value` case.
16    TrueValue,
17    /// Represents the `false_value` case.
18    FalseValue,
19    /// Represents the `unknown` case.
20    Unknown,
21}
22
23impl ZeroEquivalenceVerdict {
24    /// Provides the `as_str` operation on this native type.
25    pub fn as_str(self) -> &'static str {
26        match self {
27            ZeroEquivalenceVerdict::TrueValue => "true",
28            ZeroEquivalenceVerdict::FalseValue => "false",
29            ZeroEquivalenceVerdict::Unknown => "unknown",
30        }
31    }
32}

34/// Performs the advanced native `zero_equivalent` operation.
35pub fn zero_equivalent(
36    context: &Context,
37    input: &ExpressionReference,
38    constraint: Option<&ExpressionReference>,
39) -> ZeroEquivalenceVerdict {
40    if is_structural_zero(input) {
41        return ZeroEquivalenceVerdict::TrueValue;
42    }
43    if is_exact_nonzero_constant(input) {
44        return ZeroEquivalenceVerdict::FalseValue;
45    }
46    if let Some(variable) = gudermannian_variable(context, input) {
47        return gudermannian_verdict(context, input, constraint, variable);
48    }
49    if zeta_reflection_matches(context, input) {
50        return ZeroEquivalenceVerdict::TrueValue;
51    }
52    if jacobi_residual_matches(context, input) {
53        return ZeroEquivalenceVerdict::TrueValue;
54    }
55    if complete_legendre_carlson_residual_matches(context, input) {
56        return ZeroEquivalenceVerdict::TrueValue;
57    }
58    ZeroEquivalenceVerdict::Unknown
59}

61fn jacobi_residual_matches(context: &Context, input: &ExpressionReference) -> bool {
62    let mut calls = Vec::new();
63    collect_jacobi_calls(input, &mut calls);
64    calls.into_iter().any(|(builtin, argument, parameter)| {
65        let pythagorean =
66            crate::special_functions::jacobi_elliptic::identities::pythagorean_residual(
67                context, &argument, &parameter,
68            );
69        if pythagorean.kind == input.kind {
70            return true;
71        }
72        let parameter_residual =
73            crate::special_functions::jacobi_elliptic::identities::parameter_residual(
74                context, &argument, &parameter,
75            );
76        if parameter_residual.kind == input.kind {
77            return true;
78        }
79        let Some(index) =
80            crate::special_functions::jacobi_elliptic::JacobiIndex::of_builtin(builtin)
81        else {
82            return false;
83        };
84        let first_order =
85            crate::special_functions::jacobi_elliptic::identities::first_order_residual(
86                context, index, &argument, &parameter,
87            );
88        if first_order.kind == input.kind {
89            return true;
90        }
91        crate::special_functions::jacobi_elliptic::identities::second_order_residual(
92            context, index, &argument, &parameter,
93        )
94        .kind
95            == input.kind
96    })
97}

99fn collect_jacobi_calls(
100    input: &ExpressionReference,
101    calls: &mut Vec<(BuiltinFunction, ExpressionReference, ExpressionReference)>,
102) {
103    match &input.kind {
104        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments)
105            if crate::special_functions::jacobi_elliptic::JacobiIndex::of_builtin(*builtin)
106                .is_some()
107                && arguments.len() == 2 =>
108        {
109            calls.push((*builtin, arguments[0].clone(), arguments[1].clone()));
110        }
111        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
112            for child in children {
113                collect_jacobi_calls(child, calls);
114            }
115        }
116        ExpressionKind::Power(base, exponent) => {
117            collect_jacobi_calls(base, calls);
118            collect_jacobi_calls(exponent, calls);
119        }
120        ExpressionKind::Factorial(argument) => collect_jacobi_calls(argument, calls),
121        ExpressionKind::FunctionCall(_, arguments)
122        | ExpressionKind::Derivative(_, _, arguments) => {
123            for argument in arguments {
124                collect_jacobi_calls(argument, calls);
125            }
126        }
127        _ => {}
128    }
129}

131fn complete_legendre_carlson_residual_matches(
132    context: &Context,
133    input: &ExpressionReference,
134) -> bool {
135    let ExpressionKind::Sum(terms) = &input.kind else {
136        return false;
137    };
138    terms.iter().any(|term| {
139        let ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) =
140            &term.kind
141        else {
142            return false;
143        };
144        let Some(carlson) = crate::special_functions::elliptic_integrals::complete_as_carlson(
145            context, *builtin, arguments,
146        ) else {
147            return false;
148        };
149        builders::subtract(context, term.clone(), carlson).kind == input.kind
150    })
151}

153fn is_structural_zero(input: &ExpressionReference) -> bool {
154    matches!(&input.kind, ExpressionKind::Integer(value) if integer_is_zero(value))
155}

157fn is_exact_nonzero_constant(input: &ExpressionReference) -> bool {
158    // Exactly the constants this decision can settle by arithmetic alone: rational literals,
159    // certified algebraic roots, and the Gaussian rationals built from them and the imaginary
160    // unit. A nonzero imaginary constant is as decidable as a nonzero real one, and leaving it
161    // undecided is what makes a complex characteristic root look like it might be a root of
162    // higher multiplicity than it is.
163    match &input.kind {
164        ExpressionKind::Integer(value) => !integer_is_zero(value),
165        ExpressionKind::Rational(_) | ExpressionKind::PolynomialRoot(_) => true,
166        _ => gaussian_rational_value(input).is_some_and(|(real_part, imaginary_part)| {
167            !crate::number::is_zero(&real_part) || !crate::number::is_zero(&imaginary_part)
168        }),
169    }
170}

172/// The largest integer power a Gaussian-rational constant is expanded through.
173const MAXIMUM_GAUSSIAN_RATIONAL_POWER: i64 = 64;
174
175fn gaussian_rational_value(
176    input: &ExpressionReference,
177) -> Option<(crate::number::Number, crate::number::Number)> {
178    // Evaluate a constant expression in the Gaussian rationals, as the pair of its real and
179    // imaginary parts. Returns None for anything outside that field, so nothing here decides a
180    // value it cannot compute exactly.
181    let zero = crate::number::Number::Integer(crate::exact_arithmetic::integer_zero());
182    match &input.kind {
183        ExpressionKind::Integer(value) => {
184            Some((crate::number::Number::Integer(value.clone()), zero))
185        }
186        ExpressionKind::Rational(value) => {
187            Some((crate::number::Number::Rational(value.clone()), zero))
188        }
189        ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit) => Some((
190            zero,
191            crate::number::Number::Integer(crate::exact_arithmetic::integer_one()),
192        )),
193        ExpressionKind::Sum(children) => {
194            let mut accumulated = (zero.clone(), zero);
195            for child in children {
196                let (real_part, imaginary_part) = gaussian_rational_value(child)?;
197                accumulated = (
198                    crate::number::add(&accumulated.0, &real_part),
199                    crate::number::add(&accumulated.1, &imaginary_part),
200                );
201            }
202            Some(accumulated)
203        }
204        ExpressionKind::Product(children) => {
205            let mut accumulated = (
206                crate::number::Number::Integer(crate::exact_arithmetic::integer_one()),
207                zero,
208            );
209            for child in children {
210                let factor = gaussian_rational_value(child)?;
211                accumulated = gaussian_rational_product(&accumulated, &factor);
212            }
213            Some(accumulated)
214        }
215        ExpressionKind::Power(base, exponent) => {
216            let ExpressionKind::Integer(power) = &exponent.kind else {
217                return None;
218            };
219            let power: i64 = power.to_string().parse().ok()?;
220            if power.abs() > MAXIMUM_GAUSSIAN_RATIONAL_POWER {
221                return None;
222            }
223            let value = gaussian_rational_value(base)?;
224            let mut accumulated = (
225                crate::number::Number::Integer(crate::exact_arithmetic::integer_one()),
226                zero,
227            );
228            for _ in 0..power.abs() {
229                accumulated = gaussian_rational_product(&accumulated, &value);
230            }
231            if power >= 0 {
232                return Some(accumulated);
233            }
234            gaussian_rational_reciprocal(&accumulated)
235        }
236        _ => None,
237    }
238}

240fn gaussian_rational_product(
241    left: &(crate::number::Number, crate::number::Number),
242    right: &(crate::number::Number, crate::number::Number),
243) -> (crate::number::Number, crate::number::Number) {
244    (
245        crate::number::subtract(
246            &crate::number::multiply(&left.0, &right.0),
247            &crate::number::multiply(&left.1, &right.1),
248        ),
249        crate::number::add(
250            &crate::number::multiply(&left.0, &right.1),
251            &crate::number::multiply(&left.1, &right.0),
252        ),
253    )
254}
255
256fn gaussian_rational_reciprocal(
257    value: &(crate::number::Number, crate::number::Number),
258) -> Option<(crate::number::Number, crate::number::Number)> {
259    let modulus = crate::number::add(
260        &crate::number::multiply(&value.0, &value.0),
261        &crate::number::multiply(&value.1, &value.1),
262    );
263    Some((
264        crate::number::divide(&value.0, &modulus)?,
265        crate::number::divide(&crate::number::negate(&value.1), &modulus)?,
266    ))
267}

269fn gudermannian_variable(
270    context: &Context,
271    input: &ExpressionReference,
272) -> Option<SymbolIdentifier> {
273    let ExpressionKind::Sum(terms) = &input.kind else {
274        return None;
275    };
276    for term in terms {
277        let Some(variable) = arcsinh_tangent_variable(term) else {
278            continue;
279        };
280        if gudermannian_expression(context, variable).kind == input.kind {
281            return Some(variable);
282        }
283    }
284    None
285}

287fn arcsinh_tangent_variable(input: &ExpressionReference) -> Option<SymbolIdentifier> {
288    let ExpressionKind::FunctionCall(
289        FunctionIdentifier::Builtin(BuiltinFunction::Arcsinh),
290        arguments,
291    ) = &input.kind
292    else {
293        if let ExpressionKind::Product(factors) = &input.kind {
294            for factor in factors {
295                if let Some(variable) = arcsinh_tangent_variable(factor) {
296                    return Some(variable);
297                }
298            }
299        }
300        return None;
301    };
302    let [argument] = arguments.as_slice() else {
303        return None;
304    };
305    let ExpressionKind::FunctionCall(
306        FunctionIdentifier::Builtin(BuiltinFunction::Tan),
307        tangent_arguments,
308    ) = &argument.kind
309    else {
310        return None;
311    };
312    let [tangent_argument] = tangent_arguments.as_slice() else {
313        return None;
314    };
315    let ExpressionKind::Symbol(variable) = tangent_argument.kind else {
316        return None;
317    };
318    Some(variable)
319}

321fn gudermannian_verdict(
322    context: &Context,
323    input: &ExpressionReference,
324    constraint: Option<&ExpressionReference>,
325    variable: SymbolIdentifier,
326) -> ZeroEquivalenceVerdict {
327    if has_gudermannian_region(context, constraint, variable)
328        && is_real(
329            &context.interner.intern(ExpressionKind::Symbol(variable)),
330            &context.symbol_table,
331        ) == TruthValue::Yes
332    {
333        return ZeroEquivalenceVerdict::TrueValue;
334    }
335    if constraint.is_none() && gudermannian_expression(context, variable).kind == input.kind {
336        return ZeroEquivalenceVerdict::FalseValue;
337    }
338    ZeroEquivalenceVerdict::Unknown
339}

341fn has_gudermannian_region(
342    context: &Context,
343    constraint: Option<&ExpressionReference>,
344    variable: SymbolIdentifier,
345) -> bool {
346    let Some(constraint) = constraint else {
347        return false;
348    };
349    let variable = context.interner.intern(ExpressionKind::Symbol(variable));
350    let pi = context
351        .interner
352        .intern(ExpressionKind::Constant(BuiltinConstant::Pi));
353    let two = builders::integer(context, Integer::from(2));
354    let positive_lower = builders::divide(context, Arc::clone(&pi), Arc::clone(&two));
355    let lower = builders::negate(context, positive_lower);
356    let upper = builders::divide(context, pi, two);
357    let (lower_bounds, upper_bounds) = interval_bounds(constraint, &variable);
358    lower_bounds.iter().any(|bound| {
359        let comparison =
360            builders::greater_than_or_equal(context, Arc::clone(bound), Arc::clone(&lower));
361        crate::logic::evaluate_truth_in_context(context, &comparison) == Ok(TruthValue::Yes)
362    }) && upper_bounds.iter().any(|bound| {
363        let comparison =
364            builders::less_than_or_equal(context, Arc::clone(bound), Arc::clone(&upper));
365        crate::logic::evaluate_truth_in_context(context, &comparison) == Ok(TruthValue::Yes)
366    })
367}

369fn interval_bounds(
370    constraint: &ExpressionReference,
371    variable: &ExpressionReference,
372) -> (Vec<ExpressionReference>, Vec<ExpressionReference>) {
373    let mut lower_bounds = Vec::new();
374    let mut upper_bounds = Vec::new();
375    collect_interval_bounds(constraint, variable, &mut lower_bounds, &mut upper_bounds);
376    (lower_bounds, upper_bounds)
377}

379fn collect_interval_bounds(
380    constraint: &ExpressionReference,
381    variable: &ExpressionReference,
382    lower_bounds: &mut Vec<ExpressionReference>,
383    upper_bounds: &mut Vec<ExpressionReference>,
384) {
385    match &constraint.kind {
386        ExpressionKind::FunctionCall(
387            FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
388            arguments,
389        ) => {
390            for argument in arguments {
391                collect_interval_bounds(argument, variable, lower_bounds, upper_bounds);
392            }
393        }
394        ExpressionKind::FunctionCall(
395            FunctionIdentifier::Builtin(BuiltinFunction::GreaterThan),
396            arguments,
397        ) if arguments.len() == 2 && arguments[0].kind == variable.kind => {
398            lower_bounds.push(Arc::clone(&arguments[1]));
399        }
400        ExpressionKind::FunctionCall(
401            FunctionIdentifier::Builtin(BuiltinFunction::LessThan),
402            arguments,
403        ) if arguments.len() == 2 && arguments[1].kind == variable.kind => {
404            lower_bounds.push(Arc::clone(&arguments[0]));
405        }
406        ExpressionKind::FunctionCall(
407            FunctionIdentifier::Builtin(BuiltinFunction::LessThan),
408            arguments,
409        ) if arguments.len() == 2 && arguments[0].kind == variable.kind => {
410            upper_bounds.push(Arc::clone(&arguments[1]));
411        }
412        ExpressionKind::FunctionCall(
413            FunctionIdentifier::Builtin(BuiltinFunction::GreaterThan),
414            arguments,
415        ) if arguments.len() == 2 && arguments[1].kind == variable.kind => {
416            upper_bounds.push(Arc::clone(&arguments[0]));
417        }
418        _ => {}
419    }
420}

422fn gudermannian_expression(context: &Context, variable: SymbolIdentifier) -> ExpressionReference {
423    let variable = context.interner.intern(ExpressionKind::Symbol(variable));
424    let pi = context
425        .interner
426        .intern(ExpressionKind::Constant(BuiltinConstant::Pi));
427    let two = builders::integer(context, Integer::from(2));
428    let four = builders::integer(context, Integer::from(4));
429    let half_x = builders::divide(context, Arc::clone(&variable), two);
430    let quarter_pi = builders::divide(context, pi, four);
431    let half_angle = builders::sum(context, vec![quarter_pi, half_x]);
432    let tangent_half_angle = builders::function_call(
433        context,
434        FunctionIdentifier::Builtin(BuiltinFunction::Tan),
435        vec![half_angle],
436    );
437    let logarithm = builders::function_call(
438        context,
439        FunctionIdentifier::Builtin(BuiltinFunction::Log),
440        vec![tangent_half_angle],
441    );
442    let tangent = builders::function_call(
443        context,
444        FunctionIdentifier::Builtin(BuiltinFunction::Tan),
445        vec![variable],
446    );
447    let arcsinh = builders::function_call(
448        context,
449        FunctionIdentifier::Builtin(BuiltinFunction::Arcsinh),
450        vec![tangent],
451    );
452    builders::subtract(context, logarithm, arcsinh)
453}

455fn zeta_reflection_matches(context: &Context, input: &ExpressionReference) -> bool {
456    if let Some(argument) = first_zeta_argument(input) {
457        if zeta_reflection_expression(context, &argument).kind == input.kind {
458            return true;
459        }
460    }
461    match &input.kind {
462        ExpressionKind::Product(factors) => factors
463            .iter()
464            .any(|factor| zeta_reflection_matches(context, factor)),
465        _ => false,
466    }
467}

469fn first_zeta_argument(input: &ExpressionReference) -> Option<ExpressionReference> {
470    match &input.kind {
471        ExpressionKind::FunctionCall(
472            FunctionIdentifier::Builtin(BuiltinFunction::Zeta),
473            arguments,
474        ) if arguments.len() == 1 => Some(Arc::clone(&arguments[0])),
475        ExpressionKind::Sum(terms) | ExpressionKind::Product(terms) => {
476            terms.iter().find_map(first_zeta_argument)
477        }
478        ExpressionKind::Power(base, exponent) => {
479            first_zeta_argument(base).or_else(|| first_zeta_argument(exponent))
480        }
481        ExpressionKind::FunctionCall(_, arguments) => {
482            arguments.iter().find_map(first_zeta_argument)
483        }
484        _ => None,
485    }
486}

488fn zeta_reflection_expression(
489    context: &Context,
490    argument: &ExpressionReference,
491) -> ExpressionReference {
492    let one = builders::integer(context, Integer::ONE);
493    let two = builders::integer(context, Integer::from(2));
494    let pi = context
495        .interner
496        .intern(ExpressionKind::Constant(BuiltinConstant::Pi));
497    let one_minus_argument = builders::subtract(context, one, Arc::clone(argument));
498    let exponent_one = builders::integer(context, Integer::ONE);
499    let exponent = builders::subtract(context, exponent_one, Arc::clone(argument));
500    let power_of_two = builders::power(context, two, exponent);
501    let pi_times_argument = builders::product(context, vec![pi.clone(), Arc::clone(argument)]);
502    let cosine_denominator = builders::integer(context, Integer::from(2));
503    let cosine_argument = builders::divide(context, pi_times_argument, cosine_denominator);
504    let cosine = builders::function_call(
505        context,
506        FunctionIdentifier::Builtin(BuiltinFunction::Cos),
507        vec![cosine_argument],
508    );
509    let gamma = builders::function_call(
510        context,
511        FunctionIdentifier::Builtin(BuiltinFunction::Gamma),
512        vec![Arc::clone(argument)],
513    );
514    let zeta = builders::function_call(
515        context,
516        FunctionIdentifier::Builtin(BuiltinFunction::Zeta),
517        vec![Arc::clone(argument)],
518    );
519    let reflected_zeta = builders::function_call(
520        context,
521        FunctionIdentifier::Builtin(BuiltinFunction::Zeta),
522        vec![one_minus_argument],
523    );
524    let left = builders::product(context, vec![power_of_two, cosine, gamma, zeta]);
525    let pi_power = builders::power(context, pi, Arc::clone(argument));
526    let right = builders::product(context, vec![pi_power, reflected_zeta]);
527    builders::subtract(context, left, right)
528}
