1// Solution-family semantics for scalar partial differential equations.
2//
3// A partial differential equation does not have a finite list of solutions, so the answer to a
4// solve request is a family, and a family is only as trustworthy as the data recorded beside it.
5// The name and arity of an arbitrary function say nothing about which coordinates it may involve,
6// which functions of those coordinates make the expression a solution, or whether every solution
7// of the equation is reached. Those three are different mathematical statements:
8//
9//   * the *dependence* is structural, and the verifier can refute a violation exactly;
10//   * the *admissible class* is analytic, and it is a hypothesis on the datum rather than a
11//     residual, because no substitution decides whether an unspecified function is continuous
12//     with polynomial growth;
13//   * the *completeness theorem* is a statement about the solution set, and a verified residual
14//     never establishes one.
15//
16// Recording all three separately is what keeps a constructed family from being promoted past
17// what was actually proved: a kernel that satisfies the equation is a solution, a solution with a
18// free datum is a family, and only a named theorem with stated hypotheses makes it the general
19// family on the stated chart.

29/// Public native enumeration named `partial_differential_equation_permitted_dependence`.
30///
31/// Which independent variables an arbitrary datum of a family may involve. `f_xy = 0` is solved by
32/// `g(x) + h(y)` and refuted by `g(x, y) + h(y)`, and nothing but recorded dependence separates
33/// the two, so the verifier refuses a candidate whose datum reaches a coordinate it may not.
34#[derive(Debug, Clone, PartialEq, Eq, Hash)]
35pub enum PartialDifferentialEquationPermittedDependence {
36    /// No dependence restriction was recorded, so none is checked.
37    Unrestricted,
38    /// The datum's arguments may involve exactly these independent variables.
39    Coordinates(Vec<SymbolIdentifier>),
40}
41
42/// Public native enumeration named `partial_differential_equation_admissible_class`.
43///
44/// The function class an arbitrary datum ranges over. This is a hypothesis, never a residual: a
45/// family is a solution *for data in this class*, and the class is what a uniqueness theorem is
46/// stated over. Declaring the widest class that the construction actually supports, rather than
47/// the narrowest one that makes an integral converge, is the point of recording it.
48#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
49pub enum PartialDifferentialEquationAdmissibleClass {
50    /// Any function of its arguments with enough derivatives for the governing residual to be
51    /// defined on the declared chart.
52    SufficientlyDifferentiableOnTheChart,
53    /// Continuous on the whole real line and bounded in absolute value by some polynomial. The
54    /// Gaussian kernel integrates against every such datum at every positive time, and Tychonoff
55    /// uniqueness holds inside the class.
56    ContinuousOfPolynomialGrowthOnTheRealLine,
57}
58
59impl PartialDifferentialEquationAdmissibleClass {
60    /// Returns the serialized name of the admissible class.
61    pub fn name(self) -> &'static str {
62        match self {
63            PartialDifferentialEquationAdmissibleClass::SufficientlyDifferentiableOnTheChart => {
64                "sufficiently_differentiable_on_the_chart"
65            }
66            PartialDifferentialEquationAdmissibleClass::ContinuousOfPolynomialGrowthOnTheRealLine => {
67                "continuous_of_polynomial_growth_on_the_real_line"
68            }
69        }
70    }
71}

73/// Public native data structure named `PartialDifferentialEquationArbitraryFunction`.
74#[derive(Debug, Clone, PartialEq, Eq, Hash)]
75pub struct PartialDifferentialEquationArbitraryFunction {
76    /// Stores the generated function identifier.
77    pub function: FunctionIdentifier,
78    /// Stores the validated function arity.
79    pub arity: usize,
80    /// Stores the independent variables the datum's arguments may involve.
81    pub permitted_dependence: PartialDifferentialEquationPermittedDependence,
82    /// Stores the class the datum ranges over for the family to solve the equation.
83    pub admissible_class: PartialDifferentialEquationAdmissibleClass,
84}
85
86impl PartialDifferentialEquationArbitraryFunction {
87    /// Builds an arbitrary datum with no recorded dependence or analytic restriction.
88    pub fn unrestricted(function: FunctionIdentifier, arity: usize) -> Self {
89        Self {
90            function,
91            arity,
92            permitted_dependence: PartialDifferentialEquationPermittedDependence::Unrestricted,
93            admissible_class:
94                PartialDifferentialEquationAdmissibleClass::SufficientlyDifferentiableOnTheChart,
95        }
96    }
97
98    /// Builds an arbitrary datum confined to the named coordinates.
99    pub fn of_coordinates(
100        function: FunctionIdentifier,
101        coordinates: Vec<SymbolIdentifier>,
102        admissible_class: PartialDifferentialEquationAdmissibleClass,
103    ) -> Self {
104        Self {
105            function,
106            arity: coordinates.len(),
107            permitted_dependence: PartialDifferentialEquationPermittedDependence::Coordinates(
108                coordinates,
109            ),
110            admissible_class,
111        }
112    }
113}

140/// Public native enumeration named `partial_differential_equation_family_hypothesis`.
141///
142/// A condition the completeness theorem is stated under. These are neither residuals nor
143/// discharged obligations: they restrict the chart or the data, and the family is the general one
144/// only where they hold.
145#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
146pub enum PartialDifferentialEquationFamilyHypothesis {
147    /// The chart is a connected product of intervals in the declared coordinates. On a
148    /// disconnected or non-product region the same equation has solutions this family misses.
149    ConnectedProductChart,
150    /// The declared evolution coordinate is strictly positive.
151    PositiveEvolutionTime,
152    /// The declared spatial coordinate ranges over the whole real line.
153    WholeRealLineSpatialCoordinate,
154    /// The arbitrary datum lies in the family's declared admissible class.
155    DatumInTheAdmissibleClass,
156    /// Differentiating under the integral sign is justified for data in that class, which is what
157    /// carries the exact kernel identity through to the operator.
158    DifferentiationUnderTheIntegralSign,
159    /// The radial-angular chart extends to an open disk by regularity at the origin.
160    OpenDiskWithOriginRegularity,
161    /// The angular coordinate is identified modulo one full turn.
162    AngularPeriodicity,
163    /// The coefficient series and its differentiated series converge uniformly on compact
164    /// subdisks.
165    UniformConvergenceOfTheModeSeriesOnCompactSubdisks,
166}

200/// Public native enumeration named `partial_differential_equation_family_identity`.
201///
202/// An exact identity a construction rests on, proved on its own rather than inferred from the
203/// family it produced. Each one is checked by reducing a residual that mentions no arbitrary
204/// datum, so the proof does not consist of the solver agreeing with itself.
205#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
206pub enum PartialDifferentialEquationFamilyIdentity {
207    /// The evolution kernel itself solves the governing equation at positive time.
208    KernelSolvesTheGoverningEquation,
209    /// The kernel integrates to one over the whole spatial line at every positive time, so the
210    /// operator preserves constants and the initial trace is not rescaled.
211    KernelMassIsOne,
212    /// The operator applied to a stated polynomial datum reproduces a known heat solution.
213    OperatorReproducesAPolynomialDatum,
214    /// The iterated derivative the equation sets to zero annihilates each recorded basis term.
215    IteratedDerivativeAnnihilatesEachTerm,
216    /// The polar Laplacian annihilates the constant and every recorded radial-angular mode.
217    PolarLaplacianAnnihilatesEachMode,
218    /// The radial modes selected by origin regularity extend smoothly across the origin.
219    RadialModesExtendSmoothlyToTheOrigin,
220}

264/// Public native enumeration named `partial_differential_equation_completeness_theorem`.
265///
266/// The named theorem under which a family is the general one. A family with no entry here is a
267/// set of solutions and nothing more; a verified residual is membership, never coverage.
268#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
269pub enum PartialDifferentialEquationCompletenessTheorem {
270    /// On a connected product of intervals, every solution of an iterated derivative set to zero
271    /// is the finite sum of coordinate-power multiples of arbitrary functions of the remaining
272    /// coordinates recorded by the family. Proof: integrate the vanishing derivative once in each
273    /// of its coordinates, the constant of each integration being an arbitrary function of the
274    /// coordinates not yet integrated.
275    VanishingIteratedDerivativeOnAProductChart,
276    /// Every solution in the declared admissible class equals the recorded operator applied to
277    /// its own trace at the initial time. Proof: Tychonoff uniqueness inside the class, together
278    /// with the operator's kernel identity, mass normalization, and attained initial trace.
279    InitialTraceRepresentationInTheAdmissibleClass,
280    /// Every real harmonic function on an open disk has the recorded convergent polar expansion;
281    /// conversely, every expansion satisfying its coefficient convergence hypothesis is harmonic.
282    RegularHarmonicExpansionOnADisk,
283}
284
285impl PartialDifferentialEquationCompletenessTheorem {
286    /// Returns the serialized name of the completeness theorem.
287    pub fn name(self) -> &'static str {
288        match self {
289            PartialDifferentialEquationCompletenessTheorem::VanishingIteratedDerivativeOnAProductChart => {
290                "vanishing_iterated_derivative_on_a_product_chart"
291            }
292            PartialDifferentialEquationCompletenessTheorem::InitialTraceRepresentationInTheAdmissibleClass => {
293                "initial_trace_representation_in_the_admissible_class"
294            }
295            PartialDifferentialEquationCompletenessTheorem::RegularHarmonicExpansionOnADisk => {
296                "regular_harmonic_expansion_on_a_disk"
297            }
298        }
299    }
300}

302/// Public native enumeration named `partial_differential_equation_representation_redundancy`.
303///
304/// Whether distinct data give the same member of the family. The redundancy is a property of the
305/// representation, not a defect, but a caller counting arbitrary functions as free parameters is
306/// wrong without it: `g(x) + h(y)` is unchanged when a constant moves from `g` to `h`.
307#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
308pub enum PartialDifferentialEquationRepresentationRedundancy {
309    /// Distinct data give distinct members.
310    DataDetermineTheMemberUniquely,
311    /// Data differing by a polynomial of degree below the differentiated order in each shared
312    /// coordinate give the same member; the additive constant shared by two summands is the
313    /// smallest instance.
314    SharedPolynomialFreedomBetweenSummands,
315}

317/// Public native data structure named `partial_differential_equation_completeness_certificate`.
318#[derive(Debug, Clone)]
319pub struct PartialDifferentialEquationCompletenessCertificate {
320    /// Stores the theorem under which the family is the general one.
321    pub theorem: PartialDifferentialEquationCompletenessTheorem,
322    /// Stores the hypotheses the theorem is stated under.
323    pub hypotheses: Vec<PartialDifferentialEquationFamilyHypothesis>,
324    /// Stores identities proved exactly during construction.
325    pub proved_identities: Vec<PartialDifferentialEquationProvedIdentity>,
326    /// Stores whether distinct data give distinct members.
327    pub representation_redundancy: PartialDifferentialEquationRepresentationRedundancy,
328}
329
330impl PartialDifferentialEquationCompletenessCertificate {
331    /// Reports whether every identity the certificate names was proved exactly.
332    pub fn identities_are_proved(&self) -> bool {
333        self.proved_identities
334            .iter()
335            .all(PartialDifferentialEquationProvedIdentity::is_proved)
336    }
337}

339/// Public native data structure named `partial_differential_equation_family_semantics`.
340///
341/// Everything a family states beyond its own expression. Kept in one record so that a route which
342/// establishes nothing beyond a residual constructs it by default and cannot silently acquire a
343/// completeness claim by adding a field somewhere else.
344#[derive(Debug, Clone, Default)]
345pub struct PartialDifferentialEquationFamilySemantics {
346    /// Stores the completeness theorem the family satisfies, when one was established.
347    pub completeness: Option<PartialDifferentialEquationCompletenessCertificate>,
348    /// Stores arbitrary indexed coefficient sequences with their index domains.
349    pub coefficient_families: Vec<PartialDifferentialEquationCoefficientFamily>,
350    /// Stores obligations the family states and does not discharge, so a scope it does not reach
351    /// stays visible instead of disappearing into a solved verdict.
352    pub outstanding_obligations: Vec<PartialDifferentialEquationProofObligation>,
353}

364/// Public native data structure named `partial_differential_equation_solution_family`.
365#[derive(Debug, Clone)]
366pub struct PartialDifferentialEquationSolutionFamily {
367    /// Stores the candidate solution form.
368    pub form: PartialDifferentialEquationSolutionForm,
369    /// Stores arbitrary functions that remain symbolic during verification.
370    pub arbitrary_functions: Vec<PartialDifferentialEquationArbitraryFunction>,
371    /// Stores method-specific obligations supplied with the candidate.
372    pub proof_obligations: Vec<PartialDifferentialEquationProofObligation>,
373    /// Stores what the family claims beyond solving the equation.
374    pub semantics: PartialDifferentialEquationFamilySemantics,
375}
376
377impl PartialDifferentialEquationSolutionFamily {
378    /// Returns the candidate expression underlying either solution form.
379    pub fn candidate_expression(&self) -> &ExpressionReference {
380        match &self.form {
381            PartialDifferentialEquationSolutionForm::Explicit(value) => value,
382            PartialDifferentialEquationSolutionForm::Implicit(value) => value,
383        }
384    }
385
386    /// Reports whether the family establishes the complete solution set on its declared chart.
387    ///
388    /// A certificate counts only when every identity it names was proved exactly; a certificate
389    /// carrying an unproved identity describes an intended argument, not a finished one.
390    pub fn establishes_completeness(&self) -> bool {
391        self.semantics
392            .completeness
393            .as_ref()
394            .is_some_and(PartialDifferentialEquationCompletenessCertificate::identities_are_proved)
395    }
