1// The initial-data solution operator of a one-dimensional diffusion equation.
2//
3// `u_t = a*u_xx` on the whole spatial line has no finite-dimensional general solution, and a
4// finite sum of separated modes offered as one is simply a different, much smaller, set of
5// functions. What the equation does have is a solution *operator*: for an admissible datum `g`,
6//
7//     u(x, t) = Integral(exp(-(x - s)^2/(4*a*t))*g(s), s = -infinity..infinity)/sqrt(4*pi*a*t)
8//
9// solves the equation at every positive time and tends to `g` as the time tends to zero. The
10// arbitrary datum here is an arbitrary *input profile*, not an arbitrary function already
11// required to solve the equation: a family whose free function is constrained only by the
12// original unsolved equation would be a restatement of the problem rather than an answer to it.
13//
14// Three things are proved exactly, none of them by asking the construction whether it agrees with
15// itself:
16//
17//   * the kernel alone solves the equation, checked on the kernel with the datum absent;
18//   * the kernel has unit mass at every positive time, which is what keeps the operator from
19//     rescaling the datum it is supposed to reproduce;
20//   * the operator returns known solutions on polynomial data, which is the trace check that a
21//     decaying-data chart would never exercise, since those solutions do not decay at all.
22//
23// The completeness claim is stated over a class and no wider. Inside continuous data of at most
24// polynomial growth the representation is complete, by Tychonoff uniqueness; outside it the heat
25// equation has nonzero solutions with zero initial data, so no representation by initial trace
26// can be complete there, and that obligation is retained on the family instead of being quietly
27// dropped.

57/// Public native enumeration named `partial_differential_equation_evolution_operator_decline`.
58#[derive(Debug, Clone, PartialEq, Eq)]
59pub enum PartialDifferentialEquationEvolutionOperatorDecline {
60    /// The equation is not a one-dimensional diffusion equation in the declared coordinates.
61    EquationIsNotOneDimensionalDiffusion,
62    /// The diffusion coefficient was not proved to be a positive constant, so neither the kernel
63    /// nor its convergence over the whole line is justified.
64    DiffusionCoefficientNotProvedPositiveConstant,
65    /// The evolution coordinate was not assumed positive, and the kernel exists only there.
66    EvolutionTimeNotProvedPositive,
67    /// The problem carries initial or boundary data that this operator does not fit.
68    DataNotSupported,
69    /// The constructed operator did not verify against the original problem.
70    ConstructedFamilyNotVerified,
71}

100/// Builds the whole-line initial-data solution operator of a diffusion equation.
101pub fn solve_by_initial_data_convolution(
102    context: &Context,
103    problem: &PartialDifferentialEquationProblem,
104) -> Result<
105    PartialDifferentialEquationEvolutionOperatorSolution,
106    PartialDifferentialEquationEvolutionOperatorDecline,
107> {
108    if !problem.initial_conditions().is_empty() || !problem.boundary_conditions().is_empty() {
109        return Err(PartialDifferentialEquationEvolutionOperatorDecline::DataNotSupported);
110    }
111    let shape = diffusion_shape(context, problem)?;
112    let datum = allocate_partial_differential_equation_arbitrary_function(context, "g", 1);
113    let integration_variable =
114        allocate_partial_differential_equation_symbol(context, "integration_variable");
115    let solution = build_operator(context, &shape, datum.function, integration_variable);
116    let kernel_identity = prove_kernel_solves_the_equation(context, &shape);
117    let mass_identity = prove_kernel_mass_is_one(context, &shape, integration_variable);
118    let polynomial_identity =
119        prove_operator_reproduces_a_polynomial_datum(context, &shape, integration_variable);
120    let completeness_established =
121        kernel_identity.is_proved() && mass_identity.is_proved() && polynomial_identity.is_proved();
122    let proved_identities = vec![kernel_identity, mass_identity, polynomial_identity];
123    let family = PartialDifferentialEquationSolutionFamily {
124        form: PartialDifferentialEquationSolutionForm::Explicit(solution),
125        arbitrary_functions: vec![
126            PartialDifferentialEquationArbitraryFunction::of_coordinates(
127                datum.function,
128                vec![integration_variable],
129                PartialDifferentialEquationAdmissibleClass::ContinuousOfPolynomialGrowthOnTheRealLine,
130            ),
131        ],
132        proof_obligations: vec![
133            PartialDifferentialEquationProofObligation::ConstantCoefficients,
134            PartialDifferentialEquationProofObligation::DifferentiationUnderTheIntegralSign,
135            PartialDifferentialEquationProofObligation::InitialTraceAttainedInTheAdmissibleClass,
136            PartialDifferentialEquationProofObligation::UniquenessInTheAdmissibleClass,
137        ],
138        semantics: PartialDifferentialEquationFamilySemantics {
139            completeness: completeness_established.then(|| {
140                PartialDifferentialEquationCompletenessCertificate {
141                    theorem: PartialDifferentialEquationCompletenessTheorem::InitialTraceRepresentationInTheAdmissibleClass,
142                    hypotheses: vec![
143                        PartialDifferentialEquationFamilyHypothesis::WholeRealLineSpatialCoordinate,
144                        PartialDifferentialEquationFamilyHypothesis::PositiveEvolutionTime,
145                        PartialDifferentialEquationFamilyHypothesis::DatumInTheAdmissibleClass,
146                        PartialDifferentialEquationFamilyHypothesis::DifferentiationUnderTheIntegralSign,
147                    ],
148                    proved_identities,
149                    representation_redundancy:
150                        PartialDifferentialEquationRepresentationRedundancy::DataDetermineTheMemberUniquely,
151                }
152            }),
153            coefficient_families: Vec::new(),
154            outstanding_obligations: vec![
155                PartialDifferentialEquationProofObligation::CoverageOutsideTheAdmissibleClass,
156            ],
157        },
158    };
159    let verification = verify_partial_differential_equation_solution(context, problem, &family);
160    if verification.verdict != PartialDifferentialEquationVerificationVerdict::Verified {
161        return Err(
162            PartialDifferentialEquationEvolutionOperatorDecline::ConstructedFamilyNotVerified,
163        );
164    }
165    Ok(PartialDifferentialEquationEvolutionOperatorSolution {
166        family,
167        verification,
168        discharged_obligations: vec![
169            PartialDifferentialEquationProofObligation::ConstantCoefficients,
170        ],
171    })
172}

174/// Recognises `u_t = a*u_xx` with a positive constant coefficient and a positive evolution time.
175pub fn diffusion_shape(
176    context: &Context,
177    problem: &PartialDifferentialEquationProblem,
178) -> Result<
179    PartialDifferentialEquationDiffusionShape,
180    PartialDifferentialEquationEvolutionOperatorDecline,
181> {
182    // Read the two jets out of the residual and divide. The sign of the resulting coefficient is
183    // the whole content of the recognition: the same syntax with the opposite sign is the
184    // backward heat equation, whose initial-value problem is ill posed and for which this kernel
185    // is not a solution operator at all.
186    let variables = problem.independent_variables();
187    if variables.len() != 2 {
188        return Err(PartialDifferentialEquationEvolutionOperatorDecline::EquationIsNotOneDimensionalDiffusion);
189    }
190    let terms = match &problem.equation().kind {
191        ExpressionKind::Sum(terms) => terms.clone(),
192        _ => vec![problem.equation().clone()],
193    };
194    let mut evolution_term = None;
195    let mut diffusion_term = None;
196    for term in &terms {
197        let Some((coefficient, orders)) = split_jet_term(context, problem, term) else {
198            return Err(PartialDifferentialEquationEvolutionOperatorDecline::EquationIsNotOneDimensionalDiffusion);
199        };
200        let slot = match (orders[0], orders[1]) {
201            (0, 1) | (1, 0) => &mut evolution_term,
202            (2, 0) | (0, 2) => &mut diffusion_term,
203            _ => {
204                return Err(PartialDifferentialEquationEvolutionOperatorDecline::EquationIsNotOneDimensionalDiffusion);
205            }
206        };
207        if slot.is_some() {
208            return Err(PartialDifferentialEquationEvolutionOperatorDecline::EquationIsNotOneDimensionalDiffusion);
209        }
210        *slot = Some((coefficient, orders));
211    }
212    let (
213        Some((evolution_coefficient, evolution_orders)),
214        Some((diffusion_coefficient, diffusion_orders)),
215    ) = (evolution_term, diffusion_term)
216    else {
217        return Err(PartialDifferentialEquationEvolutionOperatorDecline::EquationIsNotOneDimensionalDiffusion);
218    };
219    let evolution_index = if evolution_orders[0] == 1 { 0 } else { 1 };
220    let spatial_index = if diffusion_orders[0] == 2 { 0 } else { 1 };
221    if evolution_index == spatial_index {
222        return Err(PartialDifferentialEquationEvolutionOperatorDecline::EquationIsNotOneDimensionalDiffusion);
223    }
224    let coefficient = builders::negate(
225        context,
226        builders::divide(context, diffusion_coefficient, evolution_coefficient),
227    );
228    let evolution_coordinate = variables[evolution_index];
229    let spatial_coordinate = variables[spatial_index];
230    if depends_on(&coefficient, evolution_coordinate)
231        || depends_on(&coefficient, spatial_coordinate)
232    {
233        return Err(PartialDifferentialEquationEvolutionOperatorDecline::DiffusionCoefficientNotProvedPositiveConstant);
234    }
235    if crate::assumptions::queries::is_positive(&coefficient, &context.symbol_table)
236        != TruthValue::Yes
237    {
238        return Err(PartialDifferentialEquationEvolutionOperatorDecline::DiffusionCoefficientNotProvedPositiveConstant);
239    }
240    let evolution_expression = context
241        .interner
242        .intern(ExpressionKind::Symbol(evolution_coordinate));
243    if crate::assumptions::queries::is_positive(&evolution_expression, &context.symbol_table)
244        != TruthValue::Yes
245    {
246        return Err(
247            PartialDifferentialEquationEvolutionOperatorDecline::EvolutionTimeNotProvedPositive,
248        );
249    }
250    Ok(PartialDifferentialEquationDiffusionShape {
251        spatial_coordinate,
252        evolution_coordinate,
253        diffusion_coefficient: coefficient,
254    })
255}

257fn split_jet_term(
258    context: &Context,
259    problem: &PartialDifferentialEquationProblem,
260    term: &ExpressionReference,
261) -> Option<(ExpressionReference, Vec<usize>)> {
262    // Split one additive term into its coefficient and the derivative orders of the single jet it
263    // carries, with the jet required to be taken at the declared coordinates themselves.
264    let (factors, derivative): (Vec<ExpressionReference>, ExpressionReference) = match &term.kind {
265        ExpressionKind::Derivative(_, _, _) => (Vec::new(), term.clone()),
266        ExpressionKind::Product(factors) => {
267            let mut derivative = None;
268            let mut coefficient = Vec::new();
269            for factor in factors {
270                match factor.kind {
271                    ExpressionKind::Derivative(function, _, _)
272                        if function == problem.dependent_function() =>
273                    {
274                        if derivative.is_some() {
275                            return None;
276                        }
277                        derivative = Some(factor.clone());
278                    }
279                    _ => coefficient.push(factor.clone()),
280                }
281            }
282            (coefficient, derivative?)
283        }
284        _ => return None,
285    };
286    let ExpressionKind::Derivative(function, orders, arguments) = &derivative.kind else {
287        return None;
288    };
289    if *function != problem.dependent_function() || orders.len() != 2 {
290        return None;
291    }
292    if !arguments
293        .iter()
294        .zip(problem.independent_variables())
295        .all(|(argument, variable)| {
296            matches!(argument.kind, ExpressionKind::Symbol(found) if found == *variable)
297        })
298    {
299        return None;
300    }
301    Some((builders::product(context, factors), orders.clone()))
302}

304fn kernel(
305    context: &Context,
306    shape: &PartialDifferentialEquationDiffusionShape,
307    displacement: ExpressionReference,
308) -> ExpressionReference {
309    // The Gauss-Weierstrass kernel at a stated displacement: exp(-d^2/(4*a*t))/sqrt(4*pi*a*t).
310    let four = builders::integer_value(context, 4);
311    let evolution = context
312        .interner
313        .intern(ExpressionKind::Symbol(shape.evolution_coordinate));
314    let spread = builders::product(
315        context,
316        vec![
317            four.clone(),
318            shape.diffusion_coefficient.clone(),
319            evolution.clone(),
320        ],
321    );
322    let exponent = builders::negate(
323        context,
324        builders::divide(
325            context,
326            builders::power(context, displacement, builders::integer_value(context, 2)),
327            spread,
328        ),
329    );
330    let exponential = builders::function_call(
331        context,
332        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
333        vec![exponent],
334    );
335    let pi = context
336        .interner
337        .intern(ExpressionKind::Constant(BuiltinConstant::Pi));
338    let normalizer = builders::power(
339        context,
340        builders::product(
341            context,
342            vec![four, pi, shape.diffusion_coefficient.clone(), evolution],
343        ),
344        builders::rational_value(context, 1, 2),
345    );
346    builders::divide(context, exponential, normalizer)
347}

349fn build_operator(
350    context: &Context,
351    shape: &PartialDifferentialEquationDiffusionShape,
352    datum: FunctionIdentifier,
353    integration_variable: SymbolIdentifier,
354) -> ExpressionReference {
355    let spatial = context
356        .interner
357        .intern(ExpressionKind::Symbol(shape.spatial_coordinate));
358    let source = context
359        .interner
360        .intern(ExpressionKind::Symbol(integration_variable));
361    let displacement = builders::subtract(context, spatial, Arc::clone(&source));
362    let integrand = builders::product(
363        context,
364        vec![
365            kernel(context, shape, displacement),
366            builders::function_call(context, datum, vec![source]),
367        ],
368    );
369    let infinity = context
370        .interner
371        .intern(ExpressionKind::Constant(BuiltinConstant::Infinity));
372    builders::integral(
373        context,
374        integrand,
375        integration_variable,
376        Some(builders::negate(context, Arc::clone(&infinity))),
377        Some(infinity),
378    )
379}

381fn prove_kernel_solves_the_equation(
382    context: &Context,
383    shape: &PartialDifferentialEquationDiffusionShape,
384) -> PartialDifferentialEquationProvedIdentity {
385    // The kernel with the datum absent, checked against the equation directly. This is the
386    // identity the whole operator rests on, and it involves no arbitrary function, so it is
387    // decided exactly rather than carried as a hypothesis.
388    let spatial = context
389        .interner
390        .intern(ExpressionKind::Symbol(shape.spatial_coordinate));
391    let value = kernel(context, shape, spatial);
392    let evolution_derivative =
393        partial_derivative(context, &value, &[(shape.evolution_coordinate, 1)]);
394    let spatial_derivative = partial_derivative(context, &value, &[(shape.spatial_coordinate, 2)]);
395    let residual = builders::subtract(
396        context,
397        evolution_derivative,
398        builders::product(
399            context,
400            vec![shape.diffusion_coefficient.clone(), spatial_derivative],
401        ),
402    );
403    PartialDifferentialEquationProvedIdentity {
404        verdict: verification_verdict(prove_residual_zero(context, &residual)),
405        identity: PartialDifferentialEquationFamilyIdentity::KernelSolvesTheGoverningEquation,
406        residual,
407    }
408}

410fn prove_kernel_mass_is_one(
411    context: &Context,
412    shape: &PartialDifferentialEquationDiffusionShape,
413    integration_variable: SymbolIdentifier,
414) -> PartialDifferentialEquationProvedIdentity {
415    // The kernel integrates to one over the whole line at every positive time. Without this the
416    // operator could solve the equation and still return a rescaled datum, so the initial trace
417    // would not be the datum at all.
418    let residual = builders::subtract(
419        context,
420        operator_applied_to(context, shape, integration_variable, builders::one(context)),
421        builders::one(context),
422    );
423    PartialDifferentialEquationProvedIdentity {
424        verdict: verification_verdict(prove_residual_zero(context, &residual)),
425        identity: PartialDifferentialEquationFamilyIdentity::KernelMassIsOne,
426        residual,
427    }
428}

430fn prove_operator_reproduces_a_polynomial_datum(
431    context: &Context,
432    shape: &PartialDifferentialEquationDiffusionShape,
433    integration_variable: SymbolIdentifier,
434) -> PartialDifferentialEquationProvedIdentity {
435    // The operator applied to the datum `s^2` must return `x^2 + 2*a*t`, the known heat solution
436    // with that initial trace. The datum grows without bound and the solution never decays, so
437    // this is the trace check a rapidly decaying test datum cannot perform: an operator whose
438    // convergence had been secured by a decay assumption would not survive it.
439    let source = context
440        .interner
441        .intern(ExpressionKind::Symbol(integration_variable));
442    let datum = builders::power(context, source, builders::integer_value(context, 2));
443    let spatial = context
444        .interner
445        .intern(ExpressionKind::Symbol(shape.spatial_coordinate));
446    let evolution = context
447        .interner
448        .intern(ExpressionKind::Symbol(shape.evolution_coordinate));
449    let expected = builders::sum(
450        context,
451        vec![
452            builders::power(context, spatial, builders::integer_value(context, 2)),
453            builders::product(
454                context,
455                vec![
456                    builders::integer_value(context, 2),
457                    shape.diffusion_coefficient.clone(),
458                    evolution,
459                ],
460            ),
461        ],
462    );
463    let residual = builders::subtract(
464        context,
465        operator_applied_to(context, shape, integration_variable, datum),
466        expected,
467    );
468    PartialDifferentialEquationProvedIdentity {
469        verdict: verification_verdict(prove_residual_zero(context, &residual)),
470        identity: PartialDifferentialEquationFamilyIdentity::OperatorReproducesAPolynomialDatum,
471        residual,
472    }
473}

475/// Applies the whole-line solution operator to a concrete datum expression.
476pub fn operator_applied_to(
477    context: &Context,
478    shape: &PartialDifferentialEquationDiffusionShape,
479    integration_variable: SymbolIdentifier,
480    datum: ExpressionReference,
481) -> ExpressionReference {
482    let spatial = context
483        .interner
484        .intern(ExpressionKind::Symbol(shape.spatial_coordinate));
485    let source = context
486        .interner
487        .intern(ExpressionKind::Symbol(integration_variable));
488    let displacement = builders::subtract(context, spatial, source);
489    let integrand = builders::product(context, vec![kernel(context, shape, displacement), datum]);
490    let infinity = context
491        .interner
492        .intern(ExpressionKind::Constant(BuiltinConstant::Infinity));
493    crate::integrate::definite::integrate_definite(
494        context,
495        &integrand,
496        integration_variable,
497        builders::negate(context, Arc::clone(&infinity)),
498        infinity,
499    )
