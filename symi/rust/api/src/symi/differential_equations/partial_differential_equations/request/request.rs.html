1// Executable request semantics for scalar partial differential equation solving.
2//
3// A partial-equation request states more than its residual. Wester's polar problem is not "solve
4// the Laplace equation": it asks for the family on a disk, regular at the origin and periodic in
5// the angle, and those two words are what make the answer the constant mode together with every
6// positive integer cosine and sine mode rather than an arbitrary separated product. A request
7// that carries the geometry as an untagged string beside the problem lets a solver return a
8// family that is correct on a punctured annulus and be credited with the disk problem.
9//
10// The scope, the geometry, and the regularity are therefore part of the validated request, and a
11// method that cannot honour the declared geometry declines it rather than answering the
12// unrestricted problem under the same name.
13

17/// Public native enumeration named `partial_differential_equation_geometric_domain`.
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
19pub enum PartialDifferentialEquationGeometricDomain {
20    /// The whole coordinate space of the declared variables, with no boundary.
21    WholeSpace,
22    /// The open disk whose radial coordinate is the first declared variable and whose angular
23    /// coordinate is the second, with the origin an interior point of the region.
24    Disk,
25    /// An evolution chart: the first declared variable is a spatial coordinate ranging over the
26    /// whole real line and the second is a strictly positive evolution coordinate. Declaring this
27    /// is what lets a transform route know that the spatial axis is the whole line; inferring it
28    /// from a variable being spelled `x` would be guessing the geometry from a name.
29    RealLineEvolution,
30}
31
32impl PartialDifferentialEquationGeometricDomain {
33    /// Returns the serialized name of the geometric domain.
34    pub fn name(self) -> &'static str {
35        match self {
36            PartialDifferentialEquationGeometricDomain::WholeSpace => "whole_space",
37            PartialDifferentialEquationGeometricDomain::Disk => "disk",
38            PartialDifferentialEquationGeometricDomain::RealLineEvolution => "real_line_evolution",
39        }
40    }
41
42    /// Parses a serialized geometric-domain name.
43    pub fn from_name(name: &str) -> Option<Self> {
44        match name {
45            "whole_space" => Some(PartialDifferentialEquationGeometricDomain::WholeSpace),
46            "disk" => Some(PartialDifferentialEquationGeometricDomain::Disk),
47            "real_line_evolution" => {
48                Some(PartialDifferentialEquationGeometricDomain::RealLineEvolution)
49            }
50            _ => None,
51        }
52    }
53
54    /// Returns the number of independent variables the geometry is stated in.
55    pub fn required_variable_count(self) -> Option<usize> {
56        match self {
57            PartialDifferentialEquationGeometricDomain::WholeSpace => None,
58            PartialDifferentialEquationGeometricDomain::Disk => Some(2),
59            PartialDifferentialEquationGeometricDomain::RealLineEvolution => Some(2),
60        }
61    }
62}

97/// Public native enumeration named `partial_differential_equation_solution_class`.
98///
99/// The analytic class of solutions the answer has to cover. Scope says how much of a solution set
100/// is wanted; this says which solution set. The heat equation has nonzero solutions with zero
101/// initial data once growth is unrestricted, so a representation complete among solutions of
102/// polynomial growth is not an answer to the unrestricted request, and only a declared class
103/// makes that difference decidable instead of rhetorical.
104#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
105pub enum PartialDifferentialEquationSolutionClass {
106    /// Every solution of the equation on the declared geometry, with no growth restriction.
107    Unrestricted,
108    /// Solutions of at most polynomial growth in the spatial coordinate.
109    PolynomialGrowth,
110}
111

115        match self {

117            PartialDifferentialEquationSolutionClass::PolynomialGrowth => "polynomial_growth",
118        }
119    }
120
121    /// Parses a serialized solution-class name.
122    pub fn from_name(name: &str) -> Option<Self> {
123        match name {
124            "unrestricted" => Some(PartialDifferentialEquationSolutionClass::Unrestricted),
125            "polynomial_growth" => Some(PartialDifferentialEquationSolutionClass::PolynomialGrowth),
126            _ => None,
127        }

140/// Public native data structure named `partial_differential_equation_request`.
141#[derive(Debug, Clone)]
142pub struct PartialDifferentialEquationRequest {
143    problem: PartialDifferentialEquationProblem,
144    scope: DifferentialEquationSolutionScope,
145    geometric_domain: PartialDifferentialEquationGeometricDomain,
146    regularity: PartialDifferentialEquationRegularity,
147    solution_class: PartialDifferentialEquationSolutionClass,
148}
149
150impl PartialDifferentialEquationRequest {
151    /// Constructs a validated request whose geometry and regularity are mutually consistent.
152    pub fn new(
153        problem: PartialDifferentialEquationProblem,
154        scope: DifferentialEquationSolutionScope,
155        geometric_domain: PartialDifferentialEquationGeometricDomain,
156        regularity: PartialDifferentialEquationRegularity,
157        solution_class: PartialDifferentialEquationSolutionClass,
158    ) -> Result<Self, PartialDifferentialEquationRequestError> {
159        if let Some(required) = geometric_domain.required_variable_count()
160            && problem.independent_variables().len() != required
161        {
162            return Err(PartialDifferentialEquationRequestError::GeometryVariableCountMismatch);
163        }
164        if regularity == PartialDifferentialEquationRegularity::OriginRegularAngularPeriodic
165            && geometric_domain != PartialDifferentialEquationGeometricDomain::Disk
166        {
167            return Err(PartialDifferentialEquationRequestError::RegularityDoesNotSuitTheGeometry);
168        }
169        Ok(Self {
170            problem,
171            scope,
172            geometric_domain,
173            regularity,
174            solution_class,
175        })
176    }
177
178    /// Returns the validated governing problem.
179    pub fn problem(&self) -> &PartialDifferentialEquationProblem {
180        &self.problem
181    }
182
183    /// Returns the family scope the answer must satisfy.
184    pub fn scope(&self) -> DifferentialEquationSolutionScope {
185        self.scope
186    }
187
188    /// Returns the geometry the family is requested on.
189    pub fn geometric_domain(&self) -> PartialDifferentialEquationGeometricDomain {
190        self.geometric_domain
191    }
192
193    /// Returns the regularity the family must satisfy on that geometry.
194    pub fn regularity(&self) -> PartialDifferentialEquationRegularity {
195        self.regularity
196    }
197
198    /// Returns the analytic class of solutions the answer has to cover.
199    pub fn solution_class(&self) -> PartialDifferentialEquationSolutionClass {
200        self.solution_class
201    }
202
203    /// Reports whether the request restricts the family beyond the residual itself.
204    pub fn restricts_the_family(&self) -> bool {
205        self.geometric_domain != PartialDifferentialEquationGeometricDomain::WholeSpace
206            || self.regularity != PartialDifferentialEquationRegularity::Unrestricted
207    }
208}

210/// Public native enumeration named `partial_differential_equation_request_decline`.
211///
212/// Why a validated request was not attempted, or why its answer does not satisfy it. These are
213/// separate from the method declines of the dispatcher: the dispatcher reports that a method did
214/// not close, and these report that the request itself was never in the supported class, so a
215/// correct answer to a weaker question is not reported under the stronger one.
216#[derive(Debug, Clone, PartialEq, Eq)]
217pub enum PartialDifferentialEquationRequestDecline {
218    /// No implemented method constructs a family on the declared geometry.
219    DeclaredGeometryNotSupported,
220    /// No implemented method constructs a family satisfying the declared regularity.
221    DeclaredRegularityNotSupported,
222    /// The request asked for the complete family and the family returned carries no completeness
223    /// theorem: a verified residual proves membership in the solution set, never coverage of it.
224    CompleteFamilyNotEstablished,
225    /// A complete family was constructed, but its completeness theorem holds over a narrower
226    /// class of solutions than the request asked for. This is the honest verdict for the
227    /// unrestricted heat request: the initial-trace representation is complete among solutions of
228    /// polynomial growth and the equation has others, so answering the wider question with the
229    /// narrower theorem would be a false completeness claim rather than a useful approximation.
230    CompletenessHoldsOnlyOnANarrowerSolutionClass,
231}

233/// Solves a validated partial differential equation request, honouring its declared geometry.
234pub fn solve_partial_differential_equation_for_request(
235    context: &crate::interner::Context,
236    request: &PartialDifferentialEquationRequest,
237    method: crate::differential_equations::partial_differential_equations::PartialDifferentialEquationMethod,
238) -> Result<
239    crate::differential_equations::partial_differential_equations::PartialDifferentialEquationSolveResult,
240    PartialDifferentialEquationRequestDecline,
241>{
242    // Geometry and regularity are checked before the dispatcher runs, because the dispatcher has
243    // no way to express them: handing back a family correct on the unrestricted coordinate space
244    // for a disk problem regular at the origin would answer a different question. Completeness is
245    // checked after, because a solved result is exactly where an unproved coverage claim would
246    // otherwise enter, and it is checked against the request's declared solution class rather
247    // than against the mere presence of a certificate.
248    if request.geometric_domain() == PartialDifferentialEquationGeometricDomain::Disk
249        && request.regularity() == PartialDifferentialEquationRegularity::OriginRegularAngularPeriodic
250        && method == crate::differential_equations::partial_differential_equations::PartialDifferentialEquationMethod::Automatic
251    {
252        return crate::differential_equations::partial_differential_equations::solve_regular_polar_disk_harmonic_family(context, request)
253            .map(|solution| crate::differential_equations::partial_differential_equations::PartialDifferentialEquationSolveResult {
254                verdict: crate::differential_equations::partial_differential_equations::PartialDifferentialEquationSolveVerdict::Solved,
255                solution: Some(solution.family),
256                classification: crate::differential_equations::partial_differential_equations::classify_partial_differential_equation(context, request.problem()).ok(),
257                method: Some(crate::differential_equations::partial_differential_equations::PartialDifferentialEquationMethod::Separation),
258                verification: Some(solution.verification),
259                decline: None,
260                provenance: None,
261                portfolio_declines: Vec::new(),
262            })
263            .map_err(|_| PartialDifferentialEquationRequestDecline::DeclaredRegularityNotSupported);
264    }
265    if !geometry_is_supported(request.geometric_domain()) {
266        return Err(PartialDifferentialEquationRequestDecline::DeclaredGeometryNotSupported);
267    }
268    if request.regularity() != PartialDifferentialEquationRegularity::Unrestricted {
269        return Err(PartialDifferentialEquationRequestDecline::DeclaredRegularityNotSupported);
270    }
271    let result = match partial_differential_equation_transform_plan_for_request(context, request) {
272        Some(plan) => crate::differential_equations::partial_differential_equations::solve_partial_differential_equation_with_transform_plan(
273            context,
274            request.problem(),
275            method,
276            &plan,
277        ),
278        None => crate::differential_equations::partial_differential_equations::solve_partial_differential_equation(
279            context,
280            request.problem(),
281            method,
282        ),
283    };
284    if request.scope() != DifferentialEquationSolutionScope::General
285        || result.verdict
286            != crate::differential_equations::partial_differential_equations::PartialDifferentialEquationSolveVerdict::Solved
287    {
288        return Ok(result);
289    }
290    let Some(family) = result.solution.as_ref() else {
291        return Err(PartialDifferentialEquationRequestDecline::CompleteFamilyNotEstablished);
292    };
293    if !family.establishes_completeness() {
294        return Err(PartialDifferentialEquationRequestDecline::CompleteFamilyNotEstablished);
295    }
296    let certificate = family
297        .semantics
298        .completeness
299        .as_ref()
300        .ok_or(PartialDifferentialEquationRequestDecline::CompleteFamilyNotEstablished)?;
301    if !completeness_covers_the_request(certificate.theorem, request) {
302        return Err(
303            PartialDifferentialEquationRequestDecline::CompletenessHoldsOnlyOnANarrowerSolutionClass,
304        );
305    }
306    Ok(result)
307}

309fn geometry_is_supported(geometry: PartialDifferentialEquationGeometricDomain) -> bool {
310    match geometry {
311        PartialDifferentialEquationGeometricDomain::WholeSpace
312        | PartialDifferentialEquationGeometricDomain::RealLineEvolution => true,
313        PartialDifferentialEquationGeometricDomain::Disk => false,
314    }
315}
316
317fn completeness_covers_the_request(
318    theorem: crate::differential_equations::partial_differential_equations::PartialDifferentialEquationCompletenessTheorem,
319    request: &PartialDifferentialEquationRequest,
320) -> bool {
321    // Match the theorem against what was asked for. The repeated-integration theorem covers every
322    // solution on a connected product chart and so answers either solution class; the initial-
323    // trace representation is a theorem about one class and answers only a request stated over
324    // that class.
325    use crate::differential_equations::partial_differential_equations::PartialDifferentialEquationCompletenessTheorem as Theorem;
326    match theorem {
327        Theorem::VanishingIteratedDerivativeOnAProductChart => matches!(
328            request.geometric_domain(),
329            PartialDifferentialEquationGeometricDomain::WholeSpace
330                | PartialDifferentialEquationGeometricDomain::RealLineEvolution
331        ),
332        Theorem::InitialTraceRepresentationInTheAdmissibleClass => {
333            request.solution_class() == PartialDifferentialEquationSolutionClass::PolynomialGrowth
334        }
335        Theorem::RegularHarmonicExpansionOnADisk => {
336            request.geometric_domain() == PartialDifferentialEquationGeometricDomain::Disk
337                && request.regularity()
338                    == PartialDifferentialEquationRegularity::OriginRegularAngularPeriodic
339        }
340    }
341}

343/// Builds the transform plan a request's declared geometry justifies, or none when it declares
344/// no transformable axis.
345pub fn partial_differential_equation_transform_plan_for_request(
346    context: &crate::interner::Context,
347    request: &PartialDifferentialEquationRequest,
348) -> Option<crate::differential_equations::partial_differential_equations::PartialDifferentialEquationTransformPlan>
349{
350    // Request-driven transform geometry. A declared evolution chart says that the first
351    // coordinate is the whole real line, which is exactly the hypothesis a Fourier axis needs, so
352    // the plan is built from the declaration. An unspecified geometry builds no plan at all: the
353    // transform route would otherwise acquire a whole-line hypothesis from the fact that someone
354    // named a variable `x`.
355    use crate::differential_equations::partial_differential_equations::{
356        PartialDifferentialEquationTransformAxisKind,
357        PartialDifferentialEquationTransformAxisRequest,
358        build_partial_differential_equation_transform_plan,
359    };
360    if request.geometric_domain() != PartialDifferentialEquationGeometricDomain::RealLineEvolution {
361        return None;
362    }
363    let spatial_coordinate = *request.problem().independent_variables().first()?;
364    build_partial_differential_equation_transform_plan(
365        context,
366        request.problem(),
367        &[PartialDifferentialEquationTransformAxisRequest {
368            source_variable: spatial_coordinate,
369            kind: PartialDifferentialEquationTransformAxisKind::FourierWholeRealLine,
370            transform_variable: None,
371        }],
372    )
373    .ok()
