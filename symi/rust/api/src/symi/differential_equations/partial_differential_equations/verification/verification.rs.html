54    pub verdict: PartialDifferentialEquationVerificationVerdict,

59    /// Stores boundary-condition residual reports in insertion order.
60    pub boundary_condition_residuals: Vec<PartialDifferentialEquationVerificationResidual>,

62    pub proof_obligations: Vec<PartialDifferentialEquationProofObligation>,
63    /// Stores the branch checked after exact implicit isolation.
64    pub explicit_branch: Option<ExpressionReference>,
65    /// Stores why exact verification could not conclude, when applicable.
66    pub inconclusive_reason: Option<PartialDifferentialEquationVerificationInconclusiveReason>,
67    /// Stores whether every arbitrary datum stayed inside the dependence its family declared.
68    ///
69    /// This is a structural refutation rather than a residual one: `g(x, y) + h(y)` has a
70    /// vanishing mixed derivative for no reason the equation supplies, and a family that declared
71    /// `g` a function of `x` alone is refuted by the occurrence, not by an identity.
72    pub declared_dependence_respected: bool,
73}

75/// Allocates a collision-free generated arbitrary function in one mathematical context.
76pub fn allocate_partial_differential_equation_arbitrary_function(
77    context: &Context,
78    preferred_name: &str,
79    arity: usize,
80) -> PartialDifferentialEquationArbitraryFunction {
81    let existing_names: BTreeSet<String> = (0..context.symbol_table.number_of_symbols())
82        .map(|number| {
83            context
84                .symbol_table
85                .name(SymbolIdentifier(number as u32))
86                .to_string()
87        })
88        .collect();
89    let mut suffix = 0usize;
90    let name = loop {
91        let candidate = if suffix == 0 {
92            preferred_name.to_string()
93        } else {
94            format!("{preferred_name}_{suffix}")
95        };
96        if !existing_names.contains(&candidate) {
97            break candidate;
98        }
99        suffix += 1;
100    };
101    PartialDifferentialEquationArbitraryFunction::unrestricted(
102        FunctionIdentifier::User(context.symbol_table.intern_symbol(&name)),
103        arity,
104    )
105}

134/// Reports every arbitrary datum of a family whose occurrences reach a forbidden coordinate.
135pub fn partial_differential_equation_dependence_violations(
136    candidate: &PartialDifferentialEquationSolutionFamily,
137) -> Vec<PartialDifferentialEquationArbitraryFunction> {
138    let expression = candidate.candidate_expression();
139    candidate
140        .arbitrary_functions
141        .iter()
142        .filter(|datum| {
143            let PartialDifferentialEquationPermittedDependence::Coordinates(permitted) =
144                &datum.permitted_dependence
145            else {
146                return false;
147            };
148            let mut arguments = Vec::new();
149            collect_call_arguments(expression, datum.function, &mut arguments);
150            arguments.iter().any(|argument| {
151                let mut reached = BTreeSet::new();
152                collect_free_symbols(argument, &mut reached);
153                reached.iter().any(|symbol| !permitted.contains(symbol))
154            })
155        })
156        .cloned()
157        .collect()
158}

160fn collect_call_arguments(
161    expression: &ExpressionReference,
162    function: FunctionIdentifier,
163    arguments: &mut Vec<ExpressionReference>,
164) {
165    match &expression.kind {
166        ExpressionKind::FunctionCall(called, called_arguments)
167        | ExpressionKind::Derivative(called, _, called_arguments)
168            if *called == function =>
169        {
170            arguments.extend(called_arguments.iter().cloned());
171        }
172        _ => {}
173    }
174    let _ =
175        crate::differential_equations::partial_differential_equations::problem::traverse_children(
176            expression,
177            |child| {
178                collect_call_arguments(child, function, arguments);
179                Ok(())
180            },
181        );
182}

201/// Verifies a PDE candidate by independent exact substitution and zero reasoning.
202pub fn verify_partial_differential_equation_solution(
203    context: &Context,
204    problem: &PartialDifferentialEquationProblem,
205    candidate: &PartialDifferentialEquationSolutionFamily,
206) -> PartialDifferentialEquationVerificationReport {
207    // The declared dependence is checked before any residual is reduced, because a datum that
208    // reaches a coordinate its family forbids makes the family a different mathematical object
209    // even when the residual happens to vanish.
210    let mut report = match &candidate.form {
211        PartialDifferentialEquationSolutionForm::Explicit(solution) => {
212            verify_explicit_solution(context, problem, candidate, solution, None)
213        }
214        PartialDifferentialEquationSolutionForm::Implicit(residual) => {
215            verify_implicit_solution(context, problem, candidate, residual)
216        }
217    };
218    if !partial_differential_equation_dependence_violations(candidate).is_empty() {
219        report.declared_dependence_respected = false;
220        report.verdict = PartialDifferentialEquationVerificationVerdict::Rejected;
221        report.inconclusive_reason = None;
222    }
223    report
224}

226fn verify_implicit_solution(
227    context: &Context,
228    problem: &PartialDifferentialEquationProblem,
229    candidate: &PartialDifferentialEquationSolutionFamily,
230    residual: &ExpressionReference,
231) -> PartialDifferentialEquationVerificationReport {
232    if contains_dependent_derivative(residual, problem) {
233        return inconclusive_report(
234            context,
235            candidate,
236            PartialDifferentialEquationVerificationInconclusiveReason::ImplicitSolutionContainsDerivative,
237        );
238    }
239    let Some(solution) = isolate_unit_dependent_call(context, residual, problem) else {
240        return inconclusive_report(
241            context,
242            candidate,
243            PartialDifferentialEquationVerificationInconclusiveReason::ImplicitSolutionNotIsolatable,
244        );
245    };
246    verify_explicit_solution(
247        context,
248        problem,
249        candidate,
250        &solution,
251        Some(solution.clone()),
252    )
253}

255fn isolate_unit_dependent_call(
256    context: &Context,
257    residual: &ExpressionReference,
258    problem: &PartialDifferentialEquationProblem,
259) -> Option<ExpressionReference> {
260    let dependent_call = dependent_call(context, problem);
261    let without_positive_call =
262        builders::subtract(context, residual.clone(), dependent_call.clone());
263    if !contains_dependent_occurrence(&without_positive_call, problem) {
264        return Some(builders::negate(context, without_positive_call));
265    }
266    let without_negative_call = builders::sum(context, vec![residual.clone(), dependent_call]);
267    if !contains_dependent_occurrence(&without_negative_call, problem) {
268        return Some(without_negative_call);
269    }
270    None
271}

273fn verify_explicit_solution(
274    context: &Context,
275    problem: &PartialDifferentialEquationProblem,
276    candidate: &PartialDifferentialEquationSolutionFamily,
277    solution: &ExpressionReference,
278    explicit_branch: Option<ExpressionReference>,
279) -> PartialDifferentialEquationVerificationReport {
280    let mut derivative_cache = BTreeMap::new();
281    let governing_residual = reduce_residual(
282        context,
283        problem,
284        solution,
285        problem.equation(),
286        &mut derivative_cache,
287    );
288    let initial_condition_residuals: Vec<_> = problem
289        .initial_conditions()
290        .iter()
291        .map(|condition| {
292            reduce_residual(context, problem, solution, condition, &mut derivative_cache)
293        })
294        .collect();
295    let boundary_condition_residuals: Vec<_> = problem
296        .boundary_conditions()
297        .iter()
298        .map(|condition| {
299            reduce_residual(context, problem, solution, condition, &mut derivative_cache)
300        })
301        .collect();
302    let all_residuals = std::iter::once(&governing_residual)
303        .chain(initial_condition_residuals.iter())
304        .chain(boundary_condition_residuals.iter());
305    let verdict = if all_residuals.clone().any(|residual| {
306        residual.verdict == PartialDifferentialEquationVerificationVerdict::Rejected
307    }) {
308        PartialDifferentialEquationVerificationVerdict::Rejected
309    } else if all_residuals.clone().any(|residual| {
310        residual.verdict == PartialDifferentialEquationVerificationVerdict::Inconclusive
311    }) {
312        PartialDifferentialEquationVerificationVerdict::Inconclusive
313    } else {
314        PartialDifferentialEquationVerificationVerdict::Verified
315    };
316    PartialDifferentialEquationVerificationReport {
317        verdict,
318        governing_residual,
319        initial_condition_residuals,
320        boundary_condition_residuals,
321        proof_obligations: candidate.proof_obligations.clone(),
322        explicit_branch,
323        declared_dependence_respected: true,
324        inconclusive_reason: (verdict
325            == PartialDifferentialEquationVerificationVerdict::Inconclusive)
326            .then_some(
327                PartialDifferentialEquationVerificationInconclusiveReason::ResidualZeroNotProved,
328            ),
329    }
330}

332fn reduce_residual(
333    context: &Context,
334    problem: &PartialDifferentialEquationProblem,
335    solution: &ExpressionReference,
336    residual: &ExpressionReference,
337    derivative_cache: &mut BTreeMap<Vec<usize>, ExpressionReference>,
338) -> PartialDifferentialEquationVerificationResidual {
339    let mut occurrences = Vec::new();
340    collect_dependent_occurrences(residual, problem, &mut occurrences);
341    let mut mapping = SubstitutionMap::new();
342    for occurrence in occurrences {
343        let Some((orders, arguments)) = dependent_occurrence_data(&occurrence, problem) else {
344            continue;
345        };
346        let derivative = derivative_cache
347            .entry(orders.clone())
348            .or_insert_with(|| {
349                derivative_for_orders(context, solution, problem.independent_variables(), &orders)
350            })
351            .clone();
352        let mut argument_mapping = SubstitutionMap::new();
353        for (variable, argument) in problem.independent_variables().iter().zip(arguments) {
354            argument_mapping.insert(
355                context.interner.intern(ExpressionKind::Symbol(*variable)),
356                argument,
357            );
358        }
359        mapping.insert(
360            occurrence,
361            substitute(context, &derivative, &argument_mapping),
362        );
363    }
364    let reduced = substitute(context, residual, &mapping);
365    PartialDifferentialEquationVerificationResidual {
366        verdict: verification_verdict(prove_residual_zero(context, &reduced)),
367        residual: reduced,
368    }
369}

371fn derivative_for_orders(
372    context: &Context,
373    solution: &ExpressionReference,
374    independent_variables: &[SymbolIdentifier],
375    orders: &[usize],
376) -> ExpressionReference {
377    let derivative_orders: Vec<_> = independent_variables
378        .iter()
379        .copied()
380        .zip(orders.iter().copied())
381        .collect();
382    partial_derivative(context, solution, &derivative_orders)
383}

403pub(crate) fn dependent_occurrence_data(
404    expression: &ExpressionReference,
405    problem: &PartialDifferentialEquationProblem,
406) -> Option<(Vec<usize>, Vec<ExpressionReference>)> {
407    match &expression.kind {
408        ExpressionKind::FunctionCall(function, arguments)
409            if *function == problem.dependent_function() =>
410        {
411            (arguments.len() == problem.independent_variables().len())
412                .then(|| (vec![0; arguments.len()], arguments.clone()))
413        }
414        ExpressionKind::Derivative(function, orders, arguments)
415            if *function == problem.dependent_function() =>
416        {
417            (orders.len() == arguments.len()
418                && arguments.len() == problem.independent_variables().len())
419            .then(|| (orders.clone(), arguments.clone()))
420        }
421        _ => None,
422    }
423}

469pub(crate) fn prove_residual_zero(
470    context: &Context,
471    residual: &ExpressionReference,
472) -> ZeroEquivalenceVerdict {
473    let direct = zero_equivalent(context, residual, None);
474    if direct != ZeroEquivalenceVerdict::Unknown {
475        return direct;
476    }
477    let cancelled = crate::simplify::rational::cancel(context, residual);
478    let cancelled_verdict = zero_equivalent(context, &cancelled, None);
479    if cancelled_verdict != ZeroEquivalenceVerdict::Unknown {
480        return cancelled_verdict;
481    }
482    let expanded = crate::expand::expand(context, &cancelled);
483    let expanded_verdict = zero_equivalent(context, &expanded, None);
484    if expanded_verdict != ZeroEquivalenceVerdict::Unknown {
485        return expanded_verdict;
486    }
487    let combined = crate::simplify::rational::together(context, &expanded);
488    let expanded_combined = crate::expand::expand(context, &combined);
489    let simplified = crate::simplify::simplify(context, expanded_combined);
490    let verdict = zero_equivalent(context, &simplified, None);
491    if verdict != ZeroEquivalenceVerdict::Unknown {
492        return verdict;
493    }
494    if residual_vanishes_by_integral_linearity(context, &simplified) {
495        return ZeroEquivalenceVerdict::TrueValue;
496    }
497    ZeroEquivalenceVerdict::Unknown
498}
499
500fn residual_vanishes_by_integral_linearity(
501    context: &Context,
502    residual: &ExpressionReference,
503) -> bool {
504    // Linearity of the definite integral, used in one direction only. A solution built as an
505    // integral operator differentiates into several integrals over the same variable and the same
506    // bounds with different scalar factors outside, and no amount of expanding compares their
507    // integrands. Folding each factor into its integrand and summing turns the comparison into a
508    // pointwise identity the zero prover can decide, and a pointwise zero integrand makes the
509    // integral zero.
510    //
511    // The converse is deliberately not used: a nonzero integrand does not make the integral
512    // nonzero, so this route can establish a vanishing residual and can never refute one.
513    let mut groups: Vec<(
514        SymbolIdentifier,
515        ExpressionReference,
516        ExpressionReference,
517        Vec<ExpressionReference>,
518    )> = Vec::new();
519    let mut remaining = Vec::new();
520    for term in sum_terms(residual) {
521        match split_scalar_multiple_of_definite_integral(context, &term) {
522            Some((coefficient, integrand, variable, lower, upper)) => {
523                let scaled = builders::product(context, vec![coefficient, integrand]);
524                match groups
525                    .iter_mut()
526                    .find(|group| group.0 == variable && group.1 == lower && group.2 == upper)
527                {
528                    Some(group) => group.3.push(scaled),
529                    None => groups.push((variable, lower, upper, vec![scaled])),
530                }
531            }
532            None => remaining.push(term),
533        }
534    }
535    if groups.is_empty() {
536        return false;
537    }
538    let remaining_sum = builders::sum(context, remaining);
539    if !reduces_to_zero(context, &remaining_sum) {
540        return false;
541    }
542    groups.into_iter().all(|(_, _, _, integrands)| {
543        let integrand = builders::sum(context, integrands);
544        reduces_to_zero(context, &integrand)
545    })
546}

548fn reduces_to_zero(context: &Context, expression: &ExpressionReference) -> bool {
549    let cancelled = crate::simplify::rational::cancel(context, expression);
550    let expanded = crate::expand::expand(context, &cancelled);
551    let combined = crate::simplify::rational::together(context, &expanded);
552    let simplified = crate::simplify::simplify(context, crate::expand::expand(context, &combined));
553    zero_equivalent(context, &simplified, None) == ZeroEquivalenceVerdict::TrueValue
554}

556fn sum_terms(expression: &ExpressionReference) -> Vec<ExpressionReference> {
557    match &expression.kind {
558        ExpressionKind::Sum(terms) => terms.clone(),
559        _ => vec![expression.clone()],
560    }
561}

563fn split_scalar_multiple_of_definite_integral(
564    context: &Context,
565    term: &ExpressionReference,
566) -> Option<(
567    ExpressionReference,
568    ExpressionReference,
569    SymbolIdentifier,
570    ExpressionReference,
571    ExpressionReference,
572)> {
573    // A term joins a group when it is exactly one definite integral multiplied by factors free of
574    // that integral's bound variable. A factor involving the bound variable cannot be carried
575    // inside, and a term carrying two integrals has no single integrand, so both are left where
576    // they are rather than rewritten into something they do not equal.
577    match &term.kind {
578        ExpressionKind::Integral(integrand, variable, Some(lower), Some(upper)) => Some((
579            builders::one(context),
580            integrand.clone(),
581            *variable,
582            lower.clone(),
583            upper.clone(),
584        )),
585        ExpressionKind::Product(factors) => {
586            let mut integral = None;
587            let mut scalars = Vec::new();
588            for factor in factors {
589                match &factor.kind {
590                    ExpressionKind::Integral(integrand, variable, Some(lower), Some(upper)) => {
591                        if integral.is_some() {
592                            return None;
593                        }
594                        integral =
595                            Some((integrand.clone(), *variable, lower.clone(), upper.clone()));
596                    }
597                    _ => scalars.push(factor.clone()),
598                }
599            }
600            let (integrand, variable, lower, upper) = integral?;
601            if scalars
602                .iter()
603                .any(|factor| expression_involves_symbol(factor, variable))
604            {
605                return None;
606            }
607            Some((
608                builders::product(context, scalars),
609                integrand,
610                variable,
611                lower,
612                upper,
613            ))
614        }
615        _ => None,
616    }
617}

619fn expression_involves_symbol(expression: &ExpressionReference, symbol: SymbolIdentifier) -> bool {
620    if matches!(expression.kind, ExpressionKind::Symbol(found) if found == symbol) {
621        return true;
622    }
623    let mut found = false;
624    let _ =
625        crate::differential_equations::partial_differential_equations::problem::traverse_children(
626            expression,
627            |child| {
628                found |= expression_involves_symbol(child, symbol);
629                Ok(())
630            },
631        );
632    found
633}

635pub(crate) fn verification_verdict(
636    verdict: ZeroEquivalenceVerdict,
637) -> PartialDifferentialEquationVerificationVerdict {
638    match verdict {
639        ZeroEquivalenceVerdict::TrueValue => {
640            PartialDifferentialEquationVerificationVerdict::Verified
641        }
642        ZeroEquivalenceVerdict::FalseValue => {
643            PartialDifferentialEquationVerificationVerdict::Rejected
644        }
645        ZeroEquivalenceVerdict::Unknown => {
646            PartialDifferentialEquationVerificationVerdict::Inconclusive
647        }
648    }

651fn inconclusive_report(
652    context: &Context,
653    candidate: &PartialDifferentialEquationSolutionFamily,
654    reason: PartialDifferentialEquationVerificationInconclusiveReason,
655) -> PartialDifferentialEquationVerificationReport {
656    let residual = PartialDifferentialEquationVerificationResidual {
657        residual: context
658            .interner
659            .intern(ExpressionKind::Constant(BuiltinConstant::Undefined)),
660        verdict: PartialDifferentialEquationVerificationVerdict::Inconclusive,
661    };
662    PartialDifferentialEquationVerificationReport {
663        verdict: PartialDifferentialEquationVerificationVerdict::Inconclusive,
664        governing_residual: residual,
665        initial_condition_residuals: Vec::new(),
666        boundary_condition_residuals: Vec::new(),
667        proof_obligations: candidate.proof_obligations.clone(),
668        explicit_branch: None,
669        declared_dependence_respected: true,
670        inconclusive_reason: Some(reason),
671    }
