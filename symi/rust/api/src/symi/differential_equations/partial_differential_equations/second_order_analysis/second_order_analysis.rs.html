17    Elliptic,
18    /// Represents the `hyperbolic` case.
19    Hyperbolic,

21    Parabolic,

26/// Public native enumeration named `partial_differential_equation_second_order_type_verdict`.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
28pub enum PartialDifferentialEquationSecondOrderTypeVerdict {
29    /// Represents one type proved under the active assumptions.
30    Definite,
31    /// Represents an exhaustive set of proved mutually exclusive type cases.

63/// Public native data structure named `partial_differential_equation_second_order_classification`.
64#[derive(Debug, Clone)]
65pub struct PartialDifferentialEquationSecondOrderClassification {
66    /// Stores A's frozen symmetric second-order principal matrix.
67    pub principal_matrix: Matrix,
68    /// Stores the exact discriminant `b^2-a*c` for the symmetric matrix.
69    pub discriminant: ExpressionReference,
70    /// Stores whether the result is definite, conditional, or unknown.
71    pub verdict: PartialDifferentialEquationSecondOrderTypeVerdict,
72    /// Stores the one proved type for a definite result.
73    pub equation_type: Option<PartialDifferentialEquationSecondOrderType>,
74    /// Stores complete exact cases for a conditional result.
75    pub cases: Vec<PartialDifferentialEquationSecondOrderTypeCase>,
76    /// Stores obligations relevant to downstream canonical-coordinate methods.
77    pub proof_obligations: Vec<PartialDifferentialEquationProofObligation>,
78    /// Stores why an unknown result could not be completed.
79    pub decline: Option<PartialDifferentialEquationSecondOrderClassificationDecline>,
80}

82/// Classifies a two-variable linear second-order PDE over real coordinates.
83pub fn classify_partial_differential_equation_second_order(
84    context: &Context,
85    problem: &PartialDifferentialEquationProblem,
86) -> PartialDifferentialEquationSecondOrderClassification {
87    classify_partial_differential_equation_second_order_with_budget(
88        context,
89        problem,
90        DEFAULT_SECOND_ORDER_CLASSIFICATION_BUDGET,
91    )
92}

94fn classify_partial_differential_equation_second_order_with_budget(
95    context: &Context,
96    problem: &PartialDifferentialEquationProblem,
97    budget: usize,
98) -> PartialDifferentialEquationSecondOrderClassification {
99    let classification = match classify_partial_differential_equation(context, problem) {
100        Ok(classification) => classification,
101        Err(_) => return unavailable_classification(context, PartialDifferentialEquationSecondOrderClassificationDecline::NotTwoVariableSecondOrderLinear),
102    };
103    let Some(principal_part) = classification.second_order_principal_part else {
104        return unavailable_classification(
105            context,
106            PartialDifferentialEquationSecondOrderClassificationDecline::PrincipalPartNotAvailable,
107        );
108    };
109    if classification.order != 2
110        || classification.linearity != PartialDifferentialEquationLinearity::Linear
111        || problem.independent_variables().len() != 2
112    {
113        return unknown_classification(
114            principal_part.matrix,
115            builders::zero(context),
116            PartialDifferentialEquationSecondOrderClassificationDecline::NotTwoVariableSecondOrderLinear,
117        );
118    }
119    let source_variables = problem.independent_variables();
120    let source_coordinates = source_variables
121        .iter()
122        .map(|identifier| {
123            context
124                .interner
125                .intern(crate::expression::ExpressionKind::Symbol(*identifier))
126        })
127        .collect::<Vec<_>>();
128    let coordinate_reality =
129        AssumptionFormula::and(source_coordinates.into_iter().map(crate::assumptions::real));
130    match context.with_temporary_assumptions(coordinate_reality, |scoped_context| {
131        classify_with_real_coordinates(scoped_context, principal_part.matrix, budget)
132    }) {
133        Ok(result) => result,
134        // A bounded consistency checker that gives up has not shown the operation-local real
135        // coordinate facts to be contradictory, so it is reported as an exhausted budget rather
136        // than as an inconsistency the user could act on.
137        Err(crate::assumptions::AssumptionMutationError::ResourceExhausted) => {
138            unavailable_classification(
139                context,
140                PartialDifferentialEquationSecondOrderClassificationDecline::BudgetExceeded,
141            )
142        }
143        Err(_) => unavailable_classification(
144            context,
145            PartialDifferentialEquationSecondOrderClassificationDecline::InconsistentAssumptions,
146        ),
147    }
148}

150fn classify_with_real_coordinates(
151    context: &Context,
152    principal_matrix: Matrix,
153    mut remaining_budget: usize,
154) -> PartialDifferentialEquationSecondOrderClassification {
155    let (a, b, c) =
156        principal_entries(&principal_matrix).expect("validated two-variable principal matrix");
157    for coefficient in [&a, &b, &c] {
158        if !consume_budget(&mut remaining_budget) {
159            return unknown_classification(
160                principal_matrix,
161                builders::zero(context),
162                PartialDifferentialEquationSecondOrderClassificationDecline::BudgetExceeded,
163            );
164        }
165        if crate::assumptions::queries::is_real(coefficient, &context.symbol_table)
166            != TruthValue::Yes
167        {
168            return unknown_classification(principal_matrix, builders::zero(context), PartialDifferentialEquationSecondOrderClassificationDecline::RealPrincipalCoefficientsNotProved);
169        }
170    }
171    let exponent = builders::integer(context, 2.into());
172    let squared_mixed_coefficient = builders::power(context, b.clone(), exponent);
173    let pure_coefficient_product = builders::product(context, vec![a.clone(), c.clone()]);
174    let discriminant =
175        builders::subtract(context, squared_mixed_coefficient, pure_coefficient_product);
176    if !consume_budget(&mut remaining_budget) {
177        return unknown_classification(
178            principal_matrix,
179            discriminant,
180            PartialDifferentialEquationSecondOrderClassificationDecline::BudgetExceeded,
181        );
182    }
183    if let Some(equation_type) =
184        direct_type(context, &a, &b, &c, &discriminant, &mut remaining_budget)
185    {
186        return definite_classification(principal_matrix, discriminant, equation_type);
187    }
188    conditional_classification(
189        context,
190        principal_matrix,
191        discriminant,
192        a,
193        b,
194        c,
195        remaining_budget,
196    )
197}

199fn conditional_classification(
200    context: &Context,
201    principal_matrix: Matrix,
202    discriminant: ExpressionReference,
203    a: ExpressionReference,
204    b: ExpressionReference,
205    c: ExpressionReference,
206    mut remaining_budget: usize,
207) -> PartialDifferentialEquationSecondOrderClassification {
208    let zero = builders::zero(context);
209    let conditions = vec![
210        AssumptionFormula::less(discriminant.clone(), zero.clone()),
211        AssumptionFormula::equal(discriminant.clone(), zero.clone()),
212        AssumptionFormula::less(zero, discriminant.clone()),
213    ];
214    let mut cases = Vec::new();
215    for condition in conditions {
216        if !consume_budget(&mut remaining_budget) {
217            return unknown_classification(
218                principal_matrix,
219                discriminant,
220                PartialDifferentialEquationSecondOrderClassificationDecline::BudgetExceeded,
221            );
222        }
223        if context.symbol_table.could_hold_assumptions(&condition) == TruthValue::No {
224            continue;
225        }
226        let scoped_result =
227            context.with_temporary_assumptions(condition.clone(), |scoped_context| {
228                let mut branch_budget = remaining_budget;
229                direct_type(
230                    scoped_context,
231                    &a,
232                    &b,
233                    &c,
234                    &discriminant,
235                    &mut branch_budget,
236                )
237            });
238        let Ok(Some(equation_type)) = scoped_result else {
239            return unknown_classification(
240                principal_matrix,
241                discriminant,
242                PartialDifferentialEquationSecondOrderClassificationDecline::TypeNotProved,
243            );
244        };
245        cases.push(PartialDifferentialEquationSecondOrderTypeCase {
246            condition,
247            equation_type,
248        });
249    }
250    if cases.is_empty() {
251        unknown_classification(
252            principal_matrix,
253            discriminant,
254            PartialDifferentialEquationSecondOrderClassificationDecline::InconsistentAssumptions,
255        )
256    } else {
257        PartialDifferentialEquationSecondOrderClassification {
258            principal_matrix,
259            discriminant,
260            verdict: PartialDifferentialEquationSecondOrderTypeVerdict::Conditional,
261            equation_type: None,
262            cases,
263            proof_obligations: vec![
264                PartialDifferentialEquationProofObligation::EquationTypeEvidence,
265            ],
266            decline: None,
267        }
268    }
269}

271fn direct_type(
272    context: &Context,
273    a: &ExpressionReference,
274    b: &ExpressionReference,
275    c: &ExpressionReference,
276    discriminant: &ExpressionReference,
277    remaining_budget: &mut usize,
278) -> Option<PartialDifferentialEquationSecondOrderType> {
279    let zero = builders::zero(context);
280    if assumed_negative(context, discriminant, &zero, remaining_budget) {
281        return Some(PartialDifferentialEquationSecondOrderType::Elliptic);
282    }
283    if assumed_positive(context, discriminant, &zero, remaining_budget) {
284        return Some(PartialDifferentialEquationSecondOrderType::Hyperbolic);
285    }
286    if !consume_budget(remaining_budget)
287        || context
288            .symbol_table
289            .ask_assumptions(&AssumptionFormula::equal(
290                discriminant.clone(),
291                zero.clone(),
292            ))
293            != TruthValue::Yes
294    {
295        return None;
296    }
297    let coefficients = [a, b, c];
298    for coefficient in coefficients {
299        if !consume_budget(remaining_budget) {
300            return None;
301        }
302        if context
303            .symbol_table
304            .ask_assumptions(&AssumptionFormula::not_equal(
305                coefficient.clone(),
306                zero.clone(),
307            ))
308            == TruthValue::Yes
309        {
310            return Some(PartialDifferentialEquationSecondOrderType::Parabolic);
311        }
312    }
313    let all_zero = coefficients.iter().all(|coefficient| {
314        consume_budget(remaining_budget)
315            && context
316                .symbol_table
317                .ask_assumptions(&AssumptionFormula::equal(
318                    (*coefficient).clone(),
319                    zero.clone(),
320                ))
321                == TruthValue::Yes
322    });
323    all_zero.then_some(PartialDifferentialEquationSecondOrderType::Degenerate)
324}

326fn assumed_negative(
327    context: &Context,
328    expression: &ExpressionReference,
329    zero: &ExpressionReference,
330    remaining_budget: &mut usize,
331) -> bool {
332    if consume_budget(remaining_budget)
333        && context
334            .symbol_table
335            .ask_assumptions(&AssumptionFormula::less(expression.clone(), zero.clone()))
336            == TruthValue::Yes
337    {
338        return true;
339    }
340    if !consume_budget(remaining_budget) {
341        return false;
342    }
343    let negated_expression = builders::negate(context, expression.clone());
344    context
345        .symbol_table
346        .ask_assumptions(&AssumptionFormula::less(zero.clone(), negated_expression))
347        == TruthValue::Yes
348}

350fn assumed_positive(
351    context: &Context,
352    expression: &ExpressionReference,
353    zero: &ExpressionReference,
354    remaining_budget: &mut usize,
355) -> bool {
356    if consume_budget(remaining_budget)
357        && context
358            .symbol_table
359            .ask_assumptions(&AssumptionFormula::less(zero.clone(), expression.clone()))
360            == TruthValue::Yes
361    {
362        return true;
363    }
364    if !consume_budget(remaining_budget) {
365        return false;
366    }
367    let negated_expression = builders::negate(context, expression.clone());
368    context
369        .symbol_table
370        .ask_assumptions(&AssumptionFormula::less(negated_expression, zero.clone()))
371        == TruthValue::Yes
372}

374fn principal_entries(
375    principal_matrix: &Matrix,
376) -> Option<(
377    ExpressionReference,
378    ExpressionReference,
379    ExpressionReference,
380)> {
381    if principal_matrix.shape() != (2, 2) {
382        return None;
383    }
384    Some((
385        principal_matrix.entry_unchecked(0, 0).clone(),
386        principal_matrix.entry_unchecked(0, 1).clone(),
387        principal_matrix.entry_unchecked(1, 1).clone(),
388    ))
389}
390
391fn consume_budget(remaining_budget: &mut usize) -> bool {
392    if *remaining_budget == 0 {
393        false
394    } else {
395        *remaining_budget -= 1;
396        true
397    }
398}

432fn unavailable_classification(
433    context: &Context,
434    decline: PartialDifferentialEquationSecondOrderClassificationDecline,
435) -> PartialDifferentialEquationSecondOrderClassification {
436    let zero = builders::zero(context);
437    unknown_classification(
438        crate::matrix::matrix_from_rows(vec![
439            vec![zero.clone(), zero.clone()],
440            vec![zero.clone(), zero.clone()],
441        ])
442        .expect("fixed two-by-two matrix"),
443        zero,
444        decline,
445    )
