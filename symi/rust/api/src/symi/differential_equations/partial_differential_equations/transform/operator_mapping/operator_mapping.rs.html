1// Derivative multi-index to transform-operator mapping.
2//
3// The shared one-variable transform rules recognise a derivative of an unknown
4// function of one variable. A partial differential equation jet is a multi-index
5// over several declared coordinates, and asking those rules to understand it by
6// incidental syntax matching would be a second, weaker theorem. This module
7// states the operator theorem once, over the classifier's multi-index, so that
8// the producer and the independent verifier can each apply it from the same
9// declared data.
10

24/// Public native data structure named `partial_differential_equation_transformed_unknown`.
25#[derive(Debug, Clone, PartialEq, Eq)]
26pub struct PartialDifferentialEquationTransformedUnknown {
27    /// Stores the generated one-argument function representing the transform image.
28    pub function: FunctionIdentifier,
29    /// Stores the untransformed coordinate the image still depends on.
30    pub remaining_variable: SymbolIdentifier,
31}
32
33impl PartialDifferentialEquationTransformedUnknown {
34    /// Builds the transform image differentiated to the requested order.
35    pub fn derivative_call(&self, context: &Context, order: usize) -> ExpressionReference {
36        let argument = context
37            .interner
38            .intern(ExpressionKind::Symbol(self.remaining_variable));
39        if order == 0 {
40            return builders::function_call(context, self.function, vec![argument]);
41        }
42        builders::derivative(context, self.function, vec![order], vec![argument])
43    }
44}

46/// Maps one dependent jet through two trace-free transform axes into an algebraic image.
47pub fn map_jet_through_two_transform_axes(
48    context: &Context,
49    jet: &PartialDifferentialEquationJet,
50    first_axis: &PartialDifferentialEquationTransformAxis,
51    second_axis: &PartialDifferentialEquationTransformAxis,
52    transformed_value: ExpressionReference,
53    transformed_traces: &[PartialDifferentialEquationOriginTrace],
54) -> Result<ExpressionReference, PartialDifferentialEquationTransformDecline> {
55    if jet.derivative_orders.len() != 2
56        || first_axis.source_variable_index == second_axis.source_variable_index
57    {
58        return Err(PartialDifferentialEquationTransformDecline::TransformDomainIncompatible);
59    }
60    let first_order = jet.derivative_orders[first_axis.source_variable_index];
61    let second_order = jet.derivative_orders[second_axis.source_variable_index];
62    if first_axis.kind == PartialDifferentialEquationTransformAxisKind::LaplaceNonnegativeHalfLine
63        && first_order > 0
64    {
65        if second_order > 0 {
66            return Err(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported);
67        }
68        let frequency = context
69            .interner
70            .intern(ExpressionKind::Symbol(first_axis.transform_variable));
71        let exponent = builders::integer_value(context, first_order as i64);
72        let leading_multiplier = builders::power(context, frequency.clone(), exponent);
73        let leading = builders::product(context, vec![leading_multiplier, transformed_value]);
74        let mut terms = vec![leading];
75        for boundary_order in 0..first_order {
76            let mut required_orders = vec![0usize; jet.derivative_orders.len()];
77            required_orders[first_axis.source_variable_index] = boundary_order;
78            let Some(trace) = transformed_traces
79                .iter()
80                .find(|candidate| candidate.derivative_orders == required_orders)
81            else {
82                return Err(
83                    PartialDifferentialEquationTransformDecline::RequiredOriginTraceMissing,
84                );
85            };
86            let frequency_exponent = first_order - 1 - boundary_order;
87            let frequency_factor = if frequency_exponent == 0 {
88                builders::one(context)
89            } else {
90                let exponent = builders::integer_value(context, frequency_exponent as i64);
91                builders::power(context, frequency.clone(), exponent)
92            };
93            let boundary_term =
94                builders::product(context, vec![frequency_factor, trace.value.clone()]);
95            terms.push(builders::negate(context, boundary_term));
96        }
97        return Ok(builders::sum(context, terms));
98    }
99    let mut factors = vec![transformed_value];
100    for axis in [first_axis, second_axis] {
101        let order = jet.derivative_orders[axis.source_variable_index];
102        if order == 0 {
103            continue;
104        }
105        match axis.kind {
106            PartialDifferentialEquationTransformAxisKind::FourierWholeRealLine => {
107                let frequency = context
108                    .interner
109                    .intern(ExpressionKind::Symbol(axis.transform_variable));
110                let imaginary = context
111                    .interner
112                    .intern(ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit));
113                let symbol_factor = builders::product(context, vec![imaginary, frequency]);
114                let exponent = builders::integer_value(context, order as i64);
115                factors.push(builders::power(context, symbol_factor, exponent));
116            }
117            PartialDifferentialEquationTransformAxisKind::LaplaceNonnegativeHalfLine => {
118                return Err(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported);
119            }
120        }
121    }
122    Ok(builders::product(context, factors))
123}

125/// Maps one dependent jet through the whole-line Fourier operator theorem.
126pub fn map_jet_through_fourier_axis(
127    context: &Context,
128    jet: &PartialDifferentialEquationJet,
129    axis: &PartialDifferentialEquationTransformAxis,
130    remaining_variable_index: usize,
131    unknown: &PartialDifferentialEquationTransformedUnknown,
132) -> Result<ExpressionReference, PartialDifferentialEquationTransformDecline> {
133    // F{D_x^m D_t^n u}(omega, t) = (i · omega)^m · D_t^n U(t).
134    // The identity carries no boundary term because the source coordinate is
135    // declared to be the whole real line; that declaration is the axis
136    // geometry the plan validated, not an inference from the equation.
137    let (source_order, remaining_order) =
138        split_jet_orders(jet, axis.source_variable_index, remaining_variable_index)?;
139    let image = unknown.derivative_call(context, remaining_order);
140    if source_order == 0 {
141        return Ok(image);
142    }
143    let frequency = context
144        .interner
145        .intern(ExpressionKind::Symbol(axis.transform_variable));
146    let imaginary = context
147        .interner
148        .intern(ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit));
149    let symbol_factor = builders::product(context, vec![imaginary, frequency]);
150    let exponent = builders::integer_value(context, source_order as i64);
151    let multiplier = builders::power(context, symbol_factor, exponent);
152    Ok(builders::product(context, vec![multiplier, image]))
153}

155/// Maps one dependent jet through the unilateral Laplace operator theorem.
156pub fn map_jet_through_laplace_axis(
157    context: &Context,
158    jet: &PartialDifferentialEquationJet,
159    axis: &PartialDifferentialEquationTransformAxis,
160    remaining_variable_index: usize,
161    unknown: &PartialDifferentialEquationTransformedUnknown,
162    traces: &[PartialDifferentialEquationOriginTrace],
163) -> Result<ExpressionReference, PartialDifferentialEquationTransformDecline> {
164    // L{D_x^m D_t^n u}(s, t)
165    //   = s^m · D_t^n U(t) − sum_{k=0}^{m-1} s^(m-1-k) · T_{k,n}(t),
166    // where T_{k,n}(t) = (D_x^k D_t^n u)(0, t). Every boundary term is looked
167    // up in the resolved trace set; a missing term is a decline rather than an
168    // implicit zero.
169    let (source_order, remaining_order) =
170        split_jet_orders(jet, axis.source_variable_index, remaining_variable_index)?;
171    let image = unknown.derivative_call(context, remaining_order);
172    let frequency = context
173        .interner
174        .intern(ExpressionKind::Symbol(axis.transform_variable));
175    let leading = if source_order == 0 {
176        image
177    } else {
178        let exponent = builders::integer_value(context, source_order as i64);
179        let multiplier = builders::power(context, Arc::clone(&frequency), exponent);
180        builders::product(context, vec![multiplier, image])
181    };
182    let mut terms = vec![leading];
183    for boundary_order in 0..source_order {
184        let mut required_orders = vec![0usize; jet.derivative_orders.len()];
185        required_orders[axis.source_variable_index] = boundary_order;
186        required_orders[remaining_variable_index] = remaining_order;
187        let Some(trace) = traces
188            .iter()
189            .find(|candidate| candidate.derivative_orders == required_orders)
190        else {
191            return Err(PartialDifferentialEquationTransformDecline::RequiredOriginTraceMissing);
192        };
193        let frequency_exponent = source_order - 1 - boundary_order;
194        let frequency_factor = if frequency_exponent == 0 {
195            builders::one(context)
196        } else {
197            let exponent = builders::integer_value(context, frequency_exponent as i64);
198            builders::power(context, Arc::clone(&frequency), exponent)
199        };
200        let term = builders::product(context, vec![frequency_factor, trace.value.clone()]);
201        terms.push(builders::negate(context, term));
202    }
203    Ok(builders::sum(context, terms))
204}

206/// Enumerates every origin trace the Laplace operator theorem needs for a jet set.
207pub fn required_origin_trace_orders(
208    jets: &[PartialDifferentialEquationJet],
209    source_variable_index: usize,
210    remaining_variable_index: usize,
211) -> Vec<Vec<usize>> {
212    // The boundary terms of the derivative theorem are determined entirely by
213    // the multi-index inventory, so the requirement list is computed once and
214    // consumed both by resolution and by independent replay.
215    let mut required = Vec::new();
216    for jet in jets {
217        let source_order = jet.derivative_orders[source_variable_index];
218        let remaining_order = jet.derivative_orders[remaining_variable_index];
219        for boundary_order in 0..source_order {
220            let mut orders = vec![0usize; jet.derivative_orders.len()];
221            orders[source_variable_index] = boundary_order;
222            orders[remaining_variable_index] = remaining_order;
223            if !required.contains(&orders) {
224                required.push(orders);
225            }
226        }
227    }
228    required.sort();
229    required
230}

232fn split_jet_orders(
233    jet: &PartialDifferentialEquationJet,
234    source_variable_index: usize,
235    remaining_variable_index: usize,
236) -> Result<(usize, usize), PartialDifferentialEquationTransformDecline> {
237    if jet.derivative_orders.len() != 2 {
238        return Err(PartialDifferentialEquationTransformDecline::UnsupportedVariableCount);
239    }
240    if source_variable_index == remaining_variable_index {
241        return Err(PartialDifferentialEquationTransformDecline::TransformDomainIncompatible);
242    }
243    Ok((
244        jet.derivative_orders[source_variable_index],
245        jet.derivative_orders[remaining_variable_index],
246    ))
