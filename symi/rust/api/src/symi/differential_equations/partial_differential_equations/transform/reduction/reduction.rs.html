1// Exact transform-domain reduction of a validated linear PDE problem.
2//
3// The reduction consumes the classifier's coefficient data and the validated
4// axis geometry, applies the operator theorem to every dependent jet, and
5// transforms the forcing term through the shared transform subsystem. It
6// produces a transform-domain residual and the evidence needed to replay every
7// step; it produces no solution and claims none.

45/// Public native data structure named `partial_differential_equation_transform_reduction`.
46#[derive(Debug, Clone)]
47pub struct PartialDifferentialEquationTransformReduction {
48    /// Stores the stable method identity of the applied transform route.
49    pub method: PartialDifferentialEquationMethod,
50    /// Stores the validated transform-axis geometry that was applied.
51    pub plan: PartialDifferentialEquationTransformPlan,
52    /// Stores the exact transform-domain residual read as equal to zero.
53    pub transformed_residual: ExpressionReference,
54    /// Stores the generated one-axis transform image when the residual is an ODE.
55    pub transformed_unknown: Option<PartialDifferentialEquationTransformedUnknown>,
56    /// Stores the generated two-axis transform image when the residual is algebraic.
57    pub transformed_value: Option<ExpressionReference>,
58    /// Stores every origin trace the operator theorem consumed.
59    pub traces: Vec<PartialDifferentialEquationOriginTrace>,
60    /// Stores exact transforms applied to origin traces during sequential reduction.
61    pub trace_transform_certificates: Vec<IntegralTransformCertificate>,
62    /// Stores the exact forward transform of the forcing term.
63    pub transformed_forcing: ExpressionReference,
64    /// Stores every sequential forcing transform certificate, omitting only exact zero steps.
65    pub forcing_certificates: Vec<IntegralTransformCertificate>,
66    /// Stores the obligations this reduction requires a later stage to discharge.
67    pub proof_obligations: Vec<PartialDifferentialEquationProofObligation>,
68}

70/// Reduces a validated linear constant-coefficient PDE along one transform axis.
71pub fn reduce_partial_differential_equation_by_transform(
72    context: &Context,
73    problem: &PartialDifferentialEquationProblem,
74    plan: &PartialDifferentialEquationTransformPlan,
75) -> Result<
76    PartialDifferentialEquationTransformReduction,
77    PartialDifferentialEquationTransformDecline,
78> {
79    let classification = classify_partial_differential_equation(context, problem)
80        .map_err(PartialDifferentialEquationTransformDecline::Classification)?;
81    reduce_partial_differential_equation_by_transform_with_classification(
82        context,
83        problem,
84        &classification,
85        plan,
86    )
87}

89/// Reduces along one transform axis using previously computed classification data.
90pub fn reduce_partial_differential_equation_by_transform_with_classification(
91    context: &Context,
92    problem: &PartialDifferentialEquationProblem,
93    classification: &PartialDifferentialEquationClassification,
94    plan: &PartialDifferentialEquationTransformPlan,
95) -> Result<
96    PartialDifferentialEquationTransformReduction,
97    PartialDifferentialEquationTransformDecline,
98> {
99    if plan.axes.len() == 2 {
100        return reduce_partial_differential_equation_by_two_transform_axes_with_classification(
101            context,
102            problem,
103            classification,
104            plan,
105        );
106    }
107    let (axis, remaining_variable_index) = single_axis_plan(plan)?;
108    let coefficients = eligible_coefficients(problem, classification)?;
109    let jets: Vec<PartialDifferentialEquationJet> = coefficients
110        .iter()
111        .map(|record| record.jet.clone())
112        .collect();
113    let unknown = allocate_transformed_unknown(context, plan)?;
114    let traces = resolve_axis_traces(context, problem, axis, remaining_variable_index, &jets)?;
115    let mut terms = Vec::with_capacity(coefficients.len() + 1);
116    for record in &coefficients {
117        let image = match axis.kind {
118            PartialDifferentialEquationTransformAxisKind::FourierWholeRealLine => {
119                map_jet_through_fourier_axis(
120                    context,
121                    &record.jet,
122                    axis,
123                    remaining_variable_index,
124                    &unknown,
125                )?
126            }
127            PartialDifferentialEquationTransformAxisKind::LaplaceNonnegativeHalfLine => {
128                map_jet_through_laplace_axis(
129                    context,
130                    &record.jet,
131                    axis,
132                    remaining_variable_index,
133                    &unknown,
134                    &traces,
135                )?
136            }
137        };
138        terms.push(builders::product(
139            context,
140            vec![record.coefficient.clone(), image],
141        ));
142    }
143    let remainder = classification
144        .coefficient_data
145        .remainder
146        .clone()
147        .ok_or(PartialDifferentialEquationTransformDecline::EquationNotLinear)?;
148    let (transformed_forcing, forcing_certificate) = transform_forcing(context, &remainder, axis)?;
149    terms.push(transformed_forcing.clone());
150    let transformed_residual = builders::sum(context, terms);
151    validate_transformed_residual(problem, &transformed_residual, std::slice::from_ref(axis))?;
152    Ok(PartialDifferentialEquationTransformReduction {
153        method: axis.kind.method(),
154        plan: plan.clone(),
155        transformed_residual,
156        transformed_unknown: Some(unknown),
157        transformed_value: None,
158        traces,
159        trace_transform_certificates: Vec::new(),
160        transformed_forcing,
161        forcing_certificates: forcing_certificate.into_iter().collect(),
162        proof_obligations: reduction_obligations(axis),
163    })
164}

166fn reduce_partial_differential_equation_by_two_transform_axes_with_classification(
167    context: &Context,
168    problem: &PartialDifferentialEquationProblem,
169    classification: &PartialDifferentialEquationClassification,
170    plan: &PartialDifferentialEquationTransformPlan,
171) -> Result<
172    PartialDifferentialEquationTransformReduction,
173    PartialDifferentialEquationTransformDecline,
174> {
175    let (first_axis, second_axis) = two_axis_plan(plan)?;
176    let coefficients = eligible_coefficients(problem, classification)?;
177    let jets: Vec<PartialDifferentialEquationJet> = coefficients
178        .iter()
179        .map(|record| record.jet.clone())
180        .collect();
181    let traces = if first_axis.kind.requires_origin_traces() {
182        resolve_axis_traces(
183            context,
184            problem,
185            first_axis,
186            second_axis.source_variable_index,
187            &jets,
188        )?
189    } else {
190        Vec::new()
191    };
192    let (transformed_traces, trace_transform_certificates) =
193        transform_traces_for_second_axis(context, &traces, second_axis)?;
194    let transformed_value = allocate_two_axis_transformed_value(context)?;
195    let mut terms = Vec::with_capacity(coefficients.len() + 1);
196    for record in &coefficients {
197        let image = map_jet_through_two_transform_axes(
198            context,
199            &record.jet,
200            first_axis,
201            second_axis,
202            transformed_value.clone(),
203            &transformed_traces,
204        )?;
205        terms.push(builders::product(
206            context,
207            vec![record.coefficient.clone(), image],
208        ));
209    }
210    let remainder = classification
211        .coefficient_data
212        .remainder
213        .clone()
214        .ok_or(PartialDifferentialEquationTransformDecline::EquationNotLinear)?;
215    let (first_transformed_forcing, first_forcing_certificate) =
216        transform_forcing(context, &remainder, first_axis)?;
217    let (transformed_forcing, second_forcing_certificate) =
218        transform_forcing(context, &first_transformed_forcing, second_axis)?;
219    let forcing_certificates = [first_forcing_certificate, second_forcing_certificate]
220        .into_iter()
221        .flatten()
222        .collect();
223    terms.push(transformed_forcing.clone());
224    let transformed_residual = builders::sum(context, terms);
225    validate_transformed_residual(problem, &transformed_residual, &plan.axes)?;
226    Ok(PartialDifferentialEquationTransformReduction {
227        method: first_axis.kind.method(),
228        plan: plan.clone(),
229        transformed_residual,
230        transformed_unknown: None,
231        transformed_value: Some(transformed_value),
232        traces,
233        trace_transform_certificates,
234        transformed_forcing,
235        forcing_certificates,
236        proof_obligations: reduction_obligations_for_axes(&plan.axes),
237    })
238}

240pub(crate) fn transform_traces_for_second_axis(
241    context: &Context,
242    traces: &[PartialDifferentialEquationOriginTrace],
243    second_axis: &PartialDifferentialEquationTransformAxis,
244) -> Result<
245    (
246        Vec<PartialDifferentialEquationOriginTrace>,
247        Vec<IntegralTransformCertificate>,
248    ),
249    PartialDifferentialEquationTransformDecline,
250> {
251    // Applies the declared second transform to already verified first-axis
252    // boundary traces, retaining every certificate needed for independent replay.
253    let mut transformed_traces = Vec::with_capacity(traces.len());
254    let mut certificates = Vec::with_capacity(traces.len());
255    for trace in traces {
256        let attempt = analyze_forward(
257            context,
258            second_axis.kind.family(),
259            trace.value.clone(),
260            second_axis.source_variable,
261            second_axis.transform_variable,
262        );
263        let IntegralTransformAttempt::Closed(certificate) = attempt else {
264            return Err(PartialDifferentialEquationTransformDecline::DataTransformUnavailable);
265        };
266        if !certificate.hypotheses_are_proved() {
267            return Err(PartialDifferentialEquationTransformDecline::TransformHypothesisNotProved);
268        }
269        let mut transformed_trace = trace.clone();
270        transformed_trace.value = certificate.result.clone();
271        transformed_traces.push(transformed_trace);
272        certificates.push(certificate);
273    }
274    Ok((transformed_traces, certificates))
275}

277/// Returns the transform axis and remaining-coordinate index of a one-axis plan.
278pub(crate) fn single_axis_plan(
279    plan: &PartialDifferentialEquationTransformPlan,
280) -> Result<
281    (&PartialDifferentialEquationTransformAxis, usize),
282    PartialDifferentialEquationTransformDecline,
283> {
284    if !plan.is_single_axis() {
285        return Err(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported);
286    }
287    let axis = &plan.axes[0];
288    let remaining_variable_index = plan
289        .remaining_variable_index
290        .ok_or(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported)?;
291    Ok((axis, remaining_variable_index))
292}

294/// Returns the two ordered transform axes of an algebraic plan.
295pub(crate) fn two_axis_plan(
296    plan: &PartialDifferentialEquationTransformPlan,
297) -> Result<
298    (
299        &PartialDifferentialEquationTransformAxis,
300        &PartialDifferentialEquationTransformAxis,
301    ),
302    PartialDifferentialEquationTransformDecline,
303> {
304    if plan.axes.len() != 2 || plan.remaining_variable.is_some() {
305        return Err(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported);
306    }
307    Ok((&plan.axes[0], &plan.axes[1]))
308}

310/// Returns the classifier coefficient records a transform route may consume.
311pub(crate) fn eligible_coefficients(
312    problem: &PartialDifferentialEquationProblem,
313    classification: &PartialDifferentialEquationClassification,
314) -> Result<
315    Vec<crate::differential_equations::partial_differential_equations::classification::PartialDifferentialEquationJetCoefficient>,
316    PartialDifferentialEquationTransformDecline,
317>{
318    // The first transform tier requires every dependent-jet coefficient to be
319    // constant with respect to every declared independent variable. The
320    // restriction is stronger than a one-axis transform strictly needs; it
321    // closes a predictable tier before variable-coefficient transformed
322    // equations are considered.
323    if problem.independent_variables().len() != 2 {
324        return Err(PartialDifferentialEquationTransformDecline::UnsupportedVariableCount);
325    }
326    if classification.linearity != PartialDifferentialEquationLinearity::Linear {
327        return Err(PartialDifferentialEquationTransformDecline::EquationNotLinear);
328    }
329    let Some(coefficients) = classification.coefficient_data.full_coefficients.clone() else {
330        return Err(PartialDifferentialEquationTransformDecline::EquationNotLinear);
331    };
332    for record in &coefficients {
333        let free = free_variables(&record.coefficient);
334        if problem
335            .independent_variables()
336            .iter()
337            .any(|variable| free.contains(variable))
338        {
339            return Err(
340                PartialDifferentialEquationTransformDecline::DependentJetCoefficientNotConstant,
341            );
342        }
343    }
344    Ok(coefficients)
345}

347/// Resolves and independently re-proves every origin trace an axis requires.
348pub(crate) fn resolve_axis_traces(
349    context: &Context,
350    problem: &PartialDifferentialEquationProblem,
351    axis: &PartialDifferentialEquationTransformAxis,
352    remaining_variable_index: usize,
353    jets: &[PartialDifferentialEquationJet],
354) -> Result<Vec<PartialDifferentialEquationOriginTrace>, PartialDifferentialEquationTransformDecline>
355{
356    if !axis.kind.requires_origin_traces() {
357        return Ok(Vec::new());
358    }
359    let required =
360        required_origin_trace_orders(jets, axis.source_variable_index, remaining_variable_index);
361    let mut traces = Vec::with_capacity(required.len());
362    for orders in required {
363        let trace = resolve_partial_differential_equation_origin_trace(
364            context,
365            problem,
366            axis.source_variable,
367            &orders,
368        )?;
369        match verify_partial_differential_equation_origin_trace(context, problem, &trace) {
370            PartialDifferentialEquationVerificationVerdict::Verified => traces.push(trace),
371            PartialDifferentialEquationVerificationVerdict::Rejected => {
372                return Err(
373                    PartialDifferentialEquationTransformDecline::RequiredOriginTraceAmbiguous,
374                );
375            }
376            PartialDifferentialEquationVerificationVerdict::Inconclusive => return Err(
377                PartialDifferentialEquationTransformDecline::OriginTraceVerificationInconclusive,
378            ),
379        }
380    }
381    Ok(traces)
382}

384/// Transforms the forcing term exactly, or proves it is exactly zero.
385pub(crate) fn transform_forcing(
386    context: &Context,
387    remainder: &ExpressionReference,
388    axis: &PartialDifferentialEquationTransformAxis,
389) -> Result<
390    (ExpressionReference, Option<IntegralTransformCertificate>),
391    PartialDifferentialEquationTransformDecline,
392> {
393    // A forcing term proved exactly zero transforms to zero by linearity, and
394    // that fact is recorded as the absence of a certificate rather than as a
395    // certificate the shipped dispatcher could not replay. Any other forcing
396    // must close through the certificate path.
397    let normalized_remainder = crate::expand::expand(context, remainder);
398    if verification_verdict(prove_residual_zero(context, &normalized_remainder))
399        == PartialDifferentialEquationVerificationVerdict::Verified
400    {
401        return Ok((builders::zero(context), None));
402    }
403    let attempt = analyze_forward(
404        context,
405        axis.kind.family(),
406        normalized_remainder,
407        axis.source_variable,
408        axis.transform_variable,
409    );
410    match attempt {
411        IntegralTransformAttempt::Closed(certificate) => {
412            if !certificate.hypotheses_are_proved() {
413                return Err(
414                    PartialDifferentialEquationTransformDecline::TransformHypothesisNotProved,
415                );
416            }
417            Ok((certificate.result.clone(), Some(certificate)))
418        }
419        IntegralTransformAttempt::Unevaluated(_) => {
420            Err(PartialDifferentialEquationTransformDecline::ForcingTransformUnavailable)
421        }
422        IntegralTransformAttempt::Declined(decline) => {
423            Err(PartialDifferentialEquationTransformDecline::TransformSubsystem(decline))
424        }
425    }
426}

428/// Reports the obligations a transform reduction along one axis leaves open.
429pub(crate) fn reduction_obligations(
430    axis: &PartialDifferentialEquationTransformAxis,
431) -> Vec<PartialDifferentialEquationProofObligation> {
432    let mut obligations = vec![
433        PartialDifferentialEquationProofObligation::TransformAxisDomainEstablished,
434        PartialDifferentialEquationProofObligation::ConstantCoefficients,
435        PartialDifferentialEquationProofObligation::TransformExistenceAndGrowth,
436        PartialDifferentialEquationProofObligation::TransformReductionExact,
437    ];
438    if axis.kind.requires_origin_traces() {
439        obligations.push(PartialDifferentialEquationProofObligation::OriginTraceEstablished);
440    }
441    obligations.push(PartialDifferentialEquationProofObligation::ExactInverseTransformAvailable);
442    obligations
443}

445pub(crate) fn reduction_obligations_for_axes(
446    axes: &[PartialDifferentialEquationTransformAxis],
447) -> Vec<PartialDifferentialEquationProofObligation> {
448    let mut obligations = vec![
449        PartialDifferentialEquationProofObligation::TransformAxisDomainEstablished,
450        PartialDifferentialEquationProofObligation::ConstantCoefficients,
451        PartialDifferentialEquationProofObligation::TransformExistenceAndGrowth,
452        PartialDifferentialEquationProofObligation::TransformReductionExact,
453        PartialDifferentialEquationProofObligation::ExactInverseTransformAvailable,
454    ];
455    if axes.iter().any(|axis| axis.kind.requires_origin_traces()) {
456        obligations.push(PartialDifferentialEquationProofObligation::OriginTraceEstablished);
457    }
458    obligations
459}

461fn allocate_transformed_unknown(
462    context: &Context,
463    plan: &PartialDifferentialEquationTransformPlan,
464) -> Result<
465    PartialDifferentialEquationTransformedUnknown,
466    PartialDifferentialEquationTransformDecline,
467> {
468    let remaining_variable = plan
469        .remaining_variable
470        .ok_or(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported)?;
471    let allocated =
472        allocate_partial_differential_equation_arbitrary_function(context, "transform_image", 1);
473    Ok(PartialDifferentialEquationTransformedUnknown {
474        function: allocated.function,
475        remaining_variable,
476    })
477}

479fn allocate_two_axis_transformed_value(
480    context: &Context,
481) -> Result<ExpressionReference, PartialDifferentialEquationTransformDecline> {
482    let allocated =
483        allocate_partial_differential_equation_arbitrary_function(context, "transform_image", 2);
484    let FunctionIdentifier::User(identifier) = allocated.function else {
485        return Err(PartialDifferentialEquationTransformDecline::TransformPlanUnsupported);
486    };
487    Ok(context.interner.intern(ExpressionKind::Symbol(identifier)))

490fn validate_transformed_residual(
491    problem: &PartialDifferentialEquationProblem,
492    transformed_residual: &ExpressionReference,
493    axes: &[PartialDifferentialEquationTransformAxis],
494) -> Result<(), PartialDifferentialEquationTransformDecline> {
495    // A reduction that still mentions the source dependent function, still
496    // mentions the integrated-out coordinate, or still carries a structural
497    // transform node is not a reduction at all.
498    if contains_dependent_occurrence(transformed_residual, problem) {
499        return Err(
500            PartialDifferentialEquationTransformDecline::TransformedEquationNotScalarOdeOrAlgebraic,
501        );
502    }
503    if contains_transform_placeholder(transformed_residual) {
504        return Err(PartialDifferentialEquationTransformDecline::ForcingTransformUnavailable);
505    }
506    if axes
507        .iter()
508        .any(|axis| free_variables(transformed_residual).contains(&axis.source_variable))
509    {
510        return Err(
511            PartialDifferentialEquationTransformDecline::TransformedEquationNotScalarOdeOrAlgebraic,
512        );
513    }
514    Ok(())
