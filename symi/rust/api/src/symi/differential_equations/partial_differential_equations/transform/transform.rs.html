1// Exact transform-domain reduction for scalar partial differential equations.
2//
3// The subtree turns a validated linear constant-coefficient problem into an
4// exact transform-domain equation and nothing more. It owns no transform
5// mathematics of its own: every forward and inverse step is delegated to the
6// shared `integral_transforms` subsystem through that subsystem's certificate
7// path, and every jet shape is read from the classifier's coefficient data
8// rather than rediscovered from raw derivative syntax.
9//
10// A reduction is a claim about a theorem, not a solution. Nothing here returns
11// a partial differential equation solution; the independent reduction verifier
12// exists so that a later solving stage can only build on a replayed reduction.

36/// Public native enumeration named `partial_differential_equation_transform_decline`.
37#[derive(Debug, Clone, PartialEq, Eq)]
38pub enum PartialDifferentialEquationTransformDecline {
39    /// Represents a problem whose independent-variable count is outside the tier.
40    UnsupportedVariableCount,
41    /// Represents a governing equation the classifier did not prove linear.
42    EquationNotLinear,
43    /// Represents a dependent-jet coefficient depending on an independent variable.
44    DependentJetCoefficientNotConstant,
45    /// Represents a caller that supplied no explicit transform-axis geometry.
46    TransformDomainNotSpecified,
47    /// Represents transform-axis geometry incompatible with the declared problem.
48    TransformDomainIncompatible,
49    /// Represents a transform axis whose source variable is not declared.
50    TransformAxisNotApplicable,
51    /// Represents a transform plan outside the supported reduction tier.
52    TransformPlanUnsupported,
53    /// Represents a transform hypothesis the assumption engine did not prove.
54    TransformHypothesisNotProved,
55    /// Represents a required unilateral origin trace with no supplied condition.
56    RequiredOriginTraceMissing,
57    /// Represents conflicting supplied conditions for one origin trace.
58    RequiredOriginTraceAmbiguous,
59    /// Represents an origin trace whose isolating relation was not proved.
60    OriginTraceVerificationInconclusive,
61    /// Represents a forcing term whose exact forward transform did not close.
62    ForcingTransformUnavailable,
63    /// Represents supplied data whose exact forward transform did not close.
64    DataTransformUnavailable,
65    /// Represents a reduction the independent verifier could not replay.
66    TransformReductionInconclusive,
67    /// Represents a transformed equation that is neither a scalar ODE nor algebraic.
68    TransformedEquationNotScalarOdeOrAlgebraic,
69    /// Represents a transformed ordinary differential equation that declined.
70    TransformedOdeDeclined,
71    /// Represents a transformed algebraic solve that declined.
72    TransformedAlgebraicSolveDeclined,
73    /// Represents transformed-candidate verification that did not conclude.
74    TransformedSolutionVerificationInconclusive,
75    /// Represents an inverse transform with no exact closed path.
76    ExactInverseTransformUnavailable,
77    /// Represents a forward round trip that did not conclude.
78    TransformRoundTripInconclusive,
79    /// Represents a candidate the central PDE verifier proved wrong.
80    FinalPdeVerificationRejected,
81    /// Represents a candidate the central PDE verifier could not decide.
82    FinalPdeVerificationInconclusive,
83    /// Represents exhaustion of the deterministic work budget.
84    ResourceBudgetExceeded,
85    /// Represents a classification decline reported by the classifier.
86    Classification(PartialDifferentialEquationClassificationDecline),
87    /// Represents a decline reported by the shared transform subsystem.
88    TransformSubsystem(IntegralTransformDecline),
89}
