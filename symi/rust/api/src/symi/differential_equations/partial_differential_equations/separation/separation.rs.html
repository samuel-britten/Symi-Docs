63/// Public native enumeration named `partial_differential_equation_separation_decline`.
64#[derive(Debug, Clone, PartialEq, Eq)]
65pub enum PartialDifferentialEquationSeparationDecline {
66    /// The validated problem does not have exactly two independent variables.
67    TwoIndependentVariablesRequired,
68    /// A dependent-function derivative has total order greater than two.
69    DerivativeOrderNotSupported,
70    /// No exact dependency-pure multiplicative decomposition was found.
71    MultiplicativeDecompositionNotFound,
72    /// No exact dependency-pure additive decomposition was found.
73    AdditiveDecompositionNotFound,
74    /// The deterministic expression-analysis budget was exhausted.
75    SeparationBudgetExceeded,
76    /// A coordinate-dependent expression entangles both factor sides.
77    MixedDependency,
78    /// The independently replayed certificate did not verify.
79    VerificationNotVerified,
80    /// No original-coordinate decomposition or verified canonical retry closed.
81    CanonicalPreprocessingNotClosed,
82}

160/// Separates a supported two-variable PDE through a directed verified ansatz.
161pub fn separate_partial_differential_equation(
162    context: &Context,
163    problem: &PartialDifferentialEquationProblem,
164    options: PartialDifferentialEquationSeparationOptions,
165) -> Result<PartialDifferentialEquationSeparationResult, PartialDifferentialEquationSeparationDecline>
166{
167    match separate_working_problem(context, problem, problem, &options) {
168        Ok(result) => Ok(result),
169        Err(original_decline) if options.allow_canonical_preprocessing => {
170            let canonicalization = canonicalize_partial_differential_equation(context, problem)
171                .map_err(|_| original_decline.clone())?;
172            let mut retry_options = options;
173            retry_options.allow_canonical_preprocessing = false;
174            let mut result = separate_working_problem(
175                context,
176                &canonicalization.transformation.target_problem,
177                &canonicalization.transformation.target_problem,
178                &retry_options,
179            )
180            .map_err(|_| {
181                PartialDifferentialEquationSeparationDecline::CanonicalPreprocessingNotClosed
182            })?;
183            result.preprocessing = Some(canonicalization);
184            let verification =
185                verify_partial_differential_equation_separation(context, problem, &result);
186            if verification.verdict != PartialDifferentialEquationVerificationVerdict::Verified {
187                return Err(PartialDifferentialEquationSeparationDecline::VerificationNotVerified);
188            }
189            result.verification = verification;
190            Ok(result)
191        }
192        Err(decline) => Err(decline),
193    }
194}

196/// Replays a recorded separation certificate without using separator state.
197pub fn verify_partial_differential_equation_separation(
198    context: &Context,
199    problem: &PartialDifferentialEquationProblem,
200    separation: &PartialDifferentialEquationSeparationResult,
201) -> PartialDifferentialEquationSeparationVerificationReport {
202    if let Some(preprocessing) = &separation.preprocessing {
203        let coordinate_report = verify_partial_differential_equation_coordinate_transformation(
204            context,
205            problem,
206            &preprocessing.transformation,
207        );
208        if coordinate_report.verdict != PartialDifferentialEquationVerificationVerdict::Verified {
209            return rejected_report(context);
210        }
211    }
212    let working_problem = separation
213        .preprocessing
214        .as_ref()
215        .map(|preprocessing| &preprocessing.transformation.target_problem)
216        .unwrap_or(problem);
217    let factor_calls = factor_calls(context, separation, working_problem);
218    let Some(reconstruction) = reconstruction(context, separation.strategy, &factor_calls) else {
219        return rejected_report(context);
220    };
221    let substituted = substituted_residual(context, working_problem, &reconstruction);
222    let normalized_substituted_residual = builders::product(
223        context,
224        vec![separation.normalization_multiplier.clone(), substituted],
225    );
226    let first_recombination_term = builders::product(
227        context,
228        vec![
229            separation.second_weight.clone(),
230            separation.first_equation.residual.clone(),
231        ],
232    );
233    let second_recombination_term = builders::product(
234        context,
235        vec![
236            separation.first_weight.clone(),
237            separation.second_equation.residual.clone(),
238        ],
239    );
240    let recombination_residual = builders::sum(
241        context,
242        vec![first_recombination_term, second_recombination_term],
243    );
244    let difference = builders::subtract(
245        context,
246        recombination_residual.clone(),
247        normalized_substituted_residual.clone(),
248    );
249    let difference_verdict =
250        verdict_from_zero_equivalence(separation_zero_verdict(context, &difference));
251    let first_pure = is_side_pure(
252        &separation.first_equation.residual,
253        working_problem.independent_variables()[0],
254        working_problem.independent_variables()[1],
255        separation.first_factor.function,
256        separation.second_factor.function,
257        true,
258    );
259    let second_pure = is_side_pure(
260        &separation.second_equation.residual,
261        working_problem.independent_variables()[0],
262        working_problem.independent_variables()[1],
263        separation.first_factor.function,
264        separation.second_factor.function,
265        false,
266    );
267    let expanded_recombination = crate::expand::expand(context, &recombination_residual);
268    let simplified_recombination = crate::simplify::simplify(context, expanded_recombination);
269    let constant_cancels =
270        !contains_symbol(&simplified_recombination, separation.separation_constant);
271    let normalization_verdict =
272        normalization_verdict(context, &separation.normalization_multiplier);
273    let recorded_shape_is_valid = separation.first_factor.independent_variable
274        == working_problem.independent_variables()[0]
275        && separation.second_factor.independent_variable
276            == working_problem.independent_variables()[1]
277        && separation.first_equation.dependent_function == separation.first_factor.function
278        && separation.second_equation.dependent_function == separation.second_factor.function
279        && separation.first_equation.independent_variable
280            == separation.first_factor.independent_variable
281        && separation.second_equation.independent_variable
282            == separation.second_factor.independent_variable;
283    let verdict = if !first_pure
284        || !second_pure
285        || !constant_cancels
286        || !recorded_shape_is_valid
287        || normalization_verdict == PartialDifferentialEquationVerificationVerdict::Rejected
288    {
289        PartialDifferentialEquationVerificationVerdict::Rejected
290    } else if normalization_verdict == PartialDifferentialEquationVerificationVerdict::Inconclusive
291    {
292        PartialDifferentialEquationVerificationVerdict::Inconclusive
293    } else {
294        difference_verdict
295    };
296    PartialDifferentialEquationSeparationVerificationReport {
297        verdict,
298        normalized_substituted_residual,
299        recombination_residual,
300        difference,
301    }
302}

304fn separate_working_problem(
305    context: &Context,
306    source_problem: &PartialDifferentialEquationProblem,
307    working_problem: &PartialDifferentialEquationProblem,
308    options: &PartialDifferentialEquationSeparationOptions,
309) -> Result<PartialDifferentialEquationSeparationResult, PartialDifferentialEquationSeparationDecline>
310{
311    validate_problem_shape(working_problem, options.work_budget)?;
312    let first_variable = working_problem.independent_variables()[0];
313    let second_variable = working_problem.independent_variables()[1];
314    let first_factor = PartialDifferentialEquationSeparatedFactor {
315        function: fresh_function(
316            context,
317            options
318                .first_factor_name
319                .as_deref()
320                .unwrap_or("separation_first"),
321        ),
322        independent_variable: first_variable,
323    };
324    let second_factor = PartialDifferentialEquationSeparatedFactor {
325        function: fresh_function(
326            context,
327            options
328                .second_factor_name
329                .as_deref()
330                .unwrap_or("separation_second"),
331        ),
332        independent_variable: second_variable,
333    };
334    let separation_constant = fresh_symbol(
335        context,
336        options
337            .separation_constant_name
338            .as_deref()
339            .unwrap_or("separation_constant"),
340    );
341    let first_variable_expression = symbol_expression(context, first_variable);
342    let first_call = builders::function_call(
343        context,
344        first_factor.function,
345        vec![first_variable_expression],
346    );
347    let second_variable_expression = symbol_expression(context, second_variable);
348    let second_call = builders::function_call(
349        context,
350        second_factor.function,
351        vec![second_variable_expression],
352    );
353    let calls = FactorCalls {
354        first: first_call,
355        second: second_call,
356    };
357    let strategies = strategies(options.strategy);
358    let mut last_decline =
359        PartialDifferentialEquationSeparationDecline::MultiplicativeDecompositionNotFound;
360    for strategy in strategies {
361        let Some(reconstruction) = reconstruction(context, strategy, &calls) else {
362            return Err(
363                PartialDifferentialEquationSeparationDecline::MultiplicativeDecompositionNotFound,
364            );
365        };
366        let residual = substituted_residual(context, working_problem, &reconstruction);
367        let decomposition = match strategy {
368            PartialDifferentialEquationSeparationStrategy::Multiplicative => {
369                multiplicative_decomposition(
370                    context,
371                    &residual,
372                    working_problem,
373                    &first_factor,
374                    &second_factor,
375                    &calls,
376                    options.work_budget,
377                )
378            }
379            PartialDifferentialEquationSeparationStrategy::Additive => additive_decomposition(
380                context,
381                &residual,
382                working_problem,
383                &first_factor,
384                &second_factor,
385                options.work_budget,
386            ),
387            PartialDifferentialEquationSeparationStrategy::Automatic => {
388                return Err(
389                    PartialDifferentialEquationSeparationDecline::MultiplicativeDecompositionNotFound,
390                );
391            }
392        };
393        let Ok((first_operator, second_operator, first_weight, second_weight)) = decomposition
394        else {
395            last_decline = decomposition.unwrap_err();
396            continue;
397        };
398        let constant_expression = symbol_expression(context, separation_constant);
399        let (first_residual, second_residual) = match strategy {
400            PartialDifferentialEquationSeparationStrategy::Multiplicative => {
401                let first_constant_term = builders::product(
402                    context,
403                    vec![constant_expression.clone(), first_weight.clone()],
404                );
405                let second_constant_term =
406                    builders::product(context, vec![constant_expression, second_weight.clone()]);
407                (
408                    builders::subtract(context, first_operator, first_constant_term),
409                    builders::sum(context, vec![second_operator, second_constant_term]),
410                )
411            }
412            PartialDifferentialEquationSeparationStrategy::Additive => (
413                builders::subtract(context, first_operator, constant_expression.clone()),
414                builders::sum(context, vec![second_operator, constant_expression]),
415            ),
416            PartialDifferentialEquationSeparationStrategy::Automatic => {
417                return Err(
418                    PartialDifferentialEquationSeparationDecline::MultiplicativeDecompositionNotFound,
419                );
420            }
421        };
422        let mut result = PartialDifferentialEquationSeparationResult {
423            strategy,
424            separation_constant,
425            first_factor: first_factor.clone(),
426            second_factor: second_factor.clone(),
427            first_equation: PartialDifferentialEquationSeparatedEquation {
428                residual: first_residual,
429                dependent_function: first_factor.function,
430                independent_variable: first_variable,
431            },
432            second_equation: PartialDifferentialEquationSeparatedEquation {
433                residual: second_residual,
434                dependent_function: second_factor.function,
435                independent_variable: second_variable,
436            },
437            reconstruction,
438            normalization_multiplier: builders::one(context),
439            first_weight,
440            second_weight,
441            preprocessing: None,
442            proof_obligations: vec![
443                PartialDifferentialEquationProofObligation::SeparabilityEstablished,
444            ],
445            verification: rejected_report(context),
446        };
447        let verification =
448            verify_partial_differential_equation_separation(context, source_problem, &result);
449        if verification.verdict == PartialDifferentialEquationVerificationVerdict::Verified {
450            result.verification = verification;
451            return Ok(result);
452        }
453        last_decline = PartialDifferentialEquationSeparationDecline::VerificationNotVerified;
454    }
455    Err(last_decline)
456}

458fn validate_problem_shape(
459    problem: &PartialDifferentialEquationProblem,
460    budget: usize,
461) -> Result<(), PartialDifferentialEquationSeparationDecline> {
462    if problem.independent_variables().len() != 2 {
463        return Err(PartialDifferentialEquationSeparationDecline::TwoIndependentVariablesRequired);
464    }
465    let inventory = collect_partial_differential_equation_jets(problem, budget)
466        .map_err(|_| PartialDifferentialEquationSeparationDecline::SeparationBudgetExceeded)?;
467    if inventory.order > 2 {
468        return Err(PartialDifferentialEquationSeparationDecline::DerivativeOrderNotSupported);
469    }
470    Ok(())
471}

485fn multiplicative_decomposition(
486    context: &Context,
487    residual: &ExpressionReference,
488    problem: &PartialDifferentialEquationProblem,
489    first_factor: &PartialDifferentialEquationSeparatedFactor,
490    second_factor: &PartialDifferentialEquationSeparatedFactor,
491    calls: &FactorCalls,
492    budget: usize,
493) -> Result<
494    (
495        ExpressionReference,
496        ExpressionReference,
497        ExpressionReference,
498        ExpressionReference,
499    ),
500    PartialDifferentialEquationSeparationDecline,
501> {
502    let terms = product_terms(
503        context,
504        residual,
505        problem,
506        first_factor.function,
507        second_factor.function,
508        budget,
509    )?;
510    let first_weights = candidate_weights(&terms, true, &calls.first);
511    let second_weights = candidate_weights(&terms, false, &calls.second);
512    for first_weight in first_weights {
513        for second_weight in &second_weights {
514            let mut first_operators = Vec::new();
515            let mut second_operators = Vec::new();
516            let mut valid = true;
517            for term in &terms {
518                if term.second == *second_weight {
519                    first_operators.push(term.first.clone());
520                } else if term.first == first_weight {
521                    second_operators.push(term.second.clone());
522                } else {
523                    valid = false;
524                    break;
525                }
526            }
527            if valid && !first_operators.is_empty() && !second_operators.is_empty() {
528                return Ok((
529                    builders::sum(context, first_operators),
530                    builders::sum(context, second_operators),
531                    first_weight,
532                    second_weight.clone(),
533                ));
534            }
535        }
536    }
537    Err(PartialDifferentialEquationSeparationDecline::MultiplicativeDecompositionNotFound)
538}

540fn additive_decomposition(
541    context: &Context,
542    residual: &ExpressionReference,
543    problem: &PartialDifferentialEquationProblem,
544    first_factor: &PartialDifferentialEquationSeparatedFactor,
545    second_factor: &PartialDifferentialEquationSeparatedFactor,
546    budget: usize,
547) -> Result<
548    (
549        ExpressionReference,
550        ExpressionReference,
551        ExpressionReference,
552        ExpressionReference,
553    ),
554    PartialDifferentialEquationSeparationDecline,
555> {
556    let mut first_terms = Vec::new();
557    let mut second_terms = Vec::new();
558    let summands = flatten_sum(residual);
559    if summands.len() > budget {
560        return Err(PartialDifferentialEquationSeparationDecline::SeparationBudgetExceeded);
561    }
562    for term in summands {
563        let dependency = dependency_of(
564            &term,
565            problem.independent_variables()[0],
566            problem.independent_variables()[1],
567            first_factor.function,
568            second_factor.function,
569        );
570        match dependency {
571            SideDependency::First | SideDependency::Parameter => first_terms.push(term),
572            SideDependency::Second => second_terms.push(term),
573            SideDependency::Mixed => {
574                return Err(PartialDifferentialEquationSeparationDecline::MixedDependency);
575            }
576        }
577    }
578    if first_terms.is_empty() || second_terms.is_empty() {
579        return Err(PartialDifferentialEquationSeparationDecline::AdditiveDecompositionNotFound);
580    }
581    Ok((
582        builders::sum(context, first_terms),
583        builders::sum(context, second_terms),
584        builders::one(context),
585        builders::one(context),
586    ))
587}

589fn product_terms(
590    context: &Context,
591    residual: &ExpressionReference,
592    problem: &PartialDifferentialEquationProblem,
593    first_function: FunctionIdentifier,
594    second_function: FunctionIdentifier,
595    budget: usize,
596) -> Result<Vec<ProductTerm>, PartialDifferentialEquationSeparationDecline> {
597    let summands = flatten_sum(residual);
598    if summands.len() > budget {
599        return Err(PartialDifferentialEquationSeparationDecline::SeparationBudgetExceeded);
600    }
601    summands
602        .into_iter()
603        .map(|term| {
604            let mut first_factors = Vec::new();
605            let mut second_factors = Vec::new();
606            for factor in flatten_product(&term) {
607                match dependency_of(
608                    &factor,
609                    problem.independent_variables()[0],
610                    problem.independent_variables()[1],
611                    first_function,
612                    second_function,
613                ) {
614                    SideDependency::First | SideDependency::Parameter => first_factors.push(factor),
615                    SideDependency::Second => second_factors.push(factor),
616                    SideDependency::Mixed => {
617                        return Err(PartialDifferentialEquationSeparationDecline::MixedDependency);
618                    }
619                }
620            }
621            Ok(ProductTerm {
622                first: builders::product(context, first_factors),
623                second: builders::product(context, second_factors),
624            })
625        })
626        .collect()
627}

629fn candidate_weights(
630    terms: &[ProductTerm],
631    first: bool,
632    preferred: &ExpressionReference,
633) -> Vec<ExpressionReference> {
634    let mut weights = vec![preferred.clone()];
635    for term in terms {
636        let candidate = if first { &term.first } else { &term.second };
637        if !weights.contains(candidate) {
638            weights.push(candidate.clone());
639        }
640    }
641    weights
642}

644fn substituted_residual(
645    context: &Context,
646    problem: &PartialDifferentialEquationProblem,
647    reconstruction: &ExpressionReference,
648) -> ExpressionReference {
649    let mut occurrences = Vec::new();
650    collect_dependent_occurrences(problem.equation(), problem, &mut occurrences);
651    let mut mapping = crate::expression::substitute::SubstitutionMap::new();
652    for occurrence in occurrences {
653        let Some((orders, arguments)) = dependent_occurrence_data(&occurrence, problem) else {
654            continue;
655        };
656        let derivative_orders: Vec<_> = problem
657            .independent_variables()
658            .iter()
659            .copied()
660            .zip(orders)
661            .collect();
662        let derivative = partial_derivative(context, reconstruction, &derivative_orders);
663        let mut argument_mapping = crate::expression::substitute::SubstitutionMap::new();
664        for (variable, argument) in problem.independent_variables().iter().zip(arguments) {
665            argument_mapping.insert(symbol_expression(context, *variable), argument);
666        }
667        mapping.insert(
668            occurrence,
669            crate::expression::substitute::substitute(context, &derivative, &argument_mapping),
670        );
671    }
672    crate::expression::substitute::substitute(context, problem.equation(), &mapping)
673}

675fn collect_dependent_occurrences(
676    expression: &ExpressionReference,
677    problem: &PartialDifferentialEquationProblem,
678    occurrences: &mut Vec<ExpressionReference>,
679) {
680    if dependent_occurrence_data(expression, problem).is_some() {
681        occurrences.push(expression.clone());
682    }
683    let _ = traverse_children(expression, |child| {
684        collect_dependent_occurrences(child, problem, occurrences);
685        Ok(())
686    });
687}

689fn dependent_occurrence_data(
690    expression: &ExpressionReference,
691    problem: &PartialDifferentialEquationProblem,
692) -> Option<(Vec<usize>, Vec<ExpressionReference>)> {
693    match &expression.kind {
694        ExpressionKind::FunctionCall(function, arguments)
695            if *function == problem.dependent_function() =>
696        {
697            Some((vec![0; arguments.len()], arguments.clone()))
698        }
699        ExpressionKind::Derivative(function, orders, arguments)
700            if *function == problem.dependent_function() =>
701        {
702            Some((orders.clone(), arguments.clone()))
703        }
704        _ => None,
705    }
706}

757fn verdict_from_zero_equivalence(
758    verdict: ZeroEquivalenceVerdict,
759) -> PartialDifferentialEquationVerificationVerdict {
760    match verdict {
761        ZeroEquivalenceVerdict::TrueValue => {
762            PartialDifferentialEquationVerificationVerdict::Verified
763        }
764        ZeroEquivalenceVerdict::FalseValue => {
765            PartialDifferentialEquationVerificationVerdict::Rejected
766        }
767        ZeroEquivalenceVerdict::Unknown => {
768            PartialDifferentialEquationVerificationVerdict::Inconclusive
769        }
770    }
771}
772
773fn separation_zero_verdict(
774    context: &Context,
775    residual: &ExpressionReference,
776) -> ZeroEquivalenceVerdict {
777    let direct = zero_equivalent(context, residual, None);
778    if direct != ZeroEquivalenceVerdict::Unknown {
779        return direct;
780    }
781    let cancelled = crate::simplify::rational::cancel(context, residual);
782    let cancelled_verdict = zero_equivalent(context, &cancelled, None);
783    if cancelled_verdict != ZeroEquivalenceVerdict::Unknown {
784        return cancelled_verdict;
785    }
786    let expanded = crate::expand::expand(context, &cancelled);
787    let expanded_verdict = zero_equivalent(context, &expanded, None);
788    if expanded_verdict != ZeroEquivalenceVerdict::Unknown {
789        return expanded_verdict;
790    }
791    let combined = crate::simplify::rational::together(context, &expanded);
792    let expanded_combined = crate::expand::expand(context, &combined);
793    let simplified = crate::simplify::simplify(context, expanded_combined);
794    zero_equivalent(context, &simplified, None)
795}
796
797fn normalization_verdict(
798    context: &Context,
799    multiplier: &ExpressionReference,
800) -> PartialDifferentialEquationVerificationVerdict {
801    let one = builders::one(context);
802    let difference_from_one = builders::subtract(context, multiplier.clone(), one);
803    if separation_zero_verdict(context, &difference_from_one) == ZeroEquivalenceVerdict::TrueValue {
804        return PartialDifferentialEquationVerificationVerdict::Verified;
805    }
806    match separation_zero_verdict(context, multiplier) {
807        ZeroEquivalenceVerdict::FalseValue => {
808            PartialDifferentialEquationVerificationVerdict::Verified
809        }
810        ZeroEquivalenceVerdict::TrueValue => {
811            PartialDifferentialEquationVerificationVerdict::Rejected
812        }
813        ZeroEquivalenceVerdict::Unknown => {
814            PartialDifferentialEquationVerificationVerdict::Inconclusive
815        }
816    }
817}

819fn is_side_pure(
820    expression: &ExpressionReference,
821    first_variable: SymbolIdentifier,
822    second_variable: SymbolIdentifier,
823    first_function: FunctionIdentifier,
824    second_function: FunctionIdentifier,
825    first_side: bool,
826) -> bool {
827    matches!(
828        dependency_of(expression, first_variable, second_variable, first_function, second_function),
829        SideDependency::Parameter | SideDependency::First if first_side
830    ) || matches!(
831        dependency_of(expression, first_variable, second_variable, first_function, second_function),
832        SideDependency::Parameter | SideDependency::Second if !first_side
833    )
834}
835
836#[derive(Clone, Copy, PartialEq, Eq)]
837enum SideDependency {
838    First,
839    Second,
840    Parameter,
841    Mixed,
842}
843
844fn dependency_of(
845    expression: &ExpressionReference,
846    first_variable: SymbolIdentifier,
847    second_variable: SymbolIdentifier,
848    first_function: FunctionIdentifier,
849    second_function: FunctionIdentifier,
850) -> SideDependency {
851    let variables = free_variables(expression);
852    let first =
853        variables.contains(&first_variable) || contains_function(expression, first_function);
854    let second =
855        variables.contains(&second_variable) || contains_function(expression, second_function);
856    match (first, second) {
857        (true, false) => SideDependency::First,
858        (false, true) => SideDependency::Second,
859        (false, false) => SideDependency::Parameter,
860        (true, true) => SideDependency::Mixed,
861    }
862}
863
864fn contains_function(expression: &ExpressionReference, function: FunctionIdentifier) -> bool {
865    if matches!(expression.kind, ExpressionKind::FunctionCall(found, _) | ExpressionKind::Derivative(found, _, _) if found == function)
866    {
867        return true;
868    }
869    let mut found = false;
870    let _ = traverse_children(expression, |child| {
871        found |= contains_function(child, function);
872        Ok(())
873    });
874    found
875}
876
877fn contains_symbol(expression: &ExpressionReference, symbol: SymbolIdentifier) -> bool {
878    if matches!(expression.kind, ExpressionKind::Symbol(found) if found == symbol) {
879        return true;
880    }
881    let mut found = false;
882    let _ = traverse_children(expression, |child| {
883        found |= contains_symbol(child, symbol);
884        Ok(())
885    });
886    found
887}
