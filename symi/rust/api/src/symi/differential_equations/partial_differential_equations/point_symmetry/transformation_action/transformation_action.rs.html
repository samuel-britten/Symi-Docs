1// Finite point transformations acting on solutions, and their certificates.
2//
3// A fiber-linear point transformation acts on a function by `T[f](x) = A(x) f(X(x))`. It is a
4// symmetry of a linear homogeneous equation `L u = 0` exactly when
5//
6//     L(A f(X)) = M(x) (L f)(X)
7//
8// holds identically in the jets of an arbitrary seed `f`, for some multiplier `M`. That identity is
9// what is checked here, and it is checked on the transformation as given, with no reference to the
10// generators it may have been built from: the seed's jets at the transformed point become
11// independent symbols, the chain rule is applied through the coordinate map, and every coefficient
12// of the difference must vanish. Dividing by the amplitude through its logarithmic derivatives keeps
13// exponential and fractional-power factors out of the identities being decided.
14//
15// Invertibility is a separate certificate: the coordinate map's Jacobian must not vanish on the
16// chart, and every denominator and fractional power in the closed forms becomes a published chart
17// restriction rather than a silent assumption. The seed must also be a solution at the transformed
18// coordinates, which is a statement about where the seed lives and is recorded as an obligation.
19//
20// Equivalence of two transformation families is decided infinitesimally. Two connected local group
21// actions with the same Lie algebra of generators agree near the identity, so a family built by
22// composing integrated generators reproduces a published family exactly when the generators both
23// induce at the identity span the same space.

143    /// Stores what the action does not claim.

147impl PartialDifferentialEquationPointSymmetryAction {
148    /// Returns `a(x) + A(x) f(X(x))` for a seed and, when admitted, an additive solution.
149    pub fn applied_to_seed(
150        &self,
151        context: &Context,
152        seed: FunctionIdentifier,
153        additive_solution: Option<FunctionIdentifier>,
154    ) -> ExpressionReference {
155        let transformed = self.transformation.applied_to_seed(context, seed);
156        match (self.additive_solution_function, additive_solution) {
157            (Some(_), Some(function)) => {
158                let arguments = self
159                    .transformation
160                    .independent_variables
161                    .iter()
162                    .map(|variable| context.interner.intern(ExpressionKind::Symbol(*variable)))
163                    .collect();
164                builders::sum(
165                    context,
166                    vec![
167                        builders::function_call(context, function, arguments),
168                        transformed,
169                    ],
170                )
171            }
172            _ => transformed,
173        }
174    }
175}

177/// Returns the transformation `f -> outer[inner[f]]`.
178pub fn compose_point_transformations(
179    context: &Context,
180    outer: &PartialDifferentialEquationPointTransformation,
181    inner: &PartialDifferentialEquationPointTransformation,
182) -> Result<
183    PartialDifferentialEquationPointTransformation,
184    PartialDifferentialEquationSymmetryActionDecline,
185> {
186    // `outer[inner[f]](x) = A_o(x) inner[f](X_o(x)) = A_o(x) A_i(X_o(x)) f(X_i(X_o(x)))`.
187    if outer.independent_variables != inner.independent_variables {
188        return Err(PartialDifferentialEquationSymmetryActionDecline::IncompatibleTransformations);
189    }
190    let mapping = coordinate_mapping(
191        context,
192        &outer.independent_variables,
193        &outer.transformed_coordinates,
194    );
195    let mut parameters = outer.parameters.clone();
196    for parameter in &inner.parameters {
197        if !parameters.contains(parameter) {
198            parameters.push(*parameter);
199        }
200    }
201    Ok(PartialDifferentialEquationPointTransformation {
202        independent_variables: outer.independent_variables.clone(),
203        parameters,
204        transformed_coordinates: inner
205            .transformed_coordinates
206            .iter()
207            .map(|coordinate| substitute(context, coordinate, &mapping))
208            .collect(),
209        amplitude: builders::product(
210            context,
211            vec![
212                outer.amplitude.clone(),
213                substitute(context, &inner.amplitude, &mapping),
214            ],
215        ),
216    })
217}

219/// Independently verifies that a transformation carries solutions of an equation to solutions.
220pub fn verify_point_transformation_action(
221    context: &Context,
222    problem: &PartialDifferentialEquationProblem,
223    transformation: &PartialDifferentialEquationPointTransformation,
224) -> PartialDifferentialEquationPointTransformationVerification {
225    let mut chart_obligations = Vec::new();
226    for expression in transformation
227        .transformed_coordinates
228        .iter()
229        .chain(std::iter::once(&transformation.amplitude))
230    {
231        collect_chart_obligations(context, expression, &mut chart_obligations);
232    }
233    let (intertwining, multiplier) = intertwining_identity(context, problem, transformation);
234    let (invertibility, jacobian) =
235        invertibility_on_the_chart(context, transformation, &mut chart_obligations);
236    PartialDifferentialEquationPointTransformationVerification {
237        intertwining,
238        multiplier,
239        jacobian,
240        invertibility,
241        chart_obligations,
242    }
243}

245/// Returns the generators a transformation family induces at the identity.
246pub fn infinitesimal_generators_of_point_transformation(
247    context: &Context,
248    problem: &PartialDifferentialEquationProblem,
249    transformation: &PartialDifferentialEquationPointTransformation,
250) -> Result<
251    Vec<PartialDifferentialEquationPointSymmetryGenerator>,
252    PartialDifferentialEquationSymmetryActionDecline,
253> {
254    // With every parameter at zero the family must be the identity, and then
255    // `d/de (A f(X)) = (dA/de) f + sum_a (dX^a/de) f_a` at the origin is `-(alpha u - xi.grad u)`,
256    // the negated characteristic of `xi d/dx + alpha u d/du`.
257    let mut origin = SubstitutionMap::new();
258    for parameter in &transformation.parameters {
259        origin.insert(
260            context.interner.intern(ExpressionKind::Symbol(*parameter)),
261            builders::zero(context),
262        );
263    }
264    for (coordinate, variable) in transformation
265        .transformed_coordinates
266        .iter()
267        .zip(&transformation.independent_variables)
268    {
269        let residual = builders::subtract(
270            context,
271            substitute(context, coordinate, &origin),
272            context.interner.intern(ExpressionKind::Symbol(*variable)),
273        );
274        if !identity_holds(context, &residual) {
275            return Err(
276                PartialDifferentialEquationSymmetryActionDecline::NotTheIdentityAtTheOrigin,
277            );
278        }
279    }
280    let amplitude_residual = builders::subtract(
281        context,
282        substitute(context, &transformation.amplitude, &origin),
283        builders::one(context),
284    );
285    if !identity_holds(context, &amplitude_residual) {
286        return Err(PartialDifferentialEquationSymmetryActionDecline::NotTheIdentityAtTheOrigin);
287    }
288    let arguments: Vec<ExpressionReference> = transformation
289        .independent_variables
290        .iter()
291        .map(|variable| context.interner.intern(ExpressionKind::Symbol(*variable)))
292        .collect();
293    let dependent = builders::function_call(context, problem.dependent_function(), arguments);
294    Ok(transformation
295        .parameters
296        .iter()
297        .map(|parameter| {
298            let coordinate_infinitesimals = transformation
299                .transformed_coordinates
300                .iter()
301                .map(|coordinate| {
302                    simplified(
303                        context,
304                        &substitute(
305                            context,
306                            &differentiate(context, coordinate, *parameter),
307                            &origin,
308                        ),
309                    )
310                })
311                .collect();
312            let amplitude_rate = substitute(
313                context,
314                &differentiate(context, &transformation.amplitude, *parameter),
315                &origin,
316            );
317            PartialDifferentialEquationPointSymmetryGenerator {
318                coordinate_infinitesimals,
319                dependent_infinitesimal: simplified(
320                    context,
321                    &builders::product(
322                        context,
323                        vec![builders::negate(context, amplitude_rate), dependent.clone()],
324                    ),
325                ),
326            }
327        })
328        .collect())
329}

331/// Decides whether two lists of fiber-linear generators are bases of the same space.
332pub fn point_symmetry_generators_span_the_same_algebra(
333    context: &Context,
334    problem: &PartialDifferentialEquationProblem,
335    first: &[PartialDifferentialEquationPointSymmetryGenerator],
336    second: &[PartialDifferentialEquationPointSymmetryGenerator],
337) -> PartialDifferentialEquationVerificationVerdict {
338    // Each generator becomes the row of its polynomial coefficients over constants, component by
339    // component, so the span over constants is a matrix rank. Both lists must be independent, of
340    // the same length, and their union may not raise the rank.
341    use PartialDifferentialEquationVerificationVerdict as Verdict;
342    if first.is_empty() && second.is_empty() {
343        return Verdict::Verified;
344    }
345    if first.len() != second.len() || first.is_empty() {
346        return Verdict::Rejected;
347    }
348    let jet_space = JetSpace::new(
349        context,
350        problem.independent_variables(),
351        problem.dependent_function(),
352        0,
353    );
354    let (Some(first_rows), Some(second_rows)) = (
355        coefficient_rows(context, problem, &jet_space, first),
356        coefficient_rows(context, problem, &jet_space, second),
357    ) else {
358        return Verdict::Inconclusive;
359    };
360    let keys: BTreeSet<(usize, Vec<i64>)> = first_rows
361        .iter()
362        .chain(&second_rows)
363        .flat_map(|row| row.keys().cloned())
364        .collect();
365    let rank = |rows: &[BTreeMap<(usize, Vec<i64>), ExpressionReference>]| {
366        let matrix_rows = rows
367            .iter()
368            .map(|row| {
369                keys.iter()
370                    .map(|key| {
371                        row.get(key)
372                            .cloned()
373                            .unwrap_or_else(|| builders::zero(context))
374                    })
375                    .collect()
376            })
377            .collect();
378        crate::matrix::matrix_from_rows(matrix_rows)
379            .ok()
380            .and_then(|matrix| crate::matrix::matrix_rank(context, &matrix).ok())
381    };
382    let union: Vec<_> = first_rows.iter().chain(&second_rows).cloned().collect();
383    match (rank(&first_rows), rank(&second_rows), rank(&union)) {
384        (Some(first_rank), Some(second_rank), Some(union_rank)) => {
385            if first_rank == first.len() && second_rank == second.len() && union_rank == first.len()
386            {
387                Verdict::Verified
388            } else {
389                Verdict::Rejected
390            }
391        }
392        _ => Verdict::Inconclusive,
393    }
394}

396/// Integrates and composes every basis generator into the connected local symmetry action.
397pub fn construct_connected_point_symmetry_action(
398    context: &Context,
399    problem: &PartialDifferentialEquationProblem,
400    algebra: &PartialDifferentialEquationPointSymmetryAlgebra,
401) -> Result<
402    PartialDifferentialEquationPointSymmetryAction,
403    PartialDifferentialEquationSymmetryActionDecline,
404> {
405    // Each one-parameter group is verified directly by the intertwining identity, which never
406    // consults its generator. The composed action is then certified by composition rather than by
407    // expanding its own identity, whose coefficients are rational functions of every group parameter
408    // at once: if `L(T_o g) = M_o (L g)(X_o)` and `L(T_i f) = M_i (L f)(X_i)`, then
409    // `L(T_o T_i f) = M_o M_i(X_o) (L f)(X_i(X_o))`, and the Jacobian of `X_i(X_o)` is
410    // `J_o J_i(X_o)` by the chain rule. Both products are exact, and each factor's chart restrictions
411    // are carried to the coordinates they are evaluated at. The composed action's generators at the
412    // identity must still span the derived algebra, which is what makes it the whole connected action.
413    use PartialDifferentialEquationSymmetryActionDecline as Decline;
414    let mut groups = Vec::with_capacity(algebra.basis.len());
415    let mut group_verifications = Vec::with_capacity(algebra.basis.len());
416    for generator in &algebra.basis {
417        let parameter = allocate_partial_differential_equation_symbol(context, "group_parameter");
418        let group = integrate_point_symmetry_generator(
419            context,
420            problem.independent_variables(),
421            problem.dependent_function(),
422            generator,
423            parameter,
424        )
425        .map_err(Decline::OneParameterGroupNotIntegrated)?;
426        let verification =
427            verify_point_transformation_action(context, problem, &group.transformation);
428        if verification.intertwining != PartialDifferentialEquationVerificationVerdict::Verified
429            || verification.invertibility
430                != PartialDifferentialEquationVerificationVerdict::Verified
431        {
432            return Err(Decline::ActionNotVerified);
433        }
434        groups.push(group);
435        group_verifications.push(verification);
436    }
437    let mut transformation = identity_transformation(context, problem.independent_variables());
438    let mut multiplier = builders::one(context);
439    let mut jacobian = builders::one(context);
440    let mut chart_obligations: Vec<PartialDifferentialEquationChartObligation> = Vec::new();
441    for (group, verification) in groups.iter().zip(&group_verifications) {
442        let (Some(group_multiplier), Some(group_jacobian)) =
443            (&verification.multiplier, &verification.jacobian)
444        else {
445            return Err(Decline::ActionNotVerified);
446        };
447        let at_outer = coordinate_mapping(
448            context,
449            &transformation.independent_variables,
450            &transformation.transformed_coordinates,
451        );
452        multiplier = builders::product(
453            context,
454            vec![multiplier, substitute(context, group_multiplier, &at_outer)],
455        );
456        jacobian = builders::product(
457            context,
458            vec![jacobian, substitute(context, group_jacobian, &at_outer)],
459        );
460        for obligation in &verification.chart_obligations {
461            let carried = match obligation {
462                PartialDifferentialEquationChartObligation::NonzeroExpression(expression) => {
463                    PartialDifferentialEquationChartObligation::NonzeroExpression(substitute(
464                        context, expression, &at_outer,
465                    ))
466                }
467                PartialDifferentialEquationChartObligation::PositiveExpression(expression) => {
468                    PartialDifferentialEquationChartObligation::PositiveExpression(substitute(
469                        context, expression, &at_outer,
470                    ))
471                }
472            };
473            if !chart_obligations.contains(&carried) {
474                chart_obligations.push(carried);
475            }
476        }
477        transformation =
478            compose_point_transformations(context, &transformation, &group.transformation)?;
479    }
480    let verification = PartialDifferentialEquationPointTransformationVerification {
481        intertwining: PartialDifferentialEquationVerificationVerdict::Verified,
482        multiplier: Some(multiplier),
483        jacobian: Some(jacobian),
484        invertibility: PartialDifferentialEquationVerificationVerdict::Verified,
485        chart_obligations,
486    };
487    let induced =
488        infinitesimal_generators_of_point_transformation(context, problem, &transformation)?;
489    if point_symmetry_generators_span_the_same_algebra(context, problem, &induced, &algebra.basis)
490        != PartialDifferentialEquationVerificationVerdict::Verified
491    {
492        return Err(Decline::GeneratorsDoNotSpanTheAlgebra);
493    }
494    let mut proved_identities: Vec<PartialDifferentialEquationSymmetryProvedIdentity> = groups
495        .iter()
496        .flat_map(|group| group.proved_identities.iter().cloned())
497        .collect();
498    proved_identities.push(PartialDifferentialEquationSymmetryProvedIdentity {
499        identity:
500            PartialDifferentialEquationSymmetryIdentity::ActionIntertwinesTheGoverningOperator,
501        verdict: verification.intertwining,
502    });
503    proved_identities.push(PartialDifferentialEquationSymmetryProvedIdentity {
504        identity: PartialDifferentialEquationSymmetryIdentity::CoordinateMapIsInvertibleOnTheChart,
505        verdict: verification.invertibility,
506    });
507    if algebra.additive_solution_function.is_some() {
508        proved_identities.push(PartialDifferentialEquationSymmetryProvedIdentity {
509            identity: PartialDifferentialEquationSymmetryIdentity::LinearSuperposition,
510            verdict: verify_linear_superposition(context, problem, jet_order(problem)),
511        });
512    }
513    let additive_solution_function = algebra.additive_solution_function.map(|_| {
514        allocate_partial_differential_equation_arbitrary_function(
515            context,
516            "additive_solution",
517            problem.independent_variables().len(),
518        )
519        .function
520    });
521    Ok(PartialDifferentialEquationPointSymmetryAction {
522        transformation,
523        additive_solution_function,
524        one_parameter_groups: groups,
525        verification,
526        proved_identities,
527        outstanding_obligations: vec![
528            PartialDifferentialEquationSymmetryObligation::TransformedCoordinatesLieInTheSeedChart,
529            PartialDifferentialEquationSymmetryObligation::DiscretePointSymmetriesNotDerived,
530        ],
531    })
532}

557    problem: &PartialDifferentialEquationProblem,
558    transformation: &PartialDifferentialEquationPointTransformation,

569fn intertwining_identity_or_none(
570    context: &Context,
571    problem: &PartialDifferentialEquationProblem,
572    transformation: &PartialDifferentialEquationPointTransformation,
573) -> Option<(
574    PartialDifferentialEquationVerificationVerdict,
575    Option<ExpressionReference>,
576)> {
577    use PartialDifferentialEquationVerificationVerdict as Verdict;
578    let order = jet_order(problem);
579    let variables = problem.independent_variables();
580    if order == 0 || transformation.independent_variables != variables {
581        return None;
582    }
583    let equation_space = JetSpace::new(context, variables, problem.dependent_function(), order);
584    let Some(residual) = equation_space.to_jet_coordinates(context, problem.equation()) else {
585        return None;
586    };
587    let Some(principal) = equation_space.solve_for_principal_jet(context, &residual) else {
588        return None;
589    };
590    let seed =
591        allocate_partial_differential_equation_arbitrary_function(context, "seed", variables.len())
592            .function;
593    let seed_space = JetSpace::new(context, variables, seed, order);
594    let dimension = variables.len();
595    let mut composite: BTreeMap<Vec<usize>, ExpressionReference> = BTreeMap::new();
596    let Some(seed_value) = seed_space.coordinate(context, &vec![0; dimension]) else {
597        return None;
598    };
599    composite.insert(vec![0; dimension], seed_value);
600    for total in 1..=order {
601        for orders in multi_indices_of_total_order(dimension, total) {
602            let Some(direction) = orders.iter().position(|value| *value > 0) else {
603                return None;
604            };
605            let mut lowered = orders.clone();
606            lowered[direction] -= 1;
607            let Some(derivative) = composite_derivative(
608                context,
609                &seed_space,
610                composite.get(&lowered)?,
611                variables[direction],
612                &transformation.transformed_coordinates,
613            ) else {
614                return None;
615            };
616            composite.insert(orders, derivative);
617        }
618    }
619    let at_transformed =
620        coordinate_mapping(context, variables, &transformation.transformed_coordinates);
621    let linear = verify_linear_superposition(context, problem, order) == Verdict::Verified;
622    let (transformed_side, seed_side) = if linear {
623        let logarithmic: Vec<ExpressionReference> = variables
624            .iter()
625            .map(|variable| logarithmic_derivative(context, &transformation.amplitude, *variable))
626            .collect();
627        let mut ratios: BTreeMap<Vec<usize>, ExpressionReference> = BTreeMap::new();
628        ratios.insert(vec![0; dimension], builders::one(context));
629        for total in 1..=order {
630            for orders in multi_indices_of_total_order(dimension, total) {
631                let direction = orders.iter().position(|value| *value > 0).unwrap_or(0);
632                let mut lowered = orders.clone();
633                lowered[direction] -= 1;
634                let previous = ratios.get(&lowered)?.clone();
635                let value = builders::sum(
636                    context,
637                    vec![
638                        differentiate(context, &previous, variables[direction]),
639                        builders::product(context, vec![logarithmic[direction].clone(), previous]),
640                    ],
641                );
642                ratios.insert(orders, value);
643            }
644        }
645        let mut transformed_terms = Vec::new();
646        let mut seed_terms = Vec::new();
647        for total in 0..=order {
648            for orders in multi_indices_of_total_order(dimension, total) {
649                let Some(jet) = equation_space.coordinate(context, &orders) else {
650                    return None;
651                };
652                let ExpressionKind::Symbol(jet_symbol) = jet.kind else {
653                    return None;
654                };
655                let coefficient = differentiate(context, &residual, jet_symbol);
656                if is_structural_zero(&coefficient) {
657                    continue;
658                }
659                for lower in dominated_multi_indices(&orders) {
660                    let difference: Vec<usize> = orders
661                        .iter()
662                        .zip(&lower)
663                        .map(|(high, low)| high - low)
664                        .collect();
665                    transformed_terms.push(builders::product(
666                        context,
667                        vec![
668                            coefficient.clone(),
669                            builders::integer_value(context, multi_binomial(&orders, &lower)),
670                            ratios.get(&difference)?.clone(),
671                            composite.get(&lower)?.clone(),
672                        ],
673                    ));
674                }
675                let Some(seed_jet) = seed_space.coordinate(context, &orders) else {
676                    return None;
677                };
678                seed_terms.push(builders::product(
679                    context,
680                    vec![substitute(context, &coefficient, &at_transformed), seed_jet],
681                ));
682            }
683        }
684        (
685            builders::sum(context, transformed_terms),
686            builders::sum(context, seed_terms),
687        )
688    } else {
689        let mut transformed_jets: BTreeMap<Vec<usize>, ExpressionReference> = BTreeMap::new();
690        let amplified = builders::product(
691            context,
692            vec![
693                transformation.amplitude.clone(),
694                composite.get(&vec![0; dimension])?.clone(),
695            ],
696        );
697        transformed_jets.insert(vec![0; dimension], amplified);
698        for total in 1..=order {
699            for orders in multi_indices_of_total_order(dimension, total) {
700                let direction = orders.iter().position(|value| *value > 0).unwrap_or(0);
701                let mut lowered = orders.clone();
702                lowered[direction] -= 1;
703                let Some(derivative) = composite_derivative(
704                    context,
705                    &seed_space,
706                    transformed_jets.get(&lowered)?,
707                    variables[direction],
708                    &transformation.transformed_coordinates,
709                ) else {
710                    return None;
711                };
712                transformed_jets.insert(orders, derivative);
713            }
714        }
715        let mut to_transformed = SubstitutionMap::new();
716        let mut to_seed = at_transformed.clone();
717        for total in 0..=order {
718            for orders in multi_indices_of_total_order(dimension, total) {
719                let (Some(jet), Some(seed_jet)) = (
720                    equation_space.coordinate(context, &orders),
721                    seed_space.coordinate(context, &orders),
722                ) else {
723                    return None;
724                };
725                to_transformed.insert(jet.clone(), transformed_jets.get(&orders)?.clone());
726                to_seed.insert(jet, seed_jet);
727            }
728        }
729        (
730            substitute(context, &residual, &to_transformed),
731            substitute(context, &residual, &to_seed),
732        )
733    };
734    let Some(principal_seed_jet) = seed_space.coordinate(context, &principal.principal_orders)
735    else {
736        return None;
737    };
738    let ExpressionKind::Symbol(principal_seed_symbol) = principal_seed_jet.kind else {
739        return None;
740    };
741    let multiplier = simplified(
742        context,
743        &builders::divide(
744            context,
745            differentiate(context, &transformed_side, principal_seed_symbol),
746            differentiate(context, &seed_side, principal_seed_symbol),
747        ),
748    );
749    let difference = builders::subtract(
750        context,
751        transformed_side,
752        builders::product(context, vec![multiplier.clone(), seed_side]),
753    );
754    let coefficients =
755        match laurent_monomial_coefficients(context, &difference, &seed_space.all_symbols()) {
756            Some(coefficients) => coefficients.into_iter().map(|(_, value)| value).collect(),
757            None => vec![difference],
758        };
759    let mut verdict = Verdict::Verified;
760    let mut undecided = Vec::new();
761    for coefficient in coefficients {
762        match decide_identity(context, &coefficient) {
763            ZeroEquivalenceVerdict::TrueValue => {}
764            ZeroEquivalenceVerdict::FalseValue => {
765                return Some((Verdict::Rejected, Some(multiplier)));
766            }
767            ZeroEquivalenceVerdict::Unknown => {
768                verdict = Verdict::Inconclusive;
769                undecided.push(coefficient);
770            }
771        }
772    }
773    if refuted_on_a_one_parameter_subfamily(context, &transformation.parameters, &undecided) {
774        return Some((Verdict::Rejected, Some(multiplier)));
775    }
776    if verdict == Verdict::Verified
777        && prove_residual_zero(context, &multiplier) == ZeroEquivalenceVerdict::TrueValue
778    {
779        verdict = Verdict::Rejected;
780    }
781    Some((verdict, Some(multiplier)))
782}

784fn refuted_on_a_one_parameter_subfamily(
785    context: &Context,
786    parameters: &[SymbolIdentifier],
787    undecided: &[ExpressionReference],
788) -> bool {
789    // A family is claimed to act for every parameter value near the origin, so its identities must
790    // survive setting all parameters but one to zero. The one-parameter subfamily often leaves only
791    // rational dependence, where a nonzero numerator refutes the identity outright; an identity
792    // that is undecided on every subfamily stays undecided.
793    if parameters.len() < 2 {
794        return false;
795    }
796    for kept in parameters {
797        let mapping: SubstitutionMap = parameters
798            .iter()
799            .filter(|parameter| *parameter != kept)
800            .map(|parameter| {
801                (
802                    context.interner.intern(ExpressionKind::Symbol(*parameter)),
803                    builders::zero(context),
804                )
805            })
806            .collect();
807        for coefficient in undecided {
808            let specialized = substitute(context, coefficient, &mapping);
809            if decide_identity(context, &specialized) == ZeroEquivalenceVerdict::FalseValue {
810                return true;
811            }
812        }
813    }
814    false
815}

817fn composite_derivative(
818    context: &Context,
819    seed_space: &JetSpace,
820    expression: &ExpressionReference,
821    variable: SymbolIdentifier,
822    transformed_coordinates: &[ExpressionReference],
823) -> Option<ExpressionReference> {
824    // The chain rule through `X`: explicit dependence on the variable, plus every seed jet raised
825    // in each slot and weighted by that slot's coordinate derivative.
826    let mut terms = vec![differentiate(context, expression, variable)];
827    let coordinate_derivatives: Vec<ExpressionReference> = transformed_coordinates
828        .iter()
829        .map(|coordinate| differentiate(context, coordinate, variable))
830        .collect();
831    for symbol in seed_space.all_symbols() {
832        if !depends_on(expression, symbol) {
833            continue;
834        }
835        let orders = seed_space.multi_index_of(symbol)?;
836        let mut chain = Vec::new();
837        for (slot, coordinate_derivative) in coordinate_derivatives.iter().enumerate() {
838            if is_structural_zero(coordinate_derivative) {
839                continue;
840            }
841            let mut raised = orders.clone();
842            raised[slot] += 1;
843            if raised.iter().sum::<usize>() > seed_space.order() {
844                return None;
845            }
846            chain.push(builders::product(
847                context,
848                vec![
849                    seed_space.coordinate(context, &raised)?,
850                    coordinate_derivative.clone(),
851                ],
852            ));
853        }
854        terms.push(builders::product(
855            context,
856            vec![
857                builders::sum(context, chain),
858                differentiate(context, expression, symbol),
859            ],
860        ));
861    }
862    Some(builders::sum(context, terms))
863}

865/// Returns the logarithmic derivative of a product of exponentials, powers, and other factors.
866pub(crate) fn logarithmic_derivative(
867    context: &Context,
868    expression: &ExpressionReference,
869    variable: SymbolIdentifier,
870) -> ExpressionReference {
871    // Structural and exact wherever the factors are nonzero: a product contributes the sum of its
872    // factors' logarithmic derivatives, `exp(E)` contributes `E'`, and `B^r` with constant `r`
873    // contributes `r B'/B`. No exponential or fractional power survives into the result.
874    if !depends_on(expression, variable) {
875        return builders::zero(context);
876    }
877    match &expression.kind {
878        ExpressionKind::Product(factors) => builders::sum(
879            context,
880            factors
881                .iter()
882                .map(|factor| logarithmic_derivative(context, factor, variable))
883                .collect(),
884        ),
885        ExpressionKind::FunctionCall(
886            FunctionIdentifier::Builtin(BuiltinFunction::Exp),
887            arguments,
888        ) if arguments.len() == 1 => differentiate(context, &arguments[0], variable),
889        ExpressionKind::Power(base, exponent) if !depends_on(exponent, variable) => {
890            builders::product(
891                context,
892                vec![
893                    exponent.clone(),
894                    logarithmic_derivative(context, base, variable),
895                ],
896            )
897        }
898        _ => builders::divide(
899            context,
900            differentiate(context, expression, variable),
901            expression.clone(),
902        ),
903    }
904}

906fn invertibility_on_the_chart(
907    context: &Context,
908    transformation: &PartialDifferentialEquationPointTransformation,
909    chart_obligations: &mut Vec<PartialDifferentialEquationChartObligation>,
910) -> (
911    PartialDifferentialEquationVerificationVerdict,
912    Option<ExpressionReference>,
913) {
914    // The Jacobian is decided factor by factor after combining over a common denominator: an
915    // exponential never vanishes, a nonzero constant never does, and any other factor is either
916    // identically zero, which refutes invertibility, or becomes a nonvanishing chart restriction.
917    use PartialDifferentialEquationVerificationVerdict as Verdict;
918    let variables = &transformation.independent_variables;
919    let rows = transformation
920        .transformed_coordinates
921        .iter()
922        .map(|coordinate| {
923            variables
924                .iter()
925                .map(|variable| differentiate(context, coordinate, *variable))
926                .collect()
927        })
928        .collect();
929    let Some(jacobian) = crate::matrix::matrix_from_rows(rows)
930        .ok()
931        .and_then(|matrix| crate::matrix::matrix_determinant(context, &matrix).ok())
932    else {
933        return (Verdict::Inconclusive, None);
934    };
935    let jacobian = simplified(context, &jacobian);
936    if prove_residual_zero(context, &jacobian) == ZeroEquivalenceVerdict::TrueValue {
937        return (Verdict::Rejected, Some(jacobian));
938    }
939    let combined = crate::simplify::rational::together(context, &jacobian);
940    for factor in product_factors(&combined) {
941        nonvanishing_factor(context, variables, &factor, chart_obligations);
942    }
943    (Verdict::Verified, Some(jacobian))
944}
945
946fn nonvanishing_factor(
947    context: &Context,
948    variables: &[SymbolIdentifier],
949    factor: &ExpressionReference,
950    chart_obligations: &mut Vec<PartialDifferentialEquationChartObligation>,
951) {
952    match &factor.kind {
953        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(BuiltinFunction::Exp), _) => {}
954        ExpressionKind::Power(base, _) => {
955            nonvanishing_factor(context, variables, base, chart_obligations);
956        }
957        _ => {
958            let constant = !free_variables(factor)
959                .iter()
960                .any(|symbol| variables.contains(symbol));
961            if constant
962                && zero_equivalent(context, factor, None) == ZeroEquivalenceVerdict::FalseValue
963            {
964                return;
965            }
966            let obligation =
967                PartialDifferentialEquationChartObligation::NonzeroExpression(factor.clone());
968            if !chart_obligations.contains(&obligation) {
969                chart_obligations.push(obligation);
970            }
971        }
972    }
973}
974
975fn collect_chart_obligations(
976    context: &Context,
977    expression: &ExpressionReference,
978    chart_obligations: &mut Vec<PartialDifferentialEquationChartObligation>,
979) {
980    let obligation = match &expression.kind {
981        ExpressionKind::Power(base, exponent) => match &exponent.kind {
982            ExpressionKind::Rational(_) => {
983                Some(PartialDifferentialEquationChartObligation::PositiveExpression(base.clone()))
984            }
985            ExpressionKind::Integer(value) if *value < crate::exact_arithmetic::Integer::ZERO => {
986                Some(PartialDifferentialEquationChartObligation::NonzeroExpression(base.clone()))
987            }
988            _ => None,
989        },
990        ExpressionKind::FunctionCall(
991            FunctionIdentifier::Builtin(BuiltinFunction::Log),
992            arguments,
993        ) if arguments.len() == 1 => Some(
994            PartialDifferentialEquationChartObligation::PositiveExpression(arguments[0].clone()),
995        ),
996        _ => None,
997    };
998    if let Some(obligation) = obligation {
999        let base = match &obligation {
1000            PartialDifferentialEquationChartObligation::NonzeroExpression(base)
1001            | PartialDifferentialEquationChartObligation::PositiveExpression(base) => base.clone(),
1002        };
1003        let constant = free_variables(&base).is_empty()
1004            && zero_equivalent(context, &base, None) == ZeroEquivalenceVerdict::FalseValue;
1005        if !constant && !chart_obligations.contains(&obligation) {
1006            chart_obligations.push(obligation);
1007        }
1008    }
1009    let _ =
1010        crate::differential_equations::partial_differential_equations::problem::traverse_children(
1011            expression,
1012            |child| {
1013                collect_chart_obligations(context, child, chart_obligations);
1014                Ok(())
1015            },
1016        );
1017}

1019fn coefficient_rows(
1020    context: &Context,
1021    problem: &PartialDifferentialEquationProblem,
1022    jet_space: &JetSpace,
1023    generators: &[PartialDifferentialEquationPointSymmetryGenerator],
1024) -> Option<Vec<BTreeMap<(usize, Vec<i64>), ExpressionReference>>> {
1025    let variables = problem.independent_variables();
1026    let dependent_value = jet_space.dependent_value();
1027    let value_expression = context
1028        .interner
1029        .intern(ExpressionKind::Symbol(dependent_value));
1030    let mut rows = Vec::with_capacity(generators.len());
1031    for generator in generators {
1032        if generator.coordinate_infinitesimals.len() != variables.len() {
1033            return None;
1034        }
1035        let mut components = Vec::with_capacity(variables.len() + 1);
1036        for component in &generator.coordinate_infinitesimals {
1037            let in_jets = jet_space.to_jet_coordinates(context, component)?;
1038            if depends_on(&in_jets, dependent_value) {
1039                return None;
1040            }
1041            components.push(in_jets);
1042        }
1043        let dependent =
1044            jet_space.to_jet_coordinates(context, &generator.dependent_infinitesimal)?;
1045        let coefficient = differentiate(context, &dependent, dependent_value);
1046        if depends_on(&coefficient, dependent_value)
1047            || !identity_holds(
1048                context,
1049                &builders::subtract(
1050                    context,
1051                    dependent,
1052                    builders::product(context, vec![coefficient.clone(), value_expression.clone()]),
1053                ),
1054            )
1055        {
1056            return None;
1057        }
1058        components.push(coefficient);
1059        let mut row = BTreeMap::new();
1060        for (index, component) in components.iter().enumerate() {
1061            let expanded = crate::expand::expand(context, &simplified(context, component));
1062            for (exponents, value) in laurent_monomial_coefficients(context, &expanded, variables)?
1063            {
1064                row.insert((index, exponents), value);
1065            }
1066        }
1067        rows.push(row);
1068    }
1069    Some(rows)
1070}

1089fn dominated_multi_indices(orders: &[usize]) -> Vec<Vec<usize>> {
1090    let mut result = vec![Vec::new()];
1091    for order in orders {
1092        let mut extended = Vec::new();
1093        for prefix in &result {
1094            for value in 0..=*order {
1095                let mut next = prefix.clone();
1096                next.push(value);
1097                extended.push(next);
1098            }
1099        }
1100        result = extended;
1101    }
1102    result
1103}

1105fn multi_binomial(orders: &[usize], lower: &[usize]) -> i64 {
1106    orders
1107        .iter()
1108        .zip(lower)
1109        .map(|(high, low)| binomial(*high, *low))
1110        .product()
1111}
1112
1113fn binomial(high: usize, low: usize) -> i64 {
1114    (0..low).fold(1i64, |accumulated, step| {
1115        accumulated * (high - step) as i64 / (step + 1) as i64
1116    })
1117}
