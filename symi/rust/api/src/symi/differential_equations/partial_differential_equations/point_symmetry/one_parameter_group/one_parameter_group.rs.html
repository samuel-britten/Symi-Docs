1// Finite one-parameter transformation groups generated by fiber-linear point symmetries.
2//
3// A generator `V = sum_i xi^i(x) d/dx_i + alpha(x) u d/du` integrates to a flow on the base
4// coordinates and a multiplicative flow on the dependent value:
5//
6//     dY/d(eps) = xi(Y),  Y(0) = x,        d(log A)/d(eps) = alpha(Y),  log A(0) = 0.
7//
8// The base flow is solved one coordinate at a time, in an order in which each infinitesimal depends
9// only on coordinates already solved and on its own, which is the triangular structure the
10// generators of evolution equations have. A coordinate whose infinitesimal does not involve itself
11// is a quadrature; one that does is a scalar first-order equation handed to the tracked
12// first-order routes, with its integration constant fitted exactly to the initial value. Nothing is
13// trusted because a route returned it: every flow coordinate is checked by differentiation and at
14// the origin, and the logarithmic amplitude likewise.
15//
16// The group acts on solutions through graphs. `exp(-eps V)` carries the graph of a solution `f` to
17// the graph of `f(Y_eps(x)) / A_eps(x)`, so the action is recorded as the transformation with
18// coordinates `Y_eps` and amplitude `exp(-log A_eps)`, which composes by substitution and needs no
19// inverse map to be computed.

49/// Public native enumeration named `partial_differential_equation_one_parameter_group_decline`.
50#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
51pub enum PartialDifferentialEquationOneParameterGroupDecline {
52    /// The generator's coordinate count does not match the independent variables.
53    DimensionMismatch,
54    /// The generator's coordinate infinitesimals depend on the dependent value, or its dependent
55    /// infinitesimal is not a coordinate function times the dependent value.
56    GeneratorNotFiberLinear,
57    /// No coordinate ordering makes the base flow triangular.
58    FlowNotTriangular,
59    /// A scalar flow equation or quadrature was not solved exactly.
60    FlowEquationNotSolved,
61    /// A flow coordinate did not verify against its equation and initial value.
62    FlowNotVerified,
63    /// The amplitude quadrature was not evaluated exactly.
64    AmplitudeQuadratureNotEvaluated,
65    /// The logarithmic amplitude did not verify against its equation and initial value.
66    AmplitudeNotVerified,
67}

88/// Integrates a fiber-linear point-symmetry generator to its one-parameter group.
89pub fn integrate_point_symmetry_generator(
90    context: &Context,
91    independent_variables: &[SymbolIdentifier],
92    dependent_function: FunctionIdentifier,
93    generator: &PartialDifferentialEquationPointSymmetryGenerator,
94    parameter: SymbolIdentifier,
95) -> Result<
96    PartialDifferentialEquationOneParameterGroup,
97    PartialDifferentialEquationOneParameterGroupDecline,
98> {
99    use PartialDifferentialEquationOneParameterGroupDecline as Decline;
100    if generator.coordinate_infinitesimals.len() != independent_variables.len() {
101        return Err(Decline::DimensionMismatch);
102    }
103    let (coordinate_infinitesimals, coefficient) = fiber_linear_parts(
104        context,
105        independent_variables,
106        dependent_function,
107        generator,
108    )
109    .ok_or(Decline::GeneratorNotFiberLinear)?;
110    let mut chart_obligations = Vec::new();
111    let mut flow: Vec<Option<ExpressionReference>> = vec![None; independent_variables.len()];
112    while flow.iter().any(Option::is_none) {
113        let next = (0..independent_variables.len()).find(|index| {
114            flow[*index].is_none()
115                && independent_variables
116                    .iter()
117                    .enumerate()
118                    .all(|(other, variable)| {
119                        other == *index
120                            || flow[other].is_some()
121                            || !depends_on(&coordinate_infinitesimals[*index], *variable)
122                    })
123        });
124        let index = next.ok_or(Decline::FlowNotTriangular)?;
125        let solved_mapping = solved_coordinate_mapping(context, independent_variables, &flow);
126        let coordinate = solve_flow_coordinate(
127            context,
128            independent_variables[index],
129            &coordinate_infinitesimals[index],
130            &solved_mapping,
131            parameter,
132            &mut chart_obligations,
133        )?;
134        let coordinate =
135            logarithms_in_exponentials_as_powers(context, &coordinate, &mut chart_obligations);
136        flow[index] = Some(coordinate);
137    }
138    let flow: Vec<ExpressionReference> = flow.into_iter().flatten().collect();
139    let along_flow = solved_coordinate_mapping(
140        context,
141        independent_variables,
142        &flow.iter().cloned().map(Some).collect::<Vec<_>>(),
143    );
144    let coefficient_along_flow = substitute(context, &coefficient, &along_flow);
145    let logarithmic_amplitude = definite_quadrature(
146        context,
147        &coefficient_along_flow,
148        parameter,
149        &mut chart_obligations,
150    )
151    .ok_or(Decline::AmplitudeQuadratureNotEvaluated)?;
152    let amplitude_verified = identity_holds(
153        context,
154        &builders::subtract(
155            context,
156            differentiate(context, &logarithmic_amplitude, parameter),
157            coefficient_along_flow,
158        ),
159    ) && identity_holds(
160        context,
161        &at_origin(context, &logarithmic_amplitude, parameter),
162    );
163    if !amplitude_verified {
164        return Err(Decline::AmplitudeNotVerified);
165    }
166    let amplitude = builders::function_call(
167        context,
168        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
169        vec![builders::negate(context, logarithmic_amplitude.clone())],
170    );
171    let amplitude =
172        logarithms_in_exponentials_as_powers(context, &amplitude, &mut chart_obligations);
173    Ok(PartialDifferentialEquationOneParameterGroup {
174        generator: generator.clone(),
175        parameter,
176        transformation: PartialDifferentialEquationPointTransformation {
177            independent_variables: independent_variables.to_vec(),
178            parameters: vec![parameter],
179            transformed_coordinates: flow.clone(),
180            amplitude,
181        },
182        flow,
183        logarithmic_amplitude,
184        proved_identities: vec![PartialDifferentialEquationSymmetryProvedIdentity {
185            identity: PartialDifferentialEquationSymmetryIdentity::FlowIntegratesTheGenerator,
186            verdict: PartialDifferentialEquationVerificationVerdict::Verified,
187        }],
188        chart_obligations,
189    })
190}

192/// Splits a fiber-linear generator into its base field and the coefficient of the dependent value.
193///
194/// Returns `(xi(x), alpha(x))` for `V = xi d/dx + alpha(x) u d/du`, or `None` when a coordinate
195/// infinitesimal depends on the dependent value or the dependent infinitesimal is not a coordinate
196/// function times it.
197pub(crate) fn fiber_linear_parts(
198    context: &Context,
199    independent_variables: &[SymbolIdentifier],
200    dependent_function: FunctionIdentifier,
201    generator: &PartialDifferentialEquationPointSymmetryGenerator,
202) -> Option<(Vec<ExpressionReference>, ExpressionReference)> {
203    if generator.coordinate_infinitesimals.len() != independent_variables.len() {
204        return None;
205    }
206    let jet_space = JetSpace::new(context, independent_variables, dependent_function, 0);
207    let dependent_value = jet_space.dependent_value();
208    let mut coordinate_infinitesimals = Vec::with_capacity(independent_variables.len());
209    for component in &generator.coordinate_infinitesimals {
210        let in_jets = jet_space.to_jet_coordinates(context, component)?;
211        if depends_on(&in_jets, dependent_value) {
212            return None;
213        }
214        coordinate_infinitesimals.push(in_jets);
215    }
216    let dependent_in_jets =
217        jet_space.to_jet_coordinates(context, &generator.dependent_infinitesimal)?;
218    let coefficient = differentiate(context, &dependent_in_jets, dependent_value);
219    let value_expression = context
220        .interner
221        .intern(ExpressionKind::Symbol(dependent_value));
222    if depends_on(&coefficient, dependent_value)
223        || prove_residual_zero(
224            context,
225            &builders::subtract(
226                context,
227                dependent_in_jets,
228                builders::product(context, vec![coefficient.clone(), value_expression]),
229            ),
230        ) != ZeroEquivalenceVerdict::TrueValue
231    {
232        return None;
233    }
234    Some((coordinate_infinitesimals, coefficient))
235}

237fn solved_coordinate_mapping(
238    context: &Context,
239    independent_variables: &[SymbolIdentifier],
240    flow: &[Option<ExpressionReference>],
241) -> SubstitutionMap {
242    let mut mapping = SubstitutionMap::new();
243    for (variable, value) in independent_variables.iter().zip(flow) {
244        if let Some(value) = value {
245            mapping.insert(
246                context.interner.intern(ExpressionKind::Symbol(*variable)),
247                value.clone(),
248            );
249        }
250    }
251    mapping
252}

254fn solve_flow_coordinate(
255    context: &Context,
256    variable: SymbolIdentifier,
257    infinitesimal: &ExpressionReference,
258    solved_mapping: &SubstitutionMap,
259    parameter: SymbolIdentifier,
260    chart_obligations: &mut Vec<PartialDifferentialEquationChartObligation>,
261) -> Result<ExpressionReference, PartialDifferentialEquationOneParameterGroupDecline> {
262    // The initial value of the coordinate is the coordinate symbol itself. A right-hand side free
263    // of that coordinate is integrated directly; otherwise the running coordinate becomes a fresh
264    // unknown function of the parameter, the equation goes to the tracked first-order routes, and
265    // each route's closed form is fitted and checked until one verifies.
266    use PartialDifferentialEquationOneParameterGroupDecline as Decline;
267    let initial = context.interner.intern(ExpressionKind::Symbol(variable));
268    if !depends_on(infinitesimal, variable) {
269        let right_hand_side = substitute(context, infinitesimal, solved_mapping);
270        let displacement =
271            definite_quadrature(context, &right_hand_side, parameter, chart_obligations)
272                .ok_or(Decline::FlowEquationNotSolved)?;
273        let coordinate = builders::sum(context, vec![initial.clone(), displacement]);
274        return verify_flow_coordinate(
275            context,
276            coordinate,
277            infinitesimal,
278            solved_mapping,
279            variable,
280            parameter,
281        );
282    }
283    let running_function =
284        allocate_partial_differential_equation_arbitrary_function(context, "flow_coordinate", 1)
285            .function;
286    let parameter_expression = context.interner.intern(ExpressionKind::Symbol(parameter));
287    let running = builders::function_call(
288        context,
289        running_function,
290        vec![parameter_expression.clone()],
291    );
292    let mut mapping = solved_mapping.clone();
293    mapping.insert(initial.clone(), running.clone());
294    let right_hand_side = substitute(context, infinitesimal, &mapping);
295    let equation = builders::subtract(
296        context,
297        builders::derivative(
298            context,
299            running_function,
300            vec![1],
301            vec![parameter_expression],
302        ),
303        right_hand_side,
304    );
305    let candidates =
306        solve_ordinary_differential_equation_candidates(context, &equation, &running, parameter);
307    let zero = builders::zero(context);
308    for candidate in candidates {
309        if candidate.generated_constants.len() != 1 {
310            continue;
311        }
312        let constant = candidate.generated_constants[0];
313        let running_value = allocate_partial_differential_equation_symbol(context, "flow_value");
314        let running_value_expression = context
315            .interner
316            .intern(ExpressionKind::Symbol(running_value));
317        let mut to_value = SubstitutionMap::new();
318        to_value.insert(running.clone(), running_value_expression.clone());
319        let relation = match candidate.form {
320            OrdinaryDifferentialEquationSolutionForm::Explicit => builders::subtract(
321                context,
322                candidate.solution.clone(),
323                running_value_expression.clone(),
324            ),
325            OrdinaryDifferentialEquationSolutionForm::Implicit => {
326                substitute(context, &candidate.solution, &to_value)
327            }
328        };
329        let mut at_start = SubstitutionMap::new();
330        at_start.insert(
331            context.interner.intern(ExpressionKind::Symbol(parameter)),
332            zero.clone(),
333        );
334        at_start.insert(running_value_expression.clone(), initial.clone());
335        let initial_relation = substitute(context, &relation, &at_start);
336        let Some(constant_value) = unique_exact_root(context, &initial_relation, constant) else {
337            continue;
338        };
339        let mut fit = SubstitutionMap::new();
340        fit.insert(
341            context.interner.intern(ExpressionKind::Symbol(constant)),
342            constant_value,
343        );
344        let fitted = substitute(context, &relation, &fit);
345        let Some(coordinate) = unique_exact_root(context, &fitted, running_value) else {
346            continue;
347        };
348        if let Ok(verified) = verify_flow_coordinate(
349            context,
350            coordinate,
351            infinitesimal,
352            solved_mapping,
353            variable,
354            parameter,
355        ) {
356            return Ok(verified);
357        }
358    }
359    Err(Decline::FlowEquationNotSolved)
360}

362fn verify_flow_coordinate(
363    context: &Context,
364    coordinate: ExpressionReference,
365    infinitesimal: &ExpressionReference,
366    solved_mapping: &SubstitutionMap,
367    variable: SymbolIdentifier,
368    parameter: SymbolIdentifier,
369) -> Result<ExpressionReference, PartialDifferentialEquationOneParameterGroupDecline> {
370    let mut mapping = solved_mapping.clone();
371    mapping.insert(
372        context.interner.intern(ExpressionKind::Symbol(variable)),
373        coordinate.clone(),
374    );
375    let velocity = substitute(context, infinitesimal, &mapping);
376    let residual = builders::subtract(
377        context,
378        differentiate(context, &coordinate, parameter),
379        velocity,
380    );
381    let initial = builders::subtract(
382        context,
383        at_origin(context, &coordinate, parameter),
384        context.interner.intern(ExpressionKind::Symbol(variable)),
385    );
386    if identity_holds(context, &residual) && identity_holds(context, &initial) {
387        Ok(coordinate)
388    } else {
389        Err(PartialDifferentialEquationOneParameterGroupDecline::FlowNotVerified)
390    }
391}

393fn definite_quadrature(
394    context: &Context,
395    integrand: &ExpressionReference,
396    parameter: SymbolIdentifier,
397    chart_obligations: &mut Vec<PartialDifferentialEquationChartObligation>,
398) -> Option<ExpressionReference> {
399    // `I(eps) - I(0)` for an exact antiderivative `I`. A logarithm of an absolute value is taken
400    // on the chart where its argument is positive, which is the component containing the origin of
401    // the parameter for every flow the identity transformation starts from; the restriction is
402    // published rather than assumed.
403    if !depends_on(integrand, parameter) {
404        let parameter_expression = context.interner.intern(ExpressionKind::Symbol(parameter));
405        return Some(builders::product(
406            context,
407            vec![integrand.clone(), parameter_expression],
408        ));
409    }
410    let antiderivative = crate::integrate::integrate(context, integrand, parameter);
411    if crate::differential_equations::linear_first_order::contains_unevaluated_integral(
412        &antiderivative,
413    ) {
414        return None;
415    }
416    let antiderivative =
417        normalize_on_the_chart(context, &antiderivative, parameter, chart_obligations)?;
418    let value = builders::subtract(
419        context,
420        antiderivative.clone(),
421        at_origin(context, &antiderivative, parameter),
422    );
423    identity_holds(
424        context,
425        &builders::subtract(
426            context,
427            differentiate(context, &value, parameter),
428            integrand.clone(),
429        ),
430    )
431    .then_some(value)
432}

434fn normalize_on_the_chart(
435    context: &Context,
436    expression: &ExpressionReference,
437    parameter: SymbolIdentifier,
438    chart_obligations: &mut Vec<PartialDifferentialEquationChartObligation>,
439) -> Option<ExpressionReference> {
440    // Two presentation choices an antiderivative makes are resolved in favour of the chart the
441    // flow starts on. A split between `E = 0` and `E != 0` for a coordinate expression `E` keeps
442    // its generic branch and publishes `E != 0`; any other piecewise result is refused. A logarithm
443    // is taken of the orientation of its argument that is positive at the origin of the parameter,
444    // which changes the antiderivative only by a constant and so leaves the definite value unchanged.
445    let mut piecewise_nodes = Vec::new();
446    collect_nodes(expression, &mut piecewise_nodes, &|candidate| {
447        matches!(candidate.kind, ExpressionKind::Piecewise(_))
448    });
449    let mut mapping = SubstitutionMap::new();
450    for node in piecewise_nodes {
451        let ExpressionKind::Piecewise(branches) = &node.kind else {
452            return None;
453        };
454        if branches.len() != 2 {
455            return None;
456        }
457        let mut generic = None;
458        let mut excluded = None;
459        for (condition, value) in branches {
460            match &condition.kind {
461                ExpressionKind::FunctionCall(
462                    FunctionIdentifier::Builtin(BuiltinFunction::NotEqual),
463                    sides,
464                ) if sides.len() == 2 => {
465                    generic = Some((
466                        builders::subtract(context, sides[0].clone(), sides[1].clone()),
467                        value.clone(),
468                    ));
469                }
470                ExpressionKind::FunctionCall(
471                    FunctionIdentifier::Builtin(BuiltinFunction::Equal),
472                    sides,
473                ) if sides.len() == 2 => {
474                    excluded = Some(builders::subtract(
475                        context,
476                        sides[0].clone(),
477                        sides[1].clone(),
478                    ));
479                }
480                _ => return None,
481            }
482        }
483        let (Some((split, value)), Some(other)) = (generic, excluded) else {
484            return None;
485        };
486        if depends_on(&split, parameter)
487            || !identity_holds(context, &builders::subtract(context, split.clone(), other))
488        {
489            return None;
490        }
491        let obligation = PartialDifferentialEquationChartObligation::NonzeroExpression(split);
492        if !chart_obligations.contains(&obligation) {
493            chart_obligations.push(obligation);
494        }
495        mapping.insert(node, value);
496    }
497    let resolved = substitute(context, expression, &mapping);
498    let mut logarithms = Vec::new();
499    collect_nodes(
500        &resolved,
501        &mut logarithms,
502        &|candidate| matches!(&candidate.kind, ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(BuiltinFunction::Log), arguments) if arguments.len() == 1),
503    );
504    let mut orientation = SubstitutionMap::new();
505    for logarithm in logarithms {
506        let ExpressionKind::FunctionCall(_, arguments) = &logarithm.kind else {
507            return None;
508        };
509        let argument = match &arguments[0].kind {
510            ExpressionKind::FunctionCall(
511                FunctionIdentifier::Builtin(BuiltinFunction::Abs),
512                inner,
513            ) if inner.len() == 1 => inner[0].clone(),
514            _ => arguments[0].clone(),
515        };
516        let start = crate::simplify::simplify(context, at_origin(context, &argument, parameter));
517        let oriented = if is_negative_number(&start) {
518            builders::negate(context, argument)
519        } else {
520            argument
521        };
522        let obligation =
523            PartialDifferentialEquationChartObligation::PositiveExpression(oriented.clone());
524        if !chart_obligations.contains(&obligation) {
525            chart_obligations.push(obligation);
526        }
527        orientation.insert(
528            logarithm.clone(),
529            builders::function_call(
530                context,
531                FunctionIdentifier::Builtin(BuiltinFunction::Log),
532                vec![oriented],
533            ),
534        );
535    }
536    Some(substitute(context, &resolved, &orientation))
537}

539fn is_negative_number(expression: &ExpressionReference) -> bool {
540    match &expression.kind {
541        ExpressionKind::Integer(value) => *value < crate::exact_arithmetic::Integer::ZERO,
542        ExpressionKind::Rational(value) => {
543            *value.numerator() < crate::exact_arithmetic::Integer::ZERO
544        }
545        _ => false,
546    }
547}

549fn collect_nodes(
550    expression: &ExpressionReference,
551    found: &mut Vec<ExpressionReference>,
552    predicate: &dyn Fn(&ExpressionReference) -> bool,
553) {
554    if predicate(expression) {
555        if !found.contains(expression) {
556            found.push(expression.clone());
557        }
558        return;
559    }
560    let _ =
561        crate::differential_equations::partial_differential_equations::problem::traverse_children(
562            expression,
563            |child| {
564                collect_nodes(child, found, predicate);
565                Ok(())
566            },
567        );
568}

570pub(crate) fn at_origin(
571    context: &Context,
572    expression: &ExpressionReference,
573    parameter: SymbolIdentifier,
574) -> ExpressionReference {
575    let mut mapping = SubstitutionMap::new();
576    mapping.insert(
577        context.interner.intern(ExpressionKind::Symbol(parameter)),
578        builders::zero(context),
579    );
580    substitute(context, expression, &mapping)
581}
