1// Exact reduction of a linear determining system for point-symmetry infinitesimals.
2//
3// The determining equations of a point symmetry are a linear homogeneous system of partial
4// differential equations in the unknown infinitesimals. They are not solved by guessing an ansatz:
5// a polynomial trial of bounded degree finds some generators and cannot say it found them all, and
6// a trial for the additive component of a linear equation would truncate a solution space to a
7// handful of polynomial solutions. Instead the system is rewritten by steps that each preserve its
8// solution set exactly on a connected product chart:
9//
10//   * an equation that depends on a coordinate none of its unknowns depend on is a Laurent
11//     polynomial identity in that coordinate, and splits into its coefficients;
12//   * `c * D^alpha f = 0` with `c` not identically zero forces `D^alpha f = 0` by continuity, and on
13//     a product chart that is solved by repeated integration, each integration contributing an
14//     arbitrary function of the coordinates not integrated;
15//   * `c * d_v^m f + g = 0` with `c` a nonzero constant and `g` a known Laurent polynomial in `v`
16//     is integrated `m` times, with the same kernel added;
17//   * `c * f + g = 0` with `c` a nonzero constant eliminates `f` outright.
18//
19// When no step applies the system is either empty, so the generators form a finite-dimensional
20// family parameterized by the surviving constants, or it consists only of the governing equation
21// itself imposed on one function of the independent coordinates: the additive symmetry of a linear
22// equation, which is retained as that constraint rather than solved. Any other residue declines.
23// Nothing here consults the equation's name or shape beyond the linear algebra of its determining
24// system, and a decline never pretends a partial reduction is the whole algebra.
25

47const MAXIMUM_REDUCTION_STEPS: usize = 512;
48const MAXIMUM_EQUATION_COUNT: usize = 4096;

50/// Why a determining system could not be reduced exactly.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub(crate) enum DeterminingSystemDecline {
53    /// An equation is not linear in the unknown infinitesimals.
54    NonlinearInTheUnknowns,
55    /// No exact reduction step applies and the residue is not an additive solution symmetry.
56    ResidueNotReduced,
57    /// The residue constrains an unknown that is not a function of the independent coordinates
58    /// alone, or the governing equation is not linear and homogeneous.
59    AdditiveComponentNotRecognized,
60    /// The reduction exceeded its step or equation budget.
61    WorkLimitExceeded,
62    /// The enclosing operation was cancelled.
63    Cancelled,
64}

66/// The general solution of a reduced determining system.
67#[derive(Debug, Clone)]
68pub(crate) struct DeterminingSystemSolution {
69    /// Stores the coordinate infinitesimals followed by the dependent infinitesimal, in jet
70    /// coordinates, expressed through the surviving parameters and the additive function.
71    pub components: Vec<ExpressionReference>,
72    /// Stores the constants the general solution is parameterized by, in allocation order.
73    pub parameters: Vec<SymbolIdentifier>,
74    /// Stores the function of the independent coordinates constrained to solve the governing
75    /// equation, when the equation admits an additive solution symmetry.
76    pub additive_function: Option<FunctionIdentifier>,
77}

98    context: &'a Context,
99    jet_space: &'a JetSpace,
100    coordinates: Vec<SymbolIdentifier>,
101    unknowns: Vec<Option<Unknown>>,

107/// Reduces a determining system to its general solution.
108pub(crate) fn solve_determining_system(
109    context: &Context,
110    jet_space: &JetSpace,
111    component_functions: &[FunctionIdentifier],
112    equations: Vec<ExpressionReference>,
113    governing_residual: &ExpressionReference,
114    principal: &PrincipalJetSolution,
115) -> Result<DeterminingSystemSolution, DeterminingSystemDecline> {
116    let mut coordinates = jet_space.independent_variables().to_vec();
117    coordinates.push(jet_space.dependent_value());
118    let components = component_functions
119        .iter()
120        .map(|function| {
121            let arguments = coordinates
122                .iter()
123                .map(|symbol| context.interner.intern(ExpressionKind::Symbol(*symbol)))
124                .collect();
125            builders::function_call(context, *function, arguments)
126        })
127        .collect();
128    let unknowns = component_functions
129        .iter()
130        .map(|function| {
131            Some(Unknown {
132                function: *function,
133                arguments: coordinates.clone(),
134            })
135        })
136        .collect();
137    let mut system = DeterminingSystem {
138        context,
139        jet_space,
140        coordinates,
141        unknowns,
142        parameters: Vec::new(),
143        equations,
144        components,
145    };
146    for _ in 0..MAXIMUM_REDUCTION_STEPS {
147        if context.operation().is_cancelled() {
148            return Err(DeterminingSystemDecline::Cancelled);
149        }
150        system.normalize()?;
151        if system.equations.len() > MAXIMUM_EQUATION_COUNT {
152            return Err(DeterminingSystemDecline::WorkLimitExceeded);
153        }
154        if system.equations.is_empty() {
155            return system.finish(None);
156        }
157        if system.split_in_a_free_coordinate()?
158            || system.integrate_a_vanishing_derivative()?
159            || system.integrate_a_known_derivative()?
160            || system.eliminate_an_undifferentiated_unknown()?
161            || system.separate_an_additive_split()?
162        {
163            continue;
164        }
165        let additive =
166            system.recognize_additive_solution_symmetry(governing_residual, principal)?;
167        return system.finish(Some(additive));
168    }
169    Err(DeterminingSystemDecline::WorkLimitExceeded)
170}

172impl DeterminingSystem<'_> {
173    fn normalize(&mut self) -> Result<(), DeterminingSystemDecline> {
174        // Expand, discard identities, and strip denominators that are free of every unknown: an
175        // equation holding wherever its coefficients are defined holds for its numerator there.
176        let mut normalized = Vec::with_capacity(self.equations.len());
177        for equation in &self.equations {
178            let expanded = crate::expand::expand(self.context, equation);
179            if is_structural_zero(&expanded) {
180                continue;
181            }
182            let numerator = self.numerator_of(&expanded);
183            if is_structural_zero(&numerator) {
184                continue;
185            }
186            if !normalized.contains(&numerator) {
187                normalized.push(numerator);
188            }
189        }
190        normalized.sort_by_key(term_count);
191        self.equations = normalized;
192        Ok(())
193    }
194
195    fn numerator_of(&self, expression: &ExpressionReference) -> ExpressionReference {
196        let has_denominator = sum_terms(expression).iter().any(|term| {
197            product_factors(term).iter().any(|factor| {
198                matches!(&factor.kind, ExpressionKind::Power(_, exponent) if is_negative_literal(exponent))
199            })
200        });
201        if !has_denominator {
202            return expression.clone();
203        }
204        let combined = crate::simplify::rational::together(self.context, expression);
205        let kept = product_factors(&combined)
206            .into_iter()
207            .filter(|factor| {
208                !matches!(&factor.kind, ExpressionKind::Power(_, exponent)
209                    if is_negative_literal(exponent) && !self.contains_unknown(factor))
210            })
211            .collect();
212        crate::expand::expand(self.context, &builders::product(self.context, kept))
213    }

215    fn finish(
216        self,
217        additive_function: Option<FunctionIdentifier>,
218    ) -> Result<DeterminingSystemSolution, DeterminingSystemDecline> {
219        if self
220            .unknowns
221            .iter()
222            .flatten()
223            .any(|unknown| Some(unknown.function) != additive_function)
224        {
225            return Err(DeterminingSystemDecline::ResidueNotReduced);
226        }
227        let components = self
228            .components
229            .iter()
230            .map(|component| crate::expand::expand(self.context, component))
231            .collect();
232        Ok(DeterminingSystemSolution {
233            components,
234            parameters: self.parameters.iter().flatten().copied().collect(),
235            additive_function,
236        })
237    }

239    fn contains_unknown(&self, expression: &ExpressionReference) -> bool {
240        let functions: BTreeSet<FunctionIdentifier> = self
241            .unknowns
242            .iter()
243            .flatten()
244            .map(|unknown| unknown.function)
245            .collect();
246        let parameters: BTreeSet<SymbolIdentifier> =
247            self.parameters.iter().flatten().copied().collect();
248        mentions_any(expression, &functions, &parameters)
249    }

251    fn linear_form(
252        &self,
253        equation: &ExpressionReference,
254    ) -> Result<LinearForm, DeterminingSystemDecline> {
255        let mut grouped: BTreeMap<Occurrence, Vec<ExpressionReference>> = BTreeMap::new();
256        let mut remainder = Vec::new();
257        for term in sum_terms(equation) {
258            let mut occurrence = None;
259            let mut coefficient = Vec::new();
260            for factor in product_factors(&term) {
261                match self.occurrence_of(&factor) {
262                    Some(found) => {
263                        if occurrence.is_some() {
264                            return Err(DeterminingSystemDecline::NonlinearInTheUnknowns);
265                        }
266                        occurrence = Some(found);
267                    }
268                    None => {
269                        if self.contains_unknown(&factor) {
270                            return Err(DeterminingSystemDecline::NonlinearInTheUnknowns);
271                        }
272                        coefficient.push(factor);
273                    }
274                }
275            }
276            let coefficient = builders::product(self.context, coefficient);
277            match occurrence {
278                Some(found) => grouped.entry(found).or_default().push(coefficient),
279                None => remainder.push(coefficient),
280            }
281        }
282        Ok(LinearForm {
283            terms: grouped
284                .into_iter()
285                .map(|(occurrence, coefficients)| {
286                    (occurrence, builders::sum(self.context, coefficients))
287                })
288                .filter(|(_, coefficient)| !is_structural_zero(coefficient))
289                .collect(),
290            remainder: builders::sum(self.context, remainder),
291        })
292    }
293
294    fn occurrence_of(&self, factor: &ExpressionReference) -> Option<Occurrence> {
295        match &factor.kind {
296            ExpressionKind::Symbol(symbol) => self
297                .parameters
298                .iter()
299                .position(|parameter| *parameter == Some(*symbol))
300                .map(Occurrence::Parameter),
301            ExpressionKind::FunctionCall(function, arguments) => {
302                let unknown = self.unknown_index(*function, arguments)?;
303                Some(Occurrence::Function {
304                    unknown,
305                    orders: vec![0; arguments.len()],
306                })
307            }
308            ExpressionKind::Derivative(function, orders, arguments) => {
309                let unknown = self.unknown_index(*function, arguments)?;
310                Some(Occurrence::Function {
311                    unknown,
312                    orders: orders.clone(),
313                })
314            }
315            _ => None,
316        }
317    }
318
319    fn unknown_index(
320        &self,
321        function: FunctionIdentifier,
322        arguments: &[ExpressionReference],
323    ) -> Option<usize> {
324        self.unknowns.iter().position(|unknown| {
325            unknown.as_ref().is_some_and(|unknown| {
326                unknown.function == function
327                    && unknown.arguments.len() == arguments.len()
328                    && unknown.arguments.iter().zip(arguments).all(|(symbol, argument)| {
329                        matches!(argument.kind, ExpressionKind::Symbol(found) if found == *symbol)
330                    })
331            })
332        })
333    }

335    fn occurrence_arguments(&self, occurrence: &Occurrence) -> Vec<SymbolIdentifier> {
336        match occurrence {
337            Occurrence::Function { unknown, .. } => self.unknowns[*unknown]
338                .as_ref()
339                .map(|unknown| unknown.arguments.clone())
340                .unwrap_or_default(),
341            Occurrence::Parameter(_) => Vec::new(),
342        }
343    }

345    fn split_in_a_free_coordinate(&mut self) -> Result<bool, DeterminingSystemDecline> {
346        for index in 0..self.equations.len() {
347            let equation = self.equations[index].clone();
348            let form = self.linear_form(&equation)?;
349            let used: BTreeSet<SymbolIdentifier> = form
350                .terms
351                .keys()
352                .flat_map(|occurrence| self.occurrence_arguments(occurrence))
353                .collect();
354            for coordinate in self.coordinates.clone() {
355                if used.contains(&coordinate) || !depends_on(&equation, coordinate) {
356                    continue;
357                }
358                let Some(coefficients) =
359                    laurent_monomial_coefficients(self.context, &equation, &[coordinate])
360                else {
361                    continue;
362                };
363                self.equations.remove(index);
364                self.equations
365                    .extend(coefficients.into_iter().map(|(_, coefficient)| coefficient));
366                return Ok(true);
367            }
368        }
369        Ok(false)
370    }

372    fn integrate_a_vanishing_derivative(&mut self) -> Result<bool, DeterminingSystemDecline> {
373        for index in 0..self.equations.len() {
374            let equation = self.equations[index].clone();
375            let form = self.linear_form(&equation)?;
376            if form.terms.len() != 1 || !is_structural_zero(&form.remainder) {
377                continue;
378            }
379            let Some((Occurrence::Function { unknown, orders }, coefficient)) =
380                form.terms.iter().next()
381            else {
382                continue;
383            };
384            if orders.iter().all(|order| *order == 0) {
385                continue;
386            }
387            if zero_equivalent(self.context, coefficient, None)
388                != ZeroEquivalenceVerdict::FalseValue
389            {
390                continue;
391            }
392            let (unknown, orders) = (*unknown, orders.clone());
393            let kernel = self.kernel_of_iterated_derivative(unknown, &orders);
394            self.equations.remove(index);
395            self.replace_unknown(unknown, kernel);
396            return Ok(true);
397        }
398        Ok(false)
399    }

401    fn integrate_a_known_derivative(&mut self) -> Result<bool, DeterminingSystemDecline> {
402        for index in 0..self.equations.len() {
403            let equation = self.equations[index].clone();
404            let form = self.linear_form(&equation)?;
405            for (occurrence, coefficient) in &form.terms {
406                let Occurrence::Function { unknown, orders } = occurrence else {
407                    continue;
408                };
409                let nonzero_slots: Vec<usize> = orders
410                    .iter()
411                    .enumerate()
412                    .filter(|(_, order)| **order > 0)
413                    .map(|(slot, _)| slot)
414                    .collect();
415                if nonzero_slots.len() != 1 || !self.is_nonzero_constant(coefficient) {
416                    continue;
417                }
418                let slot = nonzero_slots[0];
419                let multiplicity = orders[slot];
420                let Some(unknown_record) = self.unknowns[*unknown].clone() else {
421                    continue;
422                };
423                let variable = unknown_record.arguments[slot];
424                if form.terms.keys().any(|other| {
425                    other != occurrence
426                        && matches!(other, Occurrence::Function { unknown: found, .. } if found == unknown)
427                }) {
428                    continue;
429                }
430                if form.terms.keys().any(|other| {
431                    other != occurrence && self.occurrence_arguments(other).contains(&variable)
432                }) {
433                    continue;
434                }
435                let Some(known_terms) = self.known_terms_except(&form, occurrence) else {
436                    continue;
437                };
438                let integrand = builders::negate(
439                    self.context,
440                    builders::divide(
441                        self.context,
442                        builders::sum(self.context, known_terms),
443                        coefficient.clone(),
444                    ),
445                );
446                if !self.dependence_within(&integrand, &unknown_record.arguments) {
447                    continue;
448                }
449                let Some(integral) =
450                    repeated_polynomial_integral(self.context, &integrand, variable, multiplicity)
451                else {
452                    continue;
453                };
454                let (unknown, orders) = (*unknown, orders.clone());
455                let kernel = self.kernel_of_iterated_derivative(unknown, &orders);
456                let value = builders::sum(self.context, vec![integral, kernel]);
457                self.equations.remove(index);
458                self.replace_unknown(unknown, value);
459                return Ok(true);
460            }
461        }
462        Ok(false)
463    }

465    fn eliminate_an_undifferentiated_unknown(&mut self) -> Result<bool, DeterminingSystemDecline> {
466        for index in 0..self.equations.len() {
467            let equation = self.equations[index].clone();
468            let form = self.linear_form(&equation)?;
469            for (occurrence, coefficient) in &form.terms {
470                if !self.is_nonzero_constant(coefficient) {
471                    continue;
472                }
473                let target_arguments = match occurrence {
474                    Occurrence::Function { unknown, orders }
475                        if orders.iter().all(|order| *order == 0) =>
476                    {
477                        if form.terms.keys().any(|other| {
478                            other != occurrence
479                                && matches!(other, Occurrence::Function { unknown: found, .. } if found == unknown)
480                        }) {
481                            continue;
482                        }
483                        self.occurrence_arguments(occurrence)
484                    }
485                    Occurrence::Parameter(_) => Vec::new(),
486                    Occurrence::Function { .. } => continue,
487                };
488                let Some(known_terms) = self.known_terms_except(&form, occurrence) else {
489                    continue;
490                };
491                let value = builders::negate(
492                    self.context,
493                    builders::divide(
494                        self.context,
495                        builders::sum(self.context, known_terms),
496                        coefficient.clone(),
497                    ),
498                );
499                if !self.dependence_within(&value, &target_arguments) {
500                    continue;
501                }
502                let occurrence = occurrence.clone();
503                self.equations.remove(index);
504                match occurrence {
505                    Occurrence::Function { unknown, .. } => self.replace_unknown(unknown, value),
506                    Occurrence::Parameter(parameter) => self.replace_parameter(parameter, value),
507                }
508                return Ok(true);
509            }
510        }
511        Ok(false)
512    }

514    fn recognize_additive_solution_symmetry(
515        &self,
516        governing_residual: &ExpressionReference,
517        principal: &PrincipalJetSolution,
518    ) -> Result<FunctionIdentifier, DeterminingSystemDecline> {
519        // Every remaining equation must mention one and the same unknown, that unknown must be a
520        // function of exactly the independent coordinates, and read back as the dependent function
521        // each equation must be a jet-free multiple of the linear homogeneous governing residual.
522        let independent = self.jet_space.independent_variables().to_vec();
523        let mut additive = None;
524        for equation in &self.equations {
525            let form = self.linear_form(equation)?;
526            if !is_structural_zero(&form.remainder) || form.terms.is_empty() {
527                return Err(DeterminingSystemDecline::ResidueNotReduced);
528            }
529            for occurrence in form.terms.keys() {
530                let Occurrence::Function { unknown, .. } = occurrence else {
531                    return Err(DeterminingSystemDecline::ResidueNotReduced);
532                };
533                match additive {
534                    None => additive = Some(*unknown),
535                    Some(found) if found == *unknown => {}
536                    Some(_) => return Err(DeterminingSystemDecline::ResidueNotReduced),
537                }
538            }
539        }
540        let unknown_index = additive.ok_or(DeterminingSystemDecline::ResidueNotReduced)?;
541        let unknown = self.unknowns[unknown_index]
542            .clone()
543            .ok_or(DeterminingSystemDecline::ResidueNotReduced)?;
544        if unknown.arguments != independent {
545            return Err(DeterminingSystemDecline::AdditiveComponentNotRecognized);
546        }
547        if !is_linear_homogeneous_in_jets(self.context, self.jet_space, governing_residual) {
548            return Err(DeterminingSystemDecline::AdditiveComponentNotRecognized);
549        }
550        let governing_principal_coefficient = &principal.coefficient;
551        for equation in &self.equations {
552            let form = self.linear_form(equation)?;
553            let mut jet_terms = Vec::new();
554            for (occurrence, coefficient) in &form.terms {
555                let Occurrence::Function { orders, .. } = occurrence else {
556                    return Err(DeterminingSystemDecline::AdditiveComponentNotRecognized);
557                };
558                let jet = self
559                    .jet_space
560                    .coordinate(self.context, orders)
561                    .ok_or(DeterminingSystemDecline::AdditiveComponentNotRecognized)?;
562                jet_terms.push(builders::product(
563                    self.context,
564                    vec![coefficient.clone(), jet],
565                ));
566            }
567            let as_residual = builders::sum(self.context, jet_terms);
568            let equation_principal_coefficient =
569                differentiate(self.context, &as_residual, principal.principal_symbol);
570            let ratio = builders::divide(
571                self.context,
572                equation_principal_coefficient,
573                governing_principal_coefficient.clone(),
574            );
575            if self
576                .jet_space
577                .all_symbols()
578                .iter()
579                .any(|symbol| depends_on(&ratio, *symbol))
580            {
581                return Err(DeterminingSystemDecline::AdditiveComponentNotRecognized);
582            }
583            let difference = builders::subtract(
584                self.context,
585                as_residual,
586                builders::product(
587                    self.context,
588                    vec![ratio.clone(), governing_residual.clone()],
589                ),
590            );
591            let difference = crate::simplify::rational::together(
592                self.context,
593                &crate::expand::expand(self.context, &difference),
594            );
595            if zero_equivalent(self.context, &difference, None) != ZeroEquivalenceVerdict::TrueValue
596                || zero_equivalent(self.context, &ratio, None) != ZeroEquivalenceVerdict::FalseValue
597            {
598                return Err(DeterminingSystemDecline::AdditiveComponentNotRecognized);
599            }
600        }
601        Ok(unknown.function)
602    }

604    fn occurrence_expression(&self, occurrence: &Occurrence) -> Option<ExpressionReference> {
605        match occurrence {
606            Occurrence::Parameter(index) => {
607                let parameter = (*self.parameters.get(*index)?)?;
608                Some(
609                    self.context
610                        .interner
611                        .intern(ExpressionKind::Symbol(parameter)),
612                )
613            }
614            Occurrence::Function { unknown, orders } => {
615                let record = self.unknowns.get(*unknown)?.as_ref()?;
616                let arguments = record
617                    .arguments
618                    .iter()
619                    .map(|symbol| {
620                        self.context
621                            .interner
622                            .intern(ExpressionKind::Symbol(*symbol))
623                    })
624                    .collect::<Vec<_>>();
625                Some(if orders.iter().all(|order| *order == 0) {
626                    builders::function_call(self.context, record.function, arguments)
627                } else {
628                    builders::derivative(self.context, record.function, orders.clone(), arguments)
629                })
630            }
631        }
632    }

634    fn known_terms_except(
635        &self,
636        form: &LinearForm,
637        excluded: &Occurrence,
638    ) -> Option<Vec<ExpressionReference>> {
639        let mut known_terms = vec![form.remainder.clone()];
640        for (other, other_coefficient) in &form.terms {
641            if other != excluded {
642                known_terms.push(builders::product(
643                    self.context,
644                    vec![
645                        other_coefficient.clone(),
646                        self.occurrence_expression(other)?,
647                    ],
648                ));
649            }
650        }
651        Some(known_terms)
652    }

654    fn dependence_within(
655        &self,
656        expression: &ExpressionReference,
657        permitted: &[SymbolIdentifier],
658    ) -> bool {
659        free_variables(expression)
660            .into_iter()
661            .filter(|symbol| self.coordinates.contains(symbol))
662            .all(|symbol| permitted.contains(&symbol))

665    fn kernel_of_iterated_derivative(
666        &mut self,
667        unknown: usize,
668        orders: &[usize],
669    ) -> ExpressionReference {
670        // `D^alpha f = 0` on a product chart: `f = sum_i sum_(j < alpha_i) v_i^j g_ij(args \ v_i)`.
671        let Some(record) = self.unknowns[unknown].clone() else {
672            return builders::zero(self.context);
673        };
674        let mut terms = Vec::new();
675        for (slot, order) in orders.iter().enumerate() {
676            if *order == 0 {
677                continue;
678            }
679            let variable = record.arguments[slot];
680            let remaining: Vec<SymbolIdentifier> = record
681                .arguments
682                .iter()
683                .copied()
684                .filter(|argument| *argument != variable)
685                .collect();
686            for power in 0..*order {
687                let datum = self.allocate_unknown(&remaining);
688                let monomial = builders::power(
689                    self.context,
690                    self.context
691                        .interner
692                        .intern(ExpressionKind::Symbol(variable)),
693                    builders::integer_value(self.context, power as i64),
694                );
695                terms.push(builders::product(self.context, vec![monomial, datum]));
696            }
697        }
698        builders::sum(self.context, terms)
699    }
700
701    fn allocate_unknown(&mut self, arguments: &[SymbolIdentifier]) -> ExpressionReference {
702        if arguments.is_empty() {
703            let parameter =
704                allocate_partial_differential_equation_symbol(self.context, "symmetry_parameter");
705            self.parameters.push(Some(parameter));
706            return self
707                .context
708                .interner
709                .intern(ExpressionKind::Symbol(parameter));
710        }
711        let function = allocate_partial_differential_equation_arbitrary_function(
712            self.context,
713            "symmetry_function",
714            arguments.len(),
715        )
716        .function;
717        self.unknowns.push(Some(Unknown {
718            function,
719            arguments: arguments.to_vec(),
720        }));
721        let arguments = arguments
722            .iter()
723            .map(|symbol| {
724                self.context
725                    .interner
726                    .intern(ExpressionKind::Symbol(*symbol))
727            })
728            .collect();
729        builders::function_call(self.context, function, arguments)
730    }
731
732    fn replace_unknown(&mut self, unknown: usize, value: ExpressionReference) {
733        let Some(record) = self.unknowns[unknown].take() else {
734            return;
735        };
736        let replace = |context: &Context, expression: &ExpressionReference| {
737            substitute_function(context, expression, &record, &value)
738        };
739        self.equations = self
740            .equations
741            .iter()
742            .map(|equation| replace(self.context, equation))
743            .collect();
744        self.components = self
745            .components
746            .iter()
747            .map(|component| replace(self.context, component))
748            .collect();
749    }
750
751    fn replace_parameter(&mut self, parameter: usize, value: ExpressionReference) {
752        let Some(symbol) = self.parameters[parameter].take() else {
753            return;
754        };
755        let mut mapping = SubstitutionMap::new();
756        mapping.insert(
757            self.context.interner.intern(ExpressionKind::Symbol(symbol)),
758            value,
759        );
760        self.equations = self
761            .equations
762            .iter()
763            .map(|equation| substitute(self.context, equation, &mapping))
764            .collect();
765        self.components = self
766            .components
767            .iter()
768            .map(|component| substitute(self.context, component, &mapping))
769            .collect();
770    }
771}

773fn substitute_function(
774    context: &Context,
775    expression: &ExpressionReference,
776    unknown: &Unknown,
777    value: &ExpressionReference,
778) -> ExpressionReference {
779    // Every call of the unknown and every derivative of it is replaced by the value and the
780    // corresponding partial derivative of the value, so no derivative placeholder of a function that
781    // no longer exists can survive the replacement.
782    let mut occurrences = Vec::new();
783    crate::differential_equations::partial_differential_equations::point_symmetry::jet_space::collect_function_occurrences(
784        expression,
785        unknown.function,
786        &mut occurrences,
787    );
788    let mut mapping = SubstitutionMap::new();
789    for occurrence in occurrences {
790        let orders = match &occurrence.kind {
791            ExpressionKind::FunctionCall(_, arguments) => vec![0; arguments.len()],
792            ExpressionKind::Derivative(_, orders, _) => orders.clone(),
793            _ => continue,
794        };
795        let requested: Vec<(SymbolIdentifier, usize)> = unknown
796            .arguments
797            .iter()
798            .copied()
799            .zip(orders)
800            .filter(|(_, order)| *order > 0)
801            .collect();
802        mapping.insert(occurrence, partial_derivative(context, value, &requested));
803    }
804    substitute(context, expression, &mapping)
805}

807fn repeated_polynomial_integral(
808    context: &Context,
809    integrand: &ExpressionReference,
810    variable: SymbolIdentifier,
811    multiplicity: usize,
812) -> Option<ExpressionReference> {
813    // Termwise antidifferentiation of a Laurent polynomial in one coordinate. A term whose
814    // repeated antiderivative would pass through `v^-1` is logarithmic and is refused rather than
815    // integrated by a different rule than the one whose completeness is being relied on.
816    let monomials = laurent_monomial_coefficients(context, integrand, &[variable])?;
817    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
818    let mut terms = Vec::new();
819    for (exponents, coefficient) in monomials {
820        let exponent = exponents[0];
821        let mut divisor: i64 = 1;
822        for step in 1..=multiplicity as i64 {
823            let raised = exponent + step;
824            if raised == 0 {
825                return None;
826            }
827            divisor = divisor.checked_mul(raised)?;
828        }
829        terms.push(builders::product(
830            context,
831            vec![
832                coefficient,
833                builders::rational_value(context, 1, divisor),
834                builders::power(
835                    context,
836                    variable_expression.clone(),
837                    builders::integer_value(context, exponent + multiplicity as i64),
838                ),
839            ],
840        ));
841    }
842    Some(builders::sum(context, terms))
843}

845fn is_linear_homogeneous_in_jets(
846    context: &Context,
847    jet_space: &JetSpace,
848    residual: &ExpressionReference,
849) -> bool {
850    let symbols = jet_space.all_symbols();
851    let Some(monomials) = laurent_monomial_coefficients(context, residual, &symbols) else {
852        return false;
853    };
854    monomials.iter().all(|(exponents, _)| {
855        exponents.iter().sum::<i64>() == 1 && exponents.iter().all(|e| *e >= 0)
856    })
857}

859impl DeterminingSystem<'_> {
860    fn separate_an_additive_split(&mut self) -> Result<bool, DeterminingSystemDecline> {
861        // `G + R = 0` with `G` depending on a coordinate `v` that `R` does not, and `R` on a
862        // coordinate `w` that `G` does not, forces both to be functions of what they share: `G`
863        // cannot vary with `v` because `R` does not, and symmetrically. A fresh function `k` of the
864        // shared coordinates therefore replaces the equation by `G = k` and `R = -k`, an exact
865        // equivalence that lets each side be integrated on its own.
866        for index in 0..self.equations.len() {
867            let equation = self.equations[index].clone();
868            let form = self.linear_form(&equation)?;
869            let Some(mut pieces) = form
870                .terms
871                .iter()
872                .map(|(occurrence, coefficient)| {
873                    let mut dependence = self.dependence_of(coefficient);
874                    dependence.extend(self.occurrence_arguments(occurrence));
875                    let expression = builders::product(
876                        self.context,
877                        vec![coefficient.clone(), self.occurrence_expression(occurrence)?],
878                    );
879                    Some((expression, dependence))
880                })
881                .collect::<Option<Vec<(ExpressionReference, BTreeSet<SymbolIdentifier>)>>>()
882            else {
883                continue;
884            };
885            if !is_structural_zero(&form.remainder) {
886                for term in sum_terms(&form.remainder) {
887                    let dependence = self.dependence_of(&term);
888                    pieces.push((term, dependence));
889                }
890            }
891            for varying in self.coordinates.clone() {
892                let (with_varying, without_varying): (Vec<_>, Vec<_>) = pieces
893                    .iter()
894                    .cloned()
895                    .partition(|(_, dependence)| dependence.contains(&varying));
896                if with_varying.is_empty() || without_varying.is_empty() {
897                    continue;
898                }
899                let first_dependence: BTreeSet<SymbolIdentifier> = with_varying
900                    .iter()
901                    .flat_map(|(_, dependence)| dependence.iter().copied())
902                    .collect();
903                let second_dependence: BTreeSet<SymbolIdentifier> = without_varying
904                    .iter()
905                    .flat_map(|(_, dependence)| dependence.iter().copied())
906                    .collect();
907                if second_dependence.is_subset(&first_dependence) {
908                    continue;
909                }
910                let shared: Vec<SymbolIdentifier> = self
911                    .coordinates
912                    .iter()
913                    .copied()
914                    .filter(|coordinate| {
915                        first_dependence.contains(coordinate)
916                            && second_dependence.contains(coordinate)
917                    })
918                    .collect();
919                let separation = self.allocate_unknown(&shared);
920                let first = builders::sum(
921                    self.context,
922                    with_varying.into_iter().map(|(piece, _)| piece).collect(),
923                );
924                let second = builders::sum(
925                    self.context,
926                    without_varying
927                        .into_iter()
928                        .map(|(piece, _)| piece)
929                        .collect(),
930                );
931                self.equations.remove(index);
932                self.equations
933                    .push(builders::subtract(self.context, first, separation.clone()));
934                self.equations
935                    .push(builders::sum(self.context, vec![second, separation]));
936                return Ok(true);
937            }
938        }
939        Ok(false)
940    }
941
942    fn dependence_of(&self, expression: &ExpressionReference) -> BTreeSet<SymbolIdentifier> {
943        free_variables(expression)
944            .into_iter()
945            .filter(|symbol| self.coordinates.contains(symbol))
946            .collect()
947    }
948
949    fn is_nonzero_constant(&self, expression: &ExpressionReference) -> bool {
950        // Constant means free of every coordinate of the chart; a symbolic coefficient such as a
951        // diffusivity qualifies when the context's assumptions prove it nonzero, and does not when
952        // it might vanish, because dividing by it would then change the equation being reduced.
953        if is_nonzero_number(expression) {
954            return true;
955        }
956        if !self.dependence_of(expression).is_empty() {
957            return false;
958        }
959        if free_variables(expression).is_empty() {
960            return zero_equivalent(self.context, expression, None)
961                == ZeroEquivalenceVerdict::FalseValue;
962        }
963        self.context.symbol_table.ask_assumptions(
964            &crate::assumptions::AssumptionFormula::not_equal(
965                expression.clone(),
966                builders::zero(self.context),
967            ),
968        ) == crate::assumptions::TruthValue::Yes
969    }
970}

973    match &expression.kind {
974        ExpressionKind::Integer(value) => *value < Integer::ZERO,
975        ExpressionKind::Rational(value) => *value.numerator() < Integer::ZERO,

980fn mentions_any(
981    expression: &ExpressionReference,
982    functions: &BTreeSet<FunctionIdentifier>,
983    parameters: &BTreeSet<SymbolIdentifier>,
984) -> bool {
985    match &expression.kind {
986        ExpressionKind::Symbol(symbol) if parameters.contains(symbol) => return true,
987        ExpressionKind::FunctionCall(function, _) | ExpressionKind::Derivative(function, _, _)
988            if functions.contains(function) =>
989        {
990            return true;
991        }
992        _ => {}
993    }
994    let mut found = false;
995    let _ =
996        crate::differential_equations::partial_differential_equations::problem::traverse_children(
997            expression,
998            |child| {
999                found |= mentions_any(child, functions, parameters);
1000                Ok(())
1001            },
1002        );
1003    found
1004}
