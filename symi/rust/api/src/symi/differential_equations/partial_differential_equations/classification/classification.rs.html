123/// Classifies a validated PDE using an explicit deterministic work budget.
124pub fn classify_partial_differential_equation_with_budget(
125    context: &Context,
126    problem: &PartialDifferentialEquationProblem,
127    budget: usize,
128) -> Result<
129    PartialDifferentialEquationClassification,
130    PartialDifferentialEquationClassificationDecline,
131> {
132    let inventory = match collect_partial_differential_equation_jets(problem, budget) {
133        Ok(inventory) => inventory,
134        Err(crate::differential_equations::partial_differential_equations::problem::PartialDifferentialEquationProblemError::DiscoveryFailed) => return Err(PartialDifferentialEquationClassificationDecline::BudgetExceeded),
135        Err(error) => return Err(PartialDifferentialEquationClassificationDecline::Problem(error)),
136    };
137    if inventory.order == 0 {
138        return Err(PartialDifferentialEquationClassificationDecline::UnsupportedExpressionShape);
139    }
140    let all_jets: BTreeSet<_> = inventory.jets.iter().cloned().collect();
141    let highest_jets: BTreeSet<_> = inventory.highest_order_jets.iter().cloned().collect();
142    let mut remaining_budget = budget;
143    let full_affine = decompose_affine(
144        context,
145        problem.equation(),
146        problem,
147        &all_jets,
148        &mut remaining_budget,
149    )?;
150    let highest_affine = if full_affine.is_some() {
151        None
152    } else {
153        decompose_affine(
154            context,
155            problem.equation(),
156            problem,
157            &highest_jets,
158            &mut remaining_budget,
159        )?
160    };
161
162    let (linearity, coefficient_data, principal_coefficients) = if let Some(full_form) = full_affine
163    {
164        let coefficients = coefficient_records(&full_form.coefficients);
165        let homogeneous = is_zero(context, &full_form.constant);
166        (
167            PartialDifferentialEquationLinearity::Linear,
168            PartialDifferentialEquationCoefficientData {
169                full_coefficients: Some(coefficients.clone()),
170                highest_order_coefficients: Some(filter_highest_coefficients(
171                    &coefficients,
172                    inventory.order,
173                )),
174                remainder: Some(full_form.constant),
175                homogeneous: Some(homogeneous),
176            },
177            Some(coefficients),
178        )
179    } else if let Some(highest_form) = highest_affine {
180        let highest_coefficients = coefficient_records(&highest_form.coefficients);
181        let coefficient_depends_on_any_jet = highest_coefficients
182            .iter()
183            .any(|record| expression_contains_any_jet(&record.coefficient, problem, &all_jets));
184        let linearity = if coefficient_depends_on_any_jet {
185            PartialDifferentialEquationLinearity::Quasilinear
186        } else {
187            PartialDifferentialEquationLinearity::Semilinear
188        };
189        (
190            linearity,
191            PartialDifferentialEquationCoefficientData {
192                full_coefficients: None,
193                highest_order_coefficients: Some(highest_coefficients.clone()),
194                remainder: Some(highest_form.constant),
195                homogeneous: None,
196            },
197            Some(highest_coefficients),
198        )
199    } else {
200        (
201            PartialDifferentialEquationLinearity::Nonlinear,
202            PartialDifferentialEquationCoefficientData {
203                full_coefficients: None,
204                highest_order_coefficients: None,
205                remainder: None,
206                homogeneous: None,
207            },
208            None,
209        )
210    };
211    let second_order_principal_part = if inventory.order == 2 {
212        principal_coefficients
213            .as_ref()
214            .map(|coefficients| principal_part(context, problem, coefficients))
215    } else {
216        None
217    };
218    Ok(PartialDifferentialEquationClassification {
219        order: inventory.order,
220        dependent_function: problem.dependent_function(),
221        independent_variables: problem.independent_variables().to_vec(),
222        linearity,
223        coefficient_data,
224        second_order_principal_part,
225        method_candidates: method_candidates(problem, inventory.order, linearity),
226    })
227}

229fn decompose_affine(
230    context: &Context,
231    expression: &ExpressionReference,
232    problem: &PartialDifferentialEquationProblem,
233    target_jets: &BTreeSet<PartialDifferentialEquationJet>,
234    remaining_budget: &mut usize,
235) -> Result<Option<AffineJetForm>, PartialDifferentialEquationClassificationDecline> {
236    if *remaining_budget == 0 {
237        return Err(PartialDifferentialEquationClassificationDecline::BudgetExceeded);
238    }
239    *remaining_budget -= 1;
240    let jet = jet_for_expression(expression, problem.signature())
241        .map_err(PartialDifferentialEquationClassificationDecline::Problem)?;
242    if let Some(jet) = jet {
243        if target_jets.contains(&jet) {
244            let mut coefficients = BTreeMap::new();
245            coefficients.insert(jet, builders::one(context));
246            return Ok(Some(AffineJetForm {
247                constant: builders::zero(context),
248                coefficients,
249            }));
250        }
251    }
252    match &expression.kind {
253        ExpressionKind::Sum(children) => {
254            let mut form = AffineJetForm {
255                constant: builders::zero(context),
256                coefficients: BTreeMap::new(),
257            };
258            for child in children {
259                let Some(child_form) =
260                    decompose_affine(context, child, problem, target_jets, remaining_budget)?
261                else {
262                    return Ok(None);
263                };
264                form.constant = builders::sum(context, vec![form.constant, child_form.constant]);
265                for (jet, coefficient) in child_form.coefficients {
266                    let current = form
267                        .coefficients
268                        .remove(&jet)
269                        .unwrap_or_else(|| builders::zero(context));
270                    form.coefficients
271                        .insert(jet, builders::sum(context, vec![current, coefficient]));
272                }
273            }
274            Ok(Some(form))
275        }
276        ExpressionKind::Product(children) => {
277            let mut constant_factors = Vec::new();
278            let mut jet_form = None;
279            for child in children {
280                let Some(child_form) =
281                    decompose_affine(context, child, problem, target_jets, remaining_budget)?
282                else {
283                    return Ok(None);
284                };
285                if child_form.coefficients.is_empty() {
286                    constant_factors.push(child_form.constant);
287                } else if jet_form.is_some() {
288                    return Ok(None);
289                } else {
290                    jet_form = Some(child_form);
291                }
292            }
293            let scalar = builders::product(context, constant_factors);
294            if let Some(form) = jet_form {
295                let constant = builders::product(context, vec![scalar.clone(), form.constant]);
296                let coefficients = form
297                    .coefficients
298                    .into_iter()
299                    .map(|(jet, coefficient)| {
300                        (
301                            jet,
302                            builders::product(context, vec![scalar.clone(), coefficient]),
303                        )
304                    })
305                    .collect();
306                Ok(Some(AffineJetForm {
307                    constant,
308                    coefficients,
309                }))
310            } else {
311                Ok(Some(AffineJetForm {
312                    constant: scalar,
313                    coefficients: BTreeMap::new(),
314                }))
315            }
316        }
317        ExpressionKind::Power(_, _)
318        | ExpressionKind::FunctionCall(_, _)
319        | ExpressionKind::Factorial(_) => {
320            if expression_contains_any_jet(expression, problem, target_jets) {
321                Ok(None)
322            } else {
323                Ok(Some(AffineJetForm {
324                    constant: expression.clone(),
325                    coefficients: BTreeMap::new(),
326                }))
327            }
328        }
329        _ => {
330            if expression_contains_any_jet(expression, problem, target_jets) {
331                Err(PartialDifferentialEquationClassificationDecline::UnsupportedExpressionShape)
332            } else {
333                Ok(Some(AffineJetForm {
334                    constant: expression.clone(),
335                    coefficients: BTreeMap::new(),
336                }))
337            }
338        }
339    }
340}

342fn expression_contains_any_jet(
343    expression: &ExpressionReference,
344    problem: &PartialDifferentialEquationProblem,
345    target_jets: &BTreeSet<PartialDifferentialEquationJet>,
346) -> bool {
347    let Ok(jet) = jet_for_expression(expression, problem.signature()) else {
348        return true;
349    };
350    if jet.is_some_and(|jet| target_jets.contains(&jet)) {
351        return true;
352    }
353    match &expression.kind {
354        ExpressionKind::Sum(children)
355        | ExpressionKind::Product(children)
356        | ExpressionKind::FiniteSet(children)
357        | ExpressionKind::SetUnion(children)
358        | ExpressionKind::SetIntersection(children)
359        | ExpressionKind::SpecialFunctionParameterVector(children)
360        | ExpressionKind::FunctionCall(_, children)
361        | ExpressionKind::Derivative(_, _, children)
362        | ExpressionKind::Distribution(_, children) => children
363            .iter()
364            .any(|child| expression_contains_any_jet(child, problem, target_jets)),
365        ExpressionKind::Power(left, right)
366        | ExpressionKind::SetComplement(left, right)
367        | ExpressionKind::Interval(left, right, _, _)
368        | ExpressionKind::ImageSet(_, left, right)
369        | ExpressionKind::ConditionSet(_, left, right) => {
370            expression_contains_any_jet(left, problem, target_jets)
371                || expression_contains_any_jet(right, problem, target_jets)
372        }
373        ExpressionKind::Factorial(argument)
374        | ExpressionKind::RandomVariable(_, argument)
375        | ExpressionKind::LaplaceTransform(argument, _, _)
376        | ExpressionKind::InverseLaplaceTransform(argument, _, _)
377        | ExpressionKind::FourierTransform(argument, _, _)
378        | ExpressionKind::InverseFourierTransform(argument, _, _) => {
379            expression_contains_any_jet(argument, problem, target_jets)
380        }
381        ExpressionKind::Integral(argument, _, lower, upper)
382        | ExpressionKind::Summation(argument, _, lower, upper)
383        | ExpressionKind::IndexedProduct(argument, _, lower, upper) => {
384            expression_contains_any_jet(argument, problem, target_jets)
385                || lower
386                    .as_ref()
387                    .is_some_and(|bound| expression_contains_any_jet(bound, problem, target_jets))
388                || upper
389                    .as_ref()
390                    .is_some_and(|bound| expression_contains_any_jet(bound, problem, target_jets))
391        }
392        ExpressionKind::Piecewise(branches) => branches.iter().any(|(condition, value)| {
393            expression_contains_any_jet(condition, problem, target_jets)
394                || expression_contains_any_jet(value, problem, target_jets)
395        }),
396        ExpressionKind::OrdinaryDifferentialEquation(equation, dependent_call, _)
397        | ExpressionKind::RecurrenceEquation(equation, dependent_call, _) => {
398            expression_contains_any_jet(equation, problem, target_jets)
399                || expression_contains_any_jet(dependent_call, problem, target_jets)
400        }
401        ExpressionKind::Integer(_)
402        | ExpressionKind::Rational(_)
403        | ExpressionKind::PolynomialRoot(_)
404        | ExpressionKind::Symbol(_)
405        | ExpressionKind::Constant(_) => false,
406    }
407}

409fn coefficient_records(
410    coefficients: &BTreeMap<PartialDifferentialEquationJet, ExpressionReference>,
411) -> Vec<PartialDifferentialEquationJetCoefficient> {
412    coefficients
413        .iter()
414        .map(
415            |(jet, coefficient)| PartialDifferentialEquationJetCoefficient {
416                jet: jet.clone(),
417                coefficient: coefficient.clone(),
418            },
419        )
420        .collect()
421}

423fn filter_highest_coefficients(
424    coefficients: &[PartialDifferentialEquationJetCoefficient],
425    order: usize,
426) -> Vec<PartialDifferentialEquationJetCoefficient> {
427    coefficients
428        .iter()
429        .filter(|record| record.jet.total_order() == order)
430        .cloned()
431        .collect()
432}

438fn principal_part(
439    context: &Context,
440    problem: &PartialDifferentialEquationProblem,
441    coefficients: &[PartialDifferentialEquationJetCoefficient],
442) -> PartialDifferentialEquationPrincipalPart {
443    let variable_count = problem.independent_variables().len();
444    let coefficient_map: BTreeMap<_, _> = coefficients
445        .iter()
446        .filter(|record| record.jet.total_order() == 2)
447        .map(|record| (record.jet.clone(), record.coefficient.clone()))
448        .collect();
449    let zero = builders::zero(context);
450    let two = builders::integer(context, 2.into());
451    let mut rows = Vec::with_capacity(variable_count);
452    for row in 0..variable_count {
453        let mut entries = Vec::with_capacity(variable_count);
454        for column in 0..variable_count {
455            let mut orders = vec![0; variable_count];
456            orders[row] += 1;
457            orders[column] += 1;
458            let coefficient = coefficient_map
459                .get(&PartialDifferentialEquationJet {
460                    derivative_orders: orders,
461                })
462                .cloned()
463                .unwrap_or_else(|| zero.clone());
464            entries.push(if row == column {
465                coefficient
466            } else {
467                builders::divide(context, coefficient, two.clone())
468            });
469        }
470        rows.push(entries);
471    }
472    PartialDifferentialEquationPrincipalPart {
473        matrix: matrix_from_rows(rows).expect("positive PDE independent-variable count"),
474        derivative_coefficients: coefficient_map
475            .into_iter()
476            .map(
477                |(jet, coefficient)| PartialDifferentialEquationPrincipalCoefficient {
478                    jet,
479                    coefficient,
480                },
481            )
482            .collect(),
483    }
484}

486fn method_candidates(
487    problem: &PartialDifferentialEquationProblem,
488    order: usize,
489    linearity: PartialDifferentialEquationLinearity,
490) -> Vec<PartialDifferentialEquationMethodCandidate> {
491    let mut candidates = Vec::new();
492    if order == 1 && linearity != PartialDifferentialEquationLinearity::Nonlinear {
493        candidates.push(PartialDifferentialEquationMethodCandidate {
494            method: PartialDifferentialEquationMethod::Characteristics,
495            proof_obligations: vec![
496                PartialDifferentialEquationProofObligation::CoefficientNonzero,
497                PartialDifferentialEquationProofObligation::CharacteristicTransversality,
498            ],
499        });
500    }
501    if order == 2
502        && linearity == PartialDifferentialEquationLinearity::Linear
503        && problem.independent_variables().len() == 2
504    {
505        candidates.push(PartialDifferentialEquationMethodCandidate {
506            method: PartialDifferentialEquationMethod::CanonicalCoordinates,
507            proof_obligations: vec![
508                PartialDifferentialEquationProofObligation::EquationTypeEvidence,
509                PartialDifferentialEquationProofObligation::CoordinateJacobianNonzero,
510            ],
511        });
512    }
513    if linearity == PartialDifferentialEquationLinearity::Linear {
514        candidates.push(PartialDifferentialEquationMethodCandidate {
515            method: PartialDifferentialEquationMethod::RepeatedIntegration,
516            proof_obligations: vec![PartialDifferentialEquationProofObligation::CoefficientNonzero],
517        });
518        if order == 2 && problem.independent_variables().len() == 2 {
519            candidates.push(PartialDifferentialEquationMethodCandidate {
520                method: PartialDifferentialEquationMethod::InitialDataConvolution,
521                proof_obligations: vec![
522                    PartialDifferentialEquationProofObligation::ConstantCoefficients,
523                    PartialDifferentialEquationProofObligation::DifferentiationUnderTheIntegralSign,
524                    PartialDifferentialEquationProofObligation::InitialTraceAttainedInTheAdmissibleClass,
525                    PartialDifferentialEquationProofObligation::UniquenessInTheAdmissibleClass,
526                ],
527            });
528        }
529        candidates.push(PartialDifferentialEquationMethodCandidate {
530            method: PartialDifferentialEquationMethod::Separation,
531            proof_obligations: vec![
532                PartialDifferentialEquationProofObligation::SeparabilityEstablished,
533                PartialDifferentialEquationProofObligation::BoundaryEigenvalueConditionsEstablished,
534            ],
535        });
536        let transform_obligations = vec![
537            PartialDifferentialEquationProofObligation::ConstantCoefficients,
538            PartialDifferentialEquationProofObligation::TransformExistenceAndGrowth,
539            PartialDifferentialEquationProofObligation::ExactInverseTransformAvailable,
540        ];
541        candidates.push(PartialDifferentialEquationMethodCandidate {
542            method: PartialDifferentialEquationMethod::LaplaceTransform,
543            proof_obligations: transform_obligations.clone(),
544        });
545        candidates.push(PartialDifferentialEquationMethodCandidate {
546            method: PartialDifferentialEquationMethod::FourierTransform,
547            proof_obligations: transform_obligations,
548        });
549    }
550    candidates
