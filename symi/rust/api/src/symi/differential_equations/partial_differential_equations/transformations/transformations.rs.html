23/// Public native enumeration named `partial_differential_equation_transformation_decline`.
24#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
25pub enum PartialDifferentialEquationTransformationDecline {
26    /// Represents the `two_source_and_target_variables_required` case.
27    TwoSourceAndTargetVariablesRequired,
28    /// Represents the `duplicate_or_colliding_coordinate_variables` case.
29    DuplicateOrCollidingCoordinateVariables,
30    /// Represents the `coordinate_map_direction_missing` case.
31    CoordinateMapDirectionMissing,
32    /// Represents the `coordinate_map_arity_mismatch` case.
33    CoordinateMapArityMismatch,
34    /// Represents the `coordinate_map_contains_dependent_function` case.
35    CoordinateMapContainsDependentFunction,
36    /// Represents the `coordinate_map_uses_wrong_variables` case.
37    CoordinateMapUsesWrongVariables,
38    /// Represents the `target_dependent_function_invalid` case.
39    TargetDependentFunctionInvalid,
40    /// Represents the `inverse_not_recovered` case.
41    InverseNotRecovered,
42    /// Represents the `inverse_map_rejected` case.
43    InverseMapRejected,
44    /// Represents the `inverse_map_inconclusive` case.
45    InverseMapInconclusive,
46    /// Represents the `jacobian_is_zero` case.
47    JacobianIsZero,
48    /// Represents the `jacobian_nonzero_not_proved` case.
49    JacobianNonzeroNotProved,
50    /// Represents the `condition_derivative_order_not_supported` case.
51    ConditionDerivativeOrderNotSupported,
52    /// Represents the `target_problem_invalid` case.
53    TargetProblemInvalid,

114/// Rebuilds a validated coordinate change from its stored canonical parts.
115pub fn rebuild_partial_differential_equation_coordinate_change(
116    source_variables: Vec<SymbolIdentifier>,
117    target_variables: Vec<SymbolIdentifier>,
118    forward_coordinates: Vec<ExpressionReference>,
119    inverse_coordinates: Vec<ExpressionReference>,
120    jacobian: ExpressionReference,
121    target_dependent_function: FunctionIdentifier,
122) -> Result<
123    PartialDifferentialEquationCoordinateChange,
124    PartialDifferentialEquationTransformationDecline,
125> {
126    // Revalidate the structural invariants a stored record must still satisfy before it
127    // becomes a coordinate change again, so a record that no longer describes a valid map
128    // fails loudly instead of rebuilding into an unvalidated object. Exact mathematical
129    // identities stay the caller's to re-establish through the independent verifier.
130    if source_variables.len() != 2 || target_variables.len() != 2 {
131        return Err(
132            PartialDifferentialEquationTransformationDecline::TwoSourceAndTargetVariablesRequired,
133        );
134    }
135    if forward_coordinates.len() != 2 || inverse_coordinates.len() != 2 {
136        return Err(PartialDifferentialEquationTransformationDecline::CoordinateMapArityMismatch);
137    }
138    if source_variables[0] == source_variables[1]
139        || target_variables[0] == target_variables[1]
140        || target_variables
141            .iter()
142            .any(|target| source_variables.contains(target))
143    {
144        return Err(PartialDifferentialEquationTransformationDecline::DuplicateOrCollidingCoordinateVariables);
145    }
146    for coordinate in &forward_coordinates {
147        if free_variables(coordinate)
148            .iter()
149            .any(|variable| target_variables.contains(variable))
150        {
151            return Err(
152                PartialDifferentialEquationTransformationDecline::CoordinateMapUsesWrongVariables,
153            );
154        }
155    }
156    for coordinate in &inverse_coordinates {
157        if free_variables(coordinate)
158            .iter()
159            .any(|variable| source_variables.contains(variable))
160        {
161            return Err(
162                PartialDifferentialEquationTransformationDecline::CoordinateMapUsesWrongVariables,
163            );
164        }
165    }
166    if !matches!(target_dependent_function, FunctionIdentifier::User(_)) {
167        return Err(
168            PartialDifferentialEquationTransformationDecline::TargetDependentFunctionInvalid,
169        );
170    }
171    Ok(PartialDifferentialEquationCoordinateChange {
172        source_variables,
173        target_variables,
174        forward_coordinates,
175        inverse_coordinates,
176        jacobian,
177        target_dependent_function,
178    })
179}

181/// Transforms a two-variable PDE through a validated exact coordinate change.
182pub fn transform_partial_differential_equation(
183    context: &Context,
184    source_problem: &PartialDifferentialEquationProblem,
185    request: PartialDifferentialEquationCoordinateChangeRequest,
186) -> Result<
187    PartialDifferentialEquationTransformationResult,
188    PartialDifferentialEquationTransformationDecline,
189> {
190    if source_problem.independent_variables().len() != 2 || request.target_variables.len() != 2 {
191        return Err(
192            PartialDifferentialEquationTransformationDecline::TwoSourceAndTargetVariablesRequired,
193        );
194    }
195    if request.target_variables[0] == request.target_variables[1]
196        || request
197            .target_variables
198            .iter()
199            .any(|target| source_problem.independent_variables().contains(target))
200    {
201        return Err(PartialDifferentialEquationTransformationDecline::DuplicateOrCollidingCoordinateVariables);
202    }
203    let target_dependent_function =
204        target_dependent_function(context, source_problem, request.target_dependent_function)?;
205    let (forward_coordinates, inverse_coordinates) =
206        resolve_coordinate_directions(context, source_problem, &request)?;
207    validate_coordinate_hygiene(
208        source_problem,
209        &request.target_variables,
210        &forward_coordinates,
211        &inverse_coordinates,
212    )?;
213    let jacobian = forward_jacobian(
214        context,
215        source_problem.independent_variables(),
216        &forward_coordinates,
217    );
218    validate_compositions(
219        context,
220        source_problem.independent_variables(),
221        &request.target_variables,
222        &forward_coordinates,
223        &inverse_coordinates,
224    )?;
225    validate_jacobian(context, &jacobian)?;
226    let coordinate_change = PartialDifferentialEquationCoordinateChange {
227        source_variables: source_problem.independent_variables().to_vec(),
228        target_variables: request.target_variables,
229        forward_coordinates,
230        inverse_coordinates,
231        jacobian,
232        target_dependent_function,
233    };
234    let target_problem = transform_problem(context, source_problem, &coordinate_change)?;
235    let normalization_multiplier = builders::one(context);
236    let result = PartialDifferentialEquationTransformationResult {
237        coordinate_change,
238        target_problem,
239        normalization_multiplier,
240    };
241    let verification = verify_partial_differential_equation_coordinate_transformation(
242        context,
243        source_problem,
244        &result,
245    );
246    match verification.verdict {
247        PartialDifferentialEquationVerificationVerdict::Verified => Ok(result),
248        PartialDifferentialEquationVerificationVerdict::Rejected => {
249            Err(PartialDifferentialEquationTransformationDecline::InverseMapRejected)
250        }
251        PartialDifferentialEquationVerificationVerdict::Inconclusive => {
252            Err(PartialDifferentialEquationTransformationDecline::InverseMapInconclusive)
253        }
254    }
255}

257/// Independently verifies an exact PDE coordinate transformation.
258pub fn verify_partial_differential_equation_coordinate_transformation(
259    context: &Context,
260    source_problem: &PartialDifferentialEquationProblem,
261    transformation: &PartialDifferentialEquationTransformationResult,
262) -> PartialDifferentialEquationTransformationVerificationReport {
263    let change = &transformation.coordinate_change;
264    let composition_residuals = composition_residuals(
265        context,
266        &change.source_variables,
267        &change.target_variables,
268        &change.forward_coordinates,
269        &change.inverse_coordinates,
270    );
271    let jacobian = forward_jacobian(
272        context,
273        &change.source_variables,
274        &change.forward_coordinates,
275    );
276    let transformed_governing_residual = transform_residual(
277        context,
278        source_problem,
279        source_problem.equation(),
280        change,
281        true,
282    );
283    let claimed_governing_residual = transformation.target_problem.equation().clone();
284    let (transformed_governing_residual, governing_verdict) = match transformed_governing_residual {
285        Ok(transformed_governing_residual) => {
286            let normalized_expected = builders::product(
287                context,
288                vec![
289                    transformation.normalization_multiplier.clone(),
290                    transformed_governing_residual.clone(),
291                ],
292            );
293            let governing_residual = builders::subtract(
294                context,
295                claimed_governing_residual.clone(),
296                normalized_expected,
297            );
298            (
299                transformed_governing_residual,
300                exact_zero_verdict(context, &governing_residual),
301            )
302        }
303        Err(_) => (builders::zero(context), ZeroEquivalenceVerdict::Unknown),
304    };
305    let principal_congruence_residuals =
306        principal_congruence_residuals(context, source_problem, transformation, change);
307    let mut verdicts = composition_residuals
308        .iter()
309        .map(|residual| exact_zero_verdict(context, residual))
310        .collect::<Vec<_>>();
311    verdicts.push(governing_verdict);
312    verdicts.extend(
313        principal_congruence_residuals
314            .iter()
315            .map(|residual| exact_zero_verdict(context, residual)),
316    );
317    let jacobian_verdict = exact_nonzero_verdict(context, &jacobian);
318    let normalization_verdict =
319        exact_nonzero_verdict(context, &transformation.normalization_multiplier);
320    let verdict = aggregate_verdict(&verdicts, jacobian_verdict, normalization_verdict);
321    PartialDifferentialEquationTransformationVerificationReport {
322        verdict,
323        composition_residuals,
324        jacobian,
325        transformed_governing_residual,
326        claimed_governing_residual,
327        principal_congruence_residuals,
328    }
329}

331fn target_dependent_function(
332    context: &Context,
333    source_problem: &PartialDifferentialEquationProblem,
334    requested_function: Option<FunctionIdentifier>,
335) -> Result<FunctionIdentifier, PartialDifferentialEquationTransformationDecline> {
336    match requested_function {
337        Some(FunctionIdentifier::User(identifier))
338            if FunctionIdentifier::User(identifier) != source_problem.dependent_function() =>
339        {
340            Ok(FunctionIdentifier::User(identifier))
341        }
342        Some(_) => {
343            Err(PartialDifferentialEquationTransformationDecline::TargetDependentFunctionInvalid)
344        }
345        None => {
346            Ok(allocate_partial_differential_equation_arbitrary_function(context, "v", 2).function)
347        }
348    }
349}

351fn resolve_coordinate_directions(
352    context: &Context,
353    source_problem: &PartialDifferentialEquationProblem,
354    request: &PartialDifferentialEquationCoordinateChangeRequest,
355) -> Result<
356    (Vec<ExpressionReference>, Vec<ExpressionReference>),
357    PartialDifferentialEquationTransformationDecline,
358> {
359    match (&request.forward_coordinates, &request.inverse_coordinates) {
360        (Some(forward_coordinates), Some(inverse_coordinates)) => {
361            Ok((forward_coordinates.clone(), inverse_coordinates.clone()))
362        }
363        (Some(forward_coordinates), None) => recover_inverse_coordinates(
364            context,
365            forward_coordinates,
366            source_problem.independent_variables(),
367            &request.target_variables,
368        )
369        .map(|inverse_coordinates| (forward_coordinates.clone(), inverse_coordinates)),
370        (None, Some(inverse_coordinates)) => recover_inverse_coordinates(
371            context,
372            inverse_coordinates,
373            &request.target_variables,
374            source_problem.independent_variables(),
375        )
376        .map(|forward_coordinates| (forward_coordinates, inverse_coordinates.clone())),
377        (None, None) => {
378            Err(PartialDifferentialEquationTransformationDecline::CoordinateMapDirectionMissing)
379        }
380    }
381}

383fn recover_inverse_coordinates(
384    context: &Context,
385    coordinate_expressions: &[ExpressionReference],
386    input_variables: &[SymbolIdentifier],
387    output_variables: &[SymbolIdentifier],
388) -> Result<Vec<ExpressionReference>, PartialDifferentialEquationTransformationDecline> {
389    if coordinate_expressions.len() != 2
390        || input_variables.len() != 2
391        || output_variables.len() != 2
392    {
393        return Err(PartialDifferentialEquationTransformationDecline::CoordinateMapArityMismatch);
394    }
395    let output_expressions = output_variables
396        .iter()
397        .map(|variable| context.interner.intern(ExpressionKind::Symbol(*variable)))
398        .collect::<Vec<_>>();
399    let equations = coordinate_expressions
400        .iter()
401        .zip(output_expressions)
402        .map(|(coordinate, output)| builders::subtract(context, coordinate.clone(), output))
403        .collect::<Vec<_>>();
404    let SystemSolution::Finite { solutions, .. } =
405        solve_polynomial_system(context, &equations, input_variables)
406    else {
407        return Err(PartialDifferentialEquationTransformationDecline::InverseNotRecovered);
408    };
409    let candidates = solutions
410        .iter()
411        .map(|solution| solution_components(context, solution, input_variables))
412        .collect::<Option<Vec<_>>>()
413        .ok_or(PartialDifferentialEquationTransformationDecline::InverseNotRecovered)?;
414    let Some(first_candidate) = candidates.first().cloned() else {
415        return Err(PartialDifferentialEquationTransformationDecline::InverseNotRecovered);
416    };
417    for candidate in candidates.iter().skip(1) {
418        if !coordinate_lists_equivalent(context, &first_candidate, candidate) {
419            return Err(PartialDifferentialEquationTransformationDecline::InverseNotRecovered);
420        }
421    }
422    Ok(first_candidate)
423}

425fn solution_components(
426    context: &Context,
427    solution: &[(SymbolIdentifier, ExpressionReference)],
428    variables: &[SymbolIdentifier],
429) -> Option<Vec<ExpressionReference>> {
430    let assignments = solution.iter().cloned().collect::<HashMap<_, _>>();
431    variables
432        .iter()
433        .map(|variable| assignments.get(variable).cloned())
434        .collect::<Option<Vec<_>>>()
435        .map(|components| {
436            components
437                .into_iter()
438                .map(|component| substitute(context, &component, &SubstitutionMap::new()))
439                .collect()
440        })
441}

443fn coordinate_lists_equivalent(
444    context: &Context,
445    left: &[ExpressionReference],
446    right: &[ExpressionReference],
447) -> bool {
448    for (left_coordinate, right_coordinate) in left.iter().zip(right) {
449        let residual =
450            builders::subtract(context, left_coordinate.clone(), right_coordinate.clone());
451        if exact_zero_verdict(context, &residual) != ZeroEquivalenceVerdict::TrueValue {
452            return false;
453        }
454    }
455    true
456}

458fn validate_coordinate_hygiene(
459    source_problem: &PartialDifferentialEquationProblem,
460    target_variables: &[SymbolIdentifier],
461    forward_coordinates: &[ExpressionReference],
462    inverse_coordinates: &[ExpressionReference],
463) -> Result<(), PartialDifferentialEquationTransformationDecline> {
464    if forward_coordinates.len() != 2 || inverse_coordinates.len() != 2 {
465        return Err(PartialDifferentialEquationTransformationDecline::CoordinateMapArityMismatch);
466    }
467    let source_variables = source_problem.independent_variables();
468    for coordinate in forward_coordinates {
469        if free_variables(coordinate)
470            .iter()
471            .any(|variable| target_variables.contains(variable))
472        {
473            return Err(
474                PartialDifferentialEquationTransformationDecline::CoordinateMapUsesWrongVariables,
475            );
476        }
477        if contains_dependent_function(coordinate, source_problem) {
478            return Err(PartialDifferentialEquationTransformationDecline::CoordinateMapContainsDependentFunction);
479        }
480    }
481    for coordinate in inverse_coordinates {
482        if free_variables(coordinate)
483            .iter()
484            .any(|variable| source_variables.contains(variable))
485        {
486            return Err(
487                PartialDifferentialEquationTransformationDecline::CoordinateMapUsesWrongVariables,
488            );
489        }
490        if contains_dependent_function(coordinate, source_problem) {
491            return Err(PartialDifferentialEquationTransformationDecline::CoordinateMapContainsDependentFunction);
492        }
493    }
494    Ok(())
495}

497fn contains_dependent_function(
498    expression: &ExpressionReference,
499    problem: &PartialDifferentialEquationProblem,
500) -> bool {
501    let mut result = matches!(expression.kind, ExpressionKind::FunctionCall(function, _) | ExpressionKind::Derivative(function, _, _) if function == problem.dependent_function());
502    let _ = traverse_children(expression, |child| {
503        result |= contains_dependent_function(child, problem);
504        Ok(())
505    });
506    result
507}

509fn validate_compositions(
510    context: &Context,
511    source_variables: &[SymbolIdentifier],
512    target_variables: &[SymbolIdentifier],
513    forward_coordinates: &[ExpressionReference],
514    inverse_coordinates: &[ExpressionReference],
515) -> Result<(), PartialDifferentialEquationTransformationDecline> {
516    let residuals = composition_residuals(
517        context,
518        source_variables,
519        target_variables,
520        forward_coordinates,
521        inverse_coordinates,
522    );
523    if residuals
524        .iter()
525        .any(|residual| exact_zero_verdict(context, residual) == ZeroEquivalenceVerdict::FalseValue)
526    {
527        return Err(PartialDifferentialEquationTransformationDecline::InverseMapRejected);
528    }
529    if residuals
530        .iter()
531        .any(|residual| exact_zero_verdict(context, residual) != ZeroEquivalenceVerdict::TrueValue)
532    {
533        return Err(PartialDifferentialEquationTransformationDecline::InverseMapInconclusive);
534    }
535    Ok(())
536}

538fn composition_residuals(
539    context: &Context,
540    source_variables: &[SymbolIdentifier],
541    target_variables: &[SymbolIdentifier],
542    forward_coordinates: &[ExpressionReference],
543    inverse_coordinates: &[ExpressionReference],
544) -> Vec<ExpressionReference> {
545    let source_expressions = symbol_expressions(context, source_variables);
546    let target_expressions = symbol_expressions(context, target_variables);
547    let forward_mapping = mapping_from_symbols(context, target_variables, forward_coordinates);
548    let inverse_mapping = mapping_from_symbols(context, source_variables, inverse_coordinates);
549    let mut residuals = Vec::with_capacity(4);
550    for (coordinate, source) in inverse_coordinates.iter().zip(source_expressions.iter()) {
551        let composed = substitute(context, coordinate, &forward_mapping);
552        residuals.push(builders::subtract(context, composed, source.clone()));
553    }
554    for (coordinate, target) in forward_coordinates.iter().zip(target_expressions.iter()) {
555        let composed = substitute(context, coordinate, &inverse_mapping);
556        residuals.push(builders::subtract(context, composed, target.clone()));
557    }
558    residuals
559}

561fn forward_jacobian(
562    context: &Context,
563    source_variables: &[SymbolIdentifier],
564    forward_coordinates: &[ExpressionReference],
565) -> ExpressionReference {
566    let first_with_respect_to_first = partial_derivative(
567        context,
568        &forward_coordinates[0],
569        &[(source_variables[0], 1)],
570    );
571    let first_with_respect_to_second = partial_derivative(
572        context,
573        &forward_coordinates[0],
574        &[(source_variables[1], 1)],
575    );
576    let second_with_respect_to_first = partial_derivative(
577        context,
578        &forward_coordinates[1],
579        &[(source_variables[0], 1)],
580    );
581    let second_with_respect_to_second = partial_derivative(
582        context,
583        &forward_coordinates[1],
584        &[(source_variables[1], 1)],
585    );
586    let diagonal = builders::product(
587        context,
588        vec![first_with_respect_to_first, second_with_respect_to_second],
589    );
590    let off_diagonal = builders::product(
591        context,
592        vec![first_with_respect_to_second, second_with_respect_to_first],
593    );
594    builders::subtract(context, diagonal, off_diagonal)
595}

597fn validate_jacobian(
598    context: &Context,
599    jacobian: &ExpressionReference,
600) -> Result<(), PartialDifferentialEquationTransformationDecline> {
601    let zero = builders::zero(context);
602    match context
603        .symbol_table
604        .ask_assumptions(&AssumptionFormula::not_equal(
605            jacobian.clone(),
606            zero.clone(),
607        )) {
608        TruthValue::Yes => Ok(()),
609        TruthValue::No => Err(PartialDifferentialEquationTransformationDecline::JacobianIsZero),
610        TruthValue::Unknown => match zero_equivalent(context, jacobian, None) {
611            ZeroEquivalenceVerdict::FalseValue => {
612                Err(PartialDifferentialEquationTransformationDecline::JacobianIsZero)
613            }
614            _ => Err(PartialDifferentialEquationTransformationDecline::JacobianNonzeroNotProved),
615        },
616    }
617}

619fn transform_problem(
620    context: &Context,
621    source_problem: &PartialDifferentialEquationProblem,
622    coordinate_change: &PartialDifferentialEquationCoordinateChange,
623) -> Result<PartialDifferentialEquationProblem, PartialDifferentialEquationTransformationDecline> {
624    let equation = transform_residual(
625        context,
626        source_problem,
627        source_problem.equation(),
628        coordinate_change,
629        true,
630    )?;
631    let initial_conditions = source_problem
632        .initial_conditions()
633        .iter()
634        .map(|condition| {
635            transform_residual(context, source_problem, condition, coordinate_change, false)
636        })
637        .collect::<Result<Vec<_>, _>>()?;
638    let boundary_conditions = source_problem
639        .boundary_conditions()
640        .iter()
641        .map(|condition| {
642            transform_residual(context, source_problem, condition, coordinate_change, false)
643        })
644        .collect::<Result<Vec<_>, _>>()?;
645    PartialDifferentialEquationProblem::new(
646        equation,
647        PartialDifferentialEquationSignature {
648            dependent_function: coordinate_change.target_dependent_function,
649            independent_variables: coordinate_change.target_variables.clone(),
650        },
651        initial_conditions,
652        boundary_conditions,
653    )
654    .map_err(|_| PartialDifferentialEquationTransformationDecline::TargetProblemInvalid)
655}

657fn transform_residual(
658    context: &Context,
659    source_problem: &PartialDifferentialEquationProblem,
660    residual: &ExpressionReference,
661    coordinate_change: &PartialDifferentialEquationCoordinateChange,
662    rewrite_source_coordinates: bool,
663) -> Result<ExpressionReference, PartialDifferentialEquationTransformationDecline> {
664    let mut occurrences = Vec::new();
665    collect_dependent_occurrences(residual, source_problem, &mut occurrences);
666    let mut mapping = SubstitutionMap::new();
667    for occurrence in occurrences {
668        let transformed_occurrence = transform_dependent_occurrence(
669            context,
670            source_problem,
671            &occurrence,
672            coordinate_change,
673            rewrite_source_coordinates,
674        )?;
675        mapping.insert(occurrence.clone(), transformed_occurrence);
676    }
677    let transformed = substitute(context, residual, &mapping);
678    if rewrite_source_coordinates {
679        let inverse_mapping = mapping_from_symbols(
680            context,
681            &coordinate_change.source_variables,
682            &coordinate_change.inverse_coordinates,
683        );
684        Ok(substitute(context, &transformed, &inverse_mapping))
685    } else {
686        Ok(transformed)
687    }
688}
689
690fn collect_dependent_occurrences(
691    expression: &ExpressionReference,
692    problem: &PartialDifferentialEquationProblem,
693    occurrences: &mut Vec<ExpressionReference>,
694) {
695    if dependent_occurrence_data(expression, problem).is_some() {
696        occurrences.push(expression.clone());
697    }
698    let _ = traverse_children(expression, |child| {
699        collect_dependent_occurrences(child, problem, occurrences);
700        Ok(())
701    });
702}
703
704fn dependent_occurrence_data(
705    expression: &ExpressionReference,
706    problem: &PartialDifferentialEquationProblem,
707) -> Option<(Vec<usize>, Vec<ExpressionReference>)> {
708    match &expression.kind {
709        ExpressionKind::FunctionCall(function, arguments)
710            if *function == problem.dependent_function() =>
711        {
712            Some((vec![0; arguments.len()], arguments.clone()))
713        }
714        ExpressionKind::Derivative(function, orders, arguments)
715            if *function == problem.dependent_function() =>
716        {
717            Some((orders.clone(), arguments.clone()))
718        }
719        _ => None,
720    }
721}
722
723fn transform_dependent_occurrence(
724    context: &Context,
725    source_problem: &PartialDifferentialEquationProblem,
726    occurrence: &ExpressionReference,
727    coordinate_change: &PartialDifferentialEquationCoordinateChange,
728    use_target_coordinate_arguments: bool,
729) -> Result<ExpressionReference, PartialDifferentialEquationTransformationDecline> {
730    let Some((orders, arguments)) = dependent_occurrence_data(occurrence, source_problem) else {
731        return Err(PartialDifferentialEquationTransformationDecline::TargetProblemInvalid);
732    };
733    if orders.len() != 2 || orders.iter().sum::<usize>() > 2 {
734        return Err(
735            PartialDifferentialEquationTransformationDecline::ConditionDerivativeOrderNotSupported,
736        );
737    }
738    let target_arguments = if use_target_coordinate_arguments {
739        symbol_expressions(context, &coordinate_change.target_variables)
740    } else {
741        let source_coordinate_mapping =
742            mapping_from_symbols(context, &coordinate_change.source_variables, &arguments);
743        coordinate_change
744            .forward_coordinates
745            .iter()
746            .map(|coordinate| substitute(context, coordinate, &source_coordinate_mapping))
747            .collect::<Vec<_>>()
748    };
749    chain_rule_operator(context, &orders, coordinate_change, &target_arguments)
750}
751
752fn chain_rule_operator(
753    context: &Context,
754    orders: &[usize],
755    coordinate_change: &PartialDifferentialEquationCoordinateChange,
756    target_arguments: &[ExpressionReference],
757) -> Result<ExpressionReference, PartialDifferentialEquationTransformationDecline> {
758    let target_value = target_jet(
759        context,
760        coordinate_change.target_dependent_function,
761        vec![0, 0],
762        target_arguments,
763    );
764    if orders == [0, 0] {
765        return Ok(target_value);
766    }
767    let source_variables = &coordinate_change.source_variables;
768    let forward = &coordinate_change.forward_coordinates;
769    let first_target = [
770        target_jet(
771            context,
772            coordinate_change.target_dependent_function,
773            vec![1, 0],
774            target_arguments,
775        ),
776        target_jet(
777            context,
778            coordinate_change.target_dependent_function,
779            vec![0, 1],
780            target_arguments,
781        ),
782    ];
783    let second_target = [
784        target_jet(
785            context,
786            coordinate_change.target_dependent_function,
787            vec![2, 0],
788            target_arguments,
789        ),
790        target_jet(
791            context,
792            coordinate_change.target_dependent_function,
793            vec![1, 1],
794            target_arguments,
795        ),
796        target_jet(
797            context,
798            coordinate_change.target_dependent_function,
799            vec![0, 2],
800            target_arguments,
801        ),
802    ];
803    let first_first = partial_derivative(context, &forward[0], &[(source_variables[0], 1)]);
804    let first_second = partial_derivative(context, &forward[0], &[(source_variables[1], 1)]);
805    let second_first = partial_derivative(context, &forward[1], &[(source_variables[0], 1)]);
806    let second_second = partial_derivative(context, &forward[1], &[(source_variables[1], 1)]);
807    let (first_source, second_source) = match orders {
808        [1, 0] => {
809            let first_term = builders::product(context, vec![first_target[0].clone(), first_first]);
810            let second_term =
811                builders::product(context, vec![first_target[1].clone(), second_first]);
812            return Ok(builders::sum(context, vec![first_term, second_term]));
813        }
814        [0, 1] => {
815            let first_term =
816                builders::product(context, vec![first_target[0].clone(), first_second]);
817            let second_term =
818                builders::product(context, vec![first_target[1].clone(), second_second]);
819            return Ok(builders::sum(context, vec![first_term, second_term]));
820        }
821        [2, 0] => (0, 0),
822        [1, 1] => (0, 1),
823        [0, 2] => (1, 1),
824        _ => return Err(
825            PartialDifferentialEquationTransformationDecline::ConditionDerivativeOrderNotSupported,
826        ),
827    };
828    let first_first =
829        partial_derivative(context, &forward[0], &[(source_variables[first_source], 1)]);
830    let first_second = partial_derivative(
831        context,
832        &forward[0],
833        &[(source_variables[second_source], 1)],
834    );
835    let second_first =
836        partial_derivative(context, &forward[1], &[(source_variables[first_source], 1)]);
837    let second_second = partial_derivative(
838        context,
839        &forward[1],
840        &[(source_variables[second_source], 1)],
841    );
842    let first_mixed_product =
843        builders::product(context, vec![first_first.clone(), second_second.clone()]);
844    let second_mixed_product =
845        builders::product(context, vec![first_second.clone(), second_first.clone()]);
846    let mixed_first_target_coefficient =
847        builders::sum(context, vec![first_mixed_product, second_mixed_product]);
848    let first_second_derivative = partial_derivative(
849        context,
850        &forward[0],
851        &[
852            (source_variables[first_source], 1),
853            (source_variables[second_source], 1),
854        ],
855    );
856    let second_second_derivative = partial_derivative(
857        context,
858        &forward[1],
859        &[
860            (source_variables[first_source], 1),
861            (source_variables[second_source], 1),
862        ],
863    );
864    let terms = vec![
865        builders::product(
866            context,
867            vec![
868                second_target[0].clone(),
869                first_first.clone(),
870                first_second.clone(),
871            ],
872        ),
873        builders::product(
874            context,
875            vec![second_target[1].clone(), mixed_first_target_coefficient],
876        ),
877        builders::product(
878            context,
879            vec![
880                second_target[2].clone(),
881                second_first.clone(),
882                second_second.clone(),
883            ],
884        ),
885        builders::product(
886            context,
887            vec![first_target[0].clone(), first_second_derivative],
888        ),
889        builders::product(
890            context,
891            vec![first_target[1].clone(), second_second_derivative],
892        ),
893    ];
894    Ok(builders::sum(context, terms))
895}

897fn target_jet(
898    context: &Context,
899    function: FunctionIdentifier,
900    orders: Vec<usize>,
901    arguments: &[ExpressionReference],
902) -> ExpressionReference {
903    if orders.iter().all(|order| *order == 0) {
904        builders::function_call(context, function, arguments.to_vec())
905    } else {
906        builders::derivative(context, function, orders, arguments.to_vec())
907    }
908}

911    context: &Context,
912    symbols: &[SymbolIdentifier],
913    values: &[ExpressionReference],

915    symbols
916        .iter()
917        .zip(values)
918        .map(|(symbol, value)| {

921                value.clone(),

923        })
924        .collect()
925}

934fn principal_congruence_residuals(
935    context: &Context,
936    source_problem: &PartialDifferentialEquationProblem,
937    transformation: &PartialDifferentialEquationTransformationResult,
938    coordinate_change: &PartialDifferentialEquationCoordinateChange,
939) -> Vec<ExpressionReference> {
940    let Ok(source_classification) = classify_partial_differential_equation(context, source_problem)
941    else {
942        return Vec::new();
943    };
944    let Ok(target_classification) =
945        classify_partial_differential_equation(context, &transformation.target_problem)
946    else {
947        return Vec::new();
948    };
949    let (Some(source_principal), Some(target_principal)) = (
950        source_classification.second_order_principal_part,
951        target_classification.second_order_principal_part,
952    ) else {
953        return Vec::new();
954    };
955    let differential = matrix_from_rows(vec![
956        vec![
957            partial_derivative(
958                context,
959                &coordinate_change.forward_coordinates[0],
960                &[(coordinate_change.source_variables[0], 1)],
961            ),
962            partial_derivative(
963                context,
964                &coordinate_change.forward_coordinates[0],
965                &[(coordinate_change.source_variables[1], 1)],
966            ),
967        ],
968        vec![
969            partial_derivative(
970                context,
971                &coordinate_change.forward_coordinates[1],
972                &[(coordinate_change.source_variables[0], 1)],
973            ),
974            partial_derivative(
975                context,
976                &coordinate_change.forward_coordinates[1],
977                &[(coordinate_change.source_variables[1], 1)],
978            ),
979        ],
980    ])
981    .expect("fixed two-by-two matrix");
982    let source_product = matrix_multiply(context, &differential, &source_principal.matrix);
983    let expected = matrix_multiply(context, &source_product, &matrix_transpose(&differential));
984    let inverse_mapping = mapping_from_symbols(
985        context,
986        &coordinate_change.source_variables,
987        &coordinate_change.inverse_coordinates,
988    );
989    let mut residuals = Vec::with_capacity(expected.entries.len() + 1);
990    for (expected_entry, actual_entry) in expected
991        .entries
992        .iter()
993        .zip(target_principal.matrix.entries.iter())
994    {
995        let expected_target_entry = substitute(context, expected_entry, &inverse_mapping);
996        let normalized_expected_target_entry = builders::product(
997            context,
998            vec![
999                transformation.normalization_multiplier.clone(),
1000                expected_target_entry,
1001            ],
1002        );
1003        residuals.push(builders::subtract(
1004            context,
1005            actual_entry.clone(),
1006            normalized_expected_target_entry,
1007        ));
1008    }
1009    let source_determinant = crate::matrix::matrix_determinant(context, &source_principal.matrix)
1010        .expect("fixed two-by-two principal matrix");
1011    let expected_determinant = substitute(context, &source_determinant, &inverse_mapping);
1012    let square_exponent = builders::integer(context, 2.into());
1013    let jacobian_square = builders::power(
1014        context,
1015        coordinate_change.jacobian.clone(),
1016        square_exponent.clone(),
1017    );
1018    let determinant_target = crate::matrix::matrix_determinant(context, &target_principal.matrix)
1019        .expect("fixed two-by-two principal matrix");
1020    let normalization_square = builders::power(
1021        context,
1022        transformation.normalization_multiplier.clone(),
1023        square_exponent,
1024    );
1025    let expected_scaled_determinant = builders::product(
1026        context,
1027        vec![normalization_square, jacobian_square, expected_determinant],
1028    );
1029    residuals.push(builders::subtract(
1030        context,
1031        determinant_target,
1032        expected_scaled_determinant,
1033    ));
1034    residuals
1035}

1046fn exact_nonzero_verdict(context: &Context, expression: &ExpressionReference) -> TruthValue {
1047    // The relation query is structural, so a Jacobian or normalization multiplier that is a
1048    // nonzero constant only after its cancelling terms are combined is asked twice: once as
1049    // written and once expanded. An undecided answer stays `unknown`, which the aggregate reports
1050    // as inconclusive rather than as a rejection.
1051    let direct = nonzero_relation(context, expression);
1052    if direct != TruthValue::Unknown {
1053        return direct;
1054    }
1055    let expanded = crate::expand::expand(context, expression);
1056    let simplified = crate::simplify::simplify(context, expanded);
1057    nonzero_relation(context, &simplified)
1058}
1059
1060fn nonzero_relation(context: &Context, expression: &ExpressionReference) -> TruthValue {
1061    let zero = builders::zero(context);
1062    context
1063        .symbol_table
1064        .ask_assumptions(&AssumptionFormula::not_equal(expression.clone(), zero))
1065}

1067fn aggregate_verdict(
1068    verdicts: &[ZeroEquivalenceVerdict],
1069    jacobian_verdict: TruthValue,
1070    normalization_verdict: TruthValue,
1071) -> PartialDifferentialEquationVerificationVerdict {
1072    if verdicts
1073        .iter()
1074        .any(|verdict| *verdict == ZeroEquivalenceVerdict::FalseValue)
1075        || jacobian_verdict == TruthValue::No
1076        || normalization_verdict == TruthValue::No
1077    {
1078        PartialDifferentialEquationVerificationVerdict::Rejected
1079    } else if verdicts
1080        .iter()
1081        .all(|verdict| *verdict == ZeroEquivalenceVerdict::TrueValue)
1082        && jacobian_verdict == TruthValue::Yes
1083        && normalization_verdict == TruthValue::Yes
1084    {
1085        PartialDifferentialEquationVerificationVerdict::Verified
1086    } else {
1087        PartialDifferentialEquationVerificationVerdict::Inconclusive
1088    }
1089}

1091fn exact_zero_verdict(context: &Context, residual: &ExpressionReference) -> ZeroEquivalenceVerdict {
1092    let direct = zero_equivalent(context, residual, None);
1093    if direct != ZeroEquivalenceVerdict::Unknown {
1094        return direct;
1095    }
1096    let cancelled = crate::simplify::rational::cancel(context, residual);
1097    let cancelled_verdict = zero_equivalent(context, &cancelled, None);
1098    if cancelled_verdict != ZeroEquivalenceVerdict::Unknown {
1099        return cancelled_verdict;
1100    }
1101    let mut variables = free_variables(&cancelled).into_iter().collect::<Vec<_>>();
1102    variables.sort();
1103    if !variables.is_empty()
1104        && crate::polynomial::conversion::expression_to_polynomial(&cancelled, &variables)
1105            .is_some_and(|polynomial| crate::polynomial::is_zero(&polynomial))
1106    {
1107        return ZeroEquivalenceVerdict::TrueValue;
1108    }
1109    // A residual carrying derivative nodes is not a polynomial in its coordinates, so the
1110    // polynomial route above cannot see it. Distributing products over sums and then combining
1111    // the result over a common denominator is the last exact attempt; a residual still undecided
1112    // here stays unknown, which the caller reports as inconclusive rather than as a rejection.
1113    let expanded = crate::expand::expand(context, &cancelled);
1114    let expanded_verdict = zero_equivalent(context, &expanded, None);
1115    if expanded_verdict != ZeroEquivalenceVerdict::Unknown {
1116        return expanded_verdict;
1117    }
1118    let combined = crate::simplify::rational::together(context, &expanded);
1119    let combined = crate::expand::expand(context, &combined);
1120    let simplified = crate::simplify::simplify(context, combined);
1121    zero_equivalent(context, &simplified, None)
1122}
