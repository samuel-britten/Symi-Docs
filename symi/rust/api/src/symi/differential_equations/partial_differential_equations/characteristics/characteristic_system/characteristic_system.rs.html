1// Projection of PDE-A first-order coefficient data onto the canonical
2// characteristic field, and deterministic proved-safe carrier selection.
3//
4// The classifier remains the sole authority on the shape of the equation; this
5// module never re-traverses the raw governing residual to decide order or
6// linearity. For the residual `a u_x + b u_y + r = 0` the extended
7// characteristic field is `(dx/ds, dy/ds, du/ds) = (a, b, -r)`, and that sign
8// convention is shared by the linear and quasilinear paths alike.

23/// The stable reason the characteristic method returned no solution.
24#[derive(Debug, Clone, PartialEq, Eq, Hash)]
25pub enum PartialDifferentialEquationCharacteristicDecline {
26    /// The equation is not of derivative order one.
27    NotFirstOrder,
28    /// The equation does not have exactly two independent variables.
29    RequiresExactlyTwoIndependentVariables,
30    /// The equation is nonlinear in its first derivatives.
31    FullyNonlinearFirstOrderNotSupported,
32    /// Exact affine first-order coefficient data was unavailable.
33    FirstOrderCoefficientDataUnavailable,
34    /// Both characteristic direction coefficients are exactly zero.
35    DegenerateCharacteristicDirection,
36    /// No carrier coefficient was proved nonzero.
37    CarrierNonzeroNotProved,
38    /// Every proved-safe carrier failed to close the construction.
39    NoSafeCharacteristicCarrier,
40    /// The two characteristic equations form a dependency cycle.
41    CharacteristicEquationsNotTriangular,
42    /// A scalar characteristic ordinary differential equation did not close.
43    CharacteristicScalarOrdinaryDifferentialEquationUnsolved,
44    /// A scalar characteristic route reported no tracked integration constant.
45    CharacteristicOrdinaryDifferentialEquationGeneratedConstantUntracked,
46    /// Exact elimination of a tracked integration constant failed.
47    CharacteristicConstantEliminationFailed,
48    /// A characteristic invariant could not be recovered exactly.
49    CharacteristicInvariantNotRecovered,
50    /// An independently checked invariant residual was proved nonzero.
51    CharacteristicInvariantRejected,
52    /// An invariant residual could not be decided exactly.
53    CharacteristicInvariantVerificationInconclusive,
54    /// Two functionally useful invariants were not obtained.
55    InsufficientIndependentInvariants,
56    /// The dependent value could not be isolated exactly.
57    DependentValueNotIsolatable,
58    /// The natural arbitrary-function family stayed implicit and unverifiable.
59    ImplicitCharacteristicFamilyNotVerifiable,
60    /// The supplied initial condition is not one supported Cauchy value relation.
61    CauchyConditionNotSupported,
62    /// More than one initial condition could drive the characteristic construction.
63    MultipleCauchyConditionsAmbiguous,
64    /// No unique curve parameter could be identified.
65    CauchyParameterAmbiguous,
66    /// The initial curve has a proved-zero tangent.
67    InitialCurveDegenerate,
68    /// Initial-curve regularity could not be established exactly.
69    InitialCurveRegularityUnknown,
70    /// The characteristic transversality determinant could not be decided.
71    TransversalityUnknown,
72    /// The characteristic field vanishes on the initial curve.
73    DegenerateCharacteristicFieldOnInitialCurve,
74    /// Characteristic initial data contradict the characteristic value equation.
75    InconsistentCharacteristicData,
76    /// Compatible characteristic data do not determine a local solution.
77    CharacteristicDataNotLocallyDetermining,
78    /// Compatibility of characteristic initial data could not be decided.
79    CharacteristicDataCompatibilityUnknown,
80    /// The invariant label could not be inverted exactly along the initial curve.
81    CauchyLabelInversionFailed,
82    /// Exact label inversion did not produce one unambiguous branch.
83    CauchyLabelBranchAmbiguous,
84    /// A characteristic projection or required definedness fact was unresolved.
85    CharacteristicProjectionSingularOrUnknown,
86    /// The central verifier could not decide the candidate.
87    CandidateVerificationInconclusive,
88    /// The central verifier proved the candidate wrong.
89    CandidateRejectedByVerifier,
90}

92/// Which independent variable carries the characteristic evolution.
93#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
94pub enum PartialDifferentialEquationCharacteristicCarrier {
95    /// The first declared independent variable carries the evolution.
96    FirstIndependentVariable,
97    /// The second declared independent variable carries the evolution.
98    SecondIndependentVariable,
99}

101/// The canonical first-order characteristic form of a two-variable equation.
102#[derive(Debug, Clone)]
103pub struct PartialDifferentialEquationFirstOrderForm {
104    /// Stores the first declared independent variable.
105    pub first_independent_variable: SymbolIdentifier,
106    /// Stores the second declared independent variable.
107    pub second_independent_variable: SymbolIdentifier,
108    /// Stores the dependent function identifier.
109    pub dependent_function: FunctionIdentifier,
110    /// Stores the coefficient of the derivative with multi-index `[1, 0]`.
111    pub first_derivative_coefficient: ExpressionReference,
112    /// Stores the coefficient of the derivative with multi-index `[0, 1]`.
113    pub second_derivative_coefficient: ExpressionReference,
114    /// Stores the residual left after removing both derivative terms.
115    pub remainder: ExpressionReference,
116}
117
118impl PartialDifferentialEquationFirstOrderForm {
119    /// Returns the coefficient dividing when the named carrier is selected.
120    pub fn carrier_coefficient(
121        &self,
122        carrier: PartialDifferentialEquationCharacteristicCarrier,
123    ) -> &ExpressionReference {
124        match carrier {
125            PartialDifferentialEquationCharacteristicCarrier::FirstIndependentVariable => {
126                &self.first_derivative_coefficient
127            }
128            PartialDifferentialEquationCharacteristicCarrier::SecondIndependentVariable => {
129                &self.second_derivative_coefficient
130            }
131        }
132    }
133
134    /// Returns the coefficient of the variable evolving against the carrier.
135    pub fn transverse_coefficient(
136        &self,
137        carrier: PartialDifferentialEquationCharacteristicCarrier,
138    ) -> &ExpressionReference {
139        match carrier {
140            PartialDifferentialEquationCharacteristicCarrier::FirstIndependentVariable => {
141                &self.second_derivative_coefficient
142            }
143            PartialDifferentialEquationCharacteristicCarrier::SecondIndependentVariable => {
144                &self.first_derivative_coefficient
145            }
146        }
147    }
148
149    /// Returns the independent variable carrying the evolution.
150    pub fn carrier_variable(
151        &self,
152        carrier: PartialDifferentialEquationCharacteristicCarrier,
153    ) -> SymbolIdentifier {
154        match carrier {
155            PartialDifferentialEquationCharacteristicCarrier::FirstIndependentVariable => {
156                self.first_independent_variable
157            }
158            PartialDifferentialEquationCharacteristicCarrier::SecondIndependentVariable => {
159                self.second_independent_variable
160            }
161        }
162    }
163
164    /// Returns the independent variable evolving against the carrier.
165    pub fn transverse_variable(
166        &self,
167        carrier: PartialDifferentialEquationCharacteristicCarrier,
168    ) -> SymbolIdentifier {
169        match carrier {
170            PartialDifferentialEquationCharacteristicCarrier::FirstIndependentVariable => {
171                self.second_independent_variable
172            }
173            PartialDifferentialEquationCharacteristicCarrier::SecondIndependentVariable => {
174                self.first_independent_variable
175            }
176        }
177    }
178}

180pub(crate) fn first_order_form(
181    context: &Context,
182    problem: &PartialDifferentialEquationProblem,
183    classification: &PartialDifferentialEquationClassification,
184) -> Result<
185    PartialDifferentialEquationFirstOrderForm,
186    PartialDifferentialEquationCharacteristicDecline,
187> {
188    // Consume the classifier's affine coefficient data rather than reading the
189    // governing residual again. A linear equation reports the zero-order jet
190    // separately from the jet-free remainder, so its coefficient of the
191    // dependent value is folded back into the characteristic remainder here;
192    // the semilinear and quasilinear reports already carry it.
193    if classification.order != 1 {
194        return Err(PartialDifferentialEquationCharacteristicDecline::NotFirstOrder);
195    }
196    if classification.independent_variables.len() != 2 {
197        return Err(
198            PartialDifferentialEquationCharacteristicDecline::RequiresExactlyTwoIndependentVariables,
199        );
200    }
201    if classification.linearity == PartialDifferentialEquationLinearity::Nonlinear {
202        return Err(
203            PartialDifferentialEquationCharacteristicDecline::FullyNonlinearFirstOrderNotSupported,
204        );
205    }
206    let data = &classification.coefficient_data;
207    let (records, jet_free_remainder) = match (
208        data.full_coefficients.as_ref(),
209        data.highest_order_coefficients.as_ref(),
210        data.remainder.as_ref(),
211    ) {
212        (Some(full), _, Some(remainder)) => (full, remainder.clone()),
213        (None, Some(highest), Some(remainder)) => (highest, remainder.clone()),
214        _ => {
215            return Err(
216                PartialDifferentialEquationCharacteristicDecline::FirstOrderCoefficientDataUnavailable,
217            );
218        }
219    };
220    let zero = builders::zero(context);
221    let coefficient_for = |orders: &[usize]| {
222        records
223            .iter()
224            .find(|record| record.jet.derivative_orders == orders)
225            .map(|record| record.coefficient.clone())
226    };
227    let first_derivative_coefficient = coefficient_for(&[1, 0]).unwrap_or_else(|| zero.clone());
228    let second_derivative_coefficient = coefficient_for(&[0, 1]).unwrap_or_else(|| zero.clone());
229    let dependent_function = classification.dependent_function;
230    let dependent_call = dependent_call(context, problem);
231    let mut remainder_terms = vec![jet_free_remainder];
232    for record in records {
233        if record.jet.derivative_orders.iter().all(|order| *order == 0) {
234            remainder_terms.push(builders::product(
235                context,
236                vec![record.coefficient.clone(), dependent_call.clone()],
237            ));
238        }
239    }
240    let remainder = builders::sum(context, remainder_terms);
241    let form = PartialDifferentialEquationFirstOrderForm {
242        first_independent_variable: classification.independent_variables[0],
243        second_independent_variable: classification.independent_variables[1],
244        dependent_function,
245        first_derivative_coefficient,
246        second_derivative_coefficient,
247        remainder,
248    };
249    if contains_dependent_derivative(&form.first_derivative_coefficient, dependent_function)
250        || contains_dependent_derivative(&form.second_derivative_coefficient, dependent_function)
251        || contains_dependent_derivative(&form.remainder, dependent_function)
252    {
253        return Err(
254            PartialDifferentialEquationCharacteristicDecline::FullyNonlinearFirstOrderNotSupported,
255        );
256    }
257    Ok(form)
258}

260/// Builds the canonical dependent-function call in the declared variables.
261pub(crate) fn dependent_call(
262    context: &Context,
263    problem: &PartialDifferentialEquationProblem,
264) -> ExpressionReference {
265    let arguments = problem
266        .independent_variables()
267        .iter()
268        .map(|variable| context.interner.intern(ExpressionKind::Symbol(*variable)))
269        .collect();
270    builders::function_call(context, problem.dependent_function(), arguments)
271}

273pub(crate) fn proved_safe_carriers(
274    context: &Context,
275    form: &PartialDifferentialEquationFirstOrderForm,
276) -> Result<
277    Vec<PartialDifferentialEquationCharacteristicCarrier>,
278    PartialDifferentialEquationCharacteristicDecline,
279> {
280    // Only a coefficient proved nonzero may be divided by. An unknown verdict
281    // is never treated as permission; when neither coefficient is decided the
282    // method declines instead of choosing a branch it cannot justify.
283    let first_verdict = is_nonzero(&form.first_derivative_coefficient, &context.symbol_table);
284    let second_verdict = is_nonzero(&form.second_derivative_coefficient, &context.symbol_table);
285    if first_verdict == TruthValue::No && second_verdict == TruthValue::No {
286        return Err(
287            PartialDifferentialEquationCharacteristicDecline::DegenerateCharacteristicDirection,
288        );
289    }
290    let mut eligible = Vec::new();
291    if first_verdict == TruthValue::Yes {
292        eligible.push(PartialDifferentialEquationCharacteristicCarrier::FirstIndependentVariable);
293    }
294    if second_verdict == TruthValue::Yes {
295        eligible.push(PartialDifferentialEquationCharacteristicCarrier::SecondIndependentVariable);
296    }
297    if eligible.is_empty() {
298        return Err(PartialDifferentialEquationCharacteristicDecline::CarrierNonzeroNotProved);
299    }
300    let mut scored: Vec<_> = eligible
301        .into_iter()
302        .map(|carrier| (carrier_score(context, form, carrier), carrier))
303        .collect();
304    scored.sort_by(|left, right| left.0.cmp(&right.0));
305    Ok(scored.into_iter().map(|(_, carrier)| carrier).collect())
306}

308fn carrier_score(
309    context: &Context,
310    form: &PartialDifferentialEquationFirstOrderForm,
311    carrier: PartialDifferentialEquationCharacteristicCarrier,
312) -> (usize, usize, usize, usize) {
313    // Deterministic carrier preference: an exact unit coefficient first, then a
314    // division that leaves no reciprocal behind, then the lighter pair of
315    // characteristic right-hand sides, then the declared variable order.
316    let coefficient = form.carrier_coefficient(carrier).clone();
317    let transverse = form.transverse_coefficient(carrier).clone();
318    let base_quotient = builders::divide(context, transverse, coefficient.clone());
319    let negated_remainder = builders::negate(context, form.remainder.clone());
320    let value_quotient = builders::divide(context, negated_remainder, coefficient.clone());
321    let unit_rank = usize::from(!is_exact_unit(context, &coefficient));
322    let one = builders::one(context);
323    let reciprocal = builders::divide(context, one, coefficient);
324    let denominator_rank = usize::from(
325        contains_reference(&base_quotient, &reciprocal)
326            || contains_reference(&value_quotient, &reciprocal),
327    );
328    let weight = crate::simplify::heuristic::node_count(&base_quotient)
329        + crate::simplify::heuristic::node_count(&value_quotient);
330    let declared_position = match carrier {
331        PartialDifferentialEquationCharacteristicCarrier::FirstIndependentVariable => 0,
332        PartialDifferentialEquationCharacteristicCarrier::SecondIndependentVariable => 1,
333    };
334    (unit_rank, denominator_rank, weight, declared_position)
335}

337fn is_exact_unit(context: &Context, coefficient: &ExpressionReference) -> bool {
338    let one = builders::one(context);
339    let negative_one = builders::negate(context, one.clone());
340    coefficient.kind == one.kind || coefficient.kind == negative_one.kind
341}

343fn contains_reference(expression: &ExpressionReference, target: &ExpressionReference) -> bool {
344    if expression.kind == target.kind {
345        return true;
346    }
347    let mut found = false;
348    let _ = traverse_children(expression, |child| {
349        found |= contains_reference(child, target);
350        Ok(())
351    });
352    found
353}

355pub(crate) fn contains_dependent_derivative(
356    expression: &ExpressionReference,
357    dependent_function: FunctionIdentifier,
358) -> bool {
359    let mut found = matches!(
360        expression.kind,
361        ExpressionKind::Derivative(function, _, _) if function == dependent_function
362    );
363    let _ = traverse_children(expression, |child| {
364        found |= contains_dependent_derivative(child, dependent_function);
365        Ok(())
366    });
367    found
368}
