1// Riccati classifier. Recognises `y'(x) = q_0(x) + q_1(x) * y(x) +
2// q_2(x) * y(x)^2` with q_2 ≢ 0. Two closure paths:
3//
4//   (a) When the caller supplies a particular solution y_1(x), the
5//       substitution y = y_1 + 1/v(x) linearises the equation to a
6//       first-order linear ODE on v, which the linear-first-order
7//       classifier then closes; the
8//       result is back-substituted to recover y(x).
9//   (b) Without a particular solution, the substitution
10//       y = -u'(x) / (q_2(x) * u(x)) reduces the Riccati to the
11//       second-order linear ODE
12//         u'' - (q_1 + q_2'/q_2) * u' + (q_0 * q_2) * u = 0
13//       on u. This classifier routes that linear equation through `kovacic::solve`;
14//       a successful closure produces u(x) from which y = -u'/(q_2 * u)
15//       follows by differentiation. When Kovacic returns None the
16//       Riccati path also returns None, and the dispatcher falls
17//       through to the unsolved-ODE placeholder.

33/// Performs the advanced native `solve` operation.
34pub fn solve(
35    context: &Context,
36    equation: &ExpressionReference,
37    dependent_variable_call: &ExpressionReference,
38    independent_variable: SymbolIdentifier,
39    particular_solution: Option<&ExpressionReference>,
40) -> Option<ExpressionReference> {
41    // Eager entry. With a particular solution, the substitution
42    // y = y_1 + 1/v linearises the equation through the linear-first-order
43    // classifier. Without a
44    // particular solution, this routes the Riccati-corresponding
45    // second-order linear ODE through `kovacic::solve` and recovers
46    // y via the back-substitution y = -u'/(q_2 * u).
47    let function = dependent_function_identifier(dependent_variable_call)?;
48    let (q_zero, q_one, q_two) =
49        extract_riccati_coefficients(context, equation, function, independent_variable)?;
50    let particular = match particular_solution {
51        Some(value) => value,
52        None => {
53            return solve_via_kovacic_closure(
54                context,
55                &q_zero,
56                &q_one,
57                &q_two,
58                independent_variable,
59            );
60        }
61    };
62    let auxiliary_function_identifier = context.symbol_table.fresh("v_riccati");
63    let auxiliary_function = FunctionIdentifier::User(auxiliary_function_identifier);
64    let x_symbol = context
65        .interner
66        .intern(ExpressionKind::Symbol(independent_variable));
67    let v_call = builders::function_call(context, auxiliary_function, vec![Arc::clone(&x_symbol)]);
68    let v_prime_call = builders::derivative(
69        context,
70        auxiliary_function,
71        vec![1],
72        vec![Arc::clone(&x_symbol)],
73    );
74    let one = builders::one(context);
75    let reciprocal_v = builders::divide(context, Arc::clone(&one), Arc::clone(&v_call));
76    let y_substitute = builders::sum(
77        context,
78        vec![Arc::clone(particular), Arc::clone(&reciprocal_v)],
79    );
80    let particular_derivative =
81        crate::expression::differentiate::differentiate(context, particular, independent_variable);
82    let two = builders::integer_value(context, 2);
83    let v_squared = builders::power(context, Arc::clone(&v_call), two);
84    let v_prime_over_v_squared = builders::divide(context, Arc::clone(&v_prime_call), v_squared);
85    let negative_v_prime_over_v_squared = builders::negate(context, v_prime_over_v_squared);
86    let y_prime_substitute = builders::sum(
87        context,
88        vec![particular_derivative, negative_v_prime_over_v_squared],
89    );
90    let y_call = derivative_call(context, function, independent_variable, 0);
91    let y_prime_call = derivative_call(context, function, independent_variable, 1);
92    let mut substitution: SubstitutionMap = HashMap::new();
93    substitution.insert(Arc::clone(&y_call), y_substitute);
94    substitution.insert(Arc::clone(&y_prime_call), y_prime_substitute);
95    let substituted_equation = substitute(context, equation, &substitution);
96    if depends_on_function(&substituted_equation, function) {
97        return None;
98    }
99    let v_closed_form = crate::differential_equations::linear_first_order::solve(
100        context,
101        &substituted_equation,
102        &v_call,
103        independent_variable,
104    )?;
105    let one_over_v = builders::divide(context, one, v_closed_form);
106    Some(builders::sum(
107        context,
108        vec![Arc::clone(particular), one_over_v],
109    ))
110}

112fn extract_riccati_coefficients(
113    context: &Context,
114    equation: &ExpressionReference,
115    function: FunctionIdentifier,
116    independent_variable: SymbolIdentifier,
117) -> Option<(
118    ExpressionReference,
119    ExpressionReference,
120    ExpressionReference,
121)> {
122    // Pull (q_0(x), q_1(x), q_2(x)) out of the canonical Riccati shape
123    // `y'(x) = q_0 + q_1 * y + q_2 * y^2`. The strategy is identical to
124    // the Bernoulli extractor's approach: substitute fresh-symbol
125    // surrogates for y'(x) and y(x), then read off the quadratic
126    // polynomial in y from the residual after isolating y'(x).
127    let y_call = derivative_call(context, function, independent_variable, 0);
128    let y_prime_call = derivative_call(context, function, independent_variable, 1);
129    let y_surrogate_identifier = context.symbol_table.fresh("y_riccati");
130    let y_prime_surrogate_identifier = context.symbol_table.fresh("y_prime_riccati");
131    let y_surrogate = context
132        .interner
133        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
134    let y_prime_surrogate = context
135        .interner
136        .intern(ExpressionKind::Symbol(y_prime_surrogate_identifier));
137    let mut atom_mapping: SubstitutionMap = HashMap::new();
138    atom_mapping.insert(Arc::clone(&y_call), Arc::clone(&y_surrogate));
139    atom_mapping.insert(Arc::clone(&y_prime_call), Arc::clone(&y_prime_surrogate));
140    let surrogate_equation = substitute(context, equation, &atom_mapping);
141    if depends_on_function(&surrogate_equation, function) {
142        return None;
143    }
144    let coefficient_of_y_prime = crate::expression::differentiate::differentiate(
145        context,
146        &surrogate_equation,
147        y_prime_surrogate_identifier,
148    );
149    let zero = builders::zero(context);
150    let mut zero_y_prime: SubstitutionMap = HashMap::new();
151    zero_y_prime.insert(Arc::clone(&y_prime_surrogate), Arc::clone(&zero));
152    let a_of_x = substitute(context, &coefficient_of_y_prime, &zero_y_prime);
153    if is_integer_zero(&a_of_x)
154        || crate::expression::queries::depends_on(&a_of_x, y_surrogate_identifier)
155    {
156        return None;
157    }
158    let a_times_y_prime = builders::product(
159        context,
160        vec![Arc::clone(&a_of_x), Arc::clone(&y_prime_surrogate)],
161    );
162    let residual = builders::subtract(context, Arc::clone(&surrogate_equation), a_times_y_prime);
163    if crate::expression::queries::depends_on(&residual, y_prime_surrogate_identifier) {
164        return None;
165    }
166    let two = builders::integer_value(context, 2);
167    let y_squared = builders::power(context, Arc::clone(&y_surrogate), two);
168    let q_two_numerator = coefficient_of_polynomial_term(context, &residual, &y_squared)?;
169    if crate::expression::queries::depends_on(&q_two_numerator, y_surrogate_identifier) {
170        return None;
171    }
172    let q_two_times_y_squared =
173        builders::product(context, vec![Arc::clone(&q_two_numerator), y_squared]);
174    let residual_without_q_two =
175        builders::subtract(context, Arc::clone(&residual), q_two_times_y_squared);
176    let q_one_numerator =
177        coefficient_of_polynomial_term(context, &residual_without_q_two, &y_surrogate)?;
178    if crate::expression::queries::depends_on(&q_one_numerator, y_surrogate_identifier) {
179        return None;
180    }
181    let q_one_times_y = builders::product(
182        context,
183        vec![Arc::clone(&q_one_numerator), Arc::clone(&y_surrogate)],
184    );
185    let q_zero_numerator = builders::subtract(context, residual_without_q_two, q_one_times_y);
186    if crate::expression::queries::depends_on(&q_zero_numerator, y_surrogate_identifier) {
187        return None;
188    }
189    let negate_and_divide = |context: &Context,
190                             value: ExpressionReference,
191                             divisor: &ExpressionReference|
192     -> ExpressionReference {
193        let negated = builders::negate(context, value);
194        builders::divide(context, negated, Arc::clone(divisor))
195    };
196    let q_two = negate_and_divide(context, q_two_numerator, &a_of_x);
197    let q_one = negate_and_divide(context, q_one_numerator, &a_of_x);
198    let q_zero = negate_and_divide(context, q_zero_numerator, &a_of_x);
199    if is_integer_zero(&q_two) {
200        return None;
201    }
202    Some((q_zero, q_one, q_two))
203}

205fn coefficient_of_polynomial_term(
206    context: &Context,
207    target: &ExpressionReference,
208    needle: &ExpressionReference,
209) -> Option<ExpressionReference> {
210    // Sum the coefficients of every summand of `target` whose product
211    // contains `needle` as a factor exactly once. Other factors in such a
212    // summand contribute to the coefficient verbatim. Returns None when
213    // any summand contains `needle` as a non-product factor or as a power
214    // > 1 (which would indicate a polynomial degree mismatch with the
215    // canonical Riccati template).
216    let summands: Vec<ExpressionReference> = match &target.kind {
217        ExpressionKind::Sum(children) => children.iter().map(Arc::clone).collect(),
218        _ => vec![Arc::clone(target)],
219    };
220    let mut coefficient_terms: Vec<ExpressionReference> = Vec::new();
221    for summand in summands {
222        let factors: Vec<ExpressionReference> = match &summand.kind {
223            ExpressionKind::Product(children) => children.iter().map(Arc::clone).collect(),
224            _ => vec![Arc::clone(&summand)],
225        };
226        let mut needle_count: usize = 0;
227        let mut remainder_factors: Vec<ExpressionReference> = Vec::new();
228        for factor in factors {
229            if factor.kind == needle.kind {
230                needle_count += 1;
231            } else {
232                remainder_factors.push(factor);
233            }
234        }
235        if needle_count == 0 {
236            continue;
237        }
238        if needle_count != 1 {
239            return None;
240        }
241        let coefficient = if remainder_factors.is_empty() {
242            builders::one(context)
243        } else {
244            builders::product(context, remainder_factors)
245        };
246        coefficient_terms.push(coefficient);
247    }
248    if coefficient_terms.is_empty() {
249        return Some(builders::zero(context));
250    }
251    Some(builders::sum(context, coefficient_terms))
252}

254fn is_integer_zero(target: &ExpressionReference) -> bool {
255    // Predicate identifying the integer literal zero. Used by this
256    // extractor to detect a degenerate a(x) ≡ 0 case before dividing
257    // through, and to detect a q_2 ≡ 0 case (which reduces to the
258    // linear-first-order classifier).
259    use crate::exact_arithmetic::Integer;
260    matches!(&target.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
261}

263fn solve_via_kovacic_closure(
264    context: &Context,
265    q_zero: &ExpressionReference,
266    q_one: &ExpressionReference,
267    q_two: &ExpressionReference,
268    independent_variable: SymbolIdentifier,
269) -> Option<ExpressionReference> {
270    // Riccati closure via Kovacic. The Riccati y' = q_0 + q_1*y + q_2*y^2
271    // corresponds under y = -u'/(q_2 * u) to the second-order linear
272    // ODE
273    //   u'' - (q_1 + q_2'/q_2) * u' + (q_0 * q_2) * u = 0.
274    // We build that ODE explicitly with a fresh auxiliary function u,
275    // route it through `kovacic::solve`, then differentiate u to
276    // recover y. The two-parameter family in u collapses to a one-
277    // parameter family in y through the ratio invariance of the
278    // substitution; the closed form returned here keeps both u-
279    // parameters visible so the user can specialise them.
280    let auxiliary_function_identifier = context.symbol_table.fresh("u_riccati_kovacic");
281    let auxiliary_function = FunctionIdentifier::User(auxiliary_function_identifier);
282    let x_symbol = context
283        .interner
284        .intern(ExpressionKind::Symbol(independent_variable));
285    let u_call = builders::function_call(context, auxiliary_function, vec![Arc::clone(&x_symbol)]);
286    let u_prime_call = builders::derivative(
287        context,
288        auxiliary_function,
289        vec![1],
290        vec![Arc::clone(&x_symbol)],
291    );
292    let u_double_prime_call = builders::derivative(
293        context,
294        auxiliary_function,
295        vec![2],
296        vec![Arc::clone(&x_symbol)],
297    );
298    let q_two_prime =
299        crate::expression::differentiate::differentiate(context, q_two, independent_variable);
300    let q_two_prime_over_q_two = builders::divide(context, q_two_prime, Arc::clone(q_two));
301    let p_coefficient = builders::sum(context, vec![Arc::clone(q_one), q_two_prime_over_q_two]);
302    let p_times_u_prime = builders::product(
303        context,
304        vec![Arc::clone(&p_coefficient), Arc::clone(&u_prime_call)],
305    );
306    let negative_p_times_u_prime = builders::negate(context, p_times_u_prime);
307    let q_coefficient = builders::product(context, vec![Arc::clone(q_zero), Arc::clone(q_two)]);
308    let q_times_u = builders::product(context, vec![q_coefficient, Arc::clone(&u_call)]);
309    let auxiliary_equation = builders::sum(
310        context,
311        vec![u_double_prime_call, negative_p_times_u_prime, q_times_u],
312    );
313    let u_solution = crate::differential_equations::kovacic::solve(
314        context,
315        &auxiliary_equation,
316        &u_call,
317        independent_variable,
318    )?;
319    let u_solution_derivative =
320        crate::expression::differentiate::differentiate(context, &u_solution, independent_variable);
321    let q_two_times_u_solution = builders::product(context, vec![Arc::clone(q_two), u_solution]);
322    let ratio = builders::divide(context, u_solution_derivative, q_two_times_u_solution);
323    Some(builders::negate(context, ratio))
324}
