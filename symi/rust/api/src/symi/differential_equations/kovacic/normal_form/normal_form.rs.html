1// Normal-form reduction. Given a homogeneous second-order linear
2// ODE
3//
4//   a(x) * y''(x) + b(x) * y'(x) + c(x) * y(x) = 0
5//
6// with a, b, c in Q(x), the substitution y = u * exp(-(1/2) *
7// integral(b/a) dx) eliminates the first-derivative term and yields
8// the Kovacic normal form
9//
10//   u''(x) = r(x) * u(x),    r = (b^2 + 2*a*b' - 2*a'*b - 4*a*c) / (4*a^2)
11//
12// where the derivation follows Kovacic 1986 §3. r is itself a rational
13// function in Q(x). The transform factor exp(-(1/2) * integral(b/a))
14// is returned alongside r so the back-substitution in
15// [reconstruction.rs](./reconstruction.rs) can recover y from u.
16

37/// Performs the advanced native `reduce` operation.
38pub fn reduce(
39    context: &Context,
40    a_expression: &ExpressionReference,
41    b_expression: &ExpressionReference,
42    c_expression: &ExpressionReference,
43    independent_variable: SymbolIdentifier,
44) -> Option<(Polynomial, Polynomial, ExpressionReference)> {
45    // Compute r = (b^2 + 2*a*b' - 2*a'*b - 4*a*c) / (4*a^2) as a
46    // numerator / denominator polynomial pair in Q[x], and the
47    // transform factor exp(-(1/2) * integral(b/a) dx) as an
48    // expression_reference. Returns None when any of a, b, c falls
49    // outside Q(x) (the rational-pair extractor refuses non-rational
50    // shapes) or when a is structurally zero.
51    let (a_numerator, a_denominator) =
52        expression_to_rational_pair(a_expression, independent_variable, context.operation())?;
53    if polynomial_is_zero(&a_numerator) {
54        return None;
55    }
56    let (b_numerator, b_denominator) =
57        expression_to_rational_pair(b_expression, independent_variable, context.operation())?;
58    let (c_numerator, c_denominator) =
59        expression_to_rational_pair(c_expression, independent_variable, context.operation())?;
60    let (b_squared_numerator, b_squared_denominator) =
61        rational_multiply(&b_numerator, &b_denominator, &b_numerator, &b_denominator);
62    let (a_prime_numerator, a_prime_denominator) =
63        rational_derivative(&a_numerator, &a_denominator);
64    let (b_prime_numerator, b_prime_denominator) =
65        rational_derivative(&b_numerator, &b_denominator);
66    let (term_two_numerator, term_two_denominator) = rational_multiply_scalar(
67        2,
68        &rational_multiply(
69            &a_numerator,
70            &a_denominator,
71            &b_prime_numerator,
72            &b_prime_denominator,
73        ),
74    );
75    let (term_three_numerator, term_three_denominator) = rational_multiply_scalar(
76        2,
77        &rational_multiply(
78            &a_prime_numerator,
79            &a_prime_denominator,
80            &b_numerator,
81            &b_denominator,
82        ),
83    );
84    let (term_four_numerator, term_four_denominator) = rational_multiply_scalar(
85        4,
86        &rational_multiply(&a_numerator, &a_denominator, &c_numerator, &c_denominator),
87    );
88    let (sum_one_numerator, sum_one_denominator) = rational_add(
89        &b_squared_numerator,
90        &b_squared_denominator,
91        &term_two_numerator,
92        &term_two_denominator,
93    );
94    let (sum_two_numerator, sum_two_denominator) = rational_subtract(
95        &sum_one_numerator,
96        &sum_one_denominator,
97        &term_three_numerator,
98        &term_three_denominator,
99    );
100    let (numerator_of_r_times_four_a_squared, denominator_of_r_times_four_a_squared) =
101        rational_subtract(
102            &sum_two_numerator,
103            &sum_two_denominator,
104            &term_four_numerator,
105            &term_four_denominator,
106        );
107    let (four_a_squared_numerator, four_a_squared_denominator) = rational_multiply_scalar(
108        4,
109        &rational_multiply(&a_numerator, &a_denominator, &a_numerator, &a_denominator),
110    );
111    let (r_numerator_raw, r_denominator_raw) = rational_divide(
112        &numerator_of_r_times_four_a_squared,
113        &denominator_of_r_times_four_a_squared,
114        &four_a_squared_numerator,
115        &four_a_squared_denominator,
116    )?;
117    let (r_numerator, r_denominator) =
118        reduce_rational_pair(r_numerator_raw, r_denominator_raw, context.operation());
119    let transform_factor =
120        build_transform_factor(context, a_expression, b_expression, independent_variable);
121    Some((r_numerator, r_denominator, transform_factor))
122}

124fn build_transform_factor(
125    context: &Context,
126    a_expression: &ExpressionReference,
127    b_expression: &ExpressionReference,
128    independent_variable: SymbolIdentifier,
129) -> ExpressionReference {
130    // Construct exp(-(1/2) * integral(b/a) dx) as an expression. The
131    // factor multiplies the u-side solution to recover the original y.
132    // When integrate stalls the returned expression contains an
133    // unevaluated integral node; reconstruction.rs keeps that visible
134    // so the four-arms discipline still applies to the final
135    // expression.
136    let zero = builders::zero(context);
137    if b_expression.kind == zero.kind {
138        return builders::one(context);
139    }
140    let b_over_a = builders::divide(context, Arc::clone(b_expression), Arc::clone(a_expression));
141    let integral_b_over_a = crate::integrate::integrate(context, &b_over_a, independent_variable);
142    let half = builders::rational_value(context, 1, 2);
143    let negative_half = builders::negate(context, half);
144    let exponent = builders::product(context, vec![negative_half, integral_b_over_a]);
145    builders::function_call(
146        context,
147        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
148        vec![exponent],
149    )
150}

152fn rational_multiply(
153    left_numerator: &Polynomial,
154    left_denominator: &Polynomial,
155    right_numerator: &Polynomial,
156    right_denominator: &Polynomial,
157) -> (Polynomial, Polynomial) {
158    // Multiply two rational pairs without applying the gcd cancellation;
159    // the caller batches reductions at the end of the normal-form
160    // computation through reduce_rational_pair.
161    let numerator = polynomial_multiply(left_numerator, right_numerator);
162    let denominator = polynomial_multiply(left_denominator, right_denominator);
163    (numerator, denominator)
164}

166fn rational_add(
167    left_numerator: &Polynomial,
168    left_denominator: &Polynomial,
169    right_numerator: &Polynomial,
170    right_denominator: &Polynomial,
171) -> (Polynomial, Polynomial) {
172    // Add two rational pairs as (a*d + c*b) / (b*d). The reducer applies
173    // at the end of the normal-form computation, not on every step.
174    let cross_left = polynomial_multiply(left_numerator, right_denominator);
175    let cross_right = polynomial_multiply(right_numerator, left_denominator);
176    let numerator = polynomial_add(&cross_left, &cross_right);
177    let denominator = polynomial_multiply(left_denominator, right_denominator);
178    (numerator, denominator)
179}
180
181fn rational_subtract(
182    left_numerator: &Polynomial,
183    left_denominator: &Polynomial,
184    right_numerator: &Polynomial,
185    right_denominator: &Polynomial,
186) -> (Polynomial, Polynomial) {
187    // (a/b) - (c/d) = (a*d - c*b) / (b*d).
188    let cross_left = polynomial_multiply(left_numerator, right_denominator);
189    let cross_right = polynomial_multiply(right_numerator, left_denominator);
190    let numerator = polynomial_subtract(&cross_left, &cross_right);
191    let denominator = polynomial_multiply(left_denominator, right_denominator);
192    (numerator, denominator)
193}
194
195fn rational_divide(
196    left_numerator: &Polynomial,
197    left_denominator: &Polynomial,
198    right_numerator: &Polynomial,
199    right_denominator: &Polynomial,
200) -> Option<(Polynomial, Polynomial)> {
201    // (a/b) / (c/d) = (a*d) / (b*c). Returns None on division by zero
202    // (right_numerator is the zero polynomial).
203    if polynomial_is_zero(right_numerator) {
204        return None;
205    }
206    let numerator = polynomial_multiply(left_numerator, right_denominator);
207    let denominator = polynomial_multiply(left_denominator, right_numerator);
208    Some((numerator, denominator))
209}

211fn rational_multiply_scalar(
212    scalar: i64,
213    pair: &(Polynomial, Polynomial),
214) -> (Polynomial, Polynomial) {
215    // Multiply a rational pair by an integer scalar. Implemented as a
216    // multiplication against a constant polynomial so the polynomial-
217    // arithmetic path stays uniform.
218    let main_variable = pair.0.main_variable;
219    let scalar_polynomial =
220        constant_polynomial(main_variable, Number::Integer(Integer::from(scalar)));
221    let numerator = polynomial_multiply(&pair.0, &scalar_polynomial);
222    (numerator, pair.1.clone())
223}

225fn rational_derivative(
226    numerator: &Polynomial,
227    denominator: &Polynomial,
228) -> (Polynomial, Polynomial) {
229    // Quotient rule: d/dx (n/d) = (n' * d - n * d') / d^2.
230    let numerator_prime = formal_derivative(numerator);
231    let denominator_prime = formal_derivative(denominator);
232    let cross_left = polynomial_multiply(&numerator_prime, denominator);
233    let cross_right = polynomial_multiply(numerator, &denominator_prime);
234    let result_numerator = polynomial_subtract(&cross_left, &cross_right);
235    let result_denominator = polynomial_multiply(denominator, denominator);
236    (result_numerator, result_denominator)
237}

239fn reduce_rational_pair(
240    numerator: Polynomial,
241    denominator: Polynomial,
242    operation: &OperationContext,
243) -> (Polynomial, Polynomial) {
244    // Cancel the polynomial gcd from numerator and denominator and
245    // normalise the sign of the denominator's leading coefficient. The
246    // canonicaliser used by the rational_function struct produces the
247    // same shape; we replicate the contract locally so the normal-form
248    // result can be threaded through the polynomial-only pole-analysis
249    // path without round-tripping through rational_function.
250    if polynomial_is_zero(&numerator) {
251        let main_variable = denominator.main_variable;
252        return (
253            crate::polynomial::zero_polynomial(main_variable),
254            constant_polynomial(main_variable, Number::Integer(Integer::ONE)),
255        );
256    }
257    let shared = polynomial_gcd_with_operation(&numerator, &denominator, operation);
258    let reduced_numerator = divide_exact(&numerator, &shared).unwrap_or_else(|| numerator.clone());
259    let reduced_denominator =
260        divide_exact(&denominator, &shared).unwrap_or_else(|| denominator.clone());
261    normalise_sign(reduced_numerator, reduced_denominator)
262}
263
264fn normalise_sign(numerator: Polynomial, denominator: Polynomial) -> (Polynomial, Polynomial) {
265    // Force the leading coefficient of the denominator to be positive
266    // (when it is a numeric integer or rational), matching the
267    // canonicalisation contract enforced by rational_function. When the
268    // leading coefficient is nested or otherwise non-numeric, the pair
269    // is returned unchanged.
270    use crate::polynomial::{coefficient_to_number, leading_coefficient};
271    let denominator_leading = leading_coefficient(&denominator);
272    if let Some(coefficient) = denominator_leading {
273        if let Some(numeric) = coefficient_to_number(&coefficient) {
274            if crate::number::is_negative(&numeric) {
275                return (
276                    crate::polynomial::negate(&numerator),
277                    crate::polynomial::negate(&denominator),
278                );
279            }
280        }
281    }
282    (numerator, denominator)
283}

292#[cfg(test)]
293mod tests {
294    use super::*;
295    use crate::interner::Context;
296
297    fn intern_symbol(context: &Context, name: &str) -> SymbolIdentifier {
298        context.symbol_table.intern_symbol(name)
299    }
300
301    #[test]
302    fn constant_coefficients_y_prime_prime_minus_y_gives_r_one() {
303        let mut context = Context::new();
304        let x_identifier = intern_symbol(&mut context, "x");
305        let one = builders::one(&mut context);
306        let zero = builders::zero(&mut context);
307        let negative_one = builders::integer_value(&mut context, -1);
308        let (r_numerator, r_denominator, transform_factor) =
309            reduce(&mut context, &one, &zero, &negative_one, x_identifier)
310                .expect("normal form reduction should succeed");
311        // r = (0 + 0 - 0 - 4*1*(-1)) / (4*1) = 4 / 4 = 1
312        // After cancellation: numerator = 1, denominator = 1.
313        let one_polynomial = constant_polynomial(x_identifier, Number::Integer(Integer::from(1)));
314        assert_eq!(r_numerator, one_polynomial);
315        assert_eq!(r_denominator, one_polynomial);
316        let expected_transform = builders::one(&mut context);
317        assert_eq!(transform_factor.kind, expected_transform.kind);
318    }
319
320    #[test]
321    fn constant_coefficients_y_prime_prime_plus_y_gives_r_negative_one() {
322        let mut context = Context::new();
323        let x_identifier = intern_symbol(&mut context, "x");
324        let one = builders::one(&mut context);
325        let zero = builders::zero(&mut context);
326        let (r_numerator, r_denominator, _transform_factor) =
327            reduce(&mut context, &one, &zero, &one, x_identifier)
328                .expect("normal form reduction should succeed");
329        // r = (0 + 0 - 0 - 4*1*1) / (4*1) = -4 / 4 = -1.
330        let negative_one_polynomial =
331            constant_polynomial(x_identifier, Number::Integer(Integer::from(-1)));
332        let one_polynomial = constant_polynomial(x_identifier, Number::Integer(Integer::from(1)));
333        assert_eq!(r_numerator, negative_one_polynomial);
334        assert_eq!(r_denominator, one_polynomial);
335    }
