1// Pole and infinity analysis. Given r in Q(x) as a (numerator,
2// denominator) polynomial pair, factor the denominator over Z and
3// identify each rational pole's location and order. For order-2 poles
4// compute alpha_plus / alpha_minus = (1 +/- sqrt(1 + 4 * residue)) / 2
5// when the discriminant 1 + 4 * residue is a rational square. The
6// infinity-order o(infinity) = deg(denominator) - deg(numerator) is
7// the Kovacic convention; the asymptotic behaviour at infinity is
8// summarised by infinity_behavior so case_one.rs can build the
9// infinity component of omega without re-running the analysis.
10//
11// Algebraic-pole reasoning (poles at algebraic numbers in Q-bar) is
12// deferred to the future algebraic-numbers infrastructure; the
13// analysis returns None if any pole location
14// falls outside Q.
15

64/// Performs the advanced native `compute` operation.
65pub fn compute(
66    context: &Context,
67    r_numerator: &Polynomial,
68    r_denominator: &Polynomial,
69    independent_variable: SymbolIdentifier,
70) -> Option<PoleData> {
71    // Build the pole_data summary by walking the squarefree decomposition
72    // of r's denominator. Returns None when any pole falls outside the
73    // rationals (the algebraic-pole extension is deferred); pole_data is
74    // otherwise total over the rational-pole input set.
75    let _ = (context, independent_variable);
76    if r_denominator.main_variable != r_numerator.main_variable {
77        return None;
78    }
79    let denominator_degree = degree(r_denominator).unwrap_or(0);
80    let numerator_degree = degree(r_numerator).unwrap_or(0);
81    let infinity_order = denominator_degree as i64 - numerator_degree as i64;
82    let r_is_constant =
83        denominator_degree == 0 && numerator_degree == 0 && !polynomial_is_zero(r_numerator);
84    let r_constant_value = if r_is_constant {
85        let numerator_value = coefficient_to_number(&leading_coefficient(r_numerator)?)?;
86        let denominator_value = coefficient_to_number(&leading_coefficient(r_denominator)?)?;
87        Some(number_division_to_rational(
88            &numerator_value,
89            &denominator_value,
90        )?)
91    } else {
92        None
93    };
94    let mut poles: Vec<PoleInfo> = Vec::new();
95    if denominator_degree > 0 {
96        let squarefree_factors =
97            yun_squarefree_factorization_with_operation(r_denominator, context.operation());
98        for (multiplicity, squarefree_factor) in squarefree_factors {
99            if polynomial_is_zero(&squarefree_factor) {
100                continue;
101            }
102            let linear_factors = factor_squarefree_univariate(&squarefree_factor);
103            for linear in linear_factors {
104                let (location, is_linear) = extract_rational_root(&linear)?;
105                if !is_linear {
106                    return None;
107                }
108                let residue_at_order_two = if multiplicity == 2 {
109                    Some(compute_order_two_residue(
110                        r_numerator,
111                        r_denominator,
112                        &location,
113                    )?)
114                } else {
115                    None
116                };
117                let (alpha_plus, alpha_minus) = if let Some(residue_value) = &residue_at_order_two {
118                    compute_alpha_pair_for_order_two(residue_value)
119                } else {
120                    (None, None)
121                };
122                poles.push(PoleInfo {
123                    location,
124                    order: multiplicity,
125                    residue_at_order_two,
126                    alpha_plus,
127                    alpha_minus,
128                });
129            }
130        }
131    }
132    let (infinity_leading_constant, infinity_alpha_plus, infinity_alpha_minus) =
133        compute_infinity_data(r_numerator, r_denominator, infinity_order);
134    Some(PoleData {
135        poles,
136        infinity_order,
137        infinity_leading_constant,
138        infinity_alpha_plus,
139        infinity_alpha_minus,
140        r_is_constant,
141        r_constant_value,
142    })
143}

145fn extract_rational_root(value: &Polynomial) -> Option<(Rational, bool)> {
146    // From a linear polynomial a + b*x return the rational root -a/b
147    // and a flag noting whether the polynomial is truly linear. Returns
148    // None for non-linear shapes or for shapes whose coefficients fall
149    // outside Q (the algebraic-pole extension is deferred).
150    if degree(value).unwrap_or(0) != 1 {
151        return Some((Rational::ZERO, false));
152    }
153    let mut constant_term = Number::Integer(Integer::ZERO);
154    let mut linear_term = Number::Integer(Integer::ZERO);
155    for (exponent, coefficient) in &value.terms {
156        let numeric = coefficient_to_number(coefficient)?;
157        match *exponent {
158            0 => constant_term = numeric,
159            1 => linear_term = numeric,
160            _ => return None,
161        }
162    }
163    let constant_rational = number_to_rational(&constant_term)?;
164    let linear_rational = number_to_rational(&linear_term)?;
165    if linear_rational == Rational::ZERO {
166        return None;
167    }
168    let location = -constant_rational / linear_rational;
169    Some((location, true))
170}

172fn compute_order_two_residue(
173    r_numerator: &Polynomial,
174    r_denominator: &Polynomial,
175    location: &Rational,
176) -> Option<Rational> {
177    // For a pole c of order 2, the local expansion is
178    //   r(x) = residue / (x - c)^2 + lower order singular + analytic.
179    // The residue is computed as the limit
180    //   residue = lim_{x -> c} (x - c)^2 * r(x).
181    // Implementation: divide the denominator by (x - c)^2 to remove the
182    // pole, then evaluate the resulting rational function at x = c.
183    let main_variable = r_numerator.main_variable;
184    let linear_polynomial = build_linear_polynomial(main_variable, location);
185    let pole_squared = polynomial_multiply(&linear_polynomial, &linear_polynomial);
186    let denominator_without_pole = divide_exact(r_denominator, &pole_squared)?;
187    if polynomial_is_zero(&denominator_without_pole) {
188        return None;
189    }
190    let numerator_at_c = evaluate_polynomial_at_rational(r_numerator, location)?;
191    let denominator_at_c = evaluate_polynomial_at_rational(&denominator_without_pole, location)?;
192    if denominator_at_c == Rational::ZERO {
193        return None;
194    }
195    Some(numerator_at_c / denominator_at_c)
196}

198fn build_linear_polynomial(main_variable: SymbolIdentifier, location: &Rational) -> Polynomial {
199    // Construct (x - location) as a polynomial in main_variable. The
200    // location is rational; the construction promotes through the
201    // numeric coefficient type.
202    let negative_location = -location.clone();
203    let constant_coefficient = number_from_rational(&negative_location);
204    let linear_coefficient = Number::Integer(Integer::ONE);
205    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
206    if constant_coefficient != Number::Integer(Integer::ZERO) {
207        terms.push((0, PolynomialCoefficient::Numeric(constant_coefficient)));
208    }
209    terms.push((1, PolynomialCoefficient::Numeric(linear_coefficient)));
210    Polynomial {
211        main_variable,
212        terms,
213    }
214}

216fn evaluate_polynomial_at_rational(value: &Polynomial, location: &Rational) -> Option<Rational> {
217    // Horner-style evaluation of a polynomial at a rational point. Each
218    // coefficient must be a numeric in Q; nested or non-numeric
219    // coefficients (the algebraic-
220    // extension boundary) return None.
221    let degree_value = degree(value).unwrap_or(0);
222    let mut accumulator = Rational::ZERO;
223    let mut power = Rational::ONE;
224    let mut coefficient_at_each_degree: Vec<Rational> = vec![Rational::ZERO; degree_value + 1];
225    for (exponent, coefficient) in &value.terms {
226        let numeric = coefficient_to_number(coefficient)?;
227        let rational_coefficient = number_to_rational(&numeric)?;
228        coefficient_at_each_degree[*exponent] = rational_coefficient;
229    }
230    for coefficient in coefficient_at_each_degree.iter() {
231        let term = coefficient.clone() * power.clone();
232        accumulator += term;
233        power *= location.clone();
234    }
235    Some(accumulator)
236}

238fn compute_alpha_pair_for_order_two(
239    residue_value: &Rational,
240) -> (Option<Rational>, Option<Rational>) {
241    // alpha_{+/-} = (1 +/- sqrt(1 + 4 * residue)) / 2.
242    // When the discriminant is a rational square the alphas are rational;
243    // otherwise they require an algebraic extension and we return None
244    // on both sides (Cases 2/3 cover the algebraic-extension regime).
245    let discriminant = Rational::ONE + Rational::from(4) * residue_value.clone();
246    match rational_square_root(&discriminant) {
247        Some(square_root) => {
248            let alpha_plus = (Rational::ONE + square_root.clone()) / Rational::from(2);
249            let alpha_minus = (Rational::ONE - square_root) / Rational::from(2);
250            (Some(alpha_plus), Some(alpha_minus))
251        }
252        None => (None, None),
253    }
254}

256fn compute_infinity_data(
257    r_numerator: &Polynomial,
258    r_denominator: &Polynomial,
259    infinity_order: i64,
260) -> (Option<Rational>, Option<Rational>, Option<Rational>) {
261    // Compute the asymptotic data at infinity. Three sub-cases drive
262    // case_one.rs:
263    //   o(infinity) = 0:  r approaches a non-zero constant L at infinity;
264    //                     the analysis records L so the constant-r path
265    //                     can short-circuit.
266    //   o(infinity) = 2:  r = L / x^2 + lower; alpha_{+/-} pair follows
267    //                     the same (1 +/- sqrt(1 + 4*L))/2 formula as
268    //                     finite order-2 poles.
269    //   other:            alpha pair returned as None; case_one.rs only
270    //                     closes the order-0 and order-2-at-infinity
271    //                     sub-cases currently implemented.
272    if infinity_order == 0 {
273        let numerator_leading = leading_coefficient(r_numerator);
274        let denominator_leading = leading_coefficient(r_denominator);
275        if let (Some(numerator_coefficient), Some(denominator_coefficient)) =
276            (numerator_leading, denominator_leading)
277        {
278            if let (Some(numerator_value), Some(denominator_value)) = (
279                coefficient_to_number(&numerator_coefficient),
280                coefficient_to_number(&denominator_coefficient),
281            ) {
282                if let Some(constant) =
283                    number_division_to_rational(&numerator_value, &denominator_value)
284                {
285                    return (Some(constant), None, None);
286                }
287            }
288        }
289        return (None, None, None);
290    }
291    if infinity_order == 2 {
292        let numerator_leading = leading_coefficient(r_numerator);
293        let denominator_leading = leading_coefficient(r_denominator);
294        if let (Some(numerator_coefficient), Some(denominator_coefficient)) =
295            (numerator_leading, denominator_leading)
296        {
297            if let (Some(numerator_value), Some(denominator_value)) = (
298                coefficient_to_number(&numerator_coefficient),
299                coefficient_to_number(&denominator_coefficient),
300            ) {
301                if let Some(infinity_residue) =
302                    number_division_to_rational(&numerator_value, &denominator_value)
303                {
304                    let (alpha_plus, alpha_minus) =
305                        compute_alpha_pair_for_order_two(&infinity_residue);
306                    return (Some(infinity_residue), alpha_plus, alpha_minus);
307                }
308            }
309        }
310        return (None, None, None);
311    }
312    (None, None, None)
313}

315fn rational_square_root(value: &Rational) -> Option<Rational> {
316    // Return sqrt(value) as a rational, or None when value is not a
317    // perfect rational square or is negative. The Kovacic Case 1 alpha
318    // computation is gated on this: when the discriminant 1 + 4 * residue
319    // fails to be a rational square, the corresponding pole's alpha pair
320    // requires Case 2 / Case 3 algebraic-extension reasoning.
321    if crate::exact_arithmetic::rational_is_negative(value) {
322        return None;
323    }
324    let (numerator_natural, denominator_natural) = value.numerator_and_denominator_ref();
325    let numerator_integer = Integer::from(numerator_natural.clone());
326    let denominator_integer = Integer::from(denominator_natural.clone());
327    let numerator_root = integer_square_root(&numerator_integer)?;
328    let denominator_root = integer_square_root(&denominator_integer)?;
329    Some(Rational::from(numerator_root) / Rational::from(denominator_root))
330}

332fn integer_square_root(value: &Integer) -> Option<Integer> {
333    // Return floor(sqrt(value)) when value is a perfect square; None
334    // otherwise. Used by rational_square_root; encapsulates the
335    // integer-only square-root check.
336    if value.sign_comparison() == std::cmp::Ordering::Less {
337        return None;
338    }
339    let candidate = crate::exact_arithmetic::integer_floor_square_root(value)?;
340    let squared = candidate.clone() * candidate.clone();
341    if squared == *value {
342        Some(candidate)
343    } else {
344        None
345    }
346}

348fn number_to_rational(value: &Number) -> Option<Rational> {
349    // Convert a Q-number into a Rational. Reused across pole-analysis
350    // helpers to keep all arithmetic in the exact rational layer.
351    match value {
352        Number::Integer(integer_value) => Some(Rational::from(integer_value.clone())),
353        Number::Rational(rational_value) => Some(rational_value.clone()),
354    }
355}
356
357fn number_from_rational(value: &Rational) -> Number {
358    // Construct a Q-number from a Rational. The reverse of
359    // number_to_rational; promotes integer-valued rationals to the
360    // integer constructor so the downstream comparisons against
361    // Integer::ZERO stay structural.
362    if let Ok(integer_value) = Integer::try_from(value.clone()) {
363        Number::Integer(integer_value)
364    } else {
365        Number::Rational(value.clone())
366    }
367}

369fn number_division_to_rational(
370    numerator_value: &Number,
371    denominator_value: &Number,
372) -> Option<Rational> {
373    // Compute numerator / denominator in Q, returning None on division
374    // by zero. Reused by the constant-r and infinity-leading-coefficient
375    // computations.
376    let numerator_rational = number_to_rational(numerator_value)?;
377    let denominator_rational = number_to_rational(denominator_value)?;
378    if denominator_rational == Rational::ZERO {
379        return None;
380    }
381    Some(numerator_rational / denominator_rational)
