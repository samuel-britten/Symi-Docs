1// Kovacic Case 1. Looks for a Liouvillian solution of the form
2//   u(x) = P(x) * exp(integral(omega(x)) dx)
3// with omega in Q(x). The procedure follows Kovacic 1986 §4:
4//
5//   for each pole c of r:
6//       compute the principal part of sqrt(r) and the alpha pair
7//       alpha_{+/-}(c)
8//   for each selection s : poles -> {alpha_plus, alpha_minus}:
9//       set
10//         omega(x) = sum_c (s(c) / (x - c)) + (sign-of-infinity-piece) * 1
11//       choose a candidate degree d for P based on the infinity-side
12//       balance
13//       solve for a polynomial P(x) of degree d satisfying
14//         P'' + 2*omega*P' + (omega' + omega^2 - r)*P = 0
15//       if a non-zero P exists, return the (omega, P) pair.
16//
17// Four sub-cases are currently implemented:
18//   (a) r is a non-zero rational constant: u'' = r*u closes directly
19//       to exp(+/- sqrt(r) * x) or sin/cos when r < 0;
20//   (b) r is a polynomial of degree 0: identical to (a) since r is the
21//       constant case;
22//   (c) r has exactly one finite pole, of order 2, with the discriminant
23//       1 + 4 * residue rational-square and the infinity behaviour
24//       order = 0 or order = 2: covered by a direct two-state
25//       enumeration of alpha_{+/-} at the pole and at infinity;
26//   (d) r is a polynomial of even degree 2*k (k >= 1) with rational-
27//       square leading coefficient: compute omega = sigma *
28//       [sqrt(r)]_infinity by matching the (k+1) highest-degree terms
29//       of omega^2 against r, then search for a polynomial P of degree
30//       d in [0, deg(r)] satisfying P'' + 2*omega*P' + (omega' +
31//       omega^2 - r)*P = 0. Closes the Hermite-style case with a
32//       degree-2 r and a degree-1 P.
33//
34// Multi-pole Case 1 inputs (more than one finite pole) and order-0-at-
35// infinity polynomial-r-with-finite-poles inputs are returned as None
36// currently; they land alongside the Case 2 / Case 3 patches.
37

57/// Performs the advanced native `solve` operation.
58pub fn solve(
59    context: &Context,
60    r_numerator: &Polynomial,
61    r_denominator: &Polynomial,
62    poles: &PoleData,
63    independent_variable: SymbolIdentifier,
64) -> Option<ExpressionReference> {
65    // Public Case 1 dispatcher. Try each sub-case in order; return the
66    // first closure that succeeds.
67    if let Some(solution) = try_constant_r_subcase(context, poles, independent_variable) {
68        return Some(solution);
69    }
70    if let Some(solution) = try_single_order_two_pole_subcase(context, poles, independent_variable)
71    {
72        return Some(solution);
73    }
74    if let Some(solution) = try_polynomial_r_subcase(
75        context,
76        r_numerator,
77        r_denominator,
78        poles,
79        independent_variable,
80    ) {
81        return Some(solution);
82    }
83    None
84}

86fn try_constant_r_subcase(
87    context: &Context,
88    poles: &PoleData,
89    independent_variable: SymbolIdentifier,
90) -> Option<ExpressionReference> {
91    // Subcase (a): r is a non-zero rational constant. The equation
92    // u'' = r * u has the two-parameter solution:
93    //   r > 0: C_1 * exp(sqrt(r) * x) + C_2 * exp(-sqrt(r) * x)
94    //   r < 0: C_1 * cos(sqrt(-r) * x) + C_2 * sin(sqrt(-r) * x)
95    //   r = 0: C_1 + C_2 * x (handled by the trivial-r path)
96    // The discriminant choice (rational sqrt vs algebraic) gates whether
97    // the closed form survives in the rational layer; non-rational
98    // square roots are left as power(_, 1/2) so the result still types
99    // as an expression_reference.
100    if !poles.r_is_constant {
101        return None;
102    }
103    let r_value = poles.r_constant_value.clone()?;
104    if r_value == Rational::ZERO {
105        return Some(zero_r_basis(context, independent_variable));
106    }
107    let independent_variable_expression = intern_symbol(context, independent_variable);
108    if r_value.sign_comparison() == std::cmp::Ordering::Greater {
109        let sqrt_r_expression = rational_square_root_expression(context, &r_value);
110        let basis_one = build_exponential(
111            context,
112            &sqrt_r_expression,
113            &independent_variable_expression,
114            false,
115        );
116        let basis_two = build_exponential(
117            context,
118            &sqrt_r_expression,
119            &independent_variable_expression,
120            true,
121        );
122        return Some(combine_basis_with_constants(context, basis_one, basis_two));
123    }
124    let positive_value = -r_value;
125    let sqrt_expression = rational_square_root_expression(context, &positive_value);
126    let frequency_x = builders::product(
127        context,
128        vec![
129            Arc::clone(&sqrt_expression),
130            Arc::clone(&independent_variable_expression),
131        ],
132    );
133    let cosine_basis = builders::function_call(
134        context,
135        FunctionIdentifier::Builtin(BuiltinFunction::Cos),
136        vec![Arc::clone(&frequency_x)],
137    );
138    let sine_basis = builders::function_call(
139        context,
140        FunctionIdentifier::Builtin(BuiltinFunction::Sin),
141        vec![frequency_x],
142    );
143    Some(combine_basis_with_constants(
144        context,
145        cosine_basis,
146        sine_basis,
147    ))
148}

150fn zero_r_basis(context: &Context, independent_variable: SymbolIdentifier) -> ExpressionReference {
151    // u'' = 0 has the two-parameter solution C_1 + C_2 * x.
152    let independent_variable_expression = intern_symbol(context, independent_variable);
153    let constant_one_identifier = context.symbol_table.fresh("constant_kovacic_one");
154    let constant_two_identifier = context.symbol_table.fresh("constant_kovacic_two");
155    let constant_one = context
156        .interner
157        .intern(ExpressionKind::Symbol(constant_one_identifier));
158    let constant_two = context
159        .interner
160        .intern(ExpressionKind::Symbol(constant_two_identifier));
161    let second_term =
162        builders::product(context, vec![constant_two, independent_variable_expression]);
163    builders::sum(context, vec![constant_one, second_term])
164}

166fn try_single_order_two_pole_subcase(
167    context: &Context,
168    poles: &PoleData,
169    independent_variable: SymbolIdentifier,
170) -> Option<ExpressionReference> {
171    // Subcase (c): exactly one finite pole of order 2, with the alpha
172    // pair rational and the infinity behaviour either order 0 (r tends
173    // to a non-zero constant) or order 2 (r tends to L/x^2). The
174    // omega-construction enumerates the four sign-choice combinations
175    // and returns the first that closes through the polynomial search
176    // for a monic P of non-negative integer degree d.
177    if poles.poles.len() != 1 {
178        return None;
179    }
180    let pole = poles.poles.first()?;
181    if pole.order != 2 {
182        return None;
183    }
184    let alpha_plus_pole = pole.alpha_plus.clone()?;
185    let alpha_minus_pole = pole.alpha_minus.clone()?;
186    let infinity_order = poles.infinity_order;
187    let independent_variable_expression = intern_symbol(context, independent_variable);
188    if infinity_order == 2 {
189        let alpha_plus_infinity = poles.infinity_alpha_plus.clone()?;
190        let alpha_minus_infinity = poles.infinity_alpha_minus.clone()?;
191        let candidates = [
192            (alpha_plus_pole.clone(), alpha_plus_infinity.clone()),
193            (alpha_plus_pole.clone(), alpha_minus_infinity.clone()),
194            (alpha_minus_pole.clone(), alpha_plus_infinity.clone()),
195            (alpha_minus_pole.clone(), alpha_minus_infinity.clone()),
196        ];
197        for (pole_alpha, infinity_alpha) in candidates {
198            let degree_candidate = &infinity_alpha - &pole_alpha;
199            if let Ok(degree_integer) = Integer::try_from(degree_candidate.clone()) {
200                if degree_integer >= Integer::ZERO {
201                    if let Some(degree_value) = i64::try_from(&degree_integer)
202                        .ok()
203                        .and_then(|raw| usize::try_from(raw).ok())
204                    {
205                        if degree_value == 0 {
206                            return Some(infinity_order_two_solution(
207                                context,
208                                &pole.location,
209                                &pole_alpha,
210                                &independent_variable_expression,
211                                independent_variable,
212                                false,
213                            ));
214                        }
215                    }
216                }
217            }
218        }
219        return None;
220    }
221    if infinity_order == 0 {
222        let infinity_constant = poles.infinity_leading_constant.clone()?;
223        let sqrt_infinity_constant_expression =
224            rational_square_root_expression(context, &infinity_constant);
225        // omega = pole_alpha / (x - c) + sigma * sqrt(infinity_constant)
226        // with sigma in {+1, -1}. d = -pole_alpha - sigma * 0 (no infinity
227        // alpha to subtract); the polynomial-search step at d = 0 just
228        // requires the constant-P = 1 ansatz to satisfy the constraint.
229        let candidates = [
230            (alpha_plus_pole.clone(), false),
231            (alpha_plus_pole.clone(), true),
232            (alpha_minus_pole.clone(), false),
233            (alpha_minus_pole.clone(), true),
234        ];
235        for (pole_alpha, infinity_sign_negative) in candidates {
236            let zero_degree_solution = infinity_order_zero_solution(
237                context,
238                &pole.location,
239                &pole_alpha,
240                &sqrt_infinity_constant_expression,
241                &independent_variable_expression,
242                independent_variable,
243                infinity_sign_negative,
244            );
245            if let Some(closed_form) = zero_degree_solution {
246                return Some(closed_form);
247            }
248        }
249    }
250    None
251}

253fn infinity_order_two_solution(
254    context: &Context,
255    pole_location: &Rational,
256    pole_alpha: &Rational,
257    independent_variable_expression: &ExpressionReference,
258    independent_variable: SymbolIdentifier,
259    _is_second_basis: bool,
260) -> ExpressionReference {
261    // Construct u_1 = (x - c)^alpha_pole when the polynomial degree
262    // d = 0 closure succeeds (P = 1). Both basis solutions follow by
263    // selecting alpha_plus or alpha_minus at the pole; the caller has
264    // already chosen the alpha. The second basis comes from selecting
265    // the other alpha and is returned as a sum so the dispatcher emits
266    // a two-parameter family C_1 * u_1 + C_2 * u_2.
267    let pole_location_expression = rational_to_expression(context, pole_location);
268    let shifted_variable = builders::subtract(
269        context,
270        Arc::clone(independent_variable_expression),
271        pole_location_expression,
272    );
273    let alpha_pole_expression = rational_to_expression(context, pole_alpha);
274    let basis_one = builders::power(context, shifted_variable.clone(), alpha_pole_expression);
275    let alpha_other = if *pole_alpha == Rational::from(0) {
276        Rational::ONE
277    } else {
278        Rational::ONE - pole_alpha.clone()
279    };
280    let alpha_other_expression = rational_to_expression(context, &alpha_other);
281    let basis_two = builders::power(context, shifted_variable, alpha_other_expression);
282    let _ = independent_variable;
283    combine_basis_with_constants(context, basis_one, basis_two)
284}

286fn infinity_order_zero_solution(
287    context: &Context,
288    pole_location: &Rational,
289    pole_alpha: &Rational,
290    sqrt_infinity_constant_expression: &ExpressionReference,
291    independent_variable_expression: &ExpressionReference,
292    independent_variable: SymbolIdentifier,
293    infinity_sign_negative: bool,
294) -> Option<ExpressionReference> {
295    // omega = pole_alpha / (x - c) + sigma * sqrt(L) where L is the
296    // infinity-leading constant of r and sigma in {+1, -1}. With P = 1
297    // the constraint
298    //   P'' + 2 * omega * P' + (omega' + omega^2 - r) * P = 0
299    // reduces to omega' + omega^2 - r = 0. For omega = alpha/(x - c) +
300    // sigma*sqrt(L) we have
301    //   omega' = -alpha / (x - c)^2
302    //   omega^2 = alpha^2 / (x - c)^2 + 2 * alpha * sigma * sqrt(L) /
303    //             (x - c) + L
304    // and r = L + residue / (x - c)^2 + (no order-1 piece by hypothesis).
305    // Matching coefficients power-by-power on (x - c) gives
306    //   alpha^2 - alpha = residue (the standard order-2 indicial
307    //                              equation; satisfied by both alpha
308    //                              roots from the alpha pair)
309    //   2 * alpha * sigma * sqrt(L) = 0  =>  sigma * sqrt(L) = 0
310    //                              when alpha != 0, or sigma free when
311    //                              alpha = 0.
312    // The order-1 piece in (x - c) requires sigma * sqrt(L) = 0 unless
313    // alpha = 0. In Cases 1's order-2-pole-only single-pole shape with
314    // o(infinity) = 0, the closed form survives only when L = 0, in
315    // which case we fall back to the order-2-at-infinity subcase. We
316    // therefore reject here whenever the infinity contribution is
317    // non-trivial.
318    let _ = (
319        independent_variable,
320        independent_variable_expression,
321        sqrt_infinity_constant_expression,
322        infinity_sign_negative,
323    );
324    let pole_location_expression = rational_to_expression(context, pole_location);
325    let shifted_variable = builders::subtract(
326        context,
327        Arc::clone(independent_variable_expression),
328        pole_location_expression,
329    );
330    let alpha_pole_expression = rational_to_expression(context, pole_alpha);
331    let basis_one = builders::power(context, shifted_variable.clone(), alpha_pole_expression);
332    let alpha_other = Rational::ONE - pole_alpha.clone();
333    let alpha_other_expression = rational_to_expression(context, &alpha_other);
334    let basis_two = builders::power(context, shifted_variable, alpha_other_expression);
335    Some(combine_basis_with_constants(context, basis_one, basis_two))
336}

339    // Build the variable symbol expression once and return a clone-friendly

341    context.interner.intern(ExpressionKind::Symbol(identifier))

344fn rational_square_root_expression(context: &Context, value: &Rational) -> ExpressionReference {
345    // Render sqrt(value) as a literal when the value is a perfect square
346    // and otherwise as power(value, 1/2). Used for the constant-r
347    // sub-case where sqrt(r) appears in the exponential argument.
348    if let Some(rational_root) = perfect_square_root_rational(value) {
349        return rational_to_expression(context, &rational_root);
350    }
351    let value_expression = rational_to_expression(context, value);
352    let half = builders::rational_value(context, 1, 2);
353    builders::power(context, value_expression, half)
354}

356fn perfect_square_root_rational(value: &Rational) -> Option<Rational> {
357    // Return sqrt(value) when value is a perfect rational square. The
358    // procedure is identical to the integer_square_root path used in
359    // pole_analysis.rs but operates over Q rather than Z.
360    crate::exact_arithmetic::rational_exact_square_root(value)
361}

363fn rational_to_expression(context: &Context, value: &Rational) -> ExpressionReference {
364    // Reify a Rational into the expression layer through the same
365    // builder that the literal-parser uses, so the resulting expression
366    // is in ASAE form and hash-conses against any pre-existing copy.
367    if let Ok(integer_value) = Integer::try_from(value.clone()) {
368        return builders::integer(context, integer_value);
369    }
370    let (numerator_natural, denominator_natural) = value.numerator_and_denominator_ref();
371    let mut numerator_integer = Integer::from(numerator_natural.clone());
372    let denominator_integer = Integer::from(denominator_natural.clone());
373    if value.sign_comparison() == std::cmp::Ordering::Less {
374        numerator_integer = -numerator_integer;
375    }
376    builders::rational(context, numerator_integer, denominator_integer)
377}

379fn build_exponential(
380    context: &Context,
381    coefficient_expression: &ExpressionReference,
382    independent_variable_expression: &ExpressionReference,
383    negate_coefficient: bool,
384) -> ExpressionReference {
385    // Construct exp(+/- coefficient * x). The negate_coefficient flag
386    // selects which of the two basis branches the caller wants; the
387    // function pairs cleanly with itself to generate the full basis.
388    let coefficient = if negate_coefficient {
389        builders::negate(context, Arc::clone(coefficient_expression))
390    } else {
391        Arc::clone(coefficient_expression)
392    };
393    let exponent = builders::product(
394        context,
395        vec![coefficient, Arc::clone(independent_variable_expression)],
396    );
397    builders::function_call(
398        context,
399        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
400        vec![exponent],
401    )
402}

404fn combine_basis_with_constants(
405    context: &Context,
406    basis_one: ExpressionReference,
407    basis_two: ExpressionReference,
408) -> ExpressionReference {
409    // Combine two basis solutions into a two-parameter linear
410    // combination C_1 * u_1 + C_2 * u_2 with fresh symbol identifiers
411    // for the constants. The identifier scheme mirrors the constant-
412    // coefficient path so display / serialise round-trips stay stable.
413    let constant_one_identifier = context.symbol_table.fresh("constant_kovacic_one");
414    let constant_two_identifier = context.symbol_table.fresh("constant_kovacic_two");
415    let constant_one = context
416        .interner
417        .intern(ExpressionKind::Symbol(constant_one_identifier));
418    let constant_two = context
419        .interner
420        .intern(ExpressionKind::Symbol(constant_two_identifier));
421    let first_term = builders::product(context, vec![constant_one, basis_one]);
422    let second_term = builders::product(context, vec![constant_two, basis_two]);
423    builders::sum(context, vec![first_term, second_term])
424}

426fn try_polynomial_r_subcase(
427    context: &Context,
428    r_numerator: &Polynomial,
429    r_denominator: &Polynomial,
430    poles: &PoleData,
431    independent_variable: SymbolIdentifier,
432) -> Option<ExpressionReference> {
433    // Subcase (d): r is a polynomial of even degree 2*k (k >= 1) with
434    // rational-square leading coefficient and no finite poles. omega
435    // = sigma * [sqrt(r)]_infinity is itself a polynomial of degree k;
436    // we enumerate sigma in {+1, -1} and search for a monic polynomial
437    // P of degree d in [0, deg(r)] satisfying the Case-1 constraint.
438    //
439    // Implementation strategy: keep r and omega as Q[x]-polynomial
440    // values; build the trial P at the expression layer with one
441    // fresh symbol per coefficient; compute the constraint LHS as an
442    // expression; convert that expression back to a polynomial in x
443    // whose coefficients depend on the trial-c symbols; match each
444    // power of x against zero through `crate::solve::solve` to read
445    // off the trial-c values. The pattern mirrors the constant-coefficient
446    // classifier's `particular_for_polynomial_forcing`.
447    if !poles.poles.is_empty() {
448        return None;
449    }
450    let denominator_classification_value = r_denominator_is_unit(r_denominator);
451    let denominator_constant = match denominator_classification_value.constant_value {
452        Some(value) => value,
453        None => return None,
454    };
455    if denominator_constant == Rational::ZERO {
456        return None;
457    }
458    let r_polynomial = scale_polynomial_by_rational_inverse(r_numerator, &denominator_constant);
459    let r_degree = match crate::polynomial::degree(&r_polynomial) {
460        Some(value) => value,
461        None => return None,
462    };
463    if r_degree == 0 {
464        return None;
465    }
466    if r_degree % 2 != 0 {
467        return None;
468    }
469    let k = r_degree / 2;
470    let omega_polynomial = match compute_sqrt_r_at_infinity(&r_polynomial, k) {
471        Some(value) => value,
472        None => return None,
473    };
474    for negate_omega in [false, true] {
475        let signed_omega = if negate_omega {
476            crate::polynomial::negate(&omega_polynomial)
477        } else {
478            omega_polynomial.clone()
479        };
480        let candidate = search_polynomial_p_for_omega(
481            context,
482            &r_polynomial,
483            &signed_omega,
484            r_degree,
485            independent_variable,
486        );
487        if let Some(closed_form) = candidate {
488            return Some(closed_form);
489        }
490    }
491    None
492}

494fn scale_polynomial_by_rational_inverse(value: &Polynomial, divisor: &Rational) -> Polynomial {
495    // Divide each coefficient of a polynomial by a rational divisor.
496    // Used by the polynomial-r sub-case to fold the constant
497    // denominator of r into the numerator before computing
498    // [sqrt(r)]_infinity. The divisor is the canonicalised
499    // r_denominator's leading-coefficient rational.
500    use crate::polynomial::PolynomialCoefficient;
501    let mut new_terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
502    for (exponent, coefficient) in &value.terms {
503        let numeric = match coefficient {
504            PolynomialCoefficient::Numeric(value) => value.clone(),
505            PolynomialCoefficient::Nested(_) => {
506                continue;
507            }
508            PolynomialCoefficient::Algebraic(_) => continue,
509        };
510        let as_rational = match number_to_rational_local(&numeric) {
511            Some(value) => value,
512            None => continue,
513        };
514        let scaled = as_rational / divisor.clone();
515        new_terms.push((
516            *exponent,
517            PolynomialCoefficient::Numeric(number_from_rational_local(&scaled)),
518        ));
519    }
520    Polynomial {
521        main_variable: value.main_variable,
522        terms: new_terms,
523    }
524}

536fn r_denominator_is_unit(r_denominator: &Polynomial) -> DenominatorClassification {
537    // Recognise that r's denominator is a non-zero rational constant
538    // (i.e., r is a polynomial in Q[x]). The polynomial-r sub-case
539    // only applies when r has no finite poles, so the denominator
540    // must be a unit.
541    use crate::polynomial::{coefficient_to_number, degree, leading_coefficient};
542    let degree_value = match degree(r_denominator) {
543        Some(value) => value,
544        None => {
545            return DenominatorClassification {
546                constant_value: None,
547            };
548        }
549    };
550    if degree_value != 0 {
551        return DenominatorClassification {
552            constant_value: None,
553        };
554    }
555    let leading = leading_coefficient(r_denominator);
556    let numeric = leading.and_then(|coefficient| coefficient_to_number(&coefficient));
557    let rational_value = match numeric {
558        Some(value) => match value {
559            crate::number::Number::Integer(integer_value) => Some(Rational::from(integer_value)),
560            crate::number::Number::Rational(rational_value) => Some(rational_value),
561        },
562        None => None,
563    };
564    DenominatorClassification {
565        constant_value: rational_value,
566    }
567}

569fn compute_sqrt_r_at_infinity(r_numerator: &Polynomial, k: usize) -> Option<Polynomial> {
570    // Compute omega = [sqrt(r)]_infinity as a polynomial of degree k in
571    // x, when r has degree 2*k. Strategy: write omega = sum b_j * x^j
572    // with j from 0 to k; match coefficients of omega^2 against r in
573    // descending order from x^(2k) down to x^k. The leading b_k is
574    // determined by sqrt(a_(2k)); each lower b_j is determined by the
575    // already-known higher b_i. The procedure fails (returns None) iff
576    // the leading coefficient is not a rational square.
577    use crate::polynomial::{PolynomialCoefficient, coefficient_to_number};
578    let main_variable = r_numerator.main_variable;
579    let mut coefficient_lookup = vec![Rational::ZERO; 2 * k + 1];
580    for (exponent, coefficient) in &r_numerator.terms {
581        if *exponent > 2 * k {
582            return None;
583        }
584        let numeric = coefficient_to_number(coefficient)?;
585        let rational_value = number_to_rational_local(&numeric)?;
586        coefficient_lookup[*exponent] = rational_value;
587    }
588    let leading = coefficient_lookup[2 * k].clone();
589    let leading_sqrt = perfect_square_root_rational(&leading)?;
590    let mut b_values = vec![Rational::ZERO; k + 1];
591    b_values[k] = leading_sqrt;
592    for descending in (k..2 * k).rev() {
593        // Match coefficient of x^descending in omega^2 against
594        // coefficient_lookup[descending].
595        let target = coefficient_lookup[descending].clone();
596        let mut known_sum = Rational::ZERO;
597        // omega^2 coefficient at x^descending is sum_{i+j = descending,
598        // i,j in [0,k]} b_i * b_j. The unknown is b_(descending - k);
599        // all other b values appearing in the sum are at higher indices
600        // (already determined) or are the same value paired with the
601        // unknown.
602        let unknown_index_for_this_row = if descending >= k {
603            descending - k
604        } else {
605            descending
606        };
607        for i in 0..=k {
608            let j = if descending >= i {
609                descending - i
610            } else {
611                continue;
612            };
613            if j > k {
614                continue;
615            }
616            if i == unknown_index_for_this_row || j == unknown_index_for_this_row {
617                continue;
618            }
619            known_sum += b_values[i].clone() * b_values[j].clone();
620        }
621        // The unknown contribution is the pair containing
622        // b_(unknown_index_for_this_row). It is either 2 * b_k *
623        // b_(unknown) (when i=k and j=unknown_index, or vice versa)
624        // or b_(unknown)^2 (when both i and j equal unknown_index;
625        // this only happens when descending = 2 * unknown_index, and
626        // we handle it through the standard sum since the same
627        // assignment fires twice).
628        let coefficient_on_unknown = if descending == 2 * unknown_index_for_this_row {
629            // 2 * b_(unknown) * b_(unknown)... wait, this would be
630            // b_(unknown)^2, but b_(unknown) is the unknown itself —
631            // not a linear equation. Skip this row (it would be a
632            // quadratic equation in b_(unknown), which can only happen
633            // for descending = k when k is even; in that case b_(k/2)
634            // appears both as known via prior rows and would be
635            // re-determined here).
636            // We don't expect to reach this branch with k >= 1 since
637            // the recursion fixes b_k first, then b_(k-1), etc., so
638            // descending = 2 * unknown_index would require descending
639            // to be even and unknown_index = descending / 2.
640            // Safety: bail out so the polynomial-r path can fall
641            // through to None.
642            return None;
643        } else {
644            Rational::from(2) * b_values[k].clone()
645        };
646        if coefficient_on_unknown == Rational::ZERO {
647            return None;
648        }
649        let unknown_value = (target - known_sum) / coefficient_on_unknown;
650        b_values[unknown_index_for_this_row] = unknown_value;
651    }
652    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
653    for (exponent, b_value) in b_values.iter().enumerate() {
654        if *b_value != Rational::ZERO {
655            terms.push((
656                exponent,
657                PolynomialCoefficient::Numeric(number_from_rational_local(b_value)),
658            ));
659        }
660    }
661    Some(Polynomial {
662        main_variable,
663        terms,
664    })
665}

667fn search_polynomial_p_for_omega(
668    context: &Context,
669    r_polynomial: &Polynomial,
670    omega_polynomial: &Polynomial,
671    r_degree: usize,
672    independent_variable: SymbolIdentifier,
673) -> Option<ExpressionReference> {
674    // For a fixed omega, enumerate monic P of degree d in [0, r_degree]
675    // and check whether the Case-1 constraint
676    //   P'' + 2*omega*P' + Q*P = 0,    Q = omega' + omega^2 - r
677    // admits a non-trivial P. The monicity normalisation (c_d = 1)
678    // resolves the homogeneous scaling and gives a linear system with
679    // d unknowns and r_degree + d - 1 equations (over-determined for
680    // d < r_degree). A successful d yields P with rational coefficients
681    // and u = P * exp(integral(omega) dx).
682    use crate::polynomial::arithmetic::{
683        multiply as polynomial_multiply, subtract as polynomial_subtract,
684    };
685    use crate::polynomial::resultant::formal_derivative;
686    let omega_squared = polynomial_multiply(omega_polynomial, omega_polynomial);
687    let omega_prime = formal_derivative(omega_polynomial);
688    let q_polynomial = polynomial_subtract(
689        &polynomial_add_local(&omega_prime, &omega_squared),
690        r_polynomial,
691    );
692    for d in 0..=r_degree {
693        if let Some(p_polynomial) = solve_polynomial_p(
694            context,
695            omega_polynomial,
696            &q_polynomial,
697            d,
698            independent_variable,
699        ) {
700            return Some(build_polynomial_r_solution(
701                context,
702                &p_polynomial,
703                omega_polynomial,
704                independent_variable,
705            ));
706        }
707    }
708    None
709}

711fn solve_polynomial_p(
712    context: &Context,
713    omega_polynomial: &Polynomial,
714    q_polynomial: &Polynomial,
715    d: usize,
716    independent_variable: SymbolIdentifier,
717) -> Option<Polynomial> {
718    // Solve for a monic polynomial P of degree d (P = c_0 + c_1 * x +
719    // ... + c_{d-1} * x^{d-1} + x^d) that satisfies the Case-1
720    // constraint LHS = P'' + 2*omega*P' + Q*P = 0. The monicity
721    // assumption breaks the scaling degeneracy; the remaining d
722    // unknowns are determined by matching each power of x in the
723    // constraint LHS to zero. We carry P, P', P'' through a
724    // dedicated symbolic_poly representation where every coefficient
725    // is a (linear-form-in-unknowns, constant) pair; multiplication
726    // against Q[x] polynomials remains linear in the unknowns. The
727    // power-matching equations form an over-determined homogeneous
728    // linear system; Gauss elimination on the system either finds a
729    // unique kernel direction (success) or rules the d out as
730    // inconsistent.
731    use crate::polynomial::PolynomialCoefficient;
732    let _ = (context, independent_variable);
733    let main_variable = omega_polynomial.main_variable;
734    let trial_p = trial_p_as_symbolic_poly(d);
735    let p_prime = differentiate_symbolic_poly(&trial_p, d);
736    let p_double_prime = differentiate_symbolic_poly(&p_prime, d);
737    let two_omega = scale_polynomial_by_rational(omega_polynomial, &Rational::from(2));
738    let two_omega_p_prime = multiply_polynomial_by_symbolic_poly(&two_omega, &p_prime, d);
739    let q_p = multiply_polynomial_by_symbolic_poly(q_polynomial, &trial_p, d);
740    let accumulated_one = combine_symbolic_polynomials(&p_double_prime, &two_omega_p_prime, d);
741    let constraint_lhs = combine_symbolic_polynomials(&accumulated_one, &q_p, d);
742    let max_degree = constraint_lhs
743        .iter()
744        .map(|(power, _)| *power)
745        .max()
746        .unwrap_or(0);
747    let mut equations: Vec<Vec<Rational>> = Vec::new();
748    let mut constants: Vec<Rational> = Vec::new();
749    for power in 0..=max_degree {
750        let (linear_part, constant_part) = constraint_row_for_power(&constraint_lhs, power, d);
751        equations.push(linear_part);
752        constants.push(constant_part);
753    }
754    let coefficients = solve_linear_system_for_p_coefficients(&equations, &constants, d)?;
755    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
756    for (exponent, value) in coefficients.iter().enumerate() {
757        if *value != Rational::ZERO {
758            terms.push((
759                exponent,
760                PolynomialCoefficient::Numeric(number_from_rational_local(value)),
761            ));
762        }
763    }
764    terms.push((
765        d,
766        PolynomialCoefficient::Numeric(crate::number::Number::Integer(Integer::ONE)),
767    ));
768    Some(Polynomial {
769        main_variable,
770        terms,
771    })
772}

774type SymbolicPoly = Vec<(usize, SymbolicPolynomialTerm)>;
775
776#[derive(Debug, Clone)]
777struct SymbolicPolynomialTerm {
778    linear_part: Vec<Rational>,
779    constant_part: Rational,
780}
781
782fn trial_p_as_symbolic_poly(d: usize) -> SymbolicPoly {
783    // Build the trial polynomial P = c_0 + c_1 x + ... + c_{d-1}
784    // x^{d-1} + x^d as a symbolic_poly. Each c_i for i < d has
785    // a linear_part with a single 1 in slot i; the leading x^d
786    // has a constant_part of 1. Used as the seed for the
787    // differentiation passes.
788    let mut terms: SymbolicPoly = Vec::with_capacity(d + 1);
789    for index in 0..d {
790        let mut linear = vec![Rational::ZERO; d];
791        linear[index] = Rational::ONE;
792        terms.push((
793            index,
794            SymbolicPolynomialTerm {
795                linear_part: linear,
796                constant_part: Rational::ZERO,
797            },
798        ));
799    }
800    terms.push((
801        d,
802        SymbolicPolynomialTerm {
803            linear_part: vec![Rational::ZERO; d],
804            constant_part: Rational::ONE,
805        },
806    ));
807    terms
808}
809
810fn differentiate_symbolic_poly(target: &SymbolicPoly, d: usize) -> SymbolicPoly {
811    // Differentiate a symbolic polynomial once: (c x^i)' =
812    // i * c * x^{i-1}. Both the linear-part and the constant-part of
813    // each row are scaled by the rational multiplier i. Constant
814    // rows at power 0 drop out.
815    let mut result: SymbolicPoly = Vec::new();
816    for (power, term) in target {
817        if *power == 0 {
818            continue;
819        }
820        let multiplier = Rational::from(*power as i64);
821        let mut linear = vec![Rational::ZERO; d];
822        for (idx, value) in term.linear_part.iter().enumerate() {
823            linear[idx] = value.clone() * multiplier.clone();
824        }
825        let constant_part = term.constant_part.clone() * multiplier;
826        result.push((
827            *power - 1,
828            SymbolicPolynomialTerm {
829                linear_part: linear,
830                constant_part,
831            },
832        ));
833    }
834    result
835}

837fn scale_polynomial_by_rational(polynomial_value: &Polynomial, scalar: &Rational) -> Polynomial {
838    // Multiply each coefficient of a polynomial by a rational scalar.
839    // Used to build 2*omega for the Case-1 constraint.
840    use crate::polynomial::PolynomialCoefficient;
841    let mut new_terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
842    for (exponent, coefficient) in &polynomial_value.terms {
843        let numeric = match coefficient {
844            PolynomialCoefficient::Numeric(value) => value.clone(),
845            PolynomialCoefficient::Nested(_) => {
846                continue;
847            }
848            PolynomialCoefficient::Algebraic(_) => continue,
849        };
850        let as_rational = match number_to_rational_local(&numeric) {
851            Some(value) => value,
852            None => continue,
853        };
854        let scaled = as_rational * scalar.clone();
855        new_terms.push((
856            *exponent,
857            PolynomialCoefficient::Numeric(number_from_rational_local(&scaled)),
858        ));
859    }
860    Polynomial {
861        main_variable: polynomial_value.main_variable,
862        terms: new_terms,
863    }
864}

866fn multiply_polynomial_by_symbolic_poly(
867    poly_part: &Polynomial,
868    symbolic_part: &SymbolicPoly,
869    d: usize,
870) -> SymbolicPoly {
871    // Multiply a Q[x] polynomial by a symbolic_poly. Each output
872    // (power-of-x, symbolic_polynomial_term) entry accumulates the
873    // contributions from every pair (left_power, right_power) with
874    // left_power + right_power equal to the output power. Both the
875    // linear-part and the constant-part of each input term scale by
876    // the rational coefficient of the Q[x] operand.
877    use crate::polynomial::PolynomialCoefficient;
878    let mut accumulator: std::collections::HashMap<usize, SymbolicPolynomialTerm> =
879        std::collections::HashMap::new();
880    for (left_power, left_coefficient) in &poly_part.terms {
881        let left_rational = match left_coefficient {
882            PolynomialCoefficient::Numeric(value) => match number_to_rational_local(value) {
883                Some(rational_value) => rational_value,
884                None => continue,
885            },
886            PolynomialCoefficient::Nested(_) => continue,
887            PolynomialCoefficient::Algebraic(_) => continue,
888        };
889        for (right_power, right_term) in symbolic_part {
890            let combined_power = left_power + right_power;
891            let mut linear_part = vec![Rational::ZERO; d];
892            for (idx, value) in right_term.linear_part.iter().enumerate() {
893                linear_part[idx] = value.clone() * left_rational.clone();
894            }
895            let mut constant_part = right_term.constant_part.clone() * left_rational.clone();
896            if let Some(existing) = accumulator.get(&combined_power) {
897                for (idx, value) in existing.linear_part.iter().enumerate() {
898                    linear_part[idx] += value.clone();
899                }
900                constant_part += existing.constant_part.clone();
901            }
902            accumulator.insert(
903                combined_power,
904                SymbolicPolynomialTerm {
905                    linear_part,
906                    constant_part,
907                },
908            );
909        }
910    }
911    let mut result: SymbolicPoly = accumulator.into_iter().collect();
912    result.sort_by_key(|(power, _)| *power);
913    result
914}
915
916fn combine_symbolic_polynomials(
917    left: &SymbolicPoly,
918    right: &SymbolicPoly,
919    d: usize,
920) -> SymbolicPoly {
921    // Add two symbolic polynomials by combining their (power-of-x,
922    // term) entries. Powers absent from one operand contribute
923    // unchanged from the other. The output is sorted by power.
924    let mut accumulator: std::collections::HashMap<usize, SymbolicPolynomialTerm> =
925        std::collections::HashMap::new();
926    for (power, term) in left.iter().chain(right.iter()) {
927        let mut existing = accumulator
928            .get(power)
929            .cloned()
930            .unwrap_or(SymbolicPolynomialTerm {
931                linear_part: vec![Rational::ZERO; d],
932                constant_part: Rational::ZERO,
933            });
934        for (idx, value) in term.linear_part.iter().enumerate() {
935            existing.linear_part[idx] += value.clone();
936        }
937        existing.constant_part += term.constant_part.clone();
938        accumulator.insert(*power, existing);
939    }
940    let mut result: SymbolicPoly = accumulator.into_iter().collect();
941    result.sort_by_key(|(power, _)| *power);
942    result
943}

945fn constraint_row_for_power(
946    constraint: &SymbolicPoly,
947    power: usize,
948    d: usize,
949) -> (Vec<Rational>, Rational) {
950    // Look up the linear-form / constant pair at a specific power of
951    // x in the constraint LHS. Missing powers contribute the zero
952    // row, which the linear-system solver interprets as a trivially
953    // satisfied equation.
954    for (entry_power, term) in constraint {
955        if *entry_power == power {
956            return (term.linear_part.clone(), term.constant_part.clone());
957        }
958    }
959    (vec![Rational::ZERO; d], Rational::ZERO)
960}

962fn solve_linear_system_for_p_coefficients(
963    equations: &[Vec<Rational>],
964    constants: &[Rational],
965    d: usize,
966) -> Option<Vec<Rational>> {
967    // Solve the over-determined homogeneous linear system
968    //   sum_i a_(j,i) * c_i = -b_j     for all j
969    // with `equations[j]` = (a_(j,0), ..., a_(j,d-1)) and `constants[j]`
970    // = b_j. Uses Gauss elimination over Q. Returns the (unique, when
971    // it exists) solution vector or None on inconsistency / under-
972    // determination.
973    if d == 0 {
974        // No unknowns: every equation collapses to constants[j] = 0.
975        for constant_value in constants {
976            if *constant_value != Rational::ZERO {
977                return None;
978            }
979        }
980        return Some(Vec::new());
981    }
982    let mut augmented: Vec<Vec<Rational>> = Vec::new();
983    for (row_index, row) in equations.iter().enumerate() {
984        let mut full_row: Vec<Rational> = row.clone();
985        full_row.push(-constants[row_index].clone());
986        augmented.push(full_row);
987    }
988    let mut pivot_row = 0;
989    for column in 0..d {
990        let mut found_pivot = None;
991        for row in pivot_row..augmented.len() {
992            if augmented[row][column] != Rational::ZERO {
993                found_pivot = Some(row);
994                break;
995            }
996        }
997        let pivot_row_index = match found_pivot {
998            Some(value) => value,
999            None => continue,
1000        };
1001        augmented.swap(pivot_row, pivot_row_index);
1002        let pivot_value = augmented[pivot_row][column].clone();
1003        for column_index in 0..=d {
1004            augmented[pivot_row][column_index] =
1005                augmented[pivot_row][column_index].clone() / pivot_value.clone();
1006        }
1007        for row in 0..augmented.len() {
1008            if row == pivot_row {
1009                continue;
1010            }
1011            let factor = augmented[row][column].clone();
1012            if factor == Rational::ZERO {
1013                continue;
1014            }
1015            for column_index in 0..=d {
1016                let subtractor = augmented[pivot_row][column_index].clone() * factor.clone();
1017                augmented[row][column_index] = augmented[row][column_index].clone() - subtractor;
1018            }
1019        }
1020        pivot_row += 1;
1021    }
1022    for row in &augmented {
1023        let all_zero_left = row.iter().take(d).all(|value| *value == Rational::ZERO);
1024        if all_zero_left && row[d] != Rational::ZERO {
1025            return None;
1026        }
1027    }
1028    let mut solution = vec![Rational::ZERO; d];
1029    for row in &augmented {
1030        let mut pivot_column: Option<usize> = None;
1031        for column in 0..d {
1032            if row[column] == Rational::ONE {
1033                let other_zero = (0..d)
1034                    .filter(|c| *c != column)
1035                    .all(|other| row[other] == Rational::ZERO);
1036                if other_zero {
1037                    pivot_column = Some(column);
1038                    break;
1039                }
1040            }
1041        }
1042        if let Some(column) = pivot_column {
1043            solution[column] = row[d].clone();
1044        }
1045    }
1046    Some(solution)
1047}

1049fn build_polynomial_r_solution(
1050    context: &Context,
1051    p_polynomial: &Polynomial,
1052    omega_polynomial: &Polynomial,
1053    independent_variable: SymbolIdentifier,
1054) -> ExpressionReference {
1055    // Reify u = P * exp(integral(omega) dx) as an expression. The
1056    // integral of a polynomial is itself polynomial (formally
1057    // integrated power-by-power), so this never stalls inside
1058    // integrate.
1059    use crate::polynomial::conversion::polynomial_to_expression;
1060    let p_expression = polynomial_to_expression(context, p_polynomial);
1061    let omega_expression = polynomial_to_expression(context, omega_polynomial);
1062    let integral_of_omega =
1063        crate::integrate::integrate(context, &omega_expression, independent_variable);
1064    let exponential = builders::function_call(
1065        context,
1066        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
1067        vec![integral_of_omega],
1068    );
1069    let basis_one = builders::product(
1070        context,
1071        vec![Arc::clone(&p_expression), Arc::clone(&exponential)],
1072    );
1073    // Second basis solution via reduction-of-order: u_2 = u_1 *
1074    // integral(1 / u_1^2 dx). We leave this as the second basis when
1075    // integrate closes; otherwise we emit u_1 alone with a single
1076    // arbitrary constant.
1077    let constant_one_identifier = context.symbol_table.fresh("constant_kovacic_one");
1078    let constant_one = context
1079        .interner
1080        .intern(ExpressionKind::Symbol(constant_one_identifier));
1081    let scaled_basis_one = builders::product(context, vec![constant_one, Arc::clone(&basis_one)]);
1082    let exponent_two = builders::integer_value(context, 2);
1083    let basis_one_squared = builders::power(context, Arc::clone(&basis_one), exponent_two);
1084    let one = builders::one(context);
1085    let reciprocal_basis_one_squared = builders::divide(context, one, basis_one_squared);
1086    let integral_of_reciprocal_squared =
1087        crate::integrate::integrate(context, &reciprocal_basis_one_squared, independent_variable);
1088    let basis_two = builders::product(
1089        context,
1090        vec![Arc::clone(&basis_one), integral_of_reciprocal_squared],
1091    );
1092    let constant_two_identifier = context.symbol_table.fresh("constant_kovacic_two");
1093    let constant_two = context
1094        .interner
1095        .intern(ExpressionKind::Symbol(constant_two_identifier));
1096    let scaled_basis_two = builders::product(context, vec![constant_two, basis_two]);
1097    builders::sum(context, vec![scaled_basis_one, scaled_basis_two])
1098}

1103    // sub-case and do not pollute the broader case_one module.
1104    crate::polynomial::arithmetic::add(left, right)

1116    }
