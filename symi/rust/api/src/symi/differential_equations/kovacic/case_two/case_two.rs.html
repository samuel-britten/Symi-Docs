1// Kovacic Case 2 (minimal slice). Looks for a Liouvillian solution
2// u of u'' = r*u such that u^2 has logarithmic derivative
3// phi(x) = theta(x) + 2*P'(x)/P(x)
4// with theta and P in Q(x) and Q[x] respectively. The Liouvillian
5// solutions are then
6//   u_{+/-}(x) = exp(integral omega_{+/-}(x) dx)
7// where omega_{+/-} are the two roots of the quadratic
8//   omega^2 - phi(x) * omega + (phi' + phi^2 - 2*r) / 2 = 0
9// derived from the relation omega' + omega^2 = r evaluated on each
10// branch (Vieta on the two branches gives the (phi, p)
11// parameters). The two branches differ by an algebraic-extension
12// element sqrt(phi^2 + 4 r - 2 phi') which is realised inside the
13// expression layer as power(value, 1/2).
14//
15// This first slice ships only the single-order-2-pole-and-
16// infinity-order-2 sub-case (the canonical
17// Case-2-requires-this-because-Case-1's-alpha-pair-is-irrational
18// example). The broader Case 2 enumeration (mixed pole orders,
19// non-square infinity behaviour, polynomial-search at higher d)
20// lands in a follow-up patch.
21//
22// For the single-order-2-pole sub-case with rational pole location
23// c and residue b, the closed form admits a direct simplification:
24// the local indicial equation alpha(alpha - 1) = b gives alpha =
25// (1 +/- sqrt(1 + 4 * b)) / 2 directly, and the two basis
26// solutions are u_{+/-} = (x - c)^alpha_{+/-}. The sqrt(1 + 4 * b)
27// piece is the algebraic-extension element that Case 2 introduces
28// in place of Case 1's rationality requirement.

43/// Performs the advanced native `solve` operation.
44pub fn solve(
45    context: &Context,
46    r_numerator: &Polynomial,
47    r_denominator: &Polynomial,
48    poles: &PoleData,
49    independent_variable: SymbolIdentifier,
50) -> Option<ExpressionReference> {
51    // Public Case 2 dispatcher. The current implementation ships a
52    // single sub-case: r has exactly one finite pole of order 2, the
53    // order at infinity is 2, the discriminant 1 + 4 * residue (with
54    // residue the order-2 leading coefficient at the finite pole) is
55    // NOT a rational square (Case 1 must have failed for this dispatch
56    // to make sense), the corresponding infinity-side discriminant is
57    // identical, and the polynomial-search step closes at d = 0 with
58    // P = 1.
59    let _ = (r_numerator, r_denominator);
60    if poles.poles.len() != 1 {
61        return None;
62    }
63    let pole = poles.poles.first()?;
64    if pole.order != 2 {
65        return None;
66    }
67    if poles.infinity_order != 2 {
68        return None;
69    }
70    let residue = pole.residue_at_order_two.clone()?;
71    let infinity_residue = poles.infinity_leading_constant.clone()?;
72    let _ = candidate_exponent_for_order_two_pole(&residue)?;
73    let _ = candidate_exponent_for_order_two_pole(&infinity_residue)?;
74    Some(build_case_two_closed_form(
75        context,
76        &pole.location,
77        &residue,
78        independent_variable,
79    ))
80}

82fn candidate_exponent_for_order_two_pole(residue_value: &Rational) -> Option<Rational> {
83    // The Case 2 exponent set at an order-2 pole with residue b is
84    //   E_c = {2 + 2 * k * sqrt(1 + 4 * b) : k in {-1, 0, 1}} intersected with Z.
85    // When sqrt(1 + 4 * b) is irrational, only the k = 0 entry, namely
86    // 2, survives intersection with Z; that is the regime this slice
87    // closes. When sqrt(1 + 4 * b) is a rational square, Case 1 has
88    // already closed the input, and Case 2 defers (returns None) so
89    // the dispatcher does not emit a duplicate-and-ungrouped pair.
90    let discriminant = Rational::ONE + Rational::from(4) * residue_value.clone();
91    if rational_square_root_local(&discriminant).is_some() {
92        return None;
93    }
94    Some(Rational::from(2))
95}

97fn build_case_two_closed_form(
98    context: &Context,
99    pole_location: &Rational,
100    residue_value: &Rational,
101    independent_variable: SymbolIdentifier,
102) -> ExpressionReference {
103    // Reify the two basis solutions u_{+/-} = (x - c)^alpha_{+/-} with
104    //   alpha_{+/-} = (1 +/- sqrt(1 + 4 * residue)) / 2.
105    // The sqrt is symbolic: when 1 + 4 * residue is a perfect square
106    // we use the rational root directly; otherwise we emit the
107    // power(value, 1/2) form so the expression layer carries the
108    // algebraic extension element verbatim. Result:
109    //   C_1 * (x - c)^alpha_+ + C_2 * (x - c)^alpha_-.
110    let independent_variable_expression = context
111        .interner
112        .intern(ExpressionKind::Symbol(independent_variable));
113    let location_expression = rational_to_expression_local(context, pole_location);
114    let shifted_variable = builders::subtract(
115        context,
116        Arc::clone(&independent_variable_expression),
117        location_expression,
118    );
119    let alpha_plus = build_alpha_expression(context, residue_value, false);
120    let alpha_minus = build_alpha_expression(context, residue_value, true);
121    let basis_plus = builders::power(context, Arc::clone(&shifted_variable), alpha_plus);
122    let basis_minus = builders::power(context, shifted_variable, alpha_minus);
123    let constant_one_identifier = context.symbol_table.fresh("constant_kovacic_one");
124    let constant_two_identifier = context.symbol_table.fresh("constant_kovacic_two");
125    let constant_one = context
126        .interner
127        .intern(ExpressionKind::Symbol(constant_one_identifier));
128    let constant_two = context
129        .interner
130        .intern(ExpressionKind::Symbol(constant_two_identifier));
131    let term_one = builders::product(context, vec![constant_one, basis_plus]);
132    let term_two = builders::product(context, vec![constant_two, basis_minus]);
133    builders::sum(context, vec![term_one, term_two])
134}

136fn build_alpha_expression(
137    context: &Context,
138    residue_value: &Rational,
139    negate_branch: bool,
140) -> ExpressionReference {
141    // Build alpha = (1 +/- sqrt(1 + 4 * residue_value)) / 2 as an
142    // expression. The +/- selection is controlled by negate_branch:
143    // false yields the (+sqrt) branch, true yields the (-sqrt)
144    // branch. When 1 + 4 * residue happens to be a rational square
145    // (the Case-1-applicable case), the sqrt collapses to a rational
146    // literal; otherwise the closed form carries a power(value, 1/2)
147    // algebraic-extension element.
148    let discriminant = Rational::ONE + Rational::from(4) * residue_value.clone();
149    let sqrt_discriminant_expression =
150        if let Some(perfect_root) = rational_square_root_local(&discriminant) {
151            rational_to_expression_local(context, &perfect_root)
152        } else {
153            let discriminant_expression = rational_to_expression_local(context, &discriminant);
154            let half_exponent = builders::rational_value(context, 1, 2);
155            builders::power(context, discriminant_expression, half_exponent)
156        };
157    let signed_sqrt_expression = if negate_branch {
158        builders::negate(context, sqrt_discriminant_expression)
159    } else {
160        sqrt_discriminant_expression
161    };
162    let one = builders::one(context);
163    let one_plus_sqrt = builders::sum(context, vec![one, signed_sqrt_expression]);
164    let half = builders::rational_value(context, 1, 2);
165    builders::product(context, vec![half, one_plus_sqrt])
166}

168fn rational_to_expression_local(context: &Context, value: &Rational) -> ExpressionReference {
169    // Reify a Rational into the expression layer through the existing
170    // builder. The auto-simplifier collapses integer-valued rationals
171    // to the integer constructor automatically.
172    if let Ok(integer_value) = Integer::try_from(value.clone()) {
173        return builders::integer(context, integer_value);
174    }
175    let (numerator_natural, denominator_natural) = value.numerator_and_denominator_ref();
176    let mut numerator_integer = Integer::from(numerator_natural.clone());
177    let denominator_integer = Integer::from(denominator_natural.clone());
178    if value.sign_comparison() == std::cmp::Ordering::Less {
179        numerator_integer = -numerator_integer;
180    }
181    builders::rational(context, numerator_integer, denominator_integer)
182}

184fn rational_square_root_local(value: &Rational) -> Option<Rational> {
185    // Return sqrt(value) as a Rational when value is a perfect rational
186    // square. Used to gate Case 2 entry: Case 1 should have closed the
187    // input first whenever the alpha-pair discriminant is a rational
188    // square, so Case 2 only fires when this returns None.
189    crate::exact_arithmetic::rational_exact_square_root(value)
190}
