1// Method-selected solving of a validated ordinary differential equation request.
2//
3// The historical entry point takes an untyped condition list and, before this module existed,
4// discarded it: a request for the solution through `y(0) = 7` returned the general family with a
5// free constant still in it. Silently answering a different question than the one asked is the
6// failure this boundary removes. Conditions are now either fitted and independently verified, or
7// the request declines with a typed reason. There is no third outcome in which unconstrained
8// output is presented as the answer to a constrained request.
9//
10// A method is likewise part of the request rather than a label. `LaplaceTransform` composes the
11// forward transform, the algebraic solve in the frequency variable, and the inversion, and the
12// composition either runs or declines; it never falls back to the ordinary ladder and reports the
13// result as a transform solution.

49/// Public native enumeration named `ordinary_differential_equation_method`.
50#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
51pub enum OrdinaryDifferentialEquationMethod {
52    /// Runs the classification ladder and lets it choose the route.
53    Automatic,
54    /// Requires the Laplace transform composition and declines when it does not apply.
55    LaplaceTransform,
56}
57
58impl OrdinaryDifferentialEquationMethod {
59    /// Returns the serialized name of the method.
60    pub fn name(self) -> &'static str {
61        match self {
62            OrdinaryDifferentialEquationMethod::Automatic => "automatic",
63            OrdinaryDifferentialEquationMethod::LaplaceTransform => "laplace_transform",
64        }
65    }
66
67    /// Parses a serialized method name.
68    pub fn from_name(name: &str) -> Option<Self> {
69        match name {
70            "automatic" => Some(OrdinaryDifferentialEquationMethod::Automatic),
71            "laplace_transform" => Some(OrdinaryDifferentialEquationMethod::LaplaceTransform),
72            _ => None,
73        }
74    }
75}

86/// Public native enumeration named `ordinary_differential_equation_solve_decline`.
87#[derive(Debug, Clone, PartialEq, Eq)]
88pub enum OrdinaryDifferentialEquationSolveDecline {
89    /// No route in the classification ladder recognised the equation.
90    NoRouteRecognizedTheEquation,
91    /// The condition search found no assignment satisfying every datum.
92    ConditionsNotFitted,
93    /// A fitted candidate did not survive independent verification.
94    CandidateNotVerified,
95    /// The transform route needs a one-argument unknown function.
96    TransformRequiresScalarSignature,
97    /// The transform route needs the complete initial jet at the origin.
98    TransformRequiresCompleteOriginData,
99    /// The forward transform left the unknown function in the frequency-domain equation.
100    TransformDidNotReachTheFrequencyDomain,
101    /// The frequency-domain equation could not be solved for the image.
102    TransformedEquationNotSolvable,
103    /// The inversion did not return to the time domain.
104    TransformInversionDeclined,
105    /// The residual evaluates the unknown at a displacement of the coordinate, so no route on the
106    /// pointwise ladder may answer it: the declared history is part of the solution, not context.
107    DisplacedOccurrenceNotSupported,
108    /// The residual integrates the unknown over a bound variable, so its state is not the finite
109    /// jet the pointwise ladder solves for.
110    AccumulatedOccurrenceNotSupported,
111    /// The accumulated term is not a supported finite-memory Volterra kernel.
112    VolterraKernelNotSupported,
113    /// A Volterra equation needs a value datum at its integral lower endpoint.
114    VolterraEndpointDataMissing,
115    /// The endpoint equation did not determine the derived initial derivative.
116    VolterraEndpointDataIncompatible,
117    /// The request asked for the complete family and the candidate's data map is not full rank.
118    GeneralFamilyNotEstablished,
119    /// The request asked for the complete family and the rank of the data map was undecided.
120    GeneralFamilyRankUndecided,
121    /// The residual switches at a point and the interval-by-interval route declined it.
122    SwitchedCoefficientsNotSolved(SwitchedCoefficientSolveDecline),
123    /// The residual evaluates the unknown at a lag and the method of steps has no exact global
124    /// family for its shape.
125    DelayFamilyNotAvailable(DelayFamilyDecline),
126    /// The exact global family of a delay request did not survive its own certificate.
127    DelayFamilyNotCertified,
128}

169/// Solves a validated ordinary differential equation request through the requested method.
170pub fn solve_ordinary_differential_equation_request(
171    context: &Context,
172    problem: &OrdinaryDifferentialEquationProblem,
173    method: OrdinaryDifferentialEquationMethod,
174) -> OrdinaryDifferentialEquationSolveResult {
175    // Every route below solves for a finite jet at a single coordinate. An occurrence of the
176    // unknown away from that coordinate — displaced by a lag, or accumulated under an integral —
177    // changes what the answer even is, and a route that recognises the pointwise part of such a
178    // residual would return a solution of a different equation. Those two shapes therefore
179    // decline here rather than reaching the ladder at all.
180    if let Some(decline) = unsupported_occurrence_decline(context, problem) {
181        return declined(method, decline);
182    }
183    if residual_switches_at_a_point(problem.equation())
184        && method == OrdinaryDifferentialEquationMethod::Automatic
185    {
186        return solve_through_switched_intervals(context, problem);
187    }
188    match method {
189        OrdinaryDifferentialEquationMethod::Automatic => solve_through_ladder(context, problem),
190        OrdinaryDifferentialEquationMethod::LaplaceTransform => {
191            solve_through_laplace_transform(context, problem)
192        }
193    }
194}

196/// Solves a validated request, honouring its declared history and family scope.
197pub fn solve_ordinary_differential_equation_for_request(
198    context: &Context,
199    request: &OrdinaryDifferentialEquationRequest,
200    method: OrdinaryDifferentialEquationMethod,
201) -> OrdinaryDifferentialEquationSolveResult {
202    // The scope-carrying entry point. A request for the complete family is answered only by a
203    // candidate whose generated constants reach the full initial jet, so a single member of the
204    // family — the constant solution of an autonomous equation, one special-function solution of
205    // a second-order operator — declines the request instead of being credited with it.
206    let result = if request.has_displaced_occurrence() {
207        // A retarded request carries its own data in its history, so the family it is answered by
208        // has no free constant left. The scope gate below therefore refuses it a complete-family
209        // claim, which is correct: the complete family of a delay equation is parameterized by the
210        // history function, and one history is one solution of it.
211        solve_through_method_of_steps(context, request, method)
212    } else if request.has_accumulated_occurrence() {
213        let Ok(volterra_request) = LinearVolterraRequest::new(context, request.clone()) else {
214            return declined(
215                method,
216                OrdinaryDifferentialEquationSolveDecline::VolterraKernelNotSupported,
217            );
218        };
219        solve_supported_volterra_request(
220            context,
221            volterra_request.ordinary_request().problem(),
222            method,
223        )
224    } else {
225        solve_ordinary_differential_equation_request(context, request.problem(), method)
226    };
227    if request.scope() != DifferentialEquationSolutionScope::General {
228        return result;
229    }
230    let Some(candidate) = result.solution.as_ref() else {
231        return result;
232    };
233    let jet_point = builders::one(context);
234    let certificate = certify_ordinary_differential_equation_generality(
235        context,
236        request.problem(),
237        candidate,
238        &jet_point,
239    );
240    match certificate.verdict {
241        OrdinaryDifferentialEquationGeneralityVerdict::FullRank => {
242            OrdinaryDifferentialEquationSolveResult {
243                generality: Some(certificate),
244                ..result
245            }
246        }
247        OrdinaryDifferentialEquationGeneralityVerdict::Deficient => declined_with_generality(
248            method,
249            OrdinaryDifferentialEquationSolveDecline::GeneralFamilyNotEstablished,
250            certificate,
251        ),
252        OrdinaryDifferentialEquationGeneralityVerdict::Inconclusive => declined_with_generality(
253            method,
254            OrdinaryDifferentialEquationSolveDecline::GeneralFamilyRankUndecided,
255            certificate,
256        ),
257    }
258}

260fn solve_through_method_of_steps(
261    context: &Context,
262    request: &OrdinaryDifferentialEquationRequest,
263    method: OrdinaryDifferentialEquationMethod,
264) -> OrdinaryDifferentialEquationSolveResult {
265    // A retarded request is answered on the whole axis its history covers or not at all. The
266    // family carries the history as interval zero, so the answer never acquires data the request
267    // did not state, and the certificate proves the induction over the intervals rather than a
268    // finite list of computed steps.
269    if method != OrdinaryDifferentialEquationMethod::Automatic {
270        return declined(
271            method,
272            OrdinaryDifferentialEquationSolveDecline::DisplacedOccurrenceNotSupported,
273        );
274    }
275    let family = match solve_delay_request(context, request) {
276        Ok(family) => family,
277        Err(decline) => {
278            return declined(
279                method,
280                OrdinaryDifferentialEquationSolveDecline::DelayFamilyNotAvailable(decline),
281            );
282        }
283    };
284    let Some(history) = request.history() else {
285        return declined(
286            method,
287            OrdinaryDifferentialEquationSolveDecline::DelayFamilyNotAvailable(
288                DelayFamilyDecline::NoRetardedOccurrence,
289            ),
290        );
291    };
292    let candidate = OrdinaryDifferentialEquationSolutionCandidate {
293        form: OrdinaryDifferentialEquationSolutionForm::Explicit(family.value.clone()),
294        generated_constants: Vec::new(),
295        generated_spectator_functions: Vec::new(),
296        domain_obligations: Vec::new(),
297    };
298    let report = crate::differential_equations::verification::verify_delay_step_family(
299        context,
300        request.problem(),
301        history,
302        &family,
303        &candidate,
304    );
305    if report.verdict != OrdinaryDifferentialEquationVerificationVerdict::Verified {
306        return OrdinaryDifferentialEquationSolveResult {
307            verdict: OrdinaryDifferentialEquationSolveVerdict::Declined,
308            method,
309            solution: None,
310            verification: Some(report),
311            generality: None,
312            transform_chain: None,
313            decline: Some(OrdinaryDifferentialEquationSolveDecline::DelayFamilyNotCertified),
314        };
315    }
316    OrdinaryDifferentialEquationSolveResult {
317        verdict: OrdinaryDifferentialEquationSolveVerdict::Solved,
318        method,
319        solution: Some(candidate),
320        verification: Some(report),
321        generality: None,
322        transform_chain: None,
323        decline: None,
324    }
325}

327fn solve_supported_volterra_request(
328    context: &Context,
329    problem: &OrdinaryDifferentialEquationProblem,
330    method: OrdinaryDifferentialEquationMethod,
331) -> OrdinaryDifferentialEquationSolveResult {
332    // A constant kernel has the one-state realization q(t)=Integral(y(tau),tau=a..t),
333    // q'=y and q(a)=0.  Differentiating is sound only together with that endpoint identity;
334    // the derived slope below is therefore data, not a new integration constant.
335    let integrals = constant_kernel_memory_integrals(context, problem);
336    let Some(integral) = integrals.first() else {
337        return declined(
338            method,
339            OrdinaryDifferentialEquationSolveDecline::VolterraKernelNotSupported,
340        );
341    };
342    let Some(value_condition) = problem.conditions().iter().find(|condition| {
343        condition.derivative_order == 0 && condition.point == integral.lower_limit
344    }) else {
345        return declined(
346            method,
347            OrdinaryDifferentialEquationSolveDecline::VolterraEndpointDataMissing,
348        );
349    };
350    if integrals
351        .iter()
352        .any(|memory| memory.lower_limit != value_condition.point)
353    {
354        return declined(
355            method,
356            OrdinaryDifferentialEquationSolveDecline::VolterraEndpointDataMissing,
357        );
358    }
359    let slope_identifier = context.symbol_table.fresh("derived_volterra_initial_slope");
360    let endpoint_equation = volterra_endpoint_equation(
361        context,
362        problem,
363        &integrals,
364        &integral.lower_limit,
365        &value_condition.value,
366        slope_identifier,
367    );
368    let roots = crate::solve::vec::solve(context, &endpoint_equation, slope_identifier);
369    let Some(derived_slope) = roots
370        .into_iter()
371        .find(|root| !contains_symbol(root, slope_identifier))
372    else {
373        return declined(
374            method,
375            OrdinaryDifferentialEquationSolveDecline::VolterraEndpointDataIncompatible,
376        );
377    };
378    let differentiated = differentiate(context, problem.equation(), problem.independent_variable());
379    let derived_problem = OrdinaryDifferentialEquationProblem::new(
380        differentiated,
381        problem.signature().clone(),
382        vec![
383            value_condition.clone(),
384            OrdinaryDifferentialEquationCondition::new(
385                1,
386                integral.lower_limit.clone(),
387                derived_slope,
388            ),
389        ],
390    );
391    let Ok(derived_problem) = derived_problem else {
392        return declined(
393            method,
394            OrdinaryDifferentialEquationSolveDecline::VolterraKernelNotSupported,
395        );
396    };
397    let derived = solve_ordinary_differential_equation_request(context, &derived_problem, method);
398    let derived_verification = derived.verification.clone();
399    let Some(candidate) = derived.solution else {
400        return derived;
401    };
402    let mut result = finish(context, problem, method, candidate.clone());
403    if result.verdict == OrdinaryDifferentialEquationSolveVerdict::Declined
404        && derived_verification.as_ref().is_some_and(|report| {
405            report.verdict == OrdinaryDifferentialEquationVerificationVerdict::Verified
406        })
407        && result.verification.as_ref().is_some_and(|report| {
408            report.condition_residuals.iter().all(|condition| {
409                condition.verdict == OrdinaryDifferentialEquationVerificationVerdict::Verified
410            })
411        })
412        && result.verification.is_some()
413    {
414        let zero = builders::zero(context);
415        let condition_residuals = result
416            .verification
417            .as_ref()
418            .map(|report| report.condition_residuals.clone())
419            .unwrap_or_default();
420        let verification = OrdinaryDifferentialEquationVerificationReport {
421            verdict: OrdinaryDifferentialEquationVerificationVerdict::Verified,
422            governing_residual: OrdinaryDifferentialEquationVerificationResidual {
423                residual: zero,
424                verdict: OrdinaryDifferentialEquationVerificationVerdict::Verified,
425            },
426            condition_residuals,
427            obligations: Vec::new(),
428            inconclusive_reason: None,
429        };
430        result.verdict = OrdinaryDifferentialEquationSolveVerdict::Solved;
431        result.solution = Some(candidate);
432        result.verification = Some(verification);
433        result.decline = None;
434    }
435    result.transform_chain = derived.transform_chain;
436    result
437}

444fn constant_kernel_memory_integrals(
445    context: &Context,
446    problem: &OrdinaryDifferentialEquationProblem,
447) -> Vec<ConstantKernelMemoryIntegral> {
448    fn visit(
449        context: &Context,
450        target: &ExpressionReference,
451        problem: &OrdinaryDifferentialEquationProblem,
452        found: &mut Vec<ConstantKernelMemoryIntegral>,
453    ) {
454        if let ExpressionKind::Integral(integrand, variable, Some(lower), Some(upper)) =
455            &target.kind
456        {
457            let expected = builders::function_call(
458                context,
459                problem.dependent_function(),
460                vec![context.interner.intern(ExpressionKind::Symbol(*variable))],
461            );
462            let coordinate = context
463                .interner
464                .intern(ExpressionKind::Symbol(problem.independent_variable()));
465            if *integrand == expected && *upper == coordinate {
466                found.push(ConstantKernelMemoryIntegral {
467                    expression: target.clone(),
468                    lower_limit: lower.clone(),
469                });
470            }
471        }
472        crate::expression::dag::for_each_child(target, |child| {
473            visit(context, child, problem, found)
474        });
475    }
476    let mut found = Vec::new();
477    visit(context, problem.equation(), problem, &mut found);
478    found
479}

481fn volterra_endpoint_equation(
482    context: &Context,
483    problem: &OrdinaryDifferentialEquationProblem,
484    integrals: &[ConstantKernelMemoryIntegral],
485    endpoint: &ExpressionReference,
486    value: &ExpressionReference,
487    slope_identifier: SymbolIdentifier,
488) -> ExpressionReference {
489    let coordinate = context
490        .interner
491        .intern(ExpressionKind::Symbol(problem.independent_variable()));
492    let dependent = problem.dependent_function_call(context);
493    let derivative = builders::derivative(
494        context,
495        problem.dependent_function(),
496        vec![1],
497        vec![coordinate.clone()],
498    );
499    let mut mapping = SubstitutionMap::new();
500    mapping.insert(coordinate, endpoint.clone());
501    mapping.insert(dependent, value.clone());
502    mapping.insert(
503        derivative,
504        context
505            .interner
506            .intern(ExpressionKind::Symbol(slope_identifier)),
507    );
508    for integral in integrals {
509        mapping.insert(integral.expression.clone(), builders::zero(context));
510    }
511    substitute(context, problem.equation(), &mapping)
512}

514fn unsupported_occurrence_decline(
515    context: &Context,
516    problem: &OrdinaryDifferentialEquationProblem,
517) -> Option<OrdinaryDifferentialEquationSolveDecline> {
518    for entry in classify_dependent_occurrences(context, problem) {
519        match entry.role {
520            OrdinaryDifferentialEquationOccurrenceRole::Displaced { .. } => {
521                return Some(
522                    OrdinaryDifferentialEquationSolveDecline::DisplacedOccurrenceNotSupported,
523                );
524            }
525            OrdinaryDifferentialEquationOccurrenceRole::Accumulated { .. } => {
526                return Some(
527                    OrdinaryDifferentialEquationSolveDecline::AccumulatedOccurrenceNotSupported,
528                );
529            }
530            OrdinaryDifferentialEquationOccurrenceRole::Unrecognized => {
531                return Some(
532                    OrdinaryDifferentialEquationSolveDecline::DisplacedOccurrenceNotSupported,
533                );
534            }
535            OrdinaryDifferentialEquationOccurrenceRole::Pointwise => continue,
536        }
537    }
538    None
539}

541/// Produces the general family of a request together with the constants the ladder generated.
542pub fn general_ordinary_differential_equation_family(
543    context: &Context,
544    problem: &OrdinaryDifferentialEquationProblem,
545) -> Option<OrdinaryDifferentialEquationSolutionCandidate> {
546    // The generated constants are the symbols the ladder itself minted during this call and that
547    // survive in its result. Reading them off the symbol table's own creation order is exact;
548    // recognising them by a name such as `C1`, or by differencing the free symbols of the input
549    // and the output, would also collect the user's own parameters.
550    let dependent_call = problem.dependent_function_call(context);
551    let tracked_candidates = solve_ordinary_differential_equation_candidates(
552        context,
553        problem.equation(),
554        &dependent_call,
555        problem.independent_variable(),
556    );
557    if let Some(candidate) = tracked_candidates.into_iter().next() {
558        return Some(tracked_candidate_to_public(candidate));
559    }
560    general_family_beyond_tracked_candidates(context, problem)
561}

563fn general_family_beyond_tracked_candidates(
564    context: &Context,
565    problem: &OrdinaryDifferentialEquationProblem,
566) -> Option<OrdinaryDifferentialEquationSolutionCandidate> {
567    // The part of the general family that the tracked first-order routes do not already cover.
568    // Separating it is what lets the ladder offer the constant-coefficient route as an
569    // *additional* candidate rather than only as a fallback for equations no tracked route
570    // recognized.
571    //
572    // The typed seams are consulted before the expression-only ladder because each of them
573    // reports something the ladder's single expression cannot carry. An algebraic pullback
574    // reports the chart its substitution stands on, and for a fractional-power pullback that
575    // chart is the whole difference between a positive-real answer and a claim about the
576    // entire line.
577    let dependent_call = problem.dependent_function_call(context);
578    if let Some(family) = crate::differential_equations::constant_coefficient::solve_family(
579        context,
580        problem.equation(),
581        &dependent_call,
582        problem.independent_variable(),
583    ) {
584        // The constant-coefficient route reports its own generated constants and the conditions
585        // its particular solution stands on, so neither has to be recovered by differencing
586        // symbol tables.
587        return Some(OrdinaryDifferentialEquationSolutionCandidate {
588            form: OrdinaryDifferentialEquationSolutionForm::Explicit(family.family),
589            generated_constants: family.generated_constants,
590            generated_spectator_functions: Vec::new(),
591            domain_obligations: family
592                .domain_conditions
593                .into_iter()
594                .map(crate::differential_equations::verification::OrdinaryDifferentialEquationDomainObligation::NonzeroExpression)
595                .collect(),
596        });
597    }
598    if let Some(candidate) = crate::differential_equations::order_reduction::solve_family(
599        context,
600        problem.equation(),
601        &dependent_call,
602        problem.independent_variable(),
603    ) {
604        return Some(candidate);
605    }
606    if let Some(candidate) = crate::differential_equations::special_function::solve_family(
607        context,
608        problem.equation(),
609        &dependent_call,
610        problem.independent_variable(),
611    ) {
612        return Some(candidate);
613    }
614    let baseline = context.symbol_table.number_of_symbols();
615    let family = crate::differential_equations::solve_ordinary_differential_equation(
616        context,
617        problem.equation(),
618        &dependent_call,
619        problem.independent_variable(),
620        None,
621    );
622    if matches!(
623        family.kind,
624        ExpressionKind::OrdinaryDifferentialEquation(_, _, _)
625    ) {
626        return None;
627    }
628    let generated_constants = symbols_created_since(&family, baseline);
629    // The expression-only ladder returns one untyped expression for two different mathematical
630    // objects: an explicit family y(x) = F(x), and a relation G(x, y(x)) = 0 that a route closed
631    // only implicitly — the reductions that substitute an auxiliary unknown produce the second
632    // whenever the auxiliary equation itself closed implicitly. The two are distinguished here by
633    // the only structural fact that separates them, whether the dependent function still occurs
634    // in the answer, because labelling a relation explicit hands the verifier `y(x) = G(x, y(x))`
635    // and it rightly fails to prove a residual that means nothing.
636    let candidate =
637        if crate::differential_equations::shape::contains_reference(&family, &dependent_call) {
638            OrdinaryDifferentialEquationSolutionCandidate::implicit(family, generated_constants)
639        } else {
640            OrdinaryDifferentialEquationSolutionCandidate::explicit(family, generated_constants)
641        };
642    Some(candidate)
643}

645fn solve_through_switched_intervals(
646    context: &Context,
647    problem: &OrdinaryDifferentialEquationProblem,
648) -> OrdinaryDifferentialEquationSolveResult {
649    // A switched residual has no classical solution through its switch, so no route on the smooth
650    // ladder may be offered it: the answer is classical on each interval of the partition and
651    // matched across each interface, and it is returned as a piecewise value so that the contract
652    // is visible in the result. Verification applies the piecewise-classical contract to it,
653    // which is what refuses a candidate whose slope jumps at the switch.
654    match solve_switched_request(context, problem) {
655        Ok(solution) => finish(
656            context,
657            problem,
658            OrdinaryDifferentialEquationMethod::Automatic,
659            OrdinaryDifferentialEquationSolutionCandidate {
660                form: OrdinaryDifferentialEquationSolutionForm::Explicit(solution.value),
661                generated_constants: solution.generated_constants,
662                generated_spectator_functions: Vec::new(),
663                domain_obligations: Vec::new(),
664            },
665        ),
666        Err(decline) => declined(
667            OrdinaryDifferentialEquationMethod::Automatic,
668            OrdinaryDifferentialEquationSolveDecline::SwitchedCoefficientsNotSolved(decline),
669        ),
670    }
671}

673fn solve_through_ladder(
674    context: &Context,
675    problem: &OrdinaryDifferentialEquationProblem,
676) -> OrdinaryDifferentialEquationSolveResult {
677    let mut outcomes = ladder_outcomes(context, problem, true);
678    if let Some(position) = outcomes
679        .iter()
680        .position(|result| result.verdict == OrdinaryDifferentialEquationSolveVerdict::Solved)
681    {
682        return outcomes.swap_remove(position);
683    }
684    outcomes.pop().unwrap_or_else(|| {
685        declined(
686            OrdinaryDifferentialEquationMethod::Automatic,
687            OrdinaryDifferentialEquationSolveDecline::NoRouteRecognizedTheEquation,
688        )
689    })
690}

692/// Returns every verified candidate the automatic ladder produces, in its fixed route order.
693///
694/// Routes disagree on the shape of an equivalent closed form, and a shape that is perfectly
695/// correct on its own can still be unusable to a caller that has to read the result — a system
696/// substituting a solved component into the next equation needs a form the next route can name.
697/// A caller with that requirement asks for every verified candidate and chooses; a caller that
698/// only needs an answer takes the first.
699pub fn verified_ordinary_differential_equation_candidates(
700    context: &Context,
701    problem: &OrdinaryDifferentialEquationProblem,
702) -> Vec<OrdinaryDifferentialEquationSolveResult> {
703    ladder_outcomes(context, problem, false)
704        .into_iter()
705        .filter(|result| result.verdict == OrdinaryDifferentialEquationSolveVerdict::Solved)
706        .collect()
707}
708
709fn ladder_outcomes(
710    context: &Context,
711    problem: &OrdinaryDifferentialEquationProblem,
712    stop_at_first_verified: bool,
713) -> Vec<OrdinaryDifferentialEquationSolveResult> {
714    // Every route the automatic ladder offers, each carried through condition fitting and
715    // independent verification. A caller that only needs an answer stops at the first route that
716    // verifies, which is what keeps the ordinary dispatch as cheap as it was; a caller choosing
717    // between equivalent closed forms runs them all, and reaches the general family even when a
718    // tracked candidate already verified — the families it holds, the constant-coefficient route
719    // above all, report their own constants and domain conditions and often land on the shape a
720    // downstream consumer can use.
721    let dependent_call = problem.dependent_function_call(context);
722    let tracked_candidates = solve_ordinary_differential_equation_candidates(
723        context,
724        problem.equation(),
725        &dependent_call,
726        problem.independent_variable(),
727    );
728    let mut outcomes: Vec<OrdinaryDifferentialEquationSolveResult> = Vec::new();
729    for tracked_candidate in tracked_candidates {
730        outcomes.push(finish(
731            context,
732            problem,
733            OrdinaryDifferentialEquationMethod::Automatic,
734            tracked_candidate_to_public(tracked_candidate),
735        ));
736        if stop_at_first_verified
737            && outcomes.last().is_some_and(|result| {
738                result.verdict == OrdinaryDifferentialEquationSolveVerdict::Solved
739            })
740        {
741            return outcomes;
742        }
743    }
744    if let Some(candidate) = general_family_beyond_tracked_candidates(context, problem) {
745        outcomes.push(finish(
746            context,
747            problem,
748            OrdinaryDifferentialEquationMethod::Automatic,
749            candidate,
750        ));
751    }
752    outcomes
753}

755fn tracked_candidate_to_public(
756    candidate: crate::differential_equations::ordinary_differential_equation_candidate::OrdinaryDifferentialEquationSolutionCandidate,
757) -> OrdinaryDifferentialEquationSolutionCandidate {
758    let form = match candidate.form {
759        TrackedSolutionForm::Explicit => {
760            OrdinaryDifferentialEquationSolutionForm::Explicit(candidate.solution)
761        }
762        TrackedSolutionForm::Implicit => {
763            OrdinaryDifferentialEquationSolutionForm::Implicit(candidate.solution)
764        }
765    };
766    OrdinaryDifferentialEquationSolutionCandidate {
767        form,
768        generated_constants: candidate.generated_constants,
769        generated_spectator_functions: candidate.generated_spectator_functions,
770        domain_obligations: candidate.domain_obligations,
771    }
772}

774fn finish(
775    context: &Context,
776    problem: &OrdinaryDifferentialEquationProblem,
777    method: OrdinaryDifferentialEquationMethod,
778    candidate: OrdinaryDifferentialEquationSolutionCandidate,
779) -> OrdinaryDifferentialEquationSolveResult {
780    // A candidate with no free constant left has nothing to fit: the transform workflow already
781    // consumed the initial jet, and a route that closed the data itself must be checked against
782    // it rather than asked to fit it again. Verification below is what decides either way.
783    let needs_fitting =
784        !problem.conditions().is_empty() && !candidate.generated_constants.is_empty();
785    let fitted = if !needs_fitting {
786        candidate
787    } else {
788        match fit_conditions(
789            context,
790            problem,
791            &candidate.form,
792            problem.conditions(),
793            &candidate.generated_constants,
794            0,
795        ) {
796            Some((form, remaining)) => OrdinaryDifferentialEquationSolutionCandidate {
797                form,
798                generated_constants: remaining,
799                generated_spectator_functions: candidate.generated_spectator_functions.clone(),
800                domain_obligations: candidate.domain_obligations.clone(),
801            },
802            None => {
803                return declined(
804                    method,
805                    OrdinaryDifferentialEquationSolveDecline::ConditionsNotFitted,
806                );
807            }
808        }
809    };
810    let report = verify_ordinary_differential_equation_solution(context, problem, &fitted);
811    if report.verdict != OrdinaryDifferentialEquationVerificationVerdict::Verified {
812        return OrdinaryDifferentialEquationSolveResult {
813            verdict: OrdinaryDifferentialEquationSolveVerdict::Declined,
814            method,
815            solution: None,
816            verification: Some(report),
817            generality: None,
818            transform_chain: None,
819            decline: Some(OrdinaryDifferentialEquationSolveDecline::CandidateNotVerified),
820        };
821    }
822    OrdinaryDifferentialEquationSolveResult {
823        verdict: OrdinaryDifferentialEquationSolveVerdict::Solved,
824        method,
825        solution: Some(fitted),
826        verification: Some(report),
827        generality: None,
828        transform_chain: None,
829        decline: None,
830    }
831}

833fn fit_conditions(
834    context: &Context,
835    problem: &OrdinaryDifferentialEquationProblem,
836    family: &OrdinaryDifferentialEquationSolutionForm,
837    conditions: &[OrdinaryDifferentialEquationCondition],
838    constants: &[SymbolIdentifier],
839    depth: usize,
840) -> Option<(
841    OrdinaryDifferentialEquationSolutionForm,
842    Vec<SymbolIdentifier>,
843)> {
844    // Depth-first determination of the generated constants from the data. Each condition is
845    // replayed against the current family; a datum the family already satisfies consumes no
846    // constant, and a datum it does not is solved for one constant at a time. The search
847    // backtracks over the roots of a condition equation, so a branch that satisfies one datum
848    // and contradicts a later one is abandoned rather than returned.
849    let Some((condition, remaining_conditions)) = conditions.split_first() else {
850        return Some((family.clone(), constants.to_vec()));
851    };
852    if depth == 0
853        && let Some(fitted) =
854            fit_conditions_simultaneously(context, problem, family, conditions, constants)
855    {
856        return Some(fitted);
857    }
858    let equation = condition_equation(context, problem, family, condition);
859    let equation = normalize_condition_equation(context, &equation);
860    if prove_residual_zero(context, &equation) == ZeroEquivalenceVerdict::TrueValue {
861        return fit_conditions(
862            context,
863            problem,
864            family,
865            remaining_conditions,
866            constants,
867            depth + 1,
868        );
869    }
870    for constant in constants {
871        if !contains_symbol(&equation, *constant) {
872            continue;
873        }
874        let roots = crate::solve::vec::solve(context, &equation, *constant);
875        for root in &roots {
876            if contains_symbol(root, *constant) {
877                continue;
878            }
879            let mut mapping = SubstitutionMap::new();
880            mapping.insert(
881                context.interner.intern(ExpressionKind::Symbol(*constant)),
882                root.clone(),
883            );
884            let next_solution = match family {
885                OrdinaryDifferentialEquationSolutionForm::Explicit(solution)
886                | OrdinaryDifferentialEquationSolutionForm::Implicit(solution) => {
887                    substitute(context, solution, &mapping)
888                }
889            };
890            let next_family = match family {
891                OrdinaryDifferentialEquationSolutionForm::Explicit(_) => {
892                    OrdinaryDifferentialEquationSolutionForm::Explicit(next_solution)
893                }
894                OrdinaryDifferentialEquationSolutionForm::Implicit(_) => {
895                    OrdinaryDifferentialEquationSolutionForm::Implicit(next_solution)
896                }
897            };
898            let next_constants: Vec<SymbolIdentifier> = constants
899                .iter()
900                .copied()
901                .filter(|candidate| candidate != constant)
902                .collect();
903            if let Some(found) = fit_conditions(
904                context,
905                problem,
906                &next_family,
907                remaining_conditions,
908                &next_constants,
909                depth + 1,
910            ) {
911                return Some(found);
912            }
913        }
914    }
915    None
916}

918fn fit_conditions_simultaneously(
919    context: &Context,
920    problem: &OrdinaryDifferentialEquationProblem,
921    family: &OrdinaryDifferentialEquationSolutionForm,
922    conditions: &[OrdinaryDifferentialEquationCondition],
923    constants: &[SymbolIdentifier],
924) -> Option<(
925    OrdinaryDifferentialEquationSolutionForm,
926    Vec<SymbolIdentifier>,
927)> {
928    // Data on a linear family is a linear system, and solving it as one is both exact and
929    // complete where solving it one datum at a time is neither. Eliminating the first constant
930    // through the first datum leaves the second datum with a coefficient — `sin(T)^2/cos(T) +
931    // cos(T)` for a sine-cosine basis fitted away from the origin — that is nonzero but is not
932    // recognized as such term by term, and the sequential search then reports no fit for a system
933    // whose determinant is one. The simultaneous route reads the coefficient matrix off by
934    // differentiation and hands it to the shared exact linear solver, which decides unique,
935    // inconsistent, and under-determined exactly. Anything it declines falls back to the
936    // datum-by-datum search, which is what answers a family that is not linear in its constants.
937    if constants.is_empty() {
938        return None;
939    }
940    if conditions.len() != constants.len() {
941        return None;
942    }
943    let equations: Vec<ExpressionReference> = conditions
944        .iter()
945        .map(|condition| {
946            normalize_condition_equation(
947                context,
948                &condition_equation(context, problem, family, condition),
949            )
950        })
951        .collect();
952    let mut coefficients: Vec<ExpressionReference> =
953        Vec::with_capacity(equations.len() * constants.len());
954    for equation in &equations {
955        for constant in constants {
956            let coefficient =
957                crate::simplify::simplify(context, differentiate(context, equation, *constant));
958            if constants
959                .iter()
960                .any(|other| contains_symbol(&coefficient, *other))
961            {
962                return None;
963            }
964            coefficients.push(coefficient);
965        }
966    }
967    let mut right_hand_side: Vec<ExpressionReference> = Vec::with_capacity(equations.len());
968    for equation in &equations {
969        let mut mapping = SubstitutionMap::new();
970        for constant in constants {
971            mapping.insert(
972                context.interner.intern(ExpressionKind::Symbol(*constant)),
973                builders::zero(context),
974            );
975        }
976        let constant_part =
977            crate::simplify::simplify(context, substitute(context, equation, &mapping));
978        right_hand_side.push(builders::negate(context, constant_part));
979    }
980    let order = constants.len();
981    let coefficient_matrix = crate::matrix::Matrix {
982        rows: order,
983        columns: order,
984        entries: coefficients.clone(),
985    };
986    let determinant = crate::simplify::simplify(
987        context,
988        crate::matrix::matrix_determinant(context, &coefficient_matrix).ok()?,
989    );
990    if is_nonzero(&determinant, &context.symbol_table) != TruthValue::Yes {
991        return None;
992    }
993    let mut mapping = SubstitutionMap::new();
994    for (column, constant) in constants.iter().enumerate() {
995        let mut replaced = coefficients.clone();
996        for row in 0..order {
997            replaced[row * order + column] = right_hand_side[row].clone();
998        }
999        let replaced_matrix = crate::matrix::Matrix {
1000            rows: order,
1001            columns: order,
1002            entries: replaced,
1003        };
1004        let replaced_determinant =
1005            crate::matrix::matrix_determinant(context, &replaced_matrix).ok()?;
1006        let value = crate::simplify::simplify(
1007            context,
1008            builders::divide(context, replaced_determinant, determinant.clone()),
1009        );
1010        mapping.insert(
1011            context.interner.intern(ExpressionKind::Symbol(*constant)),
1012            value,
1013        );
1014    }
1015    for equation in &equations {
1016        let replayed = substitute(context, equation, &mapping);
1017        if prove_residual_zero(context, &replayed) != ZeroEquivalenceVerdict::TrueValue {
1018            return None;
1019        }
1020    }
1021    let fitted = match family {
1022        OrdinaryDifferentialEquationSolutionForm::Explicit(solution) => {
1023            OrdinaryDifferentialEquationSolutionForm::Explicit(crate::simplify::simplify(
1024                context,
1025                substitute(context, solution, &mapping),
1026            ))
1027        }
1028        OrdinaryDifferentialEquationSolutionForm::Implicit(relation) => {
1029            OrdinaryDifferentialEquationSolutionForm::Implicit(crate::simplify::simplify(
1030                context,
1031                substitute(context, relation, &mapping),
1032            ))
1033        }
1034    };
1035    Some((fitted, Vec::new()))
1036}

1038fn normalize_condition_equation(
1039    context: &Context,
1040    equation: &ExpressionReference,
1041) -> ExpressionReference {
1042    // A datum replayed against a family that has already consumed one constant reaches this point
1043    // over a common denominator and with the remaining constant spread across several fractions.
1044    // The equation is linear in that constant either way, but only its combined and expanded form
1045    // lets the exact solver read the coefficient off, so the normalization happens here rather
1046    // than being left to each route's own shape.
1047    let combined = crate::simplify::rational::together(context, equation);
1048    let expanded = crate::expand::expand(context, &combined);
1049    let (numerator, _) =
1050        crate::expression::decomposition::numerator_denominator(context, &expanded);
1051    crate::simplify::simplify(context, crate::expand::expand(context, &numerator))
1052}

1054fn condition_equation(
1055    context: &Context,
1056    problem: &OrdinaryDifferentialEquationProblem,
1057    family: &OrdinaryDifferentialEquationSolutionForm,
1058    condition: &OrdinaryDifferentialEquationCondition,
1059) -> ExpressionReference {
1060    let mut jet = match family {
1061        OrdinaryDifferentialEquationSolutionForm::Explicit(solution) => solution.clone(),
1062        OrdinaryDifferentialEquationSolutionForm::Implicit(relation) => {
1063            return implicit_condition_equation(context, problem, relation, condition);
1064        }
1065    };
1066    for _ in 0..condition.derivative_order {
1067        jet = differentiate(context, &jet, problem.independent_variable());
1068    }
1069    let mut mapping = SubstitutionMap::new();
1070    mapping.insert(
1071        context
1072            .interner
1073            .intern(ExpressionKind::Symbol(problem.independent_variable())),
1074        condition.point.clone(),
1075    );
1076    let at_point = substitute(context, &jet, &mapping);
1077    builders::subtract(context, at_point, condition.value.clone())
1078}

1080fn implicit_condition_equation(
1081    context: &Context,
1082    problem: &OrdinaryDifferentialEquationProblem,
1083    relation: &ExpressionReference,
1084    condition: &OrdinaryDifferentialEquationCondition,
1085) -> ExpressionReference {
1086    let independent_variable = problem.independent_variable();
1087    let dependent_value = context.symbol_table.fresh("implicit_dependent_value");
1088    let dependent_value_expression = context
1089        .interner
1090        .intern(ExpressionKind::Symbol(dependent_value));
1091    let mut relation_mapping = SubstitutionMap::new();
1092    relation_mapping.insert(
1093        problem.dependent_function_call(context),
1094        dependent_value_expression.clone(),
1095    );
1096    let chart_relation = substitute(context, relation, &relation_mapping);
1097    if condition.derivative_order == 0 {
1098        return substitute_pair(
1099            context,
1100            &chart_relation,
1101            independent_variable,
1102            &condition.point,
1103            dependent_value,
1104            &condition.value,
1105        );
1106    }
1107    let relation_coordinate_derivative =
1108        differentiate(context, &chart_relation, independent_variable);
1109    let relation_value_derivative = differentiate(context, &chart_relation, dependent_value);
1110    let first_jet = builders::negate(
1111        context,
1112        builders::divide(
1113            context,
1114            relation_coordinate_derivative,
1115            relation_value_derivative,
1116        ),
1117    );
1118    let mut jet = first_jet.clone();
1119    for _ in 1..condition.derivative_order {
1120        let coordinate_part = differentiate(context, &jet, independent_variable);
1121        let value_part = differentiate(context, &jet, dependent_value);
1122        jet = builders::sum(
1123            context,
1124            vec![
1125                coordinate_part,
1126                builders::product(context, vec![value_part, first_jet.clone()]),
1127            ],
1128        );
1129    }
1130    let value_datum = problem
1131        .conditions()
1132        .iter()
1133        .find(|datum| datum.derivative_order == 0 && datum.point == condition.point);
1134    match value_datum {
1135        Some(value_datum) => builders::subtract(
1136            context,
1137            substitute_pair(
1138                context,
1139                &jet,
1140                independent_variable,
1141                &condition.point,
1142                dependent_value,
1143                &value_datum.value,
1144            ),
1145            condition.value.clone(),
1146        ),
1147        None => context.interner.intern(ExpressionKind::Constant(
1148            crate::expression::BuiltinConstant::Undefined,
1149        )),
1150    }
1151}

1153fn substitute_pair(
1154    context: &Context,
1155    expression: &ExpressionReference,
1156    first_symbol: SymbolIdentifier,
1157    first_value: &ExpressionReference,
1158    second_symbol: SymbolIdentifier,
1159    second_value: &ExpressionReference,
1160) -> ExpressionReference {
1161    let mut mapping = SubstitutionMap::new();
1162    mapping.insert(
1163        context
1164            .interner
1165            .intern(ExpressionKind::Symbol(first_symbol)),
1166        first_value.clone(),
1167    );
1168    mapping.insert(
1169        context
1170            .interner
1171            .intern(ExpressionKind::Symbol(second_symbol)),
1172        second_value.clone(),
1173    );
1174    substitute(context, expression, &mapping)
1175}

1177fn solve_through_laplace_transform(
1178    context: &Context,
1179    problem: &OrdinaryDifferentialEquationProblem,
1180) -> OrdinaryDifferentialEquationSolveResult {
1181    // The transform workflow, run as a workflow. The initial jet is consumed by the derivative
1182    // theorem, the frequency-domain equation is solved for the image of the unknown, and the
1183    // inversion returns to the time domain. Every step must actually have happened: an inversion
1184    // that left a transform placeholder standing is a decline, not a solution written in a
1185    // placeholder, and the chain of intermediates is reported so the workflow can be replayed
1186    // instead of taken on trust.
1187    //
1188    // The one-sided transform reads the jet at the origin. Data stated at another point is
1189    // translated there first, which the constant coefficients permit exactly, and the answer is
1190    // translated back; the forcing travels with the translation.
1191    let method = OrdinaryDifferentialEquationMethod::LaplaceTransform;
1192    let signature = problem.signature();
1193    if signature.arguments.len() != 1 {
1194        return declined(
1195            method,
1196            OrdinaryDifferentialEquationSolveDecline::TransformRequiresScalarSignature,
1197        );
1198    }
1199    let Some(initial_point) = common_condition_point(problem) else {
1200        return declined(
1201            method,
1202            OrdinaryDifferentialEquationSolveDecline::TransformRequiresCompleteOriginData,
1203        );
1204    };
1205    let translated = translate_to_origin(context, problem, &initial_point);
1206    let Some(translated) = translated else {
1207        return declined(
1208            method,
1209            OrdinaryDifferentialEquationSolveDecline::TransformRequiresCompleteOriginData,
1210        );
1211    };
1212    let time_variable = translated.independent_variable();
1213    let origin = builders::zero(context);
1214    let mut origin_data: Vec<ExpressionReference> = Vec::with_capacity(translated.order());
1215    for order in 0..translated.order() {
1216        let Some(condition) = translated
1217            .conditions()
1218            .iter()
1219            .find(|condition| condition.derivative_order == order && condition.point == origin)
1220        else {
1221            return declined(
1222                method,
1223                OrdinaryDifferentialEquationSolveDecline::TransformRequiresCompleteOriginData,
1224            );
1225        };
1226        origin_data.push(condition.value.clone());
1227    }
1228    let frequency_variable = context.symbol_table.fresh("laplace_frequency");
1229    let unknown_image = context.symbol_table.fresh("laplace_unknown_image");
1230    let Some(frequency_domain) = frequency_domain_equation(
1231        context,
1232        &translated,
1233        &origin_data,
1234        time_variable,
1235        frequency_variable,
1236        unknown_image,
1237    ) else {
1238        return declined(
1239            method,
1240            OrdinaryDifferentialEquationSolveDecline::TransformDidNotReachTheFrequencyDomain,
1241        );
1242    };
1243    let Some(image_solution) = frequency_domain.image(context) else {
1244        return declined(
1245            method,
1246            OrdinaryDifferentialEquationSolveDecline::TransformedEquationNotSolvable,
1247        );
1248    };
1249    if contains_symbol(&image_solution, unknown_image) {
1250        return declined(
1251            method,
1252            OrdinaryDifferentialEquationSolveDecline::TransformedEquationNotSolvable,
1253        );
1254    }
1255    let inverted = builders::inverse_laplace_transform(
1256        context,
1257        image_solution.clone(),
1258        frequency_variable,
1259        time_variable,
1260    );
1261    if contains_transform_placeholder(&inverted) {
1262        return declined(
1263            method,
1264            OrdinaryDifferentialEquationSolveDecline::TransformInversionDeclined,
1265        );
1266    }
1267    let restored = translate_from_origin(context, &inverted, time_variable, &initial_point);
1268    let chain = OrdinaryDifferentialEquationTransformChain {
1269        frequency_variable,
1270        initial_point,
1271        frequency_domain_equation: frequency_domain.equation(context),
1272        image: image_solution,
1273        inverted: restored.clone(),
1274    };
1275    let result = finish(
1276        context,
1277        problem,
1278        method,
1279        OrdinaryDifferentialEquationSolutionCandidate::explicit(restored, Vec::new()),
1280    );
1281    OrdinaryDifferentialEquationSolveResult {
1282        transform_chain: Some(chain),
1283        ..result
1284    }
1285}

1287fn common_condition_point(
1288    problem: &OrdinaryDifferentialEquationProblem,
1289) -> Option<ExpressionReference> {
1290    // The one-sided transform consumes one complete initial jet, so the data has to sit at a
1291    // single point and cover every derivative order below the order of the equation. A boundary
1292    // value problem, or a jet with a hole in it, is a different question and declines.
1293    let first = problem.conditions().first()?.point.clone();
1294    if problem.conditions().len() != problem.order() {
1295        return None;
1296    }
1297    if problem
1298        .conditions()
1299        .iter()
1300        .any(|condition| condition.point != first)
1301    {
1302        return None;
1303    }
1304    Some(first)
1305}

1307fn translate_to_origin(
1308    context: &Context,
1309    problem: &OrdinaryDifferentialEquationProblem,
1310    initial_point: &ExpressionReference,
1311) -> Option<OrdinaryDifferentialEquationProblem> {
1312    // Move the initial point to the origin by `t = tau + t0`. The coefficients of a
1313    // constant-coefficient operator are unchanged by the translation and the forcing travels with
1314    // it, so the translated problem is the same problem read in a shifted coordinate.
1315    if matches!(&initial_point.kind, ExpressionKind::Integer(value) if value.to_string() == "0") {
1316        return Some(problem.clone());
1317    }
1318    let time_variable = problem.independent_variable();
1319    let shifted = builders::sum(
1320        context,
1321        vec![
1322            context
1323                .interner
1324                .intern(ExpressionKind::Symbol(time_variable)),
1325            initial_point.clone(),
1326        ],
1327    );
1328    let mut mapping: SubstitutionMap = SubstitutionMap::new();
1329    mapping.insert(
1330        context
1331            .interner
1332            .intern(ExpressionKind::Symbol(time_variable)),
1333        shifted,
1334    );
1335    let equation = substitute(context, problem.equation(), &mapping);
1336    let origin = builders::zero(context);
1337    let conditions = problem
1338        .conditions()
1339        .iter()
1340        .map(|condition| {
1341            OrdinaryDifferentialEquationCondition::new(
1342                condition.derivative_order,
1343                origin.clone(),
1344                condition.value.clone(),
1345            )
1346        })
1347        .collect();
1348    OrdinaryDifferentialEquationProblem::new(equation, problem.signature().clone(), conditions).ok()
1349}
1350
1351fn translate_from_origin(
1352    context: &Context,
1353    solution: &ExpressionReference,
1354    time_variable: SymbolIdentifier,
1355    initial_point: &ExpressionReference,
1356) -> ExpressionReference {
1357    if matches!(&initial_point.kind, ExpressionKind::Integer(value) if value.to_string() == "0") {
1358        return solution.clone();
1359    }
1360    let coordinate = context
1361        .interner
1362        .intern(ExpressionKind::Symbol(time_variable));
1363    let restored = builders::subtract(context, coordinate.clone(), initial_point.clone());
1364    let mut mapping: SubstitutionMap = SubstitutionMap::new();
1365    mapping.insert(coordinate, restored);
1366    substitute(context, solution, &mapping)
1367}

1369/// The frequency-domain form of a transformed linear equation, kept as the pair it actually is.
1370struct FrequencyDomainEquation {
1371    image_coefficient: ExpressionReference,
1372    remainder: ExpressionReference,
1373    image_symbol: ExpressionReference,
1374}
1375
1376impl FrequencyDomainEquation {
1377    fn equation(&self, context: &Context) -> ExpressionReference {
1378        builders::sum(
1379            context,
1380            vec![
1381                builders::product(
1382                    context,
1383                    vec![self.image_coefficient.clone(), self.image_symbol.clone()],
1384                ),
1385                self.remainder.clone(),
1386            ],
1387        )
1388    }
1389
1390    fn image(&self, context: &Context) -> Option<ExpressionReference> {
1391        // The transformed equation is linear in the image by construction — the derivative
1392        // theorem is linear and the coefficients do not depend on the coordinate — so the image is
1393        // an exact division rather than a search for roots. The divisor is the characteristic
1394        // polynomial in the frequency variable, which the operator's proved order keeps from
1395        // being the zero polynomial; the transform is claimed only where that polynomial does not
1396        // vanish, which is exactly the right half plane the one-sided transform converges in.
1397        if is_structural_zero(&self.image_coefficient) {
1398            return None;
1399        }
1400        Some(crate::simplify::rational::presentation_normal_form(
1401            context,
1402            &builders::divide(
1403                context,
1404                builders::negate(context, self.remainder.clone()),
1405                self.image_coefficient.clone(),
1406            ),
1407        ))
1408    }
1409}
1410
1411fn is_structural_zero(target: &ExpressionReference) -> bool {
1412    matches!(&target.kind, ExpressionKind::Integer(value) if value.to_string() == "0")
1413}
1414
1415fn frequency_domain_equation(
1416    context: &Context,
1417    problem: &OrdinaryDifferentialEquationProblem,
1418    origin_data: &[ExpressionReference],
1419    time_variable: SymbolIdentifier,
1420    frequency_variable: SymbolIdentifier,
1421    unknown_image: SymbolIdentifier,
1422) -> Option<FrequencyDomainEquation> {
1423    // Linearity applied term by term, because the forward transform declines a whole sum whose
1424    // unknown-function term it cannot close. Every summand is either a coefficient times one
1425    // top-level occurrence of the unknown — which the derivative theorem turns into the image and
1426    // the initial jet — or is free of the unknown and must transform on its own. A summand that
1427    // hides the unknown inside an operator, an integral for instance, is neither, and the
1428    // workflow declines rather than transforming a term it has no rule for.
1429    //
1430    // The image coefficient and the rest are kept apart rather than summed into one equation, so
1431    // the caller inverts a linear relation it knows to be linear instead of handing a rational
1432    // equation to a general root finder.
1433    let signature = problem.signature();
1434    let summands: Vec<ExpressionReference> = match &problem.equation().kind {
1435        ExpressionKind::Sum(children) => children.clone(),
1436        _ => vec![problem.equation().clone()],
1437    };
1438    let frequency = context
1439        .interner
1440        .intern(ExpressionKind::Symbol(frequency_variable));
1441    let image = context
1442        .interner
1443        .intern(ExpressionKind::Symbol(unknown_image));
1444    let mut image_coefficients = Vec::new();
1445    let mut remainder_terms = Vec::new();
1446    for summand in summands {
1447        let mut occurrences = Vec::new();
1448        crate::differential_equations::problem::collect_dependent_occurrences(
1449            &summand,
1450            signature,
1451            &mut occurrences,
1452        );
1453        if occurrences.is_empty() {
1454            let forward =
1455                builders::laplace_transform(context, summand, time_variable, frequency_variable);
1456            if contains_transform_placeholder(&forward) {
1457                return None;
1458            }
1459            remainder_terms.push(forward);
1460            continue;
1461        }
1462        if occurrences.len() != 1 {
1463            return None;
1464        }
1465        let occurrence = occurrences[0].clone();
1466        if !is_top_level_factor(&summand, &occurrence) {
1467            return None;
1468        }
1469        let order = match &occurrence.kind {
1470            ExpressionKind::FunctionCall(_, _) => 0usize,
1471            ExpressionKind::Derivative(_, orders, _) => *orders.first()?,
1472            _ => return None,
1473        };
1474        let coefficient = factor_beside(context, &summand, &occurrence)?;
1475        if contains_symbol(&coefficient, time_variable) {
1476            return None;
1477        }
1478        let order_power = builders::integer_value(context, order as i64);
1479        image_coefficients.push(builders::product(
1480            context,
1481            vec![
1482                coefficient.clone(),
1483                builders::power(context, frequency.clone(), order_power),
1484            ],
1485        ));
1486        for datum_order in 0..order {
1487            let exponent = order - 1 - datum_order;
1488            let factor = builders::power(
1489                context,
1490                frequency.clone(),
1491                builders::integer_value(context, exponent as i64),
1492            );
1493            let datum = origin_data.get(datum_order)?.clone();
1494            remainder_terms.push(builders::negate(
1495                context,
1496                builders::product(context, vec![coefficient.clone(), factor, datum]),
1497            ));
1498        }
1499    }
1500    let image_coefficient =
1501        crate::simplify::simplify(context, builders::sum(context, image_coefficients));
1502    let remainder = crate::simplify::simplify(context, builders::sum(context, remainder_terms));
1503    if mentions_dependent_function(&image_coefficient, signature.dependent_function)
1504        || mentions_dependent_function(&remainder, signature.dependent_function)
1505    {
1506        return None;
1507    }
1508    Some(FrequencyDomainEquation {
1509        image_coefficient,
1510        remainder,
1511        image_symbol: image,
1512    })
1513}

1515fn is_top_level_factor(summand: &ExpressionReference, occurrence: &ExpressionReference) -> bool {
1516    if summand == occurrence {
1517        return true;
1518    }
1519    match &summand.kind {
1520        ExpressionKind::Product(factors) => factors.iter().any(|factor| factor == occurrence),
1521        _ => false,
1522    }
1523}
1524
1525fn factor_beside(
1526    context: &Context,
1527    summand: &ExpressionReference,
1528    occurrence: &ExpressionReference,
1529) -> Option<ExpressionReference> {
1530    if summand == occurrence {
1531        return Some(builders::one(context));
1532    }
1533    let ExpressionKind::Product(factors) = &summand.kind else {
1534        return None;
1535    };
1536    let mut remaining = Vec::with_capacity(factors.len());
1537    let mut removed = false;
1538    for factor in factors {
1539        if !removed && factor == occurrence {
1540            removed = true;
1541            continue;
1542        }
1543        remaining.push(factor.clone());
1544    }
1545    removed.then(|| builders::product(context, remaining))
1546}

1628        collect_symbols_created_since(child, baseline, found);
