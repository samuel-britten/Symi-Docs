1// Separable classifier and integrator. Recognises `y'(x) = f(x) *
2// g(y)` and integrates each side via the full integration pipeline
3// (builtin-table, rational-function, and Risch stages):
4//   ∫ 1/g(y) dy = ∫ f(x) dx + C
5// The constant of integration is materialised as a fresh symbol. When
6// `solve` recovers a closed form for the auxiliary symbol that stands in
7// for y(x), the explicit form is back-substituted and returned; otherwise
8// the implicit form (the LHS - RHS = 0 expression) is returned so the
9// caller can still compose against it.
10

28/// Performs the advanced native `solve` operation.
29pub fn solve(
30    context: &Context,
31    equation: &ExpressionReference,
32    dependent_variable_call: &ExpressionReference,
33    independent_variable: SymbolIdentifier,
34) -> Option<ExpressionReference> {
35    // Expression-only projection of the tracked entry point. The generated
36    // integration constant is discarded here because the historical public
37    // ordinary-differential-equation surface returns only an expression.
38    let mut generated_constants = Vec::new();
39    solve_tracked(
40        context,
41        equation,
42        dependent_variable_call,
43        independent_variable,
44        &mut generated_constants,
45    )
46    .map(|closed_form| closed_form.solution)
47}

49pub(crate) fn solve_tracked(
50    context: &Context,
51    equation: &ExpressionReference,
52    dependent_variable_call: &ExpressionReference,
53    independent_variable: SymbolIdentifier,
54    generated_constants: &mut Vec<SymbolIdentifier>,
55) -> Option<OrdinaryDifferentialEquationRouteSolution> {
56    // Eager entry. Attempts to isolate y'(x) and factor the right-hand
57    // side into a product whose factors partition cleanly into "depends
58    // on x alone" and "depends on y alone" pieces. When that split
59    // succeeds, the antiderivatives on each side close the integration
60    // and N's solve recovers an explicit y(x). Mixed factors abort to
61    // None so the outer dispatcher tries the next classifier.
62    let function = dependent_function_identifier(dependent_variable_call)?;
63    let right_hand_side =
64        solve_for_first_derivative(context, equation, function, independent_variable)?;
65    let auxiliary_variable = context.symbol_table.fresh("y_separable");
66    let auxiliary_symbol = context
67        .interner
68        .intern(ExpressionKind::Symbol(auxiliary_variable));
69    let y_call = derivative_call(context, function, independent_variable, 0);
70    let mut y_to_auxiliary: SubstitutionMap = HashMap::new();
71    y_to_auxiliary.insert(Arc::clone(&y_call), Arc::clone(&auxiliary_symbol));
72    let right_hand_side_in_auxiliary = substitute(context, &right_hand_side, &y_to_auxiliary);
73    if crate::expression::queries::depends_on(&right_hand_side_in_auxiliary, auxiliary_variable)
74        && crate::differential_equations::shape::depends_on_function(
75            &right_hand_side_in_auxiliary,
76            function,
77        )
78    {
79        return None;
80    }
81    let (x_factor, y_factor) = partition_separable_factors_after_surrogate(
82        context,
83        &right_hand_side_in_auxiliary,
84        auxiliary_variable,
85        independent_variable,
86    )?;
87    let y_factor_in_auxiliary = y_factor;
88    let one_value = builders::one(context);
89    let reciprocal_y_factor =
90        builders::divide(context, one_value, Arc::clone(&y_factor_in_auxiliary));
91    let y_side_antiderivative = integrate(context, &reciprocal_y_factor, auxiliary_variable);
92    if contains_unevaluated_integral(&y_side_antiderivative) {
93        return None;
94    }
95    let x_side_antiderivative = integrate(context, &x_factor, independent_variable);
96    if contains_unevaluated_integral(&x_side_antiderivative) {
97        return None;
98    }
99    let integration_constant_identifier = context.symbol_table.fresh("integration_constant");
100    generated_constants.push(integration_constant_identifier);
101    let integration_constant = context
102        .interner
103        .intern(ExpressionKind::Symbol(integration_constant_identifier));
104    let right_side_with_constant = builders::sum(
105        context,
106        vec![x_side_antiderivative, Arc::clone(&integration_constant)],
107    );
108    let implicit_in_auxiliary = builders::subtract(
109        context,
110        Arc::clone(&y_side_antiderivative),
111        Arc::clone(&right_side_with_constant),
112    );
113    let auxiliary_solutions =
114        crate::solve::solve(context, &implicit_in_auxiliary, auxiliary_variable);
115    if auxiliary_solutions.len() == 1 {
116        let candidate = Arc::clone(&auxiliary_solutions[0]);
117        if !crate::differential_equations::shape::contains_reference(&candidate, &auxiliary_symbol)
118        {
119            return Some(OrdinaryDifferentialEquationRouteSolution {
120                solution: candidate,
121                form: OrdinaryDifferentialEquationSolutionForm::Explicit,
122                generated_spectator_functions: Vec::new(),
123                domain_obligations: Vec::new(),
124            });
125        }
126    }
127    let mut auxiliary_to_y: SubstitutionMap = HashMap::new();
128    auxiliary_to_y.insert(auxiliary_symbol, y_call);
129    Some(OrdinaryDifferentialEquationRouteSolution {
130        solution: substitute(context, &implicit_in_auxiliary, &auxiliary_to_y),
131        form: OrdinaryDifferentialEquationSolutionForm::Implicit,
132        generated_spectator_functions: Vec::new(),
133        domain_obligations: Vec::new(),
134    })
135}

137fn partition_separable_factors_after_surrogate(
138    context: &Context,
139    right_hand_side: &ExpressionReference,
140    auxiliary_variable: SymbolIdentifier,
141    independent_variable: SymbolIdentifier,
142) -> Option<(ExpressionReference, ExpressionReference)> {
143    // Variant of partition_separable_factors that operates after the
144    // surrogate substitution y(x) → auxiliary_symbol. With the surrogate
145    // in place, "depends on y" is just depends_on(auxiliary_variable),
146    // which lets the partition correctly classify factors whose y-
147    // dependence is mediated through the function call (a case the
148    // original function-keyed check rejected because y(x) structurally
149    // mentions x as its argument).
150    use crate::expression::queries::depends_on;
151    let factors: Vec<ExpressionReference> = match &right_hand_side.kind {
152        ExpressionKind::Product(children) => children.iter().map(Arc::clone).collect(),
153        _ => vec![Arc::clone(right_hand_side)],
154    };
155    let mut x_factors: Vec<ExpressionReference> = Vec::new();
156    let mut y_factors: Vec<ExpressionReference> = Vec::new();
157    for factor in factors {
158        let depends_on_x = depends_on(&factor, independent_variable);
159        let depends_on_y = depends_on(&factor, auxiliary_variable);
160        match (depends_on_x, depends_on_y) {
161            (false, false) => x_factors.push(factor),
162            (true, false) => x_factors.push(factor),
163            (false, true) => y_factors.push(factor),
164            (true, true) => {
165                // A mixed factor may still be separable when it is a power of a
166                // polynomial that factors into single-variable pieces — the
167                // rational normal form merges (1 - y)^(-1) * x^(-1) into
168                // (x - x*y)^(-1), so the merged denominator must be re-split
169                // here rather than aborting the classification.
170                let split = split_mixed_power_factor(
171                    context,
172                    &factor,
173                    auxiliary_variable,
174                    independent_variable,
175                )?;
176                for (piece, is_y_piece) in split {
177                    if is_y_piece {
178                        y_factors.push(piece);
179                    } else {
180                        x_factors.push(piece);
181                    }
182                }
183            }
184        }
185    }
186    let x_factor = if x_factors.is_empty() {
187        builders::one(context)
188    } else if x_factors.len() == 1 {
189        x_factors.into_iter().next().unwrap()
190    } else {
191        builders::product(context, x_factors)
192    };
193    let y_factor = if y_factors.is_empty() {
194        builders::one(context)
195    } else if y_factors.len() == 1 {
196        y_factors.into_iter().next().unwrap()
197    } else {
198        builders::product(context, y_factors)
199    };
200    Some((x_factor, y_factor))
201}

203fn split_mixed_power_factor(
204    context: &Context,
205    factor: &ExpressionReference,
206    auxiliary_variable: SymbolIdentifier,
207    independent_variable: SymbolIdentifier,
208) -> Option<Vec<(ExpressionReference, bool)>> {
209    // Attempt to separate a mixed factor of the shape power(base, n) by
210    // factoring the base over Q. Succeeds only when every irreducible piece of
211    // the base depends on at most one of the two variables, in which case each
212    // piece is returned raised to the original exponent, tagged with whether it
213    // belongs on the y side. Any genuinely mixed irreducible piece aborts.
214    use crate::expression::queries::depends_on;
215    let ExpressionKind::Power(base, exponent) = &factor.kind else {
216        return None;
217    };
218    if !matches!(&exponent.kind, ExpressionKind::Integer(_)) {
219        return None;
220    }
221    let factored_base = crate::factor::factor(context, base, None);
222    let base_pieces: Vec<ExpressionReference> = match &factored_base.kind {
223        ExpressionKind::Product(children) => children.iter().map(Arc::clone).collect(),
224        _ => return None,
225    };
226    let mut split: Vec<(ExpressionReference, bool)> = Vec::new();
227    for piece in base_pieces {
228        let piece_on_x = depends_on(&piece, independent_variable);
229        let piece_on_y = depends_on(&piece, auxiliary_variable);
230        if piece_on_x && piece_on_y {
231            return None;
232        }
233        let raised = builders::power(context, piece, Arc::clone(exponent));
234        split.push((raised, piece_on_y));
235    }
236    Some(split)
237}

239fn contains_unevaluated_integral(target: &ExpressionReference) -> bool {
240    // Probe the result of `integrate` for an embedded `integral(...)`
241    // placeholder. Used to detect that the antiderivative stack stalled
242    // partway through the integration, in which case the separable
243    // classifier refuses rather than returning a result that still
244    // contains an integral node.
245    match &target.kind {
246        ExpressionKind::Integer(_)
247        | ExpressionKind::Rational(_)
248        | ExpressionKind::Symbol(_)
249        | ExpressionKind::Constant(_) => false,
250        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
251            children.iter().any(contains_unevaluated_integral)
252        }
253        ExpressionKind::Power(base, exponent) => {
254            contains_unevaluated_integral(base) || contains_unevaluated_integral(exponent)
255        }
256        ExpressionKind::Factorial(argument) => contains_unevaluated_integral(argument),
257        ExpressionKind::FunctionCall(_, arguments)
258        | ExpressionKind::Derivative(_, _, arguments) => {
259            arguments.iter().any(contains_unevaluated_integral)
260        }
261        ExpressionKind::Integral(_, _, _, _) => true,
262        ExpressionKind::Summation(summand, _, lower, upper) => {
263            contains_unevaluated_integral(summand)
264                || lower.as_ref().is_some_and(contains_unevaluated_integral)
265                || upper.as_ref().is_some_and(contains_unevaluated_integral)
266        }
267        ExpressionKind::LaplaceTransform(target_expression, _, _)
268        | ExpressionKind::InverseLaplaceTransform(target_expression, _, _)
269        | ExpressionKind::FourierTransform(target_expression, _, _)
270        | ExpressionKind::InverseFourierTransform(target_expression, _, _) => {
271            contains_unevaluated_integral(target_expression)
272        }
273        ExpressionKind::Piecewise(branches) => branches.iter().any(|(condition, value)| {
274            contains_unevaluated_integral(condition) || contains_unevaluated_integral(value)
275        }),
276        ExpressionKind::OrdinaryDifferentialEquation(equation, dependent_variable_call, _)
277        | ExpressionKind::RecurrenceEquation(equation, dependent_variable_call, _) => {
278            contains_unevaluated_integral(equation)
279                || contains_unevaluated_integral(dependent_variable_call)
280        }
281        _ => false,
282    }
