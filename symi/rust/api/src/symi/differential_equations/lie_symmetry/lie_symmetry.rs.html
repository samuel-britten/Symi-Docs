1// Lie-symmetry first-order classifier. Recognises four patterns from
2// the Cheb-Terrab–Roche (1999) catalogue whose generators have constant
3// coefficients (or, for the quasi-homogeneous case, a single rational
4// exponent):
5//
6//   homogeneous          y' = F(y/x)                     X = x ∂_x + y ∂_y
7//   translation          y' = F(y - k x)                 X = ∂_x + k ∂_y
8//   linear-fractional    y' = (a₁x+b₁y+c₁)/(a₂x+b₂y+c₂)  reduces to either
9//                                                         homogeneous (lines
10//                                                         intersect) or
11//                                                         translation (lines
12//                                                         parallel)
13//   quasi-homogeneous    y' = x^(α-1) F(y/x^α)           X = x ∂_x + α y ∂_y
14//
15// Each pattern's canonical-coordinate transformation reduces the original
16// ODE to a separable form in a fresh auxiliary function u(x); that ODE is
17// then routed through [separable::solve]. The result is back-substituted
18// to express the closed form in terms of the original y(x). On any
19// detection failure or downstream integration stall the classifier returns
20// None so the dispatcher in [solve_ordinary_differential_equation] falls
21// through to the next arm (and ultimately to the unsolved-ODE AST
22// placeholder).
23//
24// Patterns not yet implemented: rotation
25// (`X = -y ∂_x + x ∂_y`), composite patterns with function-valued
26// coefficients (Cheb-Terrab–Kolokolnikov 2003), and the general
27// `F((linear)/(linear))` linear-fractional detection beyond the literal-
28// rational sub-case.

48/// Performs the advanced native `solve` operation.
49pub fn solve(
50    context: &Context,
51    equation: &ExpressionReference,
52    dependent_variable_call: &ExpressionReference,
53    independent_variable: SymbolIdentifier,
54) -> Option<ExpressionReference> {
55    // Dispatcher. Solves the equation for y'(x), substitutes a fresh
56    // surrogate for y(x), and tries each pattern in order: homogeneous
57    // (cheapest specialised test), translation (different shape), linear-
58    // fractional (composite — reduces to homogeneous or translation),
59    // quasi-homogeneous (general scaling — fires when α ≠ 1 and the
60    // determining ratio is constant).
61    let function = dependent_function_identifier(dependent_variable_call)?;
62    let right_hand_side =
63        solve_for_first_derivative(context, equation, function, independent_variable)?;
64    let y_surrogate_identifier = context.symbol_table.fresh("y_lie_symmetry");
65    let y_surrogate = context
66        .interner
67        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
68    let y_call = derivative_call(context, function, independent_variable, 0);
69    let mut y_to_surrogate: SubstitutionMap = HashMap::new();
70    y_to_surrogate.insert(Arc::clone(&y_call), Arc::clone(&y_surrogate));
71    let f_in_surrogate = substitute(context, &right_hand_side, &y_to_surrogate);
72    if depends_on_function(&f_in_surrogate, function) {
73        return None;
74    }
75    if let Some(solution) = try_homogeneous(
76        context,
77        &f_in_surrogate,
78        y_surrogate_identifier,
79        function,
80        independent_variable,
81    ) {
82        return Some(solution);
83    }
84    if let Some(solution) = try_translation(
85        context,
86        &f_in_surrogate,
87        y_surrogate_identifier,
88        function,
89        independent_variable,
90    ) {
91        return Some(solution);
92    }
93    if let Some(solution) = try_linear_fractional(
94        context,
95        &f_in_surrogate,
96        y_surrogate_identifier,
97        function,
98        independent_variable,
99    ) {
100        return Some(solution);
101    }
102    if let Some(solution) = try_quasi_homogeneous(
103        context,
104        &f_in_surrogate,
105        y_surrogate_identifier,
106        function,
107        independent_variable,
108    ) {
109        return Some(solution);
110    }
111    None
112}

114fn try_homogeneous(
115    context: &Context,
116    f_in_surrogate: &ExpressionReference,
117    y_surrogate_identifier: SymbolIdentifier,
118    function: FunctionIdentifier,
119    independent_variable: SymbolIdentifier,
120) -> Option<ExpressionReference> {
121    // Pattern y' = F(y/x), generator X = x ∂_x + y ∂_y. Detect by the
122    // Euler-homogeneity identity x · f_x + y · f_y = 0 (true iff f is
123    // degree-0 homogeneous in (x, y)). The detection is robust to the
124    // structural form of f (polynomial, sum-of-monomials, rational
125    // function) — for a rational f = N/D, the residual reduces to the
126    // determinantal identity (x N_x + y N_y)·D − (x D_x + y D_y)·N, which
127    // is zero exactly when N and D are homogeneous of the same degree.
128    // Once detected, F(t) is recovered by substituting (x, y_surrogate)
129    // → (1, t); the canonical separable equation u'(x) = (F(u(x)) − u(x))/x
130    // is built in a fresh auxiliary function u, routed through separable::
131    // solve, and back-substituted u(x) → y(x)/x to recover y.
132    let x_symbol = context
133        .interner
134        .intern(ExpressionKind::Symbol(independent_variable));
135    let y_surrogate = context
136        .interner
137        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
138    if !satisfies_homogeneity_identity(
139        context,
140        f_in_surrogate,
141        &x_symbol,
142        &y_surrogate,
143        independent_variable,
144        y_surrogate_identifier,
145    ) {
146        return None;
147    }
148    let t_identifier = context.symbol_table.fresh("t_homogeneous_canonical");
149    let t_symbol = context
150        .interner
151        .intern(ExpressionKind::Symbol(t_identifier));
152    let one = builders::one(context);
153    let mut canonical_substitution: SubstitutionMap = HashMap::new();
154    canonical_substitution.insert(Arc::clone(&x_symbol), one);
155    canonical_substitution.insert(Arc::clone(&y_surrogate), Arc::clone(&t_symbol));
156    let f_in_t = substitute(context, f_in_surrogate, &canonical_substitution);
157    if depends_on(&f_in_t, independent_variable) || depends_on(&f_in_t, y_surrogate_identifier) {
158        return None;
159    }
160    let f_in_t_minus_t = builders::subtract(context, f_in_t, Arc::clone(&t_symbol));
161    let f_in_t_minus_t_normalized =
162        normalize_rational_in_variable(context, &f_in_t_minus_t, t_identifier)
163            .unwrap_or(f_in_t_minus_t);
164    let auxiliary_function_identifier = context.symbol_table.fresh("u_lie_homogeneous");
165    let auxiliary_function = FunctionIdentifier::User(auxiliary_function_identifier);
166    let u_call = builders::function_call(context, auxiliary_function, vec![Arc::clone(&x_symbol)]);
167    let u_prime_call = builders::derivative(
168        context,
169        auxiliary_function,
170        vec![1],
171        vec![Arc::clone(&x_symbol)],
172    );
173    let mut t_to_u: SubstitutionMap = HashMap::new();
174    t_to_u.insert(Arc::clone(&t_symbol), Arc::clone(&u_call));
175    let f_of_u_minus_u = substitute(context, &f_in_t_minus_t_normalized, &t_to_u);
176    let rhs = builders::divide(context, f_of_u_minus_u, Arc::clone(&x_symbol));
177    let auxiliary_equation = builders::subtract(context, u_prime_call, rhs);
178    let u_solution = separable::solve(context, &auxiliary_equation, &u_call, independent_variable)?;
179    let y_call = derivative_call(context, function, independent_variable, 0);
180    if !contains_reference(&u_solution, &u_call) {
181        let y_explicit = builders::product(context, vec![u_solution, Arc::clone(&x_symbol)]);
182        return Some(y_explicit);
183    }
184    let y_over_x = builders::divide(context, Arc::clone(&y_call), Arc::clone(&x_symbol));
185    let mut back_substitution: SubstitutionMap = HashMap::new();
186    back_substitution.insert(u_call, y_over_x);
187    Some(substitute(context, &u_solution, &back_substitution))
188}

190fn try_translation(
191    context: &Context,
192    f_in_surrogate: &ExpressionReference,
193    y_surrogate_identifier: SymbolIdentifier,
194    function: FunctionIdentifier,
195    independent_variable: SymbolIdentifier,
196) -> Option<ExpressionReference> {
197    // Pattern y' = F(y − k x), generator X = ∂_x + k ∂_y. The translation
198    // slope k satisfies f_x + k · f_y = 0, so k = −f_x/f_y provided
199    // f_y ≢ 0. Confirm k is constant in (x, y_surrogate), then verify
200    // structurally by checking that f(x, k·x + u_probe) is independent
201    // of x. Build the autonomous separable ODE u'(x) = F(u) − k in a
202    // fresh auxiliary function u, route through separable::solve, and
203    // back-substitute u(x) → y(x) − k·x.
204    let f_y = differentiate(context, f_in_surrogate, y_surrogate_identifier);
205    if is_integer_zero(&f_y) {
206        return None;
207    }
208    let f_x = differentiate(context, f_in_surrogate, independent_variable);
209    let negated_f_x = builders::negate(context, f_x);
210    let k_candidate = builders::divide(context, negated_f_x, Arc::clone(&f_y));
211    if depends_on(&k_candidate, independent_variable)
212        || depends_on(&k_candidate, y_surrogate_identifier)
213    {
214        return None;
215    }
216    let k = k_candidate;
217    let x_symbol = context
218        .interner
219        .intern(ExpressionKind::Symbol(independent_variable));
220    let y_surrogate = context
221        .interner
222        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
223    let u_probe_identifier = context.symbol_table.fresh("u_translation_probe");
224    let u_probe_symbol = context
225        .interner
226        .intern(ExpressionKind::Symbol(u_probe_identifier));
227    let k_times_x = builders::product(context, vec![Arc::clone(&k), Arc::clone(&x_symbol)]);
228    let k_times_x_plus_u = builders::sum(context, vec![k_times_x, Arc::clone(&u_probe_symbol)]);
229    let mut probe_substitution: SubstitutionMap = HashMap::new();
230    probe_substitution.insert(Arc::clone(&y_surrogate), k_times_x_plus_u);
231    let f_in_u_probe = substitute(context, f_in_surrogate, &probe_substitution);
232    if depends_on(&f_in_u_probe, independent_variable) {
233        return None;
234    }
235    let auxiliary_function_identifier = context.symbol_table.fresh("u_lie_translation");
236    let auxiliary_function = FunctionIdentifier::User(auxiliary_function_identifier);
237    let u_call = builders::function_call(context, auxiliary_function, vec![Arc::clone(&x_symbol)]);
238    let u_prime_call = builders::derivative(
239        context,
240        auxiliary_function,
241        vec![1],
242        vec![Arc::clone(&x_symbol)],
243    );
244    let mut probe_to_u: SubstitutionMap = HashMap::new();
245    probe_to_u.insert(Arc::clone(&u_probe_symbol), Arc::clone(&u_call));
246    let f_of_u = substitute(context, &f_in_u_probe, &probe_to_u);
247    let f_of_u_minus_k = builders::subtract(context, f_of_u, Arc::clone(&k));
248    let auxiliary_equation = builders::subtract(context, u_prime_call, f_of_u_minus_k);
249    let u_solution = separable::solve(context, &auxiliary_equation, &u_call, independent_variable)?;
250    let y_call = derivative_call(context, function, independent_variable, 0);
251    let k_times_x_back = builders::product(context, vec![Arc::clone(&k), Arc::clone(&x_symbol)]);
252    if !contains_reference(&u_solution, &u_call) {
253        let y_explicit = builders::sum(context, vec![u_solution, k_times_x_back]);
254        return Some(y_explicit);
255    }
256    let negative_k_times_x = builders::negate(context, k_times_x_back);
257    let y_minus_kx = builders::sum(context, vec![Arc::clone(&y_call), negative_k_times_x]);
258    let mut back_substitution: SubstitutionMap = HashMap::new();
259    back_substitution.insert(u_call, y_minus_kx);
260    Some(substitute(context, &u_solution, &back_substitution))
261}

263fn try_quasi_homogeneous(
264    context: &Context,
265    f_in_surrogate: &ExpressionReference,
266    y_surrogate_identifier: SymbolIdentifier,
267    function: FunctionIdentifier,
268    independent_variable: SymbolIdentifier,
269) -> Option<ExpressionReference> {
270    // Pattern y' = x^(α−1) F(y/x^α), generator X = x ∂_x + α y ∂_y. The
271    // determining equation x f_x + α y f_y = (α−1) f rearranges to
272    //   α = −(x f_x + f) / (y f_y − f)
273    // when the denominator is non-zero. Confirm α is constant in
274    // (x, y_surrogate), then verify structurally by substituting
275    // y_surrogate → t · x^α and dividing by x^(α−1) — the residual must
276    // be independent of (x, y_surrogate). Build the canonical separable
277    // ODE u'(x) = (G(u) − α u)/x where G(u) is the residual, route
278    // through separable::solve, and back-substitute u(x) → y(x)/x^α.
279    //
280    // When α = 1 this overlaps the homogeneous classifier, so it fires
281    // only on inputs the homogeneous arm refused (which is enforced by
282    // dispatch order in `solve`).
283    let x_symbol = context
284        .interner
285        .intern(ExpressionKind::Symbol(independent_variable));
286    let y_surrogate = context
287        .interner
288        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
289    let f_x = differentiate(context, f_in_surrogate, independent_variable);
290    let f_y = differentiate(context, f_in_surrogate, y_surrogate_identifier);
291    let x_times_f_x = builders::product(context, vec![Arc::clone(&x_symbol), f_x]);
292    let x_f_x_plus_f = builders::sum(context, vec![x_times_f_x, Arc::clone(f_in_surrogate)]);
293    let y_times_f_y = builders::product(context, vec![Arc::clone(&y_surrogate), f_y]);
294    let y_f_y_minus_f = builders::subtract(context, y_times_f_y, Arc::clone(f_in_surrogate));
295    if is_integer_zero(&y_f_y_minus_f) {
296        return None;
297    }
298    let alpha = determine_quasi_homogeneity_exponent(
299        context,
300        f_in_surrogate,
301        &x_f_x_plus_f,
302        &y_f_y_minus_f,
303        &x_symbol,
304        &y_surrogate,
305        independent_variable,
306        y_surrogate_identifier,
307    )?;
308    let t_identifier = context.symbol_table.fresh("t_quasi_homogeneous_probe");
309    let t_symbol = context
310        .interner
311        .intern(ExpressionKind::Symbol(t_identifier));
312    let x_to_alpha = builders::power(context, Arc::clone(&x_symbol), Arc::clone(&alpha));
313    let t_times_x_alpha = builders::product(
314        context,
315        vec![Arc::clone(&t_symbol), Arc::clone(&x_to_alpha)],
316    );
317    let mut probe_substitution: SubstitutionMap = HashMap::new();
318    probe_substitution.insert(Arc::clone(&y_surrogate), t_times_x_alpha);
319    let f_in_t = substitute(context, f_in_surrogate, &probe_substitution);
320    let one = builders::one(context);
321    let alpha_minus_one = builders::subtract(context, Arc::clone(&alpha), one);
322    let x_to_alpha_minus_one = builders::power(context, Arc::clone(&x_symbol), alpha_minus_one);
323    let g_of_t_unsimplified = builders::divide(context, f_in_t, x_to_alpha_minus_one);
324    let g_of_t = crate::expand::expand(context, &g_of_t_unsimplified);
325    if depends_on(&g_of_t, independent_variable) || depends_on(&g_of_t, y_surrogate_identifier) {
326        return None;
327    }
328    let auxiliary_function_identifier = context.symbol_table.fresh("u_lie_quasi_homogeneous");
329    let auxiliary_function = FunctionIdentifier::User(auxiliary_function_identifier);
330    let u_call = builders::function_call(context, auxiliary_function, vec![Arc::clone(&x_symbol)]);
331    let u_prime_call = builders::derivative(
332        context,
333        auxiliary_function,
334        vec![1],
335        vec![Arc::clone(&x_symbol)],
336    );
337    let alpha_times_t = builders::product(context, vec![Arc::clone(&alpha), Arc::clone(&t_symbol)]);
338    let g_of_t_minus_alpha_t = builders::subtract(context, g_of_t, alpha_times_t);
339    let g_of_t_minus_alpha_t_normalized =
340        normalize_rational_in_variable(context, &g_of_t_minus_alpha_t, t_identifier)
341            .unwrap_or(g_of_t_minus_alpha_t);
342    let mut t_to_u: SubstitutionMap = HashMap::new();
343    t_to_u.insert(Arc::clone(&t_symbol), Arc::clone(&u_call));
344    let g_minus_alpha_u = substitute(context, &g_of_t_minus_alpha_t_normalized, &t_to_u);
345    let rhs = builders::divide(context, g_minus_alpha_u, Arc::clone(&x_symbol));
346    let auxiliary_equation = builders::subtract(context, u_prime_call, rhs);
347    let u_solution = separable::solve(context, &auxiliary_equation, &u_call, independent_variable)?;
348    let y_call = derivative_call(context, function, independent_variable, 0);
349    let x_to_alpha_back = builders::power(context, Arc::clone(&x_symbol), Arc::clone(&alpha));
350    if !contains_reference(&u_solution, &u_call) {
351        let y_explicit = builders::product(context, vec![u_solution, x_to_alpha_back]);
352        return Some(y_explicit);
353    }
354    let y_over_x_alpha = builders::divide(context, Arc::clone(&y_call), x_to_alpha_back);
355    let mut back_substitution: SubstitutionMap = HashMap::new();
356    back_substitution.insert(u_call, y_over_x_alpha);
357    Some(substitute(context, &u_solution, &back_substitution))
358}

360fn try_linear_fractional(
361    context: &Context,
362    f_in_surrogate: &ExpressionReference,
363    y_surrogate_identifier: SymbolIdentifier,
364    function: FunctionIdentifier,
365    independent_variable: SymbolIdentifier,
366) -> Option<ExpressionReference> {
367    // Pattern y' = (a₁x + b₁y + c₁) / (a₂x + b₂y + c₂) with all six
368    // coefficients constant. Detect by extracting numerator and
369    // denominator from f and verifying each is affine in (x, y_surrogate).
370    // Two sub-cases by the determinant D = a₁b₂ − a₂b₁:
371    //
372    //   D ≠ 0 (intersecting lines): shift (x, y) → (x + x₀, y + y₀) where
373    //     (x₀, y₀) solves the linear system a_i x + b_i y + c_i = 0; the
374    //     ODE in (ξ, η) is homogeneous, closed via the homogeneous path
375    //     of this same module. Back-substitute ξ ← x − x₀, η ← y − y₀.
376    //
377    //   D = 0 (parallel lines): substitute u = a₂x + b₂y; the ODE in u
378    //     is separable, closed via separable::solve. Back-substitute
379    //     u ← a₂x + b₂y.
380    let (numerator, denominator) = extract_numerator_and_denominator(context, f_in_surrogate);
381    let (a_one, b_one, c_one) = extract_affine_coefficients(
382        context,
383        &numerator,
384        independent_variable,
385        y_surrogate_identifier,
386    )?;
387    let (a_two, b_two, c_two) = extract_affine_coefficients(
388        context,
389        &denominator,
390        independent_variable,
391        y_surrogate_identifier,
392    )?;
393    if is_integer_zero(&a_two) && is_integer_zero(&b_two) {
394        return None;
395    }
396    if is_integer_zero(&a_one) && is_integer_zero(&b_one) {
397        return None;
398    }
399    if is_integer_zero(&c_one) && is_integer_zero(&c_two) {
400        return None;
401    }
402    let a_one_b_two = builders::product(context, vec![Arc::clone(&a_one), Arc::clone(&b_two)]);
403    let a_two_b_one = builders::product(context, vec![Arc::clone(&a_two), Arc::clone(&b_one)]);
404    let determinant = builders::subtract(context, a_one_b_two, a_two_b_one);
405    if !is_integer_zero(&determinant) {
406        return solve_linear_fractional_intersecting(
407            context,
408            &a_one,
409            &b_one,
410            &c_one,
411            &a_two,
412            &b_two,
413            &c_two,
414            &determinant,
415            function,
416            independent_variable,
417        );
418    }
419    solve_linear_fractional_parallel(
420        context,
421        &a_one,
422        &b_one,
423        &c_one,
424        &a_two,
425        &b_two,
426        &c_two,
427        function,
428        independent_variable,
429    )
430}

432fn solve_linear_fractional_intersecting(
433    context: &Context,
434    a_one: &ExpressionReference,
435    b_one: &ExpressionReference,
436    c_one: &ExpressionReference,
437    a_two: &ExpressionReference,
438    b_two: &ExpressionReference,
439    c_two: &ExpressionReference,
440    determinant: &ExpressionReference,
441    function: FunctionIdentifier,
442    independent_variable: SymbolIdentifier,
443) -> Option<ExpressionReference> {
444    // D ≠ 0 case. Solve the 2×2 system
445    //   a₁ x₀ + b₁ y₀ + c₁ = 0
446    //   a₂ x₀ + b₂ y₀ + c₂ = 0
447    // via Cramer's rule, then build the (ξ, η) homogeneous ODE and close
448    // it through try_homogeneous. Finally back-translate.
449    let c_one_b_two = builders::product(context, vec![Arc::clone(c_one), Arc::clone(b_two)]);
450    let c_two_b_one = builders::product(context, vec![Arc::clone(c_two), Arc::clone(b_one)]);
451    let determinant_x = builders::subtract(context, c_one_b_two, c_two_b_one);
452    let negative_determinant_x = builders::negate(context, determinant_x);
453    let x_shift = builders::divide(context, negative_determinant_x, Arc::clone(determinant));
454    let a_one_c_two = builders::product(context, vec![Arc::clone(a_one), Arc::clone(c_two)]);
455    let a_two_c_one = builders::product(context, vec![Arc::clone(a_two), Arc::clone(c_one)]);
456    let determinant_y = builders::subtract(context, a_one_c_two, a_two_c_one);
457    let negative_determinant_y = builders::negate(context, determinant_y);
458    let y_shift = builders::divide(context, negative_determinant_y, Arc::clone(determinant));
459    let xi_identifier = context.symbol_table.fresh("xi_lie_linear_fractional");
460    let xi_symbol = context
461        .interner
462        .intern(ExpressionKind::Symbol(xi_identifier));
463    let eta_function_identifier = context.symbol_table.fresh("eta_lie_linear_fractional");
464    let eta_function = FunctionIdentifier::User(eta_function_identifier);
465    let eta_call = builders::function_call(context, eta_function, vec![Arc::clone(&xi_symbol)]);
466    let eta_prime_call =
467        builders::derivative(context, eta_function, vec![1], vec![Arc::clone(&xi_symbol)]);
468    let a_one_xi = builders::product(context, vec![Arc::clone(a_one), Arc::clone(&xi_symbol)]);
469    let b_one_eta = builders::product(context, vec![Arc::clone(b_one), Arc::clone(&eta_call)]);
470    let translated_numerator = builders::sum(context, vec![a_one_xi, b_one_eta]);
471    let a_two_xi = builders::product(context, vec![Arc::clone(a_two), Arc::clone(&xi_symbol)]);
472    let b_two_eta = builders::product(context, vec![Arc::clone(b_two), Arc::clone(&eta_call)]);
473    let translated_denominator = builders::sum(context, vec![a_two_xi, b_two_eta]);
474    let translated_rhs = builders::divide(context, translated_numerator, translated_denominator);
475    let translated_equation = builders::subtract(context, eta_prime_call, translated_rhs);
476    let eta_solution = solve(context, &translated_equation, &eta_call, xi_identifier)?;
477    let x_symbol = context
478        .interner
479        .intern(ExpressionKind::Symbol(independent_variable));
480    let y_call = derivative_call(context, function, independent_variable, 0);
481    let negative_x_shift = builders::negate(context, Arc::clone(&x_shift));
482    let xi_in_x = builders::sum(context, vec![Arc::clone(&x_symbol), negative_x_shift]);
483    let negative_y_shift = builders::negate(context, Arc::clone(&y_shift));
484    let eta_in_y = builders::sum(context, vec![Arc::clone(&y_call), negative_y_shift]);
485    let mut back_substitution: SubstitutionMap = HashMap::new();
486    back_substitution.insert(xi_symbol, xi_in_x);
487    back_substitution.insert(eta_call, eta_in_y);
488    let translated_solution = substitute(context, &eta_solution, &back_substitution);
489    if !contains_reference(&translated_solution, &y_call) {
490        let y_explicit = builders::sum(context, vec![translated_solution, y_shift]);
491        return Some(y_explicit);
492    }
493    Some(translated_solution)
494}

496fn solve_linear_fractional_parallel(
497    context: &Context,
498    a_one: &ExpressionReference,
499    b_one: &ExpressionReference,
500    c_one: &ExpressionReference,
501    a_two: &ExpressionReference,
502    b_two: &ExpressionReference,
503    c_two: &ExpressionReference,
504    function: FunctionIdentifier,
505    independent_variable: SymbolIdentifier,
506) -> Option<ExpressionReference> {
507    // D = 0 case (parallel lines). Since a₁ b₂ = a₂ b₁, the numerator
508    // is λ times the linear (a₂ x + b₂ y) plus a constant: numerator =
509    // λ (a₂ x + b₂ y) + c₁ where λ is determined from (a₁, b₁) and
510    // (a₂, b₂). Choose the substitution u = a₂ x + b₂ y, in which case
511    //   du/dx = a₂ + b₂ y' = a₂ + b₂ (λ u + c₁) / (u + c₂),
512    // which is separable in u. Solve through separable::solve and back-
513    // substitute u ← a₂ x + b₂ y.
514    let lambda = compute_parallel_ratio(context, a_one, b_one, a_two, b_two)?;
515    let auxiliary_function_identifier = context
516        .symbol_table
517        .fresh("u_lie_linear_fractional_parallel");
518    let auxiliary_function = FunctionIdentifier::User(auxiliary_function_identifier);
519    let x_symbol = context
520        .interner
521        .intern(ExpressionKind::Symbol(independent_variable));
522    let u_call = builders::function_call(context, auxiliary_function, vec![Arc::clone(&x_symbol)]);
523    let u_prime_call = builders::derivative(
524        context,
525        auxiliary_function,
526        vec![1],
527        vec![Arc::clone(&x_symbol)],
528    );
529    let u_atom_identifier = context.symbol_table.fresh("u_atom_parallel");
530    let u_atom = context
531        .interner
532        .intern(ExpressionKind::Symbol(u_atom_identifier));
533    let lambda_times_u_atom =
534        builders::product(context, vec![Arc::clone(&lambda), Arc::clone(&u_atom)]);
535    let combined_numerator = builders::sum(context, vec![lambda_times_u_atom, Arc::clone(c_one)]);
536    let combined_denominator = builders::sum(context, vec![Arc::clone(&u_atom), Arc::clone(c_two)]);
537    let y_prime_in_u_atom = builders::divide(context, combined_numerator, combined_denominator);
538    let b_two_times_y_prime =
539        builders::product(context, vec![Arc::clone(b_two), y_prime_in_u_atom]);
540    let u_prime_rhs_in_atom = builders::sum(context, vec![Arc::clone(a_two), b_two_times_y_prime]);
541    let u_prime_rhs_normalized =
542        normalize_rational_in_variable(context, &u_prime_rhs_in_atom, u_atom_identifier)
543            .unwrap_or(u_prime_rhs_in_atom);
544    let mut atom_to_u_call: SubstitutionMap = HashMap::new();
545    atom_to_u_call.insert(Arc::clone(&u_atom), Arc::clone(&u_call));
546    let u_prime_rhs = substitute(context, &u_prime_rhs_normalized, &atom_to_u_call);
547    let auxiliary_equation = builders::subtract(context, u_prime_call, u_prime_rhs);
548    let u_solution = separable::solve(context, &auxiliary_equation, &u_call, independent_variable)?;
549    let y_call = derivative_call(context, function, independent_variable, 0);
550    let a_two_x = builders::product(context, vec![Arc::clone(a_two), Arc::clone(&x_symbol)]);
551    let b_two_y = builders::product(context, vec![Arc::clone(b_two), Arc::clone(&y_call)]);
552    let u_in_x_y = builders::sum(context, vec![a_two_x, b_two_y]);
553    if !contains_reference(&u_solution, &u_call) {
554        let u_minus_a_two_x = {
555            let a_two_x_back =
556                builders::product(context, vec![Arc::clone(a_two), Arc::clone(&x_symbol)]);
557            let negative_a_two_x = builders::negate(context, a_two_x_back);
558            builders::sum(context, vec![u_solution, negative_a_two_x])
559        };
560        let y_explicit = builders::divide(context, u_minus_a_two_x, Arc::clone(b_two));
561        return Some(y_explicit);
562    }
563    let mut back_substitution: SubstitutionMap = HashMap::new();
564    back_substitution.insert(u_call, u_in_x_y);
565    Some(substitute(context, &u_solution, &back_substitution))
566}

568fn compute_parallel_ratio(
569    context: &Context,
570    a_one: &ExpressionReference,
571    b_one: &ExpressionReference,
572    a_two: &ExpressionReference,
573    b_two: &ExpressionReference,
574) -> Option<ExpressionReference> {
575    // For the parallel sub-case a₁ b₂ = a₂ b₁; therefore the proportion
576    // λ = a₁/a₂ = b₁/b₂. Use whichever denominator is non-zero to avoid
577    // division by zero. Returns None if both (a₂, b₂) are zero (which
578    // contradicts the precondition checked by try_linear_fractional, but
579    // is guarded here for completeness).
580    if !is_integer_zero(a_two) {
581        return Some(builders::divide(
582            context,
583            Arc::clone(a_one),
584            Arc::clone(a_two),
585        ));
586    }
587    if !is_integer_zero(b_two) {
588        return Some(builders::divide(
589            context,
590            Arc::clone(b_one),
591            Arc::clone(b_two),
592        ));
593    }
594    None
595}

597fn extract_numerator_and_denominator(
598    context: &Context,
599    target: &ExpressionReference,
600) -> (ExpressionReference, ExpressionReference) {
601    // Decompose `target` syntactically into (numerator, denominator) such
602    // that target = numerator / denominator. Walk the top-level product
603    // structure: each child of the form `power(base, integer(-n))` for a
604    // positive integer n contributes `base^n` to the denominator; all
605    // other children remain in the numerator. A bare power with a
606    // negative integer exponent is handled identically; any other shape
607    // is treated as numerator with denominator one.
608    let factors: Vec<ExpressionReference> = match &target.kind {
609        ExpressionKind::Product(children) => children.iter().map(Arc::clone).collect(),
610        _ => vec![Arc::clone(target)],
611    };
612    let mut numerator_factors: Vec<ExpressionReference> = Vec::new();
613    let mut denominator_factors: Vec<ExpressionReference> = Vec::new();
614    for factor in factors {
615        if let Some((base, positive_exponent)) = negative_integer_power_decomposition(&factor) {
616            if positive_exponent == 1 {
617                denominator_factors.push(base);
618            } else {
619                let exponent_value = builders::integer_value(context, positive_exponent as i64);
620                let positive_power = builders::power(context, base, exponent_value);
621                denominator_factors.push(positive_power);
622            }
623        } else {
624            numerator_factors.push(factor);
625        }
626    }
627    let numerator = if numerator_factors.is_empty() {
628        builders::one(context)
629    } else if numerator_factors.len() == 1 {
630        numerator_factors.into_iter().next().unwrap()
631    } else {
632        builders::product(context, numerator_factors)
633    };
634    let denominator = if denominator_factors.is_empty() {
635        builders::one(context)
636    } else if denominator_factors.len() == 1 {
637        denominator_factors.into_iter().next().unwrap()
638    } else {
639        builders::product(context, denominator_factors)
640    };
641    (numerator, denominator)
642}
643
644fn negative_integer_power_decomposition(
645    factor: &ExpressionReference,
646) -> Option<(ExpressionReference, i64)> {
647    // Recognise factors of the form `power(base, integer(-n))` for a
648    // strictly positive integer n. Returns (base, n) so the caller can
649    // reconstruct `base^n` for the denominator product.
650    match &factor.kind {
651        ExpressionKind::Power(base, exponent) => match &exponent.kind {
652            ExpressionKind::Integer(value) => {
653                if *value < crate::exact_arithmetic::integer_zero() {
654                    let positive = -value;
655                    let as_i64: i64 = i64::try_from(&positive).ok()?;
656                    Some((Arc::clone(base), as_i64))
657                } else {
658                    None
659                }
660            }
661            _ => None,
662        },
663        _ => None,
664    }
665}

667fn extract_affine_coefficients(
668    context: &Context,
669    target: &ExpressionReference,
670    independent_variable: SymbolIdentifier,
671    y_surrogate_identifier: SymbolIdentifier,
672) -> Option<(
673    ExpressionReference,
674    ExpressionReference,
675    ExpressionReference,
676)> {
677    // Decompose `target` as a · x + b · y_surrogate + c. The coefficients
678    // are recovered by differentiating once with respect to x (gives a),
679    // once with respect to y_surrogate (gives b), and then substituting
680    // x ← 0, y_surrogate ← 0 (gives c). The decomposition is valid only
681    // when target is affine; verify by checking that the residual is
682    // identically zero.
683    let a_coefficient = differentiate(context, target, independent_variable);
684    if depends_on(&a_coefficient, independent_variable)
685        || depends_on(&a_coefficient, y_surrogate_identifier)
686    {
687        return None;
688    }
689    let b_coefficient = differentiate(context, target, y_surrogate_identifier);
690    if depends_on(&b_coefficient, independent_variable)
691        || depends_on(&b_coefficient, y_surrogate_identifier)
692    {
693        return None;
694    }
695    let zero = builders::zero(context);
696    let x_symbol = context
697        .interner
698        .intern(ExpressionKind::Symbol(independent_variable));
699    let y_symbol = context
700        .interner
701        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
702    let mut zeroing: SubstitutionMap = HashMap::new();
703    zeroing.insert(x_symbol, Arc::clone(&zero));
704    zeroing.insert(y_symbol, Arc::clone(&zero));
705    let c_coefficient = substitute(context, target, &zeroing);
706    if depends_on(&c_coefficient, independent_variable)
707        || depends_on(&c_coefficient, y_surrogate_identifier)
708    {
709        return None;
710    }
711    Some((a_coefficient, b_coefficient, c_coefficient))
712}

714fn satisfies_homogeneity_identity(
715    context: &Context,
716    f_in_surrogate: &ExpressionReference,
717    x_symbol: &ExpressionReference,
718    y_surrogate: &ExpressionReference,
719    independent_variable: SymbolIdentifier,
720    y_surrogate_identifier: SymbolIdentifier,
721) -> bool {
722    // Verify the Euler-homogeneity identity for degree 0:
723    //   x · f_x + y · f_y = 0
724    // where f is the right-hand side after y(x) → y_surrogate. For a
725    // rational f = N/D, computing this directly leaves residuals that
726    // auto-simplification cannot collapse without polynomial division;
727    // instead, extract the (N, D) decomposition and verify the
728    // equivalent polynomial residual
729    //   (x · N_x + y · N_y) · D − (x · D_x + y · D_y) · N = 0
730    // which auto-simplification reduces to integer zero when both N and
731    // D are homogeneous of the same degree (the canonical sort on
732    // product children ensures `N · D` and `D · N` share the same
733    // hash-consed form, so the difference cancels under like-term
734    // collection). The direct identity is also checked first as a fast
735    // path for polynomial f where the residual already collapses
736    // without the polynomial-form expansion.
737    let f_x = differentiate(context, f_in_surrogate, independent_variable);
738    let f_y = differentiate(context, f_in_surrogate, y_surrogate_identifier);
739    let x_f_x = builders::product(context, vec![Arc::clone(x_symbol), f_x]);
740    let y_f_y = builders::product(context, vec![Arc::clone(y_surrogate), f_y]);
741    let direct_residual = builders::sum(context, vec![x_f_x, y_f_y]);
742    let expanded_direct_residual = crate::expand::expand(context, &direct_residual);
743    if is_integer_zero(&expanded_direct_residual) {
744        return true;
745    }
746    let (numerator, denominator) = extract_numerator_and_denominator(context, f_in_surrogate);
747    let n_x = differentiate(context, &numerator, independent_variable);
748    let n_y = differentiate(context, &numerator, y_surrogate_identifier);
749    let d_x = differentiate(context, &denominator, independent_variable);
750    let d_y = differentiate(context, &denominator, y_surrogate_identifier);
751    let x_n_x = builders::product(context, vec![Arc::clone(x_symbol), n_x]);
752    let y_n_y = builders::product(context, vec![Arc::clone(y_surrogate), n_y]);
753    let euler_numerator = builders::sum(context, vec![x_n_x, y_n_y]);
754    let x_d_x = builders::product(context, vec![Arc::clone(x_symbol), d_x]);
755    let y_d_y = builders::product(context, vec![Arc::clone(y_surrogate), d_y]);
756    let euler_denominator = builders::sum(context, vec![x_d_x, y_d_y]);
757    let term_one = builders::product(context, vec![euler_numerator, Arc::clone(&denominator)]);
758    let term_two = builders::product(context, vec![euler_denominator, Arc::clone(&numerator)]);
759    let polynomial_residual = builders::subtract(context, term_one, term_two);
760    let expanded_polynomial_residual = crate::expand::expand(context, &polynomial_residual);
761    is_integer_zero(&expanded_polynomial_residual)
762}

764fn determine_quasi_homogeneity_exponent(
765    context: &Context,
766    f_in_surrogate: &ExpressionReference,
767    x_f_x_plus_f: &ExpressionReference,
768    y_f_y_minus_f: &ExpressionReference,
769    x_symbol: &ExpressionReference,
770    y_surrogate: &ExpressionReference,
771    independent_variable: SymbolIdentifier,
772    y_surrogate_identifier: SymbolIdentifier,
773) -> Option<ExpressionReference> {
774    // Extract the rational exponent α such that
775    //   x f_x + α y f_y = (α − 1) f
776    // by evaluating −(x f_x + f)/(y f_y − f) at the numeric pair
777    // (x, y_surrogate) = (2, 3). The numeric substitution collapses the
778    // rational ratio through auto-simplification's folding pass, yielding
779    // a single number expression when α is a constant. The candidate is
780    // then verified by checking the polynomial determining residual at
781    // α — only the verification step uses α; the value itself is
782    // committed only after the residual collapses to zero under expand.
783    // Returns None on any of: division by zero at the probe point, a non-
784    // numeric folded candidate, or a non-zero verification residual.
785    let probe_x = builders::integer_value(context, 2);
786    let probe_y = builders::integer_value(context, 3);
787    let mut numeric_substitution: SubstitutionMap = HashMap::new();
788    numeric_substitution.insert(Arc::clone(x_symbol), Arc::clone(&probe_x));
789    numeric_substitution.insert(Arc::clone(y_surrogate), Arc::clone(&probe_y));
790    let numerator_at_probe = substitute(context, x_f_x_plus_f, &numeric_substitution);
791    let denominator_at_probe = substitute(context, y_f_y_minus_f, &numeric_substitution);
792    if is_integer_zero(&denominator_at_probe) {
793        return None;
794    }
795    let negative_numerator_at_probe = builders::negate(context, numerator_at_probe);
796    let alpha_at_probe =
797        builders::divide(context, negative_numerator_at_probe, denominator_at_probe);
798    if depends_on(&alpha_at_probe, independent_variable)
799        || depends_on(&alpha_at_probe, y_surrogate_identifier)
800    {
801        return None;
802    }
803    if !is_numeric_constant(&alpha_at_probe) {
804        return None;
805    }
806    let alpha = alpha_at_probe;
807    let f_x = differentiate(context, f_in_surrogate, independent_variable);
808    let f_y = differentiate(context, f_in_surrogate, y_surrogate_identifier);
809    let x_f_x = builders::product(context, vec![Arc::clone(x_symbol), f_x]);
810    let alpha_y_f_y = builders::product(
811        context,
812        vec![Arc::clone(&alpha), Arc::clone(y_surrogate), f_y],
813    );
814    let one = builders::one(context);
815    let alpha_minus_one = builders::subtract(context, Arc::clone(&alpha), one);
816    let alpha_minus_one_times_f =
817        builders::product(context, vec![alpha_minus_one, Arc::clone(f_in_surrogate)]);
818    let euler_sum = builders::sum(context, vec![x_f_x, alpha_y_f_y]);
819    let direct_residual = builders::subtract(context, euler_sum, alpha_minus_one_times_f);
820    let expanded_direct_residual = crate::expand::expand(context, &direct_residual);
821    if is_integer_zero(&expanded_direct_residual) {
822        return Some(alpha);
823    }
824    let (numerator, denominator) = extract_numerator_and_denominator(context, f_in_surrogate);
825    let n_x = differentiate(context, &numerator, independent_variable);
826    let n_y = differentiate(context, &numerator, y_surrogate_identifier);
827    let d_x = differentiate(context, &denominator, independent_variable);
828    let d_y = differentiate(context, &denominator, y_surrogate_identifier);
829    let x_n_x = builders::product(context, vec![Arc::clone(x_symbol), n_x]);
830    let alpha_y_n_y = builders::product(
831        context,
832        vec![Arc::clone(&alpha), Arc::clone(y_surrogate), n_y],
833    );
834    let euler_numerator = builders::sum(context, vec![x_n_x, alpha_y_n_y]);
835    let x_d_x = builders::product(context, vec![Arc::clone(x_symbol), d_x]);
836    let alpha_y_d_y = builders::product(
837        context,
838        vec![Arc::clone(&alpha), Arc::clone(y_surrogate), d_y],
839    );
840    let euler_denominator = builders::sum(context, vec![x_d_x, alpha_y_d_y]);
841    let one_again = builders::one(context);
842    let alpha_minus_one_again = builders::subtract(context, Arc::clone(&alpha), one_again);
843    let term_one = builders::product(context, vec![euler_numerator, Arc::clone(&denominator)]);
844    let term_two = builders::product(context, vec![euler_denominator, Arc::clone(&numerator)]);
845    let cross_residual = builders::subtract(context, term_one, term_two);
846    let alpha_minus_one_n_d = builders::product(
847        context,
848        vec![
849            alpha_minus_one_again,
850            Arc::clone(&numerator),
851            Arc::clone(&denominator),
852        ],
853    );
854    let polynomial_residual = builders::subtract(context, cross_residual, alpha_minus_one_n_d);
855    let expanded_polynomial_residual = crate::expand::expand(context, &polynomial_residual);
856    if is_integer_zero(&expanded_polynomial_residual) {
857        return Some(alpha);
858    }
859    None
860}

862fn is_numeric_constant(target: &ExpressionReference) -> bool {
863    // Predicate identifying an integer or rational literal. Used by the
864    // quasi-homogeneous probe to confirm the numeric substitution
865    // collapsed alpha to a single rational number.
866    matches!(
867        &target.kind,
868        ExpressionKind::Integer(_) | ExpressionKind::Rational(_)
869    )
870}

872fn normalize_rational_in_variable(
873    context: &Context,
874    target: &ExpressionReference,
875    variable: SymbolIdentifier,
876) -> Option<ExpressionReference> {
877    // Bring a rational expression in `variable` to (numerator)/(denominator)
878    // form, where numerator and denominator are univariate polynomials in
879    // `variable` with the common gcd divided out. Wraps the gosper-module
880    // helper [expression_to_rational_pair] and converts the resulting
881    // polynomial pair back into an expression via [polynomial_to_expression].
882    // Returns None whenever the input fails to lift to a rational function
883    // in `variable` — the canonical case is a transcendental factor in the
884    // numerator or denominator that polynomial conversion cannot represent.
885    let (numerator_polynomial, denominator_polynomial) =
886        crate::summation::gosper::expression_to_rational_pair(
887            target,
888            variable,
889            context.operation(),
890        )?;
891    let numerator_expression =
892        crate::polynomial::conversion::polynomial_to_expression(context, &numerator_polynomial);
893    let denominator_expression =
894        crate::polynomial::conversion::polynomial_to_expression(context, &denominator_polynomial);
895    Some(builders::divide(
896        context,
897        numerator_expression,
898        denominator_expression,
899    ))
900}

902fn is_integer_zero(target: &ExpressionReference) -> bool {
903    // Predicate identifying the integer literal zero. Used pervasively
904    // by the pattern-detection helpers to distinguish degenerate input
905    // shapes (e.g. f_y ≡ 0 in the translation classifier) from genuine
906    // matches.
907    use crate::exact_arithmetic::Integer;
908    matches!(&target.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
909}
