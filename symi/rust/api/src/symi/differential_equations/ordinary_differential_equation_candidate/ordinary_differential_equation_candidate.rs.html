1// Structured first-order ordinary-differential-equation candidate seam.
2//
3// The historical public entry point returns only an expression, and each
4// first-order route allocates its own fresh integration constant internally.
5// Characteristic solving needs the exact identity of every generated constant
6// so that an invariant can be recovered by exact elimination, and it needs to
7// know whether the returned relation is explicit or implicit. Discovering that
8// after the fact by scanning symbol names or by differencing free-symbol sets
9// is unsound, because the internal classifiers also create surrogate symbols.
10//
11// This module keeps the tracked routes internal and first-order-focused. Only
12// the separable, linear-first-order, and exact routes report their generated
13// constants; a route that cannot report them exactly stays ineligible here.
14//
15// The three routes are peers: every one of them is offered the same equation,
16// every one runs whatever the others do, and every closed form is kept, because
17// equivalent closed forms are not equally eliminable and the caller — not this
18// seam — chooses between them. That is what makes this the one coarse parallel
19// boundary the first-order machinery has. The method ladders above it are not:
20// the public dispatcher stops at its first success, and on the equations that
21// reach it the winning route carries essentially the whole computation, so
22// speculating on the losers buys nothing and destroys the short-circuit.
23//
24// Two things must be true for the routes to run concurrently, and both are
25// arranged here rather than inside the routes. First, a route must not read
26// what a sibling produced; none does, because each classifies and integrates
27// the same equation from scratch. Second, the integration constants must be
28// named by the mathematics rather than by the schedule. Every route mints its
29// constant from the shared `integration_constant` prefix on its own success
30// path, so under concurrent execution which route receives which suffix would
31// follow completion order. When — and only when — the cost gate admits the
32// parallel route, the coordinator therefore mints the canonical constants
33// itself, in route order, before any task is formed, and rewrites each route's
34// own constant to the canonical one it was assigned; the published naming is
35// then a function of the route order alone. A serial batch is left exactly as it
36// was, because its routes already mint in route order by running in it.

50/// Whether a route closed the equation explicitly or only as a relation.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub(crate) enum OrdinaryDifferentialEquationSolutionForm {
53    /// The dependent value is isolated on its own.
54    Explicit,
55    /// The result is a residual relation interpreted as equal to zero.
56    Implicit,
57}

91pub(crate) fn solve_ordinary_differential_equation_candidates(
92    context: &Context,
93    equation: &ExpressionReference,
94    dependent_function_call: &ExpressionReference,
95    independent_variable: SymbolIdentifier,
96) -> Vec<OrdinaryDifferentialEquationSolutionCandidate> {
97    // Every first-order route that reports its generated integration constants
98    // is offered, in the dispatcher's own fixed order. Routes disagree on the
99    // shape of an equivalent closed form — the separable route writes a linear
100    // decay through a logarithm where the integrating-factor route writes it as
101    // an exponential — and only one of those shapes may admit exact elimination
102    // later, so the caller keeps the choice rather than the seam. A route that
103    // stalls contributes nothing, which preserves the standing rule that a
104    // partially closed expression is never a result.
105    let routes: [TrackedRoute; 4] = [
106        crate::differential_equations::separable::solve_tracked,
107        crate::differential_equations::linear_first_order::solve_tracked,
108        crate::differential_equations::exact::solve_tracked,
109        crate::differential_equations::bernoulli::solve_tracked,
110    ];
111    let route = candidate_route(context, equation, routes.len());
112    let canonical_constants = canonical_route_constants(context, route, routes.len());
113    let route_outcomes = ordered_map_with_identifiers(
114        route,
115        &routes,
116        &crate::parallelism::LogicalTaskIdentifier::root(context.operation().identifier().value()),
117        |route, identifier| {
118            crate::instrumentation::apply_structured_domain_test_control(
119                crate::instrumentation::StructuredDomainWorkKind::OrdinaryDifferentialEquationMode,
120                identifier.path().last().copied().unwrap_or_default() as usize,
121            );
122            let mut generated_constants = Vec::new();
123            let route_solution = route(
124                context,
125                equation,
126                dependent_function_call,
127                independent_variable,
128                &mut generated_constants,
129            );
130            (route_solution, generated_constants)
131        },
132    );
133    let mut candidates = Vec::new();
134    for (route_index, (route_solution, generated_constants)) in
135        route_outcomes.into_iter().enumerate()
136    {
137        let Some(route_solution) = route_solution else {
138            continue;
139        };
140        let candidate = match canonical_constants.get(route_index).copied() {
141            Some(canonical_constant) => rename_generated_constant(
142                context,
143                route_solution,
144                generated_constants,
145                canonical_constant,
146            ),
147            None => OrdinaryDifferentialEquationSolutionCandidate {
148                solution: route_solution.solution,
149                form: route_solution.form,
150                generated_constants,
151                generated_spectator_functions: route_solution.generated_spectator_functions,
152                domain_obligations: route_solution.domain_obligations,
153            },
154        };
155        candidates.push(candidate);
156    }
157    candidates
158}

160fn canonical_route_constants(
161    context: &Context,
162    route: ExecutionRoute,
163    route_count: usize,
164) -> Vec<SymbolIdentifier> {
165    // One integration constant per route, minted here in route order so the naming is decided
166    // before any task exists. A route that declines still owns its name; leaving a gap costs one
167    // unused symbol and buys a numbering that does not depend on which routes happened to close.
168    //
169    // Only a batch that could actually fork needs this. On the serial route the routes run one
170    // after another and mint in route order by themselves, so pre-minting would change the
171    // published naming of a path that was never at risk — and reserve names no route uses. The
172    // serial route therefore reserves nothing and every route keeps the constant it minted.
173    if route != ExecutionRoute::Parallel {
174        return Vec::new();
175    }
176    (0..route_count)
177        .map(|_| context.symbol_table.fresh("integration_constant"))
178        .collect()
179}

181fn rename_generated_constant(
182    context: &Context,
183    route_solution: OrdinaryDifferentialEquationRouteSolution,
184    generated_constants: Vec<SymbolIdentifier>,
185    canonical_constant: SymbolIdentifier,
186) -> OrdinaryDifferentialEquationSolutionCandidate {
187    // Rewrite the one constant a route minted for itself to the canonical constant its route
188    // index was assigned. A route that reports anything other than exactly one constant is left
189    // as it is: the callers that eliminate constants already require exactly one and drop the
190    // rest, so renaming a shape this seam does not model would be a guess.
191    if generated_constants.len() != 1 || generated_constants[0] == canonical_constant {
192        return OrdinaryDifferentialEquationSolutionCandidate {
193            solution: route_solution.solution,
194            form: route_solution.form,
195            generated_constants,
196            generated_spectator_functions: route_solution.generated_spectator_functions,
197            domain_obligations: route_solution.domain_obligations,
198        };
199    }
200    let minted = context
201        .interner
202        .intern(crate::expression::ExpressionKind::Symbol(
203            generated_constants[0],
204        ));
205    let canonical = context
206        .interner
207        .intern(crate::expression::ExpressionKind::Symbol(
208            canonical_constant,
209        ));
210    let mut renaming: SubstitutionMap = HashMap::new();
211    renaming.insert(minted, canonical);
212    OrdinaryDifferentialEquationSolutionCandidate {
213        solution: substitute(context, &route_solution.solution, &renaming),
214        form: route_solution.form,
215        generated_constants: vec![canonical_constant],
216        generated_spectator_functions: route_solution.generated_spectator_functions,
217        domain_obligations: route_solution.domain_obligations,
218    }
219}

221fn candidate_route(
222    context: &Context,
223    equation: &ExpressionReference,
224    route_count: usize,
225) -> ExecutionRoute {
226    // Each route classifies and integrates the whole equation, so the equation's structural size
227    // gauges every route alike and the estimate is that size times the route count. The cutoff
228    // that reads it is recorded with the measurement that placed it alongside the other work
229    // classes.
230    if route_count < 2 {
231        return ExecutionRoute::Serial;
232    }
233    // The equation's structural size below is a walk of the whole equation, so the boundary asks
234    // whether the class can fork at all before taking it.
235    if !context
236        .operation()
237        .may_fork_work_class("differential_equation_route")
238    {
239        record_counter!(differential_equation_serial_route_batches);
240        return ExecutionRoute::Serial;
241    }
242    let equation_size = crate::simplify::heuristic::node_count(equation);
243    let work_estimate = WorkEstimate::new(
244        (equation_size as u64).saturating_mul(route_count as u64),
245        route_count,
246        route_count
247            .saturating_mul(equation_size)
248            .saturating_mul(std::mem::size_of::<ExpressionReference>()),
249        0,
250        route_count,
251        "differential_equation_route",
252    );
253    let decision = context
254        .operation()
255        .execution_decision_silent(&work_estimate, route_count, None);
256    match decision.route {
257        ExecutionRoute::Parallel => {
258            record_counter!(differential_equation_parallel_route_batches)
259        }
260        ExecutionRoute::Serial => {
261            record_counter!(differential_equation_serial_route_batches)
262        }
263    }
264    decision.route
265}

275#[allow(
276    dead_code,
277    reason = "Retains candidate data for exact Cauchy-problem construction."
278)]
279pub(crate) fn specialize_ordinary_differential_equation_candidate(
280    context: &Context,
281    candidate: &OrdinaryDifferentialEquationSolutionCandidate,
282    independent_variable: SymbolIdentifier,
283    independent_value: &ExpressionReference,
284    dependent_value: &ExpressionReference,
285) -> Option<ExpressionReference> {
286    // Exact scalar initial-value specialization. The candidate is evaluated at
287    // the supplied independent value, the residual against the supplied
288    // dependent value is solved for the tracked constants alone, and a branch
289    // that still mentions a tracked constant is rejected. Every step is exact;
290    // no branch is chosen by ordering or presentation.
291    if candidate.generated_constants.is_empty() {
292        return None;
293    }
294    let variable_expression = context
295        .interner
296        .intern(crate::expression::ExpressionKind::Symbol(
297            independent_variable,
298        ));
299    let mut evaluation: SubstitutionMap = HashMap::new();
300    evaluation.insert(variable_expression, Arc::clone(independent_value));
301    let evaluated = substitute(context, &candidate.solution, &evaluation);
302    let residual = match candidate.form {
303        OrdinaryDifferentialEquationSolutionForm::Explicit => {
304            crate::expression::builders::subtract(context, evaluated, Arc::clone(dependent_value))
305        }
306        OrdinaryDifferentialEquationSolutionForm::Implicit => evaluated,
307    };
308    let mut specialized = Arc::clone(&candidate.solution);
309    for constant in &candidate.generated_constants {
310        let value = unique_exact_root(context, &residual, *constant)?;
311        if candidate
312            .generated_constants
313            .iter()
314            .any(|tracked| crate::expression::queries::depends_on(&value, *tracked))
315        {
316            return None;
317        }
318        let constant_expression = context
319            .interner
320            .intern(crate::expression::ExpressionKind::Symbol(*constant));
321        let mut mapping: SubstitutionMap = HashMap::new();
322        mapping.insert(constant_expression, value);
323        specialized = substitute(context, &specialized, &mapping);
324    }
325    for constant in &candidate.generated_constants {
326        if crate::expression::queries::depends_on(&specialized, *constant) {
327            return None;
328        }
329    }
330    Some(specialized)
331}

333pub(crate) fn unique_exact_root(
334    context: &Context,
335    residual: &ExpressionReference,
336    variable: SymbolIdentifier,
337) -> Option<ExpressionReference> {
338    // Exact elimination helper shared by constant recovery and specialization.
339    // Only a complete or independently verified root list of length one is
340    // accepted, so a solver decline and a genuinely multibranch inverse are
341    // both refused instead of silently picking the first entry.
342    let roots = match solve_with_verdict(context, residual, variable) {
343        SolveOutcome::CompleteRoots(roots) | SolveOutcome::VerifiedRoots(roots) => roots,
344        SolveOutcome::Declined => return None,
345    };
346    if roots.len() != 1 {
347        return None;
348    }
349    let root = Arc::clone(&roots[0]);
350    (!crate::expression::queries::depends_on(&root, variable)).then_some(root)
351}
