1// Linear first-order classifier. Recognises `y'(x) + p(x) * y(x) =
2// q(x)` by collecting on y and y'. The integrating-factor closed form
3//   μ(x) = exp(∫ p(x) dx)
4//   y(x) = ( ∫ μ(x) * q(x) dx + C ) / μ(x)
5// routes both inner integrals through the full integration pipeline
6// (builtin-table, rational-function, and Risch stages). A
7// stall inside either integral causes the entry point to return None so
8// the outer dispatcher emits the unsolved placeholder rather than commit a
9// partial result.
10//
11// Whenever the antiderivative of `p` carries an integer multiple of a logarithm, that part of the
12// integrating factor is written as the power it is: `exp(a + n log(u)) = u^n exp(a)` holds
13// wherever `u` is nonzero, and that restriction travels with the candidate as a domain
14// obligation rather than being assumed away. The rewrite is not cosmetic. An integrating factor
15// left as `exp(t + log(2 + sin t))` is the same function as `(2 + sin t) exp(t)`, but only the
16// second is a shape any later consumer can read — a system that substitutes this component into
17// the next equation needs the forcing term to lie in a space it can name, and an exponential of
18// a logarithm lies in none of them.

39/// Performs the advanced native `solve` operation.
40pub fn solve(
41    context: &Context,
42    equation: &ExpressionReference,
43    dependent_variable_call: &ExpressionReference,
44    independent_variable: SymbolIdentifier,
45) -> Option<ExpressionReference> {
46    // Expression-only projection of the tracked entry point. The generated
47    // integration constant is discarded here because the historical public
48    // ordinary-differential-equation surface returns only an expression.
49    let mut generated_constants = Vec::new();
50    solve_tracked(
51        context,
52        equation,
53        dependent_variable_call,
54        independent_variable,
55        &mut generated_constants,
56    )
57    .map(|closed_form| closed_form.solution)
58}

60pub(crate) fn solve_tracked(
61    context: &Context,
62    equation: &ExpressionReference,
63    dependent_variable_call: &ExpressionReference,
64    independent_variable: SymbolIdentifier,
65    generated_constants: &mut Vec<SymbolIdentifier>,
66) -> Option<OrdinaryDifferentialEquationRouteSolution> {
67    // Eager entry point. Extracts the (p(x), q(x)) pair by matching the
68    // canonical linear-first-order shape against the supplied equation. A
69    // None p / q decomposition aborts to None so the dispatcher tries the
70    // next classifier (exact, Bernoulli, Riccati, constant-coefficient).
71    // The integrating-factor closed form is then assembled and returned in
72    // its explicit y(x) = ... form; on integration stalls anywhere in the
73    // pipeline the function returns None.
74    let function = dependent_function_identifier(dependent_variable_call)?;
75    let (p_of_x, q_of_x) = extract_linear_first_order_coefficients(
76        context,
77        equation,
78        dependent_variable_call,
79        function,
80        independent_variable,
81    )?;
82    let mut domain_obligations = Vec::new();
83    let mu = match logarithmic_derivative_integrating_factor(context, &p_of_x, independent_variable)
84    {
85        Some(factor) => factor,
86        None => {
87            let p_antiderivative = integrate(context, &p_of_x, independent_variable);
88            if contains_unevaluated_integral(&p_antiderivative)
89                || matches!(
90                    p_antiderivative.kind,
91                    ExpressionKind::Constant(crate::expression::BuiltinConstant::Undefined)
92                )
93            {
94                return None;
95            }
96            exponential_of_antiderivative(context, &p_antiderivative, &mut domain_obligations)
97        }
98    };
99    let mu_times_q = builders::product(context, vec![Arc::clone(&mu), Arc::clone(&q_of_x)]);
100    let inner_antiderivative = integrate(context, &mu_times_q, independent_variable);
101    if contains_unevaluated_integral(&inner_antiderivative)
102        || matches!(
103            inner_antiderivative.kind,
104            ExpressionKind::Constant(crate::expression::BuiltinConstant::Undefined)
105        )
106    {
107        return None;
108    }
109    let integration_constant = integration_datum(
110        context,
111        dependent_variable_call,
112        independent_variable,
113        generated_constants,
114    )?;
115    let generated_spectator_functions = match &integration_constant.kind {
116        ExpressionKind::FunctionCall(function, _) => vec![*function],
117        _ => Vec::new(),
118    };
119    let numerator = builders::sum(context, vec![inner_antiderivative, integration_constant]);
120    Some(OrdinaryDifferentialEquationRouteSolution {
121        solution: crate::simplify::simplify(context, builders::divide(context, numerator, mu)),
122        form: OrdinaryDifferentialEquationSolutionForm::Explicit,
123        generated_spectator_functions,
124        domain_obligations,
125    })
126}

128fn exponential_of_antiderivative(
129    context: &Context,
130    antiderivative: &ExpressionReference,
131    domain_obligations: &mut Vec<
132        crate::differential_equations::verification::OrdinaryDifferentialEquationDomainObligation,
133    >,
134) -> ExpressionReference {
135    // `exp(a + n log(u)) = u^n exp(a)` for integer `n`, valid wherever `u` is nonzero. The power
136    // has to be an integer: for any other exponent the identity is a statement about which
137    // branch of `u^n` is meant, and no branch has been chosen here.
138    let terms = crate::differential_equations::shape::additive_terms(context, antiderivative);
139    let mut factors = Vec::new();
140    let mut exponent_terms = Vec::new();
141    for term in terms {
142        match integer_multiple_of_logarithm(context, &term) {
143            Some((argument, power)) => {
144                domain_obligations.push(
145                    crate::differential_equations::verification::OrdinaryDifferentialEquationDomainObligation::NonzeroExpression(
146                        Arc::clone(&argument),
147                    ),
148                );
149                factors.push(builders::power(
150                    context,
151                    argument,
152                    builders::integer(context, power),
153                ));
154            }
155            None => exponent_terms.push(term),
156        }
157    }
158    let remaining = builders::function_call(
159        context,
160        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
161        vec![builders::sum(context, exponent_terms)],
162    );
163    if factors.is_empty() {
164        return remaining;
165    }
166    factors.push(remaining);
167    crate::simplify::simplify(context, builders::product(context, factors))
168}

170fn integer_multiple_of_logarithm(
171    context: &Context,
172    term: &ExpressionReference,
173) -> Option<(ExpressionReference, crate::exact_arithmetic::Integer)> {
174    let logarithm_argument = |candidate: &ExpressionReference| match &candidate.kind {
175        ExpressionKind::FunctionCall(
176            FunctionIdentifier::Builtin(BuiltinFunction::Log),
177            arguments,
178        ) if arguments.len() == 1 => Some(Arc::clone(&arguments[0])),
179        _ => None,
180    };
181    if let Some(argument) = logarithm_argument(term) {
182        return Some((argument, crate::exact_arithmetic::Integer::from(1)));
183    }
184    let ExpressionKind::Product(children) = &term.kind else {
185        return None;
186    };
187    let mut argument = None;
188    let mut coefficient_factors = Vec::with_capacity(children.len());
189    for child in children {
190        match logarithm_argument(child) {
191            Some(inner) if argument.is_none() => argument = Some(inner),
192            Some(_) => return None,
193            None => coefficient_factors.push(Arc::clone(child)),
194        }
195    }
196    let argument = argument?;
197    let coefficient =
198        crate::simplify::simplify(context, builders::product(context, coefficient_factors));
199    let ExpressionKind::Integer(power) = &coefficient.kind else {
200        return None;
201    };
202    Some((argument, power.clone()))
203}

205fn logarithmic_derivative_integrating_factor(
206    context: &Context,
207    coefficient: &ExpressionReference,
208    independent_variable: SymbolIdentifier,
209) -> Option<ExpressionReference> {
210    let factors: Vec<ExpressionReference> = match &coefficient.kind {
211        ExpressionKind::Product(factors) => factors.iter().map(Arc::clone).collect(),
212        _ => vec![Arc::clone(coefficient)],
213    };
214    let coordinate = context
215        .interner
216        .intern(ExpressionKind::Symbol(independent_variable));
217    let reciprocal_coordinate = builders::power(
218        context,
219        coordinate.clone(),
220        builders::integer(context, (-1).into()),
221    );
222    let reciprocal_position = factors
223        .iter()
224        .position(|factor| factor.kind == reciprocal_coordinate.kind)?;
225    let remaining: Vec<ExpressionReference> = factors
226        .into_iter()
227        .enumerate()
228        .filter_map(|(position, factor)| (position != reciprocal_position).then_some(factor))
229        .collect();
230    let exponent = if remaining.is_empty() {
231        builders::one(context)
232    } else if remaining.len() == 1 {
233        remaining.into_iter().next()?
234    } else {
235        return None;
236    };
237    if !matches!(exponent.kind, ExpressionKind::Integer(_)) {
238        return None;
239    }
240    let integrating_factor = builders::power(context, coordinate, exponent);
241    let derivative = crate::expression::differentiate::differentiate(
242        context,
243        &integrating_factor,
244        independent_variable,
245    );
246    let required_derivative = builders::product(
247        context,
248        vec![coefficient.clone(), integrating_factor.clone()],
249    );
250    let residual = builders::subtract(context, derivative, required_derivative);
251    matches!(
252        crate::zero_equivalence::zero_equivalent(context, &residual, None),
253        crate::zero_equivalence::ZeroEquivalenceVerdict::TrueValue
254    )
255    .then_some(integrating_factor)
256}

258fn integration_datum(
259    context: &Context,
260    dependent_variable_call: &ExpressionReference,
261    independent_variable: SymbolIdentifier,
262    generated_constants: &mut Vec<SymbolIdentifier>,
263) -> Option<ExpressionReference> {
264    // Integration in one coordinate leaves a scalar constant only for a scalar unknown.  For
265    // `y(x, a, ...)`, every spectator remains free, so the datum is an arbitrary function of
266    // precisely those spectator coordinates.
267    let ExpressionKind::FunctionCall(_, arguments) = &dependent_variable_call.kind else {
268        return None;
269    };
270    let spectators: Vec<ExpressionReference> = arguments
271        .iter()
272        .filter(|argument| {
273            !matches!(&argument.kind, ExpressionKind::Symbol(identifier) if *identifier == independent_variable)
274        })
275        .map(Arc::clone)
276        .collect();
277    if spectators.is_empty() {
278        let integration_constant_identifier = context.symbol_table.fresh("integration_constant");
279        generated_constants.push(integration_constant_identifier);
280        return Some(
281            context
282                .interner
283                .intern(ExpressionKind::Symbol(integration_constant_identifier)),
284        );
285    }
286    let datum_identifier = context.symbol_table.fresh("integration_datum");
287    Some(builders::function_call(
288        context,
289        FunctionIdentifier::User(datum_identifier),
290        spectators,
291    ))
292}

294pub(crate) fn extract_linear_first_order_coefficients(
295    context: &Context,
296    equation: &ExpressionReference,
297    dependent_variable_call: &ExpressionReference,
298    function: FunctionIdentifier,
299    independent_variable: SymbolIdentifier,
300) -> Option<(ExpressionReference, ExpressionReference)> {
301    // Pull the (p(x), q(x)) pair out of the canonical form. The equation
302    // arrives in LHS - RHS = 0 form. The strategy is:
303    //   1. Treat y'(x) and y(x) as opaque atoms via fresh-symbol surrogates.
304    //   2. Extract the coefficient of y'(x) (the lead coefficient `a(x)`).
305    //   3. Extract the coefficient of y(x) (call it `b(x)`).
306    //   4. Extract the residual that depends on neither (call it `r(x)`).
307    //   5. Require a(x) != 0, then p(x) = b(x) / a(x), q(x) = -r(x) / a(x).
308    // The peeled terms cancel structurally only when the residual was written as a flat sum, so
309    // the surrogate residual is expanded before step 5 asks whether it still depends on either
310    // surrogate. A residual that states the same equation through a subtraction of a group —
311    // `y' - (q - y)` rather than `y' + y - q` — otherwise leaves the peeled occurrence inside a
312    // product and reads as nonlinear; the surrogates are plain symbols, so expanding disturbs
313    // nothing else.
314    // Coefficients that depend on y(x) or y'(x) (i.e. nonlinear in y or y'
315    // — including higher derivatives like y''(x) buried in the same term)
316    // cause the extraction to abort to None.
317    let y_call = Arc::clone(dependent_variable_call);
318    let y_prime_call = derivative_call_from_dependent_variable_call(
319        context,
320        dependent_variable_call,
321        independent_variable,
322        1,
323    )?;
324    let y_surrogate_identifier = context.symbol_table.fresh("y_linear_first_order");
325    let y_prime_surrogate_identifier = context.symbol_table.fresh("y_prime_linear_first_order");
326    let y_surrogate = context
327        .interner
328        .intern(ExpressionKind::Symbol(y_surrogate_identifier));
329    let y_prime_surrogate = context
330        .interner
331        .intern(ExpressionKind::Symbol(y_prime_surrogate_identifier));
332    let mut atom_mapping: SubstitutionMap = HashMap::new();
333    atom_mapping.insert(Arc::clone(&y_call), Arc::clone(&y_surrogate));
334    atom_mapping.insert(Arc::clone(&y_prime_call), Arc::clone(&y_prime_surrogate));
335    let surrogate_equation = substitute(context, equation, &atom_mapping);
336    if crate::differential_equations::shape::depends_on_function(&surrogate_equation, function) {
337        return None;
338    }
339    let coefficient_of_y_prime = coefficient_of_symbol(
340        context,
341        &surrogate_equation,
342        y_prime_surrogate_identifier,
343        y_surrogate_identifier,
344    )?;
345    if depends_on_surrogate(&coefficient_of_y_prime, y_surrogate_identifier) {
346        return None;
347    }
348    let after_removing_y_prime_term = subtract_symbol_term(
349        context,
350        &surrogate_equation,
351        &coefficient_of_y_prime,
352        y_prime_surrogate_identifier,
353    );
354    let coefficient_of_y = coefficient_of_symbol(
355        context,
356        &after_removing_y_prime_term,
357        y_surrogate_identifier,
358        y_prime_surrogate_identifier,
359    )?;
360    let residual = crate::simplify::simplify(
361        context,
362        crate::expand::expand(
363            context,
364            &subtract_symbol_term(
365                context,
366                &after_removing_y_prime_term,
367                &coefficient_of_y,
368                y_surrogate_identifier,
369            ),
370        ),
371    );
372    if depends_on_surrogate(&residual, y_surrogate_identifier)
373        || depends_on_surrogate(&residual, y_prime_surrogate_identifier)
374    {
375        return None;
376    }
377    if is_integer_zero(&coefficient_of_y_prime) {
378        return None;
379    }
380    let p_of_x = builders::divide(
381        context,
382        Arc::clone(&coefficient_of_y),
383        Arc::clone(&coefficient_of_y_prime),
384    );
385    let negated_residual = builders::negate(context, residual);
386    let q_of_x = builders::divide(context, negated_residual, coefficient_of_y_prime);
387    if depends_on_function(&p_of_x, function) || depends_on_function(&q_of_x, function) {
388        return None;
389    }
390    Some((p_of_x, q_of_x))
391}

393fn coefficient_of_symbol(
394    context: &Context,
395    target: &ExpressionReference,
396    symbol_of_interest: SymbolIdentifier,
397    other_symbol: SymbolIdentifier,
398) -> Option<ExpressionReference> {
399    // Differentiate `target` with respect to `symbol_of_interest` once,
400    // then substitute zero for both surrogates so the resulting
401    // expression is the coefficient of the linear term. Because the
402    // surrogates appear linearly in the original equation (the linear-
403    // first-order shape), this single differentiation recovers the
404    // coefficient exactly. Returns None when the differentiated
405    // expression still mentions either surrogate symbol (which would
406    // indicate a nonlinear dependence).
407    let derivative =
408        crate::expression::differentiate::differentiate(context, target, symbol_of_interest);
409    let zero = builders::zero(context);
410    let symbol_of_interest_expression = context
411        .interner
412        .intern(ExpressionKind::Symbol(symbol_of_interest));
413    let other_symbol_expression = context
414        .interner
415        .intern(ExpressionKind::Symbol(other_symbol));
416    let mut mapping: SubstitutionMap = HashMap::new();
417    mapping.insert(symbol_of_interest_expression, Arc::clone(&zero));
418    mapping.insert(other_symbol_expression, zero);
419    let coefficient = substitute(context, &derivative, &mapping);
420    if depends_on_surrogate(&coefficient, symbol_of_interest)
421        || depends_on_surrogate(&coefficient, other_symbol)
422    {
423        return None;
424    }
425    Some(coefficient)
426}

428fn subtract_symbol_term(
429    context: &Context,
430    target: &ExpressionReference,
431    coefficient: &ExpressionReference,
432    symbol_of_interest: SymbolIdentifier,
433) -> ExpressionReference {
434    // Compute target - coefficient * symbol_of_interest as an ASAE-form
435    // expression. Used to peel off the y' term and then the y term from
436    // the surrogate-form equation so the residual ends up with no
437    // remaining linear dependence on either.
438    let symbol_expression = context
439        .interner
440        .intern(ExpressionKind::Symbol(symbol_of_interest));
441    let product = builders::product(context, vec![Arc::clone(coefficient), symbol_expression]);
442    builders::subtract(context, Arc::clone(target), product)
443}

445fn depends_on_surrogate(
446    target: &ExpressionReference,
447    surrogate_identifier: SymbolIdentifier,
448) -> bool {
449    // Wrapper around the standard depends_on query keyed by a surrogate
450    // identifier. Mirrors the depends_on helper from queries.rs but stays
451    // local to keep the imports tight.
452    crate::expression::queries::depends_on(target, surrogate_identifier)
453}

455fn is_integer_zero(target: &ExpressionReference) -> bool {
456    // Predicate identifying the integer literal zero. The linear-first-
457    // order classifier needs this to detect a degenerate a(x) ≡ 0 case
458    // before dividing through.
459    use crate::exact_arithmetic::Integer;
460    matches!(&target.kind, ExpressionKind::Integer(value) if *value == Integer::ZERO)
461}

463pub(crate) fn contains_unevaluated_integral(target: &ExpressionReference) -> bool {
464    // Probe the result of `integrate` for an embedded `integral(...)`
465    // placeholder. Shared by the linear-first-order / exact / Bernoulli /
466    // Riccati classifier ladder
467    // so a stall partway through any integration is detected uniformly.
468    match &target.kind {
469        ExpressionKind::Integer(_)
470        | ExpressionKind::Rational(_)
471        | ExpressionKind::Symbol(_)
472        | ExpressionKind::Constant(_) => false,
473        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
474            children.iter().any(contains_unevaluated_integral)
475        }
476        ExpressionKind::Power(base, exponent) => {
477            contains_unevaluated_integral(base) || contains_unevaluated_integral(exponent)
478        }
479        ExpressionKind::Factorial(argument) => contains_unevaluated_integral(argument),
480        ExpressionKind::FunctionCall(_, arguments)
481        | ExpressionKind::Derivative(_, _, arguments) => {
482            arguments.iter().any(contains_unevaluated_integral)
483        }
484        ExpressionKind::Integral(_, _, _, _) => true,
485        ExpressionKind::Summation(summand, _, lower, upper) => {
486            contains_unevaluated_integral(summand)
487                || lower.as_ref().is_some_and(contains_unevaluated_integral)
488                || upper.as_ref().is_some_and(contains_unevaluated_integral)
489        }
490        ExpressionKind::LaplaceTransform(target_expression, _, _)
491        | ExpressionKind::InverseLaplaceTransform(target_expression, _, _)
492        | ExpressionKind::FourierTransform(target_expression, _, _)
493        | ExpressionKind::InverseFourierTransform(target_expression, _, _) => {
494            contains_unevaluated_integral(target_expression)
495        }
496        ExpressionKind::Piecewise(branches) => branches.iter().any(|(condition, value)| {
497            contains_unevaluated_integral(condition) || contains_unevaluated_integral(value)
498        }),
499        ExpressionKind::OrdinaryDifferentialEquation(equation, dependent_variable_call, _)
500        | ExpressionKind::RecurrenceEquation(equation, dependent_variable_call, _) => {
501            contains_unevaluated_integral(equation)
502                || contains_unevaluated_integral(dependent_variable_call)
503        }
504        _ => false,
505    }
