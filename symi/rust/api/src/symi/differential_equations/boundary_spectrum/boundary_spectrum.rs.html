1// Eigenparameter requests for a two-point ordinary boundary value problem.
2//
3// A request such as "for which nonzero solutions does `y'' + k^2 y = 0` satisfy `y(0) = 0` and
4// `y'(1) = 0`" has two unknowns of different kinds — a function and a parameter — so it is not a
5// solving request and no scalar solver accepts it. This boundary reads the eigenvalue problem out
6// of a validated ordinary request, hands the endpoint pair to the exact elementary spectral
7// analysis the separated Sturm-Liouville machinery already owns, and returns the indexed family
8// together with the independent verification of one mode.
9//
10// The analysis is reused rather than reimplemented, and its answer is not trusted on its own: the
11// returned eigenfunction is replayed against the equation and both endpoint operators, and a
12// family whose eigenfunction is the zero function fails that replay by construction. A single
13// eigenpair is never presented as the spectrum; the indexed family, its integer index domain, and
14// any separately recorded exceptional mode all travel in the result.
15//
16// What makes the family a spectrum rather than a list is the identity in the index: the
17// characteristic determinant of the endpoint pair vanishes at the family wave number for *every*
18// integer index, and that is proved here in the index symbol itself. Sampling two modes shows
19// that two modes are eigenpairs and nothing more; it is the symbolic identity that rules out a
20// family which happens to be right at the sampled indices. Two concrete replays are kept as well,
21// because they prove separately that the eigenfunction is not the zero function.
22//
23// The request asks for the eigenparameter, not for the wave number, and the two are not the same
24// unknown: the equation places a spectral value on the unknown function, and each eigenvalue of
25// that spectral value has its own eigenparameter preimages. Every reported preimage is proved to
26// realize the family eigenvalue by substitution, and a preimage proved to be zero is dropped — at
27// a zero wave number the endpoint problem has only the trivial solution, so zero is never a
28// member of the nontrivial spectrum however the algebra of the preimage happens to come out.
29//
30// Soundness of each entry and completeness of the set are separate claims, and the result carries
31// them separately. Completeness is taken from the set-shaped solve of the preimage equation,
32// whose finite-set shape is itself the certificate that no other eigenparameter realizes the
33// eigenvalue over the eigenparameter's domain; when that solve declines, the entries are still
34// reported and the completeness flag is false. The preimage equation is solved under the family's
35// own index domain, because the family is asserted only for indices in that domain, and a
36// solution written in terms of the index is otherwise stated for indices the family never
37// claimed.

59/// Public native enumeration named `ordinary_differential_equation_boundary_spectrum_decline`.
60#[derive(Debug, Clone, PartialEq, Eq)]
61pub enum OrdinaryDifferentialEquationBoundarySpectrumDecline {
62    /// The residual is not a second-order eigenvalue equation in the unknown function.
63    EquationIsNotAnEigenvalueProblem,
64    /// The coefficient of the unknown or of its second derivative still mentions the coordinate.
65    CoefficientDependsOnTheCoordinate,
66    /// The coefficient of the second derivative could vanish, so the equation has no fixed order.
67    LeadingCoefficientMayVanish,
68    /// The request does not carry exactly two homogeneous endpoint data.
69    TwoHomogeneousEndpointConditionsRequired,
70    /// The two endpoint data sit at the same point, or their order could not be proved.
71    EndpointOrderNotProved,
72    /// The endpoint pair is not one of the exactly analysable elementary families.
73    EndpointPairNotElementary,
74    /// The returned mode did not survive independent replay against the equation and endpoints.
75    ModeVerificationRejected,
76    /// The characteristic determinant was not proved to vanish on the family for a symbolic index.
77    FamilyIdentityNotProved,
78    /// The eigenparameter preimages of the family eigenvalue could not be solved for exactly.
79    EigenparameterPreimagesNotResolved,
80}

109    /// Stores whether no other eigenparameter over its domain realizes the family eigenvalue.

126/// Analyzes the exact spectrum of a two-point eigenvalue request in one coordinate.
127pub fn analyze_ordinary_differential_equation_boundary_spectrum(
128    context: &Context,
129    problem: &OrdinaryDifferentialEquationProblem,
130    spectral_parameter: SymbolIdentifier,
131) -> Result<
132    OrdinaryDifferentialEquationBoundarySpectrum,
133    OrdinaryDifferentialEquationBoundarySpectrumDecline,
134> {
135    let spectral_value = read_spectral_value(context, problem)?;
136    let endpoint_conditions = read_endpoint_conditions(context, problem)?;
137    let spectrum =
138        analyze_elementary_negative_second_derivative_spectrum(context, &endpoint_conditions)
139            .map_err(|_| {
140                OrdinaryDifferentialEquationBoundarySpectrumDecline::EndpointPairNotElementary
141            })?;
142    // Two concrete modes rather than the symbolic family: the endpoint residual of the family
143    // vanishes only because the index is an integer, which is a fact about the index domain and
144    // not an identity the simplifier can close in a free symbol. Replaying the two lowest modes
145    // of the declared domain proves the family is not the zero function and that its endpoint
146    // operators really do annihilate it, and it is two eigenpairs rather than one.
147    let mode_verifications =
148        replay_lowest_modes(context, &endpoint_conditions, &spectrum.indexed_family);
149    if mode_verifications.iter().any(|replay| {
150        replay.verification.verdict
151            != PartialDifferentialEquationSpectralModeVerificationVerdict::Verified
152    }) {
153        return Err(OrdinaryDifferentialEquationBoundarySpectrumDecline::ModeVerificationRejected);
154    }
155    let determinant_on_family = evaluate_determinant_on_family(context, &spectrum);
156    if crate::differential_equations::verification::prove_residual_zero(
157        context,
158        &determinant_on_family,
159    ) != crate::zero_equivalence::ZeroEquivalenceVerdict::TrueValue
160    {
161        return Err(OrdinaryDifferentialEquationBoundarySpectrumDecline::FamilyIdentityNotProved);
162    }
163    let eigenvalue = crate::simplify::simplify(
164        context,
165        ExpressionReference::clone(&spectrum.indexed_family.spectral_value),
166    );
167    let preimages = nontrivial_spectral_parameter_values(
168        context,
169        &spectral_value,
170        &spectrum.indexed_family,
171        &eigenvalue,
172        spectral_parameter,
173    )
174    .ok_or(
175        OrdinaryDifferentialEquationBoundarySpectrumDecline::EigenparameterPreimagesNotResolved,
176    )?;
177    Ok(OrdinaryDifferentialEquationBoundarySpectrum {
178        spectral_parameter,
179        spectral_value_in_parameter: spectral_value,
180        mode_index: spectrum.indexed_family.mode_index,
181        index_domain: spectrum.indexed_family.index_domain,
182        wave_number: spectrum.indexed_family.wave_number.clone(),
183        eigenfunction: spectrum.indexed_family.eigenfunction.clone(),
184        characteristic_determinant: spectrum.boundary_matrix.determinant.clone(),
185        exceptional_mode: spectrum.exceptional_zero_mode.clone(),
186        mode_verifications,
187        eigenvalue,
188        characteristic_determinant_on_family: determinant_on_family,
189        spectral_parameter_values: preimages.values,
190        spectral_parameter_values_are_complete: preimages.is_complete,
191    })
192}

194fn evaluate_determinant_on_family(
195    context: &Context,
196    spectrum: &crate::differential_equations::partial_differential_equations::spectral_boundary::PartialDifferentialEquationElementarySpectrum,
197) -> ExpressionReference {
198    let mut mapping = SubstitutionMap::new();
199    mapping.insert(
200        context
201            .interner
202            .intern(ExpressionKind::Symbol(spectrum.wave_number_symbol)),
203        spectrum.indexed_family.wave_number.clone(),
204    );
205    crate::simplify::simplify(
206        context,
207        substitute(context, &spectrum.boundary_matrix.determinant, &mapping),
208    )
209}

217fn nontrivial_spectral_parameter_values(
218    context: &Context,
219    spectral_value_in_parameter: &ExpressionReference,
220    family: &PartialDifferentialEquationSpectralModeFamily,
221    eigenvalue: &ExpressionReference,
222    spectral_parameter: SymbolIdentifier,
223) -> Option<NontrivialSpectralParameterPreimages> {
224    // The eigenparameter values are the preimages of the family eigenvalue under the spectral
225    // value the equation places on the unknown. The set-shaped solve is asked first, because a
226    // finite set is a completeness claim about the preimages while a root list is not; when it
227    // declines, the candidates come instead from the root-list solve together with the wave
228    // number and its reflection, which are preimages whenever the spectral value is even in the
229    // eigenparameter — the usual case, and the one in which a spectrum stated in a positive wave
230    // number silently drops half of the eigenparameters that realize it.
231    //
232    // Either way every candidate is proved to realize the eigenvalue by substitution, so each
233    // entry is sound on its own. Completeness survives that filter only when no candidate of a
234    // complete set was discarded for failing its proof: a preimage the solver claimed and the
235    // verifier could not confirm leaves the set unaccounted for. Dropping a preimage proved to be
236    // zero does not cost completeness, because the trivial eigenparameter is excluded from the
237    // nontrivial spectrum by definition rather than by not being found.
238    let residual = builders::subtract(
239        context,
240        ExpressionReference::clone(spectral_value_in_parameter),
241        ExpressionReference::clone(eigenvalue),
242    );
243    context
244        .with_temporary_assumptions(mode_index_domain_assumption(context, family), |context| {
245            resolve_preimages_over_the_index_domain(
246                context,
247                &residual,
248                &family.wave_number,
249                spectral_parameter,
250            )
251        })
252        .ok()
253        .flatten()
254}

256fn resolve_preimages_over_the_index_domain(
257    context: &Context,
258    residual: &ExpressionReference,
259    wave_number: &ExpressionReference,
260    spectral_parameter: SymbolIdentifier,
261) -> Option<NontrivialSpectralParameterPreimages> {
262    // Worker for nontrivial_spectral_parameter_values, running where the mode index is known to
263    // lie in the family's declared domain. A finite solution set is the solver's certificate that
264    // it enumerated every eigenparameter realizing the eigenvalue over that parameter's domain;
265    // every other set shape — a decline placeholder, an unbounded family — carries no such
266    // certificate and is not read as one.
267    let solution_set = crate::solve::solve_as_set(context, residual, spectral_parameter);
268    let (candidates, candidates_are_complete) =
269        match crate::solve::solution_set_to_vec(&solution_set) {
270            Some(complete) => (complete, true),
271            None => {
272                let mut candidates = crate::solve::solve(context, residual, spectral_parameter);
273                candidates.push(ExpressionReference::clone(wave_number));
274                candidates.push(builders::negate(
275                    context,
276                    ExpressionReference::clone(wave_number),
277                ));
278                (candidates, false)
279            }
280        };
281    let mut values: Vec<ExpressionReference> = Vec::new();
282    let mut every_candidate_accounted_for = true;
283    for candidate in candidates {
284        let candidate = crate::simplify::simplify(context, candidate);
285        if is_nonzero(&candidate, &context.symbol_table) == TruthValue::No {
286            continue;
287        }
288        if values.contains(&candidate) {
289            continue;
290        }
291        let mut mapping = SubstitutionMap::new();
292        mapping.insert(
293            context
294                .interner
295                .intern(ExpressionKind::Symbol(spectral_parameter)),
296            ExpressionReference::clone(&candidate),
297        );
298        let realized = substitute(context, residual, &mapping);
299        if crate::differential_equations::verification::prove_residual_zero(context, &realized)
300            != crate::zero_equivalence::ZeroEquivalenceVerdict::TrueValue
301        {
302            every_candidate_accounted_for = false;
303            continue;
304        }
305        values.push(candidate);
306    }
307    (!values.is_empty()).then_some(NontrivialSpectralParameterPreimages {
308        is_complete: candidates_are_complete && every_candidate_accounted_for,
309        values,
310    })
311}

313fn mode_index_domain_assumption(
314    context: &Context,
315    family: &PartialDifferentialEquationSpectralModeFamily,
316) -> AssumptionFormula {
317    // Restate the family's own declared index domain as a fact about its index symbol. The family
318    // is asserted for exactly these indices, so this adds nothing to what the spectrum already
319    // claims; it makes the fact available to the solver and the simplifier, which otherwise read
320    // the fresh index as an arbitrary complex symbol and cannot, for instance, reduce the square
321    // root of a squared wave number back to the wave number.
322    let subject = context
323        .interner
324        .intern(ExpressionKind::Symbol(family.mode_index));
325    let predicate = match family.index_domain {
326        PartialDifferentialEquationSpectralIndexDomain::PositiveInteger => {
327            CurrentScalarPredicate::PositiveInteger
328        }
329        PartialDifferentialEquationSpectralIndexDomain::NonnegativeInteger => {
330            CurrentScalarPredicate::Natural
331        }
332    };
333    AssumptionFormula::atom(PredicateIdentifier::CurrentScalar(predicate), subject)
334}

336fn replay_lowest_modes(
337    context: &Context,
338    endpoint_conditions: &PartialDifferentialEquationEndpointConditions,
339    family: &PartialDifferentialEquationSpectralModeFamily,
340) -> Vec<OrdinaryDifferentialEquationSpectralModeReplay> {
341    let first_index = match family.index_domain {
342        PartialDifferentialEquationSpectralIndexDomain::PositiveInteger => 1,
343        PartialDifferentialEquationSpectralIndexDomain::NonnegativeInteger => 0,
344    };
345    let index_expression = context
346        .interner
347        .intern(ExpressionKind::Symbol(family.mode_index));
348    (first_index..first_index + 2)
349        .map(|value| {
350            let mode_index_value = builders::integer_value(context, value);
351            let mut mapping = SubstitutionMap::new();
352            mapping.insert(index_expression.clone(), mode_index_value.clone());
353            let wave_number = substitute(context, &family.wave_number, &mapping);
354            let eigenfunction = substitute(context, &family.eigenfunction, &mapping);
355            let verification = verify_partial_differential_equation_elementary_spectral_mode(
356                context,
357                endpoint_conditions,
358                &eigenfunction,
359                &wave_number,
360            );
361            OrdinaryDifferentialEquationSpectralModeReplay {
362                mode_index_value,
363                wave_number,
364                eigenfunction,
365                verification,
366            }
367        })
368        .collect()
369}

371fn read_spectral_value(
372    context: &Context,
373    problem: &OrdinaryDifferentialEquationProblem,
374) -> Result<ExpressionReference, OrdinaryDifferentialEquationBoundarySpectrumDecline> {
375    // Reads `alpha y'' + beta y = 0` off the residual by replacing the two occurrences with
376    // working symbols and differentiating. Anything the residual carries beyond those two terms
377    // survives the subtraction and refuses the request, so a first-order term or a forcing term
378    // is a decline rather than a silently dropped part of the equation.
379    if problem.order() != 2 || problem.signature().arguments.len() != 1 {
380        return Err(
381            OrdinaryDifferentialEquationBoundarySpectrumDecline::EquationIsNotAnEigenvalueProblem,
382        );
383    }
384    let coordinate = problem.independent_variable();
385    let coordinate_expression = context.interner.intern(ExpressionKind::Symbol(coordinate));
386    let value_call = builders::function_call(
387        context,
388        problem.dependent_function(),
389        vec![coordinate_expression.clone()],
390    );
391    let second_derivative = builders::derivative(
392        context,
393        problem.dependent_function(),
394        vec![2],
395        vec![coordinate_expression],
396    );
397    let value_symbol = context.symbol_table.fresh("spectral_value_placeholder");
398    let derivative_symbol = context
399        .symbol_table
400        .fresh("spectral_derivative_placeholder");
401    let mut mapping = SubstitutionMap::new();
402    mapping.insert(
403        value_call,
404        context
405            .interner
406            .intern(ExpressionKind::Symbol(value_symbol)),
407    );
408    mapping.insert(
409        second_derivative,
410        context
411            .interner
412            .intern(ExpressionKind::Symbol(derivative_symbol)),
413    );
414    let replaced = substitute(context, problem.equation(), &mapping);
415    let mut remaining = Vec::new();
416    collect_dependent_occurrences(&replaced, problem.signature(), &mut remaining);
417    if !remaining.is_empty() {
418        return Err(
419            OrdinaryDifferentialEquationBoundarySpectrumDecline::EquationIsNotAnEigenvalueProblem,
420        );
421    }
422    let value_coefficient = crate::simplify::simplify(
423        context,
424        crate::expression::differentiate::differentiate(context, &replaced, value_symbol),
425    );
426    let derivative_coefficient = crate::simplify::simplify(
427        context,
428        crate::expression::differentiate::differentiate(context, &replaced, derivative_symbol),
429    );
430    let reconstructed = builders::sum(
431        context,
432        vec![
433            builders::product(
434                context,
435                vec![
436                    value_coefficient.clone(),
437                    context
438                        .interner
439                        .intern(ExpressionKind::Symbol(value_symbol)),
440                ],
441            ),
442            builders::product(
443                context,
444                vec![
445                    derivative_coefficient.clone(),
446                    context
447                        .interner
448                        .intern(ExpressionKind::Symbol(derivative_symbol)),
449                ],
450            ),
451        ],
452    );
453    let difference = builders::subtract(context, replaced, reconstructed);
454    if crate::differential_equations::verification::prove_residual_zero(context, &difference)
455        != crate::zero_equivalence::ZeroEquivalenceVerdict::TrueValue
456    {
457        return Err(
458            OrdinaryDifferentialEquationBoundarySpectrumDecline::EquationIsNotAnEigenvalueProblem,
459        );
460    }
461    if contains_symbol(&value_coefficient, coordinate)
462        || contains_symbol(&derivative_coefficient, coordinate)
463    {
464        return Err(
465            OrdinaryDifferentialEquationBoundarySpectrumDecline::CoefficientDependsOnTheCoordinate,
466        );
467    }
468    if is_nonzero(&derivative_coefficient, &context.symbol_table) != TruthValue::Yes {
469        return Err(
470            OrdinaryDifferentialEquationBoundarySpectrumDecline::LeadingCoefficientMayVanish,
471        );
472    }
473    Ok(builders::divide(
474        context,
475        value_coefficient,
476        derivative_coefficient,
477    ))
478}

480fn read_endpoint_conditions(
481    context: &Context,
482    problem: &OrdinaryDifferentialEquationProblem,
483) -> Result<
484    PartialDifferentialEquationEndpointConditions,
485    OrdinaryDifferentialEquationBoundarySpectrumDecline,
486> {
487    let zero = builders::zero(context);
488    let conditions = problem.conditions();
489    if conditions.len() != 2 || conditions.iter().any(|condition| condition.value != zero) {
490        return Err(
491            OrdinaryDifferentialEquationBoundarySpectrumDecline::TwoHomogeneousEndpointConditionsRequired,
492        );
493    }
494    let first = &conditions[0];
495    let second = &conditions[1];
496    let separation = builders::subtract(context, second.point.clone(), first.point.clone());
497    let ordered = is_positive(&separation, &context.symbol_table);
498    let (lower, upper) = match ordered {
499        TruthValue::Yes => (first, second),
500        TruthValue::No => {
501            let reversed = builders::subtract(context, first.point.clone(), second.point.clone());
502            if is_positive(&reversed, &context.symbol_table) != TruthValue::Yes {
503                return Err(
504                    OrdinaryDifferentialEquationBoundarySpectrumDecline::EndpointOrderNotProved,
505                );
506            }
507            (second, first)
508        }
509        TruthValue::Unknown => {
510            return Err(
511                OrdinaryDifferentialEquationBoundarySpectrumDecline::EndpointOrderNotProved,
512            );
513        }
514    };
515    let lower_condition = endpoint_condition(context, lower.derivative_order, &lower.point)?;
516    let upper_condition = endpoint_condition(context, upper.derivative_order, &upper.point)?;
517    Ok(PartialDifferentialEquationEndpointConditions {
518        spectral_coordinate: problem.independent_variable(),
519        non_spectral_coordinate: context.symbol_table.fresh("spectral_spectator_coordinate"),
520        lower_endpoint: lower.point.clone(),
521        upper_endpoint: upper.point.clone(),
522        lower_condition,
523        upper_condition,
524    })
525}

527fn endpoint_condition(
528    context: &Context,
529    derivative_order: usize,
530    point: &ExpressionReference,
531) -> Result<
532    PartialDifferentialEquationEndpointCondition,
533    OrdinaryDifferentialEquationBoundarySpectrumDecline,
534> {
535    let one = builders::one(context);
536    let zero = builders::zero(context);
537    let (kind, value_coefficient, derivative_coefficient) = match derivative_order {
538        0 => (
539            PartialDifferentialEquationEndpointConditionKind::Dirichlet,
540            one,
541            zero.clone(),
542        ),
543        1 => (
544            PartialDifferentialEquationEndpointConditionKind::Neumann,
545            zero.clone(),
546            one,
547        ),
548        _ => {
549            return Err(
550                OrdinaryDifferentialEquationBoundarySpectrumDecline::TwoHomogeneousEndpointConditionsRequired,
551            );
552        }
553    };
554    Ok(PartialDifferentialEquationEndpointCondition {
555        endpoint: point.clone(),
556        value_coefficient,
557        derivative_coefficient,
558        kind,
559        source_residual: zero,
560    })
