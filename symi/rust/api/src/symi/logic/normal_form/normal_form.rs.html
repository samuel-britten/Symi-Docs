11/// Performs the advanced native `disjunctive_normal_form` operation.
12pub fn disjunctive_normal_form(
13    context: &Context,
14    reference: &ExpressionReference,
15) -> ExpressionReference {
16    // Produce a disjunctive normal form (an OR of AND-terms over literals).
17    // The derived connectives are eliminated, negations are pushed to the atoms,
18    // the formula is distributed into product terms, and the term set is reduced
19    // (contradictory terms dropped, duplicates and subsumed terms absorbed) before
20    // it is rebuilt through the auto-simplifying builders.
21    let (atoms, reduced) = reduced_disjunctive_sets(context, reference);
22    build_clauses(
23        context,
24        reduced,
25        &atoms,
26        BuiltinFunction::LogicalOr,
27        BuiltinFunction::LogicalAnd,
28    )
29}

31/// Performs the advanced native `conjunctive_normal_form` operation.
32pub fn conjunctive_normal_form(
33    context: &Context,
34    reference: &ExpressionReference,
35) -> ExpressionReference {
36    // Produce a conjunctive normal form (an AND of OR-clauses over
37    // literals), the exact dual of disjunctive_normal_form.
38    let (atoms, reduced) = reduced_conjunctive_sets(context, reference);
39    build_clauses(
40        context,
41        reduced,
42        &atoms,
43        BuiltinFunction::LogicalAnd,
44        BuiltinFunction::LogicalOr,
45    )
46}

48pub(crate) fn reduced_disjunctive_sets(
49    context: &Context,
50    reference: &ExpressionReference,
51) -> (Vec<ExpressionReference>, Vec<BTreeSet<(usize, bool)>>) {
52    // Shared front end for the DNF path: eliminate derived connectives, push
53    // negations in, distribute to product terms, and reduce to a clause-set cover.
54    // The Espresso minimiser consumes the same reduced sets.
55    let eliminated = eliminate_derived(context, reference);
56    let normalized = negation_normal_form(context, &eliminated);
57    let mut atoms = Vec::new();
58    collect_atoms(&normalized, &mut atoms);
59    let terms = collect_dnf_terms(&normalized);
60    let reduced = reduce_clause_sets(&terms, &atoms);
61    (atoms, reduced)
62}

64pub(crate) fn reduced_conjunctive_sets(
65    context: &Context,
66    reference: &ExpressionReference,
67) -> (Vec<ExpressionReference>, Vec<BTreeSet<(usize, bool)>>) {
68    // Shared front end for the CNF path, dual to reduced_disjunctive_sets.
69    let eliminated = eliminate_derived(context, reference);
70    let normalized = negation_normal_form(context, &eliminated);
71    let mut atoms = Vec::new();
72    collect_atoms(&normalized, &mut atoms);
73    let clauses = collect_cnf_clauses(&normalized);
74    let reduced = reduce_clause_sets(&clauses, &atoms);
75    (atoms, reduced)
76}

78pub(crate) fn collect_atoms(reference: &ExpressionReference, atoms: &mut Vec<ExpressionReference>) {
79    // Gather the distinct propositional atoms of a formula: every maximal
80    // sub-expression that is neither a connective call nor a Boolean constant.
81    // Relations, symbols, and any other opaque sub-expression are atoms. Hash
82    // consing makes structurally equal atoms the same Arc, so pointer identity is
83    // an exact membership test.
84    match &reference.kind {
85        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalTrue)
86        | ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalFalse) => {}
87        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments)
88            if super::is_logical_connective(*builtin) =>
89        {
90            for argument in arguments {
91                collect_atoms(argument, atoms);
92            }
93        }
94        _ => {
95            if !atoms
96                .iter()
97                .any(|existing| Arc::ptr_eq(existing, reference))
98            {
99                atoms.push(Arc::clone(reference));
100            }
101        }
102    }
103}

105pub(crate) fn atom_index(atoms: &[ExpressionReference], atom: &ExpressionReference) -> usize {
106    // Resolve an atom expression to its index in the shared atom table.
107    atoms
108        .iter()
109        .position(|existing| Arc::ptr_eq(existing, atom))
110        .expect("atom must be present in the table built from the same formula")
111}

113pub(crate) fn evaluate_under_assignment(
114    reference: &ExpressionReference,
115    atoms: &[ExpressionReference],
116    assignment: u64,
117) -> bool {
118    // Classical two-valued evaluation of a connective tree under a bit-packed
119    // assignment of the atom table (bit i is the value of atoms[i]). Used by the
120    // truth-table oracle and the minimiser to enumerate minterms.
121    match &reference.kind {
122        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalTrue) => true,
123        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalFalse) => false,
124        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments)
125            if super::is_logical_connective(*builtin) =>
126        {
127            match builtin {
128                BuiltinFunction::LogicalAnd => arguments
129                    .iter()
130                    .all(|argument| evaluate_under_assignment(argument, atoms, assignment)),
131                BuiltinFunction::LogicalOr => arguments
132                    .iter()
133                    .any(|argument| evaluate_under_assignment(argument, atoms, assignment)),
134                BuiltinFunction::LogicalNot => {
135                    !evaluate_under_assignment(&arguments[0], atoms, assignment)
136                }
137                BuiltinFunction::Implies => {
138                    !evaluate_under_assignment(&arguments[0], atoms, assignment)
139                        || evaluate_under_assignment(&arguments[1], atoms, assignment)
140                }
141                BuiltinFunction::ExclusiveOr => arguments.iter().fold(false, |parity, argument| {
142                    parity ^ evaluate_under_assignment(argument, atoms, assignment)
143                }),
144                BuiltinFunction::Equivalent => {
145                    let mut iterator = arguments.iter();
146                    match iterator.next() {
147                        None => true,
148                        Some(first) => {
149                            let first_value = evaluate_under_assignment(first, atoms, assignment);
150                            iterator.all(|argument| {
151                                evaluate_under_assignment(argument, atoms, assignment)
152                                    == first_value
153                            })
154                        }
155                    }
156                }
157                _ => false,
158            }
159        }
160        _ => {
161            let index = atom_index(atoms, reference);
162            (assignment >> index) & 1 == 1
163        }
164    }
165}

167fn eliminate_derived(context: &Context, reference: &ExpressionReference) -> ExpressionReference {
168    // The DNF/CNF front end shares one basis eliminator with the public directed
169    // operation. Normal-form construction is itself unbounded, so it passes `None`
170    // limits: the shared kernel then performs no preflight and refuses nothing,
171    // which keeps DNF and CNF exactly as admissive as they were before extraction.
172    // A `logical_not` further down this pipeline still requires the derived heads to
173    // be gone, so an atomic refusal here would leave the distributor with a formula
174    // it cannot key.
175    match super::connective_conversion::eliminate_within(context, reference, None) {
176        super::connective_conversion::LogicalConversionOutcome::Converted(result) => result,
177        _ => Arc::clone(reference),
178    }
179}

181fn negation_normal_form(context: &Context, reference: &ExpressionReference) -> ExpressionReference {
182    // Push every negation down to the atoms over a formula already restricted to
183    // and / or / not. The result has logical_not only directly on atoms.
184    match &reference.kind {
185        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) => {
186            match builtin {
187                BuiltinFunction::LogicalNot if arguments.len() == 1 => {
188                    push_negation(context, &arguments[0])
189                }
190                BuiltinFunction::LogicalAnd => {
191                    let children = arguments
192                        .iter()
193                        .map(|argument| negation_normal_form(context, argument))
194                        .collect();
195                    builders::function_call(context, and_head(), children)
196                }
197                BuiltinFunction::LogicalOr => {
198                    let children = arguments
199                        .iter()
200                        .map(|argument| negation_normal_form(context, argument))
201                        .collect();
202                    builders::function_call(context, or_head(), children)
203                }
204                _ => Arc::clone(reference),
205            }
206        }
207        _ => Arc::clone(reference),
208    }
209}

211fn push_negation(context: &Context, operand: &ExpressionReference) -> ExpressionReference {
212    // Negation-normal-form of ¬operand: De Morgan on and / or, double-negation
213    // cancellation, constant folding, and a bare negated literal otherwise.
214    match &operand.kind {
215        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalTrue) => {
216            constant(context, false)
217        }
218        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalFalse) => {
219            constant(context, true)
220        }
221        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) => {
222            match builtin {
223                BuiltinFunction::LogicalNot if arguments.len() == 1 => {
224                    negation_normal_form(context, &arguments[0])
225                }
226                BuiltinFunction::LogicalAnd => {
227                    let children = arguments
228                        .iter()
229                        .map(|argument| push_negation(context, argument))
230                        .collect();
231                    builders::function_call(context, or_head(), children)
232                }
233                BuiltinFunction::LogicalOr => {
234                    let children = arguments
235                        .iter()
236                        .map(|argument| push_negation(context, argument))
237                        .collect();
238                    builders::function_call(context, and_head(), children)
239                }
240                _ => negation(context, operand),
241            }
242        }
243        _ => negation(context, operand),
244    }
245}

247fn collect_dnf_terms(reference: &ExpressionReference) -> Vec<Vec<ExpressionReference>> {
248    // Distribute a negation-normal-form formula into a sum of products: an outer
249    // OR of AND-terms, each term a list of literal expressions.
250    match &reference.kind {
251        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalTrue) => {
252            vec![Vec::new()]
253        }
254        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalFalse) => Vec::new(),
255        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) => {
256            match builtin {
257                BuiltinFunction::LogicalOr => {
258                    let mut terms = Vec::new();
259                    for argument in arguments {
260                        terms.extend(collect_dnf_terms(argument));
261                    }
262                    terms
263                }
264                BuiltinFunction::LogicalAnd => {
265                    let mut terms = vec![Vec::new()];
266                    for argument in arguments {
267                        let factor_terms = collect_dnf_terms(argument);
268                        terms = cartesian_combine(&terms, &factor_terms);
269                    }
270                    terms
271                }
272                _ => vec![vec![Arc::clone(reference)]],
273            }
274        }
275        _ => vec![vec![Arc::clone(reference)]],
276    }
277}

279fn collect_cnf_clauses(reference: &ExpressionReference) -> Vec<Vec<ExpressionReference>> {
280    // The dual distribution: a product of sums, an outer AND of OR-clauses.
281    match &reference.kind {
282        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalFalse) => {
283            vec![Vec::new()]
284        }
285        ExpressionKind::Constant(crate::expression::BuiltinConstant::LogicalTrue) => Vec::new(),
286        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) => {
287            match builtin {
288                BuiltinFunction::LogicalAnd => {
289                    let mut clauses = Vec::new();
290                    for argument in arguments {
291                        clauses.extend(collect_cnf_clauses(argument));
292                    }
293                    clauses
294                }
295                BuiltinFunction::LogicalOr => {
296                    let mut clauses = vec![Vec::new()];
297                    for argument in arguments {
298                        let summand_clauses = collect_cnf_clauses(argument);
299                        clauses = cartesian_combine(&clauses, &summand_clauses);
300                    }
301                    clauses
302                }
303                _ => vec![vec![Arc::clone(reference)]],
304            }
305        }
306        _ => vec![vec![Arc::clone(reference)]],
307    }
308}

310fn cartesian_combine(
311    left: &[Vec<ExpressionReference>],
312    right: &[Vec<ExpressionReference>],
313) -> Vec<Vec<ExpressionReference>> {
314    // Distribute one factor over an accumulated product: each accumulated clause
315    // is extended by every clause of the new factor.
316    let mut combined = Vec::with_capacity(left.len() * right.len());
317    for left_clause in left {
318        for right_clause in right {
319            let mut merged = left_clause.clone();
320            merged.extend(right_clause.iter().cloned());
321            combined.push(merged);
322        }
323    }
324    combined
325}

327fn literal_key(literal: &ExpressionReference, atoms: &[ExpressionReference]) -> (usize, bool) {
328    // Map a literal expression to (atom index, polarity). A logical_not(atom)
329    // literal has polarity false; a bare atom literal has polarity true.
330    if let ExpressionKind::FunctionCall(
331        FunctionIdentifier::Builtin(BuiltinFunction::LogicalNot),
332        inner,
333    ) = &literal.kind
334    {
335        if inner.len() == 1 {
336            return (atom_index(atoms, &inner[0]), false);
337        }
338    }
339    (atom_index(atoms, literal), true)
340}

342fn reduce_clause_sets(
343    clauses: &[Vec<ExpressionReference>],
344    atoms: &[ExpressionReference],
345) -> Vec<BTreeSet<(usize, bool)>> {
346    // Turn the distributed clause lists into deduplicated literal sets, drop the
347    // self-contradictory ones (an atom with both polarities), and absorb every
348    // clause that is a superset of another (the smaller subsumes it). The result
349    // is the reduced clause set ready to be rebuilt.
350    let mut sets: Vec<BTreeSet<(usize, bool)>> = Vec::new();
351    'clause: for clause in clauses {
352        let mut literals = BTreeSet::new();
353        for literal in clause {
354            let key = literal_key(literal, atoms);
355            if literals.contains(&(key.0, !key.1)) {
356                continue 'clause;
357            }
358            literals.insert(key);
359        }
360        if !sets.iter().any(|existing| *existing == literals) {
361            sets.push(literals);
362        }
363    }
364    let mut absorbed: Vec<BTreeSet<(usize, bool)>> = Vec::new();
365    for (position, candidate) in sets.iter().enumerate() {
366        let subsumed = sets.iter().enumerate().any(|(other_position, other)| {
367            other_position != position
368                && (other.len() < candidate.len()
369                    || (other.len() == candidate.len() && other_position < position))
370                && other.is_subset(candidate)
371        });
372        if !subsumed {
373            absorbed.push(candidate.clone());
374        }
375    }
376    absorbed
377}

379fn build_clauses(
380    context: &Context,
381    clauses: Vec<BTreeSet<(usize, bool)>>,
382    atoms: &[ExpressionReference],
383    outer: BuiltinFunction,
384    inner: BuiltinFunction,
385) -> ExpressionReference {
386    // Reconstruct a reduced clause set into the target normal form. An empty
387    // clause set is the identity of the outer connective (false for an OR-of-ANDs
388    // DNF, true for an AND-of-ORs CNF). Each surviving clause and the whole form
389    // are built through the auto-simplifying builders so the output is canonical.
390    if clauses.is_empty() {
391        return match outer {
392            BuiltinFunction::LogicalOr => constant(context, false),
393            _ => constant(context, true),
394        };
395    }
396    let mut clause_expressions = Vec::with_capacity(clauses.len());
397    for clause in clauses {
398        let mut literals = Vec::with_capacity(clause.len());
399        for (index, polarity) in clause {
400            let atom = Arc::clone(&atoms[index]);
401            let literal = if polarity {
402                atom
403            } else {
404                negation(context, &atom)
405            };
406            literals.push(literal);
407        }
408        clause_expressions.push(builders::function_call(
409            context,
410            FunctionIdentifier::Builtin(inner),
411            literals,
412        ));
413    }
414    builders::function_call(
415        context,
416        FunctionIdentifier::Builtin(outer),
417        clause_expressions,
418    )
419}
