11/// Public native constant `ESPRESSO_MAX_ATOMS`.
12pub const ESPRESSO_MAX_ATOMS: usize = 32;
13
14/// Performs the advanced native `minimize_sum_of_products` operation.
15pub fn minimize_sum_of_products(
16    context: &Context,
17    reference: &ExpressionReference,
18) -> Option<ExpressionReference> {
19    // Heuristic sum-of-products minimisation for formulas beyond the
20    // truth-table bound of the exact Quine–McCluskey path. It starts from the
21    // reduced disjunctive cover and applies the symbolic combine-and-absorb
22    // reduction (heuristic_cover): a good, not provably minimal, cover obtained by
23    // sound equivalence-preserving steps only. Returns None above the cube-width
24    // bound so the caller falls back to the plain DNF.
25    let (atoms, sets) = reduced_disjunctive_sets(context, reference);
26    if atoms.len() > ESPRESSO_MAX_ATOMS {
27        return None;
28    }
29    if sets.is_empty() {
30        return Some(constant(context, false));
31    }
32    if sets.iter().any(|set| set.is_empty()) {
33        return Some(constant(context, true));
34    }
35    let cover = heuristic_cover(cubes_of(&sets));
36    Some(build_two_level(
37        context,
38        &cover,
39        &atoms,
40        BuiltinFunction::LogicalOr,
41        BuiltinFunction::LogicalAnd,
42        false,
43    ))
44}

46/// Performs the advanced native `minimize_product_of_sums` operation.
47pub fn minimize_product_of_sums(
48    context: &Context,
49    reference: &ExpressionReference,
50) -> Option<ExpressionReference> {
51    // Heuristic product-of-sums minimisation, the dual of the
52    // sum-of-products path. The reduced conjunctive cover is combine-and-absorb
53    // reduced; the same single-opposite-literal merge ((a∨b) ∧ (a∨¬b) = a) is
54    // sound on OR-clauses, so the literal sets reuse the cube machinery directly
55    // (no polarity flip).
56    let (atoms, sets) = reduced_conjunctive_sets(context, reference);
57    if atoms.len() > ESPRESSO_MAX_ATOMS {
58        return None;
59    }
60    if sets.is_empty() {
61        return Some(constant(context, true));
62    }
63    if sets.iter().any(|set| set.is_empty()) {
64        return Some(constant(context, false));
65    }
66    let cover = heuristic_cover(cubes_of(&sets));
67    Some(build_two_level(
68        context,
69        &cover,
70        &atoms,
71        BuiltinFunction::LogicalAnd,
72        BuiltinFunction::LogicalOr,
73        false,
74    ))
75}
