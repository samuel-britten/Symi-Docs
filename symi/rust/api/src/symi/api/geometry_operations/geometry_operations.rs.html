8impl Session {
9    fn publish_vertex_angles(
10        &self,
11        scope: &OperationScope,
12        angles: Vec<(core_point_2d, crate::expression::ExpressionReference)>,
13    ) -> Result<Vec<VertexAngle>, ApiError> {
14        angles
15            .into_iter()
16            .map(|(vertex, angle)| {
17                Ok(VertexAngle {
18                    vertex: self.publish_point_2d(scope, vertex)?,
19                    angle: self.publish_expression(scope, angle)?,
20                })
21            })
22            .collect()
23    }
24}

26impl Segment2d {
27    /// Whether this segment's session has been reset since the segment was made.
28    pub fn is_stale(&self) -> bool {
29        self.generation != self.state.generation.get()
30    }
31
32    /// The equation of the perpendicular bisector in the two named coordinates.
33    pub fn perpendicular_bisector_equation<'a, 'b, XVariableType, YVariableType>(
34        &self,
35        variable_x: XVariableType,
36        variable_y: YVariableType,
37    ) -> Result<Expression, ApiError>
38    where
39        XVariableType: Into<VariableLike<'a>>,
40        YVariableType: Into<VariableLike<'b>>,
41    {
42        let owner = self.owner();
43        let operation = owner.begin_operation();
44        owner.validate_segment_2d(self)?;
45        let variable_x = owner.variable_identifier(variable_x.into())?;
46        let variable_y = owner.variable_identifier(variable_y.into())?;
47        let inner = self.inner.perpendicular_bisector_equation(
48            &mut operation.derived_context(),
49            variable_x,
50            variable_y,
51        );
52        owner.publish_expression(&operation, inner)
53    }
54}

63    pub fn interior_angles(&self) -> Result<Vec<VertexAngle>, ApiError> {
64        let owner = self.owner();
65        let operation = owner.begin_operation();
66        owner.validate_polygon_2d(self)?;
67        let angles = self.inner.interior_angles(&mut operation.derived_context());
68        owner.publish_vertex_angles(&operation, angles)

71    /// The number of vertices bounding this polygon.
72    pub fn vertex_count(&self) -> Result<usize, ApiError> {
73        let owner = self.owner();
74        owner.validate_polygon_2d(self)?;
75        Ok(self.inner.vertices.len())
