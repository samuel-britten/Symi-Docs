16fn is_negative_number(value: &ExpressionReference) -> bool {
17    // True when the leaf is a negative integer or negative rational literal.
18    // Used to recognise the radicand of an imaginary square-root atom of the
19    // form `power(negative, 1/2)`, which the LRT logarithmic-part construction
20    // emits as the surd surface of a complex-conjugate pair of resultant roots.
21    use crate::exact_arithmetic::compatibility::*;
22    use std::cmp::Ordering;
23    match &value.kind {
24        ExpressionKind::Integer(integer_value) => integer_value.sign_comparison() == Ordering::Less,
25        ExpressionKind::Rational(rational_value) => {
26            rational_value.sign_comparison() == Ordering::Less
27        }
28        _ => false,
29    }
30}

32fn is_one_half_exponent(exponent: &ExpressionReference) -> bool {
33    // True when the exponent is exactly the rational one half. The imaginary
34    // square-root atom `power(negative, 1/2)` squares to its radicand, which is
35    // the algebraic fact the real/imaginary decomposition in
36    // `split_real_imaginary` relies on; a different fractional power would not
37    // square to a rational and so is deliberately excluded here.
38    use crate::exact_arithmetic::Natural;
39    use crate::exact_arithmetic::compatibility::*;
40    use std::cmp::Ordering;
41    let ExpressionKind::Rational(value) = &exponent.kind else {
42        return false;
43    };
44    if value.sign_comparison() != Ordering::Greater {
45        return false;
46    }
47    let (numerator, denominator) = value.numerator_and_denominator_ref();
48    numerator == Natural::from(1u32) && denominator == Natural::from(2u32)
49}

51pub(crate) fn imaginary_atom_radicand(value: &ExpressionReference) -> Option<ExpressionReference> {
52    // Recognise an imaginary square-root atom `power(negative, 1/2)` and return
53    // its (negative) radicand. The atom equals `i * sqrt(-radicand)`, so its
54    // square is the radicand — the identity that powers the linear
55    // real/imaginary split of any polynomial in the atom.
56    let ExpressionKind::Power(base, exponent) = &value.kind else {
57        return None;
58    };
59    if is_negative_number(base) && is_one_half_exponent(exponent) {
60        Some(Arc::clone(base))
61    } else {
62        None
63    }
64}

66pub(crate) fn contains_imaginary_atom(value: &ExpressionReference) -> bool {
67    // Post-order recogniser for any embedded imaginary square-root atom. The
68    // presence of such an atom in an antiderivative is the signature of an LRT
69    // complex-logarithmic part: the closed form is correct (it differentiates
70    // back to the integrand) but is written with complex constants whose
71    // principal-branch evaluation defeats the Newton–Leibniz endpoint
72    // difference. The definite-integration entry consults this to decide
73    // whether the trig-rational Weierstrass-domain path is needed, and the
74    // realness gate in definite integration consults it to confirm a rational-function
75    // antiderivative came out real.
76    if imaginary_atom_radicand(value).is_some() {
77        return true;
78    }
79    use ExpressionKind::*;
80    match &value.kind {
81        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => false,
82        Sum(children) | Product(children) => children.iter().any(contains_imaginary_atom),
83        Power(base, exponent) => contains_imaginary_atom(base) || contains_imaginary_atom(exponent),
84        Factorial(argument) => contains_imaginary_atom(argument),
85        FunctionCall(_, arguments) | Derivative(_, _, arguments) => {
86            arguments.iter().any(contains_imaginary_atom)
87        }
88        _ => false,
89    }
90}

92pub(crate) fn contains_imaginary_unit(value: &ExpressionReference) -> bool {
93    // Post-order recogniser for the imaginary_unit constant anywhere in `value`.
94    // The LRT logarithmic part of a rational function whose Rothstein–Trager
95    // resultant has an irreducible factor of degree >= 3 with complex roots is
96    // emitted in the exp(i·k·pi/q) roots-of-unity representation, which carries
97    // the imaginary unit rather than a sqrt-of-negative atom. The realness gate
98    // in definite integration consults this alongside `contains_imaginary_atom` so both
99    // complex representations are caught.
100    if matches!(
101        &value.kind,
102        ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit)
103    ) {
104        return true;
105    }
106    use ExpressionKind::*;
107    match &value.kind {
108        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => false,
109        Sum(children) | Product(children) => children.iter().any(contains_imaginary_unit),
110        Power(base, exponent) => contains_imaginary_unit(base) || contains_imaginary_unit(exponent),
111        Factorial(argument) => contains_imaginary_unit(argument),
112        FunctionCall(_, arguments) | Derivative(_, _, arguments) => {
113            arguments.iter().any(contains_imaginary_unit)
114        }
115        _ => false,
116    }
117}

119pub(crate) fn contains_imaginary(value: &ExpressionReference) -> bool {
120    // True when `value` carries either complex representation the LRT
121    // logarithmic part can produce: a sqrt-of-negative imaginary atom (the
122    // degree-2-resultant-factor surd surface) or the imaginary_unit constant
123    // (the exp(i·theta) roots-of-unity surface of a degree->=3 irreducible
124    // resultant factor). The realness gate in definite integration is exactly its
125    // negation: a rational-function antiderivative is real iff this is false.
126    contains_imaginary_atom(value) || contains_imaginary_unit(value)
127}

129pub(crate) fn contains_atom(value: &ExpressionReference, atom: &ExpressionReference) -> bool {
130    // True when `atom` occurs anywhere in `value`, compared by hash-cons
131    // identity. The atom is an interned leaf shared across every occurrence, so
132    // pointer equality is an exact structural containment test.
133    if Arc::ptr_eq(value, atom) {
134        return true;
135    }
136    use ExpressionKind::*;
137    match &value.kind {
138        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => false,
139        Sum(children) | Product(children) => {
140            children.iter().any(|child| contains_atom(child, atom))
141        }
142        Power(base, exponent) => contains_atom(base, atom) || contains_atom(exponent, atom),
143        Factorial(argument) => contains_atom(argument, atom),
144        FunctionCall(_, arguments) | Derivative(_, _, arguments) => arguments
145            .iter()
146            .any(|argument| contains_atom(argument, atom)),
147        _ => false,
148    }
149}

151fn multiply_complex(
152    context: &Context,
153    left: &(ExpressionReference, ExpressionReference),
154    right: &(ExpressionReference, ExpressionReference),
155    radicand: &ExpressionReference,
156) -> (ExpressionReference, ExpressionReference) {
157    // Multiply two numbers written in the basis {1, s} where the atom `s`
158    // satisfies s^2 = radicand: (a + b s)(c + d s) = (a c + b d · radicand) +
159    // (a d + b c) s. Every coefficient arithmetic step routes through the
160    // auto-simplifying builders so the result stays in canonical ASAE form.
161    let (a, b) = left;
162    let (c, d) = right;
163    let a_times_c = builders::product(context, vec![Arc::clone(a), Arc::clone(c)]);
164    let b_times_d = builders::product(context, vec![Arc::clone(b), Arc::clone(d)]);
165    let b_times_d_times_radicand =
166        builders::product(context, vec![b_times_d, Arc::clone(radicand)]);
167    let real_part = builders::sum(context, vec![a_times_c, b_times_d_times_radicand]);
168    let a_times_d = builders::product(context, vec![Arc::clone(a), Arc::clone(d)]);
169    let b_times_c = builders::product(context, vec![Arc::clone(b), Arc::clone(c)]);
170    let imaginary_part = builders::sum(context, vec![a_times_d, b_times_c]);
171    (real_part, imaginary_part)
172}

174pub(crate) fn split_real_imaginary(
175    context: &Context,
176    value: &ExpressionReference,
177    atom: &ExpressionReference,
178    radicand: &ExpressionReference,
179) -> Option<(ExpressionReference, ExpressionReference)> {
180    // Decompose `value` as `real + imaginary · atom`, where `atom` squares to
181    // `radicand`, returning the two `atom`-free coefficients. Any subexpression
182    // not containing the atom is purely real and contributes (subexpression, 0).
183    // The atom itself contributes (0, 1). Sums add componentwise; products fold
184    // through `multiply_complex`; a power with a small non-negative integer
185    // exponent unrolls into repeated multiplication. Returns None when the atom
186    // sits inside a shape the linear decomposition cannot reduce (a
187    // transcendental call, a non-integer or symbolic power), so the caller can
188    // decline to realify rather than fabricate a wrong split.
189    if !contains_atom(value, atom) {
190        let zero = builders::zero(context);
191        return Some((Arc::clone(value), zero));
192    }
193    if Arc::ptr_eq(value, atom) {
194        let zero = builders::zero(context);
195        let one = builders::one(context);
196        return Some((zero, one));
197    }
198    match &value.kind {
199        ExpressionKind::Sum(children) => {
200            let mut real_part = builders::zero(context);
201            let mut imaginary_part = builders::zero(context);
202            for child in children.clone() {
203                let (child_real, child_imaginary) =
204                    split_real_imaginary(context, &child, atom, radicand)?;
205                real_part = builders::sum(context, vec![real_part, child_real]);
206                imaginary_part = builders::sum(context, vec![imaginary_part, child_imaginary]);
207            }
208            Some((real_part, imaginary_part))
209        }
210        ExpressionKind::Product(factors) => {
211            let one = builders::one(context);
212            let zero = builders::zero(context);
213            let mut accumulated = (one, zero);
214            for factor in factors.clone() {
215                let factor_split = split_real_imaginary(context, &factor, atom, radicand)?;
216                accumulated = multiply_complex(context, &accumulated, &factor_split, radicand);
217            }
218            Some(accumulated)
219        }
220        ExpressionKind::Power(base, exponent) => {
221            let ExpressionKind::Integer(exponent_value) = &exponent.kind else {
222                return None;
223            };
224            let repetitions = u32::try_from(exponent_value).ok()?;
225            if repetitions > 16 {
226                return None;
227            }
228            let base_split = split_real_imaginary(context, base, atom, radicand)?;
229            let one = builders::one(context);
230            let zero = builders::zero(context);
231            let mut accumulated = (one, zero);
232            for _ in 0..repetitions {
233                accumulated = multiply_complex(context, &accumulated, &base_split, radicand);
234            }
235            Some(accumulated)
236        }
237        _ => None,
238    }
239}

241pub(crate) fn build_real_logarithm_single(
242    context: &Context,
243    term: &ComplexLogarithmTerm,
244    imaginary_magnitude: &ExpressionReference,
245    variable: SymbolIdentifier,
246) -> ExpressionReference {
247    // Real form of a single self-conjugate complex logarithm term c·log(u)
248    // whose derivative is known (certified by the caller in exact field
249    // arithmetic) to be real: its algebraic conjugate has the same
250    // derivative, so c·log(u) differs from (c·log(u) + conj)/2 by an
251    // integration constant, and the halved conjugate-pair identity applies.
252    // This generalizes the LRT pair assembler to
253    // algebraic-function logarithms with pure-radical coefficients.
254    let one_half = builders::rational_value(context, 1, 2);
255    let halved_real = builders::product(
256        context,
257        vec![Arc::clone(&one_half), Arc::clone(&term.coefficient_real)],
258    );
259    let halved_imaginary = builders::product(
260        context,
261        vec![one_half, Arc::clone(&term.coefficient_imaginary)],
262    );
263    let halved_term = ComplexLogarithmTerm {
264        coefficient_real: halved_real,
265        coefficient_imaginary: halved_imaginary,
266        argument_real: Arc::clone(&term.argument_real),
267        argument_imaginary: Arc::clone(&term.argument_imaginary),
268    };
269    build_real_logarithm_pair(context, &halved_term, imaginary_magnitude, variable)
270}

272pub(crate) fn build_real_logarithm_pair(
273    context: &Context,
274    term: &ComplexLogarithmTerm,
275    imaginary_magnitude: &ExpressionReference,
276    variable: SymbolIdentifier,
277) -> ExpressionReference {
278    // Convert one complex-conjugate logarithm pair to its real form. With the
279    // shared atom written as `i · imaginary_magnitude`, the coefficient is
280    // `a + b·i` and the logarithm argument is `u + v·i`, where
281    // `a = coefficient_real`, `b = coefficient_imaginary · imaginary_magnitude`,
282    // `u = argument_real`, `v = argument_imaginary · imaginary_magnitude`. The
283    // pair `k·log(g) + conj(k)·log(conj(g))` equals `2·Re(k·log g) =
284    // a·log(u² + v²) − 2b·arctan(v/u)`. The construction is invariant under
285    // swapping the pair's two terms, so either representative yields the same
286    // real result. The logarithmic summand is dropped when `a` is zero, the
287    // pure-imaginary-coefficient case that covers every trigonometric-rational
288    // and real-denominator quadratic residue. The `log(u² + v²)` factor routes
289    // through the `antiderivative_logarithm` choke point so the integration
290    // log-domain policy is honored uniformly (the argument is a sum of squares,
291    // so the real-mode `abs` strips and the form is identical in both modes).
292    let real_coefficient = Arc::clone(&term.coefficient_real);
293    let imaginary_coefficient = builders::product(
294        context,
295        vec![
296            Arc::clone(&term.coefficient_imaginary),
297            Arc::clone(imaginary_magnitude),
298        ],
299    );
300    let argument_real = Arc::clone(&term.argument_real);
301    let argument_imaginary = builders::product(
302        context,
303        vec![
304            Arc::clone(&term.argument_imaginary),
305            Arc::clone(imaginary_magnitude),
306        ],
307    );
308    let arctangent_argument = builders::divide(
309        context,
310        Arc::clone(&argument_imaginary),
311        Arc::clone(&argument_real),
312    );
313    let arctangent_call = builders::function_call(
314        context,
315        FunctionIdentifier::Builtin(BuiltinFunction::Arctan),
316        vec![arctangent_argument],
317    );
318    let two = builders::integer_value(context, 2);
319    let two_imaginary_coefficient = builders::product(context, vec![two, imaginary_coefficient]);
320    let arctangent_term =
321        builders::product(context, vec![two_imaginary_coefficient, arctangent_call]);
322    let arctangent_contribution = builders::negate(context, arctangent_term);
323    if matches!(&real_coefficient.kind, ExpressionKind::Integer(integer_value) if *integer_value == crate::exact_arithmetic::integer_zero())
324    {
325        return arctangent_contribution;
326    }
327    let two_exponent = builders::integer_value(context, 2);
328    let argument_real_squared = builders::power(context, argument_real, two_exponent);
329    let two_exponent_again = builders::integer_value(context, 2);
330    let argument_imaginary_squared =
331        builders::power(context, argument_imaginary, two_exponent_again);
332    let modulus_squared = builders::sum(
333        context,
334        vec![argument_real_squared, argument_imaginary_squared],
335    );
336    let logarithm_call = super::antiderivative_logarithm(context, modulus_squared, variable);
337    let logarithm_term = builders::product(context, vec![real_coefficient, logarithm_call]);
338    builders::sum(context, vec![logarithm_term, arctangent_contribution])
339}
