32/// Performs the advanced native `integrate_in_algebraic_tower` operation.
33pub fn integrate_in_algebraic_tower(
34    context: &Context,
35    numerator: &Polynomial,
36    denominator: &Polynomial,
37    tower: &DifferentialField,
38) -> IntegrationResult {
39    // Entry point. Build the certified simple algebraic function
40    // field, run the Hermite reduction, and, when a nonzero squarefree
41    // residual remains, run the Trager logarithmic reduction. The three-way
42    // outcome is preserved exactly: a verified elementary decomposition is
43    // reified into an expression, a certified non-elementary residual
44    // produces the rigorous non-elementary verdict, and every gap remains
45    // `incomplete` so the caller emits the ordinary unevaluated integral.
46    if tower.extensions.len() != 1 || polynomial_is_zero(denominator) {
47        return IntegrationResult::Incomplete;
48    }
49    let generator = top_variable(tower);
50    let extension = match find_extension(tower, generator) {
51        Some(value) => value,
52        None => return IntegrationResult::Incomplete,
53    };
54    let defining_polynomial = match &extension.kind {
55        ExtensionKind::Algebraic(value) => value,
56        _ => return IntegrationResult::Incomplete,
57    };
58    let field = match make_simple_algebraic_function_field(
59        tower.base_variable,
60        generator,
61        defining_polynomial,
62        context.operation(),
63    ) {
64        Some(value) => value,
65        None => return IntegrationResult::Incomplete,
66    };
67    let target = match tower_rational_to_algebraic_function_element(numerator, denominator, &field)
68    {
69        Some(value) => value,
70        None => return IntegrationResult::Incomplete,
71    };
72    let reduction = match reduce_algebraic_function(&target, context.operation()) {
73        Some(value) => value,
74        None => return IntegrationResult::Incomplete,
75    };
76    if polynomial_is_zero(&reduction.residual_denominator) {
77        return IntegrationResult::Incomplete;
78    }
79    if is_zero(&reduction.residual) {
80        return IntegrationResult::Elementary(algebraic_function_element_to_expression(
81            context,
82            &reduction.derivative_part,
83        ));
84    }
85    let constant_generator = context
86        .symbol_table
87        .intern_symbol("algebraic_residue_radical");
88    let logarithmic = reduce_algebraic_logarithmic_part(
89        &context.algebraic_field_interner,
90        constant_generator,
91        &reduction.residual,
92    );
93    match logarithmic {
94        AlgebraicLogarithmicReductionResult::Elementary(decomposition) => {
95            match reify_logarithmic_reduction(
96                context,
97                &reduction.derivative_part,
98                &decomposition,
99                &field,
100                tower.base_variable,
101            ) {
102                Some(expression) => IntegrationResult::Elementary(expression),
103                None => IntegrationResult::Incomplete,
104            }
105        }
106        AlgebraicLogarithmicReductionResult::NonElementary(_) => IntegrationResult::NonElementary,
107        AlgebraicLogarithmicReductionResult::Incomplete => IntegrationResult::Incomplete,
108    }
109}

111fn reify_logarithmic_reduction(
112    context: &Context,
113    hermite_derivative_part: &crate::algebraic_function_field::AlgebraicFunctionFieldElement,
114    decomposition: &AlgebraicLogarithmicReduction,
115    field: &Arc<crate::algebraic_function_field::SimpleAlgebraicFunctionField>,
116    variable: SymbolIdentifier,
117) -> Option<ExpressionReference> {
118    // Assemble the complete antiderivative expression: the
119    // part, the odd de Rham primitive, the integrated even part (a rational
120    // function of the base variable, closed by the shipped complete rational
121    // integrator), and the verified logarithm terms with the shared
122    // LogToReal discipline for complex conjugate data.
123    let mut terms = Vec::new();
124    let hermite_expression =
125        algebraic_function_element_to_expression(context, hermite_derivative_part);
126    if !matches!(&hermite_expression.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::integer_zero())
127    {
128        terms.push(hermite_expression);
129    }
130    if !decomposition.odd_primitive_coefficient.is_zero() {
131        let mut primitive_element = crate::algebraic_function_field::field::zero_element(field);
132        primitive_element.representative[1] = decomposition.odd_primitive_coefficient.clone();
133        terms.push(algebraic_function_element_to_expression(
134            context,
135            &primitive_element,
136        ));
137    }
138    if !decomposition.even_part.is_zero() {
139        let even_expression = rational_function_to_expression(context, &decomposition.even_part);
140        let even_integral = crate::integrate::integrate(context, &even_expression, variable);
141        if matches!(even_integral.kind, ExpressionKind::Integral(_, _, _, _)) {
142            return None;
143        }
144        terms.push(even_integral);
145    }
146    for term in &decomposition.logarithms {
147        terms.push(reify_logarithm_term(context, term, field, variable)?);
148    }
149    if terms.is_empty() {
150        return Some(builders::zero(context));
151    }
152    if terms.len() == 1 {
153        let mut terms = terms;
154        return Some(terms.remove(0));
155    }
156    Some(builders::sum(context, terms))
157}

159fn reify_logarithm_term(
160    context: &Context,
161    term: &AlgebraicLogarithmTerm,
162    field: &Arc<crate::algebraic_function_field::SimpleAlgebraicFunctionField>,
163    variable: SymbolIdentifier,
164) -> Option<ExpressionReference> {
165    // Convert one verified logarithm term to its public expression. Rational
166    // constants reify directly through the antiderivative-logarithm choke
167    // point; real-radical constants emit the explicit conjugate sum; and
168    // imaginary-radical constants route through the shared LogToReal
169    // machinery — the full conjugate-pair identity for a term carrying an
170    // implicit conjugate partner, and the halved identity for a
171    // self-conjugate term. No branch-unsound logarithm identity is applied.
172    let divisor_multiple_rational = Rational::from(crate::exact_arithmetic::Integer::from(
173        term.divisor_multiple.clone(),
174    ));
175    match (&term.radical_radicand, &term.conjugate_closure) {
176        (None, _) | (Some(_), TermConjugateClosure::Rational) => {
177            let coefficient =
178                algebraic_element_rational_part(&term.coefficient)? / divisor_multiple_rational;
179            let argument =
180                algebraic_argument_expression(context, term, field, ConjugationMode::Direct)?;
181            let logarithm = super::antiderivative_logarithm(context, argument, variable);
182            let coefficient_expression = builders::rational_literal(context, coefficient);
183            Some(builders::product(
184                context,
185                vec![coefficient_expression, logarithm],
186            ))
187        }
188        (Some(radicand), closure) if *radicand > Rational::ZERO => {
189            // Real quadratic constants: emit c*log(u) + conjugate(c)*log(
190            // conjugate(u)) with explicit radical constants.
191            let scale = Rational::from(1) / divisor_multiple_rational;
192            let coefficient_expression = scaled_algebraic_coefficient_expression(
193                context,
194                &term.coefficient,
195                &scale,
196                ConjugationMode::Direct,
197            );
198            let argument =
199                algebraic_argument_expression(context, term, field, ConjugationMode::Direct)?;
200            let logarithm = super::antiderivative_logarithm(context, argument, variable);
201            let first = builders::product(context, vec![coefficient_expression, logarithm]);
202            if *closure == TermConjugateClosure::SelfConjugate {
203                return Some(first);
204            }
205            let conjugate_coefficient = scaled_algebraic_coefficient_expression(
206                context,
207                &term.coefficient,
208                &scale,
209                ConjugationMode::Conjugated,
210            );
211            let conjugate_argument =
212                algebraic_argument_expression(context, term, field, ConjugationMode::Conjugated)?;
213            let conjugate_logarithm =
214                super::antiderivative_logarithm(context, conjugate_argument, variable);
215            let second =
216                builders::product(context, vec![conjugate_coefficient, conjugate_logarithm]);
217            Some(builders::sum(context, vec![first, second]))
218        }
219        (Some(radicand), closure) => {
220            // Imaginary quadratic constants: w = i * sqrt(-radicand). Split
221            // the coefficient and the argument into their rational and
222            // w-coordinate parts and hand the certified pair to the shared
223            // real-form assembler.
224            let magnitude_radicand = -radicand.clone();
225            let magnitude_base = builders::rational_literal(context, magnitude_radicand);
226            let one_half_exponent = builders::rational_value(context, 1, 2);
227            let imaginary_magnitude = builders::power(context, magnitude_base, one_half_exponent);
228            let scale = Rational::from(1) / divisor_multiple_rational;
229            let coefficient_real = builders::rational_literal(
230                context,
231                algebraic_element_rational_part(&term.coefficient)? * scale.clone(),
232            );
233            let coefficient_imaginary = builders::rational_literal(
234                context,
235                algebraic_element_radical_part(&term.coefficient) * scale,
236            );
237            let (argument_real, argument_imaginary) =
238                split_argument_real_imaginary(context, term, field)?;
239            let complex_term = ComplexLogarithmTerm {
240                coefficient_real,
241                coefficient_imaginary,
242                argument_real,
243                argument_imaginary,
244            };
245            match closure {
246                TermConjugateClosure::SelfConjugate => Some(build_real_logarithm_single(
247                    context,
248                    &complex_term,
249                    &imaginary_magnitude,
250                    variable,
251                )),
252                _ => Some(build_real_logarithm_pair(
253                    context,
254                    &complex_term,
255                    &imaginary_magnitude,
256                    variable,
257                )),
258            }
259        }
260    }
261}

269fn conjugate_if_needed(value: &AlgebraicElement, mode: ConjugationMode) -> AlgebraicElement {
270    let mut result = value.clone();
271    if mode == ConjugationMode::Conjugated && result.representative.len() >= 2 {
272        result.representative[1] = -result.representative[1].clone();
273    }
274    result
275}

277fn algebraic_element_rational_part(value: &AlgebraicElement) -> Option<Rational> {

289fn scaled_algebraic_coefficient_expression(
290    context: &Context,
291    coefficient: &AlgebraicElement,
292    scale: &Rational,
293    mode: ConjugationMode,
294) -> ExpressionReference {
295    // Reify (coefficient / divisor_multiple), optionally conjugated, through
296    // the algebraic-number expression bridge so pure-radical constants print
297    // as radicals.
298    let adjusted = crate::algebraic_numbers::arithmetic::scale_by_rational(
299        &conjugate_if_needed(coefficient, mode),
300        scale,
301    );
302    algebraic_element_to_expression(context, &adjusted)
303}

305fn algebraic_argument_expression(
306    context: &Context,
307    term: &AlgebraicLogarithmTerm,
308    field: &Arc<crate::algebraic_function_field::SimpleAlgebraicFunctionField>,
309    mode: ConjugationMode,
310) -> Option<ExpressionReference> {
311    // The logarithm argument (a(x) + b(x) * alpha) / d(x) as an expression in
312    // the base variable and the tower generator, with every polynomial
313    // coefficient reified through the algebraic-number bridge.
314    let numerator_even = algebraic_polynomial_expression(
315        context,
316        &term.argument_numerator_even,
317        field.base_variable,
318        mode,
319    );
320    let numerator_odd = algebraic_polynomial_expression(
321        context,
322        &term.argument_numerator_odd,
323        field.base_variable,
324        mode,
325    );
326    let generator_expression = context
327        .interner
328        .intern(ExpressionKind::Symbol(field.generator));
329    let odd_product = builders::product(context, vec![numerator_odd, generator_expression]);
330    let numerator = builders::sum(context, vec![numerator_even, odd_product]);
331    let denominator = algebraic_polynomial_expression(
332        context,
333        &term.argument_denominator,
334        field.base_variable,
335        mode,
336    );
337    Some(builders::divide(context, numerator, denominator))
338}

340fn algebraic_polynomial_expression(
341    context: &Context,
342    coefficients: &[AlgebraicElement],
343    variable: SymbolIdentifier,
344    mode: ConjugationMode,
345) -> ExpressionReference {
346    // Reify a dense polynomial with algebraic-number coefficients.
347    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
348    let mut terms = Vec::new();
349    for (exponent, coefficient) in coefficients.iter().enumerate() {
350        let adjusted = conjugate_if_needed(coefficient, mode);
351        if crate::algebraic_numbers::element::is_zero(&adjusted) {
352            continue;
353        }
354        let coefficient_expression = algebraic_element_to_expression(context, &adjusted);
355        let term = if exponent == 0 {
356            coefficient_expression
357        } else {
358            let power = if exponent == 1 {
359                Arc::clone(&variable_expression)
360            } else {
361                let exponent_expression = builders::integer_value(context, exponent as i64);
362                builders::power(
363                    context,
364                    Arc::clone(&variable_expression),
365                    exponent_expression,
366                )
367            };
368            builders::product(context, vec![coefficient_expression, power])
369        };
370        terms.push(term);
371    }
372    if terms.is_empty() {
373        builders::zero(context)
374    } else if terms.len() == 1 {
375        terms.remove(0)
376    } else {
377        builders::sum(context, terms)
378    }
379}

381fn split_argument_real_imaginary(
382    context: &Context,
383    term: &AlgebraicLogarithmTerm,
384    field: &Arc<crate::algebraic_function_field::SimpleAlgebraicFunctionField>,
385) -> Option<(ExpressionReference, ExpressionReference)> {
386    // Split the argument U = (a + b*alpha)/d over Q(w) into exact real and
387    // w-coordinate parts. With N = N_re + N_im*w and d = d_re + d_im*w,
388    // rationalizing by the conjugate gives
389    //   U = ((N_re*d_re - N_im*d_im*D) + (N_im*d_re - N_re*d_im)*w)
390    //       / (d_re^2 - d_im^2*D),
391    // all of whose pieces are expressions in x and alpha with rational
392    // constants. The returned pair is (real part, w-coordinate); the caller
393    // multiplies the coordinate by the imaginary magnitude.
394    let radicand = term.radical_radicand.clone()?;
395    let (numerator_real, numerator_imaginary) = split_combined_numerator(context, term, field)?;
396    let (denominator_real_vector, denominator_imaginary_vector) =
397        split_algebraic_polynomial(&term.argument_denominator);
398    let denominator_real =
399        rational_dense_to_expression(context, &denominator_real_vector, field.base_variable);
400    if denominator_imaginary_vector
401        .iter()
402        .all(|value| *value == Rational::from(0))
403    {
404        let real = builders::divide(context, numerator_real, Arc::clone(&denominator_real));
405        let imaginary = builders::divide(context, numerator_imaginary, denominator_real);
406        return Some((real, imaginary));
407    }
408    let denominator_imaginary =
409        rational_dense_to_expression(context, &denominator_imaginary_vector, field.base_variable);
410    let radicand_expression = builders::rational_literal(context, radicand);
411    let two = builders::integer_value(context, 2);
412    let denominator_real_squared =
413        builders::power(context, Arc::clone(&denominator_real), Arc::clone(&two));
414    let denominator_imaginary_squared =
415        builders::power(context, Arc::clone(&denominator_imaginary), two);
416    let scaled_imaginary_squared = builders::product(
417        context,
418        vec![
419            denominator_imaginary_squared,
420            Arc::clone(&radicand_expression),
421        ],
422    );
423    let negated_scaled = builders::negate(context, scaled_imaginary_squared);
424    let norm = builders::sum(context, vec![denominator_real_squared, negated_scaled]);
425    let real_first = builders::product(
426        context,
427        vec![Arc::clone(&numerator_real), Arc::clone(&denominator_real)],
428    );
429    let real_second_product = builders::product(
430        context,
431        vec![
432            Arc::clone(&numerator_imaginary),
433            Arc::clone(&denominator_imaginary),
434            radicand_expression,
435        ],
436    );
437    let real_second = builders::negate(context, real_second_product);
438    let real_numerator = builders::sum(context, vec![real_first, real_second]);
439    let imaginary_first = builders::product(context, vec![numerator_imaginary, denominator_real]);
440    let imaginary_second_product =
441        builders::product(context, vec![numerator_real, denominator_imaginary]);
442    let imaginary_second = builders::negate(context, imaginary_second_product);
443    let imaginary_numerator = builders::sum(context, vec![imaginary_first, imaginary_second]);
444    let real = builders::divide(context, real_numerator, Arc::clone(&norm));
445    let imaginary = builders::divide(context, imaginary_numerator, norm);
446    Some((real, imaginary))
447}

449fn split_combined_numerator(
450    context: &Context,
451    term: &AlgebraicLogarithmTerm,
452    field: &Arc<crate::algebraic_function_field::SimpleAlgebraicFunctionField>,
453) -> Option<(ExpressionReference, ExpressionReference)> {
454    // Real and w-coordinate parts of a(x) + b(x)*alpha, where alpha is real
455    // on the integration domain and the w-coordinates come only from the
456    // algebraic coefficients.
457    let (even_real, even_imaginary) = split_algebraic_polynomial(&term.argument_numerator_even);
458    let (odd_real, odd_imaginary) = split_algebraic_polynomial(&term.argument_numerator_odd);
459    let generator_expression = context
460        .interner
461        .intern(ExpressionKind::Symbol(field.generator));
462    let even_real_expression =
463        rational_dense_to_expression(context, &even_real, field.base_variable);
464    let odd_real_expression = rational_dense_to_expression(context, &odd_real, field.base_variable);
465    let odd_real_product = builders::product(
466        context,
467        vec![odd_real_expression, Arc::clone(&generator_expression)],
468    );
469    let real = builders::sum(context, vec![even_real_expression, odd_real_product]);
470    let even_imaginary_expression =
471        rational_dense_to_expression(context, &even_imaginary, field.base_variable);
472    let odd_imaginary_expression =
473        rational_dense_to_expression(context, &odd_imaginary, field.base_variable);
474    let odd_imaginary_product = builders::product(
475        context,
476        vec![odd_imaginary_expression, generator_expression],
477    );
478    let imaginary = builders::sum(
479        context,
480        vec![even_imaginary_expression, odd_imaginary_product],
481    );
482    Some((real, imaginary))
483}

485fn split_algebraic_polynomial(coefficients: &[AlgebraicElement]) -> (Vec<Rational>, Vec<Rational>) {
486    // Coordinatewise split of a dense polynomial over Q(w) into its rational
487    // and w-parts.
488    let mut real = Vec::with_capacity(coefficients.len());
489    let mut imaginary = Vec::with_capacity(coefficients.len());
490    for coefficient in coefficients {
491        real.push(
492            coefficient
493                .representative
494                .first()
495                .cloned()
496                .unwrap_or(Rational::from(0)),
497        );
498        imaginary.push(
499            coefficient
500                .representative
501                .get(1)
502                .cloned()
503                .unwrap_or(Rational::from(0)),
504        );
505    }
506    (real, imaginary)
507}
