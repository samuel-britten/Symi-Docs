11const MAXIMUM_CANDIDATE_PARAMETERS: usize = 2;
12
13/// Evaluates a definite integral separately on the sign cells of one real parameter.
14pub(crate) fn integrate_over_parameter_sign_cells(
15    context: &Context,
16    target: &ExpressionReference,
17    variable: SymbolIdentifier,
18    lower_bound: &ExpressionReference,
19    upper_bound: &ExpressionReference,
20) -> Option<ExpressionReference> {
21    // A real parameter of the integrand whose sign is undecided can change which
22    // poles lie in a half-plane, where a primitive is singular, or whether a
23    // radical degenerates. The three cells p < 0, p = 0 and p > 0 partition the
24    // real parameter line; each cell the facts in force do not exclude is
25    // integrated on its own — the strict cells with the sign as a temporary
26    // fact, the zero cell by substitution. The value is the piecewise function
27    // over those cells, and it exists only if every admitted cell settles to an
28    // exact finite value: a cell that declines or diverges declines the whole
29    // request, because the integral is then undetermined for those parameters.
30    //
31    // At most two parameters are tried, in a fixed order. Within a cell the
32    // chosen parameter's sign is decided, so a nested split can only select a
33    // different parameter and the recursion terminates.
34    let mut candidates: Vec<SymbolIdentifier> = free_variables(target)
35        .into_iter()
36        .filter(|symbol| *symbol != variable)
37        .collect();
38    candidates.sort();
39    let mut attempted = 0;
40    for parameter in candidates {
41        if attempted == MAXIMUM_CANDIDATE_PARAMETERS {
42            break;
43        }
44        let Some(cells) = admitted_cells(context, parameter) else {
45            continue;
46        };
47        attempted += 1;
48        if let Some(value) = integrate_cells(
49            context,
50            target,
51            variable,
52            lower_bound,
53            upper_bound,
54            parameter,
55            &cells,
56        ) {
57            return Some(value);
58        }
59    }
60    None
61}

63#[derive(Clone, Copy, PartialEq, Eq)]
64enum SignCell {
65    Negative,
66    Zero,
67    Positive,
68}
69
70fn admitted_cells(context: &Context, parameter: SymbolIdentifier) -> Option<Vec<SignCell>> {
71    // The sign cells of a parameter proved real that the facts in force do not
72    // exclude, or None when the parameter is not proved real or its sign is
73    // already decided.
74    let symbol = context.interner.intern(ExpressionKind::Symbol(parameter));
75    let zero = builders::zero(context);
76    let is_real = crate::assumptions::range_evidence::proves_real(&context.symbol_table, &symbol)
77        || super::proves_relation(
78            context,
79            &builders::greater_than_or_equal(context, Arc::clone(&symbol), Arc::clone(&zero)),
80        )
81        || super::proves_relation(
82            context,
83            &builders::less_than_or_equal(context, Arc::clone(&symbol), Arc::clone(&zero)),
84        );
85    if !is_real {
86        return None;
87    }
88    let excludes_negative = super::proves_relation(
89        context,
90        &builders::greater_than_or_equal(context, Arc::clone(&symbol), Arc::clone(&zero)),
91    );
92    let excludes_positive = super::proves_relation(
93        context,
94        &builders::less_than_or_equal(context, Arc::clone(&symbol), Arc::clone(&zero)),
95    );
96    let excludes_zero =
97        crate::assumptions::range_evidence::nonzero_verdict(&context.symbol_table, &symbol)
98            == crate::assumptions::TruthValue::Yes
99            || super::proves_relation(context, &builders::not_equal(context, symbol, zero));
100    let cells: Vec<SignCell> = [
101        (SignCell::Negative, excludes_negative),
102        (SignCell::Zero, excludes_zero),
103        (SignCell::Positive, excludes_positive),
104    ]
105    .into_iter()
106    .filter_map(|(cell, excluded)| (!excluded).then_some(cell))
107    .collect();
108    (cells.len() >= 2).then_some(cells)
109}

111fn integrate_cells(
112    context: &Context,
113    target: &ExpressionReference,
114    variable: SymbolIdentifier,
115    lower_bound: &ExpressionReference,
116    upper_bound: &ExpressionReference,
117    parameter: SymbolIdentifier,
118    cells: &[SignCell],
119) -> Option<ExpressionReference> {
120    // Integrate every admitted cell and assemble the piecewise value, collapsing
121    // it to one expression when every cell agrees exactly.
122    let symbol = context.interner.intern(ExpressionKind::Symbol(parameter));
123    let zero = builders::zero(context);
124    let mut branches = Vec::with_capacity(cells.len());
125    for cell in cells {
126        let (condition, value) = match cell {
127            SignCell::Zero => {
128                let mut mapping: SubstitutionMap = HashMap::new();
129                mapping.insert(Arc::clone(&symbol), Arc::clone(&zero));
130                let value = super::integrate_definite(
131                    context,
132                    &substitute(context, target, &mapping),
133                    variable,
134                    substitute(context, lower_bound, &mapping),
135                    substitute(context, upper_bound, &mapping),
136                );
137                (
138                    builders::equal(context, Arc::clone(&symbol), Arc::clone(&zero)),
139                    value,
140                )
141            }
142            SignCell::Negative | SignCell::Positive => {
143                let condition = if *cell == SignCell::Negative {
144                    builders::less_than(context, Arc::clone(&symbol), Arc::clone(&zero))
145                } else {
146                    builders::greater_than(context, Arc::clone(&symbol), Arc::clone(&zero))
147                };
148                let formula = crate::assumptions::proposition_from_logical_expression(&condition)?;
149                let value = context
150                    .with_temporary_assumptions(formula, |scoped| {
151                        super::integrate_definite(
152                            scoped,
153                            target,
154                            variable,
155                            Arc::clone(lower_bound),
156                            Arc::clone(upper_bound),
157                        )
158                    })
159                    .ok()?;
160                (condition, value)
161            }
162        };
163        if !super::is_settled_definite_value(&value) {
164            return None;
165        }
166        branches.push((condition, crate::simplify::simplify(context, value)));
167    }
168    if branches
169        .iter()
170        .all(|(_, value)| value.kind == branches[0].1.kind)
171    {
172        return Some(Arc::clone(&branches[0].1));
173    }
174    Some(builders::piecewise(context, branches))
175}
