17/// Integrates a quadratic-radical integrand with parameter coefficients by an affine rescaling.
18pub(crate) fn integrate_by_quadratic_radical_rescaling(
19    context: &Context,
20    target: &ExpressionReference,
21    variable: SymbolIdentifier,
22    lower_bound: &ExpressionReference,
23    upper_bound: &ExpressionReference,
24    budget: IntegrationBudget,
25) -> Option<ExpressionReference> {
26    // Every half-integer power of a variable-dependent base shares one radicand
27    // R = A x^2 + B x + C whose coefficients involve parameters. Completing the
28    // square gives R = A (x - h)^2 + D with h = -B/(2A) and D = C - B^2/(4A).
29    // When the signs of A and D are proved, the real affine map
30    // x = h + σu with σ = sqrt(|D|/|A|) > 0 turns R into |D| (sgn(A) u^2 + sgn(D)),
31    // and since |D| is positive every half-integer power splits on the principal
32    // branch as R^(k/2) = |D|^(k/2) (sgn(A) u^2 + sgn(D))^(k/2). The positive scale
33    // preserves orientation, so the bounds map to (bound - h)/σ with infinities
34    // unchanged, and the Jacobian is the constant σ.
35    //
36    // The transformed integrand is a parameter-free multiple of an integrand
37    // whose radicand has rational coefficients, which the certified radical
38    // routes own; the parameters survive only in the constant multiple and in
39    // the mapped bounds. Anything else declines, so the rescaling never
40    // recurses into another parameterised radical.
41    if budget.substitution_exhausted() {
42        return None;
43    }
44    let radicand = shared_parameterised_radicand(context, target, variable)?;
45    let coefficients = quadratic_coefficients(context, &radicand, variable)?;
46    let [constant, linear, leading] = coefficients;
47    let leading_sign = proved_sign(context, &leading)?;
48    let two = builders::integer_value(context, 2);
49    let four = builders::integer_value(context, 4);
50    let shift = simplify_expanded(
51        context,
52        &builders::negate(
53            context,
54            builders::divide(
55                context,
56                Arc::clone(&linear),
57                builders::product(context, vec![Arc::clone(&two), Arc::clone(&leading)]),
58            ),
59        ),
60    );
61    let discriminant_part = simplify_expanded(
62        context,
63        &builders::subtract(
64            context,
65            Arc::clone(&constant),
66            builders::divide(
67                context,
68                builders::power(context, Arc::clone(&linear), Arc::clone(&two)),
69                builders::product(context, vec![four, Arc::clone(&leading)]),
70            ),
71        ),
72    );
73    let offset_sign = proved_sign(context, &discriminant_part)?;
74    let absolute_offset = signed(context, offset_sign, &discriminant_part);
75    let absolute_leading = signed(context, leading_sign, &leading);
76    let half = builders::rational_value(context, 1, 2);
77    let scale = crate::simplify::simplify(
78        context,
79        builders::power(
80            context,
81            builders::divide(context, Arc::clone(&absolute_offset), absolute_leading),
82            Arc::clone(&half),
83        ),
84    );
85    if positive_verdict(&context.symbol_table, &scale) != TruthValue::Yes {
86        return None;
87    }
88    let parameter = context
89        .symbol_table
90        .fresh("quadratic_radical_rescaling_parameter");
91    let parameter_expression = context.interner.intern(ExpressionKind::Symbol(parameter));
92    let normalized_radicand = builders::sum(
93        context,
94        vec![
95            builders::product(
96                context,
97                vec![
98                    builders::integer_value(context, leading_sign),
99                    builders::power(context, Arc::clone(&parameter_expression), two),
100                ],
101            ),
102            builders::integer_value(context, offset_sign),
103        ],
104    );
105    let split_powers = split_radical_powers(
106        context,
107        target,
108        &radicand,
109        &absolute_offset,
110        &normalized_radicand,
111    );
112    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
113    let image = builders::sum(
114        context,
115        vec![
116            Arc::clone(&shift),
117            builders::product(context, vec![Arc::clone(&scale), parameter_expression]),
118        ],
119    );
120    let mut mapping: SubstitutionMap = HashMap::new();
121    mapping.insert(variable_expression, image);
122    let transformed = builders::product(
123        context,
124        vec![
125            substitute(context, &split_powers, &mapping),
126            Arc::clone(&scale),
127        ],
128    );
129    let (multiplier, kernel) = separate_constant_factors(context, &transformed, parameter);
130    if free_variables(&kernel)
131        .iter()
132        .any(|symbol| *symbol != parameter)
133    {
134        return None;
135    }
136    let mapped_lower = mapped_bound(context, lower_bound, &shift, &scale);
137    let mapped_upper = mapped_bound(context, upper_bound, &shift, &scale);
138    let value = super::integrate_definite_with_budget(
139        context,
140        &kernel,
141        parameter,
142        mapped_lower,
143        mapped_upper,
144        budget.with_deeper_substitution(),
145    );
146    if crate::integrate::contains_integral_placeholder(&value)
147        || depends_on(&value, parameter)
148        || super::contains_nonfinite_leaf(&value)
149    {
150        return None;
151    }
152    Some(crate::simplify::simplify(
153        context,
154        builders::product(context, vec![multiplier, value]),
155    ))
156}

158fn shared_parameterised_radicand(
159    context: &Context,
160    target: &ExpressionReference,
161    variable: SymbolIdentifier,
162) -> Option<ExpressionReference> {
163    // The single base of every half-integer power that depends on the variable,
164    // provided it involves some other symbol.
165    let mut radicands: Vec<ExpressionReference> = Vec::new();
166    collect_half_integer_power_bases(target, variable, &mut radicands);
167    let first = radicands.first()?;
168    if radicands.iter().any(|radicand| radicand.kind != first.kind) {
169        return None;
170    }
171    let symbols = free_variables(first);
172    if symbols.iter().all(|symbol| *symbol == variable) {
173        return None;
174    }
175    let _ = context;
176    Some(Arc::clone(first))
177}

179fn collect_half_integer_power_bases(
180    target: &ExpressionReference,
181    variable: SymbolIdentifier,
182    radicands: &mut Vec<ExpressionReference>,
183) {
184    // Bases of powers with an odd-numerator, denominator-two exponent that depend on the variable.
185    match &target.kind {
186        ExpressionKind::Power(base, exponent) => {
187            if let ExpressionKind::Rational(value) = &exponent.kind
188                && *value.denominator_ref() == Natural::from(2u32)
189                && depends_on(base, variable)
190            {
191                radicands.push(Arc::clone(base));
192                return;
193            }
194            collect_half_integer_power_bases(base, variable, radicands);
195            collect_half_integer_power_bases(exponent, variable, radicands);
196        }
197        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
198            for child in children {
199                collect_half_integer_power_bases(child, variable, radicands);
200            }
201        }
202        ExpressionKind::FunctionCall(_, arguments) => {
203            for argument in arguments {
204                collect_half_integer_power_bases(argument, variable, radicands);
205            }
206        }
207        _ => {}
208    }
209}

211fn quadratic_coefficients(
212    context: &Context,
213    radicand: &ExpressionReference,
214    variable: SymbolIdentifier,
215) -> Option<[ExpressionReference; 3]> {
216    // The coefficients of a radicand that is exactly quadratic in the variable,
217    // lowest degree first, each free of the variable.
218    let expanded = crate::expand::expand(context, radicand);
219    let order =
220        crate::polynomial::conversion::extract_variables_in_order(&expanded, Some(variable));
221    let polynomial = crate::polynomial::conversion::expression_to_polynomial(&expanded, &order)?;
222    if crate::polynomial::degree(&polynomial) != Some(2) {
223        return None;
224    }
225    let mut coefficients = [
226        builders::zero(context),
227        builders::zero(context),
228        builders::zero(context),
229    ];
230    for (exponent, coefficient) in &polynomial.terms {
231        coefficients[*exponent] =
232            crate::polynomial::conversion::coefficient_to_expression(context, coefficient);
233    }
234    if coefficients
235        .iter()
236        .any(|coefficient| depends_on(coefficient, variable))
237    {
238        return None;
239    }
240    Some(coefficients)
241}

243fn proved_sign(context: &Context, value: &ExpressionReference) -> Option<i64> {
244    // 1 or -1 for a value proved positive or negative from the facts in force.
245    if positive_verdict(&context.symbol_table, value) == TruthValue::Yes
246        || super::proves_relation(
247            context,
248            &builders::greater_than(context, Arc::clone(value), builders::zero(context)),
249        )
250    {
251        return Some(1);
252    }
253    if negative_verdict(&context.symbol_table, value) == TruthValue::Yes
254        || super::proves_relation(
255            context,
256            &builders::less_than(context, Arc::clone(value), builders::zero(context)),
257        )
258    {
259        return Some(-1);
260    }
261    None
262}

264fn signed(context: &Context, sign: i64, value: &ExpressionReference) -> ExpressionReference {
265    // The absolute value of a value of known sign.
266    if sign > 0 {
267        Arc::clone(value)
268    } else {
269        simplify_expanded(context, &builders::negate(context, Arc::clone(value)))
270    }
271}

273fn simplify_expanded(context: &Context, value: &ExpressionReference) -> ExpressionReference {
274    // Expand and simplify.
275    crate::simplify::simplify(context, crate::expand::expand(context, value))
276}

278fn split_radical_powers(
279    context: &Context,
280    target: &ExpressionReference,
281    radicand: &ExpressionReference,
282    absolute_offset: &ExpressionReference,
283    normalized_radicand: &ExpressionReference,
284) -> ExpressionReference {
285    // Replace each power R^e of the shared radicand by |D|^e (sgn(A) u^2 + sgn(D))^e.
286    match &target.kind {
287        ExpressionKind::Power(base, exponent) if base.kind == radicand.kind => builders::product(
288            context,
289            vec![
290                builders::power(context, Arc::clone(absolute_offset), Arc::clone(exponent)),
291                builders::power(
292                    context,
293                    Arc::clone(normalized_radicand),
294                    Arc::clone(exponent),
295                ),
296            ],
297        ),
298        ExpressionKind::Power(base, exponent) => builders::power(
299            context,
300            split_radical_powers(
301                context,
302                base,
303                radicand,
304                absolute_offset,
305                normalized_radicand,
306            ),
307            Arc::clone(exponent),
308        ),
309        ExpressionKind::Sum(children) => builders::sum(
310            context,
311            children
312                .iter()
313                .map(|child| {
314                    split_radical_powers(
315                        context,
316                        child,
317                        radicand,
318                        absolute_offset,
319                        normalized_radicand,
320                    )
321                })
322                .collect(),
323        ),
324        ExpressionKind::Product(children) => builders::product(
325            context,
326            children
327                .iter()
328                .map(|child| {
329                    split_radical_powers(
330                        context,
331                        child,
332                        radicand,
333                        absolute_offset,
334                        normalized_radicand,
335                    )
336                })
337                .collect(),
338        ),
339        ExpressionKind::FunctionCall(function, arguments) => builders::function_call(
340            context,
341            *function,
342            arguments
343                .iter()
344                .map(|argument| {
345                    split_radical_powers(
346                        context,
347                        argument,
348                        radicand,
349                        absolute_offset,
350                        normalized_radicand,
351                    )
352                })
353                .collect(),
354        ),
355        _ => Arc::clone(target),
356    }
357}

359fn separate_constant_factors(
360    context: &Context,
361    transformed: &ExpressionReference,
362    parameter: SymbolIdentifier,
363) -> (ExpressionReference, ExpressionReference) {
364    // Split a product into the factors free of the new variable and the rest.
365    let factors = match &transformed.kind {
366        ExpressionKind::Product(children) => children.clone(),
367        _ => vec![Arc::clone(transformed)],
368    };
369    let (constant, kernel): (Vec<_>, Vec<_>) = factors
370        .into_iter()
371        .partition(|factor| !depends_on(factor, parameter));
372    (
373        builders::product(context, constant),
374        builders::product(context, kernel),
375    )
376}

378fn mapped_bound(
379    context: &Context,
380    bound: &ExpressionReference,
381    shift: &ExpressionReference,
382    scale: &ExpressionReference,
383) -> ExpressionReference {
384    // (bound - h)/σ, with the infinities fixed by the positive scale.
385    if matches!(
386        bound.kind,
387        ExpressionKind::Constant(BuiltinConstant::Infinity | BuiltinConstant::NegativeInfinity)
388    ) {
389        return Arc::clone(bound);
390    }
391    crate::simplify::simplify(
392        context,
393        builders::divide(
394            context,
395            builders::subtract(context, Arc::clone(bound), Arc::clone(shift)),
396            Arc::clone(scale),
397        ),
398    )
