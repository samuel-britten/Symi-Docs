13/// Evaluates the bounded collection of special-function product theorems whose convergence
14/// charts can be certified from the current real assumptions.
15pub(crate) fn integrate_special_function_product(
16    context: &Context,
17    target: &ExpressionReference,
18    variable: SymbolIdentifier,
19    lower_bound: &ExpressionReference,
20    upper_bound: &ExpressionReference,
21) -> Option<ExpressionReference> {
22    integrate_quarter_period_elliptic_kernel(context, target, variable, lower_bound, upper_bound)
23        .or_else(|| {
24            integrate_weber_schafheitlin_distinct_scale(
25                context,
26                target,
27                variable,
28                lower_bound,
29                upper_bound,
30            )
31        })
32        .or_else(|| {
33            integrate_weber_schafheitlin_equal_scale(
34                context,
35                target,
36                variable,
37                lower_bound,
38                upper_bound,
39            )
40        })
41        .or_else(|| {
42            integrate_cosine_integral_bessel_kernel(
43                context,
44                target,
45                variable,
46                lower_bound,
47                upper_bound,
48            )
49        })
50        .or_else(|| {
51            integrate_elliptic_first_kind_moment(
52                context,
53                target,
54                variable,
55                lower_bound,
56                upper_bound,
57            )
58        })
59}

61fn integrate_quarter_period_elliptic_kernel(
62    context: &Context,
63    target: &ExpressionReference,
64    variable: SymbolIdentifier,
65    lower_bound: &ExpressionReference,
66    upper_bound: &ExpressionReference,
67) -> Option<ExpressionReference> {
68    zero_to_half_pi(context, lower_bound, upper_bound)?;
69    let factors = product_factors(target);
70    let inverse_square_root_index = factors.iter().position(|factor| {
71        matches!(&factor.kind, ExpressionKind::Power(_, exponent) if Arc::ptr_eq(
72            exponent,
73            &builders::rational_value(context, -1, 2),
74        ))
75    })?;
76    let ExpressionKind::Power(radicand, _) = &factors[inverse_square_root_index].kind else {
77        return None;
78    };
79    let parameter = elliptic_quarter_period_parameter(context, radicand, variable)?;
80    let coefficient_factors = factors
81        .into_iter()
82        .enumerate()
83        .filter_map(|(index, factor)| {
84            (index != inverse_square_root_index).then(|| Arc::clone(factor))
85        })
86        .collect::<Vec<_>>();
87    if coefficient_factors
88        .iter()
89        .any(|factor| crate::expression::queries::depends_on(factor, variable))
90    {
91        return None;
92    }
93    let coefficient = match coefficient_factors.len() {
94        0 => builders::one(context),
95        1 => Arc::clone(&coefficient_factors[0]),
96        _ => builders::product(context, coefficient_factors),
97    };
98    Some(builders::product(
99        context,
100        vec![
101            coefficient,
102            crate::special_functions::elliptic_integrals::elliptic_k(context, parameter),
103        ],
104    ))
105}

107fn elliptic_quarter_period_parameter(
108    context: &Context,
109    radicand: &ExpressionReference,
110    variable: SymbolIdentifier,
111) -> Option<ExpressionReference> {
112    let sine_square = builders::power(
113        context,
114        builders::function_call(
115            context,
116            FunctionIdentifier::Builtin(BuiltinFunction::Sin),
117            vec![context.interner.intern(ExpressionKind::Symbol(variable))],
118        ),
119        builders::integer(context, crate::exact_arithmetic::Integer::from(2)),
120    );
121    let ExpressionKind::Sum(terms) = &radicand.kind else {
122        return None;
123    };
124    let negative_product = terms.iter().find_map(|term| match &term.kind {
125        ExpressionKind::Product(factors) if factors.iter().any(|factor| matches!(&factor.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::from(-1))) && factors.iter().any(|factor| Arc::ptr_eq(factor, &sine_square)) => Some(factors),
126        _ => None,
127    })?;
128    let parameter_factors = negative_product.iter().filter(|factor| !Arc::ptr_eq(factor, &sine_square) && !matches!(&factor.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::from(-1))).cloned().collect::<Vec<_>>();
129    if parameter_factors.is_empty() {
130        return None;
131    }
132    let parameter = if parameter_factors.len() == 1 {
133        Arc::clone(&parameter_factors[0])
134    } else {
135        builders::product(context, parameter_factors)
136    };
137    let expected_radicand = builders::subtract(
138        context,
139        builders::one(context),
140        builders::product(context, vec![Arc::clone(&parameter), sine_square]),
141    );
142    (radicand.kind == expected_radicand.kind && product_of_real_sine_squares(context, &parameter))
143        .then_some(parameter)
144}

146fn product_of_real_sine_squares(context: &Context, expression: &ExpressionReference) -> bool {
147    match &expression.kind {
148        ExpressionKind::Product(factors) => factors
149            .iter()
150            .all(|factor| product_of_real_sine_squares(context, factor)),
151        ExpressionKind::Power(base, exponent) if matches!(&exponent.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::from(2)) =>
152        {
153            matches!(&base.kind, ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(BuiltinFunction::Sin), arguments) if matches!(arguments.as_slice(), [argument] if real_verdict(&context.symbol_table, argument) == TruthValue::Yes))
154        }
155        _ => false,
156    }
157}

159/// Applies the positive distinct-scale Weber--Schafheitlin formula on an absolutely convergent chart.
160fn integrate_weber_schafheitlin_distinct_scale(
161    context: &Context,
162    target: &ExpressionReference,
163    variable: SymbolIdentifier,
164    lower_bound: &ExpressionReference,
165    upper_bound: &ExpressionReference,
166) -> Option<ExpressionReference> {
167    standard_positive_half_line(lower_bound, upper_bound)?;
168    let mut bessel_factors = Vec::new();
169    let mut lambda = None;
170    for factor in product_factors(target) {
171        if let Some((order, scale, multiplicity)) = bessel_factor(context, factor, variable) {
172            if multiplicity != 1 {
173                return None;
174            }
175            bessel_factors.push((order, scale));
176            continue;
177        }
178        if let Some(candidate_lambda) = negative_variable_power(context, factor, variable) {
179            if lambda.replace(candidate_lambda).is_some() {
180                return None;
181            }
182            continue;
183        }
184        return None;
185    }
186    let [(first_order, first_scale), (second_order, second_scale)] = bessel_factors.as_slice()
187    else {
188        return None;
189    };
190    let lambda = lambda?;
191    let first_is_smaller = positive_verdict(
192        &context.symbol_table,
193        &builders::subtract(context, Arc::clone(second_scale), Arc::clone(first_scale)),
194    ) == TruthValue::Yes;
195    let second_is_smaller = positive_verdict(
196        &context.symbol_table,
197        &builders::subtract(context, Arc::clone(first_scale), Arc::clone(second_scale)),
198    ) == TruthValue::Yes;
199    let (smaller_order, smaller_scale, larger_order, larger_scale) = if first_is_smaller {
200        (
201            Arc::clone(first_order),
202            Arc::clone(first_scale),
203            Arc::clone(second_order),
204            Arc::clone(second_scale),
205        )
206    } else if second_is_smaller {
207        (
208            Arc::clone(second_order),
209            Arc::clone(second_scale),
210            Arc::clone(first_order),
211            Arc::clone(first_scale),
212        )
213    } else {
214        return None;
215    };
216    let one = builders::one(context);
217    let half = builders::rational_value(context, 1, 2);
218    let endpoint_margin = builders::sum(
219        context,
220        vec![
221            Arc::clone(&smaller_order),
222            Arc::clone(&larger_order),
223            builders::negate(context, Arc::clone(&lambda)),
224            Arc::clone(&one),
225        ],
226    );
227    let lower_gamma_argument =
228        builders::sum(context, vec![Arc::clone(&smaller_order), Arc::clone(&one)]);
229    let second_denominator_argument = builders::product(
230        context,
231        vec![
232            builders::sum(
233                context,
234                vec![
235                    Arc::clone(&larger_order),
236                    builders::negate(context, Arc::clone(&smaller_order)),
237                    Arc::clone(&lambda),
238                    Arc::clone(&one),
239                ],
240            ),
241            Arc::clone(&half),
242        ],
243    );
244    if positive_verdict(&context.symbol_table, &smaller_scale) != TruthValue::Yes
245        || positive_verdict(&context.symbol_table, &larger_scale) != TruthValue::Yes
246        || real_verdict(&context.symbol_table, &smaller_order) != TruthValue::Yes
247        || real_verdict(&context.symbol_table, &larger_order) != TruthValue::Yes
248        || positive_verdict(&context.symbol_table, &lambda) != TruthValue::Yes
249        || positive_verdict(&context.symbol_table, &endpoint_margin) != TruthValue::Yes
250        || positive_verdict(&context.symbol_table, &lower_gamma_argument) != TruthValue::Yes
251        || positive_verdict(&context.symbol_table, &second_denominator_argument) != TruthValue::Yes
252    {
253        return None;
254    }
255    let first_upper_argument = builders::product(
256        context,
257        vec![Arc::clone(&endpoint_margin), Arc::clone(&half)],
258    );
259    let second_upper_argument = builders::product(
260        context,
261        vec![
262            builders::sum(
263                context,
264                vec![
265                    Arc::clone(&smaller_order),
266                    builders::negate(context, Arc::clone(&larger_order)),
267                    builders::negate(context, Arc::clone(&lambda)),
268                    Arc::clone(&one),
269                ],
270            ),
271            Arc::clone(&half),
272        ],
273    );
274    let hypergeometric_argument = builders::power(
275        context,
276        builders::divide(
277            context,
278            Arc::clone(&smaller_scale),
279            Arc::clone(&larger_scale),
280        ),
281        builders::integer_value(context, 2),
282    );
283    let hypergeometric = crate::special_functions::hypergeometric::hypergeometric_2f1(
284        context,
285        Arc::clone(&first_upper_argument),
286        second_upper_argument,
287        lower_gamma_argument.clone(),
288        hypergeometric_argument,
289    );
290    let gamma = |argument| {
291        builders::function_call(
292            context,
293            FunctionIdentifier::Builtin(BuiltinFunction::Gamma),
294            vec![argument],
295        )
296    };
297    let scale_exponent = builders::sum(
298        context,
299        vec![
300            Arc::clone(&smaller_order),
301            builders::negate(context, Arc::clone(&lambda)),
302            Arc::clone(&one),
303        ],
304    );
305    let denominator = builders::product(
306        context,
307        vec![
308            builders::power(
309                context,
310                builders::integer_value(context, 2),
311                Arc::clone(&lambda),
312            ),
313            builders::power(context, larger_scale, scale_exponent),
314            gamma(lower_gamma_argument),
315            gamma(second_denominator_argument),
316        ],
317    );
318    Some(builders::divide(
319        context,
320        builders::product(
321            context,
322            vec![
323                builders::power(context, smaller_scale, smaller_order),
324                gamma(first_upper_argument),
325                hypergeometric,
326            ],
327        ),
328        denominator,
329    ))
330}

332/// Applies the equal-order, equal-scale Weber--Schafheitlin formula on its convergent chart.
333fn integrate_weber_schafheitlin_equal_scale(
334    context: &Context,
335    target: &ExpressionReference,
336    variable: SymbolIdentifier,
337    lower_bound: &ExpressionReference,
338    upper_bound: &ExpressionReference,
339) -> Option<ExpressionReference> {
340    standard_positive_half_line(lower_bound, upper_bound)?;
341    let mut order = None;
342    let mut bessel_count = 0usize;
343    let mut lambda = None;
344    for factor in product_factors(target) {
345        if let Some((candidate_order, candidate_scale, multiplicity)) =
346            bessel_factor(context, factor, variable)
347        {
348            if let Some((known_order, known_scale)) = &order {
349                if zero_equivalent(context, known_order, Some(&candidate_order))
350                    != ZeroEquivalenceVerdict::TrueValue
351                    || zero_equivalent(context, known_scale, Some(&candidate_scale))
352                        != ZeroEquivalenceVerdict::TrueValue
353                {
354                    return None;
355                }
356            } else {
357                order = Some((candidate_order, candidate_scale));
358            }
359            bessel_count = bessel_count.checked_add(multiplicity)?;
360            continue;
361        }
362        if let Some(candidate_lambda) = negative_variable_power(context, factor, variable) {
363            if lambda.replace(candidate_lambda).is_some() {
364                return None;
365            }
366            continue;
367        }
368        return None;
369    }
370    if bessel_count != 2 {
371        return None;
372    }
373    let (order, scale) = order?;
374    let lambda = lambda?;
375    if positive_verdict(&context.symbol_table, &scale) != TruthValue::Yes
376        || real_verdict(&context.symbol_table, &order) != TruthValue::Yes
377        || positive_verdict(&context.symbol_table, &lambda) != TruthValue::Yes
378    {
379        return None;
380    }
381    let one = builders::one(context);
382    let admissibility_margin = builders::subtract(
383        context,
384        builders::sum(
385            context,
386            vec![
387                builders::product(
388                    context,
389                    vec![builders::integer_value(context, 2), Arc::clone(&order)],
390                ),
391                one,
392            ],
393        ),
394        Arc::clone(&lambda),
395    );
396    if positive_verdict(&context.symbol_table, &admissibility_margin) != TruthValue::Yes {
397        return None;
398    }
399    let half = builders::rational_value(context, 1, 2);
400    let lambda_minus_one = builders::subtract(context, Arc::clone(&lambda), builders::one(context));
401    let second_gamma_argument = builders::sum(
402        context,
403        vec![
404            order.clone(),
405            builders::product(
406                context,
407                vec![
408                    Arc::clone(&half),
409                    builders::subtract(context, builders::one(context), Arc::clone(&lambda)),
410                ],
411            ),
412        ],
413    );
414    let third_gamma_argument = builders::product(
415        context,
416        vec![
417            builders::sum(context, vec![Arc::clone(&lambda), builders::one(context)]),
418            builders::rational_value(context, 1, 2),
419        ],
420    );
421    let fourth_gamma_argument = builders::sum(
422        context,
423        vec![
424            order,
425            builders::product(
426                context,
427                vec![
428                    builders::rational_value(context, 1, 2),
429                    builders::sum(context, vec![Arc::clone(&lambda), builders::one(context)]),
430                ],
431            ),
432        ],
433    );
434    let gamma = |argument| {
435        builders::function_call(
436            context,
437            FunctionIdentifier::Builtin(BuiltinFunction::Gamma),
438            vec![argument],
439        )
440    };
441    let denominator = builders::product(
442        context,
443        vec![
444            builders::power(
445                context,
446                builders::integer_value(context, 2),
447                Arc::clone(&lambda),
448            ),
449            builders::power(
450                context,
451                gamma(third_gamma_argument),
452                builders::integer_value(context, 2),
453            ),
454            gamma(fourth_gamma_argument),
455        ],
456    );
457    Some(builders::divide(
458        context,
459        builders::product(
460            context,
461            vec![
462                builders::power(context, scale, lambda_minus_one),
463                gamma(lambda),
464                gamma(second_gamma_argument),
465            ],
466        ),
467        denominator,
468    ))
469}

471/// Applies the damped-transform identity for a cosine integral times a square-root Bessel kernel.
472fn integrate_cosine_integral_bessel_kernel(
473    context: &Context,
474    target: &ExpressionReference,
475    variable: SymbolIdentifier,
476    lower_bound: &ExpressionReference,
477    upper_bound: &ExpressionReference,
478) -> Option<ExpressionReference> {
479    standard_positive_half_line(lower_bound, upper_bound)?;
480    let mut cosine_scale = None;
481    let mut bessel_parameter = None;
482    for factor in product_factors(target) {
483        if let Some(candidate) = cosine_integral_linear_scale(context, factor, variable) {
484            if cosine_scale.replace(candidate).is_some() {
485                return None;
486            }
487            continue;
488        }
489        if let Some(candidate) = square_root_bessel_parameter(context, factor, variable) {
490            if bessel_parameter.replace(candidate).is_some() {
491                return None;
492            }
493            continue;
494        }
495        return None;
496    }
497    let cosine_scale = cosine_scale?;
498    let bessel_parameter = bessel_parameter?;
499    if positive_verdict(&context.symbol_table, &cosine_scale) != TruthValue::Yes
500        || positive_verdict(&context.symbol_table, &bessel_parameter) != TruthValue::Yes
501    {
502        return None;
503    }
504    let quotient = builders::divide(context, Arc::clone(&bessel_parameter), cosine_scale);
505    let cosine = builders::function_call(
506        context,
507        FunctionIdentifier::Builtin(BuiltinFunction::Cos),
508        vec![quotient],
509    );
510    Some(builders::divide(
511        context,
512        builders::subtract(context, cosine, builders::one(context)),
513        bessel_parameter,
514    ))
515}

517/// Applies the Euler--hypergeometric complete-elliptic first-kind moment for absolute scale at most one.
518fn integrate_elliptic_first_kind_moment(
519    context: &Context,
520    target: &ExpressionReference,
521    variable: SymbolIdentifier,
522    lower_bound: &ExpressionReference,
523    upper_bound: &ExpressionReference,
524) -> Option<ExpressionReference> {
525    zero_to_half_pi(context, lower_bound, upper_bound)?;
526    let mut scale = None;
527    let mut squared_scale = None;
528    let mut sine_seen = false;
529    let mut elliptic_seen = false;
530    for factor in product_factors(target) {
531        if is_sine_of_variable(factor, variable) {
532            if sine_seen {
533                return None;
534            }
535            sine_seen = true;
536            continue;
537        }
538        if let Some(candidate) = elliptic_first_kind_parameter(factor, variable) {
539            if elliptic_seen {
540                return None;
541            }
542            elliptic_seen = true;
543            squared_scale = Some(candidate);
544            continue;
545        }
546        if !crate::expression::queries::depends_on(factor, variable) {
547            if scale.replace(Arc::clone(factor)).is_some() {
548                return None;
549            }
550            continue;
551        }
552        return None;
553    }
554    let scale = scale?;
555    let squared_scale = squared_scale?;
556    let expected_squared_scale =
557        builders::product(context, vec![Arc::clone(&scale), Arc::clone(&scale)]);
558    if !sine_seen
559        || !elliptic_seen
560        || !matches!(prove_rational_ratio(context, &squared_scale, &expected_squared_scale), Some(ratio) if ratio == crate::exact_arithmetic::rational_one())
561        || !scale_is_certified_in_closed_unit_interval(context, &scale)
562    {
563        return None;
564    }
565    let pi = builders::constant(context, BuiltinConstant::Pi);
566    let arcsine = builders::function_call(
567        context,
568        FunctionIdentifier::Builtin(BuiltinFunction::Arcsin),
569        vec![scale],
570    );
571    Some(builders::product(
572        context,
573        vec![builders::rational_value(context, 1, 2), pi, arcsine],
574    ))
575}

577fn scale_is_certified_in_closed_unit_interval(
578    context: &Context,
579    scale: &ExpressionReference,
580) -> bool {
581    matches!(&scale.kind, ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(BuiltinFunction::Sin), arguments) if matches!(arguments.as_slice(), [argument] if real_verdict(&context.symbol_table, argument) == TruthValue::Yes))
582        || (nonnegative_verdict(
583            &context.symbol_table,
584            &builders::sum(context, vec![builders::one(context), Arc::clone(scale)]),
585        ) == TruthValue::Yes
586            && nonnegative_verdict(
587                &context.symbol_table,
588                &builders::subtract(context, builders::one(context), Arc::clone(scale)),
589            ) == TruthValue::Yes)
590}

592/// Recognises a Bessel factor, including a square represented as a power.
593fn bessel_factor(
594    context: &Context,
595    factor: &ExpressionReference,
596    variable: SymbolIdentifier,
597) -> Option<(ExpressionReference, ExpressionReference, usize)> {
598    if let ExpressionKind::Power(base, exponent) = &factor.kind
599        && matches!(&exponent.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::from(2))
600    {
601        let (order, scale, _) = bessel_factor(context, base, variable)?;
602        return Some((order, scale, 2));
603    }
604    let ExpressionKind::FunctionCall(
605        FunctionIdentifier::Builtin(BuiltinFunction::BesselJ),
606        arguments,
607    ) = &factor.kind
608    else {
609        return None;
610    };
611    let [order, argument] = arguments.as_slice() else {
612        return None;
613    };
614    Some((
615        Arc::clone(order),
616        linear_scale(context, argument, variable)?,
617        1,
618    ))
619}

621/// Recognises x^(-lambda) with a variable-independent positive lambda.
622fn negative_variable_power(
623    context: &Context,
624    factor: &ExpressionReference,
625    variable: SymbolIdentifier,
626) -> Option<ExpressionReference> {
627    let ExpressionKind::Power(base, exponent) = &factor.kind else {
628        return None;
629    };
630    if !matches!(&base.kind, ExpressionKind::Symbol(identifier) if *identifier == variable) {
631        return None;
632    }
633    Some(builders::negate(context, Arc::clone(exponent)))
634}

655/// Recognises J_0(2*sqrt(b*x)).
656fn square_root_bessel_parameter(
657    context: &Context,
658    factor: &ExpressionReference,
659    variable: SymbolIdentifier,
660) -> Option<ExpressionReference> {
661    let ExpressionKind::FunctionCall(
662        FunctionIdentifier::Builtin(BuiltinFunction::BesselJ),
663        arguments,
664    ) = &factor.kind
665    else {
666        return None;
667    };
668    let [order, argument] = arguments.as_slice() else {
669        return None;
670    };
671    if !matches!(&order.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::ZERO)
672    {
673        return None;
674    }
675    let ExpressionKind::Product(factors) = &argument.kind else {
676        return None;
677    };
678    let mut square_root = None;
679    let mut two_seen = false;
680    for argument_factor in factors {
681        if matches!(&argument_factor.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::from(2))
682        {
683            two_seen = true;
684        } else if let ExpressionKind::Power(base, exponent) = &argument_factor.kind
685            && matches!(&exponent.kind, ExpressionKind::Rational(value) if *value == crate::exact_arithmetic::Rational::from_integers(crate::exact_arithmetic::Integer::from(1), crate::exact_arithmetic::Integer::from(2)))
686        {
687            square_root = Some(linear_scale(context, base, variable)?);
688        } else {
689            return None;
690        }
691    }
692    two_seen.then_some(square_root?)
693}

695/// Recognises K(k^2 sin(x)^2) and returns k squared.
696fn elliptic_first_kind_parameter(
697    factor: &ExpressionReference,
698    variable: SymbolIdentifier,
699) -> Option<ExpressionReference> {
700    let ExpressionKind::FunctionCall(
701        FunctionIdentifier::Builtin(BuiltinFunction::EllipticK),
702        arguments,
703    ) = &factor.kind
704    else {
705        return None;
706    };
707    let [parameter] = arguments.as_slice() else {
708        return None;
709    };
710    let ExpressionKind::Product(factors) = &parameter.kind else {
711        return None;
712    };
713    let mut scale_squared = None;
714    let mut sine_squared_seen = false;
715    for parameter_factor in factors {
716        if let ExpressionKind::Power(base, exponent) = &parameter_factor.kind
717            && matches!(&exponent.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::from(2))
718        {
719            if is_sine_of_variable(base, variable) {
720                sine_squared_seen = true;
721            } else {
722                scale_squared = Some(Arc::clone(parameter_factor));
723            }
724        } else if !crate::expression::queries::depends_on(parameter_factor, variable) {
725            scale_squared = Some(Arc::clone(parameter_factor));
726        } else {
727            return None;
728        }
729    }
730    sine_squared_seen.then_some(scale_squared?)
731}

733/// Reads c*x in the canonical product grammar, with c=1 for a bare variable.
734fn linear_scale(
735    context: &Context,
736    target: &ExpressionReference,
737    variable: SymbolIdentifier,
738) -> Option<ExpressionReference> {
739    if matches!(&target.kind, ExpressionKind::Symbol(identifier) if *identifier == variable) {
740        return Some(builders::one(context));
741    }
742    let ExpressionKind::Product(factors) = &target.kind else {
743        return None;
744    };
745    let mut variable_count = 0usize;
746    let mut scale_factors = Vec::new();
747    for factor in factors {
748        if matches!(&factor.kind, ExpressionKind::Symbol(identifier) if *identifier == variable) {
749            variable_count += 1;
750        } else if crate::expression::queries::depends_on(factor, variable) {
751            return None;
752        } else {
753            scale_factors.push(Arc::clone(factor));
754        }
755    }
756    (variable_count == 1 && !scale_factors.is_empty())
757        .then(|| builders::product(context, scale_factors))
758}

760/// Returns the top-level factors of a product.
761fn product_factors(target: &ExpressionReference) -> Vec<&ExpressionReference> {
762    match &target.kind {
763        ExpressionKind::Product(factors) => factors.iter().collect(),
764        _ => vec![target],
765    }
766}

768/// Tests the exact [0,infinity] integration interval.
769fn standard_positive_half_line(
770    lower_bound: &ExpressionReference,
771    upper_bound: &ExpressionReference,
772) -> Option<()> {
773    (matches!(&lower_bound.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::ZERO)
774        && matches!(&upper_bound.kind, ExpressionKind::Constant(BuiltinConstant::Infinity))).then_some(())
775}

777/// Tests whether a factor is sin(x).
778fn is_sine_of_variable(factor: &ExpressionReference, variable: SymbolIdentifier) -> bool {
779    matches!(&factor.kind, ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(BuiltinFunction::Sin), arguments) if matches!(arguments.as_slice(), [argument] if matches!(&argument.kind, ExpressionKind::Symbol(identifier) if *identifier == variable)))
780}

782/// Tests the exact [0, pi/2] interval.
783fn zero_to_half_pi(
784    context: &Context,
785    lower_bound: &ExpressionReference,
786    upper_bound: &ExpressionReference,
787) -> Option<()> {
788    let expected_upper = builders::product(
789        context,
790        vec![
791            builders::rational_value(context, 1, 2),
792            builders::constant(context, BuiltinConstant::Pi),
793        ],
794    );
795    (matches!(&lower_bound.kind, ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::Integer::ZERO)
796        && (Arc::ptr_eq(upper_bound, &expected_upper)
797            || zero_equivalent(context, upper_bound, Some(&expected_upper)) == ZeroEquivalenceVerdict::TrueValue)).then_some(())
798}
