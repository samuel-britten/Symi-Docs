28/// Evaluates a whole-line rational integral with rational coefficients through its exact residue sum.
29pub(crate) fn whole_line_rational_value(
30    context: &Context,
31    numerator: &[Rational],
32    denominator: &[Rational],
33    variable: SymbolIdentifier,
34    admits_simple_real_poles: bool,
35) -> Option<ExpressionReference> {
36    // For a real rational R = P/Q with deg Q > deg P and only simple real poles, closing the
37    // symmetric truncation [-T, T] by the upper semicircle, indenting each real pole from above
38    // and letting the arc contribute -i*pi*c for R(z) = c/z + O(z^-2) gives
39    //   PV ∫ R = 2πi Σ_upper Res + πi Σ_real Res − πi c.
40    // The sum of every finite residue is c, so this equals πi (Σ_upper Res − Σ_lower Res): the
41    // real residues and the arc term cancel exactly, and the value is π times the real algebraic
42    // number σ = i Σ_nonreal sign(Im r) Res(R, r). An ordinary integral requires degree decay of
43    // at least two and no real pole; a principal value admits decay one and simple real poles.
44    // Hermite reduction first removes a derivative of a proper rational function whose poles are
45    // the repeated (hence nonreal) roots; such a derivative integrates to zero over the line, so
46    // the residues are taken of a squarefree denominator. Each irreducible factor then
47    // contributes on its own: cyclotomic factors by exact sines of rational multiples of π, and
48    // any other factor by a certified real algebraic number.
49    let mut numerator = numerator.to_vec();
50    let mut denominator = denominator.to_vec();
51    dense::normalize(&mut numerator);
52    dense::normalize(&mut denominator);
53    if denominator.is_empty() {
54        return None;
55    }
56    if numerator.is_empty() {
57        return Some(builders::zero(context));
58    }
59    let common = dense::monic_gcd(&numerator, &denominator);
60    let (numerator, numerator_remainder) = dense::divide(&numerator, &common);
61    let (denominator, denominator_remainder) = dense::divide(&denominator, &common);
62    if !numerator_remainder.is_empty() || !denominator_remainder.is_empty() {
63        return None;
64    }
65    let required_decay = if admits_simple_real_poles { 1 } else { 2 };
66    if dense::degree_of(&denominator) < dense::degree_of(&numerator) + required_decay {
67        return None;
68    }
69    let repeated_part = dense::monic_gcd(&denominator, &dense::derivative(&denominator));
70    if dense::degree_of(&repeated_part) > 0
71        && count_distinct_real_roots(&dense::from_dense(variable, &repeated_part))? != 0
72    {
73        return None;
74    }
75    if !admits_simple_real_poles
76        && count_distinct_real_roots(&dense::from_dense(variable, &denominator))? != 0
77    {
78        return None;
79    }
80    let (reduced_numerator, reduced_denominator) = if dense::degree_of(&repeated_part) > 0 {
81        let (_, numerator_polynomial, denominator_polynomial) =
82            crate::integrate::rational::hermite_reduce(
83                context,
84                &dense::from_dense(variable, &numerator),
85                &dense::from_dense(variable, &denominator),
86                variable,
87            )?;
88        (
89            dense::to_dense(&numerator_polynomial)?,
90            dense::to_dense(&denominator_polynomial)?,
91        )
92    } else {
93        (numerator, denominator)
94    };
95    if dense::degree_of(&dense::monic_gcd(
96        &reduced_denominator,
97        &dense::derivative(&reduced_denominator),
98    )) != 0
99        || dense::degree_of(&reduced_denominator) <= dense::degree_of(&reduced_numerator)
100    {
101        return None;
102    }
103    let denominator_derivative = dense::derivative(&reduced_denominator);
104    let factors = crate::polynomial::factorization::factor_univariate_over_integers_in_context(
105        context,
106        &dense::from_dense(variable, &reduced_denominator),
107    );
108    let mut factor_degree_total = 0;
109    let mut contributions = Vec::new();
110    for (multiplicity, factor_polynomial) in &factors {
111        let factor = dense::to_dense(factor_polynomial)?;
112        let factor_degree = dense::degree_of(&factor);
113        if factor_degree == 0 {
114            continue;
115        }
116        if *multiplicity != 1 {
117            return None;
118        }
119        factor_degree_total += factor_degree;
120        let real_root_total = count_distinct_real_roots(factor_polynomial)?;
121        if real_root_total == factor_degree {
122            continue;
123        }
124        let residue_polynomial =
125            residue_polynomial_modulo_factor(&reduced_numerator, &denominator_derivative, &factor)?;
126        if residue_polynomial.is_empty() {
127            continue;
128        }
129        let contribution = match cyclotomic_order(&factor) {
130            Some(order) => cyclotomic_contribution(context, &residue_polynomial, order),
131            None => {
132                generic_factor_contribution(context, &factor, real_root_total, &residue_polynomial)?
133            }
134        };
135        contributions.push(contribution);
136    }
137    if factor_degree_total != dense::degree_of(&reduced_denominator) {
138        return None;
139    }
140    Some(builders::product(
141        context,
142        vec![
143            builders::constant(context, BuiltinConstant::Pi),
144            builders::sum(context, contributions),
145        ],
146    ))
147}

149fn residue_polynomial_modulo_factor(
150    numerator: &[Rational],
151    denominator_derivative: &[Rational],
152    factor: &[Rational],
153) -> Option<Vec<Rational>> {
154    // At a simple root r of the squarefree denominator Q the residue of P/Q is P(r)/Q'(r). Every
155    // root of one irreducible factor f shares the representative A = P · (Q' mod f)^(-1) mod f,
156    // so the residues at the conjugate roots are the values A(r) of one polynomial.
157    let (_, numerator_residue) = dense::divide(numerator, factor);
158    let (_, derivative_residue) = dense::divide(denominator_derivative, factor);
159    let inverse = modular_inverse(&derivative_residue, factor)?;
160    let (_, representative) = dense::divide(&dense_multiply(&numerator_residue, &inverse), factor);
161    Some(representative)
162}

164fn modular_inverse(value: &[Rational], modulus: &[Rational]) -> Option<Vec<Rational>> {
165    // The inverse of value in Q[x]/(modulus) by the extended Euclidean algorithm; None when the
166    // two polynomials share a nonconstant factor.
167    let mut previous_remainder = modulus.to_vec();
168    let mut remainder = value.to_vec();
169    dense::normalize(&mut previous_remainder);
170    dense::normalize(&mut remainder);
171    let mut previous_coefficient: Vec<Rational> = Vec::new();
172    let mut coefficient = vec![rational_one()];
173    while !remainder.is_empty() {
174        let (quotient, next_remainder) = dense::divide(&previous_remainder, &remainder);
175        let next_coefficient = dense_subtract(
176            &previous_coefficient,
177            &dense_multiply(&quotient, &coefficient),
178        );
179        previous_remainder = remainder;
180        remainder = next_remainder;
181        previous_coefficient = coefficient;
182        coefficient = next_coefficient;
183    }
184    if dense::degree_of(&previous_remainder) != 0 || previous_remainder.is_empty() {
185        return None;
186    }
187    let scale = rational_one() / previous_remainder[0].clone();
188    let scaled: Vec<Rational> = previous_coefficient
189        .iter()
190        .map(|entry| entry * &scale)
191        .collect();
192    let (_, reduced) = dense::divide(&scaled, modulus);
193    Some(reduced)
194}

196fn dense_multiply(left: &[Rational], right: &[Rational]) -> Vec<Rational> {
197    // The product of two dense ascending rational polynomials.
198    if left.is_empty() || right.is_empty() {
199        return Vec::new();
200    }
201    let mut result = vec![rational_zero(); left.len() + right.len() - 1];
202    for (left_index, left_value) in left.iter().enumerate() {
203        for (right_index, right_value) in right.iter().enumerate() {
204            result[left_index + right_index] += left_value * right_value;
205        }
206    }
207    dense::normalize(&mut result);
208    result
209}

211fn dense_subtract(left: &[Rational], right: &[Rational]) -> Vec<Rational> {
212    // The difference of two dense ascending rational polynomials.
213    let length = left.len().max(right.len());
214    let mut result = vec![rational_zero(); length];
215    for (index, value) in left.iter().enumerate() {
216        result[index] += value;
217    }
218    for (index, value) in right.iter().enumerate() {
219        result[index] -= value;
220    }
221    dense::normalize(&mut result);
222    result
223}

225fn cyclotomic_order(factor: &[Rational]) -> Option<usize> {
226    // An irreducible factor is a cyclotomic polynomial exactly when x^n ≡ 1 modulo the factor for
227    // some n; the least such n is the common multiplicative order of its roots, and an
228    // irreducible polynomial with a primitive n-th root of unity is Φ_n. Since φ(n) ≥ sqrt(n/2),
229    // a factor of degree d can only be Φ_n for n ≤ 2d².
230    let mut monic = factor.to_vec();
231    dense::make_monic(&mut monic);
232    let degree = dense::degree_of(&monic);
233    if degree == 0 {
234        return None;
235    }
236    let bound = 2 * degree * degree;
237    let mut power = vec![rational_one()];
238    for order in 1..=bound {
239        let mut shifted = vec![rational_zero()];
240        shifted.extend(power);
241        let (_, reduced) = dense::divide(&shifted, &monic);
242        power = reduced;
243        if power.len() == 1 && power[0] == rational_one() {
244            return Some(order);
245        }
246    }
247    None
248}

250fn cyclotomic_contribution(
251    context: &Context,
252    residue_polynomial: &[Rational],
253    order: usize,
254) -> ExpressionReference {
255    // The roots of Φ_n are exp(2πik/n) with gcd(k, n) = 1, in the upper half-plane for k < n/2
256    // and conjugate to exp(-2πik/n) below. With A = Σ a_j x^j the factor contributes
257    //   σ = i Σ_upper (A(ζ^k) − conj A(ζ^k)) = −2 Σ_{k<n/2} Σ_j a_j sin(2πjk/n).
258    let pi = builders::constant(context, BuiltinConstant::Pi);
259    let mut terms = Vec::new();
260    for index in 1..order {
261        if 2 * index >= order || greatest_common_divisor(index, order) != 1 {
262            continue;
263        }
264        for (exponent, coefficient) in residue_polynomial.iter().enumerate() {
265            let angle_numerator = (exponent * index) % order;
266            if angle_numerator == 0 || *coefficient == rational_zero() {
267                continue;
268            }
269            let angle = builders::product(
270                context,
271                vec![
272                    builders::rational(
273                        context,
274                        Integer::from(2 * angle_numerator as i64),
275                        Integer::from(order as i64),
276                    ),
277                    pi.clone(),
278                ],
279            );
280            terms.push(builders::product(
281                context,
282                vec![
283                    builders::rational_literal(context, -(coefficient * Rational::from(2))),
284                    builders::function_call(
285                        context,
286                        FunctionIdentifier::Builtin(BuiltinFunction::Sin),
287                        vec![angle],
288                    ),
289                ],
290            ));
291        }
292    }
293    builders::sum(context, terms)
294}

296fn greatest_common_divisor(left: usize, right: usize) -> usize {
297    // Euclid's algorithm on machine integers.
298    let (mut first, mut second) = (left, right);
299    while second != 0 {
300        let remainder = first % second;
301        first = second;
302        second = remainder;
303    }
304    first
305}

307fn generic_factor_contribution(
308    context: &Context,
309    factor: &[Rational],
310    real_root_total: usize,
311    residue_polynomial: &[Rational],
312) -> Option<ExpressionReference> {
313    // For a real irreducible factor of degree d with ρ real roots and u = (d − ρ)/2 upper roots,
314    // write w_k = A(r_k). For the subset S of the upper and real roots, of size s = u + ρ, the
315    // lower roots are the complement, so σ = i(Σ_upper w − Σ_lower w) = i(2 y_S − T) with
316    // y_S = Σ_S w and T = Σ w the trace of A. Every s-subset sum is a root of the polynomial H
317    // built from the power sums of the w_k, so σ is a real root of H((T − iσ)/2); the real and
318    // imaginary parts of that polynomial share exactly those real roots. The correct one is
319    // selected by an exact complex-interval enclosure of σ = −2 Σ_upper Im A(r) over certified
320    // isolating rectangles, refined until the enclosure holds a single real root.
321    let degree = dense::degree_of(factor);
322    if degree > MAXIMUM_GENERIC_FACTOR_DEGREE || (degree - real_root_total) % 2 != 0 {
323        return None;
324    }
325    let upper_root_total = (degree - real_root_total) / 2;
326    let subset_size = upper_root_total + real_root_total;
327    let subset_count = binomial(degree, subset_size)?;
328    let root_power_sums = root_power_sums(factor);
329    let traces = residue_power_traces(residue_polynomial, factor, &root_power_sums, subset_count);
330    let subset_power_sums = subset_sum_power_sums(&traces, subset_size, subset_count);
331    let subset_polynomial = polynomial_from_power_sums(&subset_power_sums);
332    let annihilator = real_annihilator(&subset_polynomial, &traces[1])?;
333    let (value_lower, value_upper) = certified_enclosure(
334        context,
335        factor,
336        residue_polynomial,
337        upper_root_total,
338        &annihilator,
339    )?;
340    present_real_algebraic_value(context, &annihilator, value_lower, value_upper)
341}

343fn binomial(total: usize, chosen: usize) -> Option<usize> {
344    // The binomial coefficient, None on overflow.
345    let mut value: usize = 1;
346    for index in 0..chosen {
347        value = value.checked_mul(total - index)? / (index + 1);
348    }
349    Some(value)
350}

352fn root_power_sums(factor: &[Rational]) -> Vec<Rational> {
353    // Newton's identities for the power sums p_0, ..., p_{d-1} of the roots of the factor.
354    let mut monic = factor.to_vec();
355    dense::make_monic(&mut monic);
356    let degree = dense::degree_of(&monic);
357    let mut power_sums = vec![Rational::from(degree as i64)];
358    for order in 1..degree {
359        let mut value = monic[degree - order].clone() * Rational::from(order as i64);
360        for index in 1..order {
361            value += &monic[degree - index] * &power_sums[order - index];
362        }
363        power_sums.push(-value);
364    }
365    power_sums
366}

368fn residue_power_traces(
369    residue_polynomial: &[Rational],
370    factor: &[Rational],
371    root_power_sums: &[Rational],
372    highest_order: usize,
373) -> Vec<Rational> {
374    // The traces Tr(A^m) = Σ_k A(r_k)^m for m = 0, ..., highest_order, read off the reduced
375    // powers of A in Q[x]/(factor) against the root power sums.
376    let mut traces = vec![root_power_sums[0].clone()];
377    let mut power = vec![rational_one()];
378    for _ in 1..=highest_order {
379        let (_, reduced) = dense::divide(&dense_multiply(&power, residue_polynomial), factor);
380        power = reduced;
381        let mut trace = rational_zero();
382        for (exponent, coefficient) in power.iter().enumerate() {
383            trace += coefficient * &root_power_sums[exponent];
384        }
385        traces.push(trace);
386    }
387    traces
388}

390fn subset_sum_power_sums(
391    traces: &[Rational],
392    subset_size: usize,
393    subset_count: usize,
394) -> Vec<Rational> {
395    // Σ_{|S|=s} exp(u y_S) is the t^s coefficient of Π_k (1 + t exp(u w_k))
396    //   = exp(Σ_j (−1)^(j+1) t^j / j · Σ_m (j u)^m Tr(A^m) / m!),
397    // truncated at t^s and u^N with N the number of subsets. Its u^m coefficient times m! is the
398    // m-th power sum of the subset sums.
399    let width = subset_count + 1;
400    let mut factorials = vec![rational_one()];
401    for order in 1..width {
402        let next = &factorials[order - 1] * Rational::from(order as i64);
403        factorials.push(next);
404    }
405    let mut exponent = vec![vec![rational_zero(); width]; subset_size + 1];
406    for (power_of_t, row) in exponent.iter_mut().enumerate().skip(1) {
407        let sign = if power_of_t % 2 == 1 {
408            Rational::from(1)
409        } else {
410            Rational::from(-1)
411        };
412        let reciprocal = sign / Rational::from(power_of_t as i64);
413        let mut scale = rational_one();
414        for (order, entry) in row.iter_mut().enumerate() {
415            *entry = &reciprocal * &scale * &traces[order] / &factorials[order];
416            scale *= Rational::from(power_of_t as i64);
417        }
418    }
419    let mut total = vec![vec![rational_zero(); width]; subset_size + 1];
420    total[0][0] = rational_one();
421    let mut term = total.clone();
422    for power in 1..=subset_size {
423        let mut next = vec![vec![rational_zero(); width]; subset_size + 1];
424        for (left_t, left_row) in term.iter().enumerate() {
425            for (right_t, right_row) in exponent.iter().enumerate().skip(1) {
426                if left_t + right_t > subset_size {
427                    break;
428                }
429                for (left_u, left_value) in left_row.iter().enumerate() {
430                    if *left_value == rational_zero() {
431                        continue;
432                    }
433                    for (right_u, right_value) in right_row.iter().enumerate() {
434                        if left_u + right_u >= width {
435                            break;
436                        }
437                        next[left_t + right_t][left_u + right_u] += left_value * right_value;
438                    }
439                }
440            }
441        }
442        let divisor = Rational::from(power as i64);
443        for (row_index, row) in next.iter_mut().enumerate() {
444            for (column_index, entry) in row.iter_mut().enumerate() {
445                *entry /= &divisor;
446                total[row_index][column_index] += &*entry;
447            }
448        }
449        term = next;
450    }
451    total[subset_size]
452        .iter()
453        .enumerate()
454        .map(|(order, coefficient)| coefficient * &factorials[order])
455        .collect()
456}

458fn polynomial_from_power_sums(power_sums: &[Rational]) -> Vec<Rational> {
459    // Newton's identities recover the elementary symmetric functions e_k of N values from their
460    // power sums, and the monic polynomial with those roots is Σ (−1)^k e_k y^(N−k).
461    let count = power_sums.len() - 1;
462    let mut elementary = vec![rational_one()];
463    for order in 1..=count {
464        let mut value = rational_zero();
465        for index in 1..=order {
466            let term = &elementary[order - index] * &power_sums[index];
467            if index % 2 == 1 {
468                value += term;
469            } else {
470                value -= term;
471            }
472        }
473        elementary.push(value / Rational::from(order as i64));
474    }
475    let mut polynomial = vec![rational_zero(); count + 1];
476    for (order, value) in elementary.iter().enumerate() {
477        polynomial[count - order] = if order % 2 == 0 {
478            value.clone()
479        } else {
480            -value.clone()
481        };
482    }
483    polynomial
484}

486fn real_annihilator(subset_polynomial: &[Rational], trace: &Rational) -> Option<Vec<Rational>> {
487    // Compose H with y = (T − iσ)/2 over Q(i) and return the squarefree greatest common divisor of
488    // the real and imaginary coefficient parts: exactly the real σ with H((T − iσ)/2) = 0.
489    let linear: ExactPolynomial = vec![
490        GaussianRational {
491            real: trace / Rational::from(2),
492            imaginary: rational_zero(),
493        },
494        GaussianRational {
495            real: rational_zero(),
496            imaginary: Rational::from(-1) / Rational::from(2),
497        },
498    ];
499    let mut composed: ExactPolynomial = Vec::new();
500    for coefficient in subset_polynomial.iter().rev() {
501        composed = polynomial_add(
502            &polynomial_multiply(&composed, &linear),
503            &vec![GaussianRational {
504                real: coefficient.clone(),
505                imaginary: rational_zero(),
506            }],
507        );
508    }
509    let mut real_part: Vec<Rational> = composed.iter().map(|value| value.real.clone()).collect();
510    let mut imaginary_part: Vec<Rational> = composed
511        .iter()
512        .map(|value| value.imaginary.clone())
513        .collect();
514    dense::normalize(&mut real_part);
515    dense::normalize(&mut imaginary_part);
516    let common = if imaginary_part.is_empty() {
517        real_part
518    } else {
519        dense::monic_gcd(&real_part, &imaginary_part)
520    };
521    if dense::degree_of(&common) == 0 {
522        return None;
523    }
524    Some(dense::squarefree_part(&common))
525}

539fn interval_add(left: &RationalInterval, right: &RationalInterval) -> RationalInterval {
540    // The exact sum of two closed rational intervals.
541    RationalInterval {
542        lower: &left.lower + &right.lower,
543        upper: &left.upper + &right.upper,
544    }

547fn interval_subtract(left: &RationalInterval, right: &RationalInterval) -> RationalInterval {
548    // The exact difference of two closed rational intervals.
549    RationalInterval {
550        lower: &left.lower - &right.upper,
551        upper: &left.upper - &right.lower,
552    }

555fn interval_multiply(left: &RationalInterval, right: &RationalInterval) -> RationalInterval {
556    // The exact product of two closed rational intervals from its four corner products.
557    let corners = [
558        &left.lower * &right.lower,
559        &left.lower * &right.upper,
560        &left.upper * &right.lower,
561        &left.upper * &right.upper,
562    ];
563    let mut lower = corners[0].clone();
564    let mut upper = corners[0].clone();
565    for corner in &corners[1..] {
566        if *corner < lower {
567            lower = corner.clone();
568        }
569        if *corner > upper {
570            upper = corner.clone();
571        }
572    }
573    RationalInterval { lower, upper }
574}

576fn enclosure_of_polynomial_at_rectangle(
577    polynomial: &[Rational],
578    rectangle: &ComplexIsolatingRectangle,
579) -> ComplexEnclosure {
580    // Horner evaluation of a rational polynomial over a complex rectangle in exact interval
581    // arithmetic; the result contains the polynomial's value at every point of the rectangle.
582    let point = ComplexEnclosure {
583        real: RationalInterval {
584            lower: rectangle.real_lower.clone(),
585            upper: rectangle.real_upper.clone(),
586        },
587        imaginary: RationalInterval {
588            lower: rectangle.imaginary_lower.clone(),
589            upper: rectangle.imaginary_upper.clone(),
590        },
591    };
592    let zero_interval = RationalInterval {
593        lower: rational_zero(),
594        upper: rational_zero(),
595    };
596    let mut value = ComplexEnclosure {
597        real: zero_interval.clone(),
598        imaginary: zero_interval,
599    };
600    for coefficient in polynomial.iter().rev() {
601        let real = interval_subtract(
602            &interval_multiply(&value.real, &point.real),
603            &interval_multiply(&value.imaginary, &point.imaginary),
604        );
605        let imaginary = interval_add(
606            &interval_multiply(&value.real, &point.imaginary),
607            &interval_multiply(&value.imaginary, &point.real),
608        );
609        value = ComplexEnclosure {
610            real: interval_add(
611                &real,
612                &RationalInterval {
613                    lower: coefficient.clone(),
614                    upper: coefficient.clone(),
615                },
616            ),
617            imaginary,
618        };
619    }
620    value
621}

623fn certified_enclosure(
624    context: &Context,
625    factor: &[Rational],
626    residue_polynomial: &[Rational],
627    upper_root_total: usize,
628    annihilator: &[Rational],
629) -> Option<(Rational, Rational)> {
630    // Isolate the factor's roots, keep the rectangles certified to lie in the open upper
631    // half-plane, and refine them until the enclosure of σ = −2 Σ_upper Im A(r) contains exactly
632    // one root of the annihilator in the closed interval.
633    let primitive = PrimitiveIntegerPolynomial::from_rational_coefficients(factor)?;
634    let mut budget = RootIsolationWorkBudget::default();
635    let regions =
636        isolate_all_roots_with_operation(&primitive, &mut budget, context.operation()).ok()?;
637    let mut rectangles: Vec<ComplexIsolatingRectangle> = regions
638        .into_iter()
639        .filter_map(|region| match region {
640            AlgebraicIsolatingRegion::NonReal(rectangle) => Some(rectangle),
641            AlgebraicIsolatingRegion::Real(_) => None,
642        })
643        .collect();
644    let annihilator_polynomial = dense::from_dense(SymbolIdentifier(u32::MAX - 17), annihilator);
645    for _ in 0..MAXIMUM_ENCLOSURE_REFINEMENTS {
646        if context.operation().is_cancelled() {
647            return None;
648        }
649        let undecided = rectangles.iter().any(|rectangle| {
650            rectangle.imaginary_lower <= rational_zero()
651                && rectangle.imaginary_upper >= rational_zero()
652        });
653        let upper: Vec<&ComplexIsolatingRectangle> = rectangles
654            .iter()
655            .filter(|rectangle| rectangle.imaginary_lower > rational_zero())
656            .collect();
657        if !undecided {
658            if upper.len() != upper_root_total {
659                return None;
660            }
661            let mut imaginary_total = RationalInterval {
662                lower: rational_zero(),
663                upper: rational_zero(),
664            };
665            for rectangle in &upper {
666                let enclosure = enclosure_of_polynomial_at_rectangle(residue_polynomial, rectangle);
667                imaginary_total = interval_add(&imaginary_total, &enclosure.imaginary);
668            }
669            let lower = &imaginary_total.upper * Rational::from(-2);
670            let upper_bound = &imaginary_total.lower * Rational::from(-2);
671            if lower == upper_bound {
672                return (dense::evaluate(annihilator, &lower) == rational_zero())
673                    .then(|| (lower.clone(), lower));
674            }
675            let lower_is_root = dense::evaluate(annihilator, &lower) == rational_zero();
676            let open_count = real_root_count(
677                &annihilator_polynomial,
678                &crate::real_algebraic_numbers::dense::rational_to_number(&lower),
679                &crate::real_algebraic_numbers::dense::rational_to_number(&upper_bound),
680            )?;
681            let closed_count = open_count + usize::from(lower_is_root);
682            if closed_count == 0 {
683                return None;
684            }
685            if closed_count == 1 {
686                return Some((lower, upper_bound));
687            }
688        }
689        let mut refined = Vec::with_capacity(rectangles.len());
690        for rectangle in &rectangles {
691            let width = (&rectangle.real_upper - &rectangle.real_lower)
692                .max(&rectangle.imaginary_upper - &rectangle.imaginary_lower);
693            let target_width = width / Rational::from(4);
694            refined.push(refine_rectangle(&primitive, rectangle, &target_width, &mut budget).ok()?);
695        }
696        rectangles = refined;
697    }
698    None
699}

701fn present_real_algebraic_value(
702    context: &Context,
703    annihilator: &[Rational],
704    lower: Rational,
705    upper: Rational,
706) -> Option<ExpressionReference> {
707    // Turn the certified real root of the annihilator in [lower, upper] into an exact expression:
708    // a rational when the bracket pins it, a radical when the root solver's complete root list
709    // has exactly one real radical provably inside the bracket, and otherwise the canonical real
710    // root of the minimal polynomial.
711    if dense::evaluate(annihilator, &lower) == rational_zero() {
712        return Some(builders::rational_literal(context, lower));
713    }
714    if dense::evaluate(annihilator, &upper) == rational_zero() {
715        return Some(builders::rational_literal(context, upper));
716    }
717    let presentation_variable = context.symbol_table.fresh("__contour_residue_value");
718    let number = crate::real_algebraic_numbers::number::from_squarefree_and_interval(
719        dense::from_dense(presentation_variable, annihilator),
720        IsolatingInterval {
721            lower: lower.clone(),
722            upper: upper.clone(),
723        },
724    );
725    let minimal = crate::real_algebraic_numbers::number::minimal_polynomial_of(&number);
726    let minimal_dense = dense::to_dense(&minimal)?;
727    if dense::degree_of(&minimal_dense) == 1 {
728        return Some(builders::rational_literal(
729            context,
730            -(&minimal_dense[0] / &minimal_dense[1]),
731        ));
732    }
733    if dense::degree_of(&minimal_dense) <= MAXIMUM_RADICAL_PRESENTATION_DEGREE
734        && let Some(radical) =
735            radical_inside_bracket(context, &minimal, presentation_variable, &lower, &upper)
736    {
737        return Some(radical);
738    }
739    let primitive = PrimitiveIntegerPolynomial::from_rational_coefficients(&minimal_dense)?;
740    let bound = cauchy_bound(&minimal_dense);
741    let index = real_root_count(
742        &minimal,
743        &dense::rational_to_number(&-bound),
744        &dense::rational_to_number(&lower),
745    )?;
746    let algebraic =
747        crate::algebraic_numbers::root_expression::algebraic_number_from_primitive_and_index(
748            context, &primitive, index,
749        )
750        .ok()?;
751    match &algebraic.isolating_region {
752        AlgebraicIsolatingRegion::Real(interval)
753            if interval.upper >= lower && interval.lower <= upper =>
754        {
755            Some(builders::polynomial_root_from_algebraic_number(
756                context, algebraic,
757            ))
758        }
759        _ => None,
760    }
761}
762
763fn cauchy_bound(polynomial: &[Rational]) -> Rational {
764    // Every root of the polynomial has absolute value below 1 + max |a_k / a_n|.
765    let degree = dense::degree_of(polynomial);
766    let leading = &polynomial[degree];
767    let mut largest = rational_zero();
768    for coefficient in &polynomial[..degree] {
769        let ratio = crate::exact_arithmetic::rational_absolute_value(&(coefficient / leading));
770        if ratio > largest {
771            largest = ratio;
772        }
773    }
774    largest + Rational::from(2)
775}

777fn radical_inside_bracket(
778    context: &Context,
779    minimal: &crate::polynomial::Polynomial,
780    presentation_variable: SymbolIdentifier,
781    lower: &Rational,
782    upper: &Rational,
783) -> Option<ExpressionReference> {
784    // A root from the solver's complete list is a root of the minimal polynomial, and the
785    // bracket isolates one real root, so a real radical strictly inside it is the value.
786    let equation = crate::polynomial::conversion::polynomial_to_expression(context, minimal);
787    let SolveOutcome::CompleteRoots(candidates) =
788        solve_with_verdict(context, &equation, presentation_variable)
789    else {
790        return None;
791    };
792    let lower_expression = builders::rational_literal(context, lower.clone());
793    let upper_expression = builders::rational_literal(context, upper.clone());
794    let mut selected = None;
795    for candidate in candidates {
796        if crate::integrate::log_to_real::contains_imaginary(&candidate)
797            || contains_polynomial_root(&candidate)
798            || crate::assumptions::range_evidence::real_verdict(&context.symbol_table, &candidate)
799                != crate::assumptions::TruthValue::Yes
800        {
801            continue;
802        }
803        let above = crate::exact_real_order::compare_exact_real_expressions(
804            context,
805            &candidate,
806            &lower_expression,
807        );
808        let below = crate::exact_real_order::compare_exact_real_expressions(
809            context,
810            &candidate,
811            &upper_expression,
812        );
813        if matches!(above, Ok(Ordering::Greater)) && matches!(below, Ok(Ordering::Less)) {
814            if selected.is_some() {
815                return None;
816            }
817            selected = Some(candidate);
818        }
819    }
820    selected
821}

823fn contains_polynomial_root(expression: &ExpressionReference) -> bool {
824    // True when an algebraic-number atom occurs anywhere in the expression.
825    use crate::expression::ExpressionKind;
826    match &expression.kind {
827        ExpressionKind::PolynomialRoot(_) => true,
828        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
829            children.iter().any(contains_polynomial_root)
830        }
831        ExpressionKind::Power(base, exponent) => {
832            contains_polynomial_root(base) || contains_polynomial_root(exponent)
833        }
834        ExpressionKind::FunctionCall(_, arguments) => {
835            arguments.iter().any(contains_polynomial_root)
836        }
837        _ => false,
838    }
839}
