33pub fn integrate_rational_function(
34    context: &Context,
35    target: &ExpressionReference,
36    variable: SymbolIdentifier,
37) -> Option<ExpressionReference> {
38    integrate_rational_function_with_budget(context, target, variable, IntegrationBudget::initial())

41pub(crate) fn integrate_rational_function_with_budget(
42    context: &Context,
43    target: &ExpressionReference,
44    variable: SymbolIdentifier,
45    budget: IntegrationBudget,
46) -> Option<ExpressionReference> {
47    // Top-level entry for the rational-function integrator. The dispatcher in [integrate.rs] calls
48    // this whenever no earlier strategy fires; it attempts to read the integrand
49    // as a quotient of two univariate polynomials in the integration variable
50    // over Q and, on success, runs polynomial division, Hermite reduction, and
51    // the Lazard–Rioboo–Trager logarithmic-part construction. Returns Some on
52    // success and None when the integrand has any non-polynomial factor or
53    // when no rational-function shape applies — letting the caller fall back
54    // to the unevaluated `integral` placeholder.
55    let (numerator, denominator) = split_rational_function(target, variable)?;
56    if degree(&denominator).unwrap_or(0) == 0 {
57        return None;
58    }
59    if is_zero(&numerator) {
60        return Some(builders::zero(context));
61    }
62    let (quotient, proper_numerator) = divide_over_rationals(&numerator, &denominator)?;
63    let polynomial_part_integral = if is_zero(&quotient) {
64        builders::zero(context)
65    } else {
66        let quotient_expression = polynomial_to_expression(context, &quotient);
67        integrate_with_depth(context, &quotient_expression, variable, budget.clone())
68    };
69    if is_zero(&proper_numerator) {
70        return Some(polynomial_part_integral);
71    }
72    let (rational_part_pieces, residue_numerator, residue_denominator) =
73        hermite_reduce(context, &proper_numerator, &denominator, variable)?;
74    let residue_integral = if !rational_part_pieces.is_empty() {
75        let residue_expression =
76            build_rational_function_expression(context, &residue_numerator, &residue_denominator);
77        integrate_with_depth(context, &residue_expression, variable, budget.clone())
78    } else {
79        match lazard_rioboo_trager(
80            context,
81            &residue_numerator,
82            &residue_denominator,
83            variable,
84            context.operation(),
85        ) {
86            Some(value) => value,
87            None => {
88                let residue_expression = build_rational_function_expression(
89                    context,
90                    &residue_numerator,
91                    &residue_denominator,
92                );
93                builders::integral(context, residue_expression, variable, None, None)
94            }
95        }
96    };
97    let rational_part_sum = combine_pieces(context, rational_part_pieces);
98    let summed = builders::sum(
99        context,
100        vec![
101            polynomial_part_integral,
102            rational_part_sum,
103            residue_integral,
104        ],
105    );
106    Some(summed)
107}

109fn split_rational_function(
110    expression: &ExpressionReference,
111    variable: SymbolIdentifier,
112) -> Option<(Polynomial, Polynomial)> {
113    // Decompose the integrand into (numerator, denominator) where both are
114    // univariate polynomials in the integration variable over Q. The walk
115    // treats a top-level product as a list of factors and classifies each
116    // factor as a numerator or denominator contribution. Returns None when
117    // any factor contains a non-polynomial node (transcendental call, non-
118    // integer exponent, unbound foreign symbol) — that case is the dispatcher's
119    // signal to fall back to the unevaluated integral placeholder.
120    let one = constant_polynomial(variable, number_value::Integer(Integer::ONE));
121    match &expression.kind {
122        ExpressionKind::Product(factors) => {
123            let mut numerator_polynomial = one.clone();
124            let mut denominator_polynomial = one;
125            for factor in factors {
126                let (num_part, den_part) = split_single_factor(factor, variable)?;
127                numerator_polynomial = multiply(&numerator_polynomial, &num_part);
128                denominator_polynomial = multiply(&denominator_polynomial, &den_part);
129            }
130            Some((numerator_polynomial, denominator_polynomial))
131        }
132        _ => split_single_factor(expression, variable),
133    }
134}

136fn split_single_factor(
137    expression: &ExpressionReference,
138    variable: SymbolIdentifier,
139) -> Option<(Polynomial, Polynomial)> {
140    // Classify a single factor as either a numerator polynomial (returning
141    // (factor, 1)) or a reciprocal of a polynomial (returning (1, base^|n|)).
142    // The classification is purely structural: a power with a negative integer
143    // exponent is reciprocal, every other shape that converts to a polynomial
144    // is a numerator contribution. Returns None when the factor cannot be lifted
145    // into a polynomial in the integration variable.
146    let one = constant_polynomial(variable, number_value::Integer(Integer::ONE));
147    match &expression.kind {
148        ExpressionKind::Power(base, exponent) => {
149            let exponent_integer = match &exponent.kind {
150                ExpressionKind::Integer(value) => value.clone(),
151                _ => {
152                    let polynomial_value = expression_to_univariate(expression, variable)?;
153                    return Some((polynomial_value, one));
154                }
155            };
156            let signed_exponent = i64::try_from(&exponent_integer).ok()?;
157            let base_polynomial = expression_to_univariate(base, variable)?;
158            if signed_exponent >= 0 {
159                let raised = polynomial_pow(&base_polynomial, signed_exponent as usize);
160                Some((raised, one))
161            } else {
162                let magnitude = (-signed_exponent) as usize;
163                let raised = polynomial_pow(&base_polynomial, magnitude);
164                Some((one, raised))
165            }
166        }
167        _ => {
168            let polynomial_value = expression_to_univariate(expression, variable)?;
169            Some((polynomial_value, one))
170        }
171    }
172}

174pub(crate) fn hermite_reduce(
175    context: &Context,
176    proper_numerator: &Polynomial,
177    denominator: &Polynomial,
178    variable: SymbolIdentifier,
179) -> Option<(Vec<ExpressionReference>, Polynomial, Polynomial)> {
180    // Mack's variant of Hermite reduction (Bronstein, Symbolic Integration I,
181    // alg. 2.2.1). Squarefree-factors the denominator as D = V_1 * V_2^2 * ...
182    // * V_k^k via Yun, then iteratively reduces every factor whose
183    // multiplicity exceeds one. Each Hermite step replaces a denominator of the
184    // form U * V^m with U * V^(m-1) and contributes a rational-function piece
185    // -B / ((m-1) V^(m-1)) to the antiderivative. Returns (pieces, new_num,
186    // new_den) where new_den is the squarefree radical of the original
187    // denominator and new_num/new_den is the residue handed off to LRT (or to
188    // the dispatcher when a builtin-table shape can close it).
189    let _ = context;
190    let squarefree_factors =
191        yun_squarefree_factorization_with_operation(denominator, context.operation());
192    let mut current_numerator = proper_numerator.clone();
193    let mut current_denominator = denominator.clone();
194    let mut rational_pieces: Vec<ExpressionReference> = Vec::new();
195    let mut sorted_factors: Vec<(usize, Polynomial)> = squarefree_factors;
196    sorted_factors.sort_by(|left, right| right.0.cmp(&left.0));
197    for (original_multiplicity, squarefree_factor) in sorted_factors.iter() {
198        if *original_multiplicity < 2 {
199            continue;
200        }
201        let mut current_multiplicity = *original_multiplicity;
202        while current_multiplicity >= 2 {
203            let (piece, new_numerator, new_denominator) = match hermite_step(
204                context,
205                &current_numerator,
206                &current_denominator,
207                squarefree_factor,
208                current_multiplicity,
209                variable,
210            ) {
211                Some(value) => value,
212                None => {
213                    return Some((rational_pieces, current_numerator, current_denominator));
214                }
215            };
216            rational_pieces.push(piece);
217            current_numerator = new_numerator;
218            current_denominator = new_denominator;
219            current_multiplicity -= 1;
220        }
221    }
222    Some((rational_pieces, current_numerator, current_denominator))
223}

225fn hermite_step(
226    context: &Context,
227    numerator: &Polynomial,
228    denominator: &Polynomial,
229    squarefree_factor: &Polynomial,
230    multiplicity: usize,
231    _variable: SymbolIdentifier,
232) -> Option<(ExpressionReference, Polynomial, Polynomial)> {
233    // One pass of the Hermite recurrence. Given current state A/(U V^m), solve
234    // the Bezout identity B * U * V' + C * V = A (with deg(B) < deg(V)) via
235    // the extended Euclidean algorithm over Q[x], then apply the algebraic
236    // identity
237    //   A / (U V^m) = d/dx(-B / ((m-1) V^(m-1)))
238    //               + (U B'/(m-1) + C) / (U V^(m-1))
239    // to produce the rational-part contribution and the new numerator and
240    // denominator at multiplicity (m-1). Returns None when the inputs violate
241    // the algorithm's preconditions (e.g., gcd(V, U V') is non-constant, which
242    // shouldn't happen for a properly squarefree-decomposed denominator).
243    let factor_power = polynomial_pow(squarefree_factor, multiplicity);
244    let upper = polynomial_module::arithmetic::divide_exact(denominator, &factor_power)?;
245    let derivative_factor = formal_derivative(squarefree_factor);
246    let upper_times_derivative = multiply(&upper, &derivative_factor);
247    let (gcd_value, multiplier_factor, multiplier_upper_derivative) =
248        extended_euclidean(squarefree_factor, &upper_times_derivative)?;
249    if degree(&gcd_value).unwrap_or(0) != 0 {
250        return None;
251    }
252    let gcd_leading = leading_coefficient(&gcd_value)?;
253    let gcd_leading_value = coefficient_to_number(&gcd_leading)?;
254    let one_value = number_value::Integer(Integer::ONE);
255    let normalisation = number::divide(&one_value, &gcd_leading_value)?;
256    let normalised_multiplier_factor = scale_polynomial(&multiplier_factor, &normalisation);
257    let normalised_multiplier_upper_derivative =
258        scale_polynomial(&multiplier_upper_derivative, &normalisation);
259    let numerator_times_upper_derivative_multiplier =
260        multiply(numerator, &normalised_multiplier_upper_derivative);
261    let (sub_quotient, b_polynomial) = divide_over_rationals(
262        &numerator_times_upper_derivative_multiplier,
263        squarefree_factor,
264    )?;
265    let numerator_times_factor_multiplier = multiply(numerator, &normalised_multiplier_factor);
266    let sub_quotient_times_upper_derivative = multiply(&sub_quotient, &upper_times_derivative);
267    let c_polynomial = add(
268        &numerator_times_factor_multiplier,
269        &sub_quotient_times_upper_derivative,
270    );
271    let multiplicity_minus_one = multiplicity - 1;
272    let multiplicity_minus_one_number =
273        number_value::Integer(Integer::from(multiplicity_minus_one as i64));
274    let inverse_multiplicity_minus_one =
275        number::divide(&one_value, &multiplicity_minus_one_number)?;
276    let scaled_b = scale_polynomial(&b_polynomial, &inverse_multiplicity_minus_one);
277    let rational_numerator = polynomial_module::negate(&scaled_b);
278    let rational_denominator = polynomial_pow(squarefree_factor, multiplicity_minus_one);
279    let rational_piece =
280        build_rational_function_expression(context, &rational_numerator, &rational_denominator);
281    let b_derivative = formal_derivative(&b_polynomial);
282    let upper_times_b_derivative = multiply(&upper, &b_derivative);
283    let upper_times_b_derivative_scaled =
284        scale_polynomial(&upper_times_b_derivative, &inverse_multiplicity_minus_one);
285    let new_numerator = add(&upper_times_b_derivative_scaled, &c_polynomial);
286    let new_denominator = multiply(
287        &upper,
288        &polynomial_pow(squarefree_factor, multiplicity_minus_one),
289    );
290    Some((rational_piece, new_numerator, new_denominator))
291}

293fn lazard_rioboo_trager(
294    context: &Context,
295    numerator: &Polynomial,
296    denominator: &Polynomial,
297    variable: SymbolIdentifier,
298    operation: &OperationContext,
299) -> Option<ExpressionReference> {
300    // Logarithmic-part construction for a proper rational function p/q with q
301    // squarefree (Bronstein, Symbolic Integration I, §2.5, IntRationalLogPart).
302    // Computes R(t) = resultant_x(p - t * q', q) together with the subresultant
303    // polynomial remainder sequence, squarefree-factors R(t) over Q retaining
304    // multiplicities, and for each squarefree factor of multiplicity i emits one
305    // log term per distinct root. Rational roots (linear factors) become
306    // explicit `c_i * log(gcd_x(p - c_i q', q))` terms; higher-degree
307    // irreducible factors carry their roots as `polynomial_root` placeholders
308    // and use the subresultant of degree i as the log argument — the genuine
309    // Lazard–Rioboo construction, which keeps the argument at degree i rather
310    // than the full degree of p - t q'. Returns None (falling back to the
311    // unevaluated integral) when a required regular subresultant is missing.
312    if is_zero(denominator) || degree(denominator).unwrap_or(0) == 0 {
313        return Some(builders::zero(context));
314    }
315    let denominator_derivative = formal_derivative(denominator);
316    if is_zero(&denominator_derivative) {
317        return Some(builders::zero(context));
318    }
319    let parameter_symbol = context.symbol_table.fresh("integration_parameter");
320    let parameter_expression = context
321        .interner
322        .intern(ExpressionKind::Symbol(parameter_symbol));
323    let numerator_minus_parameter_derivative = build_bivariate_with_parameter(
324        numerator,
325        &denominator_derivative,
326        variable,
327        parameter_symbol,
328    );
329    let denominator_bivariate =
330        lift_polynomial_with_parameter(denominator, variable, parameter_symbol);
331    let resultant_coefficient = sylvester_resultant(
332        &numerator_minus_parameter_derivative,
333        &denominator_bivariate,
334    );
335    let resultant_polynomial = match resultant_coefficient {
336        PolynomialCoefficient::Nested(boxed) => *boxed,
337        PolynomialCoefficient::Numeric(numeric_value) => {
338            if number::is_zero(&numeric_value) {
339                return Some(builders::zero(context));
340            }
341            return Some(builders::zero(context));
342        }
343        PolynomialCoefficient::Algebraic(_) => return None,
344    };
345    if is_zero(&resultant_polynomial) {
346        return Some(builders::zero(context));
347    }
348    let resultant_over_integers = clear_denominators_for_factorisation(&resultant_polynomial)?;
349    let squarefree_factors =
350        yun_squarefree_factorization_with_operation(&resultant_over_integers, operation);
351    let subresultant_chain = subresultant_remainder_sequence(
352        &denominator_bivariate,
353        &numerator_minus_parameter_derivative,
354    );
355    let mut log_terms: Vec<ExpressionReference> = Vec::new();
356    for (multiplicity, squarefree_factor) in squarefree_factors.iter() {
357        let cleared_factor = clear_denominators_for_factorisation(squarefree_factor)?;
358        let irreducible_factors = factor_resultant_for_lrt(&cleared_factor, operation);
359        for (_, factor) in irreducible_factors.iter() {
360            let factor_degree = degree(factor).unwrap_or(0);
361            if factor_degree == 0 {
362                continue;
363            }
364            if factor_degree == 1 {
365                let root_number = match rational_root_of_linear_factor(factor) {
366                    Some(value) => value,
367                    None => continue,
368                };
369                let term = build_rational_root_log_term(
370                    context,
371                    numerator,
372                    &denominator_derivative,
373                    denominator,
374                    &root_number,
375                    variable,
376                    operation,
377                )?;
378                log_terms.push(term);
379            } else {
380                let irrational_terms = build_irrational_root_log_terms(
381                    context,
382                    factor,
383                    *multiplicity,
384                    denominator,
385                    &subresultant_chain,
386                    parameter_symbol,
387                    &parameter_expression,
388                    variable,
389                )?;
390                log_terms.extend(irrational_terms);
391            }
392        }
393    }
394    if log_terms.is_empty() {
395        Some(builders::zero(context))
396    } else {
397        Some(builders::sum(context, log_terms))
398    }
399}

401fn build_bivariate_with_parameter(
402    numerator: &Polynomial,
403    denominator_derivative: &Polynomial,
404    variable: SymbolIdentifier,
405    parameter: SymbolIdentifier,
406) -> Polynomial {
407    // Build p - t * q' as a polynomial in x whose coefficients live in Q[t].
408    // The result has nested coefficients on every term so the subresultant
409    // pseudo-remainder sequence used by `resultant` sees a uniform multivariate
410    // structure with no mixed numeric/nested arithmetic.
411    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
412    for (exponent, coefficient) in numerator.terms.iter() {
413        let coefficient_number = match coefficient {
414            PolynomialCoefficient::Numeric(value) => value.clone(),
415            PolynomialCoefficient::Nested(_) => continue,
416            PolynomialCoefficient::Algebraic(_) => continue,
417        };
418        let inner = constant_polynomial(parameter, coefficient_number);
419        let nested = PolynomialCoefficient::Nested(Box::new(inner));
420        polynomial_module::arithmetic::insert_or_combine(&mut terms, *exponent, nested);
421    }
422    for (exponent, coefficient) in denominator_derivative.terms.iter() {
423        let coefficient_number = match coefficient {
424            PolynomialCoefficient::Numeric(value) => value.clone(),
425            PolynomialCoefficient::Nested(_) => continue,
426            PolynomialCoefficient::Algebraic(_) => continue,
427        };
428        let negated = number::negate(&coefficient_number);
429        let inner = Polynomial {
430            main_variable: parameter,
431            terms: vec![(1, PolynomialCoefficient::Numeric(negated))],
432        };
433        let nested = PolynomialCoefficient::Nested(Box::new(inner));
434        polynomial_module::arithmetic::insert_or_combine(&mut terms, *exponent, nested);
435    }
436    Polynomial {
437        main_variable: variable,
438        terms,
439    }
440}

442fn lift_polynomial_with_parameter(
443    value: &Polynomial,
444    variable: SymbolIdentifier,
445    parameter: SymbolIdentifier,
446) -> Polynomial {
447    // Lift a univariate polynomial in x to a bivariate polynomial in (x, t)
448    // whose coefficients are degenerate constants in t. Required so the
449    // resultant in `lazard_rioboo_trager` sees both operands sharing nested
450    // coefficient types.
451    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::with_capacity(value.terms.len());
452    for (exponent, coefficient) in value.terms.iter() {
453        let coefficient_number = match coefficient {
454            PolynomialCoefficient::Numeric(value) => value.clone(),
455            PolynomialCoefficient::Nested(_) => continue,
456            PolynomialCoefficient::Algebraic(_) => continue,
457        };
458        let inner = constant_polynomial(parameter, coefficient_number);
459        let nested = PolynomialCoefficient::Nested(Box::new(inner));
460        terms.push((*exponent, nested));
461    }
462    Polynomial {
463        main_variable: variable,
464        terms,
465    }
466}

468fn clear_denominators_for_factorisation(value: &Polynomial) -> Option<Polynomial> {
469    // Scale a univariate polynomial in t over Q so all coefficients become
470    // integers. `factor_univariate_over_integers` expects integer-coefficient
471    // input and the LRT pipeline starts from the resultant, which may carry
472    // rational coefficients when q' has rational leading factors. Clearing
473    // denominators preserves the root structure modulo a non-zero constant.
474    let mut denominator_lcm = Integer::ONE;
475    for (_, coefficient) in value.terms.iter() {
476        let coefficient_number = match coefficient {
477            PolynomialCoefficient::Numeric(numeric) => numeric.clone(),
478            PolynomialCoefficient::Nested(_) => return None,
479            PolynomialCoefficient::Algebraic(_) => return None,
480        };
481        let denominator_integer = match coefficient_number {
482            number_value::Integer(_) => Integer::ONE,
483            number_value::Rational(rational_value) => {
484                let (_, denominator_natural) = rational_value.numerator_and_denominator_ref();
485                Integer::from(denominator_natural.clone())
486            }
487        };
488        denominator_lcm = lcm_integer(&denominator_lcm, &denominator_integer);
489    }
490    if denominator_lcm == Integer::ONE {
491        return Some(value.clone());
492    }
493    let scale_factor = number_value::Integer(denominator_lcm);
494    Some(scale_polynomial(value, &scale_factor))
495}

497fn factor_resultant_for_lrt(
498    value: &Polynomial,
499    operation: &OperationContext,
500) -> Vec<(usize, Polynomial)> {
501    // Factor the LRT resultant polynomial R(t) into its irreducible parts via
502    // the polynomial factorisation pipeline (Berlekamp–Zassenhaus + quadratic
503    // multifactor Hensel + van Hoeij LLL recombination). The
504    // rational-root pre-filter that previously wrapped this call was retired
505    // once Hensel lifting reached the Mignotte bound; the upstream factoriser
506    // now correctly handles every R(t) from the LRT subresultant chain
507    // without the linear-root scaffolding.
508    if is_zero(value) {
509        return Vec::new();
510    }
511    if degree(value).unwrap_or(0) == 0 {
512        return Vec::new();
513    }
514    crate::polynomial::factorization::factor_univariate_over_integers_with_operation(
515        value, operation,
516    )
517}

519fn lcm_integer(left: &Integer, right: &Integer) -> Integer {
520    // Integer least-common-multiple via the standard gcd identity. Returns
521    // zero only when both inputs are zero, which never happens in
522    // `clear_denominators_for_factorisation` because the cleared polynomial
523    // is non-zero by precondition.
524    if *left == Integer::ZERO && *right == Integer::ZERO {
525        return Integer::ZERO;
526    }
527    let gcd_value = crate::polynomial::gcd::integer_gcd(left, right);
528    let absolute_product =
529        crate::exact_arithmetic::integer_absolute_magnitude(&(left.clone() * right.clone()));
530    (absolute_product
531        / crate::exact_arithmetic::integer_to_natural(&gcd_value)
532            .expect("integer gcd is non-negative"))
533    .into()
534}

536fn rational_root_of_linear_factor(factor: &Polynomial) -> Option<number_value> {
537    // Read the root of a linear polynomial a*t + b as -b/a. The polynomial is
538    // expected to have plain numeric coefficients; nested-coefficient inputs
539    // signal a misuse of LRT and produce None.
540    let mut linear_coefficient: Option<number_value> = None;
541    let mut constant_coefficient = number_value::Integer(Integer::ZERO);
542    for (exponent, coefficient) in factor.terms.iter() {
543        let coefficient_number = coefficient_to_number(coefficient)?;
544        match *exponent {
545            1 => linear_coefficient = Some(coefficient_number),
546            0 => constant_coefficient = coefficient_number,
547            _ => return None,
548        }
549    }
550    let linear_value = linear_coefficient?;
551    let negated_constant = number::negate(&constant_coefficient);
552    number::divide(&negated_constant, &linear_value)
553}

555fn build_rational_root_log_term(
556    context: &Context,
557    numerator: &Polynomial,
558    denominator_derivative: &Polynomial,
559    denominator: &Polynomial,
560    root: &number_value,
561    variable: SymbolIdentifier,
562    operation: &OperationContext,
563) -> Option<ExpressionReference> {
564    // For a rational root c of R(t), emit c * log(g) where g = gcd_x(p - c * q', q),
565    // routed through the shared antiderivative_logarithm choke point so the
566    // logarithm carries an absolute value only under a real-domain assumption on
567    // the integration variable — uniformly with every other factor degree, never
568    // keyed on whether g happens to be linear.
569    let scaled_derivative = scale_polynomial(denominator_derivative, root);
570    let negated_scaled_derivative = polynomial_module::negate(&scaled_derivative);
571    let shifted_numerator = add(numerator, &negated_scaled_derivative);
572    let gcd_polynomial = crate::polynomial::gcd::polynomial_gcd_with_operation(
573        &shifted_numerator,
574        denominator,
575        operation,
576    );
577    let monic_gcd = monic_normalise(&gcd_polynomial)?;
578    let argument_degree = degree(&monic_gcd).unwrap_or(0);
579    if argument_degree == 0 {
580        return Some(builders::zero(context));
581    }
582    let argument_expression = polynomial_to_expression(context, &monic_gcd);
583    let log_call = super::antiderivative_logarithm(context, argument_expression, variable);
584    let root_expression = intern_number_value(context, root.clone());
585    Some(builders::product(context, vec![root_expression, log_call]))
586}

588fn build_irrational_root_log_terms(
589    context: &Context,
590    factor: &Polynomial,
591    multiplicity: usize,
592    denominator: &Polynomial,
593    subresultant_chain: &[Polynomial],
594    parameter: SymbolIdentifier,
595    parameter_expression: &ExpressionReference,
596    variable: SymbolIdentifier,
597) -> Option<Vec<ExpressionReference>> {
598    // Lazard–Rioboo–Trager log argument for an irreducible factor f(z) of the
599    // Rothstein–Trager resultant with deg(f) > 1 and squarefree multiplicity i
600    // (Bronstein §2.5). The log argument is the subresultant of degree i in the
601    // integration variable: when i equals deg(D) it is D itself, otherwise it is
602    // the chain member of degree i (made primitive in the integration variable
603    // so the displayed argument carries no spurious parameter-only factor). Each
604    // of the deg(f) roots of f contributes one term root_k * log(S_i(root_k, x)),
605    // the root carried by a `polynomial_root` placeholder substituted for the
606    // parameter z. Returns None — falling the integrator back to an unevaluated
607    // integral — when the chain has no regular subresultant of the required
608    // degree, the defective case the construction cannot close rigorously.
609    let denominator_degree = degree(denominator).unwrap_or(0);
610    let log_argument_polynomial = if multiplicity == denominator_degree {
611        lift_polynomial_with_parameter(denominator, variable, parameter)
612    } else {
613        let selected = subresultant_chain
614            .iter()
615            .find(|member| degree(member) == Some(multiplicity))?;
616        crate::polynomial::gcd::primitive_part(selected)
617    };
618    if degree(factor) == Some(2) {
619        if let Some(real_form) = build_complex_quadratic_pair_real_form(
620            context,
621            factor,
622            &log_argument_polynomial,
623            parameter_expression,
624            variable,
625        ) {
626            return Some(vec![real_form]);
627        }
628    }
629    if let Some(real_forms) = build_quartic_binomial_pair_real_forms(
630        context,
631        factor,
632        &log_argument_polynomial,
633        parameter_expression,
634        variable,
635    ) {
636        return Some(real_forms);
637    }
638    let factor_expression = polynomial_to_expression(context, factor);
639    let log_argument_template_expression =
640        polynomial_to_expression(context, &log_argument_polynomial);
641    let degree_value = degree(factor).unwrap_or(0);
642    let mut log_terms: Vec<ExpressionReference> = Vec::with_capacity(degree_value);
643    for root_index in 0..degree_value {
644        let root_placeholder = polynomial_root_node(
645            context,
646            Arc::clone(&factor_expression),
647            parameter,
648            root_index,
649        );
650        let mut mapping: SubstitutionMap = HashMap::new();
651        mapping.insert(
652            Arc::clone(parameter_expression),
653            Arc::clone(&root_placeholder),
654        );
655        let substituted_log_argument =
656            substitute(context, &log_argument_template_expression, &mapping);
657        let log_call = super::antiderivative_logarithm(context, substituted_log_argument, variable);
658        let term = builders::product(context, vec![root_placeholder, log_call]);
659        log_terms.push(term);
660    }
661    Some(log_terms)
662}

664fn build_quartic_binomial_pair_real_forms(
665    context: &Context,
666    factor: &Polynomial,
667    log_argument_polynomial: &Polynomial,
668    parameter_expression: &ExpressionReference,
669    variable: SymbolIdentifier,
670) -> Option<Vec<ExpressionReference>> {
671    let root_part = quartic_binomial_root_part(context, factor)?;
672    let positive_pair = build_quartic_binomial_single_pair_real_form(
673        context,
674        log_argument_polynomial,
675        parameter_expression,
676        &root_part,
677        false,
678        variable,
679    )?;
680    let negative_pair = build_quartic_binomial_single_pair_real_form(
681        context,
682        log_argument_polynomial,
683        parameter_expression,
684        &root_part,
685        true,
686        variable,
687    )?;
688    Some(vec![positive_pair, negative_pair])
689}

691fn build_quartic_binomial_single_pair_real_form(
692    context: &Context,
693    log_argument_polynomial: &Polynomial,
694    parameter_expression: &ExpressionReference,
695    root_part: &ExpressionReference,
696    negate_real_part: bool,
697    variable: SymbolIdentifier,
698) -> Option<ExpressionReference> {
699    let atom_symbol = context.symbol_table.fresh("quartic_imaginary_root_atom");
700    let atom = context.interner.intern(ExpressionKind::Symbol(atom_symbol));
701    let coefficient_real = if negate_real_part {
702        builders::negate(context, Arc::clone(root_part))
703    } else {
704        Arc::clone(root_part)
705    };
706    let root_expression = builders::sum(
707        context,
708        vec![Arc::clone(&coefficient_real), Arc::clone(&atom)],
709    );
710    let log_argument_template_expression =
711        polynomial_to_expression(context, log_argument_polynomial);
712    let mut mapping: SubstitutionMap = HashMap::new();
713    mapping.insert(Arc::clone(parameter_expression), root_expression);
714    let substituted_argument = substitute(context, &log_argument_template_expression, &mapping);
715    let two = builders::integer_value(context, 2);
716    let root_part_squared = builders::power(context, Arc::clone(root_part), two);
717    let radicand = builders::negate(context, root_part_squared);
718    let (argument_real, argument_imaginary) =
719        split_real_imaginary(context, &substituted_argument, &atom, &radicand)?;
720    let coefficient_imaginary = builders::one(context);
721    let term = ComplexLogarithmTerm {
722        coefficient_real,
723        coefficient_imaginary,
724        argument_real,
725        argument_imaginary,
726    };
727    Some(build_real_logarithm_pair(
728        context, &term, root_part, variable,
729    ))
730}

732fn quartic_binomial_root_part(
733    context: &Context,
734    factor: &Polynomial,
735) -> Option<ExpressionReference> {
736    if degree(factor) != Some(4) {
737        return None;
738    }
739    let mut leading: Option<number_value> = None;
740    let mut constant: Option<number_value> = None;
741    for (exponent, coefficient) in factor.terms.iter() {
742        let coefficient_number = coefficient_to_number(coefficient)?;
743        match *exponent {
744            4 => leading = Some(coefficient_number),
745            0 => constant = Some(coefficient_number),
746            _ => return None,
747        }
748    }
749    let leading = leading?;
750    let constant = constant?;
751    if number::is_zero(&leading) || number::is_zero(&constant) {
752        return None;
753    }
754    let ratio = number::divide(&constant, &leading)?;
755    if number::is_negative(&ratio) || number::is_zero(&ratio) {
756        return None;
757    }
758    let one_quarter = Rational::from_integers(Integer::ONE, Integer::from(4));
759    let root_magnitude = number_power_expression(context, ratio, one_quarter);
760    let sqrt_two = {
761        let two = builders::integer_value(context, 2);
762        let one_half = builders::rational_value(context, 1, 2);
763        builders::power(context, two, one_half)
764    };
765    let one_half = builders::rational_value(context, 1, 2);
766    Some(builders::product(
767        context,
768        vec![root_magnitude, sqrt_two, one_half],
769    ))
770}
771
772fn number_power_expression(
773    context: &Context,
774    base: number_value,
775    exponent: Rational,
776) -> ExpressionReference {
777    if let Some(number_value) = number::power_rational(&base, &exponent) {
778        return intern_number_value(context, number_value);
779    }
780    let base_expression = intern_number_value(context, base);
781    let exponent_expression = rational_expression(context, exponent);
782    builders::power(context, base_expression, exponent_expression)
783}

785fn rational_expression(context: &Context, value: Rational) -> ExpressionReference {
786    let sign = if value < Rational::ZERO { -1 } else { 1 };
787    let (numerator, denominator) = value.into_numerator_and_denominator();
788    let signed_numerator = Integer::from(sign) * Integer::from(numerator);
789    builders::rational(context, signed_numerator, Integer::from(denominator))
790}

792fn quadratic_complex_root_parts(factor: &Polynomial) -> Option<(number_value, number_value)> {
793    // Read the complex-conjugate root pair of a degree-2 factor
794    // `c2 t^2 + c1 t + c0` over Q with strictly negative discriminant. Returns
795    // (real_part, radicand) where real_part = -c1/(2 c2) is the rational real
796    // part and radicand = discriminant/(2 c2)^2 is a strictly negative rational
797    // with the imaginary atom s = sqrt(radicand) satisfying s^2 = radicand, so
798    // the root is real_part + s = real_part + i*sqrt(-radicand). Returns None
799    // for any factor that is not a quadratic with rational coefficients, or
800    // whose discriminant is non-negative (two real roots) — both of which keep
801    // the existing placeholder path, which already renders real surd forms.
802    let mut leading = number_value::Integer(Integer::ZERO);
803    let mut linear = number_value::Integer(Integer::ZERO);
804    let mut constant = number_value::Integer(Integer::ZERO);
805    for (exponent, coefficient) in factor.terms.iter() {
806        let coefficient_number = coefficient_to_number(coefficient)?;
807        match *exponent {
808            2 => leading = coefficient_number,
809            1 => linear = coefficient_number,
810            0 => constant = coefficient_number,
811            _ => return None,
812        }
813    }
814    if number::is_zero(&leading) {
815        return None;
816    }
817    let linear_squared = number::multiply(&linear, &linear);
818    let four = number_value::Integer(Integer::from(4));
819    let four_leading_constant = number::multiply(&number::multiply(&four, &leading), &constant);
820    let discriminant = number::subtract(&linear_squared, &four_leading_constant);
821    let two = number_value::Integer(Integer::from(2));
822    let two_leading = number::multiply(&two, &leading);
823    let two_leading_squared = number::multiply(&two_leading, &two_leading);
824    let radicand = number::divide(&discriminant, &two_leading_squared)?;
825    if !number::is_negative(&radicand) {
826        return None;
827    }
828    let real_part = number::divide(&number::negate(&linear), &two_leading)?;
829    Some((real_part, radicand))
830}

832fn build_complex_quadratic_pair_real_form(
833    context: &Context,
834    factor: &Polynomial,
835    log_argument_polynomial: &Polynomial,
836    parameter_expression: &ExpressionReference,
837    variable: SymbolIdentifier,
838) -> Option<ExpressionReference> {
839    // Rioboo's LogToReal step for an irreducible quadratic factor f(t) of the
840    // Rothstein–Trager resultant whose roots are a complex-conjugate pair
841    // a ± i*m (Bronstein, Symbolic Integration I, §2.8). Emits the real
842    // antiderivative term a*log(A^2 + B^2) - 2m*arctan(B/A) directly, where the
843    // degree-i subresultant log argument S_i(a + i*m, x) splits as A(x) + i*B(x).
844    // The construction substitutes the explicit root z := a + s (with the atom
845    // s the algebraic indeterminate satisfying radicand < 0, s^2 = radicand and
846    // s = i*m) into the subresultant template and performs the {1, s}-basis split
847    // with the shared log_to_real primitives, so the complex
848    // sqrt(negative)/exp(i*theta) intermediate form is never built as an
849    // antiderivative term. The atom is a fresh symbol rather than the literal
850    // sqrt(radicand): automatic simplification rewrites a literal square root of a
851    // negative rational into the imaginary-unit form (k*i), which would dissolve
852    // the ptr-identity marker `split_real_imaginary` relies on; a fresh symbol is
853    // inert under simplification and is substituted away before the result is
854    // returned, so it never leaks into the antiderivative. The log factor routes
855    // through the antiderivative_logarithm choke point inside
856    // build_real_logarithm_pair. Returns None when the factor is not a quadratic
857    // with a negative discriminant, letting the caller keep the placeholder path
858    // for the real-root and higher-degree cases.
859    let (real_part, radicand) = quadratic_complex_root_parts(factor)?;
860    let real_part_expression = intern_number_value(context, real_part);
861    let radicand_expression = intern_number_value(context, radicand.clone());
862    let atom_symbol = context.symbol_table.fresh("imaginary_root_atom");
863    let atom = context.interner.intern(ExpressionKind::Symbol(atom_symbol));
864    let root_expression = builders::sum(
865        context,
866        vec![Arc::clone(&real_part_expression), Arc::clone(&atom)],
867    );
868    let log_argument_template_expression =
869        polynomial_to_expression(context, log_argument_polynomial);
870    let mut mapping: SubstitutionMap = HashMap::new();
871    mapping.insert(Arc::clone(parameter_expression), root_expression);
872    let substituted_argument = substitute(context, &log_argument_template_expression, &mapping);
873    let (argument_real, argument_imaginary) =
874        split_real_imaginary(context, &substituted_argument, &atom, &radicand_expression)?;
875    let imaginary_magnitude_radicand = intern_number_value(context, number::negate(&radicand));
876    let one_half_for_magnitude = builders::rational_value(context, 1, 2);
877    let imaginary_magnitude = builders::power(
878        context,
879        imaginary_magnitude_radicand,
880        one_half_for_magnitude,
881    );
882    let coefficient_imaginary = builders::one(context);
883    let term = ComplexLogarithmTerm {
884        coefficient_real: real_part_expression,
885        coefficient_imaginary,
886        argument_real,
887        argument_imaginary,
888    };
889    let real_form = build_real_logarithm_pair(context, &term, &imaginary_magnitude, variable);
890    Some(real_form)
891}

893fn monic_normalise(value: &Polynomial) -> Option<Polynomial> {
894    // Scale a polynomial by the inverse of its leading coefficient so the
895    // result is monic. Returns None when the input is zero or has a non-
896    // numeric leading coefficient — both of which signal a misuse upstream.
897    if is_zero(value) {
898        return Some(value.clone());
899    }
900    let leading = leading_coefficient(value)?;
901    let leading_number = coefficient_to_number(&leading)?;
902    if number::is_zero(&leading_number) {
903        return None;
904    }
905    let one_value = number_value::Integer(Integer::ONE);
906    let inverse = number::divide(&one_value, &leading_number)?;
907    Some(scale_polynomial(value, &inverse))
908}

910fn intern_number_value(context: &Context, value: number_value) -> ExpressionReference {
911    // Wrap a number value into an interned expression leaf, routing through
912    // the integer and rational builders so the integer-collapse invariant is
913    // preserved (a rational whose denominator equals one collapses back to the
914    // integer leaf rather than persisting as a rational with denominator one).
915    match value {
916        number_value::Integer(integer_value) => builders::integer(context, integer_value),
917        number_value::Rational(rational_value) => {
918            builders::rational_literal(context, rational_value)
919        }
920    }
921}

923fn build_rational_function_expression(
924    context: &Context,
925    numerator: &Polynomial,
926    denominator: &Polynomial,
927) -> ExpressionReference {
928    // Compose num/den back into an expression via the auto-simplifying divide
929    // builder. Constant denominators collapse through the simplifier so a
930    // proper rational with denominator 1 returns the plain numerator
931    // expression, which keeps the recursion in `integrate_rational_function`
932    // free of redundant 1/1 wrappers.
933    let numerator_expression = polynomial_to_expression(context, numerator);
934    let denominator_expression = polynomial_to_expression(context, denominator);
935    builders::divide(context, numerator_expression, denominator_expression)
936}

938fn combine_pieces(context: &Context, pieces: Vec<ExpressionReference>) -> ExpressionReference {
939    // Reduce a vector of Hermite rational-part pieces to a single expression.
940    // An empty vector folds to integer zero; a singleton returns its only
941    // element; anything else passes through the auto-simplifying sum builder
942    // so like terms combine when they happen to share canonical structure.
943    match pieces.len() {
944        0 => builders::zero(context),
945        1 => pieces.into_iter().next().unwrap(),
946        _ => builders::sum(context, pieces),
947    }
948}
