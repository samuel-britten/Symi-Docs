25// =============================================================================
26// Bronstein Chapter 6 Risch differential equation decision
27// procedure over purely transcendental exp/log towers.
28//
29// Solve D(y) + f*y = g for y in k(t), t the top monomial of `tower`, returning
30//   solved(y)     a verified rational-function solution (D(y) + f*y = g holds),
31//   no_solution   a decided Liouville witness that no elementary y exists,
32//   deferred      an out-of-scope input (algebraic generator, rational f, the
33//                 parametric cancellation cases, none of which are yet implemented).
34//
35// The base level (tower = Q(x)) runs entirely on numeric polynomials in Q[x];
36// higher levels run on `tower_polynomial` (k[t] with k a rational-function
37// field). Both share the same skeleton: RdeNormalDenominator (§6.1),
38// RdeSpecialDenominator (§6.2, an identity once the denominator is normal),
39// RdeBoundDegree (§6.4), the iterative SPDE (§6.3), and the non-cancellation
40// polynomial solver (§6.5). Every solved candidate is verified by exact
41// substitution before it is returned, so a wrong closed form is impossible;
42// `no_solution` is returned only from a bounded decision.
43// =============================================================================
44
45/// Public native enumeration named `rde_outcome`.
46pub enum RdeOutcome {
47    /// Represents the `solved` case.
48    Solved(RationalFunction),
49    /// Represents the `no_solution` case.
50    NoSolution,
51    /// Represents the `deferred` case.
52    Deferred,
53}
54
55/// Performs the advanced native `solve` operation.
56pub fn solve(f: &RationalFunction, g: &RationalFunction, tower: &DifferentialField) -> RdeOutcome {
57    // Decision-procedure entry. Dispatches on tower height, then verifies any
58    // proposed closed form against the original equation. A candidate that
59    // fails verification is downgraded to `deferred` rather than returned as a
60    // wrong solution.
61    let candidate = if tower.extensions.is_empty() {
62        solve_base(f, g, tower)
63    } else {
64        solve_transcendental(f, g, tower)
65    };
66    match candidate {
67        RdeOutcome::Solved(y) => {
68            if verify_solution(f, &y, g, tower) {
69                RdeOutcome::Solved(y)
70            } else {
71                RdeOutcome::Deferred
72            }
73        }
74        other => other,
75    }
76}
77
78fn verify_solution(
79    f: &RationalFunction,
80    y: &RationalFunction,
81    g: &RationalFunction,
82    tower: &DifferentialField,
83) -> bool {
84    // Recompute D(y) + f*y - g with the tower derivation and require it to be
85    // exactly zero. This is the hard rigor gate for the closed-form branch.
86    let derivative = derive_rational_function(tower, y);
87    let product = multiply_rational_functions(f, y);
88    let left = crate::differential_field::arithmetic::add_rational_functions(&derivative, &product);
89    let residual = subtract_rational_functions(&left, g);
90    rational_function_is_zero(&residual)
91}

93fn lower_subfield(tower: &DifferentialField) -> DifferentialField {
94    let mut lower = DifferentialField {
95        base_variable: tower.base_variable,
96        extensions: tower.extensions.clone(),
97    };
98    lower.extensions.pop();
99    lower
100}

106fn solve_base(f: &RationalFunction, g: &RationalFunction, tower: &DifferentialField) -> RdeOutcome {
107    let base_variable = tower.base_variable;
108    let f_polynomial = match base_polynomial_from_rational_function(f, base_variable) {
109        Some(value) => value,
110        None => return RdeOutcome::Deferred,
111    };
112    let numerator = match require_numeric_polynomial(&g.numerator, base_variable) {
113        Some(value) => value,
114        None => return RdeOutcome::Deferred,
115    };
116    let denominator = match require_numeric_polynomial(&g.denominator, base_variable) {
117        Some(value) => value,
118        None => return RdeOutcome::Deferred,
119    };
120    // RdeNormalDenominator with f weakly normal (f is a polynomial, so its
121    // denominator is 1): the candidate denominator's normal part is
122    // h = gcd(gd, D(gd)); the equation for q = h*y is a*q' + b*q = c.
123    let denominator_derivative = formal_derivative(&denominator);
124    let h = base_monic_gcd(&denominator, &denominator_derivative, base_variable);
125    let h_squared = polynomial_multiply(&h, &h);
126    let (h_squared_over_denominator, remainder) =
127        match base_divide(&h_squared, &denominator, base_variable) {
128            Some(value) => value,
129            None => return RdeOutcome::Deferred,
130        };
131    if !polynomial_is_zero(&remainder) {
132        return RdeOutcome::NoSolution;
133    }
134    let a = h.clone();
135    let b = polynomial_subtract(
136        &polynomial_multiply(&f_polynomial, &h),
137        &formal_derivative(&h),
138    );
139    let c = polynomial_multiply(&numerator, &h_squared_over_denominator);
140    let degree_bound = bound_degree_base(&a, &b, &c);
141    let reduced = match spde_base(a, b, c, degree_bound, base_variable) {
142        BaseSpde::Reduced {
143            b,
144            c,
145            n,
146            alpha,
147            beta,
148        } => (b, c, n, alpha, beta),
149        BaseSpde::NoSolution => return RdeOutcome::NoSolution,
150        BaseSpde::Deferred => return RdeOutcome::Deferred,
151    };
152    let (reduced_b, reduced_c, reduced_n, alpha, beta) = reduced;
153    let solution_polynomial = if polynomial_is_zero(&reduced_c) {
154        zero_polynomial(base_variable)
155    } else if reduced_n < 0 {
156        return RdeOutcome::NoSolution;
157    } else if polynomial_is_zero(&reduced_b) {
158        match base_integrate(&reduced_c, base_variable) {
159            Some(value) => value,
160            None => return RdeOutcome::Deferred,
161        }
162    } else {
163        match no_cancel_b_large_base(&reduced_b, &reduced_c, reduced_n, base_variable) {
164            BaseSolve::Solved(value) => value,
165            BaseSolve::NoSolution => return RdeOutcome::NoSolution,
166            BaseSolve::Deferred => return RdeOutcome::Deferred,
167        }
168    };
169    let assembled = polynomial_add(&polynomial_multiply(&alpha, &solution_polynomial), &beta);
170    match make_rational_function(assembled, h) {
171        Some(value) => RdeOutcome::Solved(value),
172        None => RdeOutcome::Deferred,
173    }
174}

176fn bound_degree_base(a: &Polynomial, b: &Polynomial, c: &Polynomial) -> isize {
177    // RdeBoundDegree, base case. n = max(0, deg c - max(deg b, deg a - 1)),
178    // refined by the cancellation subcase deg b = deg a - 1 with an integer
179    // leading-coefficient ratio.
180    let degree_a = polynomial_degree(a)
181        .map(|value| value as isize)
182        .unwrap_or(-1);
183    let degree_c = polynomial_degree(c)
184        .map(|value| value as isize)
185        .unwrap_or(-1);
186    if polynomial_is_zero(b) {
187        return (degree_c - degree_a + 1).max(0);
188    }
189    let degree_b = polynomial_degree(b)
190        .map(|value| value as isize)
191        .unwrap_or(-1);
192    let mut bound = (degree_c - degree_b.max(degree_a - 1)).max(0);
193    if degree_b == degree_a - 1 {
194        if let Some(ratio) = base_leading_ratio(b, a) {
195            if let Some(integer_ratio) = number_as_integer(&ratio) {
196                if let Ok(integer_value) = isize::try_from(&integer_ratio) {
197                    bound = bound.max(integer_value).max(degree_c - degree_b);
198                }
199            }
200        }
201    }
202    bound.max(0)
203}

205enum BaseSpde {
206    Reduced {
207        b: Polynomial,
208        c: Polynomial,
209        n: isize,
210        alpha: Polynomial,
211        beta: Polynomial,
212    },
213    NoSolution,
214    Deferred,
215}

217fn spde_base(
218    a: Polynomial,
219    b: Polynomial,
220    c: Polynomial,
221    n: isize,
222    base_variable: SymbolIdentifier,
223) -> BaseSpde {
224    // The Rothstein SPDE reduction (Bronstein §6.3), iterative to avoid deep
225    // recursion. Peels off a computable part of q by one division and one gcd
226    // per step until the leading coefficient a becomes a unit.
227    let mut a = a;
228    let mut b = b;
229    let mut c = c;
230    let mut n = n;
231    let mut alpha = constant_polynomial(base_variable, number_value::Integer(Integer::from(1)));
232    let mut beta = zero_polynomial(base_variable);
233    loop {
234        if polynomial_is_zero(&c) {
235            return BaseSpde::Reduced {
236                b: zero_polynomial(base_variable),
237                c: zero_polynomial(base_variable),
238                n,
239                alpha,
240                beta,
241            };
242        }
243        if n < 0 {
244            return BaseSpde::NoSolution;
245        }
246        let g = base_monic_gcd(&a, &b, base_variable);
247        let (a_reduced, a_remainder) = match base_divide(&a, &g, base_variable) {
248            Some(value) => value,
249            None => return BaseSpde::Deferred,
250        };
251        let (b_reduced, b_remainder) = match base_divide(&b, &g, base_variable) {
252            Some(value) => value,
253            None => return BaseSpde::Deferred,
254        };
255        let (c_reduced, c_remainder) = match base_divide(&c, &g, base_variable) {
256            Some(value) => value,
257            None => return BaseSpde::Deferred,
258        };
259        if !polynomial_is_zero(&a_remainder)
260            || !polynomial_is_zero(&b_remainder)
261            || !polynomial_is_zero(&c_remainder)
262        {
263            return BaseSpde::NoSolution;
264        }
265        a = a_reduced;
266        b = b_reduced;
267        c = c_reduced;
268        if polynomial_degree(&a) == Some(0) {
269            let leading = match base_constant_number(&a) {
270                Some(value) => value,
271                None => return BaseSpde::Deferred,
272            };
273            let inverse = match number::divide(&number_value::Integer(Integer::from(1)), &leading) {
274                Some(value) => value,
275                None => return BaseSpde::Deferred,
276            };
277            return BaseSpde::Reduced {
278                b: scale_polynomial(&b, &inverse),
279                c: scale_polynomial(&c, &inverse),
280                n,
281                alpha,
282                beta,
283            };
284        }
285        let degree_a = polynomial_degree(&a).unwrap();
286        let (r, z) = match base_diophantine(&b, &a, &c, base_variable) {
287            Some(value) => value,
288            None => return BaseSpde::Deferred,
289        };
290        let new_b = polynomial_add(&b, &formal_derivative(&a));
291        let new_c = polynomial_subtract(&z, &formal_derivative(&r));
292        n -= degree_a as isize;
293        beta = polynomial_add(&beta, &polynomial_multiply(&alpha, &r));
294        alpha = polynomial_multiply(&alpha, &a);
295        b = new_b;
296        c = new_c;
297    }
298}

306fn no_cancel_b_large_base(
307    b: &Polynomial,
308    c: &Polynomial,
309    n: isize,
310    base_variable: SymbolIdentifier,
311) -> BaseSolve {
312    // Non-cancellation polynomial solver (Bronstein §6.5): the top-down
313    // coefficient recurrence for D(q) + b*q = c with deg(b) large enough that
314    // the leading terms never cancel.
315    let mut q = zero_polynomial(base_variable);
316    let mut c = c.clone();
317    let mut n = n;
318    let degree_b = match polynomial_degree(b) {
319        Some(value) => value as isize,
320        None => return BaseSolve::NoSolution,
321    };
322    while !polynomial_is_zero(&c) {
323        let degree_c = polynomial_degree(&c).unwrap() as isize;
324        let m = degree_c - degree_b;
325        if m < 0 || m > n {
326            return BaseSolve::NoSolution;
327        }
328        let leading_c = match base_leading_number(&c) {
329            Some(value) => value,
330            None => return BaseSolve::Deferred,
331        };
332        let leading_b = match base_leading_number(b) {
333            Some(value) => value,
334            None => return BaseSolve::Deferred,
335        };
336        let ratio = match number::divide(&leading_c, &leading_b) {
337            Some(value) => value,
338            None => return BaseSolve::Deferred,
339        };
340        let p = base_monomial(base_variable, ratio, m as usize);
341        q = polynomial_add(&q, &p);
342        n = m - 1;
343        let subtraction = polynomial_add(&formal_derivative(&p), &polynomial_multiply(b, &p));
344        c = polynomial_subtract(&c, &subtraction);
345    }
346    BaseSolve::Solved(q)
347}

353fn base_polynomial_from_rational_function(
354    value: &RationalFunction,
355    base_variable: SymbolIdentifier,
356) -> Option<Polynomial> {
357    let denominator = base_constant_number(&value.denominator)?;
358    if number::is_zero(&denominator) {
359        return None;
360    }
361    let numerator = require_numeric_polynomial(&value.numerator, base_variable)?;
362    let inverse = number::divide(&number_value::Integer(Integer::from(1)), &denominator)?;
363    Some(scale_polynomial(&numerator, &inverse))
364}
365
366fn require_numeric_polynomial(
367    value: &Polynomial,
368    base_variable: SymbolIdentifier,
369) -> Option<Polynomial> {
370    if polynomial_is_zero(value) {
371        return Some(zero_polynomial(base_variable));
372    }
373    if value.main_variable != base_variable {
374        let constant = base_constant_number(value)?;
375        return Some(constant_polynomial(base_variable, constant));
376    }
377    for (_, coefficient) in value.terms.iter() {
378        if coefficient_to_number(coefficient).is_none() {
379            return None;
380        }
381    }
382    Some(value.clone())
383}
384
385fn base_constant_number(value: &Polynomial) -> Option<number_value> {
386    if polynomial_is_zero(value) {
387        return Some(number_value::Integer(Integer::from(0)));
388    }
389    if value.terms.len() != 1 || value.terms[0].0 != 0 {
390        return None;
391    }
392    coefficient_to_number(&value.terms[0].1)
393}

395fn base_leading_number(value: &Polynomial) -> Option<number_value> {
396    coefficient_to_number(&leading_coefficient(value)?)
397}

399fn base_leading_ratio(b: &Polynomial, a: &Polynomial) -> Option<number_value> {
400    let leading_b = base_leading_number(b)?;
401    let leading_a = base_leading_number(a)?;
402    let ratio = number::divide(&leading_b, &leading_a)?;
403    Some(number::negate(&ratio))
404}

406fn base_monomial(
407    base_variable: SymbolIdentifier,
408    coefficient: number_value,
409    exponent: usize,
410) -> Polynomial {
411    if number::is_zero(&coefficient) {
412        return zero_polynomial(base_variable);
413    }
414    Polynomial {
415        main_variable: base_variable,
416        terms: vec![(exponent, PolynomialCoefficient::Numeric(coefficient))],
417    }

420fn base_monic(value: &Polynomial, base_variable: SymbolIdentifier) -> Polynomial {
421    if polynomial_is_zero(value) {
422        return value.clone();
423    }
424    let leading = match base_leading_number(value) {
425        Some(value) => value,
426        None => return value.clone(),
427    };
428    let inverse = match number::divide(&number_value::Integer(Integer::from(1)), &leading) {
429        Some(value) => value,
430        None => return value.clone(),
431    };
432    let _ = base_variable;
433    scale_polynomial(value, &inverse)
434}

436fn base_divide(
437    dividend: &Polynomial,
438    divisor: &Polynomial,
439    base_variable: SymbolIdentifier,
440) -> Option<(Polynomial, Polynomial)> {
441    if polynomial_is_zero(divisor) {
442        return None;
443    }
444    let divisor_degree = polynomial_degree(divisor)?;
445    let divisor_leading = base_leading_number(divisor)?;
446    if number::is_zero(&divisor_leading) {
447        return None;
448    }
449    let mut remainder = dividend.clone();
450    let mut quotient = zero_polynomial(base_variable);
451    while !polynomial_is_zero(&remainder) {
452        let remainder_degree = polynomial_degree(&remainder)?;
453        if remainder_degree < divisor_degree {
454            break;
455        }
456        let remainder_leading = base_leading_number(&remainder)?;
457        let factor = number::divide(&remainder_leading, &divisor_leading)?;
458        let shift = remainder_degree - divisor_degree;
459        let term = base_monomial(base_variable, factor, shift);
460        quotient = polynomial_add(&quotient, &term);
461        remainder = polynomial_subtract(&remainder, &polynomial_multiply(&term, divisor));
462    }
463    Some((quotient, remainder))
464}

466fn base_monic_gcd(
467    left: &Polynomial,
468    right: &Polynomial,
469    base_variable: SymbolIdentifier,
470) -> Polynomial {
471    let mut a = left.clone();
472    let mut b = right.clone();
473    while !polynomial_is_zero(&b) {
474        let (_, remainder) = match base_divide(&a, &b, base_variable) {
475            Some(value) => value,
476            None => break,
477        };
478        a = b;
479        b = remainder;
480    }
481    base_monic(&a, base_variable)
482}

484fn base_extended_gcd(
485    left: &Polynomial,
486    right: &Polynomial,
487    base_variable: SymbolIdentifier,
488) -> (Polynomial, Polynomial, Polynomial) {
489    let mut old_remainder = left.clone();
490    let mut remainder = right.clone();
491    let mut old_s = constant_polynomial(base_variable, number_value::Integer(Integer::from(1)));
492    let mut s = zero_polynomial(base_variable);
493    let mut old_t = zero_polynomial(base_variable);
494    let mut t = constant_polynomial(base_variable, number_value::Integer(Integer::from(1)));
495    while !polynomial_is_zero(&remainder) {
496        let (quotient, next_remainder) =
497            match base_divide(&old_remainder, &remainder, base_variable) {
498                Some(value) => value,
499                None => break,
500            };
501        old_remainder = remainder;
502        remainder = next_remainder;
503        let next_s = polynomial_subtract(&old_s, &polynomial_multiply(&quotient, &s));
504        old_s = s;
505        s = next_s;
506        let next_t = polynomial_subtract(&old_t, &polynomial_multiply(&quotient, &t));
507        old_t = t;
508        t = next_t;
509    }
510    if polynomial_is_zero(&old_remainder) {
511        return (old_remainder, old_s, old_t);
512    }
513    let leading = base_leading_number(&old_remainder).unwrap();
514    let inverse = number::divide(&number_value::Integer(Integer::from(1)), &leading).unwrap();
515    (
516        scale_polynomial(&old_remainder, &inverse),
517        scale_polynomial(&old_s, &inverse),
518        scale_polynomial(&old_t, &inverse),
519    )
520}

522fn base_diophantine(
523    b: &Polynomial,
524    a: &Polynomial,
525    c: &Polynomial,
526    base_variable: SymbolIdentifier,
527) -> Option<(Polynomial, Polynomial)> {
528    let (g, s, _) = base_extended_gcd(b, a, base_variable);
529    let (multiplier, remainder) = base_divide(c, &g, base_variable)?;
530    if !polynomial_is_zero(&remainder) {
531        return None;
532    }
533    let u_unreduced = polynomial_multiply(&s, &multiplier);
534    let (_, r) = base_divide(&u_unreduced, a, base_variable)?;
535    let numerator = polynomial_subtract(c, &polynomial_multiply(b, &r));
536    let (z, z_remainder) = base_divide(&numerator, a, base_variable)?;
537    if !polynomial_is_zero(&z_remainder) {
538        return None;
539    }
540    Some((r, z))
541}

543fn base_integrate(value: &Polynomial, base_variable: SymbolIdentifier) -> Option<Polynomial> {
544    if polynomial_is_zero(value) {
545        return Some(zero_polynomial(base_variable));
546    }
547    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::with_capacity(value.terms.len());
548    for (exponent, coefficient) in value.terms.iter() {
549        let coefficient_number = coefficient_to_number(coefficient)?;
550        let new_exponent = exponent + 1;
551        let divisor = number_value::Integer(Integer::from(new_exponent as i64));
552        let scaled = number::divide(&coefficient_number, &divisor)?;
553        terms.push((new_exponent, PolynomialCoefficient::Numeric(scaled)));
554    }
555    terms.sort_by(|left, right| right.0.cmp(&left.0));
556    Some(Polynomial {
557        main_variable: base_variable,
558        terms,
559    })
560}

569// =============================================================================
570// Transcendental level: t = exp/log monomial, k = field of the lower sub-tower.
571// Runs on tower_polynomial (k[t]).
572// =============================================================================
573
574fn solve_transcendental(
575    f: &RationalFunction,
576    g: &RationalFunction,
577    tower: &DifferentialField,
578) -> RdeOutcome {
579    let monomial = top_variable(tower);
580    let lower = lower_subfield(tower);
581    let extension = match find_extension(tower, monomial) {
582        Some(value) => value,
583        None => return RdeOutcome::Deferred,
584    };
585    let is_exponential = match &extension.kind {
586        ExtensionKind::Exponential(_) => true,
587        ExtensionKind::Logarithmic(_) => false,
588        ExtensionKind::Algebraic(_) => return RdeOutcome::Deferred,
589    };
590    // f must be a polynomial in t (weak normalization is then the identity);
591    // otherwise the denominator-bound machinery for a rational f is out of
592    // scope and we defer.
593    let f_polynomial = match tower_polynomial_from_polynomial_input(f, monomial, &lower) {
594        Some(value) => value,
595        None => return RdeOutcome::Deferred,
596    };
597    let numerator = tp::from_polynomial_in_t(&g.numerator, monomial, &lower);
598    let denominator = tp::from_polynomial_in_t(&g.denominator, monomial, &lower);
599    // Require the denominator of g to be free of the special polynomial t (for
600    // an exponential monomial that is a positive t-adic valuation): the special
601    // denominator clearing for a genuine t-pole needs the parametric machinery,
602    // which is not yet implemented.
603    if is_exponential && tp::valuation(&denominator).unwrap_or(0) > 0 {
604        return RdeOutcome::Deferred;
605    }
606    // RdeNormalDenominator, f weakly normal.
607    let denominator_derivative = match tp::derivation(tower, &denominator) {
608        Some(value) => value,
609        None => return RdeOutcome::Deferred,
610    };
611    let h = tp::gcd(&denominator, &denominator_derivative);
612    let h_squared = tp::multiply(&h, &h);
613    let (h_squared_over_denominator, remainder) =
614        match tp::divide_with_remainder(&h_squared, &denominator) {
615            Some(value) => value,
616            None => return RdeOutcome::Deferred,
617        };
618    if !tp::is_zero(&remainder) {
619        return RdeOutcome::NoSolution;
620    }
621    let a = h.clone();
622    let h_derivative = match tp::derivation(tower, &h) {
623        Some(value) => value,
624        None => return RdeOutcome::Deferred,
625    };
626    let b = tp::subtract(&tp::multiply(&f_polynomial, &h), &h_derivative);
627    let c = tp::multiply(&numerator, &h_squared_over_denominator);
628    // RdeSpecialDenominator. For a normal denominator with polynomial b and c
629    // the special part is the identity, except the exponential cancellation
630    // subcase (b has a non-zero t-constant term) which needs parametric_log_deriv.
631    if is_exponential && !tp::is_zero(&b) && tp::valuation(&b).unwrap_or(0) == 0 {
632        return RdeOutcome::Deferred;
633    }
634    let degree_bound = match bound_degree_transcendental(&a, &b, &c, is_exponential) {
635        Some(value) => value,
636        None => return RdeOutcome::Deferred,
637    };
638    let reduced = match spde_transcendental(a, b, c, degree_bound, tower, monomial, &lower) {
639        TowerSpde::Reduced {
640            b,
641            c,
642            n,
643            alpha,
644            beta,
645        } => (b, c, n, alpha, beta),
646        TowerSpde::NoSolution => return RdeOutcome::NoSolution,
647        TowerSpde::Deferred => return RdeOutcome::Deferred,
648    };
649    let (reduced_b, reduced_c, reduced_n, alpha, beta) = reduced;
650    let delta: isize = if is_exponential { 1 } else { 0 };
651    let solution_polynomial = if tp::is_zero(&reduced_c) {
652        tp::zero(monomial, lower.clone())
653    } else if reduced_n < 0 {
654        return RdeOutcome::NoSolution;
655    } else if tp::is_zero(&reduced_b) {
656        return RdeOutcome::Deferred;
657    } else if (tp::degree(&reduced_b).unwrap_or(0) as isize) > delta.max(1) - 1 {
658        match no_cancel_b_large_transcendental(
659            &reduced_b, &reduced_c, reduced_n, tower, monomial, &lower,
660        ) {
661            TowerSolve::Solved(value) => value,
662            TowerSolve::NoSolution => return RdeOutcome::NoSolution,
663            TowerSolve::Deferred => return RdeOutcome::Deferred,
664        }
665    } else {
666        // Cancellation regime (deg b small): cancel_exp / cancel_primitive need
667        // the parametric Risch machinery, which is not yet implemented.
668        return RdeOutcome::Deferred;
669    };
670    let assembled = tp::add(&tp::multiply(&alpha, &solution_polynomial), &beta);
671    let numerator_rational = tp::to_rational_function(&assembled);
672    let denominator_rational = tp::to_rational_function(&h);
673    match crate::differential_field::arithmetic::divide_rational_functions(
674        &numerator_rational,
675        &denominator_rational,
676    ) {
677        Some(value) => RdeOutcome::Solved(value),
678        None => RdeOutcome::Deferred,
679    }
680}

682fn bound_degree_transcendental(
683    a: &tp::TowerPolynomial,
684    b: &tp::TowerPolynomial,
685    c: &tp::TowerPolynomial,
686    is_exponential: bool,
687) -> Option<isize> {
688    // RdeBoundDegree for the primitive / hyperexponential monomials. Returns
689    // None (defer) on the cancellation subcase, whose refinement needs the
690    // deferred parametric machinery.
691    if tp::is_zero(b) {
692        return None;
693    }
694    let degree_a = tp::degree(a).map(|value| value as isize).unwrap_or(-1);
695    let degree_b = tp::degree(b).map(|value| value as isize).unwrap_or(-1);
696    let degree_c = tp::degree(c).map(|value| value as isize).unwrap_or(-1);
697    if is_exponential {
698        if degree_a == degree_b {
699            return None;
700        }
701        Some((degree_c - degree_b.max(degree_a)).max(0))
702    } else {
703        if degree_b == degree_a || degree_b == degree_a - 1 {
704            return None;
705        }
706        if degree_b > degree_a {
707            Some((degree_c - degree_b).max(0))
708        } else {
709            Some((degree_c - degree_a + 1).max(0))
710        }
711    }
712}

714enum TowerSpde {
715    Reduced {
716        b: tp::TowerPolynomial,
717        c: tp::TowerPolynomial,
718        n: isize,
719        alpha: tp::TowerPolynomial,
720        beta: tp::TowerPolynomial,
721    },
722    NoSolution,
723    Deferred,
724}
725
726fn spde_transcendental(
727    a: tp::TowerPolynomial,
728    b: tp::TowerPolynomial,
729    c: tp::TowerPolynomial,
730    n: isize,
731    tower: &DifferentialField,
732    monomial: SymbolIdentifier,
733    lower: &DifferentialField,
734) -> TowerSpde {
735    let mut a = a;
736    let mut b = b;
737    let mut c = c;
738    let mut n = n;
739    let mut alpha = tp::constant(monomial, lower.clone(), tower_field_one(lower));
740    let mut beta = tp::zero(monomial, lower.clone());
741    loop {
742        if tp::is_zero(&c) {
743            return TowerSpde::Reduced {
744                b: tp::zero(monomial, lower.clone()),
745                c: tp::zero(monomial, lower.clone()),
746                n,
747                alpha,
748                beta,
749            };
750        }
751        if n < 0 {
752            return TowerSpde::NoSolution;
753        }
754        let g = tp::gcd(&a, &b);
755        let (a_reduced, a_remainder) = match tp::divide_with_remainder(&a, &g) {
756            Some(value) => value,
757            None => return TowerSpde::Deferred,
758        };
759        let (b_reduced, b_remainder) = match tp::divide_with_remainder(&b, &g) {
760            Some(value) => value,
761            None => return TowerSpde::Deferred,
762        };
763        let (c_reduced, c_remainder) = match tp::divide_with_remainder(&c, &g) {
764            Some(value) => value,
765            None => return TowerSpde::Deferred,
766        };
767        if !tp::is_zero(&a_remainder) || !tp::is_zero(&b_remainder) || !tp::is_zero(&c_remainder) {
768            return TowerSpde::NoSolution;
769        }
770        a = a_reduced;
771        b = b_reduced;
772        c = c_reduced;
773        if tp::degree(&a) == Some(0) {
774            let reduced_b = match tower_divide_by_unit(&b, &a) {
775                Some(value) => value,
776                None => return TowerSpde::Deferred,
777            };
778            let reduced_c = match tower_divide_by_unit(&c, &a) {
779                Some(value) => value,
780                None => return TowerSpde::Deferred,
781            };
782            return TowerSpde::Reduced {
783                b: reduced_b,
784                c: reduced_c,
785                n,
786                alpha,
787                beta,
788            };
789        }
790        let degree_a = tp::degree(&a).unwrap();
791        let (r, z) = match tp::gcdex_diophantine(&b, &a, &c) {
792            Some(value) => value,
793            None => return TowerSpde::Deferred,
794        };
795        let a_derivative = match tp::derivation(tower, &a) {
796            Some(value) => value,
797            None => return TowerSpde::Deferred,
798        };
799        let r_derivative = match tp::derivation(tower, &r) {
800            Some(value) => value,
801            None => return TowerSpde::Deferred,
802        };
803        let new_b = tp::add(&b, &a_derivative);
804        let new_c = tp::subtract(&z, &r_derivative);
805        n -= degree_a as isize;
806        beta = tp::add(&beta, &tp::multiply(&alpha, &r));
807        alpha = tp::multiply(&alpha, &a);
808        b = new_b;
809        c = new_c;
810    }
811}
812
813enum TowerSolve {
814    Solved(tp::TowerPolynomial),
815    NoSolution,
816    Deferred,
817}
818
819fn no_cancel_b_large_transcendental(
820    b: &tp::TowerPolynomial,
821    c: &tp::TowerPolynomial,
822    n: isize,
823    tower: &DifferentialField,
824    monomial: SymbolIdentifier,
825    lower: &DifferentialField,
826) -> TowerSolve {
827    let mut q = tp::zero(monomial, lower.clone());
828    let mut c = c.clone();
829    let mut n = n;
830    let degree_b = match tp::degree(b) {
831        Some(value) => value as isize,
832        None => return TowerSolve::NoSolution,
833    };
834    while !tp::is_zero(&c) {
835        let degree_c = tp::degree(&c).unwrap() as isize;
836        let m = degree_c - degree_b;
837        if m < 0 || m > n {
838            return TowerSolve::NoSolution;
839        }
840        let leading_c = tp::leading_coefficient(&c).unwrap();
841        let leading_b = tp::leading_coefficient(b).unwrap();
842        let ratio = match crate::differential_field::arithmetic::divide_rational_functions(
843            &leading_c, &leading_b,
844        ) {
845            Some(value) => value,
846            None => return TowerSolve::Deferred,
847        };
848        let p = tp::monomial_term(monomial, lower.clone(), m as usize, ratio);
849        q = tp::add(&q, &p);
850        n = m - 1;
851        let p_derivative = match tp::derivation(tower, &p) {
852            Some(value) => value,
853            None => return TowerSolve::Deferred,
854        };
855        let subtraction = tp::add(&p_derivative, &tp::multiply(b, &p));
856        c = tp::subtract(&c, &subtraction);
857    }
858    TowerSolve::Solved(q)
859}

861fn tower_field_one(lower: &DifferentialField) -> RationalFunction {
862    crate::differential_field::arithmetic::one(top_variable(lower))
863}
864
865fn tower_divide_by_unit(
866    value: &tp::TowerPolynomial,
867    unit: &tp::TowerPolynomial,
868) -> Option<tp::TowerPolynomial> {
869    // Divide by a degree-zero (unit) tower_polynomial by scaling with the
870    // inverse of its single k-coefficient.
871    let coefficient = tp::coefficient_at(unit, 0);
872    if rational_function_is_zero(&coefficient) {
873        return None;
874    }
875    let inverse = crate::differential_field::arithmetic::divide_rational_functions(
876        &tower_field_one(&value.lower),
877        &coefficient,
878    )?;
879    Some(tp::scale(value, &inverse))
880}
881
882fn tower_polynomial_from_polynomial_input(
883    value: &RationalFunction,
884    monomial: SymbolIdentifier,
885    lower: &DifferentialField,
886) -> Option<tp::TowerPolynomial> {
887    // Convert a k(t) element that is actually a polynomial in t (denominator of
888    // degree zero in t) into a tower_polynomial; return None when a genuine
889    // t-denominator is present (rational f is out of scope).
890    if polynomial_degree(&value.denominator).unwrap_or(0) > 0
891        && value.denominator.main_variable == monomial
892    {
893        return None;
894    }
895    let numerator = tp::from_polynomial_in_t(&value.numerator, monomial, lower);
896    let denominator = tp::from_polynomial_in_t(&value.denominator, monomial, lower);
897    if tp::degree(&denominator).unwrap_or(0) > 0 {
898        return None;
899    }
900    let denominator_coefficient = tp::coefficient_at(&denominator, 0);
901    if rational_function_is_zero(&denominator_coefficient) {
902        return None;
903    }
904    let inverse = crate::differential_field::arithmetic::divide_rational_functions(
905        &tower_field_one(lower),
906        &denominator_coefficient,
907    )?;
908    Some(tp::scale(&numerator, &inverse))
909}
