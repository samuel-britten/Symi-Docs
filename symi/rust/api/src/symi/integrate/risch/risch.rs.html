71/// Performs the advanced native `integrate_using_risch` operation.
72pub fn integrate_using_risch(
73    context: &Context,
74    target: &ExpressionReference,
75    variable: SymbolIdentifier,
76) -> Option<IntegrationResult> {
77    // Top-level dispatcher entry. Builds an elementary differential-field tower
78    // covering the integrand, runs `integrate_in_tower`, and reifies the result
79    // back into an expression by substituting the tower generators with their
80    // original `exp` / `log` subexpressions. Returns None when the integrand
81    // contains a free symbol outside the integration variable, when an `exp`
82    // or `log` argument cannot be expressed in the partially-built tower, or
83    // when the rewritten integrand is not a polynomial in the tower
84    // variables — the caller then falls back to the unevaluated `integral`
85    // placeholder rather than risking an unsound verdict. Returns
86    // Some(non_elementary) when the Risch decision procedure produces a
87    // Liouville witness; the caller wraps that into a
88    // `non_elementary_integral` user-function placeholder.
89    let setup = build_tower_from_integrand(context, target, variable)?;
90    match integrate_in_tower(context, &setup.rewritten, &setup.tower) {
91        IntegrationResult::Elementary(value) => {
92            let reified = reify_with_generators(context, &value, &setup);
93            Some(IntegrationResult::Elementary(reified))
94        }
95        IntegrationResult::NonElementary => Some(IntegrationResult::NonElementary),
96        IntegrationResult::Incomplete => None,
97    }
98}

100/// Performs the advanced native `integrate_in_tower` operation.
101pub fn integrate_in_tower(
102    context: &Context,
103    target: &ExpressionReference,
104    tower: &DifferentialField,
105) -> IntegrationResult {
106    // Decision-procedure entry. Convert the integrand into a
107    // polynomial in the topmost extension over the lower field and dispatch on
108    // the extension kind. Inputs that aren't polynomial in θ_n (proper
109    // rationals in θ) now route into the Hermite + LRT-in-θ pipeline rather
110    // than returning `incomplete`. Anything outside both subsets falls through
111    // to `incomplete` so the caller emits the unevaluated `integral`
112    // placeholder.
113    let variables = variable_order(tower);
114    if let Some(target_polynomial) = expression_to_polynomial(target, &variables) {
115        return integrate_polynomial_in_tower(context, &target_polynomial, tower);
116    }
117    if !tower.extensions.is_empty() {
118        if let Some((numerator, denominator)) = split_into_rational_in_tower(target, &variables) {
119            // A denominator that is free of the top monomial makes the
120            // integrand a polynomial in the top extension with rational (in the
121            // lower field) coefficients — e.g. exp(x)·(x−1)/x². Route it to the
122            // rational-coefficient exponential case so the Bronstein Chapter 6
123            // RDE decides it, rather than the rational-in-θ pipeline (which
124            // only handles a denominator that is a polynomial in θ).
125            let top = top_variable(tower);
126            let top_is_exponential = matches!(
127                find_extension(tower, top).map(|extension| &extension.kind),
128                Some(ExtensionKind::Exponential(_))
129            );
130            let denominator_top_degree = if denominator.main_variable == top {
131                degree(&denominator).unwrap_or(0)
132            } else {
133                0
134            };
135            if top_is_exponential
136                && denominator_top_degree == 0
137                && !polynomial_is_zero(&denominator)
138                && constant_polynomial_value(&denominator).is_none()
139            {
140                return integrate_theta_polynomial_over_lower(
141                    context,
142                    &numerator,
143                    &denominator,
144                    tower,
145                );
146            }
147            return integrate_rational_in_top_extension(context, &numerator, &denominator, tower);
148        }
149    }
150    IntegrationResult::Incomplete
151}

153fn integrate_theta_polynomial_over_lower(
154    context: &Context,
155    numerator: &Polynomial,
156    denominator: &Polynomial,
157    tower: &DifferentialField,
158) -> IntegrationResult {
159    // Integrate N(θ)/D where D is a non-constant polynomial in the lower
160    // field K_{n-1} (free of the top monomial θ). The integrand is a polynomial
161    // in θ whose coefficients are proper rational functions of the lower
162    // variables; each θ^j coefficient c_j = n_j / D is handed to the completed
163    // Risch differential equation solver. Only the exponential top extension is
164    // in scope here; the logarithmic rational-coefficient case needs the
165    // deferred primitive-cancellation solver, and the algebraic case stays on
166    // its own path.
167    let top = top_variable(tower);
168    let extension = match find_extension(tower, top) {
169        Some(value) => value,
170        None => return IntegrationResult::Incomplete,
171    };
172    match &extension.kind {
173        ExtensionKind::Exponential(argument) => {
174            let argument = argument.clone();
175            integrate_exponential_rational_coefficients(
176                context,
177                numerator,
178                denominator,
179                tower,
180                &argument,
181            )
182        }
183        _ => IntegrationResult::Incomplete,
184    }
185}

187fn integrate_exponential_rational_coefficients(
188    context: &Context,
189    numerator: &Polynomial,
190    denominator: &Polynomial,
191    tower: &DifferentialField,
192    exponential_argument: &RationalFunction,
193) -> IntegrationResult {
194    // For θ = exp(u), integrate Σ_j c_j θ^j with c_j = n_j / D ∈ K_{n-1}. Each
195    // power j ≥ 1 solves the Risch differential equation q_j' + (j u') q_j = c_j
196    // via solve_risch_de; the j = 0 term integrates c_j in the lower field. A
197    // `no_solution` from any power is the rigorous non-elementary verdict; a
198    // `deferred` falls back to the unevaluated placeholder.
199    let top = top_variable(tower);
200    let lower_tower = lower_subfield(tower);
201    let lower_variables = variable_order(&lower_tower);
202    let argument_derivative = derive_rational_function(tower, exponential_argument);
203    let argument_derivative_in_lower =
204        match project_rational_function_to_lower(&argument_derivative, &lower_variables) {
205            Some(value) => value,
206            None => return IntegrationResult::Incomplete,
207        };
208    let denominator_lower = match project_polynomial_to_lower(denominator, top, &lower_variables) {
209        Some(value) => value,
210        None => return IntegrationResult::Incomplete,
211    };
212    let mut coefficient_terms: Vec<(usize, Polynomial)> = Vec::new();
213    if numerator.main_variable == top {
214        for (exponent, coefficient) in numerator.terms.iter() {
215            let lower_coefficient =
216                match polynomial_coefficient_to_lower(coefficient, &lower_variables) {
217                    Some(value) => value,
218                    None => return IntegrationResult::Incomplete,
219                };
220            coefficient_terms.push((*exponent, lower_coefficient));
221        }
222    } else if !polynomial_is_zero(numerator) {
223        let lower_coefficient = match project_polynomial_to_lower(numerator, top, &lower_variables)
224        {
225            Some(value) => value,
226            None => return IntegrationResult::Incomplete,
227        };
228        coefficient_terms.push((0, lower_coefficient));
229    }
230    let mut result_terms: Vec<ExpressionReference> = Vec::new();
231    for (exponent, lower_numerator) in coefficient_terms {
232        let coefficient_rational =
233            match make_rational_function(lower_numerator, denominator_lower.clone()) {
234                Some(value) => value,
235                None => return IntegrationResult::Incomplete,
236            };
237        if exponent == 0 {
238            let coefficient_expression =
239                rational_function_to_expression(context, &coefficient_rational);
240            match integrate_lower_expression(context, &coefficient_expression, &lower_tower) {
241                IntegrationResult::Elementary(value) => result_terms.push(value),
242                IntegrationResult::NonElementary => return IntegrationResult::NonElementary,
243                IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
244            }
245            continue;
246        }
247        let exponent_factor = rational_function_from_constant(
248            lower_variables[0],
249            number_value::Integer(Integer::from(exponent as i64)),
250        );
251        let a_coefficient =
252            multiply_rational_functions(&exponent_factor, &argument_derivative_in_lower);
253        match solve_risch_de(&a_coefficient, &coefficient_rational, &lower_tower) {
254            RdeResult::Solved(q_value) => {
255                let q_expression = rational_function_to_expression(context, &q_value);
256                let generator_symbol = context.interner.intern(ExpressionKind::Symbol(top));
257                let exponent_expression =
258                    builders::integer(context, Integer::from(exponent as i64));
259                let power_expression =
260                    builders::power(context, generator_symbol, exponent_expression);
261                let term = builders::product(context, vec![q_expression, power_expression]);
262                result_terms.push(term);
263            }
264            RdeResult::NoSolution => return IntegrationResult::NonElementary,
265            RdeResult::Deferred => return IntegrationResult::Incomplete,
266        }
267    }
268    if result_terms.is_empty() {
269        return IntegrationResult::Elementary(builders::zero(context));
270    }
271    let combined = if result_terms.len() == 1 {
272        result_terms.into_iter().next().unwrap()
273    } else {
274        builders::sum(context, result_terms)
275    };
276    IntegrationResult::Elementary(combined)
277}

279fn project_polynomial_to_lower(
280    value: &Polynomial,
281    top: SymbolIdentifier,
282    lower_variables: &[SymbolIdentifier],
283) -> Option<Polynomial> {
284    // Re-express a polynomial that is free of the top monomial as a polynomial
285    // in the lower-tower variables. When the value's main variable is already a
286    // lower variable it is returned unchanged; when it is the top monomial
287    // carrying a single degree-zero term the nested coefficient is unwrapped.
288    if value.main_variable != top {
289        return Some(value.clone());
290    }
291    if polynomial_is_zero(value) {
292        return Some(zero_polynomial(lower_variables[0]));
293    }
294    if degree(value).unwrap_or(0) > 0 {
295        return None;
296    }
297    polynomial_coefficient_to_lower(&value.terms[0].1, lower_variables)
298}

300fn rational_function_to_expression(
301    context: &Context,
302    value: &RationalFunction,
303) -> ExpressionReference {
304    // Reify a rational function over the lower tower back into an expression,
305    // dividing the numerator by the denominator when the denominator is not the
306    // constant one.
307    let numerator = polynomial_to_expression(context, &value.numerator);
308    if polynomial_is_constant_one(&value.denominator) {
309        return numerator;
310    }
311    let denominator = polynomial_to_expression(context, &value.denominator);
312    builders::divide(context, numerator, denominator)
313}

315fn integrate_lower_expression(
316    context: &Context,
317    expression_form: &ExpressionReference,
318    lower_tower: &DifferentialField,
319) -> IntegrationResult {
320    // Integrate an expression in the lower tower, routing first through the
321    // shared attempt_integrate choke point (so the builtin-table / rational-
322    // function surfaces fire) and falling back to the lower-tower Risch
323    // dispatcher. A residual
324    // integral placeholder is reported as incomplete.
325    if let Some(value) = super::outcome_expression(super::attempt_integrate(
326        context,
327        expression_form,
328        lower_tower.base_variable,
329    )) {
330        if super::contains_integral_placeholder(&value) {
331            return IntegrationResult::Incomplete;
332        }
333        return IntegrationResult::Elementary(value);
334    }
335    if lower_tower.extensions.is_empty() {
336        return IntegrationResult::Incomplete;
337    }
338    integrate_in_tower(context, expression_form, lower_tower)
339}

341fn integrate_polynomial_in_tower(
342    context: &Context,
343    target_polynomial: &Polynomial,
344    tower: &DifferentialField,
345) -> IntegrationResult {
346    // Polynomial-in-θ dispatch shared by the direct conversion path and by
347    // the residual integration after the LRT step. Routes to the
348    // base-field power-rule when the tower has no extensions, projects to the
349    // lower field when the integrand is constant in the topmost extension,
350    // and dispatches on the extension kind otherwise.
351    if tower.extensions.is_empty() {
352        return integrate_in_base_field(context, target_polynomial, tower);
353    }
354    let top = top_variable(tower);
355    if target_polynomial.main_variable != top && !polynomial_is_zero(target_polynomial) {
356        let lower_tower = lower_subfield(tower);
357        let lower_expression = polynomial_to_expression(context, target_polynomial);
358        return integrate_in_tower(context, &lower_expression, &lower_tower);
359    }
360    let extension = find_extension(tower, top).expect("top variable is the topmost extension");
361    match &extension.kind {
362        ExtensionKind::Exponential(argument) => {
363            integrate_polynomial_exponential_case(context, target_polynomial, tower, argument)
364        }
365        ExtensionKind::Logarithmic(argument) => {
366            integrate_polynomial_logarithmic_case(context, target_polynomial, tower, argument)
367        }
368        ExtensionKind::Algebraic(_) => {
369            // Route to the algebraic-tower integrator. The
370            // implementation ships the square-root subset of Trager 1976
371            // (Bronstein 2005 §5.5–5.6): degree-2 minimal polynomial over
372            // the base field, polynomial-in-alpha integrand. Everything
373            // outside that subset returns incomplete, matching the
374            // behaviour before algebraic-tower support was added.
375            crate::integrate::algebraic_tower::integrate_in_algebraic_tower(
376                context,
377                target_polynomial,
378                &constant_polynomial(
379                    target_polynomial.main_variable,
380                    number_value::Integer(Integer::ONE),
381                ),
382                tower,
383            )
384        }
385    }
386}

388fn integrate_in_base_field(
389    context: &Context,
390    target: &Polynomial,
391    tower: &DifferentialField,
392) -> IntegrationResult {
393    // Base case: tower has no extensions, so the integrand must be a polynomial
394    // in the integration variable. Apply the power rule term-by-term over Q.
395    // Rational functions in the base variable are not in this code path —
396    // they route through the rational-function integrator (Hermite + LRT)
397    // before the Risch dispatcher fires.
398    if polynomial_is_zero(target) {
399        return IntegrationResult::Elementary(builders::zero(context));
400    }
401    let antiderivative = match integrate_numeric_polynomial(target, tower.base_variable) {
402        Some(value) => value,
403        None => return IntegrationResult::Incomplete,
404    };
405    IntegrationResult::Elementary(polynomial_to_expression(context, &antiderivative))
406}

408fn integrate_polynomial_exponential_case(
409    context: &Context,
410    target: &Polynomial,
411    tower: &DifferentialField,
412    exponential_argument: &RationalFunction,
413) -> IntegrationResult {
414    // For an exponential extension θ = exp(u), the integrand is a
415    // polynomial in θ over the lower field K_{n-1}. For each nonzero power j,
416    // solve the Risch differential equation q' + j*u'*q = p_j (Bronstein
417    // §5.9) via solve_risch_de. The constant term in θ (j = 0) recurses into
418    // the lower tower. A failure of any RDE solve produces the rigorous
419    // non-elementary verdict — Liouville's theorem guarantees that no
420    // alternative elementary antiderivative exists.
421    let top = top_variable(tower);
422    debug_assert_eq!(target.main_variable, top);
423    let lower_tower = lower_subfield(tower);
424    let lower_variables = variable_order(&lower_tower);
425    let argument_derivative = derive_rational_function(tower, exponential_argument);
426    let argument_derivative_in_lower =
427        match project_rational_function_to_lower(&argument_derivative, &lower_variables) {
428            Some(value) => value,
429            None => return IntegrationResult::Incomplete,
430        };
431    let mut antiderivative_terms: Vec<(usize, Polynomial)> = Vec::new();
432    for (exponent, coefficient) in target.terms.iter() {
433        let lower_coefficient = match polynomial_coefficient_to_lower(coefficient, &lower_variables)
434        {
435            Some(value) => value,
436            None => return IntegrationResult::Incomplete,
437        };
438        if *exponent == 0 {
439            match integrate_lower(context, &lower_coefficient, &lower_tower) {
440                IntegrationResult::Elementary(value) => {
441                    let lifted = match expression_to_polynomial(&value, &lower_variables) {
442                        Some(value) => value,
443                        None => return IntegrationResult::Incomplete,
444                    };
445                    accumulate_term(&mut antiderivative_terms, 0, lifted);
446                }
447                IntegrationResult::NonElementary => {
448                    return IntegrationResult::NonElementary;
449                }
450                IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
451            }
452            continue;
453        }
454        let exponent_factor = rational_function_from_constant(
455            lower_variables[0],
456            number_value::Integer(Integer::from(*exponent as i64)),
457        );
458        let a_coefficient =
459            multiply_rational_functions(&exponent_factor, &argument_derivative_in_lower);
460        let c_rational_function = from_polynomial(lower_coefficient);
461        match solve_risch_de(&a_coefficient, &c_rational_function, &lower_tower) {
462            RdeResult::Solved(q_value) => {
463                let q_polynomial =
464                    match rational_function_numerator_when_constant_denominator(&q_value) {
465                        Some(value) => value,
466                        None => return IntegrationResult::Incomplete,
467                    };
468                accumulate_term(&mut antiderivative_terms, *exponent, q_polynomial);
469            }
470            RdeResult::NoSolution => return IntegrationResult::NonElementary,
471            RdeResult::Deferred => return IntegrationResult::Incomplete,
472        }
473    }
474    let final_polynomial = polynomial_from_top_terms(antiderivative_terms, top);
475    IntegrationResult::Elementary(polynomial_to_expression(context, &final_polynomial))
476}

478fn integrate_polynomial_logarithmic_case(
479    context: &Context,
480    target: &Polynomial,
481    tower: &DifferentialField,
482    logarithmic_argument: &RationalFunction,
483) -> IntegrationResult {
484    // For a logarithmic extension θ = log(u), the polynomial part integrates
485    // by parts since D(θ) = u'/u lies in the lower field. For each c_j(x) *
486    // θ^j with j ≥ 1 we compute Q_j with Q_j' = c_j in K_{n-1}, then apply
487    //   ∫ c_j * θ^j dx = Q_j * θ^j - j * ∫ Q_j * D(θ) * θ^(j-1) dx.
488    // The residual integrand is polynomial in θ of degree max(j) - 1, so the
489    // recursion terminates. The j = 0 constant term recurses into the lower
490    // tower directly. Constant-residue obstructions (Liouville witnesses for
491    // proper-rational integrands in θ) are not yet handled and remain a
492    // future follow-up.
493    let top = top_variable(tower);
494    debug_assert_eq!(target.main_variable, top);
495    let lower_tower = lower_subfield(tower);
496    let lower_variables = variable_order(&lower_tower);
497    let argument_derivative = derive_rational_function(tower, logarithmic_argument);
498    let logarithmic_derivative =
499        match divide_rational_functions(&argument_derivative, logarithmic_argument) {
500            Some(value) => value,
501            None => return IntegrationResult::Incomplete,
502        };
503    let logarithmic_derivative_lower =
504        match project_rational_function_to_lower(&logarithmic_derivative, &lower_variables) {
505            Some(value) => value,
506            None => return IntegrationResult::Incomplete,
507        };
508    let mut primary_terms: Vec<(usize, Polynomial)> = Vec::new();
509    let mut residual_terms: Vec<(usize, Polynomial)> = Vec::new();
510    for (exponent, coefficient) in target.terms.iter() {
511        let lower_coefficient = match polynomial_coefficient_to_lower(coefficient, &lower_variables)
512        {
513            Some(value) => value,
514            None => return IntegrationResult::Incomplete,
515        };
516        if *exponent == 0 {
517            match integrate_lower(context, &lower_coefficient, &lower_tower) {
518                IntegrationResult::Elementary(value) => {
519                    let lifted = match expression_to_polynomial(&value, &lower_variables) {
520                        Some(value) => value,
521                        None => return IntegrationResult::Incomplete,
522                    };
523                    accumulate_term(&mut primary_terms, 0, lifted);
524                }
525                IntegrationResult::NonElementary => {
526                    return IntegrationResult::NonElementary;
527                }
528                IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
529            }
530            continue;
531        }
532        let antiderivative_of_coefficient =
533            match integrate_lower(context, &lower_coefficient, &lower_tower) {
534                IntegrationResult::Elementary(value) => value,
535                IntegrationResult::NonElementary => return IntegrationResult::NonElementary,
536                IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
537            };
538        let antiderivative_polynomial =
539            match expression_to_polynomial(&antiderivative_of_coefficient, &lower_variables) {
540                Some(value) => value,
541                None => return IntegrationResult::Incomplete,
542            };
543        accumulate_term(
544            &mut primary_terms,
545            *exponent,
546            antiderivative_polynomial.clone(),
547        );
548        let antiderivative_rational = from_polynomial(antiderivative_polynomial);
549        let exponent_factor = rational_function_from_constant(
550            lower_variables[0],
551            number_value::Integer(Integer::from(*exponent as i64)),
552        );
553        let scaled = multiply_rational_functions(&exponent_factor, &antiderivative_rational);
554        let residual_contribution =
555            multiply_rational_functions(&scaled, &logarithmic_derivative_lower);
556        let residual_polynomial =
557            match rational_function_numerator_when_constant_denominator(&residual_contribution) {
558                Some(value) => value,
559                None => return IntegrationResult::Incomplete,
560            };
561        accumulate_term(&mut residual_terms, *exponent - 1, residual_polynomial);
562    }
563    let primary_polynomial = polynomial_from_top_terms(primary_terms, top);
564    let primary_expression = polynomial_to_expression(context, &primary_polynomial);
565    if residual_terms.is_empty() {
566        return IntegrationResult::Elementary(primary_expression);
567    }
568    let residual_polynomial = polynomial_from_top_terms(residual_terms, top);
569    let residual_expression = polynomial_to_expression(context, &residual_polynomial);
570    let residual_value = match integrate_in_tower(context, &residual_expression, tower) {
571        IntegrationResult::Elementary(value) => value,
572        IntegrationResult::NonElementary => return IntegrationResult::NonElementary,
573        IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
574    };
575    let combined = builders::subtract(context, primary_expression, residual_value);
576    IntegrationResult::Elementary(combined)
577}

589/// Performs the advanced native `solve_risch_de` operation.
590pub fn solve_risch_de(
591    a: &RationalFunction,
592    c: &RationalFunction,
593    tower: &DifferentialField,
594) -> RdeResult {
595    // Solve y' + a*y = c in the supplied tower for y a rational function
596    // via Bronstein's complete Chapter 6 decision procedure
597    // (risch_differential_equation::solve). Two fast paths short-circuit first:
598    //   (i) c = 0 → y = 0.
599    //   (ii) constant-solution trial: y = c/a when a ≠ 0 and y is a
600    //        derivation-constant in the tower (D(y) = 0 so a*y = c holds).
601    // Everything else — including rational-solution RDEs (denominator bounds),
602    // non-base towers, and the non-elementary verdicts — is decided by the
603    // module, which returns `solved` only after verifying D(y) + a*y = c
604    // exactly, `no_solution` only after a bounded decision, and `deferred` for
605    // the pinned out-of-scope inputs (algebraic generators, rational f, and the
606    // parametric cancellation cases). The base-field `solve_polynomial_rde`
607    // below is retained as a differential regression oracle exercised by the
608    // unit tests.
609    if rational_function_is_zero(c) {
610        return RdeResult::Solved(rational_function_zero(tower.base_variable));
611    }
612    if let Some(constant_solution) = try_constant_solution(a, c, tower) {
613        return RdeResult::Solved(constant_solution);
614    }
615    match risch_differential_equation::solve(a, c, tower) {
616        risch_differential_equation::RdeOutcome::Solved(value) => RdeResult::Solved(value),
617        risch_differential_equation::RdeOutcome::NoSolution => RdeResult::NoSolution,
618        risch_differential_equation::RdeOutcome::Deferred => RdeResult::Deferred,
619    }
620}

628fn solve_polynomial_rde(
629    a: &Polynomial,
630    c: &Polynomial,
631    base_variable: SymbolIdentifier,
632) -> PolynomialRdeOutcome {
633    // Solve q' + a*q = c for q polynomial in base_variable. Both a and c carry
634    // numeric rational coefficients. The decision rests on the comparison of
635    // degree(c) with degree(a):
636    //   * a = 0: q = integrate(c) via the power rule; always succeeds.
637    //   * c = 0, a ≠ 0: q = 0 satisfies the equation.
638    //   * a polynomial of degree m ≥ 1 with deg(c) < m: no polynomial
639    //     solution; report no_solution so the caller reports
640    //     non_elementary.
641    //   * Otherwise: deg(q) = deg(c) - m (or = deg(c) when m = 0). Compute
642    //     each coefficient of q top-down via the recurrence
643    //       q_{k-m} = (c_k - Σ_{i<m} a_i*q_{k-i} - (k+1)*q_{k+1}) / a_m,
644    //     then verify the lower-degree constraints (only fires when m ≥ 1).
645    //     A failed verification reports no_solution.
646    if polynomial_is_zero(a) {
647        return match integrate_numeric_polynomial(c, base_variable) {
648            Some(value) => PolynomialRdeOutcome::Solved(value),
649            None => PolynomialRdeOutcome::Deferred,
650        };
651    }
652    if polynomial_is_zero(c) {
653        return PolynomialRdeOutcome::Solved(zero_polynomial(base_variable));
654    }
655    let degree_a = degree(a).unwrap();
656    let degree_c = degree(c).unwrap();
657    if degree_a >= 1 && degree_c < degree_a {
658        return PolynomialRdeOutcome::NoSolution;
659    }
660    let degree_q: usize = if degree_a == 0 {
661        degree_c
662    } else {
663        degree_c - degree_a
664    };
665    let dense_a = match dense_numeric_coefficients(a, degree_a) {
666        Some(value) => value,
667        None => return PolynomialRdeOutcome::Deferred,
668    };
669    let dense_c = match dense_numeric_coefficients(c, degree_c) {
670        Some(value) => value,
671        None => return PolynomialRdeOutcome::Deferred,
672    };
673    let a_leading = dense_a[degree_a].clone();
674    if number::is_zero(&a_leading) {
675        return PolynomialRdeOutcome::Deferred;
676    }
677    let mut dense_q: Vec<number_value> = vec![number_value::Integer(Integer::ZERO); degree_q + 1];
678    for k in (degree_a..=degree_c).rev() {
679        let mut accumulator = dense_c
680            .get(k)
681            .cloned()
682            .unwrap_or(number_value::Integer(Integer::ZERO));
683        for i in 0..degree_a {
684            let q_index = match k.checked_sub(i) {
685                Some(value) if value <= degree_q => value,
686                _ => continue,
687            };
688            let a_value = dense_a
689                .get(i)
690                .cloned()
691                .unwrap_or(number_value::Integer(Integer::ZERO));
692            let product = number::multiply(&a_value, &dense_q[q_index]);
693            accumulator = number::subtract(&accumulator, &product);
694        }
695        if k + 1 <= degree_q {
696            let factor = number_value::Integer(Integer::from((k + 1) as i64));
697            let product = number::multiply(&factor, &dense_q[k + 1]);
698            accumulator = number::subtract(&accumulator, &product);
699        }
700        let q_index = k - degree_a;
701        let coefficient_value = match number::divide(&accumulator, &a_leading) {
702            Some(value) => value,
703            None => return PolynomialRdeOutcome::Deferred,
704        };
705        dense_q[q_index] = coefficient_value;
706    }
707    for k in 0..degree_a {
708        let mut left_hand_side = number_value::Integer(Integer::ZERO);
709        for i in 0..=degree_a {
710            let q_index = match k.checked_sub(i) {
711                Some(value) if value <= degree_q => value,
712                _ => continue,
713            };
714            let a_value = dense_a
715                .get(i)
716                .cloned()
717                .unwrap_or(number_value::Integer(Integer::ZERO));
718            let product = number::multiply(&a_value, &dense_q[q_index]);
719            left_hand_side = number::add(&left_hand_side, &product);
720        }
721        if k + 1 <= degree_q {
722            let factor = number_value::Integer(Integer::from((k + 1) as i64));
723            let product = number::multiply(&factor, &dense_q[k + 1]);
724            left_hand_side = number::add(&left_hand_side, &product);
725        }
726        let right_hand_side = dense_c
727            .get(k)
728            .cloned()
729            .unwrap_or(number_value::Integer(Integer::ZERO));
730        let difference = number::subtract(&left_hand_side, &right_hand_side);
731        if !number::is_zero(&difference) {
732            return PolynomialRdeOutcome::NoSolution;
733        }
734    }
735    PolynomialRdeOutcome::Solved(univariate_from_dense(base_variable, dense_q))
736}

738fn dense_numeric_coefficients(
739    value: &Polynomial,
740    expected_degree: usize,
741) -> Option<Vec<number_value>> {
742    // Return a dense Vec of numeric coefficients indexed by degree, with
743    // length expected_degree + 1. Returns None when any coefficient is
744    // non-numeric (a genuinely multivariate slot) — that case is the
745    // RDE solver's signal to fall back to deferred.
746    let mut dense: Vec<number_value> =
747        vec![number_value::Integer(Integer::ZERO); expected_degree + 1];
748    for (exponent, coefficient) in value.terms.iter() {
749        let coefficient_number = coefficient_to_number(coefficient)?;
750        if *exponent <= expected_degree {
751            dense[*exponent] = coefficient_number;
752        } else {
753            return None;
754        }
755    }
756    Some(dense)
757}

759fn try_constant_solution(
760    a: &RationalFunction,
761    c: &RationalFunction,
762    tower: &DifferentialField,
763) -> Option<RationalFunction> {
764    // Try y = c/a as a candidate constant solution. Verifies the candidate is
765    // a derivation-constant; if so, the equation reduces to a*y = c which
766    // holds by construction. The check uses the tower's derivation so
767    // multi-tower constants (e.g., y = 1 when a = t and c = t in the
768    // exp(exp(x)) test case) are recognised correctly.
769    if rational_function_is_zero(a) {
770        return None;
771    }
772    let candidate = divide_rational_functions(c, a)?;
773    let candidate_derivative = derive_rational_function(tower, &candidate);
774    if rational_function_is_zero(&candidate_derivative) {
775        Some(candidate)
776    } else {
777        None
778    }
779}

781fn integrate_numeric_polynomial(
782    target: &Polynomial,
783    base_variable: SymbolIdentifier,
784) -> Option<Polynomial> {
785    // Power-rule antiderivative of a polynomial in the base variable with
786    // numeric rational coefficients. Returns None when any coefficient is
787    // genuinely multivariate (a nested slot that does not collapse to a
788    // number) — that should not happen for valid inputs to the Risch
789    // pipeline.
790    if polynomial_is_zero(target) {
791        return Some(zero_polynomial(base_variable));
792    }
793    let mut new_terms: Vec<(usize, PolynomialCoefficient)> = Vec::with_capacity(target.terms.len());
794    for (exponent, coefficient) in target.terms.iter() {
795        let coefficient_number = coefficient_to_number(coefficient)?;
796        let new_exponent = exponent + 1;
797        let divisor = number_value::Integer(Integer::from(new_exponent as i64));
798        let scaled = number::divide(&coefficient_number, &divisor)?;
799        new_terms.push((new_exponent, PolynomialCoefficient::Numeric(scaled)));
800    }
801    new_terms.sort_by(|left, right| right.0.cmp(&left.0));
802    Some(Polynomial {
803        main_variable: base_variable,
804        terms: new_terms,
805    })
806}

808fn rational_function_to_polynomial_in_base(
809    value: &RationalFunction,
810    base_variable: SymbolIdentifier,
811) -> Option<Polynomial> {
812    // Project a rational function to a polynomial in the base variable by
813    // dividing the numerator by the denominator's constant scalar. Returns
814    // None when either the denominator is non-constant or any coefficient is
815    // genuinely multivariate.
816    let denominator_number = constant_polynomial_value(&value.denominator)?;
817    if number::is_zero(&denominator_number) {
818        return None;
819    }
820    let mut working = value.numerator.clone();
821    if !polynomial_is_zero(&working) && working.main_variable != base_variable {
822        let constant_value = constant_polynomial_value(&working)?;
823        working = constant_polynomial(base_variable, constant_value);
824    }
825    for (_, coefficient) in working.terms.iter() {
826        if coefficient_to_number(coefficient).is_none() {
827            return None;
828        }
829    }
830    let one_value = number_value::Integer(Integer::ONE);
831    let inverse = number::divide(&one_value, &denominator_number)?;
832    Some(scale_polynomial(&working, &inverse))
833}

850fn polynomial_coefficient_to_lower(
851    coefficient: &PolynomialCoefficient,
852    lower_variables: &[SymbolIdentifier],
853) -> Option<Polynomial> {
854    // Promote a polynomial coefficient (from the top-variable expansion) into
855    // a polynomial in the lower-tower variables. Numeric coefficients become
856    // constant polynomials in the lower-tower head; nested coefficients are
857    // wrapped through the chain so each ascends from the deepest matching
858    // variable up to the lower-tower head.
859    if lower_variables.is_empty() {
860        return None;
861    }
862    let head = lower_variables[0];
863    match coefficient {
864        PolynomialCoefficient::Numeric(value) => Some(constant_polynomial(head, value.clone())),
865        PolynomialCoefficient::Nested(nested) => {
866            let inner = (**nested).clone();
867            if !lower_variables.contains(&inner.main_variable) {
868                return None;
869            }
870            Some(lift_polynomial_through_chain(lower_variables, inner))
871        }
872        PolynomialCoefficient::Algebraic(_) => None,
873    }
874}
875
876fn lift_polynomial_through_chain(variables: &[SymbolIdentifier], inner: Polynomial) -> Polynomial {
877    // Re-express a polynomial as a polynomial in the chain's head variable,
878    // wrapping intermediate levels in nested-coefficient slots until the
879    // inner polynomial's main variable matches one of the chain entries.
880    if variables.is_empty() || inner.main_variable == variables[0] {
881        return inner;
882    }
883    let promoted = lift_polynomial_through_chain(&variables[1..], inner);
884    if polynomial_is_zero(&promoted) {
885        return zero_polynomial(variables[0]);
886    }
887    Polynomial {
888        main_variable: variables[0],
889        terms: vec![(
890            0,
891            simplify_coefficient(PolynomialCoefficient::Nested(Box::new(promoted))),
892        )],
893    }
894}

896fn project_rational_function_to_lower(
897    value: &RationalFunction,
898    lower_variables: &[SymbolIdentifier],
899) -> Option<RationalFunction> {
900    // Re-express a rational function so its main variable lies in the
901    // lower-tower variable order. Two paths apply: (i) the value's
902    // numerator and denominator already use a lower-tower main variable, in
903    // which case the rational function is returned unchanged; (ii) the main
904    // variable is the top extension carrying only a degree-zero term, which
905    // gets unwrapped via the nested-coefficient slot. Anything else returns
906    // None — that signals the caller should fall back to the unevaluated
907    // integral placeholder.
908    if lower_variables.is_empty() {
909        return None;
910    }
911    let numerator_main = value.numerator.main_variable;
912    let denominator_main = value.denominator.main_variable;
913    if lower_variables.contains(&numerator_main) && lower_variables.contains(&denominator_main) {
914        return Some(value.clone());
915    }
916    if degree(&value.numerator).unwrap_or(0) > 0 {
917        return None;
918    }
919    if degree(&value.denominator).unwrap_or(0) > 0 {
920        return None;
921    }
922    let numerator_coefficient = if polynomial_is_zero(&value.numerator) {
923        PolynomialCoefficient::Numeric(number_value::Integer(Integer::ZERO))
924    } else {
925        value.numerator.terms[0].1.clone()
926    };
927    let denominator_coefficient = value.denominator.terms[0].1.clone();
928    let numerator_lower = polynomial_coefficient_to_lower(&numerator_coefficient, lower_variables)?;
929    let denominator_lower =
930        polynomial_coefficient_to_lower(&denominator_coefficient, lower_variables)?;
931    make_rational_function(numerator_lower, denominator_lower)
932}

934fn polynomial_from_top_terms(
935    mut terms: Vec<(usize, Polynomial)>,
936    top: SymbolIdentifier,
937) -> Polynomial {
938    // Assemble a polynomial in the top extension from its degree-indexed
939    // coefficients (each itself a polynomial in the lower variables). Zero
940    // coefficients are dropped; constant coefficients are collapsed back to
941    // numeric form via simplify_coefficient.
942    terms.sort_by(|left, right| right.0.cmp(&left.0));
943    let mut polynomial_terms: Vec<(usize, PolynomialCoefficient)> = Vec::with_capacity(terms.len());
944    for (exponent, coefficient_polynomial) in terms {
945        if polynomial_is_zero(&coefficient_polynomial) {
946            continue;
947        }
948        let coefficient = simplify_coefficient(PolynomialCoefficient::Nested(Box::new(
949            coefficient_polynomial,
950        )));
951        if crate::polynomial::coefficient_is_zero(&coefficient) {
952            continue;
953        }
954        polynomial_terms.push((exponent, coefficient));
955    }
956    Polynomial {
957        main_variable: top,
958        terms: polynomial_terms,
959    }
960}

962fn accumulate_term(
963    accumulator: &mut Vec<(usize, Polynomial)>,
964    exponent: usize,
965    coefficient_polynomial: Polynomial,
966) {
967    // Combine a new (exponent, polynomial) contribution into the accumulator
968    // list. Matching exponents have their coefficient polynomials summed via
969    // polynomial_add; new exponents are appended in arbitrary order (the
970    // canonical sort happens inside polynomial_from_top_terms).
971    if polynomial_is_zero(&coefficient_polynomial) {
972        return;
973    }
974    if let Some(position) = accumulator
975        .iter()
976        .position(|(existing, _)| *existing == exponent)
977    {
978        let combined = polynomial_add(&accumulator[position].1, &coefficient_polynomial);
979        if polynomial_is_zero(&combined) {
980            accumulator.remove(position);
981        } else {
982            accumulator[position].1 = combined;
983        }
984    } else {
985        accumulator.push((exponent, coefficient_polynomial));
986    }
987}

989fn constant_polynomial_value(value: &Polynomial) -> Option<number_value> {
990    // Project a polynomial to a plain numeric constant when its sole term has
991    // degree zero. Returns Some(0) for the zero polynomial; None for any
992    // non-constant input or any term whose coefficient is genuinely
993    // multivariate.
994    if polynomial_is_zero(value) {
995        return Some(number_value::Integer(Integer::ZERO));
996    }
997    if value.terms.len() != 1 {
998        return None;
999    }
1000    let (exponent, coefficient) = &value.terms[0];
1001    if *exponent != 0 {
1002        return None;
1003    }
1004    coefficient_to_number(coefficient)
1005}

1065/// Performs the advanced native `build_tower_from_integrand` operation.
1066pub fn build_tower_from_integrand(
1067    context: &Context,
1068    target: &ExpressionReference,
1069    variable: SymbolIdentifier,
1070) -> Option<TowerSetup> {
1071    // Construct a differential-field tower covering every `exp` / `log` call
1072    // appearing in the integrand. Base variable is the integration variable;
1073    // extensions are added in dependency order (innermost first) so each new
1074    // generator's argument lies in the partially-built tower. Returns None
1075    // when the integrand contains a free symbol other than the integration
1076    // variable, when an exp/log argument cannot be expressed as a rational
1077    // function in the partial tower, or when the call's argument is itself a
1078    // non-elementary subexpression — the caller then falls back to the
1079    // unevaluated `integral` placeholder.
1080    let free = free_variables(target);
1081    for symbol_value in free.iter() {
1082        if *symbol_value != variable {
1083            return None;
1084        }
1085    }
1086    let mut tower = base_field(variable);
1087    let mut generator_for_call: HashMap<ExpressionReference, SymbolIdentifier> = HashMap::new();
1088    let mut algebraic_substitutions: SubstitutionMap = HashMap::new();
1089    let mut generator_for_radicand: HashMap<ExpressionReference, SymbolIdentifier> = HashMap::new();
1090    let mut generator_to_expression: HashMap<SymbolIdentifier, ExpressionReference> =
1091        HashMap::new();
1092    if contains_imaginary_unit(target) {
1093        let generator = context.symbol_table.fresh("risch_imaginary_unit_");
1094        let minimal_polynomial = imaginary_unit_minimal_polynomial(generator);
1095        extend_algebraic(&mut tower, generator, minimal_polynomial);
1096        let imaginary_unit = builders::constant(context, BuiltinConstant::ImaginaryUnit);
1097        generator_for_call.insert(Arc::clone(&imaginary_unit), generator);
1098        generator_to_expression.insert(generator, imaginary_unit);
1099    }
1100    let calls = collect_exp_log_calls(target);
1101    for call in calls.iter() {
1102        if generator_for_call.contains_key(call) {
1103            continue;
1104        }
1105        let (head, argument) = exp_log_head_and_argument(call)?;
1106        let mapping =
1107            tower_substitution_map(context, &generator_for_call, &algebraic_substitutions);
1108        let rewritten_argument = if mapping.is_empty() {
1109            Arc::clone(&argument)
1110        } else {
1111            substitute(context, &argument, &mapping)
1112        };
1113        let argument_polynomial =
1114            expression_to_polynomial(&rewritten_argument, &variable_order(&tower))?;
1115        let argument_rational = from_polynomial(argument_polynomial);
1116        let prefix = match head {
1117            BuiltinFunction::Exp => "risch_exp_",
1118            BuiltinFunction::Log => "risch_log_",
1119            _ => return None,
1120        };
1121        let generator = context.symbol_table.fresh(prefix);
1122        match head {
1123            BuiltinFunction::Exp => {
1124                extend_exponential(&mut tower, generator, argument_rational);
1125            }
1126            BuiltinFunction::Log => {
1127                extend_logarithmic(&mut tower, generator, argument_rational);
1128            }
1129            _ => return None,
1130        }
1131        generator_for_call.insert(Arc::clone(call), generator);
1132        generator_to_expression.insert(generator, Arc::clone(call));
1133    }
1134    // After the exp / log extensions have been added, look for sqrt-shaped
1135    // sub-expressions and add them as algebraic extensions. This ships
1136    // the square-root subset of Trager 1976; recognition of higher-degree
1137    // radicals (cube_root, general power(arg, 1/n)) lands as a future-round
1138    // refinement once the integrator handles non-quadratic algebraic towers.
1139    let sqrt_calls = collect_sqrt_calls(target);
1140    for sqrt_call in sqrt_calls.iter() {
1141        let argument = sqrt_argument(sqrt_call)?;
1142        let mapping =
1143            tower_substitution_map(context, &generator_for_call, &algebraic_substitutions);
1144        let rewritten_argument = if mapping.is_empty() {
1145            Arc::clone(&argument)
1146        } else {
1147            substitute(context, &argument, &mapping)
1148        };
1149        let exponent = sqrt_generator_exponent(sqrt_call)?;
1150        if let Some(generator) = generator_for_radicand.get(&rewritten_argument).copied() {
1151            let generator_symbol = context.interner.intern(ExpressionKind::Symbol(generator));
1152            let exponent_expression = builders::integer(context, exponent);
1153            let replacement = builders::power(context, generator_symbol, exponent_expression);
1154            algebraic_substitutions.insert(Arc::clone(sqrt_call), replacement);
1155            continue;
1156        }
1157        let argument_polynomial =
1158            expression_to_polynomial(&rewritten_argument, &variable_order(&tower))?;
1159        let generator = context.symbol_table.fresh("risch_algebraic_");
1160        // Build the minimal polynomial alpha^2 - argument over the lower tower
1161        // using the algebraic generator as the polynomial's main variable.
1162        let minimal_polynomial =
1163            build_quadratic_minimal_polynomial(generator, &argument_polynomial);
1164        extend_algebraic(&mut tower, generator, minimal_polynomial);
1165        let generator_symbol = context.interner.intern(ExpressionKind::Symbol(generator));
1166        let exponent_expression = builders::integer(context, exponent);
1167        let replacement = builders::power(context, generator_symbol, exponent_expression);
1168        algebraic_substitutions.insert(Arc::clone(sqrt_call), replacement);
1169        generator_for_radicand.insert(Arc::clone(&rewritten_argument), generator);
1170        let half = builders::rational_value(context, 1, 2);
1171        let principal_square_root = builders::power(context, Arc::clone(&argument), half);
1172        generator_to_expression.insert(generator, principal_square_root);
1173    }
1174    let mapping = tower_substitution_map(context, &generator_for_call, &algebraic_substitutions);
1175    let rewritten = if mapping.is_empty() {
1176        Arc::clone(target)
1177    } else {
1178        substitute(context, target, &mapping)
1179    };
1180    Some(TowerSetup {
1181        tower,
1182        rewritten,
1183        generator_to_expression,
1184    })
1185}

1230fn sqrt_argument(target: &ExpressionReference) -> Option<ExpressionReference> {
1231    // Recognise a square-root surd shape and extract the radicand. The
1232    // recogniser accepts the canonical power(arg, 1/2) form that the
1233    // simplifier rewrites sqrt() into; the direct sqrt() user-function-call
1234    // form is also accepted for inputs that bypass the simplifier.
1235    match &target.kind {
1236        ExpressionKind::Power(base, exponent) => {
1237            if let ExpressionKind::Rational(exponent_value) = &exponent.kind {
1238                let (numerator_natural, denominator_natural) =
1239                    exponent_value.clone().into_numerator_and_denominator();
1240                let numerator_integer: Integer = Integer::from(numerator_natural);
1241                let denominator_integer: Integer = Integer::from(denominator_natural);
1242                if numerator_integer % Integer::from(2) == Integer::ONE
1243                    && denominator_integer == Integer::from(2)
1244                {
1245                    return Some(Arc::clone(base));
1246                }
1247            }
1248            None
1249        }
1250        ExpressionKind::FunctionCall(function, arguments) => {
1251            if let FunctionIdentifier::Builtin(BuiltinFunction::Sqrt) = function {
1252                if arguments.len() == 1 {
1253                    return Some(Arc::clone(&arguments[0]));
1254                }
1255            }
1256            None
1257        }
1258        _ => None,
1259    }
1260}

1262fn sqrt_generator_exponent(target: &ExpressionReference) -> Option<Integer> {
1263    match &target.kind {
1264        ExpressionKind::Power(_, exponent) => {
1265            let ExpressionKind::Rational(exponent_value) = &exponent.kind else {
1266                return None;
1267            };
1268            use crate::exact_arithmetic::compatibility::*;
1269            let (numerator_natural, denominator_natural) =
1270                exponent_value.clone().into_numerator_and_denominator();
1271            if denominator_natural != crate::exact_arithmetic::Natural::from(2u8) {
1272                return None;
1273            }
1274            let mut numerator = Integer::from(numerator_natural);
1275            if exponent_value.sign_comparison() == std::cmp::Ordering::Less {
1276                numerator = -numerator;
1277            }
1278            Some(numerator)
1279        }
1280        ExpressionKind::FunctionCall(function, arguments)
1281            if matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Sqrt))
1282                && arguments.len() == 1 =>
1283        {
1284            Some(Integer::ONE)
1285        }
1286        _ => None,
1287    }

1290fn build_quadratic_minimal_polynomial(
1291    generator: SymbolIdentifier,
1292    argument_polynomial: &Polynomial,
1293) -> Polynomial {
1294    // Build alpha^2 - argument(x) in the algebraic generator's variable. The
1295    // argument polynomial lives in the lower tower; its coefficient form is
1296    // promoted into a nested polynomial coefficient so the algebraic
1297    // polynomial's representation matches the recursive convention used
1298    // elsewhere by the polynomial module.
1299    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
1300    terms.push((
1301        2,
1302        PolynomialCoefficient::Numeric(number_value::Integer(Integer::ONE)),
1303    ));
1304    let negated_argument = crate::polynomial::negate(argument_polynomial);
1305    let constant_coefficient = if polynomial_is_zero(&negated_argument) {
1306        PolynomialCoefficient::Numeric(number_value::Integer(Integer::ZERO))
1307    } else {
1308        simplify_coefficient(PolynomialCoefficient::Nested(Box::new(negated_argument)))
1309    };
1310    if !crate::polynomial::coefficient_is_zero(&constant_coefficient) {
1311        terms.push((0, constant_coefficient));
1312    }
1313    Polynomial {
1314        main_variable: generator,
1315        terms,
1316    }
1317}

1319fn collect_sqrt_calls(target: &ExpressionReference) -> Vec<ExpressionReference> {
1320    // Collect every sqrt-shaped sub-expression (power(arg, 1/2) or sqrt(arg))
1321    // in post-order, deduplicated by pointer identity. Mirrors
1322    // collect_exp_log_calls so the same dependency-aware ordering is used for
1323    // the algebraic-extension build step.
1324    let mut accumulator: Vec<ExpressionReference> = Vec::new();
1325    let mut seen: HashSet<*const crate::expression::Expression> = HashSet::new();
1326    collect_sqrt_recursive(target, &mut accumulator, &mut seen);
1327    accumulator

1330fn collect_sqrt_recursive(
1331    target: &ExpressionReference,
1332    accumulator: &mut Vec<ExpressionReference>,
1333    seen: &mut HashSet<*const crate::expression::Expression>,
1334) {
1335    // Recursive worker for collect_sqrt_calls. Visits structural children
1336    // first, then records the call itself if it matches the sqrt shape and
1337    // has not already been seen.
1338    match &target.kind {
1339        ExpressionKind::Integer(_)
1340        | ExpressionKind::Rational(_)
1341        | ExpressionKind::Symbol(_)
1342        | ExpressionKind::Constant(_) => {}
1343        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
1344            for child in children {
1345                collect_sqrt_recursive(child, accumulator, seen);
1346            }
1347        }
1348        ExpressionKind::Power(base, exponent) => {
1349            collect_sqrt_recursive(base, accumulator, seen);
1350            collect_sqrt_recursive(exponent, accumulator, seen);
1351            if sqrt_argument(target).is_some() {
1352                let pointer = Arc::as_ptr(target);
1353                if !seen.contains(&pointer) {
1354                    seen.insert(pointer);
1355                    accumulator.push(Arc::clone(target));
1356                }
1357            }
1358        }
1359        ExpressionKind::Factorial(argument) => {
1360            collect_sqrt_recursive(argument, accumulator, seen);
1361        }
1362        ExpressionKind::FunctionCall(function, arguments) => {
1363            for argument in arguments {
1364                collect_sqrt_recursive(argument, accumulator, seen);
1365            }
1366            if let FunctionIdentifier::Builtin(BuiltinFunction::Sqrt) = function {
1367                if arguments.len() == 1 {
1368                    let pointer = Arc::as_ptr(target);
1369                    if !seen.contains(&pointer) {
1370                        seen.insert(pointer);
1371                        accumulator.push(Arc::clone(target));
1372                    }
1373                }
1374            }
1375        }
1376        _ => {}
1377    }
1378}

1416fn collect_exp_log_recursive(
1417    target: &ExpressionReference,
1418    accumulator: &mut Vec<ExpressionReference>,
1419    seen: &mut HashSet<*const crate::expression::Expression>,
1420) {
1421    // Recursive worker for collect_exp_log_calls. Visits structural children
1422    // first, then records the call itself if it has not already been seen.
1423    match &target.kind {
1424        ExpressionKind::Integer(_)
1425        | ExpressionKind::Rational(_)
1426        | ExpressionKind::Symbol(_)
1427        | ExpressionKind::Constant(_) => {}
1428        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
1429            for child in children {
1430                collect_exp_log_recursive(child, accumulator, seen);
1431            }
1432        }
1433        ExpressionKind::Power(base, exponent) => {
1434            collect_exp_log_recursive(base, accumulator, seen);
1435            collect_exp_log_recursive(exponent, accumulator, seen);
1436        }
1437        ExpressionKind::Factorial(argument) => {
1438            collect_exp_log_recursive(argument, accumulator, seen);
1439        }
1440        ExpressionKind::FunctionCall(function, arguments) => {
1441            for argument in arguments {
1442                collect_exp_log_recursive(argument, accumulator, seen);
1443            }
1444            if let FunctionIdentifier::Builtin(builtin) = function {
1445                if matches!(builtin, BuiltinFunction::Exp | BuiltinFunction::Log) {
1446                    let pointer = Arc::as_ptr(target);
1447                    if !seen.contains(&pointer) {
1448                        seen.insert(pointer);
1449                        accumulator.push(Arc::clone(target));
1450                    }
1451                }
1452            }
1453        }
1454        ExpressionKind::Derivative(_, _, arguments) => {
1455            for argument in arguments {
1456                collect_exp_log_recursive(argument, accumulator, seen);
1457            }
1458        }
1459        ExpressionKind::Integral(integrand, _, lower_bound, upper_bound) => {
1460            collect_exp_log_recursive(integrand, accumulator, seen);
1461            if let Some(lower) = lower_bound {
1462                collect_exp_log_recursive(lower, accumulator, seen);
1463            }
1464            if let Some(upper) = upper_bound {
1465                collect_exp_log_recursive(upper, accumulator, seen);
1466            }
1467        }
1468        ExpressionKind::Summation(summand, _, lower_bound, upper_bound) => {
1469            collect_exp_log_recursive(summand, accumulator, seen);
1470            if let Some(lower) = lower_bound {
1471                collect_exp_log_recursive(lower, accumulator, seen);
1472            }
1473            if let Some(upper) = upper_bound {
1474                collect_exp_log_recursive(upper, accumulator, seen);
1475            }
1476        }
1477        ExpressionKind::LaplaceTransform(target, _, _)
1478        | ExpressionKind::InverseLaplaceTransform(target, _, _)
1479        | ExpressionKind::FourierTransform(target, _, _)
1480        | ExpressionKind::InverseFourierTransform(target, _, _) => {
1481            collect_exp_log_recursive(target, accumulator, seen);
1482        }
1483        ExpressionKind::Piecewise(branches) => {
1484            for (condition, value) in branches {
1485                collect_exp_log_recursive(condition, accumulator, seen);
1486                collect_exp_log_recursive(value, accumulator, seen);
1487            }
1488        }
1489        ExpressionKind::OrdinaryDifferentialEquation(equation, dependent_variable_call, _)
1490        | ExpressionKind::RecurrenceEquation(equation, dependent_variable_call, _) => {
1491            collect_exp_log_recursive(equation, accumulator, seen);
1492            collect_exp_log_recursive(dependent_variable_call, accumulator, seen);
1493        }
1494        _ => {}
1495    }
1496}

1568// =============================================================================
1569// Proper-rational integrands in the top transcendental extension.
1570//
1571// The dispatcher below handles integrands p(θ)/q(θ) where p, q are polynomials in
1572// the topmost extension θ over the lower differential field K_{n-1}. After
1573// polynomial division separates a polynomial-in-θ part (handled by the
1574// polynomial-in-tower integrator above), the proper-rational remainder runs through:
1575//   1. a Hermite-Mack squarefree reduction in θ (Bronstein §5.3),
1576//   2. the Lazard–Rioboo–Trager construction in θ over K_{n-1}
1577//      (Bronstein §5.10), and
1578//   3. a residual computation in the differential field that hands any
1579//      polynomial-in-θ remainder back to `integrate_in_tower`.
1580//
1581// The LRT step computes R(t) = res_θ(p - t·D(q), q) using the field
1582// derivation D (not the formal partial w.r.t. θ), then strips the K_{n-1}
1583// content so the rational-root theorem applies in Q[t]. Each constant root
1584// c_i yields a log term c_i·log(g_i) with g_i = gcd_θ(p - c_i·D(q), q).
1585// =============================================================================
1586
1587fn split_into_rational_in_tower(
1588    expression: &ExpressionReference,
1589    variable_order: &[SymbolIdentifier],
1590) -> Option<(Polynomial, Polynomial)> {
1591    // Decompose an integrand into (numerator, denominator) where both pieces
1592    // are polynomials in the supplied tower variable order. A top-level
1593    // product is iterated factor by factor; reciprocal-power factors (power
1594    // node with a negative integer exponent) contribute to the denominator.
1595    // Returns None whenever any factor falls outside the polynomial fragment
1596    // — the caller then reports the integrand as outside the proper-rational
1597    // path.
1598    let head_variable = variable_order[0];
1599    let one_polynomial = constant_polynomial(head_variable, number_value::Integer(Integer::ONE));
1600    match &expression.kind {
1601        ExpressionKind::Product(factors) => {
1602            let mut numerator_polynomial = one_polynomial.clone();
1603            let mut denominator_polynomial = one_polynomial;
1604            for factor in factors {
1605                let (numerator_part, denominator_part) =
1606                    split_single_factor_in_tower(factor, variable_order)?;
1607                numerator_polynomial = polynomial_multiply(&numerator_polynomial, &numerator_part);
1608                denominator_polynomial =
1609                    polynomial_multiply(&denominator_polynomial, &denominator_part);
1610            }
1611            Some((numerator_polynomial, denominator_polynomial))
1612        }
1613        _ => split_single_factor_in_tower(expression, variable_order),
1614    }
1615}
1616
1617fn split_single_factor_in_tower(
1618    expression: &ExpressionReference,
1619    variable_order: &[SymbolIdentifier],
1620) -> Option<(Polynomial, Polynomial)> {
1621    // Classify a single factor into numerator versus denominator contribution.
1622    // A `power(base, -k)` with `k` a positive integer contributes `(1,
1623    // base^k)`; anything else that converts cleanly to a polynomial in the
1624    // tower contributes `(polynomial, 1)`.
1625    let head_variable = variable_order[0];
1626    let one_polynomial = constant_polynomial(head_variable, number_value::Integer(Integer::ONE));
1627    match &expression.kind {
1628        ExpressionKind::Power(base, exponent) => {
1629            let exponent_integer = match &exponent.kind {
1630                ExpressionKind::Integer(value) => value.clone(),
1631                _ => {
1632                    let polynomial_value = expression_to_polynomial(expression, variable_order)?;
1633                    return Some((polynomial_value, one_polynomial));
1634                }
1635            };
1636            let signed_exponent = i64::try_from(&exponent_integer).ok()?;
1637            let base_polynomial = expression_to_polynomial(base, variable_order)?;
1638            if signed_exponent >= 0 {
1639                let raised = polynomial_pow(&base_polynomial, signed_exponent as usize);
1640                Some((raised, one_polynomial))
1641            } else {
1642                let magnitude = (-signed_exponent) as usize;
1643                let raised = polynomial_pow(&base_polynomial, magnitude);
1644                Some((one_polynomial, raised))
1645            }
1646        }
1647        _ => {
1648            let polynomial_value = expression_to_polynomial(expression, variable_order)?;
1649            Some((polynomial_value, one_polynomial))
1650        }
1651    }
1652}

1654fn integrate_rational_in_top_extension(
1655    context: &Context,
1656    numerator: &Polynomial,
1657    denominator: &Polynomial,
1658    tower: &DifferentialField,
1659) -> IntegrationResult {
1660    // Integrate p(θ)/q(θ) where p, q are polynomials in the top
1661    // extension θ of `tower` (over K_{n-1}). Splits off the polynomial-in-θ
1662    // part via long division (handed back to `integrate_polynomial_in_tower`),
1663    // then runs Hermite-Mack + LRT-in-θ on the proper rational remainder. The
1664    // residual produced after subtracting the LRT contribution is required to
1665    // be polynomial in θ — otherwise the integrand falls outside this
1666    // decision procedure's scope and we report `incomplete` for the caller
1667    // to emit the unevaluated placeholder.
1668    if tower.extensions.is_empty() {
1669        return IntegrationResult::Incomplete;
1670    }
1671    let top = top_variable(tower);
1672    if denominator.main_variable != top || polynomial_is_zero(denominator) {
1673        return IntegrationResult::Incomplete;
1674    }
1675    if matches!(
1676        find_extension(tower, top).map(|extension| &extension.kind),
1677        Some(ExtensionKind::Algebraic(_))
1678    ) {
1679        return crate::integrate::algebraic_tower::integrate_in_algebraic_tower(
1680            context,
1681            numerator,
1682            denominator,
1683            tower,
1684        );
1685    }
1686    let denominator_degree = match degree(denominator) {
1687        Some(value) if value > 0 => value,
1688        _ => return IntegrationResult::Incomplete,
1689    };
1690    if !top_extension_supports_proper_rational(tower) {
1691        return IntegrationResult::Incomplete;
1692    }
1693    // Split off any polynomial-in-θ part via long division. When the
1694    // divisor's leading coefficient does not lift to a numeric K-constant
1695    // (e.g. the denominator x·θ in 1/(x·log(x))), polynomial division would
1696    // need a K_{n-1} inversion we do not yet implement. If the integrand is
1697    // already proper (degree θ < degree θ of divisor) we can skip the
1698    // division entirely and treat the polynomial part as zero — that closes
1699    // the common case where the proper-rational shape is observed directly.
1700    let numerator_degree_theta = if polynomial_is_zero(numerator) {
1701        None
1702    } else if numerator.main_variable == top {
1703        degree(numerator)
1704    } else {
1705        Some(0)
1706    };
1707    let already_proper_rational = match numerator_degree_theta {
1708        Some(value) => value < denominator_degree,
1709        None => true,
1710    };
1711    let (quotient, proper_numerator) = if already_proper_rational {
1712        let proper_lifted = if numerator.main_variable == top {
1713            numerator.clone()
1714        } else if polynomial_is_zero(numerator) {
1715            zero_polynomial(top)
1716        } else {
1717            // Numerator is constant in θ. Wrap it as a degree-zero polynomial
1718            // in θ so subsequent arithmetic stays in θ over K_{n-1}.
1719            Polynomial {
1720                main_variable: top,
1721                terms: vec![(
1722                    0,
1723                    simplify_coefficient(PolynomialCoefficient::Nested(Box::new(
1724                        numerator.clone(),
1725                    ))),
1726                )],
1727            }
1728        };
1729        (zero_polynomial(top), proper_lifted)
1730    } else {
1731        match divide_polynomial_in_top_extension(numerator, denominator, top) {
1732            Some(value) => value,
1733            None => return IntegrationResult::Incomplete,
1734        }
1735    };
1736    let polynomial_part_integral = if polynomial_is_zero(&quotient) {
1737        builders::zero(context)
1738    } else {
1739        match integrate_polynomial_in_tower(context, &quotient, tower) {
1740            IntegrationResult::Elementary(value) => value,
1741            IntegrationResult::NonElementary => return IntegrationResult::NonElementary,
1742            IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
1743        }
1744    };
1745    if polynomial_is_zero(&proper_numerator) {
1746        return IntegrationResult::Elementary(polynomial_part_integral);
1747    }
1748    let _ = denominator_degree;
1749    let (hermite_pieces, hermite_numerator, hermite_denominator) =
1750        match hermite_reduce_in_top_extension(context, &proper_numerator, denominator, tower) {
1751            Some(value) => value,
1752            None => return IntegrationResult::Incomplete,
1753        };
1754    let log_outcome = lazard_rioboo_trager_in_top_extension(
1755        context,
1756        &hermite_numerator,
1757        &hermite_denominator,
1758        tower,
1759    );
1760    let log_pairs = match log_outcome {
1761        Some(value) => value,
1762        None => return IntegrationResult::Incomplete,
1763    };
1764    let log_part_expression = build_log_part_expression(context, &log_pairs, tower.base_variable);
1765    let cumulative_derivative = match compute_lrt_cumulative_derivative(&log_pairs, tower, top) {
1766        Some(value) => value,
1767        None => return IntegrationResult::Incomplete,
1768    };
1769    let integrand_rational =
1770        match make_rational_function(hermite_numerator.clone(), hermite_denominator.clone()) {
1771            Some(value) => value,
1772            None => return IntegrationResult::Incomplete,
1773        };
1774    let residual_rational =
1775        subtract_rational_functions(&integrand_rational, &cumulative_derivative);
1776    let residual_integration = if rational_function_is_zero(&residual_rational) {
1777        builders::zero(context)
1778    } else if polynomial_is_constant_one(&residual_rational.denominator) {
1779        let residual_expression = polynomial_to_expression(context, &residual_rational.numerator);
1780        match integrate_in_tower(context, &residual_expression, tower) {
1781            IntegrationResult::Elementary(value) => value,
1782            IntegrationResult::NonElementary => return IntegrationResult::NonElementary,
1783            IntegrationResult::Incomplete => return IntegrationResult::Incomplete,
1784        }
1785    } else {
1786        return IntegrationResult::Incomplete;
1787    };
1788    let hermite_part_expression = if hermite_pieces.is_empty() {
1789        builders::zero(context)
1790    } else if hermite_pieces.len() == 1 {
1791        hermite_pieces.into_iter().next().unwrap()
1792    } else {
1793        builders::sum(context, hermite_pieces)
1794    };
1795    let combined = builders::sum(
1796        context,
1797        vec![
1798            polynomial_part_integral,
1799            hermite_part_expression,
1800            log_part_expression,
1801            residual_integration,
1802        ],
1803    );
1804    IntegrationResult::Elementary(combined)
1805}

1822fn divide_polynomial_in_top_extension(
1823    dividend: &Polynomial,
1824    divisor: &Polynomial,
1825    top: SymbolIdentifier,
1826) -> Option<(Polynomial, Polynomial)> {
1827    // Long division of two polynomials in θ over K_{n-1}. Currently requires
1828    // the divisor's leading coefficient to lift to a numeric — that is, to a
1829    // K-constant — so that division by it is well-defined without a field
1830    // inversion in K_{n-1}. Returns (quotient, remainder) with deg(remainder)
1831    // < deg(divisor).
1832    if polynomial_is_zero(divisor) || divisor.main_variable != top {
1833        return None;
1834    }
1835    if polynomial_is_zero(dividend) {
1836        return Some((zero_polynomial(top), zero_polynomial(top)));
1837    }
1838    if dividend.main_variable != top {
1839        return Some((zero_polynomial(top), dividend.clone()));
1840    }
1841    let divisor_degree = degree(divisor).unwrap();
1842    let leading_divisor = leading_coefficient(divisor)?;
1843    let leading_divisor_number = coefficient_to_number(&leading_divisor)?;
1844    if number::is_zero(&leading_divisor_number) {
1845        return None;
1846    }
1847    let one_value = number_value::Integer(Integer::ONE);
1848    let leading_divisor_inverse = number::divide(&one_value, &leading_divisor_number)?;
1849    let mut working = dividend.clone();
1850    let mut quotient = zero_polynomial(top);
1851    while !polynomial_is_zero(&working) {
1852        let working_degree = degree(&working).unwrap();
1853        if working_degree < divisor_degree {
1854            break;
1855        }
1856        let working_leading = leading_coefficient(&working)?;
1857        let working_leading_number = match coefficient_to_number(&working_leading) {
1858            Some(value) => value,
1859            None => return None,
1860        };
1861        let coefficient_ratio = number::multiply(&working_leading_number, &leading_divisor_inverse);
1862        if number::is_zero(&coefficient_ratio) {
1863            break;
1864        }
1865        let exponent = working_degree - divisor_degree;
1866        polynomial_insert_or_combine(
1867            &mut quotient.terms,
1868            exponent,
1869            PolynomialCoefficient::Numeric(coefficient_ratio.clone()),
1870        );
1871        let monomial = Polynomial {
1872            main_variable: top,
1873            terms: vec![(exponent, PolynomialCoefficient::Numeric(coefficient_ratio))],
1874        };
1875        let subtraction = polynomial_multiply(&monomial, divisor);
1876        working = polynomial_subtract(&working, &subtraction);
1877    }
1878    Some((quotient, working))
1879}

1881fn hermite_reduce_in_top_extension(
1882    context: &Context,
1883    proper_numerator: &Polynomial,
1884    denominator: &Polynomial,
1885    tower: &DifferentialField,
1886) -> Option<(Vec<ExpressionReference>, Polynomial, Polynomial)> {
1887    // Hermite-Mack reduction in θ over K_{n-1} (Bronstein §5.3). Squarefree-
1888    // factors the denominator with respect to the field derivation, then
1889    // iteratively reduces every factor whose squarefree multiplicity exceeds
1890    // one. The squarefree decomposition is computed via gcd(q, D(q)) where
1891    // D is the field derivation rather than the formal partial w.r.t. θ;
1892    // for transcendental towers this still produces a valid squarefree
1893    // factorisation because D(q) is a polynomial in θ.
1894    //
1895    // The current implementation ships the no-op pre-check: when the
1896    // denominator is already squarefree (the common case for the worked
1897    // examples in Bronstein §5.10, including 1/(exp(x)+1) and 1/(x*log(x))),
1898    // we return zero rational pieces and the unchanged residue. Multi-root
1899    // denominators fall through to `None`, which lets the caller report
1900    // `incomplete`. The full reduction is straightforward to add on top of
1901    // the same gcd-with-derivative scaffold but isn't required for the
1902    // current test corpus.
1903    let denominator_derivative_rational = derive_polynomial(tower, denominator);
1904    if !polynomial_is_constant_one(&denominator_derivative_rational.denominator) {
1905        return None;
1906    }
1907    let denominator_derivative = denominator_derivative_rational.numerator;
1908    let squarefree_witness =
1909        polynomial_gcd_with_operation(denominator, &denominator_derivative, context.operation());
1910    if degree(&squarefree_witness).unwrap_or(0) > 0 {
1911        // Denominator has a repeated factor under the field derivation; the
1912        // multi-root Hermite reduction isn't implemented yet. Bail out so
1913        // the caller can fall back to the placeholder.
1914        return None;
1915    }
1916    Some((Vec::new(), proper_numerator.clone(), denominator.clone()))
1917}

1919fn lazard_rioboo_trager_in_top_extension(
1920    context: &Context,
1921    numerator: &Polynomial,
1922    denominator: &Polynomial,
1923    tower: &DifferentialField,
1924) -> Option<Vec<(number_value, Polynomial)>> {
1925    // Compute the LRT-in-θ over K_{n-1} for a proper rational
1926    // p(θ)/q(θ) with q squarefree in θ under the field derivation. Builds
1927    // R(t) = res_θ(p - t·D(q), q) where D is the differential-field
1928    // derivation, strips the K_{n-1} content so the rational-root theorem
1929    // applies in Q[t], and emits one (c_i, g_i) pair per rational root c_i
1930    // with g_i = gcd_θ(p - c_i·D(q), q) (made monic in θ when its leading
1931    // coefficient lifts to Q). Returns None when the resultant fails to
1932    // factor cleanly over the constants — that's the documented scope
1933    // boundary; integrands beyond it fall back to the unevaluated
1934    // placeholder.
1935    let top = top_variable(tower);
1936    let denominator_derivative_rational = derive_polynomial(tower, denominator);
1937    if !polynomial_is_constant_one(&denominator_derivative_rational.denominator) {
1938        return None;
1939    }
1940    let denominator_derivative = denominator_derivative_rational.numerator;
1941    let parameter = context.symbol_table.fresh("integration_parameter");
1942    let p_minus_t_qprime =
1943        build_p_minus_t_qprime_in_top_extension(numerator, &denominator_derivative, top, parameter);
1944    let denominator_bivariate =
1945        lift_polynomial_with_parameter_in_top_extension(denominator, top, parameter);
1946    let resultant_coefficient = sylvester_resultant(&p_minus_t_qprime, &denominator_bivariate);
1947    let resultant_polynomial = match resultant_coefficient {
1948        PolynomialCoefficient::Nested(boxed) => *boxed,
1949        PolynomialCoefficient::Numeric(numeric_value) => {
1950            if number::is_zero(&numeric_value) {
1951                return Some(Vec::new());
1952            }
1953            return Some(Vec::new());
1954        }
1955        PolynomialCoefficient::Algebraic(_) => return None,
1956    };
1957    if polynomial_is_zero(&resultant_polynomial) {
1958        return Some(Vec::new());
1959    }
1960    if resultant_polynomial.main_variable != parameter {
1961        // Resultant degenerated into a K_{n-1} constant carried through the
1962        // wrong main variable — give up rather than risk a wrong factor.
1963        return None;
1964    }
1965    let q_resultant = strip_lower_content_for_t_polynomial(&resultant_polynomial)?;
1966    let roots = find_all_rational_roots_in_t(&q_resultant);
1967    if roots.is_empty() {
1968        return None;
1969    }
1970    let mut results: Vec<(number_value, Polynomial)> = Vec::new();
1971    for root in roots.iter() {
1972        let scaled_derivative = scale_polynomial(&denominator_derivative, root);
1973        let shifted_numerator = polynomial_subtract(numerator, &scaled_derivative);
1974        let gcd_polynomial =
1975            polynomial_gcd_with_operation(&shifted_numerator, denominator, context.operation());
1976        let monic = make_polynomial_monic_in_top(&gcd_polynomial, top)?;
1977        if degree(&monic).unwrap_or(0) == 0 {
1978            continue;
1979        }
1980        results.push((root.clone(), monic));
1981    }
1982    Some(results)
1983}

1985fn build_p_minus_t_qprime_in_top_extension(
1986    numerator: &Polynomial,
1987    denominator_derivative: &Polynomial,
1988    top: SymbolIdentifier,
1989    parameter: SymbolIdentifier,
1990) -> Polynomial {
1991    // Build the polynomial p - t·D(q) as a polynomial in θ whose coefficients
1992    // are themselves polynomials in t with K_{n-1} sub-coefficients. The two
1993    // contributions are inserted via `insert_or_combine` so a θ-term that
1994    // appears in both numerator and derivative gets a single combined
1995    // coefficient in t.
1996    let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
1997    for (exponent_theta, coefficient) in numerator.terms.iter() {
1998        let t_polynomial = Polynomial {
1999            main_variable: parameter,
2000            terms: vec![(0, coefficient.clone())],
2001        };
2002        let wrapped = PolynomialCoefficient::Nested(Box::new(t_polynomial));
2003        polynomial_insert_or_combine(&mut terms, *exponent_theta, wrapped);
2004    }
2005    for (exponent_theta, coefficient) in denominator_derivative.terms.iter() {
2006        let negated = negate_coefficient(coefficient);
2007        let t_polynomial = Polynomial {
2008            main_variable: parameter,
2009            terms: vec![(1, negated)],
2010        };
2011        let wrapped = PolynomialCoefficient::Nested(Box::new(t_polynomial));
2012        polynomial_insert_or_combine(&mut terms, *exponent_theta, wrapped);
2013    }
2014    Polynomial {
2015        main_variable: top,
2016        terms,
2017    }
2018}

2064fn find_all_rational_roots_in_t(value: &Polynomial) -> Vec<number_value> {
2065    // Enumerate every rational root of a univariate polynomial in t with
2066    // numeric coefficients. Uses the rational-root theorem: a rational root
2067    // p/q (lowest terms) must satisfy p | constant_coefficient and q |
2068    // leading_coefficient. Rational coefficients in the input are bailed
2069    // out of by returning an empty list — the LRT caller then reports
2070    // incomplete.
2071    if polynomial_is_zero(value) {
2072        return Vec::new();
2073    }
2074    let degree_value = match degree(value) {
2075        Some(value) => value,
2076        None => return Vec::new(),
2077    };
2078    if degree_value == 0 {
2079        return Vec::new();
2080    }
2081    let constant_coefficient_number = value
2082        .terms
2083        .iter()
2084        .find(|(exponent, _)| *exponent == 0)
2085        .and_then(|(_, coefficient)| coefficient_to_number(coefficient));
2086    let leading_coefficient_number =
2087        leading_coefficient(value).and_then(|coefficient| coefficient_to_number(&coefficient));
2088    let constant_integer = match constant_coefficient_number {
2089        Some(number_value::Integer(value)) => value,
2090        Some(number_value::Rational(_)) => return Vec::new(),
2091        None => Integer::ZERO,
2092    };
2093    let leading_integer = match leading_coefficient_number {
2094        Some(number_value::Integer(value)) => value,
2095        Some(number_value::Rational(_)) => return Vec::new(),
2096        None => return Vec::new(),
2097    };
2098    let mut roots: Vec<number_value> = Vec::new();
2099    if constant_integer == Integer::ZERO {
2100        let candidate = number_value::Integer(Integer::ZERO);
2101        if evaluate_t_polynomial_at(value, &candidate) {
2102            roots.push(candidate);
2103        }
2104    }
2105    let absolute_constant = absolute_integer_value(&constant_integer);
2106    let absolute_leading = absolute_integer_value(&leading_integer);
2107    let numerator_divisors = positive_divisors_of(&absolute_constant);
2108    let denominator_divisors = positive_divisors_of(&absolute_leading);
2109    for numerator_divisor in numerator_divisors.iter() {
2110        for denominator_divisor in denominator_divisors.iter() {
2111            for sign in [1i64, -1] {
2112                let signed_numerator = if sign == 1 {
2113                    numerator_divisor.clone()
2114                } else {
2115                    -numerator_divisor.clone()
2116                };
2117                let candidate =
2118                    match number::from_integers(signed_numerator, denominator_divisor.clone()) {
2119                        Some(value) => value,
2120                        None => continue,
2121                    };
2122                if !evaluate_t_polynomial_at(value, &candidate) {
2123                    continue;
2124                }
2125                if !roots.iter().any(|existing| *existing == candidate) {
2126                    roots.push(candidate);
2127                }
2128            }
2129        }
2130    }
2131    roots
2132}

2134fn evaluate_t_polynomial_at(value: &Polynomial, point: &number_value) -> bool {
2135    // True iff `value` evaluates to zero at `point`. The walk runs over Q so
2136    // the result is exact; no floating-point round-off can mask a zero.
2137    let mut total = number_value::Integer(Integer::ZERO);
2138    for (exponent, coefficient) in value.terms.iter() {
2139        let coefficient_number = match coefficient_to_number(coefficient) {
2140            Some(value) => value,
2141            None => return false,
2142        };
2143        let point_power = match number::power_integer(point, &Integer::from(*exponent as i64)) {
2144            Some(value) => value,
2145            None => return false,
2146        };
2147        let term = number::multiply(&coefficient_number, &point_power);
2148        total = number::add(&total, &term);
2149    }
2150    number::is_zero(&total)
2151}

2158fn positive_divisors_of(value: &Integer) -> Vec<Integer> {
2159    // Enumerate every positive integer divisor of |value| via O(√n) trial
2160    // division. For the small coefficient magnitudes that appear in LRT
2161    // resultants the cost is negligible.
2162    if *value == Integer::ZERO {
2163        return vec![Integer::ONE];
2164    }
2165    let absolute = absolute_integer_value(value);
2166    let mut divisors: Vec<Integer> = Vec::new();
2167    let mut candidate = Integer::ONE;
2168    while candidate.clone() * candidate.clone() <= absolute {
2169        if absolute.clone() % candidate.clone() == Integer::ZERO {
2170            divisors.push(candidate.clone());
2171            let complement = absolute.clone() / candidate.clone();
2172            if complement != candidate {
2173                divisors.push(complement);
2174            }
2175        }
2176        candidate += Integer::ONE;
2177    }
2178    divisors
2179}

2222fn build_log_part_expression(
2223    context: &Context,
2224    log_pairs: &[(number_value, Polynomial)],
2225    variable: SymbolIdentifier,
2226) -> ExpressionReference {
2227    // Reify the LRT (c_i, g_i) pairs as a single expression Σ c_i · log(g_i),
2228    // each logarithm routed through the shared antiderivative_logarithm choke
2229    // point so it carries an absolute value only under a real-domain assumption
2230    // on the integration variable. Empty pair list folds to zero so downstream
2231    // sum builders can absorb the result without a special-case branch.
2232    if log_pairs.is_empty() {
2233        return builders::zero(context);
2234    }
2235    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(log_pairs.len());
2236    for (root, g_polynomial) in log_pairs.iter() {
2237        let g_expression = polynomial_to_expression(context, g_polynomial);
2238        let log_call = super::antiderivative_logarithm(context, g_expression, variable);
2239        let root_expression = intern_number_value_for_log(context, root.clone());
2240        let log_term = builders::product(context, vec![root_expression, log_call]);
2241        terms.push(log_term);
2242    }
2243    if terms.len() == 1 {
2244        terms.into_iter().next().unwrap()
2245    } else {

2261fn compute_lrt_cumulative_derivative(
2262    log_pairs: &[(number_value, Polynomial)],
2263    tower: &DifferentialField,
2264    top: SymbolIdentifier,
2265) -> Option<RationalFunction> {
2266    // Compute Σ c_i · D(g_i) / g_i as a rational function in θ over K_{n-1}.
2267    // Each term arrives via `derive_polynomial` (which already handles the
2268    // tower's chain rule) divided by g_i; the c_i scale is folded in via a
2269    // constant rational-function multiplier. Returns None when a g_i is zero
2270    // or division produces a zero denominator — neither should arise for
2271    // valid LRT outputs but the guard keeps the function total.
2272    let mut cumulative = rational_function_zero(top);
2273    for (root, g_polynomial) in log_pairs.iter() {
2274        let g_derivative_rational = derive_polynomial(tower, g_polynomial);
2275        let g_as_rational = from_polynomial(g_polynomial.clone());
2276        let derivative_over_g = divide_rational_functions(&g_derivative_rational, &g_as_rational)?;
2277        let root_rational = constant_rational_function_in_top(top, root.clone());
2278        let contribution = multiply_rational_functions(&root_rational, &derivative_over_g);
2279        cumulative = add_rational_functions(&cumulative, &contribution);
2280    }
2281    Some(cumulative)
2282}

2289    // denominator share `top` as their main variable. Used to scale the LRT
2290    // cumulative derivative by each rational root c_i.
2291    RationalFunction {
2292        numerator: constant_polynomial(top, value),
2293        denominator: constant_polynomial(top, number_value::Integer(Integer::ONE)),
