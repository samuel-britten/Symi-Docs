21pub(crate) fn integrate_parametric_rational(
22    context: &Context,
23    target: &ExpressionReference,
24    variable: SymbolIdentifier,
25) -> Option<ExpressionReference> {
26    let cancelled = crate::simplify::rational::cancel(context, target);
27    let pair = extract_rational_polynomial(context, &cancelled, variable)?;
28    let (numerator, denominator) =
29        cancel_common_numeric_leading_numerator(context, pair.numerator, pair.denominator);
30    let numerator = trim_polynomial(context, numerator);
31    let denominator = trim_polynomial(context, denominator);
32    if numerator.len().saturating_sub(1) > NUMERATOR_DEGREE_CAP
33        || denominator.len() > 3
34        || denominator.is_empty()
35    {
36        return None;
37    }
38    let mut coefficient_parameters: Vec<SymbolIdentifier> = free_variables(&cancelled)
39        .into_iter()
40        .filter(|identifier| *identifier != variable)
41        .collect();
42    coefficient_parameters.sort_unstable();
43    if coefficient_parameters.is_empty() || coefficient_parameters.len() > 8 {
44        return None;
45    }
46    if !coefficients_are_rational_functions(&numerator, &coefficient_parameters)
47        || !coefficients_are_rational_functions(&denominator, &coefficient_parameters)
48    {
49        return None;
50    }
51    let mut condition_parameters: Vec<SymbolIdentifier> = denominator
52        .iter()
53        .flat_map(|coefficient| free_variables(coefficient).into_iter())
54        .filter(|identifier| *identifier != variable)
55        .collect();
56    condition_parameters.sort_unstable();
57    condition_parameters.dedup();
58    if condition_parameters.len() > 3 {
59        return None;
60    }
61    let zero = builders::zero(context);
62    let denominator_coefficient = |index: usize, context: &Context| {
63        denominator
64            .get(index)
65            .cloned()
66            .unwrap_or_else(|| builders::zero(context))
67    };
68    let constant = denominator_coefficient(0, context);
69    let linear = denominator_coefficient(1, context);
70    let quadratic = denominator_coefficient(2, context);
71    if denominator.len() == 1 {
72        return integrate_constant_branch(context, &numerator, variable, &constant);
73    }
74    if denominator.len() == 2 {
75        return integrate_affine_partition(
76            context,
77            &numerator,
78            variable,
79            &linear,
80            &constant,
81            condition_parameters,
82        );
83    }
84    let quadratic_nonzero = builders::not_equal(context, Arc::clone(&quadratic), Arc::clone(&zero));
85    let quadratic_zero = builders::equal(context, Arc::clone(&quadratic), Arc::clone(&zero));
86    let linear_nonzero = builders::not_equal(context, Arc::clone(&linear), Arc::clone(&zero));
87    let linear_zero = builders::equal(context, Arc::clone(&linear), Arc::clone(&zero));
88    let constant_nonzero = builders::not_equal(context, Arc::clone(&constant), Arc::clone(&zero));
89    let four = builders::integer_value(context, 4);
90    let four_quadratic_constant = builders::product(
91        context,
92        vec![four, Arc::clone(&quadratic), Arc::clone(&constant)],
93    );
94    let two = builders::integer_value(context, 2);
95    let linear_squared = builders::power(context, Arc::clone(&linear), two);
96    let discriminant = builders::subtract(context, four_quadratic_constant, linear_squared);
97    let discriminant_positive =
98        builders::less_than(context, Arc::clone(&zero), Arc::clone(&discriminant));
99    let discriminant_negative =
100        builders::less_than(context, Arc::clone(&discriminant), Arc::clone(&zero));
101    let discriminant_zero = builders::equal(context, Arc::clone(&discriminant), Arc::clone(&zero));
102    let quadratic_positive_condition = logical_and(
103        context,
104        vec![Arc::clone(&quadratic_nonzero), discriminant_positive],
105    );
106    let quadratic_negative_condition = logical_and(
107        context,
108        vec![Arc::clone(&quadratic_nonzero), discriminant_negative],
109    );
110    let quadratic_repeated_condition =
111        logical_and(context, vec![quadratic_nonzero, discriminant_zero]);
112    let affine_condition = logical_and(
113        context,
114        vec![Arc::clone(&quadratic_zero), Arc::clone(&linear_nonzero)],
115    );
116    let constant_condition = logical_and(
117        context,
118        vec![quadratic_zero, linear_zero, Arc::clone(&constant_nonzero)],
119    );
120    let quadratic_positive = integrate_quadratic_branch(
121        context,
122        &numerator,
123        variable,
124        &quadratic,
125        &linear,
126        &constant,
127        &discriminant,
128        QuadraticBranchKind::Positive,
129    )?;
130    let quadratic_negative = integrate_quadratic_branch(
131        context,
132        &numerator,
133        variable,
134        &quadratic,
135        &linear,
136        &constant,
137        &discriminant,
138        QuadraticBranchKind::Negative,
139    )?;
140    let quadratic_repeated = integrate_quadratic_branch(
141        context,
142        &numerator,
143        variable,
144        &quadratic,
145        &linear,
146        &constant,
147        &discriminant,
148        QuadraticBranchKind::Zero,
149    )?;
150    let affine = integrate_affine_branch(context, &numerator, variable, &linear, &constant)?;
151    let constant_value = integrate_constant_branch(context, &numerator, variable, &constant)?;
152    let defined_quadratic = builders::not_equal(context, Arc::clone(&quadratic), Arc::clone(&zero));
153    let defined_linear = builders::not_equal(context, Arc::clone(&linear), Arc::clone(&zero));
154    let defined_region = logical_or(
155        context,
156        vec![defined_quadratic, defined_linear, constant_nonzero],
157    );
158    ParameterPartition {
159        parameters: condition_parameters,
160        defined_region,
161        branches: vec![
162            ParameterBranch {
163                condition: constant_condition,
164                value: constant_value,
165            },
166            ParameterBranch {
167                condition: affine_condition,
168                value: affine,
169            },
170            ParameterBranch {
171                condition: quadratic_repeated_condition,
172                value: quadratic_repeated,
173            },
174            ParameterBranch {
175                condition: quadratic_positive_condition,
176                value: quadratic_positive,
177            },
178            ParameterBranch {
179                condition: quadratic_negative_condition,
180                value: quadratic_negative,
181            },
182        ],
183        ordered_real: true,
184        prune_infeasible: false,
185    }
186    .verify_and_reify(context)
187}

189fn cancel_common_numeric_leading_numerator(
190    context: &Context,
191    numerator: Vec<ExpressionReference>,
192    denominator: Vec<ExpressionReference>,
193) -> (Vec<ExpressionReference>, Vec<ExpressionReference>) {
194    let numerator = trim_polynomial(context, numerator);
195    let denominator = trim_polynomial(context, denominator);
196    if numerator.len() <= 1 || denominator.len() <= numerator.len() {
197        return (numerator, denominator);
198    }
199    let Some(leading) = numerator.last().cloned() else {
200        return (numerator, denominator);
201    };
202    if !free_variables(&leading).is_empty()
203        || !crate::expression::queries::is_provably_nonzero(context, &leading)
204    {
205        return (numerator, denominator);
206    }
207    let monic_candidate: Vec<ExpressionReference> = numerator
208        .iter()
209        .map(|coefficient| builders::divide(context, Arc::clone(coefficient), Arc::clone(&leading)))
210        .collect();
211    let Some((numerator_quotient, numerator_remainder)) =
212        divide_polynomials(context, &numerator, &monic_candidate)
213    else {
214        return (numerator, denominator);
215    };
216    let Some((denominator_quotient, denominator_remainder)) =
217        divide_polynomials(context, &denominator, &monic_candidate)
218    else {
219        return (numerator, denominator);
220    };
221    if !numerator_remainder.is_empty() || !denominator_remainder.is_empty() {
222        return (numerator, denominator);
223    }
224    (numerator_quotient, denominator_quotient)
225}

227fn integrate_affine_partition(
228    context: &Context,
229    numerator: &[ExpressionReference],
230    variable: SymbolIdentifier,
231    linear: &ExpressionReference,
232    constant: &ExpressionReference,
233    parameters: Vec<SymbolIdentifier>,
234) -> Option<ExpressionReference> {
235    let zero = builders::zero(context);
236    let linear_zero = builders::equal(context, Arc::clone(linear), Arc::clone(&zero));
237    let linear_nonzero = builders::not_equal(context, Arc::clone(linear), Arc::clone(&zero));
238    let constant_nonzero = builders::not_equal(context, Arc::clone(constant), Arc::clone(&zero));
239    let constant_condition = logical_and(context, vec![linear_zero, Arc::clone(&constant_nonzero)]);
240    let defined_region = logical_or(context, vec![Arc::clone(&linear_nonzero), constant_nonzero]);
241    let affine_value = integrate_affine_branch(context, numerator, variable, linear, constant)?;
242    let constant_value = integrate_constant_branch(context, numerator, variable, constant)?;
243    ParameterPartition {
244        parameters,
245        defined_region,
246        branches: vec![
247            ParameterBranch {
248                condition: constant_condition,
249                value: constant_value,
250            },
251            ParameterBranch {
252                condition: linear_nonzero,
253                value: affine_value,
254            },
255        ],
256        ordered_real: false,
257        prune_infeasible: false,
258    }
259    .verify_and_reify(context)
260}

269fn integrate_quadratic_branch(
270    context: &Context,
271    numerator: &[ExpressionReference],
272    variable: SymbolIdentifier,
273    quadratic: &ExpressionReference,
274    linear: &ExpressionReference,
275    constant: &ExpressionReference,
276    discriminant: &ExpressionReference,
277    branch_kind: QuadraticBranchKind,
278) -> Option<ExpressionReference> {
279    let denominator = vec![
280        Arc::clone(constant),
281        Arc::clone(linear),
282        Arc::clone(quadratic),
283    ];
284    let (quotient, remainder) = divide_polynomials(context, numerator, &denominator)?;
285    if !division_certificate(context, numerator, &denominator, &quotient, &remainder)
286        || !quadratic_identity_certificate(
287            context,
288            variable,
289            quadratic,
290            linear,
291            constant,
292            discriminant,
293        )
294    {
295        return None;
296    }
297    let polynomial_part = integrate_polynomial_coefficients(context, &quotient, variable);
298    let remainder_constant = coefficient_or_zero(context, &remainder, 0);
299    let remainder_linear = coefficient_or_zero(context, &remainder, 1);
300    let two = builders::integer_value(context, 2);
301    let two_quadratic = builders::product(context, vec![Arc::clone(&two), Arc::clone(quadratic)]);
302    let logarithm_coefficient = builders::divide(
303        context,
304        Arc::clone(&remainder_linear),
305        Arc::clone(&two_quadratic),
306    );
307    let doubled_quadratic_constant = builders::product(
308        context,
309        vec![two, Arc::clone(quadratic), remainder_constant],
310    );
311    let linear_remainder = builders::product(
312        context,
313        vec![Arc::clone(linear), Arc::clone(&remainder_linear)],
314    );
315    let residual_numerator =
316        builders::subtract(context, doubled_quadratic_constant, linear_remainder);
317    let residual_coefficient = builders::divide(context, residual_numerator, two_quadratic);
318    let denominator_expression = polynomial_expression(context, &denominator, variable);
319    let absolute_denominator = unary_call(context, BuiltinFunction::Abs, denominator_expression);
320    let logarithm = unary_call(context, BuiltinFunction::Log, absolute_denominator);
321    let logarithmic_part = builders::product(context, vec![logarithm_coefficient, logarithm]);
322    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
323    let two = builders::integer_value(context, 2);
324    let doubled_quadratic_variable = builders::product(
325        context,
326        vec![two, Arc::clone(quadratic), variable_expression],
327    );
328    let affine_argument = builders::sum(
329        context,
330        vec![doubled_quadratic_variable, Arc::clone(linear)],
331    );
332    let residual_integral = match branch_kind {
333        QuadraticBranchKind::Positive => {
334            let root = unary_call(context, BuiltinFunction::Sqrt, Arc::clone(discriminant));
335            let argument = builders::divide(context, affine_argument, Arc::clone(&root));
336            let arctangent = unary_call(context, BuiltinFunction::Arctan, argument);
337            let minus_one = builders::integer_value(context, -1);
338            let reciprocal_root = builders::power(context, root, minus_one);
339            let two = builders::integer_value(context, 2);
340            builders::product(context, vec![two, reciprocal_root, arctangent])
341        }
342        QuadraticBranchKind::Negative => {
343            let negated_discriminant = builders::negate(context, Arc::clone(discriminant));
344            let root = unary_call(context, BuiltinFunction::Sqrt, negated_discriminant);
345            let argument = builders::divide(context, affine_argument, Arc::clone(&root));
346            let inverse_hyperbolic_tangent =
347                unary_call(context, BuiltinFunction::Arctanh, argument);
348            let minus_one = builders::integer_value(context, -1);
349            let reciprocal_root = builders::power(context, root, minus_one);
350            let minus_two = builders::integer_value(context, -2);
351            builders::product(
352                context,
353                vec![minus_two, reciprocal_root, inverse_hyperbolic_tangent],
354            )
355        }
356        QuadraticBranchKind::Zero => {
357            let minus_two = builders::integer_value(context, -2);
358            builders::divide(context, minus_two, affine_argument)
359        }
360    };
361    let residual_part = builders::product(context, vec![residual_coefficient, residual_integral]);
362    Some(builders::sum(
363        context,
364        vec![polynomial_part, logarithmic_part, residual_part],
365    ))
366}

368fn integrate_affine_branch(
369    context: &Context,
370    numerator: &[ExpressionReference],
371    variable: SymbolIdentifier,
372    linear: &ExpressionReference,
373    constant: &ExpressionReference,
374) -> Option<ExpressionReference> {
375    let denominator = vec![Arc::clone(constant), Arc::clone(linear)];
376    let (quotient, remainder) = divide_polynomials(context, numerator, &denominator)?;
377    if !division_certificate(context, numerator, &denominator, &quotient, &remainder) {
378        return None;
379    }
380    let polynomial_part = integrate_polynomial_coefficients(context, &quotient, variable);
381    let remainder_constant = coefficient_or_zero(context, &remainder, 0);
382    let affine_expression = polynomial_expression(context, &denominator, variable);
383    let logarithm_argument = if crate::assumptions::range_evidence::proves_real(
384        &context.symbol_table,
385        &affine_expression,
386    ) {
387        unary_call(context, BuiltinFunction::Abs, affine_expression)
388    } else {
389        affine_expression
390    };
391    let logarithm = unary_call(context, BuiltinFunction::Log, logarithm_argument);
392    let logarithm_coefficient = builders::divide(context, remainder_constant, Arc::clone(linear));
393    let logarithmic_part = builders::product(context, vec![logarithm_coefficient, logarithm]);
394    Some(builders::sum(
395        context,
396        vec![polynomial_part, logarithmic_part],
397    ))
398}

415fn extract_rational_polynomial(
416    context: &Context,
417    target: &ExpressionReference,
418    variable: SymbolIdentifier,
419) -> Option<RationalPolynomial> {
420    match &target.kind {
421        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) | ExpressionKind::Constant(_) => {
422            Some(constant_pair(context, target))
423        }
424        ExpressionKind::Symbol(identifier) => {
425            if *identifier == variable {
426                Some(RationalPolynomial {
427                    numerator: vec![builders::zero(context), builders::one(context)],
428                    denominator: vec![builders::one(context)],
429                })
430            } else {
431                Some(constant_pair(context, target))
432            }
433        }
434        ExpressionKind::Sum(children) => {
435            let zero = builders::zero(context);
436            let mut accumulated = constant_pair(context, &zero);
437            for child in children {
438                let child_pair = extract_rational_polynomial(context, child, variable)?;
439                accumulated = add_pairs(context, accumulated, child_pair);
440            }
441            Some(accumulated)
442        }
443        ExpressionKind::Product(children) => {
444            let one = builders::one(context);
445            let mut accumulated = constant_pair(context, &one);
446            for child in children {
447                let child_pair = extract_rational_polynomial(context, child, variable)?;
448                accumulated = multiply_pairs(context, accumulated, child_pair);
449            }
450            Some(accumulated)
451        }
452        ExpressionKind::Power(base, exponent) => {
453            let ExpressionKind::Integer(integer_exponent) = &exponent.kind else {
454                if depends_on(target, variable) {
455                    return None;
456                }
457                return Some(constant_pair(context, target));
458            };
459            let exponent_value = crate::exact_arithmetic::integer_to_i64(integer_exponent)?;
460            if exponent_value.unsigned_abs() > 10 {
461                return None;
462            }
463            let mut pair = extract_rational_polynomial(context, base, variable)?;
464            if exponent_value < 0 {
465                std::mem::swap(&mut pair.numerator, &mut pair.denominator);
466            }
467            let one = builders::one(context);
468            let mut result = constant_pair(context, &one);
469            for _ in 0..exponent_value.unsigned_abs() {
470                result = multiply_pairs(context, result, pair.clone());
471            }
472            Some(result)
473        }
474        _ if !depends_on(target, variable) => Some(constant_pair(context, target)),
475        _ => None,
476    }
477}

486fn add_pairs(
487    context: &Context,
488    left: RationalPolynomial,
489    right: RationalPolynomial,
490) -> RationalPolynomial {
491    let left_numerator = multiply_polynomials(context, &left.numerator, &right.denominator);
492    let right_numerator = multiply_polynomials(context, &right.numerator, &left.denominator);
493    let numerator = add_polynomials(context, left_numerator, right_numerator);
494    let denominator = multiply_polynomials(context, &left.denominator, &right.denominator);
495    RationalPolynomial {
496        numerator,
497        denominator,
498    }
499}

501fn multiply_pairs(
502    context: &Context,
503    left: RationalPolynomial,
504    right: RationalPolynomial,
505) -> RationalPolynomial {
506    RationalPolynomial {
507        numerator: multiply_polynomials(context, &left.numerator, &right.numerator),
508        denominator: multiply_polynomials(context, &left.denominator, &right.denominator),
509    }

512fn add_polynomials(
513    context: &Context,
514    left: Vec<ExpressionReference>,
515    right: Vec<ExpressionReference>,
516) -> Vec<ExpressionReference> {
517    let length = left.len().max(right.len());
518    (0..length)
519        .map(|index| {
520            let left_coefficient = coefficient_or_zero(context, &left, index);
521            let right_coefficient = coefficient_or_zero(context, &right, index);
522            builders::sum(context, vec![left_coefficient, right_coefficient])
523        })
524        .collect()
525}
526
527fn multiply_polynomials(
528    context: &Context,
529    left: &[ExpressionReference],
530    right: &[ExpressionReference],
531) -> Vec<ExpressionReference> {
532    if left.is_empty() || right.is_empty() {
533        return Vec::new();
534    }
535    let mut result = vec![builders::zero(context); left.len() + right.len() - 1];
536    for (left_index, left_coefficient) in left.iter().enumerate() {
537        for (right_index, right_coefficient) in right.iter().enumerate() {
538            let product = builders::product(
539                context,
540                vec![Arc::clone(left_coefficient), Arc::clone(right_coefficient)],
541            );
542            let index = left_index + right_index;
543            result[index] = builders::sum(context, vec![Arc::clone(&result[index]), product]);
544        }
545    }
546    result
547}

549fn divide_polynomials(
550    context: &Context,
551    numerator: &[ExpressionReference],
552    denominator: &[ExpressionReference],
553) -> Option<(Vec<ExpressionReference>, Vec<ExpressionReference>)> {
554    let denominator = trim_polynomial(context, denominator.to_vec());
555    if denominator.is_empty() {
556        return None;
557    }
558    let denominator_degree = denominator.len() - 1;
559    let leading = Arc::clone(denominator.last()?);
560    let mut remainder = trim_polynomial(context, numerator.to_vec());
561    let quotient_length = remainder
562        .len()
563        .saturating_sub(denominator.len())
564        .saturating_add(1);
565    let mut quotient = vec![builders::zero(context); quotient_length];
566    while !remainder.is_empty() && remainder.len() - 1 >= denominator_degree {
567        let degree_difference = remainder.len() - 1 - denominator_degree;
568        let coefficient =
569            builders::divide(context, Arc::clone(remainder.last()?), Arc::clone(&leading));
570        quotient[degree_difference] = Arc::clone(&coefficient);
571        for (index, denominator_coefficient) in denominator.iter().enumerate() {
572            let minus_one = builders::integer_value(context, -1);
573            let subtraction = builders::product(
574                context,
575                vec![
576                    minus_one,
577                    Arc::clone(&coefficient),
578                    Arc::clone(denominator_coefficient),
579                ],
580            );
581            let remainder_index = index + degree_difference;
582            remainder[remainder_index] = builders::sum(
583                context,
584                vec![Arc::clone(&remainder[remainder_index]), subtraction],
585            );
586        }
587        remainder = trim_polynomial(context, remainder);
588    }
589    Some((trim_polynomial(context, quotient), remainder))
590}

592fn division_certificate(
593    context: &Context,
594    numerator: &[ExpressionReference],
595    denominator: &[ExpressionReference],
596    quotient: &[ExpressionReference],
597    remainder: &[ExpressionReference],
598) -> bool {
599    let product = multiply_polynomials(context, quotient, denominator);
600    let recomposed = add_polynomials(context, product, remainder.to_vec());
601    let length = numerator.len().max(recomposed.len());
602    let mut residual = Vec::with_capacity(length);
603    for index in 0..length {
604        let expected = coefficient_or_zero(context, numerator, index);
605        let actual = coefficient_or_zero(context, &recomposed, index);
606        residual.push(builders::subtract(context, actual, expected));
607    }
608    trim_polynomial(context, residual).is_empty()
609}

611fn quadratic_identity_certificate(
612    context: &Context,
613    variable: SymbolIdentifier,
614    quadratic: &ExpressionReference,
615    linear: &ExpressionReference,
616    constant: &ExpressionReference,
617    discriminant: &ExpressionReference,
618) -> bool {
619    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
620    let two = builders::integer_value(context, 2);
621    let doubled_quadratic_variable = builders::product(
622        context,
623        vec![two, Arc::clone(quadratic), variable_expression],
624    );
625    let affine = builders::sum(
626        context,
627        vec![doubled_quadratic_variable, Arc::clone(linear)],
628    );
629    let square_exponent = builders::integer_value(context, 2);
630    let affine_squared = builders::power(context, affine, square_exponent);
631    let left = builders::sum(context, vec![Arc::clone(discriminant), affine_squared]);
632    let four = builders::integer_value(context, 4);
633    let denominator = polynomial_expression(
634        context,
635        &[
636            Arc::clone(constant),
637            Arc::clone(linear),
638            Arc::clone(quadratic),
639        ],
640        variable,
641    );
642    let right = builders::product(context, vec![four, Arc::clone(quadratic), denominator]);
643    let residual = builders::subtract(context, left, right);
644    let expanded = expand(context, &residual);
645    is_zero(context, &expanded)
646}
647
648fn integrate_polynomial_coefficients(
649    context: &Context,
650    coefficients: &[ExpressionReference],
651    variable: SymbolIdentifier,
652) -> ExpressionReference {
653    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
654    let mut terms = Vec::new();
655    for (degree, coefficient) in coefficients.iter().enumerate() {
656        if is_zero(context, coefficient) {
657            continue;
658        }
659        let integrated_degree = degree + 1;
660        let integrated_degree_expression =
661            builders::integer_value(context, integrated_degree as i64);
662        let scaled_coefficient = builders::divide(
663            context,
664            Arc::clone(coefficient),
665            integrated_degree_expression,
666        );
667        let integrated_degree_expression =
668            builders::integer_value(context, integrated_degree as i64);
669        let power = builders::power(
670            context,
671            Arc::clone(&variable_expression),
672            integrated_degree_expression,
673        );
674        terms.push(builders::product(context, vec![scaled_coefficient, power]));
675    }
676    builders::sum(context, terms)
677}
678
679fn polynomial_expression(
680    context: &Context,
681    coefficients: &[ExpressionReference],
682    variable: SymbolIdentifier,
683) -> ExpressionReference {
684    let variable_expression = context.interner.intern(ExpressionKind::Symbol(variable));
685    let mut terms = Vec::new();
686    for (degree, coefficient) in coefficients.iter().enumerate() {
687        if is_zero(context, coefficient) {
688            continue;
689        }
690        if degree == 0 {
691            terms.push(Arc::clone(coefficient));
692        } else {
693            let degree_expression = builders::integer_value(context, degree as i64);
694            let power =
695                builders::power(context, Arc::clone(&variable_expression), degree_expression);
696            terms.push(builders::product(
697                context,
698                vec![Arc::clone(coefficient), power],
699            ));
700        }
701    }
702    builders::sum(context, terms)
703}
704
705fn trim_polynomial(
706    context: &Context,
707    mut coefficients: Vec<ExpressionReference>,
708) -> Vec<ExpressionReference> {
709    while coefficients
710        .last()
711        .is_some_and(|coefficient| is_zero(context, coefficient))
712    {
713        coefficients.pop();
714    }
715    coefficients
716}

718fn coefficient_or_zero(
719    context: &Context,
720    coefficients: &[ExpressionReference],
721    index: usize,
722) -> ExpressionReference {
723    coefficients
724        .get(index)

729fn coefficients_are_rational_functions(
730    coefficients: &[ExpressionReference],
731    parameters: &[SymbolIdentifier],
732) -> bool {
733    coefficients
734        .iter()
735        .all(|coefficient| rational_function_in_parameters(coefficient, parameters))
736}
737
738fn rational_function_in_parameters(
739    target: &ExpressionReference,
740    parameters: &[SymbolIdentifier],
741) -> bool {
742    match &target.kind {
743        ExpressionKind::Integer(_) | ExpressionKind::Rational(_) => true,
744        ExpressionKind::Symbol(identifier) => parameters.contains(identifier),
745        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => children
746            .iter()
747            .all(|child| rational_function_in_parameters(child, parameters)),
748        ExpressionKind::Power(base, exponent) => {
749            matches!(&exponent.kind, ExpressionKind::Integer(_))
750                && rational_function_in_parameters(base, parameters)
751        }
752        _ => false,
753    }
754}
