11pub(crate) fn decide_sign(context: &Context, value: &ExpressionReference) -> SignDecision {
12    // The single rigor chokepoint of the optimization layer (Decision 3): the
13    // sign of a classification minor is decided exactly, never by numeric
14    // evaluation. The whole decision now lives in `exact_real_order`, the one home
15    // of real ordering, and this is a delegation to its context-bearing entry
16    // point: the value is driven toward a literal through the rational normal form
17    // and the simplifier, then decided by the structural rules (literal signs, the
18    // multiplicative sign rule for products and powers, the same-sign rule for
19    // sums), the canonical real-constant form (linear combinations over pi, a
20    // squarefree surd basis and real algebraic leaves, including the exact
21    // `p^2 <=> q^2 r` rule for `p + q*sqrt(r)`), the general real-algebraic sign,
22    // and finally the assumption queries. Whatever none of these proves is
23    // reported `unknown`, which the caller turns into an honest `inconclusive`
24    // verdict rather than a guess. (Decision 3's `sign_at_real_root` sub-case
25    // is subsumed: every coordinate solve::system returns is a rational or a closed-form
26    // radical, and a minor evaluated there is a real algebraic expression of
27    // exactly the shape these exact rules cover; coordinates outside this reach
28    // are declined by solve::system upstream, so they never arrive here.)
29    crate::exact_real_order::sign_of_real_expression_in_context(context, value)
30}

32pub(crate) fn classify_by_minor_signs(
33    context: &Context,
34    bordered_hessian: &Matrix,
35    constraint_count: usize,
36    variable_count: usize,
37) -> ExtremumKind {
38    // The det(H)-keyed Sylvester decision tree (Decision 4), stated once for
39    // both the unconstrained (constraint_count = 0, the matrix is the plain
40    // Hessian) and the equality-constrained (a bordered Hessian) second-order
41    // tests. With m constraints and n variables the matrix is (n + m) square and
42    // the verdict is read from the last n - m of its leading principal minors
43    // D_{2m+1}, ..., D_{n+m}:
44    //   * the test only sees a constrained extremum when there is at least one
45    //     free direction (n > m); otherwise the feasible set is discrete and the
46    //     second-order test is silent (inconclusive);
47    //   * the full determinant D_{n+m} being zero (degenerate Hessian / singular
48    //     projected Hessian) or undecidable is honestly inconclusive — the
49    //     higher-order test is deferred;
50    //   * a strict local minimum is exactly the pattern "every relevant minor has
51    //     the sign of (-1)^m" and a strict local maximum the alternating pattern
52    //     "the order-k minor has the sign of (-1)^{k-m}" (Sylvester's criterion;
53    //     for m = 0 these reduce to all-positive and strictly-alternating);
54    //   * a nonsingular projected Hessian (D_{n+m} != 0) that matches neither
55    //     definite pattern is indefinite on the tangent space, hence a saddle.
56    // Any relevant minor the sign oracle cannot decide forces inconclusive so the
57    // routine never asserts a definiteness it has not proved.
58    let m = constraint_count;
59    let n = variable_count;
60    if n <= m {
61        return ExtremumKind::Inconclusive;
62    }
63    let total = n + m;
64    if bordered_hessian.rows != total || bordered_hessian.columns != total {
65        return ExtremumKind::Inconclusive;
66    }
67    let mut minor_signs: Vec<SignDecision> = Vec::with_capacity(total);
68    for order in 1..=total {
69        let minor = match leading_principal_minor(context, bordered_hessian, order) {
70            Some(value) => value,
71            None => return ExtremumKind::Inconclusive,
72        };
73        minor_signs.push(decide_sign(context, &minor));
74    }
75    let start_order = 2 * m + 1;
76    let determinant_sign = minor_signs[total - 1];
77    if determinant_sign == SignDecision::Zero || determinant_sign == SignDecision::Unknown {
78        return ExtremumKind::Inconclusive;
79    }
80    for order in start_order..=total {
81        if minor_signs[order - 1] == SignDecision::Unknown {
82            return ExtremumKind::Inconclusive;
83        }
84    }
85    let minimum_target = if m % 2 == 0 {
86        SignDecision::Positive
87    } else {
88        SignDecision::Negative
89    };
90    let is_minimum = (start_order..=total).all(|order| minor_signs[order - 1] == minimum_target);
91    if is_minimum {
92        return ExtremumKind::LocalMinimum;
93    }
94    let is_maximum = (start_order..=total).all(|order| {
95        let target = if (order - m) % 2 == 0 {
96            SignDecision::Positive
97        } else {
98            SignDecision::Negative
99        };
100        minor_signs[order - 1] == target
101    });
102    if is_maximum {
103        return ExtremumKind::LocalMaximum;
104    }
105    ExtremumKind::SaddlePoint
106}

108fn leading_principal_minor(
109    context: &Context,
110    source: &Matrix,
111    order: usize,
112) -> Option<ExpressionReference> {
113    // The determinant of the top-left order-by-order submatrix, taken through the
114    // existing division-free Berkowitz determinant so symbolic entries are handled
115    // without ever requiring an exact division in the entry ring.
116    let mut rows: Vec<Vec<ExpressionReference>> = Vec::with_capacity(order);
117    for row_index in 0..order {
118        let mut row: Vec<ExpressionReference> = Vec::with_capacity(order);
119        for column_index in 0..order {
120            row.push(Arc::clone(source.entry(row_index, column_index).ok()?));
121        }
122        rows.push(row);
123    }
124    let submatrix = matrix_from_rows(rows).ok()?;
125    matrix_determinant(context, &submatrix).ok()
126}

129mod tests {

131    use crate::exact_arithmetic::Integer;

135    fn sqrt(context: &Context, radicand: i64) -> ExpressionReference {
136        let base = builders::integer_value(context, radicand);
137        let half = builders::rational(context, Integer::from(1), Integer::from(2));
138        builders::power(context, base, half)
139    }
140
141    #[test]
142    fn structural_sign_decides_negated_surd_products() {
143        let mut context = Context::new();
144        let radical = sqrt(&mut context, 2);
145        let coefficient = builders::integer_value(&mut context, -2);
146        let value = builders::product(&mut context, vec![coefficient, radical]);
147        assert_eq!(decide_sign(&mut context, &value), SignDecision::Negative);
148    }
149
150    #[test]
151    fn surd_sum_sign_is_exact_for_p_plus_q_root_r() {
152        let mut context = Context::new();
153        // 3 - 2*sqrt(2) ~ 0.172 > 0
154        let radical = sqrt(&mut context, 2);
155        let coefficient = builders::integer_value(&mut context, -2);
156        let term = builders::product(&mut context, vec![coefficient, radical]);
157        let constant = builders::integer_value(&mut context, 3);
158        let positive_value = builders::sum(&mut context, vec![constant, term]);
159        assert_eq!(
160            decide_sign(&mut context, &positive_value),
161            SignDecision::Positive
162        );
163
164        // 2 - 2*sqrt(2) ~ -0.828 < 0
165        let radical = sqrt(&mut context, 2);
166        let coefficient = builders::integer_value(&mut context, -2);
167        let term = builders::product(&mut context, vec![coefficient, radical]);
168        let constant = builders::integer_value(&mut context, 2);
169        let negative_value = builders::sum(&mut context, vec![constant, term]);
170        assert_eq!(
171            decide_sign(&mut context, &negative_value),
172            SignDecision::Negative
173        );
174    }
