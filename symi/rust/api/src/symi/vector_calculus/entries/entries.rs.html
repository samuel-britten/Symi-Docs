1// Ordered execution of the independent derivative entries of one vector-calculus operator.
2//
3// Every operator in the domain is a fixed array of partial derivatives over an immutable input:
4// a gradient is one partial per coordinate, a divergence is the diagonal of the jacobian, a curl
5// is six cross partials, and a jacobian is one partial per (component, variable) pair. The entries
6// are independent units. Each is one call into the differentiation substrate over its own
7// component and its own variable, and none of them reads what another produced.
8//
9// The entries borrow the caller's own component and variable lists rather than owning copies of
10// them. An operator states its layout by listing which borrowed component each entry differentiates
11// and with respect to what, and the route is decided from that list before anything is owned, so a
12// serial operator allocates no task input, no logical identity and no per-entry record at all —
13// only the row vectors it publishes.
14//
15// The merge is the entry sequence in the operator's canonical component index order, which for a
16// matrix operator is row-major over (component, variable) and for a vector operator is the caller's
17// own variable order. That order is the operator's published layout, not an artifact of the loop
18// that built it, so it cannot follow completion order: a jacobian whose rows arrived in whichever
19// order the workers finished would be a different matrix.
20//
21// The two routes differ only in which memo the entries share, and the memo follows the route the
22// coordinator selected rather than the mode the operation is running under. An operator that has
23// chosen to run its entries in order cannot fork, so they run through one traversal-local typed
24// memo that lives exactly as long as the operator call: the shared subderivatives of its entries
25// are published once into a map that needs no lock and no type erasure, and the map is dropped with
26// the operator. A parallel operator's entries may run concurrently, so each takes the
27// operation-scoped single-flight memo instead, which publishes each key once and lets every other
28// consumer of that key wait for it rather than recompute it. Both key on the same (expression
29// identity, variable, assumptions, rule version) semantics and both publish the same canonical
30// results, so the route decides what the sharing costs and never what it answers.
31//
32// The entries this coordinator hands out are the ones the operator's contract requires. Mixed
33// partials of a hessian are computed independently rather than being mirrored across the diagonal,
34// because Schwarz commutation holds only where the field is smooth and the substrate's own memo
35// already collapses the pair to one computation wherever it canonically may — a symmetrisation
36// here would instead assert the theorem on inputs that have not earned it.

52/// One requested derivative entry: which expression to differentiate, and with respect to what.
53pub(crate) struct DerivativeEntry<'operator> {
54    /// The component or field the entry differentiates.
55    pub(crate) target: &'operator ExpressionReference,
56    /// The coordinate the entry differentiates with respect to.
57    pub(crate) variable: SymbolIdentifier,
58}
59
60pub(crate) fn derivative_entries(
61    context: &Context,
62    entries: &[DerivativeEntry<'_>],
63) -> Vec<ExpressionReference> {
64    record_counter!(vector_calculus_derivative_entries, entries.len());
65    // Differentiate every requested entry in the operator's canonical index order. No entry
66    // consults a sibling, because the entries of a formed batch may run concurrently and in any
67    // order, and the route is chosen from the borrowed entry list before either backend owns
68    // anything.
69    match derivative_entry_route(context, entries) {
70        ExecutionRoute::Serial => {
71            let mut memo =
72                ExpressionTraversalMemo::for_selected_route(context, ExecutionRoute::Serial);
73            entries
74                .iter()
75                .enumerate()
76                .map(|(index, entry)| {
77                    crate::instrumentation::apply_structured_domain_test_control(
78                        crate::instrumentation::StructuredDomainWorkKind::VectorCalculusDerivativeEntry,
79                        index,
80                    );
81                    differentiate_within_traversal(
82                        context,
83                        entry.target,
84                        entry.variable,
85                        &mut memo,
86                    )
87                })
88                .collect()
89        }
90        ExecutionRoute::Parallel => {
91            let root_identifier =
92                LogicalTaskIdentifier::root(context.operation().identifier().value());
93            ordered_map_with_identifiers(
94                ExecutionRoute::Parallel,
95                entries,
96                &root_identifier,
97                |entry, identifier| {
98                    crate::instrumentation::apply_structured_domain_test_control(
99                        crate::instrumentation::StructuredDomainWorkKind::VectorCalculusDerivativeEntry,
100                        identifier.path().last().copied().unwrap_or(0) as usize,
101                    );
102                    differentiate(context, entry.target, entry.variable)
103                },
104            )
105        }
106    }
107}

109fn derivative_entry_route(context: &Context, entries: &[DerivativeEntry<'_>]) -> ExecutionRoute {
110    // The estimate is the entry count times the total size of the *distinct* expressions the
111    // entries differentiate, and the distinctness is the whole of it.
112    //
113    // The entry count alone says nothing about the cost of one entry: an eight-by-eight jacobian
114    // of coordinate projections is sixty-four constant-time entries, and one of nested
115    // transcendental components is sixty-four whole descents of the differentiation substrate.
116    // But scaling by the widest target is wrong in the other direction, because the memo the
117    // entries share publishes each (subexpression, variable) key once for the whole operator. A
118    // gradient differentiates one shared field n ways, so its n entries traverse one node set and
119    // n - 1 of them are reading back what the first one published; a jacobian differentiates n
120    // distinct components, so its entries traverse n disjoint node sets and every one of them is
121    // real work. Measured at eight workers, that is the difference between a batch that loses at
122    // every width and one that reaches 2.25 times serial. Summing over the distinct targets rather
123    // than taking the widest is what separates them: it counts the work the memo cannot collapse.
124    if entries.len() < 2 {
125        return ExecutionRoute::Serial;
126    }
127    // The distinct-target total below is a traversal of every target the operator differentiates,
128    // so the boundary asks whether the class can fork at all before paying for it. A refusal here
129    // records the same serial batch a refusal by the policy would have recorded.
130    if !context
131        .operation()
132        .may_fork_work_class("vector_calculus_derivative_entry")
133    {
134        record_counter!(vector_calculus_serial_derivative_batches);
135        return ExecutionRoute::Serial;
136    }
137    let distinct_target_nodes = distinct_target_node_total(entries);
138    let worker_count = crate::parallelism::maximum_operation_concurrency();
139    let requested_permits = entries.len().min(worker_count);
140    let work_estimate = WorkEstimate::new(
141        (entries.len() as u64).saturating_mul(distinct_target_nodes.max(1) as u64),
142        entries.len(),
143        entries
144            .len()
145            .saturating_mul(std::mem::size_of::<DerivativeEntry<'_>>()),
146        0,
147        requested_permits,
148        "vector_calculus_derivative_entry",
149    );
150    let decision = context.operation().execution_decision_silent(
151        &work_estimate,
152        requested_permits,
153        runtime_memory_capacity(work_estimate.predicted_task_local_bytes),
154    );
155    match decision.route {
156        ExecutionRoute::Parallel => {
157            record_counter!(vector_calculus_parallel_derivative_batches)
158        }
159        ExecutionRoute::Serial => {
160            record_counter!(vector_calculus_serial_derivative_batches)
161        }
162    }
163    decision.route
164}

166fn distinct_target_node_total(entries: &[DerivativeEntry<'_>]) -> usize {
167    // The summed size of the distinct targets the entries differentiate. Distinctness is decided
168    // by interned pointer identity, which is exact here rather than a heuristic: the canonical
169    // stores give structurally equal expressions one address, and the memo keys the entries share
170    // are keyed on that same identity, so two entries share their traversal exactly when this
171    // counts their target once.
172    let mut counted: HashSet<*const crate::expression::Expression> = HashSet::new();
173    entries
174        .iter()
175        .filter(|entry| counted.insert(Arc::as_ptr(entry.target)))
176        .map(|entry| crate::simplify::heuristic::node_count(entry.target))
177        .fold(0usize, usize::saturating_add)
178}
