70/// Performs the advanced native `lookup_single` operation.
71pub fn lookup_single<'lifetime>(
72    bindings: &'lifetime Bindings,
73    name: &str,
74) -> Option<&'lifetime ExpressionReference> {
75    // Resolve a wildcard name to its single-expression binding. Returns None when
76    // the name is unbound or was bound as a sequence (in which case the caller
77    // wanted the wrong binding shape).
78    match bindings.get(name) {
79        Some(Binding::Single(reference)) => Some(reference),
80        _ => None,
81    }
82}
83
84/// Performs the advanced native `lookup_sequence` operation.
85pub fn lookup_sequence<'lifetime>(
86    bindings: &'lifetime Bindings,
87    name: &str,
88) -> Option<&'lifetime [ExpressionReference]> {
89    // Resolve a wildcard name to its sequence binding. Returns None when the name
90    // is unbound or bound as a single expression.
91    match bindings.get(name) {
92        Some(Binding::Sequence(values)) => Some(values.as_slice()),
93        _ => None,
94    }
95}

97/// Performs the advanced native `pattern_head` operation.
98pub fn pattern_head(value: &Pattern) -> PatternHeadKey {
99    // Compute the head-symbol key used by the discrimination net to bucket rules
100    // by the top-level shape of their left-hand pattern. Wildcard variants share
101    // the catch-all bucket because they may match any shape; literal patterns
102    // forward to the head of the underlying expression.
103    use Pattern::*;
104    match value {
105        Literal(reference) => expression_head(reference),
106        Wildcard(_, _) | SequenceWildcard(_) | RestrictedWildcard(_, _) => PatternHeadKey::Any,
107        Sum(_) => PatternHeadKey::Sum,
108        Product(_) => PatternHeadKey::Product,
109        Power(_, _) => PatternHeadKey::Power,
110        Factorial(_) => PatternHeadKey::Factorial,
111        FunctionCall(function, _) => PatternHeadKey::FunctionCall(*function),
112        Derivative(function, _, _) => PatternHeadKey::Derivative(*function),
113    }
114}

116/// Performs the advanced native `expression_head` operation.
117pub fn expression_head(reference: &ExpressionReference) -> PatternHeadKey {
118    // Map an expression to its head-symbol key. Mirrors pattern_head so the
119    // discrimination net can compare a target against the pre-bucketed rules.
120    use crate::expression::ExpressionKind::*;
121    match &reference.kind {
122        Integer(_) => PatternHeadKey::Integer,
123        Rational(_) => PatternHeadKey::Rational,
124        Symbol(_) => PatternHeadKey::Symbol,
125        Constant(_) => PatternHeadKey::Constant,
126        Sum(_) => PatternHeadKey::Sum,
127        Product(_) => PatternHeadKey::Product,
128        Power(_, _) => PatternHeadKey::Power,
129        Factorial(_) => PatternHeadKey::Factorial,
130        FunctionCall(function, _) => PatternHeadKey::FunctionCall(*function),
131        Derivative(function, _, _) => PatternHeadKey::Derivative(*function),
132        Integral(_, _, _, _) => PatternHeadKey::Integral,
133        Summation(_, _, _, _) => PatternHeadKey::Summation,
134        IndexedProduct(_, _, _, _) => PatternHeadKey::IndexedProduct,
135        LaplaceTransform(_, _, _) => PatternHeadKey::LaplaceTransform,
136        InverseLaplaceTransform(_, _, _) => PatternHeadKey::InverseLaplaceTransform,
137        Piecewise(_) => PatternHeadKey::Piecewise,
138        OrdinaryDifferentialEquation(_, _, _) => PatternHeadKey::OrdinaryDifferentialEquation,
139        RecurrenceEquation(_, _, _) => PatternHeadKey::RecurrenceEquation,
140        _ => PatternHeadKey::RecurrenceEquation,
141    }
142}
