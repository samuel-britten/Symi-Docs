11/// Public native type alias named `rewrite_condition`.
12pub type RewriteCondition = fn(&Bindings, &SymbolTable) -> TruthValue;

14/// Public native data structure named `rewrite_rule`.
15pub struct RewriteRule {
16    /// Stores the `left` value.
17    pub left: Pattern,
18    /// Stores the `right` value.
19    pub right: PatternTemplate,
20    /// Stores the `condition` value.
21    pub condition: Option<RewriteCondition>,
22}
23
24impl RewriteRule {
25    /// Provides the `new` operation on this native type.
26    pub fn new(left: Pattern, right: PatternTemplate) -> Self {
27        // Construct an unconditional rewrite rule. The supplied left and right
28        // patterns must share their wildcard names — every name appearing in the
29        // template must be bound by a successful match against the left side, or
30        // the rule is malformed and the rewrite will silently fail to fire.
31        RewriteRule {
32            left,
33            right,
34            condition: None,
35        }
36    }
37
38    /// Provides the `conditional` operation on this native type.
39    pub fn conditional(left: Pattern, right: PatternTemplate, condition: RewriteCondition) -> Self {
40        // Construct a rewrite rule that fires only when the condition predicate
41        // returns truth_value::yes against the captured bindings. A predicate
42        // returning unknown is treated identically to no — the rule does not
43        // fire, preserving the three-valued logic discipline used throughout.
44        RewriteRule {
45            left,
46            right,
47            condition: Some(condition),
48        }
49    }
50}

52/// Performs the advanced native `rewrite_once` operation.
53pub fn rewrite_once(
54    context: &Context,
55    target: &ExpressionReference,
56    rule: &RewriteRule,
57) -> Option<ExpressionReference> {
58    // Apply a single rule to the top-level expression once. Returns Some with the
59    // instantiated right-hand side when the left pattern matches and any
60    // condition holds; returns None otherwise. This entry point does not recurse
61    // into subexpressions — that is rewrite_bottom_up's responsibility.
62    let bindings = match_pattern(&rule.left, target, &context.symbol_table)?;
63    if let Some(condition) = rule.condition {
64        if condition(&bindings, &context.symbol_table) != TruthValue::Yes {
65            return None;
66        }
67    }
68    instantiate(context, &rule.right, &bindings)
69}

71/// Performs the advanced native `apply_first_matching_rule` operation.
72pub fn apply_first_matching_rule(
73    context: &Context,
74    target: &ExpressionReference,
75    rules: &[RewriteRule],
76) -> Option<ExpressionReference> {
77    // Try each rule in order and return the first successful rewrite. Used by
78    // both the bottom-up traversal and the discrimination net's catch-all path.
79    for rule in rules {
80        if let Some(rewritten) = rewrite_once(context, target, rule) {
81            return Some(rewritten);
82        }
83    }
84    None
85}

87/// Performs the advanced native `apply_first_matching_rule_borrowed` operation.
88pub fn apply_first_matching_rule_borrowed(
89    context: &Context,
90    target: &ExpressionReference,
91    rules: &[&RewriteRule],
92) -> Option<ExpressionReference> {
93    // Borrow-form of apply_first_matching_rule. The discrimination net stores
94    // rules in head-keyed buckets and assembles a flat slice of references for
95    // dispatch; this helper consumes that shape directly without forcing a
96    // clone of every rule.
97    for rule in rules {
98        if let Some(rewritten) = rewrite_once(context, target, rule) {
99            return Some(rewritten);
100        }
101    }
102    None
103}

105/// Performs the advanced native `rewrite_bottom_up` operation.
106pub fn rewrite_bottom_up(
107    context: &Context,
108    target: &ExpressionReference,
109    rules: &[RewriteRule],
110) -> ExpressionReference {
111    // Recursively rewrite an expression from the leaves upward. Children are
112    // rewritten first so that any rule firing at the parent sees the already-
113    // simplified subtree. After children are rebuilt, the rule list is tried at
114    // the current node; the first successful rule returns its rewritten form,
115    // otherwise the rebuilt node is returned unchanged. The traversal always
116    // routes the rebuilt parent through the auto-simplifying builders so the
117    // result remains in ASAE form.
118    rewrite_bottom_up_with(context, target, &|context, target| {
119        apply_first_matching_rule(context, target, rules)
120    })
121}

123/// Performs the advanced native `rewrite_bottom_up_with` operation.
124pub fn rewrite_bottom_up_with<ApplyFunction>(
125    context: &Context,
126    target: &ExpressionReference,
127    apply: &ApplyFunction,
128) -> ExpressionReference
129where
130    ApplyFunction: Fn(&Context, &ExpressionReference) -> Option<ExpressionReference>,
131{
132    // Bottom-up traversal parameterised on the per-node rewrite. Children are
133    // rebuilt first via the same closure so the apply function sees the
134    // already-rewritten subtree at every level. This is the foundation that
135    // both the slice-based rewrite_bottom_up and the discrimination net's
136    // bottom-up entry point share, so the head-indexed lookup does not pay a
137    // double traversal cost.
138    let rebuilt = rewrite_children_with(context, target, apply);
139    apply(context, &rebuilt).unwrap_or(rebuilt)
140}

142/// Performs the advanced native `rewrite_to_fixpoint` operation.
143pub fn rewrite_to_fixpoint(
144    context: &Context,
145    target: &ExpressionReference,
146    rules: &[RewriteRule],
147    max_iterations: usize,
148) -> ExpressionReference {
149    // Repeatedly apply rewrite_bottom_up until the result stops changing or the
150    // maximum iteration count is reached. The interner gives us cheap pointer
151    // equality on identical results so the fixpoint check is O(1) per round.
152    // The iteration cap guards against pathological non-terminating rule sets;
153    // the caller chooses a value appropriate for their domain.
154    let mut current = Arc::clone(target);
155    for _ in 0..max_iterations {
156        let next = rewrite_bottom_up(context, &current, rules);
157        if Arc::ptr_eq(&next, &current) {
158            return current;
159        }
160        current = next;
161    }
162    current
163}

165/// Performs the advanced native `rewrite_to_fixpoint_with` operation.
166pub fn rewrite_to_fixpoint_with<ApplyFunction>(
167    context: &Context,
168    target: &ExpressionReference,
169    apply: &ApplyFunction,
170    max_iterations: usize,
171) -> ExpressionReference
172where
173    ApplyFunction: Fn(&Context, &ExpressionReference) -> Option<ExpressionReference>,
174{
175    // Closure-driven counterpart of rewrite_to_fixpoint. Iterates the bottom-up
176    // traversal until pointer-equal stability is observed or the cap is
177    // exhausted. Used by the discrimination net to honour its head-indexed
178    // dispatch through the same fixpoint loop.
179    let mut current = Arc::clone(target);
180    for _ in 0..max_iterations {
181        let next = rewrite_bottom_up_with(context, &current, apply);
182        if Arc::ptr_eq(&next, &current) {
183            return current;
184        }
185        current = next;
186    }
187    current
188}

190/// Performs the advanced native `instantiate` operation.
191pub fn instantiate(
192    context: &Context,
193    template: &PatternTemplate,
194    bindings: &Bindings,
195) -> Option<ExpressionReference> {
196    // Build a concrete expression from a template by substituting wildcard
197    // bindings. Composite shapes recurse; sum and product templates expand
198    // sequence wildcards inline so a captured tail re-flattens into the parent
199    // operator. Construction always routes through the auto-simplifying builders
200    // so the returned expression is guaranteed to be in ASAE form. Returns None
201    // when the template references a wildcard that the bindings map does not
202    // resolve, or when a sequence wildcard appears outside of an AC operator.
203    match template {
204        Pattern::Literal(reference) => Some(Arc::clone(reference)),
205        Pattern::Wildcard(name, _) | Pattern::RestrictedWildcard(name, _) => {
206            match bindings.get(name)? {
207                Binding::Single(value) => Some(Arc::clone(value)),
208                Binding::Sequence(_) => None,
209            }
210        }
211        Pattern::SequenceWildcard(_) => None,
212        Pattern::Sum(children) => {
213            let expanded = expand_template_children(context, children, bindings)?;
214            Some(sum(context, expanded))
215        }
216        Pattern::Product(children) => {
217            let expanded = expand_template_children(context, children, bindings)?;
218            Some(product(context, expanded))
219        }
220        Pattern::Power(base_template, exponent_template) => {
221            let base = instantiate(context, base_template, bindings)?;
222            let exponent = instantiate(context, exponent_template, bindings)?;
223            Some(power(context, base, exponent))
224        }
225        Pattern::Factorial(argument_template) => {
226            let argument = instantiate(context, argument_template, bindings)?;
227            Some(factorial(context, argument))
228        }
229        Pattern::FunctionCall(function, argument_templates) => {
230            let arguments = expand_template_arguments(context, argument_templates, bindings)?;
231            Some(function_call(context, *function, arguments))
232        }
233        Pattern::Derivative(function, orders, argument_templates) => {
234            let arguments = expand_template_arguments(context, argument_templates, bindings)?;
235            Some(context.interner.intern(ExpressionKind::Derivative(
236                *function,
237                orders.clone(),
238                arguments,
239            )))
240        }
241    }
242}

244fn expand_template_children(
245    context: &Context,
246    templates: &[PatternTemplate],
247    bindings: &Bindings,
248) -> Option<Vec<ExpressionReference>> {
249    // Materialise the children of an AC operator from the template list.
250    // Sequence wildcards expand inline so a binding captured against e.g.
251    // sum([..., sequence_wildcard("rest")]) re-emerges as the original tail of
252    // children at the same nesting level. Single wildcards and literals
253    // contribute exactly one child apiece.
254    let mut materialised: Vec<ExpressionReference> = Vec::new();
255    for child_template in templates {
256        match child_template {
257            Pattern::SequenceWildcard(name) => match bindings.get(name)? {
258                Binding::Sequence(values) => {
259                    materialised.extend(values.iter().map(Arc::clone));
260                }
261                Binding::Single(value) => materialised.push(Arc::clone(value)),
262            },
263            _ => materialised.push(instantiate(context, child_template, bindings)?),
264        }
265    }
266    Some(materialised)
267}
268
269fn expand_template_arguments(
270    context: &Context,
271    templates: &[PatternTemplate],
272    bindings: &Bindings,
273) -> Option<Vec<ExpressionReference>> {
274    // Materialise the argument list of a function_call or derivative template.
275    // These contexts are not associative-commutative; a sequence_wildcard found
276    // here expands but preserves the surrounding fixed-arity contract — callers
277    // are responsible for ensuring the captured length matches the function's
278    // arity. Falling out of bounds simply lets the construction proceed and rely
279    // on later semantic checks.
280    let mut materialised: Vec<ExpressionReference> = Vec::new();
281    for argument_template in templates {
282        match argument_template {
283            Pattern::SequenceWildcard(name) => match bindings.get(name)? {
284                Binding::Sequence(values) => {
285                    materialised.extend(values.iter().map(Arc::clone));
286                }
287                Binding::Single(value) => materialised.push(Arc::clone(value)),
288            },
289            _ => materialised.push(instantiate(context, argument_template, bindings)?),
290        }
291    }
292    Some(materialised)
293}

295fn rewrite_children_with<ApplyFunction>(
296    context: &Context,
297    target: &ExpressionReference,
298    apply: &ApplyFunction,
299) -> ExpressionReference
300where
301    ApplyFunction: Fn(&Context, &ExpressionReference) -> Option<ExpressionReference>,
302{
303    // Closure-driven traversal of the children of a composite node. Each child
304    // is rewritten via rewrite_bottom_up_with (which itself recurses through
305    // this helper) and the parent is reassembled through the auto-simplifying
306    // builders so the result remains in ASAE form. Derivative nodes route
307    // through the interner directly because there is no auto-simplifying
308    // derivative builder — they remain symbolic placeholders by construction.
309    match &target.kind {
310        ExpressionKind::Integer(_)
311        | ExpressionKind::Rational(_)
312        | ExpressionKind::PolynomialRoot(_)
313        | ExpressionKind::Symbol(_)
314        | ExpressionKind::Constant(_) => Arc::clone(target),
315        ExpressionKind::Sum(children) => {
316            let rewritten: Vec<ExpressionReference> = children
317                .iter()
318                .map(|child| rewrite_bottom_up_with(context, child, apply))
319                .collect();
320            sum(context, rewritten)
321        }
322        ExpressionKind::Product(children) => {
323            let rewritten: Vec<ExpressionReference> = children
324                .iter()
325                .map(|child| rewrite_bottom_up_with(context, child, apply))
326                .collect();
327            product(context, rewritten)
328        }
329        ExpressionKind::Power(base, exponent) => {
330            let new_base = rewrite_bottom_up_with(context, base, apply);
331            let new_exponent = rewrite_bottom_up_with(context, exponent, apply);
332            power(context, new_base, new_exponent)
333        }
334        ExpressionKind::Factorial(argument) => {
335            let new_argument = rewrite_bottom_up_with(context, argument, apply);
336            factorial(context, new_argument)
337        }
338        ExpressionKind::FunctionCall(function, arguments) => {
339            let rewritten: Vec<ExpressionReference> = arguments
340                .iter()
341                .map(|argument| rewrite_bottom_up_with(context, argument, apply))
342                .collect();
343            function_call(context, *function, rewritten)
344        }
345        ExpressionKind::Derivative(function, orders, arguments) => {
346            let rewritten: Vec<ExpressionReference> = arguments
347                .iter()
348                .map(|argument| rewrite_bottom_up_with(context, argument, apply))
349                .collect();
350            context.interner.intern(ExpressionKind::Derivative(
351                *function,
352                orders.clone(),
353                rewritten,
354            ))
355        }
356        ExpressionKind::Integral(integrand, variable, lower_bound, upper_bound) => {
357            let rewritten_integrand = rewrite_bottom_up_with(context, integrand, apply);
358            let rewritten_lower = lower_bound
359                .as_ref()
360                .map(|bound| rewrite_bottom_up_with(context, bound, apply));
361            let rewritten_upper = upper_bound
362                .as_ref()
363                .map(|bound| rewrite_bottom_up_with(context, bound, apply));
364            crate::expression::builders::integral(
365                context,
366                rewritten_integrand,
367                *variable,
368                rewritten_lower,
369                rewritten_upper,
370            )
371        }
372        ExpressionKind::Summation(summand, index, lower_bound, upper_bound) => {
373            let rewritten_summand = rewrite_bottom_up_with(context, summand, apply);
374            let rewritten_lower = lower_bound
375                .as_ref()
376                .map(|bound| rewrite_bottom_up_with(context, bound, apply));
377            let rewritten_upper = upper_bound
378                .as_ref()
379                .map(|bound| rewrite_bottom_up_with(context, bound, apply));
380            crate::expression::builders::summation(
381                context,
382                rewritten_summand,
383                *index,
384                rewritten_lower,
385                rewritten_upper,
386            )
387        }
388        ExpressionKind::IndexedProduct(multiplicand, index, lower_bound, upper_bound) => {
389            let rewritten_multiplicand = rewrite_bottom_up_with(context, multiplicand, apply);
390            let rewritten_lower = lower_bound
391                .as_ref()
392                .map(|bound| rewrite_bottom_up_with(context, bound, apply));
393            let rewritten_upper = upper_bound
394                .as_ref()
395                .map(|bound| rewrite_bottom_up_with(context, bound, apply));
396            crate::expression::builders::indexed_product(
397                context,
398                rewritten_multiplicand,
399                *index,
400                rewritten_lower,
401                rewritten_upper,
402            )
403        }
404        ExpressionKind::LaplaceTransform(target_expression, time_variable, frequency_variable) => {
405            let rewritten_target = rewrite_bottom_up_with(context, target_expression, apply);
406            crate::expression::builders::laplace_transform_unevaluated(
407                context,
408                rewritten_target,
409                *time_variable,
410                *frequency_variable,
411            )
412        }
413        ExpressionKind::InverseLaplaceTransform(
414            target_expression,
415            frequency_variable,
416            time_variable,
417        ) => {
418            let rewritten_target = rewrite_bottom_up_with(context, target_expression, apply);
419            crate::expression::builders::inverse_laplace_transform_unevaluated(
420                context,
421                rewritten_target,
422                *frequency_variable,
423                *time_variable,
424            )
425        }
426        ExpressionKind::FourierTransform(target_expression, time_variable, frequency_variable) => {
427            let rewritten_target = rewrite_bottom_up_with(context, target_expression, apply);
428            crate::expression::builders::fourier_transform_unevaluated(
429                context,
430                rewritten_target,
431                *time_variable,
432                *frequency_variable,
433            )
434        }
435        ExpressionKind::InverseFourierTransform(
436            target_expression,
437            frequency_variable,
438            time_variable,
439        ) => {
440            let rewritten_target = rewrite_bottom_up_with(context, target_expression, apply);
441            crate::expression::builders::inverse_fourier_transform_unevaluated(
442                context,
443                rewritten_target,
444                *frequency_variable,
445                *time_variable,
446            )
447        }
448        ExpressionKind::Piecewise(branches) => {
449            let rebuilt: Vec<(ExpressionReference, ExpressionReference)> = branches
450                .iter()
451                .map(|(condition, value)| {
452                    let new_condition = rewrite_bottom_up_with(context, condition, apply);
453                    let new_value = rewrite_bottom_up_with(context, value, apply);
454                    (new_condition, new_value)
455                })
456                .collect();
457            crate::simplify::simplify_piecewise(context, rebuilt)
458        }
459        ExpressionKind::OrdinaryDifferentialEquation(
460            equation,
461            dependent_variable_call,
462            independent_variable,
463        ) => {
464            let rewritten_equation = rewrite_bottom_up_with(context, equation, apply);
465            let rewritten_dependent_call =
466                rewrite_bottom_up_with(context, dependent_variable_call, apply);
467            crate::expression::builders::ordinary_differential_equation(
468                context,
469                rewritten_equation,
470                rewritten_dependent_call,
471                *independent_variable,
472            )
473        }
474        ExpressionKind::RecurrenceEquation(equation, sequence_variable_call, index_variable) => {
475            let rewritten_equation = rewrite_bottom_up_with(context, equation, apply);
476            let rewritten_sequence_call =
477                rewrite_bottom_up_with(context, sequence_variable_call, apply);
478            crate::expression::builders::recurrence_equation(
479                context,
480                rewritten_equation,
481                rewritten_sequence_call,
482                *index_variable,
483            )
484        }
485        ExpressionKind::Interval(lower_bound, upper_bound, lower_inclusive, upper_inclusive) => {
486            let rewritten_lower = rewrite_bottom_up_with(context, lower_bound, apply);
487            let rewritten_upper = rewrite_bottom_up_with(context, upper_bound, apply);
488            crate::sets::builders::interval(
489                context,
490                rewritten_lower,
491                rewritten_upper,
492                *lower_inclusive,
493                *upper_inclusive,
494            )
495        }
496        ExpressionKind::FiniteSet(elements) => {
497            let rewritten = elements
498                .iter()
499                .map(|element| rewrite_bottom_up_with(context, element, apply))
500                .collect();
501            crate::sets::builders::finite_set(context, rewritten)
502        }
503        ExpressionKind::SetUnion(operands) => {
504            let rewritten = operands
505                .iter()
506                .map(|operand| rewrite_bottom_up_with(context, operand, apply))
507                .collect();
508            crate::sets::builders::set_union(context, rewritten)
509        }
510        ExpressionKind::SetIntersection(operands) => {
511            let rewritten = operands
512                .iter()
513                .map(|operand| rewrite_bottom_up_with(context, operand, apply))
514                .collect();
515            crate::sets::builders::set_intersection(context, rewritten)
516        }
517        ExpressionKind::SetComplement(left_operand, right_operand) => {
518            let rewritten_left = rewrite_bottom_up_with(context, left_operand, apply);
519            let rewritten_right = rewrite_bottom_up_with(context, right_operand, apply);
520            crate::sets::builders::set_complement(context, rewritten_left, rewritten_right)
521        }
522        ExpressionKind::ImageSet(parameter, domain, lambda_expression) => {
523            let rewritten_domain = rewrite_bottom_up_with(context, domain, apply);
524            let rewritten_lambda = rewrite_bottom_up_with(context, lambda_expression, apply);
525            crate::sets::builders::image_set(
526                context,
527                *parameter,
528                rewritten_domain,
529                rewritten_lambda,
530            )
531        }
532        ExpressionKind::ConditionSet(parameter, base_set, condition) => {
533            let rewritten_base = rewrite_bottom_up_with(context, base_set, apply);
534            let rewritten_condition = rewrite_bottom_up_with(context, condition, apply);
535            crate::sets::builders::condition_set(
536                context,
537                *parameter,
538                rewritten_base,
539                rewritten_condition,
540            )
541        }
542        ExpressionKind::Distribution(identifier, parameters) => {
543            let rewritten = parameters
544                .iter()
545                .map(|parameter| rewrite_bottom_up_with(context, parameter, apply))
546                .collect();
547            context
548                .interner
549                .intern(ExpressionKind::Distribution(*identifier, rewritten))
550        }
551        ExpressionKind::RandomVariable(identifier, distribution) => {
552            let rewritten_distribution = rewrite_bottom_up_with(context, distribution, apply);
553            context.interner.intern(ExpressionKind::RandomVariable(
554                *identifier,
555                rewritten_distribution,
556            ))
557        }
558        ExpressionKind::SpecialFunctionParameterVector(parameters) => {
559            let rewritten = parameters
560                .iter()
561                .map(|parameter| rewrite_bottom_up_with(context, parameter, apply))
562                .collect();
563            context
564                .interner
565                .intern(ExpressionKind::SpecialFunctionParameterVector(rewritten))
566        }
567    }

653            table: &crate::symbol::SymbolTable,
654        ) -> TruthValue {
655            match crate::pattern::lookup_single(bindings, "x") {
656                Some(reference) => is_positive(reference, table),
657                None => TruthValue::Unknown,
658            }
659        }
660        let pattern_left = Pattern::FunctionCall(
661            FunctionIdentifier::Builtin(BuiltinFunction::Abs),
662            vec![Pattern::Wildcard(name("x"), None)],
663        );
664        let pattern_right = Pattern::Wildcard(name("x"), None);
665        let rule = RewriteRule::conditional(pattern_left, pattern_right, binding_is_positive);
666        // Interned directly (not via the simplifying builder) because the
667        // construction-time fold abs(positive) -> positive would otherwise
668        // collapse the node this test needs the rule to match.
669        let abs_p = context
670            .interner
671            .intern(crate::expression::ExpressionKind::FunctionCall(
672                FunctionIdentifier::Builtin(BuiltinFunction::Abs),
673                vec![Arc::clone(&p)],
674            ));
675        let result_positive = rewrite_once(&mut context, &abs_p, &rule).unwrap();
676        assert!(Arc::ptr_eq(&result_positive, &p));
677        let abs_q = function_call(
678            &mut context,
679            FunctionIdentifier::Builtin(BuiltinFunction::Abs),
680            vec![Arc::clone(&q)],
681        );
682        assert!(rewrite_once(&mut context, &abs_q, &rule).is_none());
683    }
