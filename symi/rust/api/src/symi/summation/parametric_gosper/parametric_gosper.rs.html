1// Zeilberger's creative telescoping with passive parameters (architectural decision 6).
2//
3// Zeilberger seeks polynomials σ_0, …, σ_J — not all zero — and a rational certificate
4// G(n, k) = R(n, k)·F(n, k) satisfying the telescoping identity
5//   Σ_{j=0}^{J} σ_j F(n + j, k) = G(n, k + 1) − G(n, k).
6// Dividing by F(n, k) and writing the consecutive index ratio r = F(n, k + 1)/F(n, k) and the shift
7// ratios ρ_j = F(n + j, k)/F(n, k) gives
8//   Σ_j σ_j ρ_j = R(n, k + 1) r − R(n, k).                                              (★)
9// Multiplying (★) through by the common denominator of the ρ_j keeps the unknown certificate R a
10// polynomial: with r = A/B and ρ_j = P_j/Q_j, set Q = Π_j Q_j, A_G = A·Q, B_G = B·Q(k + 1), and
11// (★) becomes the polynomial identity
12//   R(n, k + 1)·A_G − R(n, k)·B_G = Σ_j σ_j P_j (Q/Q_j) B_G,                            (POLY)
13// linear in the unknown coefficients of the σ_j and of R. Matching every monomial yields one
14// homogeneous linear system; a kernel vector with a non-zero σ-part is a telescoping certificate.
15//
16// The shipped engine required exactly one parameter and solved (POLY) over `Q`. This one admits any
17// number of parameters: one *distinguished* parameter carries the recurrence, and the rest ride
18// along. The question is where those passive parameters live, and the answer here is the polynomial
19// layer, not the coefficient field. Both are complete — a recurrence with rational-function
20// coefficients can always be scaled to clear the passive denominators, so σ and R polynomial in
21// *every* parameter is no restriction — but they behave very differently in practice. Solving over
22// `Q(b, c)` means a multivariate gcd inside every one of the ~20,000 field multiplications a modest
23// system needs, and fraction-free elimination instead makes the entries minors whose term count
24// explodes; on Dixon both ran for hours. Keeping the passive parameters as polynomial *variables*
25// makes every coefficient a plain rational, which is what the corresponding system solves in
26// seconds. The coefficient field is still where Gosper, the term analyzer, and the recurrence
27// closure work — it is only this one linear system that is better posed over `Q`.
28//
29// The certificate is exact by construction — the kernel vector annihilates every monomial equation —
30// and `residual_vanishes` recomputes (POLY) from the reconstructed σ and R to confirm the residual is
31// the zero polynomial, so any asymmetry between assembly and reconstruction turns a flawed candidate
32// into a decline rather than a wrong answer.

66/// Public native data structure named `parameter_split`.
67pub struct ParameterSplit {
68    // How a term's parameters are divided for the recurrence search: exactly one *distinguished*
69    // parameter carries the recurrence, and every other parameter is passive. Both kinds live in the
70    // polynomial layer while the telescoping system is solved; the passive field is what the
71    // recurrence *closure* then works over.
72    /// Stores the `full_field` value.
73    pub full_field: CoefficientField,
74    /// Stores the `passive_field` value.
75    pub passive_field: CoefficientField,
76    /// Stores the `distinguished` value.
77    pub distinguished: SymbolIdentifier,
78    /// Stores the `distinguished_position` value.
79    pub distinguished_position: usize,
80    /// Stores the `passive_position_of_full` value.
81    pub passive_position_of_full: Vec<Option<usize>>,
82}

84/// Performs the advanced native `build_parameter_split` operation.
85pub fn build_parameter_split(
86    full_field: &CoefficientField,
87    distinguished: SymbolIdentifier,
88) -> Option<ParameterSplit> {
89    // Split a parameter set around a distinguished parameter, recording the position map between the
90    // full and passive rings so exponent vectors can be translated without ambiguity.
91    let distinguished_position = full_field.position_of(distinguished)?;
92    let passive_parameters: Vec<SymbolIdentifier> = full_field
93        .parameters()
94        .iter()
95        .copied()
96        .filter(|parameter| *parameter != distinguished)
97        .collect();
98    let passive_field =
99        CoefficientField::with_operation(passive_parameters, full_field.operation().clone());
100    let passive_position_of_full: Vec<Option<usize>> = full_field
101        .parameters()
102        .iter()
103        .map(|parameter| {
104            if *parameter == distinguished {
105                None
106            } else {
107                passive_field.position_of(*parameter)
108            }
109        })
110        .collect();
111    Some(ParameterSplit {
112        full_field: full_field.clone(),
113        passive_field,
114        distinguished,
115        distinguished_position,
116        passive_position_of_full,
117    })
118}

123#[derive(Clone, Debug, PartialEq, Eq, Default)]
124/// Public native data structure named `shift_polynomial`.
125pub struct ShiftPolynomial {
126    // A polynomial in every parameter and the summation index, over the rationals. Keys pair an
127    // exponent vector over the full parameter ring with a degree in the index; no zero coefficient is
128    // stored, so emptiness is the unique zero.
129    /// Stores the `terms` value.
130    pub terms: BTreeMap<MonomialKey, Rational>,
131}
132
133impl ShiftPolynomial {
134    /// Provides the `is_zero` operation on this native type.
135    pub fn is_zero(&self) -> bool {
136        // True exactly when no term is stored.
137        self.terms.is_empty()
138    }
139
140    /// Provides the `degree_in_index` operation on this native type.
141    pub fn degree_in_index(&self) -> usize {
142        // The highest power of the summation index appearing; zero for the zero polynomial.
143        self.terms
144            .keys()
145            .map(|(_, index_degree)| *index_degree)
146            .max()
147            .unwrap_or(0)
148    }
149
150    /// Provides the `total_parameter_degree` operation on this native type.
151    pub fn total_parameter_degree(&self) -> usize {
152        // The highest total degree in the parameters appearing; zero for the zero polynomial.
153        self.terms
154            .keys()
155            .map(|(exponents, _)| total_degree(exponents))
156            .max()
157            .unwrap_or(0)
158    }
159
160    fn insert(&mut self, key: MonomialKey, value: &Rational) {
161        // Accumulate a coefficient at a monomial, removing the entry when the running sum cancels.
162        if *value == Rational::ZERO {
163            return;
164        }
165        let updated = match self.terms.get(&key) {
166            Some(existing) => existing + value,
167            None => value.clone(),
168        };
169        if updated == Rational::ZERO {
170            self.terms.remove(&key);
171        } else {
172            self.terms.insert(key, updated);
173        }
174    }
175}

177fn total_degree(exponents: &ExponentVector) -> usize {
178    // The sum of a monomial's exponents.
179    exponents.iter().map(|value| *value as usize).sum()
180}

182fn multiply_exponents(left: &ExponentVector, right: &ExponentVector) -> ExponentVector {
183    // The exponent vector of a product of two monomials.
184    left.iter()
185        .zip(right.iter())
186        .map(|(first, second)| first + second)
187        .collect()
188}

190fn shift_polynomial_one(arity: usize) -> ShiftPolynomial {
191    // The multiplicative identity.
192    let mut result = ShiftPolynomial::default();
193    result.insert((vec![0; arity], 0), &Rational::ONE);
194    result
195}
196
197fn shift_polynomial_multiply(left: &ShiftPolynomial, right: &ShiftPolynomial) -> ShiftPolynomial {
198    // Product of two polynomials by convolution of their sparse terms.
199    let mut result = ShiftPolynomial::default();
200    for ((left_exponents, left_index), left_value) in &left.terms {
201        for ((right_exponents, right_index), right_value) in &right.terms {
202            let key = (
203                multiply_exponents(left_exponents, right_exponents),
204                left_index + right_index,
205            );
206            result.insert(key, &(left_value * right_value));
207        }
208    }
209    result
210}
211
212fn shift_polynomial_subtract(left: &ShiftPolynomial, right: &ShiftPolynomial) -> ShiftPolynomial {
213    // Difference of two polynomials.
214    let mut result = left.clone();
215    for (key, value) in &right.terms {
216        result.insert(key.clone(), &-value.clone());
217    }
218    result
219}

221fn shift_polynomial_shift_index(value: &ShiftPolynomial, offset: i64) -> ShiftPolynomial {
222    // Substitute `index -> index + offset`, expanding each affected power binomially.
223    if offset == 0 {
224        return value.clone();
225    }
226    let mut result = ShiftPolynomial::default();
227    for ((exponents, index_degree), coefficient) in &value.terms {
228        for (new_degree, factor) in binomial_shift_contributions(*index_degree, offset) {
229            result.insert((exponents.clone(), new_degree), &(coefficient * &factor));
230        }
231    }
232    result
233}

235fn shift_polynomial_from_parameter_polynomial(
236    value: &ParameterPolynomial,
237    index_degree: usize,
238) -> ShiftPolynomial {
239    // Lift a polynomial in the parameters into the (parameters, index) ring at one index degree.
240    let mut result = ShiftPolynomial::default();
241    for (exponents, coefficient) in &value.terms {
242        result.insert((exponents.clone(), index_degree), coefficient);
243    }
244    result
245}

247fn split_rational_function(
248    full: &CoefficientField,
249    value: &IndexRationalFunction,
250) -> Option<(ShiftPolynomial, ShiftPolynomial)> {
251    // Rewrite a rational function of the index over the parameter field as a quotient of two
252    // polynomials in (parameters, index) over the rationals.
253    //
254    // Every coefficient denominator is cleared from numerator and denominator alike, which leaves the
255    // quotient unchanged; a coefficient that still carries a denominator afterwards means the term is
256    // outside the admitted class, and the conversion declines rather than dropping it.
257    let mut clearing = full.parameter_constant(&Rational::ONE);
258    for coefficient in value
259        .numerator
260        .terms
261        .values()
262        .chain(value.denominator.terms.values())
263    {
264        let denominator = full.denominator_of(coefficient);
265        if denominator.as_rational() == Some(Rational::ONE) {
266            continue;
267        }
268        clearing = full.parameter_multiply(&clearing, &denominator);
269    }
270    let clearing_element = full.from_parameter_polynomial(&clearing);
271    let convert =
272        |source: &crate::summation::coefficient_field::FieldPolynomial| -> Option<ShiftPolynomial> {
273            let mut result = ShiftPolynomial::default();
274            for (index_degree, coefficient) in &source.terms {
275                let cleared = full.multiply(coefficient, &clearing_element);
276                if cleared.denominator.as_rational() != Some(Rational::ONE) {
277                    return None;
278                }
279                for (key, value) in
280                    shift_polynomial_from_parameter_polynomial(&cleared.numerator, *index_degree)
281                        .terms
282                {
283                    result.insert(key, &value);
284                }
285            }
286            Some(result)
287        };
288    let numerator = convert(&value.numerator)?;
289    let denominator = convert(&value.denominator)?;
290    if denominator.is_zero() {
291        return None;
292    }
293    Some((numerator, denominator))
294}

331/// Performs the advanced native `find_telescoping_recurrence` operation.
332pub fn find_telescoping_recurrence(
333    context: &Context,
334    term: &HypergeometricTerm,
335    split: &ParameterSplit,
336    acceptor: &dyn Fn(&Context, &[ExpressionReference]) -> bool,
337) -> TelescopingOutcome {
338    // Search for a verified Zeilberger certificate and return the recurrence coefficients.
339    //
340    // The order ladder is climbed to a bound *derived from the term* rather than to a fixed cap: the
341    // index-shift weight below counts how far the term's factorial arguments and rational part can
342    // move along the index, which is the quantity the proper-term existence theorem depends on.
343    // Reaching that bound without an accepted certificate is reported as exhaustion, not as "no
344    // recurrence" — the engine never converts the end of its own search into a mathematical claim.
345    //
346    // A candidate must clear two independent gates: the exact polynomial replay of (POLY), and the
347    // caller's acceptor, which checks the recurrence against the definition of the sum. An exact
348    // telescoping identity whose certificate has a pole inside the support is a true identity that
349    // does not sum correctly, so the search continues past it rather than returning it.
350    let full = &split.full_field;
351    let index_ratio = match index_shift_ratio(full, term, 1) {
352        Some(value) => value,
353        None => {
354            return TelescopingOutcome::NotApplicable(
355                "the consecutive index ratio is not rational over the parameter field".to_string(),
356            );
357        }
358    };
359    let index_ratio = rational_function_reduce(full, &index_ratio.value);
360    let (ratio_numerator, ratio_denominator) = match split_rational_function(full, &index_ratio) {
361        Some(value) => value,
362        None => {
363            return TelescopingOutcome::NotApplicable(
364                "the index ratio does not clear to polynomials over the rationals".to_string(),
365            );
366        }
367    };
368    let derived_order_bound = derived_order_bound(term).min(ORDER_SEARCH_CEILING);
369    let arity = full.arity();
370    let mut shift_numerators: Vec<ShiftPolynomial> = vec![shift_polynomial_one(arity)];
371    let mut shift_denominators: Vec<ShiftPolynomial> = vec![shift_polynomial_one(arity)];
372    let mut last_measurement: Option<ResourceMeasurement> = None;
373    for order in 1..=derived_order_bound {
374        let shift_ratio = match parameter_shift_ratio(
375            full,
376            term,
377            split.distinguished_position,
378            order as i64,
379        ) {
380            Some(value) => value,
381            None => {
382                return TelescopingOutcome::NotApplicable(format!(
383                    "the shift ratio of the distinguished parameter is not rational at order {order}"
384                ));
385            }
386        };
387        let reduced = rational_function_reduce(full, &shift_ratio.value);
388        let (numerator, denominator) = match split_rational_function(full, &reduced) {
389            Some(value) => value,
390            None => {
391                return TelescopingOutcome::NotApplicable(format!(
392                    "the order-{order} shift ratio does not clear to polynomials over the rationals"
393                ));
394            }
395        };
396        shift_numerators.push(numerator);
397        shift_denominators.push(denominator);
398        match search_order(
399            context,
400            split,
401            &ratio_numerator,
402            &ratio_denominator,
403            &shift_numerators,
404            &shift_denominators,
405            acceptor,
406        ) {
407            SearchOutcome::Found(value) => return TelescopingOutcome::Recurrence(value),
408            SearchOutcome::Exhausted(measurement) => last_measurement = Some(measurement),
409            SearchOutcome::Absent => {}
410        }
411    }
412    TelescopingOutcome::Exhausted(last_measurement.unwrap_or(ResourceMeasurement {
413        stage: "parametric_gosper.order_ladder".to_string(),
414        work_units: derived_order_bound as u64,
415        ceiling: ORDER_SEARCH_CEILING as u64,
416        detail: "no accepted certificate up to the derived order bound".to_string(),
417    }))
418}

420fn derived_order_bound(term: &HypergeometricTerm) -> usize {
421    // The order the search climbs to, derived from the term rather than fixed.
422    //
423    // A proper hypergeometric term's recurrence order is controlled by how far its factorial
424    // arguments and rational part can move along the summation index: each factorial argument
425    // contributes the magnitude of its index coefficient, and the rational part contributes its
426    // degree. Summing those gives a term-sensitive ladder height — small for the simple binomial
427    // shapes, which are order one, and larger for terms with several index-carrying factorials —
428    // instead of the shipped fixed cap of four, which was loose for the easy cases and arbitrary for
429    // the hard ones.
430    let factorial_weight: i64 = term
431        .factorial_factors
432        .iter()
433        .map(|factor| factor.argument.index_coefficient.abs() * factor.exponent.abs())
434        .sum();
435    let rational_weight = term
436        .rational_part
437        .numerator
438        .degree()
439        .unwrap_or(0)
440        .max(term.rational_part.denominator.degree().unwrap_or(0));
441    let total = 1 + (factorial_weight.unsigned_abs() as usize) + rational_weight;
442    total.max(1)
443}

445enum SearchOutcome {
446    Found(TelescopingRecurrence),
447    Exhausted(ResourceMeasurement),
448    Absent,
449}

451#[allow(clippy::too_many_arguments)]
452fn search_order(
453    context: &Context,
454    split: &ParameterSplit,
455    ratio_numerator: &ShiftPolynomial,
456    ratio_denominator: &ShiftPolynomial,
457    shift_numerators: &[ShiftPolynomial],
458    shift_denominators: &[ShiftPolynomial],
459    acceptor: &dyn Fn(&Context, &[ExpressionReference]) -> bool,
460) -> SearchOutcome {
461    // For one fixed order, assemble the concrete polynomials of (POLY) once, then grow the degree
462    // ansatz until a kernel vector with a non-zero σ-part appears and verifies.
463    let arity = split.full_field.arity();
464    let order = shift_numerators.len() - 1;
465    let mut common_denominator = shift_polynomial_one(arity);
466    for denominator in shift_denominators {
467        common_denominator = shift_polynomial_multiply(&common_denominator, denominator);
468    }
469    let shifted_common = shift_polynomial_shift_index(&common_denominator, 1);
470    let numerator_certificate = shift_polynomial_multiply(ratio_numerator, &common_denominator);
471    let denominator_certificate = shift_polynomial_multiply(ratio_denominator, &shifted_common);
472    let mut sigma_coefficients: Vec<ShiftPolynomial> = Vec::with_capacity(order + 1);
473    for shift in 0..=order {
474        let mut omitted_product = shift_polynomial_one(arity);
475        for (other_shift, denominator) in shift_denominators.iter().enumerate() {
476            if other_shift != shift {
477                omitted_product = shift_polynomial_multiply(&omitted_product, denominator);
478            }
479        }
480        sigma_coefficients.push(shift_polynomial_multiply(
481            &shift_polynomial_multiply(&shift_numerators[shift], &omitted_product),
482            &denominator_certificate,
483        ));
484    }
485    let concrete_index_degree = numerator_certificate
486        .degree_in_index()
487        .max(denominator_certificate.degree_in_index());
488    // Derive the certificate's starting degree in the index by balancing (POLY). Its left side has
489    // index degree `deg R + max(deg A_G, deg B_G)` and its right side `max_j deg C_j`, so a
490    // certificate of index degree below `max_j deg C_j − max(deg A_G, deg B_G)` cannot balance the
491    // identity, and one much above it only inflates the system. On Dixon that difference is 2, while
492    // the concrete degree alone is 5 — the ansatz the shipped engine would have used builds a system
493    // three times wider than the one that actually works.
494    let sigma_side_index_degree = sigma_coefficients
495        .iter()
496        .map(ShiftPolynomial::degree_in_index)
497        .max()
498        .unwrap_or(0);
499    let base_certificate_index_degree =
500        sigma_side_index_degree.saturating_sub(concrete_index_degree);
501    // Climb the ansatz from the bottom. Every acceptance target currently supported has a σ of total
502    // degree one, so starting at the concrete polynomial degrees — as the shipped engine did — built
503    // systems hundreds of unknowns wide before ever trying the one that works.
504    for budget in 0..=CERTIFICATE_BUDGET_CEILING {
505        let parameter_degree = budget;
506        let certificate_degree_index = base_certificate_index_degree + budget;
507        let parameter_monomials = monomials_up_to_degree(arity, parameter_degree);
508        let sigma_block = parameter_monomials.len() * (order + 1);
509        let certificate_block = parameter_monomials.len() * (certificate_degree_index + 1);
510        let unknown_count = sigma_block + certificate_block;
511        if unknown_count > SYSTEM_UNKNOWN_CEILING {
512            return SearchOutcome::Exhausted(ResourceMeasurement {
513                stage: "parametric_gosper.system_assembly".to_string(),
514                work_units: unknown_count as u64,
515                ceiling: SYSTEM_UNKNOWN_CEILING as u64,
516                detail: format!("order {order}, parameter degree budget {budget}"),
517            });
518        }
519        match attempt_certificate(
520            context,
521            split,
522            &numerator_certificate,
523            &denominator_certificate,
524            &sigma_coefficients,
525            order,
526            &parameter_monomials,
527            certificate_degree_index,
528            acceptor,
529        ) {
530            AttemptOutcome::Found(found) => return SearchOutcome::Found(found),
531            AttemptOutcome::Exhausted(measurement) => {
532                return SearchOutcome::Exhausted(measurement);
533            }
534            AttemptOutcome::Absent => {}
535        }
536    }
537    SearchOutcome::Absent
538}

540fn monomials_up_to_degree(arity: usize, degree: usize) -> Vec<ExponentVector> {
541    // Every exponent vector over the parameters with total degree at most the bound, in a
542    // deterministic order so unknown indices are reproducible across runs.
543    let mut result: Vec<ExponentVector> = Vec::new();
544    let mut current = vec![0u32; arity];
545    enumerate_monomials(arity, degree, 0, &mut current, &mut result);
546    result
547}
548
549fn enumerate_monomials(
550    arity: usize,
551    remaining_degree: usize,
552    position: usize,
553    current: &mut ExponentVector,
554    result: &mut Vec<ExponentVector>,
555) {
556    // Recursive worker for `monomials_up_to_degree`.
557    if position == arity {
558        result.push(current.clone());
559        return;
560    }
561    for exponent in 0..=remaining_degree {
562        current[position] = exponent as u32;
563        enumerate_monomials(
564            arity,
565            remaining_degree - exponent,
566            position + 1,
567            current,
568            result,
569        );
570    }
571    current[position] = 0;
572}

574#[derive(Clone, Debug, Default)]
575struct LinearShiftPolynomial {
576    // A polynomial in (parameters, index) whose coefficients are homogeneous linear forms over the
577    // unknowns: monomial -> (unknown index -> rational coefficient). The residual of (POLY) is built
578    // in this representation, and each monomial's linear form becomes one row of the system.
579    terms: BTreeMap<MonomialKey, BTreeMap<usize, Rational>>,
580}
581
582impl LinearShiftPolynomial {
583    fn insert_form(&mut self, key: MonomialKey, unknown: usize, value: &Rational) {
584        // Accumulate one unknown's coefficient at one monomial, pruning entries that cancel.
585        if *value == Rational::ZERO {
586            return;
587        }
588        let form = self.terms.entry(key.clone()).or_default();
589        let updated = match form.get(&unknown) {
590            Some(existing) => existing + value,
591            None => value.clone(),
592        };
593        if updated == Rational::ZERO {
594            form.remove(&unknown);
595            if form.is_empty() {
596                self.terms.remove(&key);
597            }
598        } else {
599            form.insert(unknown, updated);
600        }
601    }
602
603    fn multiply_by_concrete(&self, concrete: &ShiftPolynomial) -> LinearShiftPolynomial {
604        // Multiply a linear-form-valued polynomial by a concrete polynomial.
605        let mut result = LinearShiftPolynomial::default();
606        for ((left_exponents, left_index), form) in &self.terms {
607            for ((right_exponents, right_index), scalar) in &concrete.terms {
608                let key = (
609                    multiply_exponents(left_exponents, right_exponents),
610                    left_index + right_index,
611                );
612                for (unknown, value) in form {
613                    result.insert_form(key.clone(), *unknown, &(value * scalar));
614                }
615            }
616        }
617        result
618    }
619
620    fn shift_index_by_one(&self) -> LinearShiftPolynomial {
621        // Substitute `index -> index + 1`, distributing each monomial's linear form binomially.
622        let mut result = LinearShiftPolynomial::default();
623        for ((exponents, index_degree), form) in &self.terms {
624            for (new_degree, factor) in binomial_shift_contributions(*index_degree, 1) {
625                let key = (exponents.clone(), new_degree);
626                for (unknown, value) in form {
627                    result.insert_form(key.clone(), *unknown, &(value * &factor));
628                }
629            }
630        }
631        result
632    }
633
634    fn add_linear(&self, other: &LinearShiftPolynomial) -> LinearShiftPolynomial {
635        // Sum of two linear-form-valued polynomials.
636        let mut result = self.clone();
637        for (key, form) in &other.terms {
638            for (unknown, value) in form {
639                result.insert_form(key.clone(), *unknown, value);
640            }
641        }
642        result
643    }
644
645    fn subtract(&self, other: &LinearShiftPolynomial) -> LinearShiftPolynomial {
646        // Difference of two linear-form-valued polynomials.
647        let mut result = self.clone();
648        for (key, form) in &other.terms {
649            for (unknown, value) in form {
650                result.insert_form(key.clone(), *unknown, &-value.clone());
651            }
652        }
653        result
654    }
655
656    fn into_rows(self, unknown_count: usize) -> Vec<Vec<Rational>> {
657        // Materialise the homogeneous system: every monomial's linear form must vanish.
658        let mut rows: Vec<Vec<Rational>> = Vec::with_capacity(self.terms.len());
659        for (_, form) in self.terms {
660            let mut row = vec![Rational::ZERO; unknown_count];
661            for (unknown, value) in form {
662                if unknown < unknown_count {
663                    row[unknown] = value;
664                }
665            }
666            rows.push(row);
667        }
668        rows
669    }
670}

672enum AttemptOutcome {
673    Found(TelescopingRecurrence),
674    Exhausted(ResourceMeasurement),
675    Absent,
676}

678#[allow(clippy::too_many_arguments)]
679fn attempt_certificate(
680    context: &Context,
681    split: &ParameterSplit,
682    numerator_certificate: &ShiftPolynomial,
683    denominator_certificate: &ShiftPolynomial,
684    sigma_coefficients: &[ShiftPolynomial],
685    order: usize,
686    parameter_monomials: &[ExponentVector],
687    certificate_degree_index: usize,
688    acceptor: &dyn Fn(&Context, &[ExpressionReference]) -> bool,
689) -> AttemptOutcome {
690    // Build and solve the homogeneous system of (POLY) for one (order, degree) choice. The unknowns
691    // are the σ_j coefficients followed by the certificate R's coefficients. A kernel vector with a
692    // non-zero σ-part that re-verifies exactly and is accepted by the caller becomes the recurrence.
693    let monomial_count = parameter_monomials.len();
694    let sigma_block = monomial_count * (order + 1);
695    let certificate_block = monomial_count * (certificate_degree_index + 1);
696    let unknown_count = sigma_block + certificate_block;
697    let sigma_index = |shift: usize, monomial: usize| shift * monomial_count + monomial;
698    let certificate_index = |monomial: usize, index_degree: usize| {
699        sigma_block + monomial * (certificate_degree_index + 1) + index_degree
700    };
701
702    let mut sigma_forms = LinearShiftPolynomial::default();
703    for shift in 0..=order {
704        let mut sigma_variable = LinearShiftPolynomial::default();
705        for (monomial, exponents) in parameter_monomials.iter().enumerate() {
706            sigma_variable.insert_form(
707                (exponents.clone(), 0),
708                sigma_index(shift, monomial),
709                &Rational::ONE,
710            );
711        }
712        let contribution = sigma_variable.multiply_by_concrete(&sigma_coefficients[shift]);
713        sigma_forms = sigma_forms.add_linear(&contribution);
714    }
715    let mut certificate = LinearShiftPolynomial::default();
716    for (monomial, exponents) in parameter_monomials.iter().enumerate() {
717        for index_degree in 0..=certificate_degree_index {
718            certificate.insert_form(
719                (exponents.clone(), index_degree),
720                certificate_index(monomial, index_degree),
721                &Rational::ONE,
722            );
723        }
724    }
725    let shifted_certificate = certificate.shift_index_by_one();
726    let certificate_term = shifted_certificate
727        .multiply_by_concrete(numerator_certificate)
728        .subtract(&certificate.multiply_by_concrete(denominator_certificate));
729    let residual = certificate_term.subtract(&sigma_forms);
730    let row_count = residual.terms.len();
731    if row_count > SYSTEM_ROW_CEILING {
732        return AttemptOutcome::Exhausted(ResourceMeasurement {
733            stage: "parametric_gosper.system_assembly".to_string(),
734            work_units: row_count as u64,
735            ceiling: SYSTEM_ROW_CEILING as u64,
736            detail: format!("order {order}, unknowns {unknown_count}"),
737        });
738    }
739    let rows = residual.into_rows(unknown_count);
740    let basis = rational_null_space(&rows, unknown_count);
741    diagnostics::record_note(context, "parametric_gosper.system", || {
742        format!(
743            "order {order}, unknowns {unknown_count}, rows {row_count}, kernel {}",
744            basis.len()
745        )
746    });
747    // The acceptor is the expensive gate — it computes the sum directly at several parameter
748    // assignments — so only a bounded number of candidates from one kernel are put through it. With
749    // the natural boundary already proved by the support certificate, a candidate that replays
750    // exactly is expected to be accepted immediately; a term whose certificate has a pole inside the
751    // support is better served by moving on to the next degree budget than by grinding through
752    // dozens of kernel vectors.
753    let mut acceptor_calls = 0usize;
754    for vector in &basis {
755        if acceptor_calls >= ACCEPTED_CANDIDATE_CEILING {
756            break;
757        }
758        let sigma_polynomials = reconstruct_sigma(vector, order, parameter_monomials, &sigma_index);
759        if sigma_polynomials.iter().all(ShiftPolynomial::is_zero) {
760            continue;
761        }
762        let certificate_polynomial = reconstruct_certificate(
763            vector,
764            parameter_monomials,
765            certificate_degree_index,
766            &certificate_index,
767        );
768        if !residual_vanishes(
769            &sigma_polynomials,
770            &certificate_polynomial,
771            numerator_certificate,
772            denominator_certificate,
773            sigma_coefficients,
774        ) {
775            continue;
776        }
777        let coefficients: Vec<ExpressionReference> = sigma_polynomials
778            .iter()
779            .map(|polynomial| sigma_to_expression(context, split, polynomial))
780            .collect();
781        acceptor_calls += 1;
782        if acceptor(context, &coefficients) {
783            return AttemptOutcome::Found(TelescopingRecurrence {
784                coefficients,
785                sigma_polynomials,
786                order,
787                parameter_degree: parameter_monomials
788                    .iter()
789                    .map(|exponents| total_degree(exponents))
790                    .max()
791                    .unwrap_or(0),
792                certificate_degree_index,
793                unknown_count,
794            });
795        }
796    }
797    AttemptOutcome::Absent
798}

800fn reconstruct_sigma(
801    vector: &[Rational],
802    order: usize,
803    parameter_monomials: &[ExponentVector],
804    sigma_index: &dyn Fn(usize, usize) -> usize,
805) -> Vec<ShiftPolynomial> {
806    // Read the σ_0, …, σ_J polynomials out of a kernel vector.
807    let mut result: Vec<ShiftPolynomial> = Vec::with_capacity(order + 1);
808    for shift in 0..=order {
809        let mut polynomial = ShiftPolynomial::default();
810        for (monomial, exponents) in parameter_monomials.iter().enumerate() {
811            let value = vector[sigma_index(shift, monomial)].clone();
812            polynomial.insert((exponents.clone(), 0), &value);
813        }
814        result.push(polynomial);
815    }
816    result
817}
818
819fn reconstruct_certificate(
820    vector: &[Rational],
821    parameter_monomials: &[ExponentVector],
822    certificate_degree_index: usize,
823    certificate_index: &dyn Fn(usize, usize) -> usize,
824) -> ShiftPolynomial {
825    // Read the certificate polynomial R out of a kernel vector.
826    let mut polynomial = ShiftPolynomial::default();
827    for (monomial, exponents) in parameter_monomials.iter().enumerate() {
828        for index_degree in 0..=certificate_degree_index {
829            let value = vector[certificate_index(monomial, index_degree)].clone();
830            polynomial.insert((exponents.clone(), index_degree), &value);
831        }
832    }
833    polynomial
834}
835
836fn residual_vanishes(
837    sigma_polynomials: &[ShiftPolynomial],
838    certificate: &ShiftPolynomial,
839    numerator_certificate: &ShiftPolynomial,
840    denominator_certificate: &ShiftPolynomial,
841    sigma_coefficients: &[ShiftPolynomial],
842) -> bool {
843    // Recompute (POLY) with the concrete σ and R and confirm the residual is the zero polynomial.
844    // This is the exact certificate replay: a reconstruction or assembly bug can only make it fail,
845    // turning a flawed candidate into a decline rather than a wrong answer.
846    let shifted = shift_polynomial_shift_index(certificate, 1);
847    let mut residual = shift_polynomial_subtract(
848        &shift_polynomial_multiply(&shifted, numerator_certificate),
849        &shift_polynomial_multiply(certificate, denominator_certificate),
850    );
851    for (shift, sigma) in sigma_polynomials.iter().enumerate() {
852        residual = shift_polynomial_subtract(
853            &residual,
854            &shift_polynomial_multiply(sigma, &sigma_coefficients[shift]),
855        );
856    }
857    residual.is_zero()
858}

860fn sigma_to_expression(
861    context: &Context,
862    split: &ParameterSplit,
863    polynomial: &ShiftPolynomial,
864) -> ExpressionReference {
865    // Reify a σ_j — a polynomial in every parameter — as an expression. A term carrying a non-zero
866    // index degree is not expected for a σ coefficient and is skipped defensively.
867    let parameters = split.full_field.parameters().to_vec();
868    let mut terms: Vec<ExpressionReference> = Vec::new();
869    for ((exponents, index_degree), coefficient) in &polynomial.terms {
870        if *index_degree != 0 {
871            continue;
872        }
873        let mut factors: Vec<ExpressionReference> =
874            vec![rational_to_expression(context, coefficient)];
875        for (position, exponent) in exponents.iter().enumerate() {
876            if *exponent == 0 {
877                continue;
878            }
879            let symbol = context
880                .interner
881                .intern(ExpressionKind::Symbol(parameters[position]));
882            let factor = if *exponent == 1 {
883                symbol
884            } else {
885                let power = builders::integer_value(context, *exponent as i64);
886                builders::power(context, symbol, power)
887            };
888            factors.push(factor);
889        }
890        terms.push(builders::product(context, factors));
891    }
892    if terms.is_empty() {
893        return builders::zero(context);
894    }
895    builders::sum(context, terms)
896}
897
898fn rational_to_expression(context: &Context, value: &Rational) -> ExpressionReference {
899    // Convert an exact rational into an interned expression leaf.
900    if *value == Rational::ZERO {
901        return builders::zero(context);
902    }
903    let sign = value.sign_comparison();
904    let (numerator_natural, denominator_natural) = value.to_numerator_and_denominator();
905    let mut numerator = Integer::from(numerator_natural);
906    if sign == std::cmp::Ordering::Less {
907        numerator = -numerator;
908    }
909    let denominator = Integer::from(denominator_natural);
910    builders::rational(context, numerator, denominator)
911}

913/// Performs the advanced native `sigma_as_passive_polynomial` operation.
914pub fn sigma_as_passive_polynomial(
915    split: &ParameterSplit,
916    polynomial: &ShiftPolynomial,
917) -> Option<BTreeMap<usize, ParameterPolynomial>> {
918    // Rewrite a σ coefficient as a polynomial in the distinguished parameter whose coefficients live
919    // in the passive ring — the form the first-order recurrence closure consumes.
920    let mut result: BTreeMap<usize, ParameterPolynomial> = BTreeMap::new();
921    for ((exponents, index_degree), coefficient) in &polynomial.terms {
922        if *index_degree != 0 {
923            return None;
924        }
925        let distinguished_degree = exponents[split.distinguished_position] as usize;
926        let mut passive_exponents = vec![0u32; split.passive_field.arity()];
927        for (position, exponent) in exponents.iter().enumerate() {
928            if position == split.distinguished_position || *exponent == 0 {
929                continue;
930            }
931            let passive_position = split.passive_position_of_full[position]?;
932            passive_exponents[passive_position] = *exponent;
933        }
934        let mut monomial = split.passive_field.polynomial_zero_parameter();
935        monomial
936            .terms
937            .insert(passive_exponents, coefficient.clone());
938        let entry = result
939            .entry(distinguished_degree)
940            .or_insert_with(|| split.passive_field.polynomial_zero_parameter());
941        *entry = split.passive_field.parameter_add(entry, &monomial);
942    }
943    Some(result)
944}
