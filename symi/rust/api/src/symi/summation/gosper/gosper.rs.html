34/// Performs the advanced native `summation_indefinite_hypergeometric` operation.
35pub fn summation_indefinite_hypergeometric(
36    context: &Context,
37    summand: &ExpressionReference,
38    index: SymbolIdentifier,
39) -> Option<ExpressionReference> {
40    // Public entry for the indefinite hypergeometric path. Runs the complete decision procedure
41    // below and keeps only the positive verdict, so the surface behaviour is unchanged while the
42    // internal answer is now typed: a proved-negative decline and a resource-ceiling decline are
43    // different facts, recorded separately, rather than one anonymous `None`.
44    match decide_hypergeometric_antidifference(context, summand, index) {
45        GosperVerdict::Antidifference { value, .. } => Some(value),
46        _ => None,
47    }
48}

50/// Performs the advanced native `summation_indefinite_combined_hypergeometric` operation.
51pub fn summation_indefinite_combined_hypergeometric(
52    context: &Context,
53    summand: &ExpressionReference,
54    index: SymbolIdentifier,
55) -> Option<ExpressionReference> {
56    // Gosper on a top-level sum that combines into one hypergeometric term. Engages only when the
57    // combined term actually carries a factorial or exponential factor: a sum of polynomials or
58    // rational functions also "combines", but the polynomial and Abramov paths produce better forms
59    // for those, so this entry declines and leaves them alone.
60    let analysis = crate::summation::hypergeometric_term::analyse_term(context, summand, index)?;
61    if analysis.term.factorial_factors.is_empty() && analysis.term.exponential_factors.is_empty() {
62        return None;
63    }
64    match decide_hypergeometric_antidifference(context, summand, index) {
65        GosperVerdict::Antidifference { value, .. } => Some(value),
66        _ => None,
67    }
68}

71/// Public native data structure named `gosper_certificate`.
72pub struct GosperCertificate {
73    // The Gosper–Petkovšek data that proves an anti-difference: the normal-form triple (a, b, c) of
74    // the consecutive ratio, the polynomial solution x of the Gosper recurrence, and the resulting
75    // rational multiplier ρ with S(k) = ρ(k)·t(k). The multiplier is the whole certificate in
76    // practice — `ρ(k+1)·r(k) − ρ(k) = 1` is an identity in `Q(parameters)(k)` that can be replayed
77    // from these fields alone.
78    /// Stores the `field` value.
79    pub field: CoefficientField,
80    /// Stores the `a_polynomial` value.
81    pub a_polynomial: FieldPolynomial,
82    /// Stores the `b_polynomial` value.
83    pub b_polynomial: FieldPolynomial,
84    /// Stores the `c_polynomial` value.
85    pub c_polynomial: FieldPolynomial,
86    /// Stores the `x_polynomial` value.
87    pub x_polynomial: FieldPolynomial,
88    /// Stores the `multiplier` value.
89    pub multiplier: IndexRationalFunction,
90    /// Stores the `dispersion_set` value.
91    pub dispersion_set: Vec<i64>,
92}

95/// Public native enumeration named `gosper_verdict`.
96pub enum GosperVerdict {
97    // A verified hypergeometric anti-difference, its certificate, and the conditions the term
98    // normalization assumed.
99    /// Represents the `antidifference` case.
100    Antidifference {
101        /// Stores the `value` value.
102        value: ExpressionReference,
103        /// Stores the `certificate` value.
104        certificate: GosperCertificate,
105        /// Stores the `conditions` value.
106        conditions: ConditionLedger,
107    },
108    // Gosper's decision procedure ran to completion and proved that the admitted term has no
109    // hypergeometric anti-difference in its own term field. This is a mathematical result, not a
110    // failure to find one.
111    /// Represents the `no_hypergeometric_antidifference` case.
112    NoHypergeometricAntidifference {
113        /// Stores the `proof_data` value.
114        proof_data: String,
115    },
116    // The input is not a proper hypergeometric term over its parameter field, so no claim is made.
117    /// Represents the `not_hypergeometric` case.
118    NotHypergeometric,
119    // A resource ceiling stopped the procedure before its mathematical bound was reached.
120    /// Represents the `resource_exhausted` case.
121    ResourceExhausted {
122        /// Stores the `measured_budget` value.
123        measured_budget: ResourceMeasurement,
124    },
125}

131/// Performs the advanced native `decide_hypergeometric_antidifference` operation.
132pub fn decide_hypergeometric_antidifference(
133    context: &Context,
134    summand: &ExpressionReference,
135    index: SymbolIdentifier,
136) -> GosperVerdict {
137    // Gosper's algorithm as a decision procedure over `Q(parameters)(k)` rather than over `Q(k)`.
138    //
139    // Given a term `t(k)` whose consecutive ratio `r(k) = t(k+1)/t(k)` is a rational function of the
140    // index with coefficients in the parameter field, write `r = (a/b)·(c(k+1)/c(k))` in
141    // Gosper–Petkovšek normal form — `gcd(a(k), b(k+j)) = 1` for every non-negative integer `j`,
142    // enforced through a *complete* dispersion set rather than a degree-bounded scan — then decide
143    // whether `a(k)x(k+1) − b(k−1)x(k) = c(k)` has a polynomial solution within the exact degree
144    // bound. Gosper's theorem makes that decision equivalent to the existence of a hypergeometric
145    // anti-difference, which is what lets the negative answer be reported as proved.
146    //
147    // The anti-difference is `S(k) = ρ(k)·t(k)` with `ρ = b(k−1)x(k)/c(k)`, and it is verified twice:
148    // exactly in the coefficient field, then numerically in the expression layer.
149    let analysis =
150        match crate::summation::hypergeometric_term::analyse_term(context, summand, index) {
151            Some(value) => value,
152            None => {
153                diagnostics::record_verdict(
154                    context,
155                    "gosper.term_analysis",
156                    diagnostics::not_applicable("the summand is not a proper hypergeometric term"),
157                );
158                return GosperVerdict::NotHypergeometric;
159            }
160        };
161    let field = analysis.term.field.clone();
162    let ratio = match crate::summation::hypergeometric_term::index_shift_ratio(
163        &field,
164        &analysis.term,
165        1,
166    ) {
167        Some(value) => value,
168        None => {
169            diagnostics::record_verdict(
170                context,
171                "gosper.ratio_reader",
172                diagnostics::not_applicable(
173                    "the consecutive ratio is not a rational function of the index over the parameter field",
174                ),
175            );
176            return GosperVerdict::NotHypergeometric;
177        }
178    };
179    let mut conditions = analysis.conditions.clone();
180    conditions.merge(&ratio.conditions);
181    let reduced_ratio = rational_function_reduce(&field, &ratio.value);
182    if reduced_ratio.numerator.is_zero() {
183        return GosperVerdict::NotHypergeometric;
184    }
185
186    let normal_form = match construct_normal_form(
187        &field,
188        &reduced_ratio.numerator,
189        &reduced_ratio.denominator,
190        index,
191    ) {
192        NormalFormOutcome::Triple(value) => value,
193        NormalFormOutcome::Exhausted(measurement) => {
194            diagnostics::record_verdict(
195                context,
196                "gosper.normal_form",
197                diagnostics::SummationVerdict::ResourceExhausted(measurement.clone()),
198            );
199            return GosperVerdict::ResourceExhausted {
200                measured_budget: measurement,
201            };
202        }
203        NormalFormOutcome::Declined(reason) => {
204            diagnostics::record_verdict(
205                context,
206                "gosper.normal_form",
207                diagnostics::not_applicable(&reason),
208            );
209            return GosperVerdict::NotHypergeometric;
210        }
211    };
212
213    let solution = match solve_gosper_recurrence_over_field(&field, &normal_form) {
214        RecurrenceOutcome::Solution(value) => value,
215        RecurrenceOutcome::Exhausted(measurement) => {
216            diagnostics::record_verdict(
217                context,
218                "gosper.recurrence_solve",
219                diagnostics::SummationVerdict::ResourceExhausted(measurement.clone()),
220            );
221            return GosperVerdict::ResourceExhausted {
222                measured_budget: measurement,
223            };
224        }
225        RecurrenceOutcome::ProvedAbsent(reason) => {
226            diagnostics::record_verdict(
227                context,
228                "gosper.recurrence_solve",
229                diagnostics::proved_impossible(&reason),
230            );
231            return GosperVerdict::NoHypergeometricAntidifference { proof_data: reason };
232        }
233    };
234
235    let b_shifted = field.polynomial_shift_index(&normal_form.b_polynomial, -1);
236    let multiplier = rational_function_reduce(
237        &field,
238        &IndexRationalFunction {
239            numerator: field.polynomial_multiply(&b_shifted, &solution),
240            denominator: normal_form.c_polynomial.clone(),
241        },
242    );
243    if !multiplier_satisfies_gosper_identity(&field, &multiplier, &reduced_ratio) {
244        diagnostics::record_verdict(
245            context,
246            "gosper.certificate_replay",
247            diagnostics::not_applicable(
248                "the reconstructed multiplier failed the exact field identity replay",
249            ),
250        );
251        return GosperVerdict::NotHypergeometric;
252    }
253    let multiplier_expression =
254        crate::summation::hypergeometric_term::rational_function_to_expression(
255            &field,
256            context,
257            &multiplier,
258            index,
259        );
260    let value = builders::product(context, vec![multiplier_expression, Arc::clone(summand)]);
261    if !antidifference_agrees_numerically(context, &value, summand, index, &field) {
262        diagnostics::record_verdict(
263            context,
264            "gosper.expression_replay",
265            diagnostics::not_applicable(
266                "the reified anti-difference disagreed with the summand at a sampled point",
267            ),
268        );
269        return GosperVerdict::NotHypergeometric;
270    }
271    diagnostics::record_note(context, "gosper.antidifference", || {
272        format!(
273            "dispersion {:?}, degree of x {:?}",
274            normal_form.dispersion_set,
275            solution.degree()
276        )
277    });
278    GosperVerdict::Antidifference {
279        value,
280        certificate: GosperCertificate {
281            field,
282            a_polynomial: normal_form.a_polynomial,
283            b_polynomial: normal_form.b_polynomial,
284            c_polynomial: normal_form.c_polynomial,
285            x_polynomial: solution,
286            multiplier,
287            dispersion_set: normal_form.dispersion_set,
288        },
289        conditions,
290    }
291}

294struct GosperNormalForm {
295    // The Gosper–Petkovšek triple over the parameter field, plus the dispersion set that produced
296    // it, retained so the certificate can name the shifts that were extracted.
297    a_polynomial: FieldPolynomial,
298    b_polynomial: FieldPolynomial,
299    c_polynomial: FieldPolynomial,
300    dispersion_set: Vec<i64>,
301}
302
303enum NormalFormOutcome {
304    Triple(GosperNormalForm),
305    Exhausted(ResourceMeasurement),
306    Declined(String),
307}
308
309fn construct_normal_form(
310    field: &CoefficientField,
311    numerator: &FieldPolynomial,
312    denominator: &FieldPolynomial,
313    index: SymbolIdentifier,
314) -> NormalFormOutcome {
315    // Build the Gosper–Petkovšek normal form `numerator/denominator = (a/b)·(c(k+1)/c(k))` with
316    // `gcd(a(k), b(k+j)) = 1` for every non-negative integer `j`.
317    //
318    // The shipped implementation scanned `j` up to `2·(deg a + deg b) + 2`, which is not a bound on
319    // the dispersion at all: `gcd(k, (k + j) − 1000)` is non-trivial only at `j = 1000`, far outside
320    // that window, so a term whose ratio carried a large integer shift silently got the wrong triple
321    // and then failed to close. Here the dispersion set is computed exactly, so the extraction loop
322    // visits every shift that can contribute and no others.
323    let dispersion = match dispersion_set(field, numerator, denominator, index) {
324        DispersionOutcome::Shifts(values) => values,
325        DispersionOutcome::Exhausted(measurement) => {
326            return NormalFormOutcome::Exhausted(measurement);
327        }
328        DispersionOutcome::Declined(reason) => return NormalFormOutcome::Declined(reason),
329    };
330    let mut a_polynomial = numerator.clone();
331    let mut b_polynomial = denominator.clone();
332    let mut c_polynomial = field.polynomial_one();
333    for shift in &dispersion {
334        let shifted_b = field.polynomial_shift_index(&b_polynomial, *shift);
335        let shared = field.polynomial_greatest_common_divisor(&a_polynomial, &shifted_b);
336        if field.polynomial_is_unit(&shared) || shared.is_zero() {
337            continue;
338        }
339        let reduced_a = match field.polynomial_divide_exact(&a_polynomial, &shared) {
340            Some(value) => value,
341            None => {
342                return NormalFormOutcome::Declined("gcd extraction was not exact".to_string());
343            }
344        };
345        let shared_back = field.polynomial_shift_index(&shared, -*shift);
346        let reduced_b = match field.polynomial_divide_exact(&b_polynomial, &shared_back) {
347            Some(value) => value,
348            None => {
349                return NormalFormOutcome::Declined("gcd extraction was not exact".to_string());
350            }
351        };
352        a_polynomial = reduced_a;
353        b_polynomial = reduced_b;
354        for step in 1..=*shift {
355            let shifted_shared = field.polynomial_shift_index(&shared, -step);
356            c_polynomial = field.polynomial_multiply(&c_polynomial, &shifted_shared);
357        }
358    }
359    NormalFormOutcome::Triple(GosperNormalForm {
360        a_polynomial,
361        b_polynomial,
362        c_polynomial,
363        dispersion_set: dispersion,
364    })
365}

367enum DispersionOutcome {
368    Shifts(Vec<i64>),
369    Exhausted(ResourceMeasurement),
370    #[allow(
371        dead_code,
372        reason = "Retains an explicit decline outcome for future dispersion bounds."
373    )]
374    Declined(String),
375}

377fn dispersion_set(
378    field: &CoefficientField,
379    left: &FieldPolynomial,
380    right: &FieldPolynomial,
381    shift_variable: SymbolIdentifier,
382) -> DispersionOutcome {
383    // The complete set of non-negative integers `j` with `deg gcd(left(k), right(k + j)) > 0`.
384    //
385    // `Res_k(left(k), right(k + j))` vanishes exactly at those `j`, and as a polynomial in `j` it has
386    // degree at most `deg left · deg right`, so it is determined by that many consecutive integer
387    // shifts. The whole candidate search runs on a *specialization*: the parameters are pinned to
388    // integers first, so the resultants and the interpolation are plain rational arithmetic. That is
389    // sound in the direction that matters — the resultant is a polynomial in the coefficients, so it
390    // commutes with the evaluation homomorphism once the specialization preserves both degrees, and
391    // a `j` in the true dispersion set therefore always shows up as a root of the specialized
392    // polynomial. Nothing is *believed* on the strength of the specialization: every surviving
393    // candidate is confirmed by an exact gcd over the parameter field, so an over-generous candidate
394    // set costs a little work and never correctness.
395    //
396    // Doing it the other way round — interpolating the resultant over `Q(parameters)` and only then
397    // specializing — is what the first version of this function did, and it cost 2.4 seconds on the
398    // three-parameter Dixon term, essentially all of it in multivariate gcd inside a dense
399    // Vandermonde solve over a rational-function field.
400    let left_degree = match left.degree() {
401        Some(value) if value > 0 => value,
402        _ => return DispersionOutcome::Shifts(Vec::new()),
403    };
404    let right_degree = match right.degree() {
405        Some(value) if value > 0 => value,
406        _ => return DispersionOutcome::Shifts(Vec::new()),
407    };
408    let resultant_degree = left_degree * right_degree;
409    if resultant_degree > DISPERSION_DEGREE_CEILING {
410        return DispersionOutcome::Exhausted(ResourceMeasurement {
411            stage: "gosper.dispersion".to_string(),
412            work_units: resultant_degree as u64,
413            ceiling: DISPERSION_DEGREE_CEILING as u64,
414            detail: "the resultant's degree in the shift variable exceeds the ceiling".to_string(),
415        });
416    }
417    let candidates =
418        match integer_root_candidates(
419            field,
420            left,
421            right,
422            left_degree,
423            right_degree,
424            resultant_degree,
425            shift_variable,
426        ) {
427            Some(value) => value,
428            None => return DispersionOutcome::Exhausted(ResourceMeasurement {
429                stage: "gosper.dispersion".to_string(),
430                work_units: DISPERSION_SPECIALISATION_ATTEMPTS as u64,
431                ceiling: DISPERSION_SPECIALISATION_ATTEMPTS as u64,
432                detail:
433                    "no parameter specialisation preserved both degrees with a non-zero resultant"
434                        .to_string(),
435            }),
436        };
437    let mut confirmed: Vec<i64> = Vec::new();
438    for candidate in candidates {
439        if candidate < 0 {
440            continue;
441        }
442        let shifted = field.polynomial_shift_index(right, candidate);
443        let shared = field.polynomial_greatest_common_divisor(left, &shifted);
444        if !field.polynomial_is_unit(&shared) && !shared.is_zero() {
445            confirmed.push(candidate);
446        }
447    }
448    confirmed.sort_unstable();
449    confirmed.dedup();
450    DispersionOutcome::Shifts(confirmed)
451}

453fn interpolate_over_field(
454    field: &CoefficientField,
455    values: &[FieldElement],
456) -> Option<Vec<FieldElement>> {
457    // Recover the coefficients of the unique polynomial of degree < values.len() taking the given
458    // values at the shift points 0, 1, ..., by solving the Vandermonde system over the field.
459    let count = values.len();
460    let mut rows: Vec<Vec<FieldElement>> = Vec::with_capacity(count);
461    for point in 0..count {
462        let mut row: Vec<FieldElement> = Vec::with_capacity(count);
463        let mut power = field.one();
464        let point_element = field.from_integer(point as i64);
465        for _ in 0..count {
466            row.push(power.clone());
467            power = field.multiply(&power, &point_element);
468        }
469        rows.push(row);
470    }
471    field.solve_linear_system(&rows, values, count)
472}

474fn specialise_polynomial(
475    field: &CoefficientField,
476    value: &FieldPolynomial,
477    point: &[Rational],
478    trivial_field: &CoefficientField,
479    expected_degree: usize,
480) -> Option<FieldPolynomial> {
481    // Evaluate every coefficient of an index polynomial at an integer parameter point, producing the
482    // same polynomial over the rationals. Returns None when a coefficient's denominator vanishes at
483    // the point or the leading coefficient does, because a dropped degree would break the identity
484    // `Res(specialise(f), specialise(g)) = specialise(Res(f, g))` the candidate search relies on.
485    let mut result = trivial_field.polynomial_zero();
486    for (degree, coefficient) in &value.terms {
487        let evaluated = field.evaluate(coefficient, point)?;
488        result = trivial_field.polynomial_add(
489            &result,
490            &trivial_field.polynomial_monomial(*degree, &trivial_field.from_rational(&evaluated)),
491        );
492    }
493    if result.degree() != Some(expected_degree) {
494        return None;
495    }
496    Some(result)
497}

499#[allow(
500    clippy::too_many_arguments,
501    reason = "The dispersion coordinator passes the fixed polynomial-search inputs together."
502)]
503fn integer_root_candidates(
504    field: &CoefficientField,
505    left: &FieldPolynomial,
506    right: &FieldPolynomial,
507    left_degree: usize,
508    right_degree: usize,
509    resultant_degree: usize,
510    shift_variable: SymbolIdentifier,
511) -> Option<Vec<i64>> {
512    // Over-approximate the dispersion set by isolating the real roots of the specialized shift
513    // resultant. Returns None when no attempted specialization both preserves the two degrees and
514    // leaves the resultant non-zero as a polynomial in the shift.
515    let trivial_field = CoefficientField::new(Vec::new());
516    let attempts: Vec<i64> = (0..DISPERSION_SPECIALISATION_ATTEMPTS).collect();
517    let execution_route = if field
518        .operation()
519        .may_fork_work_class("summation_parameter_sample")
520    {
521        let work_estimate = WorkEstimate::new(
522            (attempts.len() as u64).saturating_mul((resultant_degree as u64).saturating_add(1)),
523            attempts.len(),
524            attempts.len().saturating_mul(
525                (resultant_degree + 1).saturating_mul(std::mem::size_of::<FieldElement>()),
526            ),
527            0,
528            attempts.len(),
529            "summation_parameter_sample",
530        );
531        field
532            .operation()
533            .execution_decision_silent(&work_estimate, attempts.len(), None)
534            .route
535    } else {
536        crate::parallelism::ExecutionRoute::Serial
537    };
538    ordered_find_first(execution_route, &attempts, |attempt| {
539        integer_root_candidates_for_sample(
540            field,
541            &trivial_field,
542            left,
543            right,
544            left_degree,
545            right_degree,
546            resultant_degree,
547            shift_variable,
548            *attempt,
549        )
550    })
551}

553#[allow(
554    clippy::too_many_arguments,
555    reason = "Each parameter-sample task receives the fixed inputs needed for exact reconstruction."
556)]
557fn integer_root_candidates_for_sample(
558    field: &CoefficientField,
559    trivial_field: &CoefficientField,
560    left: &FieldPolynomial,
561    right: &FieldPolynomial,
562    left_degree: usize,
563    right_degree: usize,
564    resultant_degree: usize,
565    shift_variable: SymbolIdentifier,
566    attempt: i64,
567) -> Option<Vec<i64>> {
568    // Evaluate one preassigned parameter sample and derive an over-approximation of the integer
569    // dispersion roots. Exact confirmation remains in `dispersion_set`, so this task only orders
570    // candidate generation and never turns a sample into proof.
571    let point: Vec<Rational> = (0..field.arity())
572        .map(|position| {
573            Rational::from(Integer::from(
574                3 + attempt * 7 + position as i64 * (attempt + 2),
575            ))
576        })
577        .collect();
578    let left_specialised = specialise_polynomial(field, left, &point, trivial_field, left_degree)?;
579    let right_specialised =
580        specialise_polynomial(field, right, &point, trivial_field, right_degree)?;
581    let mut sample_values: Vec<FieldElement> = Vec::with_capacity(resultant_degree + 1);
582    for shift in 0..=resultant_degree {
583        let shifted = trivial_field.polynomial_shift_index(&right_specialised, shift as i64);
584        sample_values.push(trivial_field.polynomial_resultant(&left_specialised, &shifted));
585    }
586    if sample_values.iter().all(FieldElement::is_zero) {
587        return None;
588    }
589    let coefficients = interpolate_over_field(trivial_field, &sample_values)?;
590    let mut specialised: Vec<number_value> = Vec::with_capacity(coefficients.len());
591    for coefficient in &coefficients {
592        specialised.push(rational_to_number(coefficient.as_rational()?));
593    }
594    if specialised.iter().all(number::is_zero) {
595        return None;
596    }
597    let specialised_polynomial = univariate_from_dense(shift_variable, specialised);
598    let intervals = crate::polynomial::real_roots::isolate_real_roots(&specialised_polynomial)?;
599    let quarter =
600        crate::exact_arithmetic::rational_from_integers(Integer::from(1), Integer::from(4))
601            .unwrap_or(Rational::ONE);
602    let mut candidates: Vec<i64> = Vec::new();
603    for interval in &intervals {
604        let refined = crate::polynomial::real_roots::refine_isolating_interval(
605            &specialised_polynomial,
606            interval,
607            &quarter,
608        );
609        for bound in [&refined.lower, &refined.upper] {
610            let floor_value = crate::exact_arithmetic::rational_floor(bound);
611            if let Ok(native) = i64::try_from(&floor_value) {
612                for offset in [0i64, 1] {
613                    let candidate = native + offset;
614                    if !candidates.contains(&candidate) {
615                        candidates.push(candidate);
616                    }
617                }
618            }
619        }
620    }
621    Some(candidates)
622}

624enum RecurrenceOutcome {
625    Solution(FieldPolynomial),
626    Exhausted(ResourceMeasurement),
627    ProvedAbsent(String),
628}
629
630fn solve_gosper_recurrence_over_field(
631    field: &CoefficientField,
632    normal_form: &GosperNormalForm,
633) -> RecurrenceOutcome {
634    // Decide whether `a(k)x(k+1) − b(k−1)x(k) = c(k)` has a polynomial solution, and return one.
635    //
636    // The degree bound is the exact one of Petkovšek–Wilf–Zeilberger §5.3. With
637    // `p = a + b(k−1)` and `q = a − b(k−1)`, the identity
638    //   a(k)x(k+1) − b(k−1)x(k) = ½[p·(x(k+1) − x(k)) + q·(x(k+1) + x(k))]
639    // makes the leading behaviour explicit: for `deg x = d ≥ 1` the top degree is
640    // `max(deg p − 1, deg q) + d`, giving `d = deg c − max(deg p − 1, deg q)`, *unless*
641    // `deg q = deg p − 1`, where the two leading terms can cancel and the extra root
642    // `d = −2·μ/λ` (λ the leading coefficient of p, μ the coefficient of `k^(deg p − 1)` in q)
643    // becomes possible. Taking the maximum of both, and always admitting `d = 0` — where the
644    // difference `x(k+1) − x(k)` vanishes and the leading analysis does not apply — covers every
645    // solvable degree, so exhausting the search is a proof of absence.
646    let a_polynomial = &normal_form.a_polynomial;
647    let b_shifted = field.polynomial_shift_index(&normal_form.b_polynomial, -1);
648    let c_polynomial = &normal_form.c_polynomial;
649    let sum_polynomial = field.polynomial_add(a_polynomial, &b_shifted);
650    let difference_polynomial = field.polynomial_subtract(a_polynomial, &b_shifted);
651    let degree_of = |value: &FieldPolynomial| -> i64 {
652        value.degree().map(|degree| degree as i64).unwrap_or(-1)
653    };
654    let degree_sum = degree_of(&sum_polynomial);
655    let degree_difference = degree_of(&difference_polynomial);
656    let degree_c = degree_of(c_polynomial);
657    let standard_bound = degree_c - (degree_sum - 1).max(degree_difference);
658    let mut degree_bound = standard_bound.max(0);
659    if degree_sum >= 1 && degree_difference == degree_sum - 1 {
660        if let (Some(leading_sum), Some(leading_difference)) = (
661            sum_polynomial.leading_coefficient(),
662            difference_polynomial.leading_coefficient(),
663        ) {
664            let doubled = field.multiply(&field.from_integer(-2), leading_difference);
665            if let Some(quotient) = field.divide(&doubled, leading_sum) {
666                if let Some(candidate) = quotient.as_integer() {
667                    if candidate > degree_bound {
668                        degree_bound = candidate;
669                    }
670                }
671            }
672        }
673    }
674    if degree_bound > ANTIDIFFERENCE_DEGREE_CEILING {
675        return RecurrenceOutcome::Exhausted(ResourceMeasurement {
676            stage: "gosper.recurrence_solve".to_string(),
677            work_units: degree_bound as u64,
678            ceiling: ANTIDIFFERENCE_DEGREE_CEILING as u64,
679            detail: "the exact degree bound for the anti-difference exceeds the ceiling"
680                .to_string(),
681        });
682    }
683    let unknown_count = (degree_bound as usize) + 1;
684    let mut contributions: Vec<FieldPolynomial> = Vec::with_capacity(unknown_count);
685    let mut highest_degree = degree_c.max(0) as usize;
686    for degree in 0..unknown_count {
687        let basis = field.polynomial_monomial(degree, &field.one());
688        let shifted_basis = field.polynomial_shift_index(&basis, 1);
689        let contribution = field.polynomial_subtract(
690            &field.polynomial_multiply(a_polynomial, &shifted_basis),
691            &field.polynomial_multiply(&b_shifted, &basis),
692        );
693        if let Some(degree_value) = contribution.degree() {
694            highest_degree = highest_degree.max(degree_value);
695        }
696        contributions.push(contribution);
697    }
698    let row_count = highest_degree + 1;
699    let mut rows: Vec<Vec<FieldElement>> = vec![vec![field.zero(); unknown_count]; row_count];
700    for (column, contribution) in contributions.iter().enumerate() {
701        for (degree, coefficient) in &contribution.terms {
702            rows[*degree][column] = coefficient.clone();
703        }
704    }
705    let mut right_hand_side: Vec<FieldElement> = vec![field.zero(); row_count];
706    for (degree, coefficient) in &c_polynomial.terms {
707        right_hand_side[*degree] = coefficient.clone();
708    }
709    let solution = match field.solve_linear_system(&rows, &right_hand_side, unknown_count) {
710        Some(value) => value,
711        None => {
712            return RecurrenceOutcome::ProvedAbsent(format!(
713                "the Gosper recurrence has no polynomial solution; the exact degree bound is {standard_bound} \
714                 from the leading analysis and the search covered every degree up to {degree_bound}"
715            ));
716        }
717    };
718    let mut result = field.polynomial_zero();
719    for (degree, coefficient) in solution.iter().enumerate() {
720        result = field.polynomial_add(&result, &field.polynomial_monomial(degree, coefficient));
721    }
722    if result.is_zero() && !c_polynomial.is_zero() {
723        return RecurrenceOutcome::ProvedAbsent(
724            "the only solution of the Gosper recurrence is zero, which cannot match a non-zero c"
725                .to_string(),
726        );
727    }
728    RecurrenceOutcome::Solution(result)
729}

731fn multiplier_satisfies_gosper_identity(
732    field: &CoefficientField,
733    multiplier: &IndexRationalFunction,
734    ratio: &IndexRationalFunction,
735) -> bool {
736    // Replay the defining identity exactly in the coefficient field: with `S(k) = ρ(k)·t(k)` and
737    // `r(k) = t(k+1)/t(k)`, the requirement `S(k+1) − S(k) = t(k)` is equivalent to
738    // `ρ(k+1)·r(k) − ρ(k) = 1` as rational functions. This is a proof, not a sample: a reconstruction
739    // slip anywhere upstream can only make it fail, turning a flawed candidate into a decline.
740    let shifted = rational_function_shift_index(field, multiplier, 1);
741    let product = rational_function_multiply(field, &shifted, ratio);
742    let residual = rational_function_reduce(
743        field,
744        &rational_function_add(
745            field,
746            &product,
747            &rational_function_multiply(
748                field,
749                multiplier,
750                &rational_function_from_polynomial(
751                    field,
752                    &field.polynomial_constant(&field.from_integer(-1)),
753                ),
754            ),
755        ),
756    );
757    residual == rational_function_reduce(field, &rational_function_one(field))
758}

760fn antidifference_agrees_numerically(
761    context: &Context,
762    antidifference: &ExpressionReference,
763    summand: &ExpressionReference,
764    index: SymbolIdentifier,
765    field: &CoefficientField,
766) -> bool {
767    // The expression-layer half of the verification: sample the discrete fundamental theorem
768    // `S(k+1) − S(k) = t(k)` at integer points. The field identity above is the proof; this is the
769    // independent second net that catches a reification or interning mistake, which no amount of
770    // field algebra would notice. Samples that fail to evaluate are skipped; a sample that evaluates
771    // and *disagrees* rejects the result outright.
772    let mut agreements = 0u32;
773    for (index_value, parameter_base) in [(2i64, 9i64), (3, 11), (1, 13)] {
774        let mut bindings: SubstitutionMap = HashMap::new();
775        for (position, parameter) in field.parameters().iter().enumerate() {
776            let symbol = context.interner.intern(ExpressionKind::Symbol(*parameter));
777            let value = builders::integer_value(context, parameter_base + position as i64);
778            bindings.insert(symbol, value);
779        }
780        let index_symbol = context.interner.intern(ExpressionKind::Symbol(index));
781        let at_index = builders::integer_value(context, index_value);
782        let at_next = builders::integer_value(context, index_value + 1);
783        let mut lower_bindings = bindings.clone();
784        lower_bindings.insert(Arc::clone(&index_symbol), at_index);
785        let mut upper_bindings = bindings;
786        upper_bindings.insert(Arc::clone(&index_symbol), at_next);
787        let at_lower = substitute(context, antidifference, &lower_bindings);
788        let at_upper = substitute(context, antidifference, &upper_bindings);
789        let summand_value = substitute(context, summand, &lower_bindings);
790        let difference = builders::subtract(context, at_upper, at_lower);
791        let (Some(difference_value), Some(expected_value)) = (
792            crate::numeric::evaluate_to_float(context, &difference),
793            crate::numeric::evaluate_to_float(context, &summand_value),
794        ) else {
795            continue;
796        };
797        if !difference_value.is_finite() || !expected_value.is_finite() {
798            continue;
799        }
800        let scale = expected_value.abs().max(1.0);
801        if (difference_value - expected_value).abs() > 1e-7 * scale {
802            return false;
803        }
804        agreements += 1;
805    }
806    agreements >= 1
807}

809pub(crate) fn expression_to_rational_pair(
810    expression: &ExpressionReference,
811    index: SymbolIdentifier,
812    operation: &OperationContext,
813) -> Option<(Polynomial, Polynomial)> {
814    // Convert an expression into a (numerator, denominator) polynomial
815    // pair in `index`. Recurses through sums (cross-multiply),
816    // products (independent multiply), and powers with integer
817    // exponents (repeated multiply or inversion). Returns None
818    // whenever the structure is not closed within Q(index).
819    rational_from_node(expression, index, operation)
820}
821
822fn rational_from_node(
823    expression: &ExpressionReference,
824    index: SymbolIdentifier,
825    operation: &OperationContext,
826) -> Option<(Polynomial, Polynomial)> {
827    use ExpressionKind::*;
828    match &expression.kind {
829        Integer(_) | Rational(_) | PolynomialRoot(_) | Symbol(_) | Constant(_) => {
830            let numerator_polynomial = expression_to_univariate(expression, index)?;
831            let denominator_polynomial = one_polynomial(index);
832            Some((numerator_polynomial, denominator_polynomial))
833        }
834        Sum(children) => {
835            let mut current_numerator = zero_polynomial(index);
836            let mut current_denominator = one_polynomial(index);
837            for child in children {
838                let (child_numerator, child_denominator) =
839                    rational_from_node(child, index, operation)?;
840                let new_numerator = polynomial_add(
841                    &polynomial_multiply(&current_numerator, &child_denominator),
842                    &polynomial_multiply(&child_numerator, &current_denominator),
843                );
844                let new_denominator = polynomial_multiply(&current_denominator, &child_denominator);
845                current_numerator = new_numerator;
846                current_denominator = new_denominator;
847            }
848            Some(reduce_rational_pair(
849                current_numerator,
850                current_denominator,
851                operation,
852            ))
853        }
854        Product(children) => {
855            let mut current_numerator = one_polynomial(index);
856            let mut current_denominator = one_polynomial(index);
857            for child in children {
858                let (child_numerator, child_denominator) =
859                    rational_from_node(child, index, operation)?;
860                current_numerator = polynomial_multiply(&current_numerator, &child_numerator);
861                current_denominator = polynomial_multiply(&current_denominator, &child_denominator);
862            }
863            Some(reduce_rational_pair(
864                current_numerator,
865                current_denominator,
866                operation,
867            ))
868        }
869        Power(base, exponent) => {
870            let exponent_value = match &exponent.kind {
871                ExpressionKind::Integer(value) => value.clone(),
872                _ => return None,
873            };
874            let exponent_signed = i64::try_from(&exponent_value).ok()?;
875            let (base_numerator, base_denominator) = rational_from_node(base, index, operation)?;
876            if exponent_signed >= 0 {
877                let absolute_exponent = exponent_signed as usize;
878                let numerator_result = polynomial_pow(&base_numerator, absolute_exponent);
879                let denominator_result = polynomial_pow(&base_denominator, absolute_exponent);
880                Some(reduce_rational_pair(
881                    numerator_result,
882                    denominator_result,
883                    operation,
884                ))
885            } else {
886                let absolute_exponent = (-exponent_signed) as usize;
887                let numerator_result = polynomial_pow(&base_denominator, absolute_exponent);
888                let denominator_result = polynomial_pow(&base_numerator, absolute_exponent);
889                if polynomial_is_zero(&denominator_result) {
890                    return None;
891                }
892                Some(reduce_rational_pair(
893                    numerator_result,
894                    denominator_result,
895                    operation,
896                ))
897            }
898        }
899        _ => None,
900    }
901}

903fn reduce_rational_pair(
904    numerator: Polynomial,
905    denominator: Polynomial,
906    operation: &OperationContext,
907) -> (Polynomial, Polynomial) {
908    // Reduce a rational pair to lowest terms via polynomial-gcd
909    // cancellation. The denominator stays nonzero so a zero numerator
910    // can be safely interpreted as the literal zero rational.
911    if polynomial_is_zero(&numerator) {
912        return (numerator, denominator);
913    }
914    let shared = polynomial_gcd_with_operation(&numerator, &denominator, operation);
915    let reduced_numerator = divide_exact(&numerator, &shared).unwrap_or(numerator.clone());
916    let reduced_denominator = divide_exact(&denominator, &shared).unwrap_or(denominator.clone());
917    (reduced_numerator, reduced_denominator)
918}

920pub(crate) fn shift_polynomial(
921    value: &Polynomial,
922    index: SymbolIdentifier,
923    delta: i64,
924) -> Polynomial {
925    // Compose value(k) → value(k + delta). Computed term by term by
926    // expanding each (k + delta)^exponent via the binomial theorem and
927    // summing the contributions. Only nonzero numeric coefficients are
928    // supported here — nested coefficients are passed through verbatim
929    // because the algebra of nested coefficients commutes with the
930    // outer-variable shift.
931    let mut accumulator = zero_polynomial(index);
932    for (exponent, coefficient) in &value.terms {
933        let shift_expansion = expand_shifted_power(*exponent, delta);
934        for (basis_exponent, basis_integer_value) in shift_expansion {
935            let basis_coefficient_value = match coefficient {
936                PolynomialCoefficient::Numeric(numeric_value) => {
937                    PolynomialCoefficient::Numeric(number::multiply(
938                        numeric_value,
939                        &number_value::Integer(basis_integer_value.clone()),
940                    ))
941                }
942                PolynomialCoefficient::Nested(nested_value) => {
943                    PolynomialCoefficient::Nested(Box::new(crate::polynomial::scale_polynomial(
944                        nested_value,
945                        &number_value::Integer(basis_integer_value.clone()),
946                    )))
947                }
948                PolynomialCoefficient::Algebraic(algebraic_value) => {
949                    PolynomialCoefficient::Algebraic(
950                        crate::algebraic_numbers::arithmetic::scale_by_rational(
951                            algebraic_value,
952                            &crate::exact_arithmetic::rational_from_integer(
953                                basis_integer_value.clone(),
954                            ),
955                        ),
956                    )
957                }
958            };
959            let term_polynomial =
960                single_term_polynomial(index, basis_exponent, basis_coefficient_value);
961            accumulator = polynomial_add(&accumulator, &term_polynomial);
962        }
963    }
964    accumulator
965}
966
967fn expand_shifted_power(exponent: usize, delta: i64) -> Vec<(usize, Integer)> {
968    // Expand (k + delta)^exponent via the binomial theorem. Each (m, c)
969    // entry contributes c * k^m. Powers of delta are computed losslessly
970    // through Integer arithmetic.
971    let mut result: Vec<(usize, Integer)> = Vec::with_capacity(exponent + 1);
972    for m in 0..=exponent {
973        let binomial_value = super::integer_binomial(exponent as u64, m as u64);
974        let delta_power = signed_power(delta, exponent - m);
975        let contribution = &binomial_value * &delta_power;
976        result.push((m, contribution));
977    }
978    result
979}
980
981fn signed_power(base: i64, exponent: usize) -> Integer {
982    // Compute base^exponent as a bignum. Used by the binomial expansion
983    // so we never lose precision through the i64 path.
984    if exponent == 0 {
985        return Integer::ONE;
986    }
987    let mut accumulator = Integer::ONE;
988    let base_bignum = Integer::from(base);
989    for _ in 0..exponent {
990        accumulator = &accumulator * &base_bignum;
991    }
992    accumulator
993}

995fn single_term_polynomial(
996    main_variable: SymbolIdentifier,
997    exponent: usize,
998    coefficient: PolynomialCoefficient,
999) -> Polynomial {
1000    // Wrap a single (exponent, coefficient) pair into a polynomial. A
1001    // zero coefficient yields the zero polynomial.
1002    if crate::polynomial::coefficient_is_zero(&coefficient) {
1003        return zero_polynomial(main_variable);
1004    }
1005    Polynomial {
1006        main_variable,
1007        terms: vec![(exponent, coefficient)],
1008    }
1009}

1022fn rational_to_number(value: Rational) -> number_value {
1023    // Lower an exact rational into the number layer. The rational
1024    // builder enforces the integer-collapse invariant so values that
1025    // turn out to be integers come back tagged accordingly.
1026
1027    use std::cmp::Ordering;
1028    if value == Rational::ZERO {
1029        return number_value::Integer(Integer::ZERO);
1030    }
1031    let sign = crate::exact_arithmetic::rational_sign(&value);
1032    let (numerator_natural, denominator_natural) = value.to_numerator_and_denominator();
1033    let mut signed_numerator = Integer::from(numerator_natural);
1034    if sign == Ordering::Less {
1035        signed_numerator = -signed_numerator;
1036    }
1037    let denominator = Integer::from(denominator_natural);
1038    number::from_integers(signed_numerator, denominator)
1039        .unwrap_or(number_value::Integer(Integer::ZERO))
1040}

1042pub(crate) fn rational_null_space(
1043    rows: &[Vec<Rational>],
1044    unknown_count: usize,
1045) -> Vec<Vec<Rational>> {
1046    // Basis for the right null space {v ∈ Q^unknown_count : M·v = 0} of a
1047    // homogeneous rational matrix, by Gauss–Jordan reduction to reduced row
1048    // echelon form. Each free (non-pivot) column contributes one basis vector:
1049    // its own coordinate is set to one, every other free coordinate to zero, and
1050    // the pivot coordinates are read straight off the reduced rows. Returns the
1051    // empty vector when the matrix has full column rank (a trivial null space).
1052    // This is the parametric-Gosper companion to solve_linear_system below: the
1053    // Zeilberger certificate search needs the kernel of a homogeneous system
1054    // rather than a particular solution of an inhomogeneous one.
1055    let row_count = rows.len();
1056    let mut matrix: Vec<Vec<Rational>> = rows
1057        .iter()
1058        .map(|row| {
1059            let mut padded = row.clone();
1060            padded.resize(unknown_count, Rational::ZERO);
1061            padded
1062        })
1063        .collect();
1064    let mut pivot_of_column: Vec<Option<usize>> = vec![None; unknown_count];
1065    let mut pivot_row = 0;
1066    for column in 0..unknown_count {
1067        let mut found_pivot = None;
1068        for row_index in pivot_row..row_count {
1069            if matrix[row_index][column] != Rational::ZERO {
1070                found_pivot = Some(row_index);
1071                break;
1072            }
1073        }
1074        let row_with_pivot = match found_pivot {
1075            Some(value) => value,
1076            None => continue,
1077        };
1078        matrix.swap(pivot_row, row_with_pivot);
1079        let pivot_value = matrix[pivot_row][column].clone();
1080        for column_to_normalise in 0..unknown_count {
1081            matrix[pivot_row][column_to_normalise] =
1082                &matrix[pivot_row][column_to_normalise] / &pivot_value;
1083        }
1084        for row_to_eliminate in 0..row_count {
1085            if row_to_eliminate == pivot_row {
1086                continue;
1087            }
1088            let factor = matrix[row_to_eliminate][column].clone();
1089            if factor == Rational::ZERO {
1090                continue;
1091            }
1092            for column_index in 0..unknown_count {
1093                let pivot_entry = matrix[pivot_row][column_index].clone();
1094                matrix[row_to_eliminate][column_index] =
1095                    &matrix[row_to_eliminate][column_index] - &factor * &pivot_entry;
1096            }
1097        }
1098        pivot_of_column[column] = Some(pivot_row);
1099        pivot_row += 1;
1100        if pivot_row >= row_count {
1101            break;
1102        }
1103    }
1104    let mut basis: Vec<Vec<Rational>> = Vec::new();
1105    for free_column in 0..unknown_count {
1106        if pivot_of_column[free_column].is_some() {
1107            continue;
1108        }
1109        let mut vector = vec![Rational::ZERO; unknown_count];
1110        vector[free_column] = Rational::ONE;
1111        for column in 0..unknown_count {
1112            if let Some(pivot_index) = pivot_of_column[column] {
1113                vector[column] = -matrix[pivot_index][free_column].clone();
1114            }
1115        }
1116        basis.push(vector);
1117    }
1118    basis
1119}

1121pub(crate) fn solve_linear_system(
1122    matrix: Vec<Vec<Rational>>,
1123    right_hand_side: Vec<Rational>,
1124    unknown_count: usize,
1125) -> Option<Vec<Rational>> {
1126    // Gauss–Jordan elimination over the rationals. Free variables (no
1127    // pivot in their column) are set to zero so an under-determined
1128    // consistent system still yields a particular solution — Gosper
1129    // needs any polynomial solution of the bounded degree, not a
1130    // parametrised family.
1131    let row_count = matrix.len();
1132    let mut augmented: Vec<Vec<Rational>> = (0..row_count)
1133        .map(|row_index| {
1134            let mut row = matrix[row_index].clone();
1135            row.push(right_hand_side[row_index].clone());
1136            row
1137        })
1138        .collect();
1139    let mut pivot_columns: Vec<Option<usize>> = vec![None; unknown_count];
1140    let mut pivot_row = 0;
1141    for column_index in 0..unknown_count {
1142        let mut found_pivot = None;
1143        for row_index in pivot_row..row_count {
1144            if augmented[row_index][column_index] != Rational::ZERO {
1145                found_pivot = Some(row_index);
1146                break;
1147            }
1148        }
1149        let row_with_pivot = match found_pivot {
1150            Some(value) => value,
1151            None => continue,
1152        };
1153        augmented.swap(pivot_row, row_with_pivot);
1154        let pivot_value = augmented[pivot_row][column_index].clone();
1155        for column_to_normalise in column_index..=unknown_count {
1156            augmented[pivot_row][column_to_normalise] =
1157                &augmented[pivot_row][column_to_normalise] / &pivot_value;
1158        }
1159        for row_to_eliminate in 0..row_count {
1160            if row_to_eliminate == pivot_row {
1161                continue;
1162            }
1163            let factor = augmented[row_to_eliminate][column_index].clone();
1164            if factor == Rational::ZERO {
1165                continue;
1166            }
1167            for column_to_subtract in column_index..=unknown_count {
1168                let pivot_row_value = augmented[pivot_row][column_to_subtract].clone();
1169                augmented[row_to_eliminate][column_to_subtract] =
1170                    &augmented[row_to_eliminate][column_to_subtract] - &factor * &pivot_row_value;
1171            }
1172        }
1173        pivot_columns[column_index] = Some(pivot_row);
1174        pivot_row += 1;
1175        if pivot_row >= row_count {
1176            break;
1177        }
1178    }
1179    // Consistency check: every all-zero left-hand-side row must also
1180    // have a zero on the right. A non-zero right-hand entry on an
1181    // all-zero row signals that the system has no solution.
1182    for row_index in 0..row_count {
1183        let mut all_zero = true;
1184        for column_index in 0..unknown_count {
1185            if augmented[row_index][column_index] != Rational::ZERO {
1186                all_zero = false;
1187                break;
1188            }
1189        }
1190        if all_zero && augmented[row_index][unknown_count] != Rational::ZERO {
1191            return None;
1192        }
1193    }
1194    let mut solution = vec![Rational::ZERO; unknown_count];
1195    for column_index in 0..unknown_count {
1196        if let Some(row_index) = pivot_columns[column_index] {
1197            solution[column_index] = augmented[row_index][unknown_count].clone();
1198        }
1199    }
1200    Some(solution)
1201}

1217    fn annihilates(rows: &[Vec<Rational>], vector: &[Rational]) -> bool {
1218        // Confirm M·v = 0 for every row, the defining property of a null-space
1219        // vector. Used by the extractor's tests to certify each basis vector.
1220        rows.iter().all(|row| {
1221            let mut accumulator = Rational::ZERO;
1222            for (entry, coordinate) in row.iter().zip(vector.iter()) {
1223                accumulator = &accumulator + &(entry * coordinate);
1224            }
1225            accumulator == Rational::ZERO
1226        })
1227    }
