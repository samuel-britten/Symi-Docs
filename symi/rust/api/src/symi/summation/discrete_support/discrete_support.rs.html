1// Finite discrete support as a proof object (architectural decision 4).
2//
3// The shipped Zeilberger orchestrator *sampled* finite support: it required a concrete integer lower
4// bound and an upper bound equal to the bare recurrence parameter, then checked the resulting
5// recurrence numerically. That is why `Σ_{k=0}^{2n}` and `Σ_{k=-min(a,b,c)}^{min(a,b,c)}` were
6// rejected before any mathematics ran — not because the support was unusable, but because it was
7// never represented.
8//
9// This module represents it. Every denominator factorial in a normalized term contributes an affine
10// bound, because the reciprocal Gamma vanishes at every non-positive integer: `factorial(A)^(-e)`
11// forces the term to zero wherever `A` is a negative integer, which is exactly the mechanism that
12// gives a binomial coefficient its combinatorial zero extension. Intersecting those bounds gives a
13// half-open-free interval `[max lower, min upper]` outside which the term is *proved* zero, and
14// that proof is what licenses two things the engines need:
15//
16//   1. replacing explicit summation bounds by the all-integer natural boundary, so the telescoped
17//      certificate's boundary terms vanish identically rather than by numerical luck; and
18//   2. computing an initial value at a base parameter point, by collapsing the support there to a
19//      finite explicit range — for Dixon at `a = 0` the interval collapses to the single point
20//      `k = 0`, which is the only reason the symbolic initial value `C(b+c, b)` is reachable.
21//
22// Symbolic comparison of bounds is done under one declared assumption: the parameters are
23// non-negative. That is the combinatorial stratum this module commits to (architectural decision
24// 3) — a parameterized binomial identity is proved there and not claimed as an unrestricted
25// Gamma-analytic identity — and the certificate records the assumption instead of hiding it.
26

37/// Public native enumeration named `bound_form`.
38pub enum BoundForm {
39    // A declared summation bound, in the shapes the admitted class allows: a plain affine form, or a
40    // `min`/`max` of affine forms (possibly arrived at through a negation, since `-min(x)` is
41    // `max(-x)`).
42    /// Represents the `single` case.
43    Single(AffineForm),
44    /// Represents the `maximum` case.
45    Maximum(Vec<AffineForm>),
46    /// Represents the `minimum` case.
47    Minimum(Vec<AffineForm>),
48}

50#[derive(Clone, Debug)]
51/// Public native data structure named `discrete_support_certificate`.
52pub struct DiscreteSupportCertificate {
53    // The proved support of a normalized term: the term is zero at every integer index outside
54    // `[max(lower_bounds), min(upper_bounds)]`. `parameter_requirements` holds the index-free forms
55    // that must be non-negative for the term to be non-zero anywhere at all, and
56    // `definedness_bounds` the numerator-factorial arguments that must be non-negative integers for
57    // the combinatorial reading to be defined.
58    /// Stores the `field` value.
59    pub field: CoefficientField,
60    /// Stores the `index` value.
61    pub index: SymbolIdentifier,
62    /// Stores the `lower_bounds` value.
63    pub lower_bounds: Vec<AffineForm>,
64    /// Stores the `upper_bounds` value.
65    pub upper_bounds: Vec<AffineForm>,
66    /// Stores the `parameter_requirements` value.
67    pub parameter_requirements: Vec<AffineForm>,
68    /// Stores the `definedness_bounds` value.
69    pub definedness_bounds: Vec<AffineForm>,
70    /// Stores the `vanishing_index_values` value.
71    pub vanishing_index_values: Vec<i64>,
72    /// Stores the `assumes_nonnegative_parameters` value.
73    pub assumes_nonnegative_parameters: bool,
74}
75
76const RATIONAL_ZERO_SEARCH_WINDOW: i64 = 64;
77const RATIONAL_ZERO_EXTENSION_CEILING: i64 = 64;

79/// Performs the advanced native `build_certificate` operation.
80pub fn build_certificate(
81    field: &CoefficientField,
82    index: SymbolIdentifier,
83    conditions: &ConditionLedger,
84    rational_numerator: &FieldPolynomial,
85) -> Option<DiscreteSupportCertificate> {
86    // Turn a term analysis's affine support bounds into a support certificate. Each recorded bound
87    // `form >= 0` is solved for the index: a unit index coefficient gives an exact symbolic
88    // endpoint, a larger one is only admitted when the remainder is a plain integer (where the
89    // floor or ceiling is computable), and a zero index coefficient is a parameter requirement
90    // rather than an endpoint. Returns None when a bound cannot be solved exactly, because a
91    // rounded endpoint would silently widen the claimed support.
92    let mut lower_bounds: Vec<AffineForm> = Vec::new();
93    let mut upper_bounds: Vec<AffineForm> = Vec::new();
94    let mut parameter_requirements: Vec<AffineForm> = Vec::new();
95    for bound in &conditions.support_bounds {
96        let mut remainder = bound.clone();
97        remainder.index_coefficient = 0;
98        match bound.index_coefficient {
99            0 => parameter_requirements.push(remainder),
100            1 => lower_bounds.push(remainder.scale(-1)),
101            -1 => upper_bounds.push(remainder),
102            coefficient if coefficient > 1 => {
103                if !remainder.is_constant() {
104                    return None;
105                }
106                lower_bounds.push(AffineForm::constant_form(
107                    field.arity(),
108                    ceiling_division(-remainder.constant, coefficient),
109                ));
110            }
111            coefficient => {
112                if !remainder.is_constant() {
113                    return None;
114                }
115                upper_bounds.push(AffineForm::constant_form(
116                    field.arity(),
117                    floor_division(remainder.constant, -coefficient),
118                ));
119            }
120        }
121    }
122    Some(DiscreteSupportCertificate {
123        field: field.clone(),
124        index,
125        lower_bounds,
126        upper_bounds,
127        parameter_requirements,
128        definedness_bounds: conditions.definedness_bounds.clone(),
129        vanishing_index_values: rational_part_integer_zeros(field, rational_numerator),
130        assumes_nonnegative_parameters: true,
131    })
132}

134fn rational_part_integer_zeros(
135    field: &CoefficientField,
136    rational_numerator: &FieldPolynomial,
137) -> Vec<i64> {
138    // Integer indices at which the term's *rational* part vanishes identically in the parameters.
139    //
140    // The factorial part is not the only thing that can make a term zero. `Σ_{k=1}^{n} C(n,k)·k²`
141    // starts at 1, one step inside the binomial support, and the sum is still equal to its
142    // all-integer form — because the `k²` factor kills the `k = 0` term. Without these zeros the
143    // natural-boundary proof refuses a sum it should accept, which is exactly how that case
144    // regressed when the support certificate first replaced the old syntactic bound test.
145    //
146    // Only zeros that hold for *every* parameter value are recorded, since a zero that depends on
147    // the parameters would not license the rewrite on the whole stratum.
148    let mut result: Vec<i64> = Vec::new();
149    if rational_numerator.is_zero() {
150        return result;
151    }
152    let degree = rational_numerator.degree().unwrap_or(0);
153    if degree == 0 {
154        return result;
155    }
156    for candidate in -RATIONAL_ZERO_SEARCH_WINDOW..=RATIONAL_ZERO_SEARCH_WINDOW {
157        let point = field.from_integer(candidate);
158        if field
159            .polynomial_evaluate(rational_numerator, &point)
160            .is_zero()
161        {
162            result.push(candidate);
163        }
164    }
165    result
166}

168fn ceiling_division(numerator: i64, denominator: i64) -> i64 {
169    // Ceiling of an exact integer quotient with a positive denominator.
170    let quotient = numerator / denominator;
171    if numerator % denominator != 0 && (numerator > 0) == (denominator > 0) {
172        quotient + 1
173    } else {
174        quotient
175    }
176}
177
178fn floor_division(numerator: i64, denominator: i64) -> i64 {
179    // Floor of an exact integer quotient with a positive denominator.
180    let quotient = numerator / denominator;
181    if numerator % denominator != 0 && (numerator < 0) != (denominator < 0) {
182        quotient - 1
183    } else {
184        quotient
185    }
186}

188impl DiscreteSupportCertificate {
189    /// Provides the `is_finite` operation on this native type.
190    pub fn is_finite(&self) -> bool {
191        // True when the support is bounded on both sides, the precondition for the natural-boundary
192        // rewrite and for every telescoped boundary term to vanish.
193        !self.lower_bounds.is_empty() && !self.upper_bounds.is_empty()
194    }
195
196    /// Provides the `dominates` operation on this native type.
197    pub fn dominates(&self, left: &AffineForm, right: &AffineForm) -> bool {
198        // Whether `left >= right` is provable under the declared non-negative-parameter stratum: the
199        // difference must be index-free with every parameter coefficient and the constant
200        // non-negative. This is a *sufficient* test, so a false result is a decline, never a claim
201        // that the inequality fails.
202        let difference = left.subtract(right);
203        if difference.depends_on_index() {
204            return false;
205        }
206        if !self.assumes_nonnegative_parameters
207            && difference
208                .parameter_coefficients
209                .iter()
210                .any(|value| *value != 0)
211        {
212            return false;
213        }
214        difference
215            .parameter_coefficients
216            .iter()
217            .all(|value| *value >= 0)
218            && difference.constant >= 0
219    }
220
221    /// Provides the `greatest_lower_bound` operation on this native type.
222    pub fn greatest_lower_bound(&self) -> Option<AffineForm> {
223        // The support's lower endpoint: a recorded lower bound that provably dominates every other.
224        // None when no single bound dominates, which is a decline rather than a wrong endpoint.
225        self.lower_bounds
226            .iter()
227            .find(|candidate| {
228                self.lower_bounds
229                    .iter()
230                    .all(|other| self.dominates(candidate, other))
231            })
232            .cloned()
233    }
234
235    /// Provides the `least_upper_bound` operation on this native type.
236    pub fn least_upper_bound(&self) -> Option<AffineForm> {
237        // The support's upper endpoint: a recorded upper bound provably dominated by every other.
238        self.upper_bounds
239            .iter()
240            .find(|candidate| {
241                self.upper_bounds
242                    .iter()
243                    .all(|other| self.dominates(other, candidate))
244            })
245            .cloned()
246    }
247
248    /// Provides the `proves_vanishing_below` operation on this native type.
249    pub fn proves_vanishing_below(&self, declared_lower: &BoundForm) -> bool {
250        // Whether the term is proved zero at every integer strictly below the declared lower bound.
251        // Since the term vanishes below `max(lower_bounds)`, it suffices to prove
252        // `max(lower_bounds) >= declared_lower`; failing that, the gap between the two is closed by
253        // the rational part's own integer zeros.
254        let dominated = match declared_lower {
255            BoundForm::Single(target) => self
256                .lower_bounds
257                .iter()
258                .any(|bound| self.dominates(bound, target)),
259            BoundForm::Maximum(targets) => targets.iter().all(|target| {
260                self.lower_bounds
261                    .iter()
262                    .any(|bound| self.dominates(bound, target))
263            }),
264            BoundForm::Minimum(targets) => targets.iter().any(|target| {
265                self.lower_bounds
266                    .iter()
267                    .any(|bound| self.dominates(bound, target))
268            }),
269        };
270        if dominated {
271            return true;
272        }
273        self.rational_zeros_close_the_lower_gap(declared_lower)
274    }
275
276    fn rational_zeros_close_the_lower_gap(&self, declared_lower: &BoundForm) -> bool {
277        // Close a gap between the support's lower endpoint and a *higher* declared lower bound by
278        // showing the term vanishes at every integer in between. Only the fully concrete case is
279        // admitted — both endpoints constant — because a symbolic gap has no finite set of indices
280        // to check.
281        let BoundForm::Single(target) = declared_lower else {
282            return false;
283        };
284        if !target.is_constant() {
285            return false;
286        }
287        let Some(support_start) = self.greatest_lower_bound() else {
288            return false;
289        };
290        if !support_start.is_constant() {
291            return false;
292        }
293        let gap = target.constant - support_start.constant;
294        if gap <= 0 || gap > RATIONAL_ZERO_EXTENSION_CEILING {
295            return false;
296        }
297        (support_start.constant..target.constant)
298            .all(|position| self.vanishing_index_values.contains(&position))
299    }
300
301    /// Provides the `proves_vanishing_above` operation on this native type.
302    pub fn proves_vanishing_above(&self, declared_upper: &BoundForm) -> bool {
303        // Whether the term is proved zero at every integer strictly above the declared upper bound.
304        // Since the term vanishes above `min(upper_bounds)`, it suffices to prove
305        // `min(upper_bounds) <= declared_upper`; failing that, the gap is closed by the rational
306        // part's own integer zeros.
307        let dominated = match declared_upper {
308            BoundForm::Single(target) => self
309                .upper_bounds
310                .iter()
311                .any(|bound| self.dominates(target, bound)),
312            BoundForm::Minimum(targets) => targets.iter().all(|target| {
313                self.upper_bounds
314                    .iter()
315                    .any(|bound| self.dominates(target, bound))
316            }),
317            BoundForm::Maximum(targets) => targets.iter().any(|target| {
318                self.upper_bounds
319                    .iter()
320                    .any(|bound| self.dominates(target, bound))
321            }),
322        };
323        if dominated {
324            return true;
325        }
326        self.rational_zeros_close_the_upper_gap(declared_upper)
327    }
328
329    fn rational_zeros_close_the_upper_gap(&self, declared_upper: &BoundForm) -> bool {
330        // The mirror image of `rational_zeros_close_the_lower_gap`, for a declared upper bound that
331        // stops short of the support's upper endpoint.
332        let BoundForm::Single(target) = declared_upper else {
333            return false;
334        };
335        if !target.is_constant() {
336            return false;
337        }
338        let Some(support_end) = self.least_upper_bound() else {
339            return false;
340        };
341        if !support_end.is_constant() {
342            return false;
343        }
344        let gap = support_end.constant - target.constant;
345        if gap <= 0 || gap > RATIONAL_ZERO_EXTENSION_CEILING {
346            return false;
347        }
348        ((target.constant + 1)..=support_end.constant)
349            .all(|position| self.vanishing_index_values.contains(&position))
350    }
351
352    /// Provides the `proves_natural_boundary` operation on this native type.
353    pub fn proves_natural_boundary(
354        &self,
355        declared_lower: &BoundForm,
356        declared_upper: &BoundForm,
357    ) -> bool {
358        // Whether the declared explicit-bound sum equals the all-integer sum of the same term. Both
359        // one-sided proofs plus two-sided finiteness are required: an unbounded side would leave the
360        // telescoped certificate with a non-vanishing boundary term.
361        if !self.is_finite() {
362            return false;
363        }
364        let below = self.proves_vanishing_below(declared_lower);
365        let above = self.proves_vanishing_above(declared_upper);
366        if !below || !above {
367            return false;
368        }
369        true
370    }

372    /// Provides the `substitute_parameter_value` operation on this native type.
373    pub fn substitute_parameter_value(
374        &self,
375        position: usize,
376        value: i64,
377    ) -> DiscreteSupportCertificate {
378        // Specialize the certificate at one parameter value, which is how the initial value of a
379        // recurrence is reached: at the base point the interval typically collapses to a single
380        // index.
381        let specialise = |forms: &[AffineForm]| -> Vec<AffineForm> {
382            forms
383                .iter()
384                .map(|form| substitute_parameter_in_form(form, position, value))
385                .collect()
386        };
387        DiscreteSupportCertificate {
388            field: self.field.clone(),
389            index: self.index,
390            lower_bounds: specialise(&self.lower_bounds),
391            upper_bounds: specialise(&self.upper_bounds),
392            parameter_requirements: specialise(&self.parameter_requirements),
393            definedness_bounds: specialise(&self.definedness_bounds),
394            vanishing_index_values: self.vanishing_index_values.clone(),
395            assumes_nonnegative_parameters: self.assumes_nonnegative_parameters,
396        }
397    }
398
399    /// Provides the `explicit_index_range` operation on this native type.
400    pub fn explicit_index_range(&self) -> Option<Vec<i64>> {
401        // The support as an explicit list of integer indices, available exactly when both endpoints
402        // resolve and their difference is a non-negative integer constant. This is what makes an
403        // initial value computable in closed symbolic form: the sum reduces to finitely many terms
404        // whose parameters remain symbolic.
405        let lower = self.greatest_lower_bound()?;
406        let upper = self.least_upper_bound()?;
407        let span = upper.subtract(&lower);
408        if !span.is_constant() || span.constant < 0 {
409            return None;
410        }
411        if !lower.is_constant() {
412            return None;
413        }
414        Some((lower.constant..=lower.constant + span.constant).collect())
415    }
416}

418/// Performs the advanced native `substitute_parameter_in_form` operation.
419pub fn substitute_parameter_in_form(form: &AffineForm, position: usize, value: i64) -> AffineForm {
420    // Substitute a concrete integer for one parameter inside an affine form.
421    let mut result = form.clone();
422    result.constant += result.parameter_coefficients[position] * value;
423    result.parameter_coefficients[position] = 0;
424    result
425}

427/// Performs the advanced native `parse_bound` operation.
428pub fn parse_bound(
429    expression: &ExpressionReference,
430    field: &CoefficientField,
431    index: SymbolIdentifier,
432) -> Option<BoundForm> {
433    // Read a declared summation bound. `min`/`max` calls are admitted directly, and a negated
434    // `min`/`max` is folded through the identity `-min(x) = max(-x)`, which is what turns Dixon's
435    // `-min(a, b, c)` lower bound into a comparable object instead of an unparsable shape.
436    if let ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments) =
437        &expression.kind
438    {
439        let mut forms: Vec<AffineForm> = Vec::with_capacity(arguments.len());
440        for argument in arguments {
441            forms.push(affine_from_expression(argument, field, index)?);
442        }
443        match builtin {
444            BuiltinFunction::Min => return Some(BoundForm::Minimum(forms)),
445            BuiltinFunction::Max => return Some(BoundForm::Maximum(forms)),
446            _ => return None,
447        }
448    }
449    if let Some(inner) = negated_operand(expression) {
450        if let Some(inner_bound) = parse_bound(&inner, field, index) {
451            return Some(match inner_bound {
452                BoundForm::Single(form) => BoundForm::Single(form.scale(-1)),
453                BoundForm::Minimum(forms) => {
454                    BoundForm::Maximum(forms.iter().map(|form| form.scale(-1)).collect())
455                }
456                BoundForm::Maximum(forms) => {
457                    BoundForm::Minimum(forms.iter().map(|form| form.scale(-1)).collect())
458                }
459            });
460        }
461    }
462    Some(BoundForm::Single(affine_from_expression(
463        expression, field, index,
464    )?))
465}
466
467fn negated_operand(expression: &ExpressionReference) -> Option<ExpressionReference> {
468    // Recognize `-1 * inner` and return `inner`, the canonical shape a negation takes in the
469    // automatically simplified product form.
470    if let ExpressionKind::Product(children) = &expression.kind {
471        if children.len() != 2 {
472            return None;
473        }
474        for (candidate, other) in [(0usize, 1usize), (1, 0)] {
475            if let ExpressionKind::Integer(value) = &children[candidate].kind {
476                if i64::try_from(value).ok() == Some(-1) {
477                    return Some(Arc::clone(&children[other]));
478                }
479            }
480        }
481    }
482    None
483}

591            vec![Arc::clone(&a), Arc::clone(&b), Arc::clone(&c)],

594        let lower = parse_bound(&negated, &field, index).expect("-min parses");

598            "-min must normalise to a maximum"

659    #[test]
660    fn a_declared_interval_narrower_than_the_support_refuses_the_rewrite() {
661        // A declared interval that *clips* the support must refuse the all-integer rewrite: for
662        // Σ_{k=0}^{n-1} C(n, k) the term at k = n is non-zero, so the two sums differ. A declared
663        // interval strictly *wider* than the support is the opposite case and stays licensed,
664        // because the extra indices contribute proved zeros.
665        let (mut context, index) = setup(&["n"]);
666        let n = builders::symbol(&mut context, "n");
667        let k = builders::symbol(&mut context, "k");
668        let summand = binomial_call(&mut context, Arc::clone(&n), k);
669        let (field, certificate) = certificate_for(&mut context, &summand, index);
670        let minus_one = builders::integer_value(&mut context, -1);
671        let clipped_upper = builders::sum(&mut context, vec![Arc::clone(&n), minus_one]);
672        let zero = builders::integer_value(&mut context, 0);
673        let lower = parse_bound(&zero, &field, index).expect("parses");
674        let clipped = parse_bound(&clipped_upper, &field, index).expect("parses");
675        assert!(
676            !certificate.proves_vanishing_above(&clipped),
677            "n - 1 clips the proved support end of n"
678        );
679        assert!(!certificate.proves_natural_boundary(&lower, &clipped));
680        let minus_five = builders::integer_value(&mut context, -5);
681        let widened = parse_bound(&minus_five, &field, index).expect("parses");
682        let upper = parse_bound(&n, &field, index).expect("parses");
683        assert!(
684            certificate.proves_natural_boundary(&widened, &upper),
685            "a wider declared interval adds only proved zeros and stays licensed"
686        );
687    }
