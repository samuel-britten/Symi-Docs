27pub(crate) fn summation_definite_trigonometric_progression(
28    context: &Context,
29    summand: &ExpressionReference,
30    index: SymbolIdentifier,
31    lower_bound: &ExpressionReference,
32    upper_bound: &ExpressionReference,
33) -> Option<ExpressionReference> {
34    let range = prove_finite_integer_range(context, lower_bound, upper_bound)?;
35    if matches!(range, FiniteIntegerRange::Empty) {
36        return Some(builders::zero(context));
37    }
38    let FiniteIntegerRange::Nonempty {
39        lower_bound,
40        upper_bound,
41        count,
42    } = range
43    else {
44        return Some(builders::zero(context));
45    };
46    let atoms = extract_atoms(context, summand, index)?;
47    if atoms.is_empty() || atoms.len() > MAXIMUM_ATOMS {
48        return None;
49    }
50    let mut strides: Vec<ExpressionReference> = Vec::new();
51    for atom in &atoms {
52        if !strides
53            .iter()
54            .any(|stride| expressions_equal(stride, &atom.argument.stride))
55        {
56            strides.push(Arc::clone(&atom.argument.stride));
57        }
58    }
59    if strides.len() > MAXIMUM_DISTINCT_STRIDES {
60        return None;
61    }
62    let branch_count = 1usize << strides.len();
63    let mut branches = Vec::with_capacity(branch_count);
64    for mask in 0..branch_count {
65        let mut conditions = Vec::with_capacity(strides.len());
66        let mut values = Vec::with_capacity(atoms.len());
67        for (stride_index, stride) in strides.iter().enumerate() {
68            let ratio = exponential_ratio(context, stride);
69            let one = builders::one(context);
70            let resonant = mask & (1usize << stride_index) != 0;
71            let condition = if resonant {
72                builders::equal(context, ratio, one)
73            } else {
74                builders::not_equal(context, ratio, one)
75            };
76            conditions.push(condition);
77        }
78        for atom in &atoms {
79            let stride_index = strides
80                .iter()
81                .position(|stride| expressions_equal(stride, &atom.argument.stride))?;
82            let resonant = mask & (1usize << stride_index) != 0;
83            let value = if resonant {
84                resonant_value(context, atom, &lower_bound, &count)
85            } else {
86                generic_value(context, atom, &lower_bound, &upper_bound, &count)
87            };
88            values.push(value);
89        }
90        let condition = logical_and(context, conditions);
91        branches.push((condition, builders::sum(context, values)));
92    }
93    Some(builders::piecewise(context, branches))
94}

96fn extract_atoms(
97    context: &Context,
98    summand: &ExpressionReference,
99    index: SymbolIdentifier,
100) -> Option<Vec<TrigonometricAtom>> {
101    let terms = match &summand.kind {
102        ExpressionKind::Sum(children) => children.as_slice(),
103        _ => std::slice::from_ref(summand),
104    };
105    let mut atoms = Vec::with_capacity(terms.len());
106    for term in terms {
107        atoms.push(extract_atom(context, term, index)?);
108    }
109    Some(atoms)
110}

112fn extract_atom(
113    context: &Context,
114    term: &ExpressionReference,
115    index: SymbolIdentifier,
116) -> Option<TrigonometricAtom> {
117    let (coefficient, call) = match &term.kind {
118        ExpressionKind::FunctionCall(_, _) => (builders::one(context), term),
119        ExpressionKind::Product(factors) => {
120            let mut call = None;
121            let mut coefficient_factors = Vec::new();
122            for factor in factors {
123                if is_supported_call(factor) {
124                    if call.is_some() {
125                        return None;
126                    }
127                    call = Some(factor);
128                } else {
129                    if depends_on(factor, index) {
130                        return None;
131                    }
132                    coefficient_factors.push(Arc::clone(factor));
133                }
134            }
135            let call = call?;
136            let coefficient = builders::product(context, coefficient_factors);
137            (coefficient, call)
138        }
139        _ => return None,
140    };
141    let ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(function), arguments) = &call.kind
142    else {
143        return None;
144    };
145    if arguments.len() != 1 || !matches!(function, BuiltinFunction::Sin | BuiltinFunction::Cos) {
146        return None;
147    }
148    let argument = affine_decomposition(context, &arguments[0], index)?;
149    Some(TrigonometricAtom {
150        coefficient,
151        function: *function,
152        argument,
153    })
154}

156fn is_supported_call(target: &ExpressionReference) -> bool {
157    matches!(
158        &target.kind,
159        ExpressionKind::FunctionCall(
160            FunctionIdentifier::Builtin(BuiltinFunction::Sin | BuiltinFunction::Cos),
161            arguments
162        ) if arguments.len() == 1
163    )
164}

166fn affine_decomposition(
167    context: &Context,
168    target: &ExpressionReference,
169    index: SymbolIdentifier,
170) -> Option<AffineArgument> {
171    if !depends_on(target, index) {
172        return Some(AffineArgument {
173            phase: Arc::clone(target),
174            stride: builders::zero(context),
175        });
176    }
177    match &target.kind {
178        ExpressionKind::Symbol(identifier) if *identifier == index => Some(AffineArgument {
179            phase: builders::zero(context),
180            stride: builders::one(context),
181        }),
182        ExpressionKind::Sum(children) => {
183            let mut phases = Vec::with_capacity(children.len());
184            let mut strides = Vec::with_capacity(children.len());
185            for child in children {
186                let part = affine_decomposition(context, child, index)?;
187                phases.push(part.phase);
188                strides.push(part.stride);
189            }
190            Some(AffineArgument {
191                phase: builders::sum(context, phases),
192                stride: builders::sum(context, strides),
193            })
194        }
195        ExpressionKind::Product(factors) => {
196            let dependent_positions: Vec<usize> = factors
197                .iter()
198                .enumerate()
199                .filter_map(|(position, factor)| depends_on(factor, index).then_some(position))
200                .collect();
201            if dependent_positions.len() != 1 {
202                return None;
203            }
204            let position = dependent_positions[0];
205            let part = affine_decomposition(context, &factors[position], index)?;
206            let constants: Vec<ExpressionReference> = factors
207                .iter()
208                .enumerate()
209                .filter_map(|(candidate, factor)| {
210                    (candidate != position).then(|| Arc::clone(factor))
211                })
212                .collect();
213            let coefficient = builders::product(context, constants);
214            Some(AffineArgument {
215                phase: builders::product(context, vec![Arc::clone(&coefficient), part.phase]),
216                stride: builders::product(context, vec![coefficient, part.stride]),
217            })
218        }
219        _ => None,
220    }
221}

223fn resonant_value(
224    context: &Context,
225    atom: &TrigonometricAtom,
226    lower_bound: &ExpressionReference,
227    count: &ExpressionReference,
228) -> ExpressionReference {
229    let shifted = builders::product(
230        context,
231        vec![Arc::clone(lower_bound), Arc::clone(&atom.argument.stride)],
232    );
233    let initial_argument = builders::sum(context, vec![Arc::clone(&atom.argument.phase), shifted]);
234    let initial = trigonometric_call(context, atom.function, initial_argument);
235    builders::product(
236        context,
237        vec![Arc::clone(&atom.coefficient), Arc::clone(count), initial],
238    )
239}

241fn generic_value(
242    context: &Context,
243    atom: &TrigonometricAtom,
244    lower_bound: &ExpressionReference,
245    upper_bound: &ExpressionReference,
246    count: &ExpressionReference,
247) -> ExpressionReference {
248    let two = builders::integer_value(context, 2);
249    let negative_one = builders::integer_value(context, -1);
250    let inverse_two = builders::power(context, Arc::clone(&two), negative_one);
251    let half_stride = builders::product(
252        context,
253        vec![Arc::clone(&atom.argument.stride), Arc::clone(&inverse_two)],
254    );
255    let count_half_stride =
256        builders::product(context, vec![Arc::clone(count), Arc::clone(&half_stride)]);
257    let amplitude = trigonometric_call(context, BuiltinFunction::Sin, count_half_stride);
258    let endpoint_sum = builders::sum(
259        context,
260        vec![Arc::clone(lower_bound), Arc::clone(upper_bound)],
261    );
262    let centered_shift = builders::product(context, vec![endpoint_sum, Arc::clone(&half_stride)]);
263    let centered_argument = builders::sum(
264        context,
265        vec![Arc::clone(&atom.argument.phase), centered_shift],
266    );
267    let centered = trigonometric_call(context, atom.function, centered_argument);
268    let denominator = trigonometric_call(context, BuiltinFunction::Sin, half_stride);
269    let numerator = builders::product(context, vec![amplitude, centered]);
270    let quotient = builders::divide(context, numerator, denominator);
271    builders::product(context, vec![Arc::clone(&atom.coefficient), quotient])
272}

274fn exponential_ratio(context: &Context, stride: &ExpressionReference) -> ExpressionReference {
275    let imaginary = context
276        .interner
277        .intern(ExpressionKind::Constant(BuiltinConstant::ImaginaryUnit));
278    let argument = builders::product(context, vec![imaginary, Arc::clone(stride)]);
279    builders::function_call(
280        context,
281        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
282        vec![argument],
283    )
284}

298fn logical_and(context: &Context, mut conditions: Vec<ExpressionReference>) -> ExpressionReference {
299    if conditions.len() == 1 {
300        return conditions.pop().unwrap();
301    }
302    builders::function_call(
303        context,
304        FunctionIdentifier::Builtin(BuiltinFunction::LogicalAnd),
305        conditions,
306    )
307}

310    Arc::ptr_eq(left, right) || left == right
