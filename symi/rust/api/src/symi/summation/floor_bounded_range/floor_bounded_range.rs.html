1// Affine floor bounds as a residue-and-support certificate (architectural decision 7).
2//
3// A summation bound such as `floor(n/2)` is not an obstacle to summation mathematics; it is an
4// obstacle to *representation*. The shipped dispatcher rejects it before any mathematics runs
5// because nothing in the stack can say what `floor(n/2)` is. This module says it: a floor of an
6// affine rational form is a Presburger object, and Euclidean division on the parameter turns it
7// into a complete, disjoint ledger of residue strata on each of which the bound is an ordinary
8// affine expression with no floor at all.
9//
10// Concretely, for `floor((a*n + b)/m)` with integer `a`, `b` and positive integer `m`, writing
11// `n = m*q + r` with `0 <= r < m` gives
12//
13//     floor((a*(m*q + r) + b)/m) = a*q + floor((a*r + b)/m)
14//
15// where the remaining floor is a *concrete integer* because `a`, `r`, `b`, and `m` all are. The
16// ledger records one stratum per residue, together with the empty strata and the forward and
17// inverse maps between lattice points, and `replay` re-derives the whole thing over a concrete
18// window: every admitted index must land in exactly one stratum, and each stratum's affine formula
19// must reproduce the original floor exactly. Disjointness and coverage are checked, not assumed.
20//
21// The support half of decision 7 lives next door in `discrete_support`, which
22// shipped first; nothing here introduces a second, incompatible support semantics. What
23// this module contributes to a support argument is the exact endpoint an erased floor is replaced
24// by, which is what makes a boundary term's vanishing checkable rather than plausible.
25

38/// Public native data structure named `affine_floor_bound`.
39pub struct AffineFloorBound {
40    // `floor((parameter_coefficient * parameter + constant) / denominator)` with an exact integer
41    // numerator and a positive integer denominator.
42    /// Stores the `parameter` value.
43    pub parameter: SymbolIdentifier,
44    /// Stores the `parameter_coefficient` value.
45    pub parameter_coefficient: i64,
46    /// Stores the `constant` value.
47    pub constant: i64,
48    /// Stores the `denominator` value.
49    pub denominator: i64,
50}

52#[derive(Clone, Debug, PartialEq, Eq)]
53/// Public native data structure named `residue_stratum`.
54pub struct ResidueStratum {
55    // One residue class of the parameter modulo the denominator. On this stratum the bound is
56    // `quotient_coefficient * quotient + quotient_constant` with no floor, where
57    // `parameter = denominator * quotient + residue`.
58    /// Stores the `residue` value.
59    pub residue: i64,
60    /// Stores the `quotient_coefficient` value.
61    pub quotient_coefficient: i64,
62    /// Stores the `quotient_constant` value.
63    pub quotient_constant: i64,
64    /// Stores the `is_empty_at_every_quotient` value.
65    pub is_empty_at_every_quotient: bool,
66}

69/// Public native data structure named `floor_bounded_range_certificate`.
70pub struct FloorBoundedRangeCertificate {
71    // The complete ledger for one floor-bounded range, together with the assumption it is proved
72    // under. `assumes_integral_nonnegative_parameter` records — rather than derives — the stratum,
73    // matching the shipped `discrete_support` convention; the dispatcher separately declines when
74    // the symbol table contradicts it.
75    /// Stores the `bound` value.
76    pub bound: AffineFloorBound,
77    /// Stores the `lower_bound` value.
78    pub lower_bound: i64,
79    /// Stores the `modulus` value.
80    pub modulus: i64,
81    /// Stores the `strata` value.
82    pub strata: Vec<ResidueStratum>,
83    /// Stores the `assumes_integral_nonnegative_parameter` value.
84    pub assumes_integral_nonnegative_parameter: bool,
85}

87/// Performs the advanced native `recognize_floor_bound` operation.
88pub fn recognize_floor_bound(
89    bound: &ExpressionReference,
90    parameter: SymbolIdentifier,
91) -> Option<AffineFloorBound> {
92    // Recognise `floor(affine rational in the parameter)`. The argument is read as `p*parameter + q`
93    // over the rationals and then cleared to an integer numerator over a positive denominator, so
94    // `floor(n/2)`, `floor((2*n + 1)/3)`, and `floor(n)` are all admitted in one shape. Anything
95    // else — a nested floor, a non-affine argument, a second parameter — declines.
96    let ExpressionKind::FunctionCall(
97        FunctionIdentifier::Builtin(BuiltinFunction::Floor),
98        arguments,
99    ) = &bound.kind
100    else {
101        return None;
102    };
103    if arguments.len() != 1 {
104        return None;
105    }
106    let (slope, intercept) = affine_rational_in(&arguments[0], parameter)?;
107    let slope_parts = crate::exact_arithmetic::rational_parts(&slope);
108    let intercept_parts = crate::exact_arithmetic::rational_parts(&intercept);
109    let slope_denominator =
110        crate::exact_arithmetic::natural_to_integer(slope_parts.denominator.clone());
111    let intercept_denominator =
112        crate::exact_arithmetic::natural_to_integer(intercept_parts.denominator.clone());
113    let common = crate::exact_arithmetic::natural_to_integer(
114        crate::exact_arithmetic::integer_least_common_multiple(
115            &slope_denominator,
116            &intercept_denominator,
117        ),
118    );
119    let denominator = integer_to_i64(&common)?;
120    if denominator <= 0 {
121        return None;
122    }
123    let parameter_coefficient =
124        integer_to_i64(&(&slope_parts.numerator * &(&common / &slope_denominator)))?;
125    let constant =
126        integer_to_i64(&(&intercept_parts.numerator * &(&common / &intercept_denominator)))?;
127    Some(AffineFloorBound {
128        parameter,
129        parameter_coefficient,
130        constant,
131        denominator,
132    })
133}

135/// Performs the advanced native `partition_by_residue` operation.
136pub fn partition_by_residue(
137    bound: &AffineFloorBound,
138    lower_bound: i64,
139) -> Option<FloorBoundedRangeCertificate> {
140    // Build the complete residue ledger. Every stratum is produced, including the ones on which the
141    // range is empty, because inheriting the nonnegative combinatorial convention silently would
142    // make an empty range indistinguishable from an unproved one. The certificate is replayed
143    // before it is returned.
144    if bound.denominator <= 0 {
145        return None;
146    }
147    if bound.denominator > RESIDUE_STRATUM_CEILING {
148        return None;
149    }
150    let mut strata = Vec::with_capacity(bound.denominator as usize);
151    for residue in 0..bound.denominator {
152        let numerator = bound
153            .parameter_coefficient
154            .checked_mul(residue)?
155            .checked_add(bound.constant)?;
156        let quotient_constant = floor_division(numerator, bound.denominator);
157        strata.push(ResidueStratum {
158            residue,
159            quotient_coefficient: bound.parameter_coefficient,
160            quotient_constant,
161            is_empty_at_every_quotient: bound.parameter_coefficient == 0
162                && quotient_constant < lower_bound,
163        });
164    }
165    let certificate = FloorBoundedRangeCertificate {
166        bound: bound.clone(),
167        lower_bound,
168        modulus: bound.denominator,
169        strata,
170        assumes_integral_nonnegative_parameter: true,
171    };
172    if !replay_certificate(&certificate) {
173        return None;
174    }
175    Some(certificate)
176}

178/// Performs the advanced native `replay_certificate` operation.
179pub fn replay_certificate(certificate: &FloorBoundedRangeCertificate) -> bool {
180    // Re-derive the whole ledger over a concrete window of parameter values. Three things are
181    // checked and none is assumed: every value lands in exactly one stratum (disjointness and
182    // coverage), the forward and inverse lattice maps agree, and the stratum's floor-free formula
183    // reproduces the original floor exactly.
184    if certificate.modulus <= 0 || certificate.strata.len() as i64 != certificate.modulus {
185        return false;
186    }
187    for parameter_value in 0..(4 * certificate.modulus + 12) {
188        let matching: Vec<&ResidueStratum> = certificate
189            .strata
190            .iter()
191            .filter(|stratum| parameter_value.rem_euclid(certificate.modulus) == stratum.residue)
192            .collect();
193        if matching.len() != 1 {
194            return false;
195        }
196        let stratum = matching[0];
197        let quotient = parameter_value.div_euclid(certificate.modulus);
198        if certificate.modulus * quotient + stratum.residue != parameter_value {
199            return false;
200        }
201        let Some(direct) = evaluate_bound(&certificate.bound, parameter_value) else {
202            return false;
203        };
204        let through_stratum = stratum.quotient_coefficient * quotient + stratum.quotient_constant;
205        if direct != through_stratum {
206            return false;
207        }
208        let is_empty = direct < certificate.lower_bound;
209        if stratum.is_empty_at_every_quotient && !is_empty {
210            return false;
211        }
212    }
213    true
214}

216/// Performs the advanced native `evaluate_bound` operation.
217pub fn evaluate_bound(bound: &AffineFloorBound, parameter_value: i64) -> Option<i64> {
218    // The bound's exact value at a concrete parameter, by integer floor division.
219    let numerator = bound
220        .parameter_coefficient
221        .checked_mul(parameter_value)?
222        .checked_add(bound.constant)?;
223    Some(floor_division(numerator, bound.denominator))
224}

226/// Performs the advanced native `floor_division` operation.
227pub fn floor_division(dividend: i64, divisor: i64) -> i64 {
228    // Floor division with true floor semantics for negative dividends, matching the `floor`
229    // built-in's measured behaviour rather than Rust's truncating quotient.
230    dividend.div_euclid(divisor)
231}

233/// Performs the advanced native `affine_rational_in` operation.
234pub fn affine_rational_in(
235    value: &ExpressionReference,
236    parameter: SymbolIdentifier,
237) -> Option<(field_element, field_element)> {
238    // Read an expression as `slope * parameter + intercept` over the rationals, declining on any
239    // shape that is not affine in the parameter or that carries a second free symbol.
240    match &value.kind {
241        ExpressionKind::Symbol(identifier) if *identifier == parameter => Some((
242            crate::exact_arithmetic::rational_one(),
243            crate::exact_arithmetic::rational_zero(),
244        )),
245        ExpressionKind::Integer(integer_value) => Some((
246            crate::exact_arithmetic::rational_zero(),
247            crate::exact_arithmetic::rational_from_integer(integer_value.clone()),
248        )),
249        ExpressionKind::Rational(rational_value) => Some((
250            crate::exact_arithmetic::rational_zero(),
251            rational_value.clone(),
252        )),
253        ExpressionKind::Sum(children) => {
254            let mut slope = crate::exact_arithmetic::rational_zero();
255            let mut intercept = crate::exact_arithmetic::rational_zero();
256            for child in children {
257                let (child_slope, child_intercept) = affine_rational_in(child, parameter)?;
258                slope += child_slope;
259                intercept += child_intercept;
260            }
261            Some((slope, intercept))
262        }
263        ExpressionKind::Product(children) => {
264            let mut scale = crate::exact_arithmetic::rational_one();
265            let mut inner: Option<(field_element, field_element)> = None;
266            for child in children {
267                if !depends_on(child, parameter) {
268                    let (child_slope, child_intercept) = affine_rational_in(child, parameter)?;
269                    if !rational_is_zero(&child_slope) {
270                        return None;
271                    }
272                    scale *= child_intercept;
273                    continue;
274                }
275                if inner.is_some() {
276                    return None;
277                }
278                inner = Some(affine_rational_in(child, parameter)?);
279            }
280            let (slope, intercept) = inner?;
281            Some((slope * &scale, intercept * &scale))
282        }
283        _ => None,
284    }
285}

287/// Performs the advanced native `sole_free_parameter` operation.
288pub fn sole_free_parameter(
289    summand: &ExpressionReference,
290    bound: &ExpressionReference,
291    index: SymbolIdentifier,
292) -> Option<SymbolIdentifier> {
293    // The single free symbol other than the summation index shared by the summand and the bound.
294    // A floor-bounded range with more than one parameter is outside the admitted class, and a
295    // range with none has no residue structure to normalize.
296    let mut found: Option<SymbolIdentifier> = None;
297    let mut conflicting = false;
298    let mut visit = |value: &ExpressionReference| {
299        collect_symbols(value, &mut |identifier| {
300            if identifier == index {
301                return;
302            }
303            match found {
304                None => found = Some(identifier),
305                Some(existing) if existing == identifier => {}
306                Some(_) => conflicting = true,
307            }
308        });
309    };
310    visit(summand);
311    visit(bound);
312    if conflicting {
313        return None;
314    }
315    found
316}
317
318fn collect_symbols(value: &ExpressionReference, visit: &mut impl FnMut(SymbolIdentifier)) {
319    // Walk every symbol leaf of an expression.
320    match &value.kind {
321        ExpressionKind::Symbol(identifier) => visit(*identifier),
322        ExpressionKind::Sum(children) | ExpressionKind::Product(children) => {
323            for child in children {
324                collect_symbols(child, visit);
325            }
326        }
327        ExpressionKind::Power(base, exponent) => {
328            collect_symbols(base, visit);
329            collect_symbols(exponent, visit);
330        }
331        ExpressionKind::Factorial(argument) => collect_symbols(argument, visit),
332        ExpressionKind::FunctionCall(_, arguments) => {
333            for argument in arguments {
334                collect_symbols(argument, visit);
335            }
336        }
337        _ => {}
338    }
339}

341/// Performs the advanced native `binomial_is_zero_at` operation.
342pub fn binomial_is_zero_at(top: i64, bottom: i64) -> bool {
343    // Whether `binomial(top, bottom)` is *proved* zero under the measured semantics of the built-in:
344    // the numeric fold returns zero exactly when both arguments are nonnegative and the bottom
345    // exceeds the top. Outside that strip the built-in declines rather than folding, and the two
346    // identity arms follow the generalized convention, so no zero may be claimed there. All six
347    // cases have been measured; this predicate is the only place they are relied on.
348    top >= 0 && bottom >= 0 && bottom > top
349}

351/// Performs the advanced native `binomial_is_defined_at` operation.
352pub fn binomial_is_defined_at(top: i64, bottom: i64) -> bool {
353    // Whether `binomial(top, bottom)` lies inside the combinatorial strip on which the built-in
354    // folds to an exact value.
355    top >= 0 && bottom >= 0
356}

358/// Performs the advanced native `exact_binomial` operation.
359pub fn exact_binomial(top: i64, bottom: i64) -> Option<field_element> {
360    // Exact `binomial(top, bottom)` inside the combinatorial strip.
361    if !binomial_is_defined_at(top, bottom) {
362        return None;
363    }
364    crate::combinatorics::binomial(&exact_integer::from(top), &exact_integer::from(bottom))
365        .map(field_element::from)
366}

368/// Performs the advanced native `bound_expression_is_floor` operation.
369pub fn bound_expression_is_floor(bound: &ExpressionReference) -> bool {
370    // Whether a summation bound is a `floor` call at the top level, the trigger for this analyzer.
371    matches!(
372        &bound.kind,
373        ExpressionKind::FunctionCall(
374            FunctionIdentifier::Builtin(BuiltinFunction::Floor),
375            arguments
376        ) if arguments.len() == 1
377    )
