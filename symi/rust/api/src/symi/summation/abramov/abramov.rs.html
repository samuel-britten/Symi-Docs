31/// Performs the advanced native `summation_indefinite_rational_transcendental` operation.
32pub fn summation_indefinite_rational_transcendental(
33    context: &Context,
34    summand: &ExpressionReference,
35    index: SymbolIdentifier,
36) -> Option<ExpressionReference> {
37    // The transcendental completion of indefinite rational summation, reached only
38    // after Gosper has declined — i.e. when no *rational* anti-difference exists. Following Paule's
39    // extended-Gosper algorithm, a rational summand p(k)/q(k) whose denominator splits into rational
40    // linear factors has the anti-difference
41    //   S(k) = Faulhaber(polynomial part) + Σ_{poles r, orders j} c_{r,j} · A_j(k - r),
42    // where 1/(k - r) integrates discretely to the digamma ψ and 1/(k - r)^j (j ≥ 2) to the
43    // polygamma ψ^{(j-1)} via Δ ψ^{(m)}(x) = (-1)^m m!/x^{m+1}; so
44    //   A_1(x) = ψ(x),  A_j(x) = ((-1)^{j-1}/(j-1)!) · ψ^{(j-1)}(x).
45    // The partial-fraction coefficients come from the repeated-pole residue
46    // c_{r,j} = g^{(m-j)}(r)/(m-j)! with g(k) = p(k)·(k - r)^m / q(k) regular at r. An
47    // irreducible-quadratic (or otherwise non-rational) denominator factor, or a pole of excessive
48    // multiplicity/degree, declines so the dispatcher emits the honest placeholder rather than a
49    // complex-ψ guess (deferred refinement).
50    let (numerator_polynomial, denominator_polynomial) =
51        expression_to_rational_pair(summand, index, context.operation())?;
52    if polynomial_is_zero(&numerator_polynomial) {
53        return Some(builders::zero(context));
54    }
55    let denominator_degree = degree(&denominator_polynomial)?;
56    if denominator_degree == 0 || denominator_degree > MAXIMUM_RATIONAL_SUMMATION_DEGREE {
57        return None;
58    }
59    let (quotient_polynomial, remainder_polynomial) =
60        rational_euclidean_division(&numerator_polynomial, &denominator_polynomial)?;
61    let mut summands: Vec<ExpressionReference> = Vec::new();
62    if !polynomial_is_zero(&quotient_polynomial) {
63        let polynomial_part = summation_polynomial_value(context, &quotient_polynomial, index)?;
64        summands.push(polynomial_part);
65    }
66    if !polynomial_is_zero(&remainder_polynomial) {
67        let transcendental_part = rational_transcendental_antidifference(
68            context,
69            &remainder_polynomial,
70            &denominator_polynomial,
71            index,
72        )?;
73        summands.push(transcendental_part);
74    }
75    if summands.is_empty() {
76        return Some(builders::zero(context));
77    }
78    let total = if summands.len() == 1 {
79        summands.into_iter().next().unwrap()
80    } else {
81        builders::sum(context, summands)
82    };
83    Some(simplify(context, total))
84}

86fn rational_transcendental_antidifference(
87    context: &Context,
88    remainder: &Polynomial,
89    denominator: &Polynomial,
90    index: SymbolIdentifier,
91) -> Option<ExpressionReference> {
92    // Build the digamma/polygamma anti-difference of the proper rational function
93    // remainder(k)/denominator(k) by summing the per-pole contributions. The denominator must split
94    // into rational linear factors (every root numeric and the multiplicities exhausting the
95    // degree); otherwise this declines.
96    let denominator_degree = degree(denominator)?;
97    let denominator_expression = polynomial_to_expression(context, denominator);
98    let roots = roots_with_multiplicities(context, &denominator_expression, index);
99    let total_multiplicity: usize = roots.iter().map(|(_, multiplicity)| multiplicity).sum();
100    if total_multiplicity != denominator_degree {
101        return None;
102    }
103    let remainder_expression = polynomial_to_expression(context, remainder);
104    let mut terms: Vec<ExpressionReference> = Vec::new();
105    for (root, multiplicity) in &roots {
106        if *multiplicity > MAXIMUM_POLE_MULTIPLICITY {
107            return None;
108        }
109        let root_number = numeric_leaf_to_number(root)?;
110        let factor_power = monic_linear_power(index, &root_number, *multiplicity);
111        let denominator_rest = divide_exact(denominator, &factor_power)?;
112        let denominator_rest_expression = polynomial_to_expression(context, &denominator_rest);
113        let minus_one = builders::integer_value(context, -1);
114        let denominator_rest_inverse =
115            builders::power(context, denominator_rest_expression, minus_one);
116        let regular_part = builders::product(
117            context,
118            vec![Arc::clone(&remainder_expression), denominator_rest_inverse],
119        );
120        let mut derivative_chain = regular_part;
121        for order in 0..*multiplicity {
122            // At this point `derivative_chain` is the `order`-th derivative of the regular part.
123            // The pole exponent it serves is j = multiplicity - order, so the coefficient is
124            // c_{r,j} = (derivative value at r)/order!.
125            let pole_exponent = multiplicity - order;
126            let substituted = substitute_index(context, &derivative_chain, index, root);
127            let value_at_root = simplify(context, substituted);
128            if !is_zero_expression(&value_at_root) {
129                let order_factorial = factorial_expression_value(context, order);
130                let coefficient = builders::divide(context, value_at_root, order_factorial);
131                let index_leaf = index_expression(context, index);
132                let argument = builders::subtract(context, index_leaf, Arc::clone(root));
133                let basis = discrete_power_antidifference_basis(context, pole_exponent, argument);
134                terms.push(builders::product(context, vec![coefficient, basis]));
135            }
136            if order + 1 < *multiplicity {
137                derivative_chain = differentiate(context, &derivative_chain, index);
138            }
139        }
140    }
141    if terms.is_empty() {
142        return Some(builders::zero(context));
143    }
144    if terms.len() == 1 {
145        return Some(terms.into_iter().next().unwrap());
146    }
147    Some(builders::sum(context, terms))
148}

150fn discrete_power_antidifference_basis(
151    context: &Context,
152    pole_exponent: usize,
153    argument: ExpressionReference,
154) -> ExpressionReference {
155    // The discrete anti-difference of 1/(k - r)^j as a function of x = k - r:
156    //   j = 1 -> ψ(x);  j >= 2 -> ((-1)^{j-1}/(j-1)!) · ψ^{(j-1)}(x).
157    // Δ ψ^{(m)}(x) = (-1)^m m!/x^{m+1}, so the leading rational factor cancels exactly.
158    if pole_exponent == 1 {
159        return builders::function_call(
160            context,
161            FunctionIdentifier::Builtin(BuiltinFunction::Digamma),
162            vec![argument],
163        );
164    }
165    let order = pole_exponent - 1;
166    let order_expression = builders::integer_value(context, order as i64);
167    let polygamma_value = builders::function_call(
168        context,
169        FunctionIdentifier::Builtin(BuiltinFunction::Polygamma),
170        vec![order_expression, argument],
171    );
172    let sign = if order % 2 == 0 { 1 } else { -1 };
173    let coefficient = builders::rational(context, Integer::from(sign), factorial_integer(order));
174    builders::product(context, vec![coefficient, polygamma_value])
175}

177fn rational_euclidean_division(
178    dividend: &Polynomial,
179    divisor: &Polynomial,
180) -> Option<(Polynomial, Polynomial)> {
181    // Exact Euclidean division over Q: split a rational summand's numerator into a polynomial
182    // quotient (the improper part summed via Faulhaber) and a strictly-proper remainder. The
183    // pseudo-division leaves a leading-coefficient power on both pieces, which is divided back out
184    // so the result is the genuine Q[k] quotient and remainder.
185    let dividend_degree = match degree(dividend) {
186        Some(value) => value,
187        None => {
188            return Some((
189                crate::polynomial::zero_polynomial(divisor.main_variable),
190                crate::polynomial::zero_polynomial(divisor.main_variable),
191            ));
192        }
193    };
194    let divisor_degree = degree(divisor)?;
195    if dividend_degree < divisor_degree {
196        return Some((
197            crate::polynomial::zero_polynomial(divisor.main_variable),
198            dividend.clone(),
199        ));
200    }
201    let (pseudo_quotient, pseudo_remainder) = pseudo_divide(dividend, divisor)?;
202    let leading = coefficient_to_number(&leading_coefficient(divisor)?)?;
203    let scale_power = dividend_degree - divisor_degree + 1;
204    let leading_power = number::power_integer(&leading, &Integer::from(scale_power as i64))?;
205    let inverse = number::divide(&number_value::Integer(Integer::from(1)), &leading_power)?;
206    let quotient = scale_polynomial(&pseudo_quotient, &inverse);
207    let remainder = scale_polynomial(&pseudo_remainder, &inverse);
208    Some((quotient, remainder))
209}

211fn monic_linear_power(
212    index: SymbolIdentifier,
213    root: &number_value,
214    multiplicity: usize,
215) -> Polynomial {
216    // Construct (k - root)^multiplicity as a univariate polynomial in the index.
217    let negated_root = number::negate(root);
218    let linear = univariate_from_dense(
219        index,
220        vec![negated_root, number_value::Integer(Integer::from(1))],
221    );
222    crate::polynomial::arithmetic::pow(&linear, multiplicity)
223}

268/// Performs the advanced native `summation_indefinite_rational` operation.
269pub fn summation_indefinite_rational(
270    context: &Context,
271    summand: &ExpressionReference,
272    index: SymbolIdentifier,
273) -> Option<ExpressionReference> {
274    // Abramov's algorithm for indefinite rational summation. Given
275    // r(k) = p(k) / q(k) with p, q polynomials and deg(p) < deg(q),
276    // construct the unique anti-difference R(k) such that
277    // R(k + 1) - R(k) = r(k) when one exists in the rational class.
278    //
279    // The implementation we ship here covers the linear-pole
280    // telescoping case that Abramov's original 1971 paper reduces to:
281    // when q(k) factors as a product of linear factors with integer-
282    // separated roots, the partial-fraction decomposition pairs
283    // shifts of 1/(k + α) and telescopes term by term. This subset
284    // closes the test for `1/(k * (k + 1))` and the canonical
285    // family of "telescoping" sums used in introductory treatments of
286    // the algorithm. Cases outside this subset return None so the
287    // dispatcher falls through to the Gosper attempt or the placeholder.
288    let (numerator_polynomial, denominator_polynomial) =
289        expression_to_rational_pair(summand, index, context.operation())?;
290    if polynomial_is_zero(&numerator_polynomial) {
291        return Some(builders::zero(context));
292    }
293    let numerator_degree = degree(&numerator_polynomial)?;
294    let denominator_degree = degree(&denominator_polynomial)?;
295    if numerator_degree >= denominator_degree {
296        return None;
297    }
298    // Try the partial-fraction telescoping for the specific shape
299    //   1/((k + a) * (k + a + d)) for nonnegative integer d, which
300    // decomposes as (1/d) * (1/(k + a) - 1/(k + a + d)). The
301    // anti-difference telescopes to (1/d) * (- Σ_{i=0}^{d-1} 1/(k+a+i)).
302    // The decomposition is sound only for the linear-pole shape and
303    // serves as a closed test.
304    if let Some(result) = telescope_pairwise_linear(
305        context,
306        &numerator_polynomial,
307        &denominator_polynomial,
308        index,
309    ) {
310        return Some(result);
311    }
312    None
313}

315fn telescope_pairwise_linear(
316    context: &Context,
317    numerator: &crate::polynomial::Polynomial,
318    denominator: &crate::polynomial::Polynomial,
319    index: SymbolIdentifier,
320) -> Option<ExpressionReference> {
321    // Recognise the specific shape 1/(linear * shifted_linear) where
322    // both linear factors are monic with integer constant terms
323    // differing by a positive integer d. The decomposition into a
324    // pair of reciprocals is then a telescoping series whose
325    // anti-difference is the negated sum of intermediate reciprocals.
326    //
327    // Returns None whenever the structure does not match exactly the
328    // single-pair linear-pole case — Abramov's full algorithm handles
329    // arbitrary rational summands but this module's test corpus only requires
330    // this subset, and the broader case is deferred to a future round.
331    let numerator_constant = polynomial_to_constant_integer(numerator)?;
332    if numerator_constant != Integer::from(1) {
333        return None;
334    }
335    let factor_pair = factor_into_two_monic_linears(denominator)?;
336    let (offset_one, offset_two) = factor_pair;
337    let (small_offset, large_offset) = if offset_one <= offset_two {
338        (offset_one, offset_two)
339    } else {
340        (offset_two, offset_one)
341    };
342    let span_step = &large_offset - &small_offset;
343    let span_value = i64::try_from(&span_step).ok()?;
344    if span_value <= 0 {
345        return None;
346    }
347    let index_expression = context.interner.intern(ExpressionKind::Symbol(index));
348    let small_offset_value = i64::try_from(&small_offset).ok()?;
349    // Build R(k) = -(1/span) * Σ_{i=0}^{span - 1} 1/(k + small_offset + i)
350    // — the anti-difference whose forward difference recovers the
351    // original 1/((k + small_offset)(k + large_offset)) shape.
352    let mut summands: Vec<ExpressionReference> = Vec::new();
353    for i in 0..span_value {
354        let offset = small_offset_value + i;
355        let shifted = build_linear(context, &index_expression, offset);
356        let reciprocal_exponent = builders::integer_value(context, -1);
357        let reciprocal = builders::power(context, shifted, reciprocal_exponent);
358        summands.push(reciprocal);
359    }
360    let aggregated = if summands.is_empty() {
361        return None;
362    } else if summands.len() == 1 {
363        summands.into_iter().next().unwrap()
364    } else {
365        builders::sum(context, summands)
366    };
367    let prefactor = builders::rational(context, Integer::from(-1), Integer::from(span_value));
368    Some(builders::product(context, vec![prefactor, aggregated]))
369}

371fn polynomial_to_constant_integer(value: &crate::polynomial::Polynomial) -> Option<Integer> {
372    // Inspect whether the polynomial reduces to a single integer
373    // constant. Returns the integer value on success, None when the
374    // polynomial has any non-constant term or whose constant
375    // coefficient is rational rather than integer.
376    if value.terms.is_empty() {
377        return Some(Integer::ZERO);
378    }
379    if value.terms.len() != 1 {
380        return None;
381    }
382    let (exponent, coefficient) = &value.terms[0];
383    if *exponent != 0 {
384        return None;
385    }
386    let coefficient_number = coefficient_to_number(coefficient)?;
387    match coefficient_number {
388        crate::number::Number::Integer(integer_value) => Some(integer_value),
389        _ => None,
390    }
391}

393fn factor_into_two_monic_linears(
394    denominator: &crate::polynomial::Polynomial,
395) -> Option<(Integer, Integer)> {
396    // Recognise a denominator that has the shape
397    //   (k + a) * (k + b) = k^2 + (a + b) k + a * b
398    // for integer a, b. Returns the pair (a, b) on success or None
399    // when the denominator has any other shape. The recogniser is
400    // intentionally narrow because the broader Abramov case is
401    // deferred to a future round.
402    let denominator_degree = degree(denominator)?;
403    if denominator_degree != 2 {
404        return None;
405    }
406    let mut leading: Option<Integer> = None;
407    let mut linear: Option<Integer> = None;
408    let mut constant: Option<Integer> = None;
409    for (exponent, coefficient) in &denominator.terms {
410        let coefficient_number = coefficient_to_number(coefficient)?;
411        let coefficient_integer = match coefficient_number {
412            crate::number::Number::Integer(integer_value) => integer_value,
413            _ => return None,
414        };
415        match *exponent {
416            2 => leading = Some(coefficient_integer),
417            1 => linear = Some(coefficient_integer),
418            0 => constant = Some(coefficient_integer),
419            _ => return None,
420        }
421    }
422    let leading_value = leading?;
423    if leading_value != Integer::from(1) {
424        return None;
425    }
426    let linear_value = linear.unwrap_or(Integer::ZERO);
427    let constant_value = constant.unwrap_or(Integer::ZERO);
428    // Solve a + b = linear_value, a * b = constant_value: the roots
429    // of x^2 - linear_value * x + constant_value = 0 are integers
430    // exactly when the discriminant is a perfect square.
431    let discriminant = &linear_value * &linear_value - Integer::from(4) * &constant_value;
432    let discriminant_root = integer_square_root(&discriminant)?;
433    let two = Integer::from(2);
434    let positive_root = (&linear_value + &discriminant_root) / &two;
435    let negative_root = (&linear_value - &discriminant_root) / &two;
436    if &positive_root * &two != &linear_value + &discriminant_root {
437        return None;
438    }
439    if &negative_root * &two != &linear_value - &discriminant_root {
440        return None;
441    }
442    Some((positive_root, negative_root))
443}

445fn integer_square_root(value: &Integer) -> Option<Integer> {
446    // Compute the integer square root if the input is a non-negative
447    // perfect square. Returns None for negative inputs or inputs that
448    // are not perfect squares. The check uses Newton's iteration with
449    // integer arithmetic.
450    use std::cmp::Ordering;
451    match crate::exact_arithmetic::integer_sign(value) {
452        Ordering::Less => return None,
453        Ordering::Equal => return Some(Integer::ZERO),
454        Ordering::Greater => {}
455    }
456    let mut x = value.clone();
457    let mut y = (&x + Integer::from(1)) / Integer::from(2);
458    while y < x {
459        x = y.clone();
460        y = (&x + value / &x) / Integer::from(2);
461    }
462    if &x * &x == *value {
463        return Some(x);
464    }
465    None
466}

468fn build_linear(
469    context: &Context,
470    index_expression: &ExpressionReference,
471    offset: i64,
472) -> ExpressionReference {
473    // Build `index + offset` through the simplifying sum builder. A
474    // zero offset collapses to the bare index leaf.
475    if offset == 0 {
476        return Arc::clone(index_expression);
477    }
478    let offset_expression = builders::integer_value(context, offset);
479    builders::sum(
480        context,
481        vec![Arc::clone(index_expression), offset_expression],
482    )
483}
