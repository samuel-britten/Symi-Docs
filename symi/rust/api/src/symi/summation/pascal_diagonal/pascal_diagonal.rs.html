1// Structural Pascal-diagonal telescoping (architectural decision 8).
2//
3// The summand `binomial(n - c*k, k)` *is* hypergeometric in `k`, so nothing about the term stopped
4// the shipped Zeilberger route. What stopped it is the bound `floor(n/(c + 1))`, which the term
5// analysis never sees because the bound recogniser rejects it first. Once the floor is normalized
6// (see `floor_bounded_range`), a general holonomic engine would still be the wrong tool: Pascal's
7// identity gives a far smaller certificate for this family, and a smaller certificate is a
8// certificate that can be replayed symbolically on every residue stratum instead of numerically at
9// sampled points.
10//
11// The derivation, for `S(T) = Σ_{k=0}^{U(T)} binomial(T - c*k, k)` with `U(T) = floor(T/(c + 1))`:
12//
13//   * the `k = 0` term is `binomial(T, 0) = 1`;
14//   * for `1 <= k <= U(T)` and `T >= c + 1` the top argument satisfies `T - c*k >= 1`, so Pascal's
15//     identity splits the term into `binomial((T-1) - c*k, k) + binomial((T-1-c) - c*(k-1), k-1)`;
16//   * the first family summed over `1 <= k <= U(T)` is `S(T-1) - 1` plus one boundary term, present
17//     only on the stratum `T ≡ 0 (mod c+1)` where `U(T) = U(T-1) + 1`. There `T = (c+1)*U(T)`, so
18//     the boundary term is `binomial(U(T) - 1, U(T))`, which is *proved* zero by the built-in's
19//     measured combinatorial semantics — both arguments nonnegative, bottom above top;
20//   * re-indexing the second family by `j = k - 1` gives exactly `S(T-1-c)`, because
21//     `U(T-1-c) = U(T) - 1` identically for `T >= c + 1`.
22//
23// Hence `S(T) = S(T-1) + S(T-c-1)`. Every one of those steps is replayed here: the residue ledger,
24// the two floor-arithmetic identities on each stratum, the boundary term's vanishing, and finally
25// the recurrence itself against directly accumulated exact sums. The resulting descriptor is then
26// handed to the shared presentation search, which proves equality with a registered sequence by
27// recurrence uniqueness — `fibonacci_number(n + 1)` for the `c = 1` case, with no name-based rewrite
28// anywhere.

50/// Public native data structure named `pascal_diagonal_certificate`.
51pub struct PascalDiagonalCertificate {
52    // Everything the replay checks, kept as data so a corrupted field is detectable. `range` is the
53    // residue ledger; `boundary_strata` names the residues on which the zero-extension boundary term
54    // appears; `boundary_terms` records that term's (top, bottom) arguments per such stratum, each
55    // of which must be proved zero; `initial_values` is the complete uniqueness window.
56    /// Stores the `diagonal_step` value.
57    pub diagonal_step: i64,
58    /// Stores the `shift` value.
59    pub shift: i64,
60    /// Stores the `range` value.
61    pub range: FloorBoundedRangeCertificate,
62    /// Stores the `boundary_strata` value.
63    pub boundary_strata: Vec<i64>,
64    /// Stores the `boundary_terms` value.
65    pub boundary_terms: Vec<(i64, i64)>,
66    /// Stores the `recurrence_order` value.
67    pub recurrence_order: usize,
68    /// Stores the `initial_values` value.
69    pub initial_values: Vec<field_element>,
70    /// Stores the `assumes_integral_nonnegative_parameter` value.
71    pub assumes_integral_nonnegative_parameter: bool,
72}

74/// Performs the advanced native `summation_definite_via_pascal_diagonal` operation.
75pub fn summation_definite_via_pascal_diagonal(
76    context: &Context,
77    summand: &ExpressionReference,
78    index: SymbolIdentifier,
79    lower_bound: &ExpressionReference,
80    upper_bound: &ExpressionReference,
81) -> Option<ExpressionReference> {
82    // Dispatcher entry. Gated on a `floor` upper bound and a binomial-diagonal summand, so it is
83    // inert for every input the shipped stack already reaches.
84    if !floor_bounded_range::bound_expression_is_floor(upper_bound) {
85        return None;
86    }
87    if !matches!(&lower_bound.kind, ExpressionKind::Integer(value)
88        if crate::exact_arithmetic::integer_is_zero(value))
89    {
90        return None;
91    }
92    let parameter = floor_bounded_range::sole_free_parameter(summand, upper_bound, index)?;
93    let bound = floor_bounded_range::recognize_floor_bound(upper_bound, parameter)?;
94    let (diagonal_step, shift) = recognize_diagonal(context, summand, index, parameter)?;
95    if diagonal_step < 1 || diagonal_step > DIAGONAL_STEP_CEILING || shift < 0 {
96        diagnostics::record_verdict(
97            context,
98            "pascal_diagonal.recognition",
99            diagnostics::not_applicable("the diagonal step is outside the admitted range"),
100        );
101        return None;
102    }
103    if bound.parameter_coefficient != 1
104        || bound.constant != shift
105        || bound.denominator != diagonal_step + 1
106    {
107        diagnostics::record_verdict(
108            context,
109            "pascal_diagonal.recognition",
110            diagnostics::not_applicable(
111                "the declared floor bound is not the diagonal family's natural boundary",
112            ),
113        );
114        return None;
115    }
116    let certificate = build_certificate(&bound, diagonal_step, shift)?;
117    let descriptor = descriptor_from_certificate(&certificate)?;
118    let mut budget = ResourceBudget::default();
119    let model = SequenceStateModel::from_descriptor(&descriptor);
120    let shifted = if shift == 0 {
121        model
122    } else {
123        match model.forward_shift(shift as usize, &mut budget) {
124            Ok(shifted) => shifted,
125            Err(rejection) => {
126                c_finite::record_closure_rejection("pascal_diagonal.presentation", rejection);
127                return None;
128            }
129        }
130    };
131    let parameter_expression = context.interner.intern(ExpressionKind::Symbol(parameter));
132    let candidates = registered_candidate_sequences();
133    c_finite::present_state_model(
134        context,
135        &shifted,
136        &parameter_expression,
137        &candidates,
138        &mut budget,
139        ClosureOperation::PascalDiagonal,
140    )
141}

143fn recognize_diagonal(
144    context: &Context,
145    summand: &ExpressionReference,
146    index: SymbolIdentifier,
147    parameter: SymbolIdentifier,
148) -> Option<(i64, i64)> {
149    // Recognise `binomial(parameter + shift - diagonal_step * index, index)`, returning the step and
150    // the shift. Anything else — a scaled parameter, a non-unit bottom argument, an extra factor —
151    // declines, because the boundary accounting below is only proved for this family.
152    let ExpressionKind::FunctionCall(
153        FunctionIdentifier::Builtin(BuiltinFunction::Binomial),
154        arguments,
155    ) = &summand.kind
156    else {
157        return None;
158    };
159    if arguments.len() != 2 {
160        return None;
161    }
162    if !matches!(&arguments[1].kind, ExpressionKind::Symbol(identifier) if *identifier == index) {
163        return None;
164    }
165    let top = Arc::clone(&arguments[0]);
166    let (index_slope, index_free) = split_affine(context, &top, index)?;
167    let diagonal_step = -index_slope;
168    let (parameter_slope, shift) = split_affine(context, &index_free, parameter)?;
169    if parameter_slope != 1 {
170        return None;
171    }
172    let shift_value = match &shift.kind {
173        ExpressionKind::Integer(integer_value) => {
174            crate::exact_arithmetic::integer_to_i64(integer_value)?
175        }
176        _ => return None,
177    };
178    Some((diagonal_step, shift_value))
179}

181fn split_affine(
182    context: &Context,
183    value: &ExpressionReference,
184    variable: SymbolIdentifier,
185) -> Option<(i64, ExpressionReference)> {
186    // Split an expression into an integer coefficient of `variable` and the remaining
187    // variable-free part, returned as an expression so the caller can split again on another
188    // variable.
189    use crate::expression::queries::depends_on;
190    if !depends_on(value, variable) {
191        return Some((0, Arc::clone(value)));
192    }
193    match &value.kind {
194        ExpressionKind::Symbol(identifier) if *identifier == variable => {
195            Some((1, crate::expression::builders::zero(context)))
196        }
197        ExpressionKind::Sum(children) => {
198            let children: Vec<ExpressionReference> = children.iter().map(Arc::clone).collect();
199            let mut coefficient = 0i64;
200            let mut remainder: Vec<ExpressionReference> = Vec::new();
201            for child in &children {
202                let (child_coefficient, child_remainder) = split_affine(context, child, variable)?;
203                coefficient = coefficient.checked_add(child_coefficient)?;
204                if !is_structural_zero(&child_remainder) {
205                    remainder.push(child_remainder);
206                }
207            }
208            let rebuilt = match remainder.len() {
209                0 => crate::expression::builders::zero(context),
210                1 => remainder.into_iter().next().unwrap(),
211                _ => crate::expression::builders::sum(context, remainder),
212            };
213            Some((coefficient, rebuilt))
214        }
215        ExpressionKind::Product(children) => {
216            let children: Vec<ExpressionReference> = children.iter().map(Arc::clone).collect();
217            let mut scale = 1i64;
218            let mut inner: Option<(i64, ExpressionReference)> = None;
219            for child in &children {
220                if !depends_on(child, variable) {
221                    let ExpressionKind::Integer(integer_value) = &child.kind else {
222                        return None;
223                    };
224                    scale = scale
225                        .checked_mul(crate::exact_arithmetic::integer_to_i64(integer_value)?)?;
226                    continue;
227                }
228                if inner.is_some() {
229                    return None;
230                }
231                inner = Some(split_affine(context, child, variable)?);
232            }
233            let (coefficient, remainder) = inner?;
234            if !is_structural_zero(&remainder) {
235                return None;
236            }
237            Some((coefficient.checked_mul(scale)?, remainder))
238        }
239        _ => None,
240    }
241}

243fn is_structural_zero(value: &ExpressionReference) -> bool {
244    // Whether an expression is the integer zero literal.
245    matches!(
246        &value.kind,
247        ExpressionKind::Integer(integer_value)
248            if crate::exact_arithmetic::integer_is_zero(integer_value)
249    )
250}

272fn build_certificate(
273    bound: &AffineFloorBound,
274    diagonal_step: i64,
275    shift: i64,
276) -> Option<PascalDiagonalCertificate> {
277    // Assemble and immediately replay the Pascal certificate. Nothing is returned until the residue
278    // ledger, the two floor identities, every boundary term, and the recurrence itself have all been
279    // re-derived from concrete exact values.
280    let range = floor_bounded_range::partition_by_residue(bound, 0)?;
281    build_certificate_from_range(bound, diagonal_step, shift, range)
282}

284fn build_certificate_from_range(
285    _bound: &AffineFloorBound,
286    diagonal_step: i64,
287    shift: i64,
288    range: FloorBoundedRangeCertificate,
289) -> Option<PascalDiagonalCertificate> {
290    let modulus = diagonal_step + 1;
291    let mut boundary_strata: Vec<i64> = Vec::new();
292    let mut boundary_terms: Vec<(i64, i64)> = Vec::new();
293    for stratum in &range.strata {
294        if stratum.residue == 0 {
295            boundary_strata.push(stratum.residue);
296        }
297    }
298    let order = modulus as usize;
299    let mut initial_values: Vec<field_element> = Vec::with_capacity(order);
300    for total in 0..order as i64 {
301        initial_values.push(diagonal_sum(total, diagonal_step)?);
302    }
303    for total in modulus..modulus + INDEPENDENT_REPLAY_WINDOW {
304        let quotient = total / modulus;
305        if total % modulus == 0 {
306            boundary_terms.push((quotient - 1, quotient));
307        }
308    }
309    let certificate = PascalDiagonalCertificate {
310        diagonal_step,
311        shift,
312        range,
313        boundary_strata,
314        boundary_terms,
315        recurrence_order: order,
316        initial_values,
317        assumes_integral_nonnegative_parameter: true,
318    };
319    if !replay_certificate(&certificate) {
320        return None;
321    }
322    Some(certificate)
323}

325/// Performs the advanced native `replay_certificate` operation.
326pub fn replay_certificate(certificate: &PascalDiagonalCertificate) -> bool {
327    // The independent replay, in four parts.
328    //
329    // 1. The residue ledger re-derives itself (disjointness, coverage, floor equality).
330    // 2. The two floor identities the derivation rests on hold at every admitted total: the upper
331    //    bound rises by one exactly on the residue-zero stratum, and the re-indexed family's bound
332    //    is exactly one lower.
333    // 3. Every recorded boundary term is *proved* zero by the built-in's combinatorial semantics.
334    // 4. The derived recurrence reproduces directly accumulated exact sums, and the recorded
335    //    initial window matches them too.
336    let step = certificate.diagonal_step;
337    let modulus = step + 1;
338    if modulus <= 0 || certificate.recurrence_order != modulus as usize {
339        return false;
340    }
341    if !floor_bounded_range::replay_certificate(&certificate.range) {
342        return false;
343    }
344    for (position, value) in certificate.initial_values.iter().enumerate() {
345        match diagonal_sum(position as i64, step) {
346            Some(direct) if direct == *value => {}
347            _ => return false,
348        }
349    }
350    for (top, bottom) in &certificate.boundary_terms {
351        if !floor_bounded_range::binomial_is_zero_at(*top, *bottom) {
352            return false;
353        }
354    }
355    for total in modulus..modulus + INDEPENDENT_REPLAY_WINDOW {
356        let upper = total.div_euclid(modulus);
357        let previous_upper = (total - 1).div_euclid(modulus);
358        let rise = upper - previous_upper;
359        let expected_rise = i64::from(total.rem_euclid(modulus) == 0);
360        if rise != expected_rise {
361            return false;
362        }
363        if rise == 1 && !certificate.boundary_strata.contains(&0) {
364            return false;
365        }
366        if rise == 1 && !floor_bounded_range::binomial_is_zero_at(upper - 1, upper) {
367            return false;
368        }
369        if (total - 1 - step).div_euclid(modulus) != upper - 1 {
370            return false;
371        }
372        let Some(at_total) = diagonal_sum(total, step) else {
373            return false;
374        };
375        let Some(at_previous) = diagonal_sum(total - 1, step) else {
376            return false;
377        };
378        let Some(at_shifted) = diagonal_sum(total - 1 - step, step) else {
379            return false;
380        };
381        if at_total != at_previous + at_shifted {
382            return false;
383        }
384    }
385    true
386}

388fn diagonal_sum(total: i64, diagonal_step: i64) -> Option<field_element> {
389    // Σ_{k=0}^{floor(total/(step+1))} binomial(total - step*k, k), accumulated exactly through the
390    // concrete binomial helper. This is the independent value stream the certificate is replayed
391    // against; it shares nothing with the recurrence.
392    if total < 0 {
393        return None;
394    }
395    let upper = total.div_euclid(diagonal_step + 1);
396    let mut accumulated = rational_zero();
397    for position in 0..=upper {
398        let top = total - diagonal_step * position;
399        accumulated += floor_bounded_range::exact_binomial(top, position)?;
400    }
401    Some(accumulated)
402}

404fn descriptor_from_certificate(
405    certificate: &PascalDiagonalCertificate,
406) -> Option<ConstantRecurrenceSequence> {
407    // Turn the proved recurrence `S(T) = S(T-1) + S(T-1-step)` into a descriptor in canonical monic
408    // forward form `S(m + step + 1) = S(m + step) + S(m)`. The construction certificate carries a
409    // window of directly summed values, which the descriptor constructor replays through the
410    // recurrence before handing the descriptor back.
411    let order = certificate.recurrence_order;
412    let mut coefficients = vec![rational_zero(); order];
413    coefficients[0] = rational_one();
414    coefficients[order - 1] = &coefficients[order - 1] + &rational_one();
415    let replay_length = (order + INDEPENDENT_REPLAY_WINDOW as usize).min(24);
416    let mut replay_values = Vec::with_capacity(replay_length);
417    for total in 0..replay_length as i64 {
418        replay_values.push(diagonal_sum(total, certificate.diagonal_step)?);
419    }
420    let construction = ConstructionCertificate {
421        operation: ClosureOperation::PascalDiagonal,
422        state_dimension: order,
423        replay_origin: 0,
424        replay_values,
425    };
426    ConstantRecurrenceSequence::from_monic_coefficients(
427        coefficients,
428        0,
429        certificate.initial_values.clone(),
430        0,
431        construction,
432    )
433    .ok()
434}

436fn registered_candidate_sequences() -> Vec<BuiltinFunction> {
437    // The built-ins that carry a defining recurrence and can therefore serve as presentation
438    // candidates. This is the registry, not a table of answers: a candidate is admitted only after
439    // recurrence uniqueness proves it equal to the derived model, and adding a sequence to the
440    // registry is what widens the vocabulary.
441    [BuiltinFunction::FibonacciNumber]
442        .into_iter()
443        .filter(|builtin| builtin_sequence_descriptor(*builtin).is_some())
444        .collect()
