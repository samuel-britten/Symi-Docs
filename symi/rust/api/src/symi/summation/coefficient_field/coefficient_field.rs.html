1// The exact parameter coefficient field `Q(parameter_1, ..., parameter_s)` and univariate
2// polynomials over it (architectural decision 1).
3//
4// Everything the summation stack needs to be *complete* rather than merely correct on numeric
5// coefficients lives here. The shipped Gosper engine read consecutive ratios in `Q(k)`, so a
6// symbolic parameter anywhere in the term blocked it even when the parameter is constant with
7// respect to the summation index. This module supplies the missing ground field: elements are
8// canonical numerator/denominator pairs of sparse multivariate polynomials over `Q` in a fixed,
9// deterministically ordered parameter list, so equality is structural, zero-testing is exact, and
10// no summation-local expression simplification is involved anywhere in the algebra.
11//
12// Two layers are exposed. `field_element` is a member of `Q(parameters)`; `field_polynomial` is a
13// univariate polynomial in the summation index whose coefficients are field elements — the ring
14// Gosper's triple, the Gosper recurrence, and Zeilberger's certificate all live in. Both are
15// manipulated through methods on `coefficient_field`, which owns the parameter list so that
16// variable identity and ordering can never drift between two elements.
17//
18// Multivariate gcd is delegated to the crate's existing recursive `polynomial` gcd (Brown/SPRS)
19// through a conversion, rather than reimplemented: canonical reduction is the only place gcd is
20// needed, and reusing the shipped implementation keeps one gcd in the codebase. Exact division is
21// implemented directly by lexicographic leading-term division, which terminates because the
22// leading monomial strictly decreases and needs no conversion round trip.

48/// Public native data structure named `parameter_polynomial`.
49pub struct ParameterPolynomial {
50    // A sparse element of `Q[parameter_1, ..., parameter_s]`. Keys are exponent vectors of length
51    // equal to the ring arity; the natural `Vec<u32>` ordering is exactly the lexicographic
52    // monomial order with the first parameter most significant, which is a valid monomial order, so
53    // `terms.last_key_value()` is the leading term and leading-term division terminates. No zero
54    // coefficient is ever stored, so `terms.is_empty()` is the unique representation of zero.
55    /// Stores the `terms` value.
56    pub terms: BTreeMap<ExponentVector, Rational>,
57}

82/// Public native data structure named `coefficient_field`.
83pub struct CoefficientField {
84    // The ground field, owning the canonical parameter list. Parameters are stored sorted and
85    // deduplicated so two fields built from the same symbol set are identical and element ordering
86    // is deterministic across runs.
87    parameters: Vec<SymbolIdentifier>,
88    // The operation the field's multivariate greatest common divisors run under, carried on the
89    // ring rather than passed to every element operation: canonical reduction is the only place a
90    // gcd is taken, and it is reached from the arithmetic of `Q(parameters)` itself. The operation
91    // takes no part in the ring's identity — two fields over the same parameter set describe the
92    // same ring whichever operation built them — and it changes no answer, only which backend the
93    // gcd runs on and when it stops.
94    operation: OperationContext,
95}

97impl ParameterPolynomial {
98    /// Provides the `is_zero` operation on this native type.
99    pub fn is_zero(&self) -> bool {
100        // True exactly when no term is stored, which is the unique representation of zero.
101        self.terms.is_empty()
102    }
103
104    /// Provides the `leading_term` operation on this native type.
105    pub fn leading_term(&self) -> Option<(&ExponentVector, &Rational)> {
106        // The lexicographically greatest monomial and its coefficient.
107        self.terms.last_key_value()
108    }
109
110    /// Provides the `total_degree` operation on this native type.
111    pub fn total_degree(&self) -> usize {
112        // The largest sum of exponents over all stored monomials; zero for the zero polynomial.
113        self.terms
114            .keys()
115            .map(|exponents| exponents.iter().map(|value| *value as usize).sum::<usize>())
116            .max()
117            .unwrap_or(0)
118    }
119
120    /// Provides the `degree_in` operation on this native type.
121    pub fn degree_in(&self, position: usize) -> usize {
122        // The largest exponent of one parameter over all stored monomials.
123        self.terms
124            .keys()
125            .map(|exponents| exponents[position] as usize)
126            .max()
127            .unwrap_or(0)
128    }
129
130    /// Provides the `monomial_count` operation on this native type.
131    pub fn monomial_count(&self) -> usize {
132        // How many monomials the polynomial stores; the resource measure the budget tracks.
133        self.terms.len()
134    }
135
136    /// Provides the `as_rational` operation on this native type.
137    pub fn as_rational(&self) -> Option<Rational> {
138        // Read the polynomial as a plain rational constant, or None when it carries a parameter.
139        if self.terms.is_empty() {
140            return Some(Rational::ZERO);
141        }
142        if self.terms.len() != 1 {
143            return None;
144        }
145        let (exponents, coefficient) = self.terms.iter().next().unwrap();
146        if exponents.iter().any(|value| *value != 0) {
147            return None;
148        }
149        Some(coefficient.clone())
150    }
151
152    fn insert_term(&mut self, exponents: ExponentVector, coefficient: Rational) {
153        // Accumulate a coefficient at a monomial, deleting the entry when the running sum cancels
154        // so the zero-free invariant is preserved.
155        if coefficient == Rational::ZERO {
156            return;
157        }
158        let entry = self
159            .terms
160            .entry(exponents.clone())
161            .or_insert_with(|| Rational::ZERO);
162        *entry = &*entry + &coefficient;
163        if *entry == Rational::ZERO {
164            self.terms.remove(&exponents);
165        }
166    }
167}

358    pub fn parameter_multiply(
359        &self,
360        left: &ParameterPolynomial,
361        right: &ParameterPolynomial,
362    ) -> ParameterPolynomial {
363        // Product of two polynomials by convolution of their sparse terms.
364        let mut result = self.polynomial_zero_parameter();
365        for (left_exponents, left_coefficient) in &left.terms {
366            for (right_exponents, right_coefficient) in &right.terms {
367                let exponents: ExponentVector = left_exponents
368                    .iter()
369                    .zip(right_exponents.iter())
370                    .map(|(left_value, right_value)| left_value + right_value)
371                    .collect();
372                result.insert_term(exponents, left_coefficient * right_coefficient);
373            }
374        }
375        result

409    /// Provides the `parameter_divide_exact` operation on this native type.
410    pub fn parameter_divide_exact(
411        &self,
412        dividend: &ParameterPolynomial,
413        divisor: &ParameterPolynomial,
414    ) -> Option<ParameterPolynomial> {
415        // Exact multivariate division by lexicographic leading-term elimination. Returns None when
416        // the division leaves a remainder, so a Some result is a proof of divisibility. Terminates
417        // because every step strictly lowers the remainder's leading monomial in a monomial order.
418        if divisor.is_zero() {
419            return None;
420        }
421        if dividend.is_zero() {
422            return Some(self.polynomial_zero_parameter());
423        }
424        let (divisor_exponents, divisor_coefficient) = divisor.leading_term()?;
425        let divisor_exponents = divisor_exponents.clone();
426        let divisor_coefficient = divisor_coefficient.clone();
427        let mut remainder = dividend.clone();
428        let mut quotient = self.polynomial_zero_parameter();
429        while !remainder.is_zero() {
430            let (remainder_exponents, remainder_coefficient) = remainder.leading_term()?;
431            let mut step_exponents: ExponentVector = Vec::with_capacity(self.arity());
432            for position in 0..self.arity() {
433                if remainder_exponents[position] < divisor_exponents[position] {
434                    return None;
435                }
436                step_exponents.push(remainder_exponents[position] - divisor_exponents[position]);
437            }
438            let step_coefficient = remainder_coefficient / &divisor_coefficient;
439            let mut step = self.polynomial_zero_parameter();
440            step.insert_term(step_exponents, step_coefficient);
441            quotient = self.parameter_add(&quotient, &step);
442            let product = self.parameter_multiply(&step, divisor);
443            remainder = self.parameter_subtract(&remainder, &product);
444        }
445        Some(quotient)
446    }

448    /// Provides the `parameter_greatest_common_divisor` operation on this native type.
449    pub fn parameter_greatest_common_divisor(
450        &self,
451        left: &ParameterPolynomial,
452        right: &ParameterPolynomial,
453    ) -> ParameterPolynomial {
454        // Multivariate gcd, delegated to the crate's recursive-polynomial gcd through a conversion.
455        // Degenerate and constant cases are handled directly so the conversion is only used where
456        // it is needed, and a conversion failure degrades to the unit gcd — which is always a
457        // sound divisor and only costs canonical-form compactness, never correctness.
458        if left.is_zero() {
459            return right.clone();
460        }
461        if right.is_zero() {
462            return left.clone();
463        }
464        if self.arity() == 0 || left.as_rational().is_some() || right.as_rational().is_some() {
465            return self.parameter_constant(&Rational::ONE);
466        }
467        // A single-monomial operand has a monomial gcd, obtained by taking exponents componentwise
468        // against the other operand's monomial content. Handling it here avoids a full multivariate
469        // gcd on the shapes that arise most often from shift substitution.
470        if left.terms.len() == 1 || right.terms.len() == 1 {
471            return self.monomial_greatest_common_divisor(left, right);
472        }
473        let recursive_left = match self.to_recursive(left) {
474            Some(value) => value,
475            None => return self.parameter_constant(&Rational::ONE),
476        };
477        let recursive_right = match self.to_recursive(right) {
478            Some(value) => value,
479            None => return self.parameter_constant(&Rational::ONE),
480        };
481        let shared =
482            polynomial_gcd_with_operation(&recursive_left, &recursive_right, &self.operation);
483        match self.from_recursive(&shared) {
484            Some(value) if !value.is_zero() => value,
485            _ => self.parameter_constant(&Rational::ONE),
486        }
487    }

489    fn monomial_greatest_common_divisor(
490        &self,
491        left: &ParameterPolynomial,
492        right: &ParameterPolynomial,
493    ) -> ParameterPolynomial {
494        // The gcd when at least one operand is a single monomial: componentwise minimum of that
495        // monomial's exponents against the other operand's monomial content, with unit coefficient
496        // (the ground ring is a field, so the numeric part carries no gcd information).
497        let mut exponents = vec![0u32; self.arity()];
498        for (position, slot) in exponents.iter_mut().enumerate() {
499            let left_minimum = left
500                .terms
501                .keys()
502                .map(|monomial| monomial[position])
503                .min()
504                .unwrap_or(0);
505            let right_minimum = right
506                .terms
507                .keys()
508                .map(|monomial| monomial[position])
509                .min()
510                .unwrap_or(0);
511            *slot = left_minimum.min(right_minimum);
512        }
513        let mut result = self.polynomial_zero_parameter();
514        result.insert_term(exponents, Rational::ONE);
515        result
516    }

518    /// Provides the `parameter_substitute` operation on this native type.
519    pub fn parameter_substitute(
520        &self,
521        value: &ParameterPolynomial,
522        position: usize,
523        replacement: &ParameterPolynomial,
524    ) -> ParameterPolynomial {
525        // Substitute one parameter by an arbitrary polynomial, expanding the replacement to each
526        // stored exponent. Used for parameter shifts `p -> p + delta` and affine strides
527        // `p -> stride * p + offset`.
528        let mut result = self.polynomial_zero_parameter();
529        let mut replacement_powers: Vec<ParameterPolynomial> =
530            vec![self.parameter_constant(&Rational::ONE)];
531        let highest = value.degree_in(position);
532        for step in 1..=highest {
533            let previous = replacement_powers[step - 1].clone();
534            replacement_powers.push(self.parameter_multiply(&previous, replacement));
535        }
536        for (exponents, coefficient) in &value.terms {
537            let mut remaining = exponents.clone();
538            let power = remaining[position] as usize;
539            remaining[position] = 0;
540            let mut term = self.polynomial_zero_parameter();
541            term.insert_term(remaining, coefficient.clone());
542            let expanded = self.parameter_multiply(&term, &replacement_powers[power]);
543            result = self.parameter_add(&result, &expanded);
544        }
545        result
546    }

548    /// Provides the `parameter_evaluate` operation on this native type.
549    pub fn parameter_evaluate(
550        &self,
551        value: &ParameterPolynomial,
552        point: &[Rational],
553    ) -> Option<Rational> {
554        // Evaluate at a rational point given one coordinate per parameter, in ring order.
555        if point.len() != self.arity() {
556            return None;
557        }
558        let mut total = Rational::ZERO;
559        for (exponents, coefficient) in &value.terms {
560            let mut term = coefficient.clone();
561            for (position, exponent) in exponents.iter().enumerate() {
562                for _ in 0..*exponent {
563                    term = &term * &point[position];
564                }
565            }
566            total = &total + &term;
567        }
568        Some(total)
569    }

571    fn parameter_normalise(&self, value: &ParameterPolynomial) -> (Rational, ParameterPolynomial) {
572        // Split a non-zero polynomial into its lexicographic leading coefficient and the monic
573        // polynomial obtained by dividing it out. This is the unique-representative choice that
574        // makes field elements canonical.
575        match value.leading_term() {
576            None => (Rational::ONE, self.polynomial_zero_parameter()),
577            Some((_, leading)) => {
578                let leading = leading.clone();
579                let inverse = &Rational::ONE / &leading;
580                (leading, self.parameter_scale(value, &inverse))
581            }
582        }
583    }

587    fn to_recursive(&self, value: &ParameterPolynomial) -> Option<Polynomial> {
588        // Convert a sparse multivariate polynomial into the crate's recursive representation, with
589        // the first ring parameter as the outermost main variable. Only used to reach the shipped
590        // multivariate gcd.
591        if self.arity() == 0 {
592            return None;
593        }
594        let entries: Vec<(ExponentVector, Rational)> = value
595            .terms
596            .iter()
597            .map(|(exponents, coefficient)| (exponents.clone(), coefficient.clone()))
598            .collect();
599        Some(self.build_recursive(&entries, 0))
600    }

602    fn build_recursive(&self, entries: &[(ExponentVector, Rational)], level: usize) -> Polynomial {
603        // Recursive worker for to_recursive: group the entries by the exponent of the level's
604        // variable, then either emit a numeric coefficient (innermost level) or recurse.
605        let variable = self.parameters[level];
606        let mut grouped: BTreeMap<usize, Vec<(ExponentVector, Rational)>> = BTreeMap::new();
607        for (exponents, coefficient) in entries {
608            grouped
609                .entry(exponents[level] as usize)
610                .or_default()
611                .push((exponents.clone(), coefficient.clone()));
612        }
613        let mut terms: Vec<(usize, PolynomialCoefficient)> = Vec::new();
614        for (degree, group) in grouped.into_iter().rev() {
615            let coefficient = if level + 1 == self.arity() {
616                PolynomialCoefficient::Numeric(rational_to_number(&group[0].1))
617            } else {
618                simplify_coefficient(PolynomialCoefficient::Nested(Box::new(
619                    self.build_recursive(&group, level + 1),
620                )))
621            };
622            if crate::polynomial::coefficient_is_zero(&coefficient) {
623                continue;
624            }
625            terms.push((degree, coefficient));
626        }
627        Polynomial {
628            main_variable: variable,
629            terms,
630        }
631    }

633    fn from_recursive(&self, value: &Polynomial) -> Option<ParameterPolynomial> {
634        // Convert a recursive polynomial back to the sparse multivariate form. Returns None when
635        // the polynomial mentions a variable outside the ring or carries an algebraic coefficient.
636        let mut result = self.polynomial_zero_parameter();
637        let exponents = vec![0u32; self.arity()];
638        self.accumulate_recursive(value, &exponents, &Rational::ONE, &mut result)?;
639        Some(result)
640    }
641
642    fn accumulate_recursive(
643        &self,
644        value: &Polynomial,
645        exponents: &ExponentVector,
646        scale: &Rational,
647        target: &mut ParameterPolynomial,
648    ) -> Option<()> {
649        // Recursive worker for from_recursive, threading the exponent vector accumulated so far.
650        let position = self.position_of(value.main_variable)?;
651        for (degree, coefficient) in &value.terms {
652            let mut updated = exponents.clone();
653            let slot = updated.get_mut(position)?;
654            *slot += u32::try_from(*degree).ok()?;
655            match coefficient {
656                PolynomialCoefficient::Numeric(numeric_value) => {
657                    let rational = number_to_rational(numeric_value);
658                    target.insert_term(updated, &rational * scale);
659                }
660                PolynomialCoefficient::Nested(nested) => {
661                    self.accumulate_recursive(nested, &updated, scale, target)?;
662                }
663                PolynomialCoefficient::Algebraic(_) => return None,
664            }
665        }
666        Some(())
667    }

713    /// Provides the `make_element` operation on this native type.
714    pub fn make_element(
715        &self,
716        numerator: ParameterPolynomial,
717        denominator: ParameterPolynomial,
718    ) -> Option<FieldElement> {
719        // Build a canonical field element from a numerator/denominator pair: cancel the gcd, then
720        // scale so the denominator is monic in the lexicographic order. Returns None for a zero
721        // denominator, which is the only way an element can fail to exist.
722        if denominator.is_zero() {
723            return None;
724        }
725        if numerator.is_zero() {
726            return Some(self.zero());
727        }
728        // A constant denominator is the overwhelmingly common case — every polynomial-valued element
729        // has one — and it is already coprime to anything and already monic once scaled out. Taking
730        // it before the gcd keeps ordinary polynomial arithmetic in this field entirely gcd-free,
731        // which is what makes the linear algebra over `Q(parameters)` affordable.
732        if let Some(constant) = denominator.as_rational() {
733            if constant == Rational::ONE {
734                return Some(FieldElement {
735                    numerator,
736                    denominator,
737                });
738            }
739            let inverse = &Rational::ONE / &constant;
740            return Some(FieldElement {
741                numerator: self.parameter_scale(&numerator, &inverse),
742                denominator: self.parameter_constant(&Rational::ONE),
743            });
744        }
745        let shared = self.parameter_greatest_common_divisor(&numerator, &denominator);
746        let (reduced_numerator, reduced_denominator) = match (
747            self.parameter_divide_exact(&numerator, &shared),
748            self.parameter_divide_exact(&denominator, &shared),
749        ) {
750            (Some(left), Some(right)) if !right.is_zero() => (left, right),
751            _ => (numerator, denominator),
752        };
753        let (leading, monic_denominator) = self.parameter_normalise(&reduced_denominator);
754        let inverse_leading = &Rational::ONE / &leading;
755        let scaled_numerator = self.parameter_scale(&reduced_numerator, &inverse_leading);
756        Some(FieldElement {
757            numerator: scaled_numerator,
758            denominator: monic_denominator,
759        })
760    }

910        &self,
911        context: &crate::interner::Context,
912        value: &FieldElement,

925    /// Provides the `parameter_to_expression` operation on this native type.
926    pub fn parameter_to_expression(
927        &self,
928        context: &crate::interner::Context,
929        value: &ParameterPolynomial,
930    ) -> ExpressionReference {
931        // Reify a polynomial in the parameters as an expression through the simplifying builders.
932        if value.is_zero() {
933            return builders::zero(context);
934        }
935        let mut terms: Vec<ExpressionReference> = Vec::new();
936        for (exponents, coefficient) in &value.terms {
937            let mut factors: Vec<ExpressionReference> =
938                vec![rational_to_expression(context, coefficient)];
939            for (position, exponent) in exponents.iter().enumerate() {
940                if *exponent == 0 {
941                    continue;
942                }
943                let symbol = context
944                    .interner
945                    .intern(ExpressionKind::Symbol(self.parameters[position]));
946                let factor = if *exponent == 1 {
947                    symbol
948                } else {
949                    let exponent_expression = builders::integer_value(context, *exponent as i64);
950                    builders::power(context, symbol, exponent_expression)
951                };
952                factors.push(factor);
953            }
954            terms.push(builders::product(context, factors));
955        }
956        builders::sum(context, terms)
957    }

959    /// Provides the `from_expression` operation on this native type.
960    pub fn from_expression(&self, value: &ExpressionReference) -> Option<FieldElement> {
961        // Convert an expression into a canonical field element. Succeeds exactly on rational
962        // functions of the ring parameters with rational coefficients; every other shape — a
963        // transcendental, a factorial, an out-of-ring symbol, a non-integer power — declines, so a
964        // Some result certifies membership in `Q(parameters)`.
965        use ExpressionKind::*;
966        match &value.kind {
967            Integer(integer_value) => Some(self.from_rational(
968                &crate::exact_arithmetic::Rational::from(integer_value.clone()),
969            )),
970            Rational(rational_value) => Some(self.from_rational(rational_value)),
971            Symbol(identifier) => {
972                let position = self.position_of(*identifier)?;
973                Some(self.parameter(position))
974            }
975            Sum(children) => {
976                let mut total = self.zero();
977                for child in children {
978                    total = self.add(&total, &self.from_expression(child)?);
979                }
980                Some(total)
981            }
982            Product(children) => {
983                let mut total = self.one();
984                for child in children {
985                    total = self.multiply(&total, &self.from_expression(child)?);
986                }
987                Some(total)
988            }
989            Power(base, exponent) => {
990                let exponent_value = match &exponent.kind {
991                    ExpressionKind::Integer(exponent_integer) => {
992                        i64::try_from(exponent_integer).ok()?
993                    }
994                    _ => return None,
995                };
996                let base_element = self.from_expression(base)?;
997                self.power(&base_element, exponent_value)
998            }
999            _ => None,
1000        }
1001    }

1048    fn polynomial_insert(&self, target: &mut FieldPolynomial, degree: usize, value: &FieldElement) {
1049        // Accumulate a coefficient at a degree, removing the entry when it cancels.
1050        if value.is_zero() {
1051            return;
1052        }
1053        let updated = match target.terms.get(&degree) {
1054            Some(existing) => self.add(existing, value),
1055            None => value.clone(),
1056        };
1057        if updated.is_zero() {
1058            target.terms.remove(&degree);
1059        } else {
1060            target.terms.insert(degree, updated);
1061        }
1062    }

1115    /// Multiplies index polynomials without intermediate rational cancellation.
1116    pub fn polynomial_multiply_uncancelled(
1117        &self,
1118        left: &FieldPolynomial,
1119        right: &FieldPolynomial,
1120    ) -> Option<FieldPolynomial> {
1121        // This is the fraction-free companion of polynomial_multiply for a caller that will clear
1122        // all denominators before inspecting coefficients. The stored elements remain exact but
1123        // intentionally noncanonical until that clearing point.
1124        let mut terms: BTreeMap<usize, Vec<FieldElement>> = BTreeMap::new();
1125        for (left_degree, left_coefficient) in &left.terms {
1126            for (right_degree, right_coefficient) in &right.terms {
1127                terms
1128                    .entry(left_degree + right_degree)
1129                    .or_default()
1130                    .push(self.multiply_uncancelled(left_coefficient, right_coefficient));
1131            }
1132        }
1133        let mut result = self.polynomial_zero();
1134        for (degree, coefficients) in terms {
1135            let coefficient = coefficients.iter().fold(self.zero(), |total, value| {
1136                self.add_uncancelled(&total, value)
1137            });
1138            if !coefficient.is_zero() {
1139                result.terms.insert(degree, coefficient);
1140            }
1141        }
1142        Some(result)
1143    }

1173    /// Provides the `polynomial_divide` operation on this native type.
1174    pub fn polynomial_divide(
1175        &self,
1176        dividend: &FieldPolynomial,
1177        divisor: &FieldPolynomial,
1178    ) -> Option<(FieldPolynomial, FieldPolynomial)> {
1179        // Division with remainder in `Q(parameters)[index]`, which is a Euclidean domain because
1180        // the coefficients form a field. Returns None only for a zero divisor.
1181        if divisor.is_zero() {
1182            return None;
1183        }
1184        let divisor_degree = divisor.degree()?;
1185        let divisor_leading = divisor.leading_coefficient()?.clone();
1186        let mut remainder = dividend.clone();
1187        let mut quotient = self.polynomial_zero();
1188        while let Some(remainder_degree) = remainder.degree() {
1189            if remainder_degree < divisor_degree {
1190                break;
1191            }
1192            let remainder_leading = remainder.leading_coefficient()?.clone();
1193            let factor = self.divide(&remainder_leading, &divisor_leading)?;
1194            let step = self.polynomial_monomial(remainder_degree - divisor_degree, &factor);
1195            quotient = self.polynomial_add(&quotient, &step);
1196            let product = self.polynomial_multiply(&step, divisor);
1197            remainder = self.polynomial_subtract(&remainder, &product);
1198        }
1199        Some((quotient, remainder))
1200    }

1231        &self,
1232        left: &FieldPolynomial,
1233        right: &FieldPolynomial,
1234    ) -> FieldPolynomial {
1235        // Monic gcd by the Euclidean algorithm, valid because the coefficients form a field.
1236        let mut current = left.clone();
1237        let mut next = right.clone();
1238        while !next.is_zero() {
1239            let remainder = match self.polynomial_divide(&current, &next) {
1240                Some((_, remainder)) => remainder,
1241                None => break,
1242            };
1243            current = next;
1244            next = remainder;
1245        }
1246        self.polynomial_make_monic(&current)
1247    }

1256    pub fn polynomial_shift_index(&self, value: &FieldPolynomial, offset: i64) -> FieldPolynomial {
1257        // Substitute `index -> index + offset`, expanding each power through the binomial theorem
1258        // so the result stays a polynomial with exact field coefficients.

1262        let mut result = self.polynomial_zero();

1264            for (new_degree, factor) in binomial_shift_contributions(*degree, offset) {

1266                self.polynomial_insert(&mut result, new_degree, &scaled);
1267            }
1268        }

1290        &self,
1291        value: &FieldPolynomial,

1295        let mut total = self.zero();

1297            Some(degree) => degree,
1298            None => return total,

1309    /// Provides the `polynomial_resultant` operation on this native type.
1310    pub fn polynomial_resultant(
1311        &self,
1312        left: &FieldPolynomial,
1313        right: &FieldPolynomial,
1314    ) -> FieldElement {
1315        // The resultant of two polynomials over the field, computed by the Euclidean recurrence
1316        //   Res(f, g) = (-1)^(deg f * deg g) * lc(g)^(deg f - deg r) * Res(g, r),  r = f mod g
1317        // with the base cases Res(f, 0) = 0 for positive-degree f and Res(f, c) = c^(deg f). The
1318        // resultant vanishes exactly when the two share a root, which is the complete test the
1319        // dispersion computation needs.
1320        if left.is_zero() || right.is_zero() {
1321            return self.zero();
1322        }
1323        let left_degree = left.degree().unwrap_or(0);
1324        let right_degree = right.degree().unwrap_or(0);
1325        if right_degree == 0 {
1326            let constant = right.leading_coefficient().unwrap().clone();
1327            return self
1328                .power(&constant, left_degree as i64)
1329                .unwrap_or_else(|| self.zero());
1330        }
1331        if left_degree == 0 {
1332            let constant = left.leading_coefficient().unwrap().clone();
1333            return self
1334                .power(&constant, right_degree as i64)
1335                .unwrap_or_else(|| self.zero());
1336        }
1337        let (_, remainder) = match self.polynomial_divide(left, right) {
1338            Some(value) => value,
1339            None => return self.zero(),
1340        };
1341        let remainder_degree = remainder.degree();
1342        let inner = self.polynomial_resultant(right, &remainder);
1343        let right_leading = right.leading_coefficient().unwrap().clone();
1344        let drop = match remainder_degree {
1345            Some(degree) => left_degree - degree,
1346            None => left_degree,
1347        };
1348        let scaling = self
1349            .power(&right_leading, drop as i64)
1350            .unwrap_or_else(|| self.zero());
1351        let mut result = self.multiply(&scaling, &inner);
1352        if (left_degree * right_degree) % 2 == 1 {
1353            result = self.negate(&result);
1354        }
1355        result
1356    }

1358    /// Provides the `polynomial_to_expression` operation on this native type.
1359    pub fn polynomial_to_expression(
1360        &self,
1361        context: &crate::interner::Context,
1362        value: &FieldPolynomial,
1363        index: SymbolIdentifier,
1364    ) -> ExpressionReference {
1365        // Reify a polynomial in the index over the field as an expression.
1366        if value.is_zero() {
1367            return builders::zero(context);
1368        }
1369        let index_expression = context.interner.intern(ExpressionKind::Symbol(index));
1370        let mut terms: Vec<ExpressionReference> = Vec::new();
1371        for (degree, coefficient) in &value.terms {
1372            let coefficient_expression = self.to_expression(context, coefficient);
1373            let power_expression = if *degree == 0 {
1374                builders::one(context)
1375            } else if *degree == 1 {
1376                Arc::clone(&index_expression)
1377            } else {
1378                let exponent = builders::integer_value(context, *degree as i64);
1379                builders::power(context, Arc::clone(&index_expression), exponent)
1380            };
1381            terms.push(builders::product(
1382                context,
1383                vec![coefficient_expression, power_expression],
1384            ));
1385        }
1386        builders::sum(context, terms)
1387    }

1391    /// Provides the `null_space` operation on this native type.
1392    pub fn null_space(
1393        &self,
1394        rows: &[Vec<FieldElement>],
1395        unknown_count: usize,
1396    ) -> Vec<Vec<FieldElement>> {
1397        // A basis for the right null space of a homogeneous matrix over the field.
1398        //
1399        // The fraction-free route is tried first and is the one that matters. Ordinary Gauss–Jordan
1400        // over a rational-function field divides by every pivot, so after the first elimination step
1401        // every entry is a genuine quotient and every subsequent add or multiply pays for a
1402        // multivariate gcd. On the three-parameter Dixon system that alone was the difference between
1403        // seconds and hours. Fraction-free (Bareiss) elimination keeps every entry a *polynomial* and
1404        // replaces the gcds with exact divisions, which are cheap and which the domain guarantees
1405        // succeed; only the final basis vectors are formed over the field, a handful of divisions in
1406        // total. The field-based elimination remains as the fallback for rows that cannot be cleared
1407        // to polynomials.
1408        if let Some(basis) = self.fraction_free_null_space(rows, unknown_count) {
1409            return basis;
1410        }
1411        let row_count = rows.len();
1412        let mut matrix: Vec<Vec<FieldElement>> = rows
1413            .iter()
1414            .map(|row| {
1415                let mut padded = row.clone();
1416                padded.resize(unknown_count, self.zero());
1417                padded
1418            })
1419            .collect();
1420        let mut pivot_of_column: Vec<Option<usize>> = vec![None; unknown_count];
1421        let mut pivot_row = 0;
1422        for column in 0..unknown_count {
1423            if pivot_row >= row_count {
1424                break;
1425            }
1426            let found = (pivot_row..row_count).find(|row| !matrix[*row][column].is_zero());
1427            let row_with_pivot = match found {
1428                Some(value) => value,
1429                None => continue,
1430            };
1431            matrix.swap(pivot_row, row_with_pivot);
1432            let pivot_value = matrix[pivot_row][column].clone();
1433            let inverse = match self.invert(&pivot_value) {
1434                Some(value) => value,
1435                None => continue,
1436            };
1437            for entry in 0..unknown_count {
1438                matrix[pivot_row][entry] = self.multiply(&matrix[pivot_row][entry], &inverse);
1439            }
1440            for row in 0..row_count {
1441                if row == pivot_row {
1442                    continue;
1443                }
1444                let factor = matrix[row][column].clone();
1445                if factor.is_zero() {
1446                    continue;
1447                }
1448                for entry in 0..unknown_count {
1449                    let pivot_entry = matrix[pivot_row][entry].clone();
1450                    let product = self.multiply(&factor, &pivot_entry);
1451                    matrix[row][entry] = self.subtract(&matrix[row][entry], &product);
1452                }
1453            }
1454            pivot_of_column[column] = Some(pivot_row);
1455            pivot_row += 1;
1456        }
1457        let mut basis: Vec<Vec<FieldElement>> = Vec::new();
1458        for free_column in 0..unknown_count {
1459            if pivot_of_column[free_column].is_some() {
1460                continue;
1461            }
1462            let mut vector = vec![self.zero(); unknown_count];
1463            vector[free_column] = self.one();
1464            for column in 0..unknown_count {
1465                if let Some(pivot_index) = pivot_of_column[column] {
1466                    vector[column] = self.negate(&matrix[pivot_index][free_column]);
1467                }
1468            }
1469            basis.push(vector);
1470        }
1471        basis
1472    }

1474    fn clear_row_denominators(&self, row: &[FieldElement]) -> Option<Vec<ParameterPolynomial>> {
1475        // Scale a homogeneous row by the product of its entries' denominators so every entry becomes
1476        // a polynomial. Scaling a row of a homogeneous system by a non-zero factor does not change
1477        // its solution set, so this is free of consequence.
1478        let mut clearing = self.parameter_constant(&Rational::ONE);
1479        for entry in row {
1480            if entry.denominator.as_rational() == Some(Rational::ONE) {
1481                continue;
1482            }
1483            clearing = self.parameter_multiply(&clearing, &entry.denominator);
1484        }
1485        let clearing_element = self.from_parameter_polynomial(&clearing);
1486        let mut result = Vec::with_capacity(row.len());
1487        for entry in row {
1488            let scaled = self.multiply(entry, &clearing_element);
1489            if scaled.denominator.as_rational() != Some(Rational::ONE) {
1490                return None;
1491            }
1492            result.push(scaled.numerator);
1493        }
1494        Some(result)
1495    }

1497    fn fraction_free_null_space(
1498        &self,
1499        rows: &[Vec<FieldElement>],
1500        unknown_count: usize,
1501    ) -> Option<Vec<Vec<FieldElement>>> {
1502        // Two-step fraction-free Gauss–Jordan elimination followed by direct read-off of the kernel.
1503        //
1504        // Each elimination step replaces a row by `(pivot·row − factor·pivot_row) / previous_pivot`.
1505        // Bareiss's identity guarantees that division is exact over an integral domain, and every
1506        // entry stays a polynomial — no gcd is ever needed.
1507        //
1508        // The catch is that those entries are *minors*, so their degrees grow with the elimination
1509        // step. Over the integers that is the classic bounded-swell win; over a multivariate
1510        // polynomial ring the term count grows like the square of the degree, and on a two-parameter
1511        // system the entries reach thousands of monomials and the exact divisions become the
1512        // bottleneck instead. So the size is watched, and exceeding the ceiling abandons the
1513        // fraction-free route in favour of elimination over the field, where gcd cancellation keeps
1514        // entries small at the cost of paying for the gcds. Neither route is right for every system;
1515        // the guard is what picks between them from the system actually in hand.
1516        //
1517        // Returns None when a row cannot be cleared to polynomials, a division turns out inexact, or
1518        // the size ceiling is exceeded.
1519        let row_count = rows.len();
1520        let mut matrix: Vec<Vec<ParameterPolynomial>> = Vec::with_capacity(row_count);
1521        for row in rows {
1522            let mut padded = row.clone();
1523            padded.resize(unknown_count, self.zero());
1524            matrix.push(self.clear_row_denominators(&padded)?);
1525        }
1526        let mut previous_pivot = self.parameter_constant(&Rational::ONE);
1527        let mut pivot_of_column: Vec<Option<usize>> = vec![None; unknown_count];
1528        let mut pivot_row = 0;
1529        for column in 0..unknown_count {
1530            if pivot_row >= row_count {
1531                break;
1532            }
1533            let found = (pivot_row..row_count).find(|row| !matrix[*row][column].is_zero());
1534            let row_with_pivot = match found {
1535                Some(value) => value,
1536                None => continue,
1537            };
1538            matrix.swap(pivot_row, row_with_pivot);
1539            let pivot = matrix[pivot_row][column].clone();
1540            for row in 0..row_count {
1541                if row == pivot_row {
1542                    continue;
1543                }
1544                let factor = matrix[row][column].clone();
1545                for entry in 0..unknown_count {
1546                    let left = self.parameter_multiply(&matrix[row][entry], &pivot);
1547                    let right = self.parameter_multiply(&factor, &matrix[pivot_row][entry]);
1548                    let difference = self.parameter_subtract(&left, &right);
1549                    let updated = self.parameter_divide_exact(&difference, &previous_pivot)?;
1550                    if updated.monomial_count() > FRACTION_FREE_MONOMIAL_CEILING {
1551                        return None;
1552                    }
1553                    matrix[row][entry] = updated;
1554                }
1555            }
1556            previous_pivot = pivot;
1557            pivot_of_column[column] = Some(pivot_row);
1558            pivot_row += 1;
1559        }
1560        let mut basis: Vec<Vec<FieldElement>> = Vec::new();
1561        for free_column in 0..unknown_count {
1562            if pivot_of_column[free_column].is_some() {
1563                continue;
1564            }
1565            let mut vector = vec![self.zero(); unknown_count];
1566            vector[free_column] = self.one();
1567            for column in 0..unknown_count {
1568                let Some(row) = pivot_of_column[column] else {
1569                    continue;
1570                };
1571                let numerator = self.parameter_negate(&matrix[row][free_column]);
1572                let denominator = matrix[row][column].clone();
1573                vector[column] = self.make_element(numerator, denominator)?;
1574            }
1575            basis.push(vector);
1576        }
1577        Some(basis)
1578    }

1580    /// Provides the `solve_linear_system` operation on this native type.
1581    pub fn solve_linear_system(
1582        &self,
1583        rows: &[Vec<FieldElement>],
1584        right_hand_side: &[FieldElement],
1585        unknown_count: usize,
1586    ) -> Option<Vec<FieldElement>> {
1587        // A particular solution of an inhomogeneous system over the field, with free variables set
1588        // to zero. Returns None when a zero row carries a non-zero right-hand entry, i.e. the
1589        // system is inconsistent.
1590        let row_count = rows.len();
1591        let mut augmented: Vec<Vec<FieldElement>> = (0..row_count)
1592            .map(|row| {
1593                let mut entries = rows[row].clone();
1594                entries.resize(unknown_count, self.zero());
1595                entries.push(right_hand_side[row].clone());
1596                entries
1597            })
1598            .collect();
1599        let mut pivot_of_column: Vec<Option<usize>> = vec![None; unknown_count];
1600        let mut pivot_row = 0;
1601        for column in 0..unknown_count {
1602            if pivot_row >= row_count {
1603                break;
1604            }
1605            let found = (pivot_row..row_count).find(|row| !augmented[*row][column].is_zero());
1606            let row_with_pivot = match found {
1607                Some(value) => value,
1608                None => continue,
1609            };
1610            augmented.swap(pivot_row, row_with_pivot);
1611            let inverse = self.invert(&augmented[pivot_row][column].clone())?;
1612            for entry in column..=unknown_count {
1613                augmented[pivot_row][entry] = self.multiply(&augmented[pivot_row][entry], &inverse);
1614            }
1615            for row in 0..row_count {
1616                if row == pivot_row {
1617                    continue;
1618                }
1619                let factor = augmented[row][column].clone();
1620                if factor.is_zero() {
1621                    continue;
1622                }
1623                for entry in column..=unknown_count {
1624                    let pivot_entry = augmented[pivot_row][entry].clone();
1625                    let product = self.multiply(&factor, &pivot_entry);
1626                    augmented[row][entry] = self.subtract(&augmented[row][entry], &product);
1627                }
1628            }
1629            pivot_of_column[column] = Some(pivot_row);
1630            pivot_row += 1;
1631        }
1632        for row in 0..row_count {
1633            let all_zero = (0..unknown_count).all(|column| augmented[row][column].is_zero());
1634            if all_zero && !augmented[row][unknown_count].is_zero() {
1635                return None;
1636            }
1637        }
1638        let mut solution = vec![self.zero(); unknown_count];
1639        for column in 0..unknown_count {
1640            if let Some(row) = pivot_of_column[column] {
1641                solution[column] = augmented[row][unknown_count].clone();
1642            }
1643        }
1644        Some(solution)
1645    }

1648/// Performs the advanced native `binomial_shift_contributions` operation.
1649pub fn binomial_shift_contributions(exponent: usize, offset: i64) -> Vec<(usize, Rational)> {
1650    // Expand `(v + offset)^exponent` by the binomial theorem, returning the (degree, coefficient)
1651    // contributions. Powers of the offset are computed over the integers so nothing is rounded.
1652    let mut result: Vec<(usize, Rational)> = Vec::with_capacity(exponent + 1);
1653    for degree in 0..=exponent {
1654        let binomial = integer_binomial(exponent as u64, degree as u64);
1655        let offset_power = integer_power(offset, exponent - degree);
1656        let coefficient = &binomial * &offset_power;
1657        if coefficient != Integer::ZERO {
1658            result.push((degree, Rational::from(coefficient)));
1659        }
1660    }
1661    result
1662}
1663
1664fn integer_binomial(top: u64, bottom: u64) -> Integer {
1665    // Binomial coefficient over the integers, interleaving multiply and exact divide so no large
1666    // factorial is ever built.
1667    if bottom > top {
1668        return Integer::ZERO;
1669    }
1670    let effective = bottom.min(top - bottom);
1671    let mut result = Integer::ONE;
1672    for step in 0..effective {
1673        result = &result * &Integer::from(top - step) / &Integer::from(step + 1);
1674    }
1675    result
1676}
1677
1678fn integer_power(base: i64, exponent: usize) -> Integer {
1679    // Raise a native integer to a non-negative power over the integers, preserving the sign.
1680    let mut result = Integer::ONE;
1681    let base_integer = Integer::from(base);
1682    for _ in 0..exponent {
1683        result = &result * &base_integer;
1684    }
1685    result
1686}
