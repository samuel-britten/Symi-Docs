27// Halvings of the pi bracket the rational-pi-affine sign rule may spend. The
28// loop starts at width 1, so the finest separation it resolves is about 2^-160
29// — far below the closest a rational with the denominators the app produces can
30// come to pi.
31const PI_AFFINE_REFINEMENT_BUDGET: u32 = 160;

33#[derive(Debug, Clone, PartialEq)]
34pub(crate) struct RealConstantForm {
35    pub(crate) rational_part: exact_rational,
36    pub(crate) pi_coefficient: exact_rational,
37    pub(crate) surd_terms: Vec<(exact_rational, exact_rational)>,
38    pub(crate) algebraic_terms: Vec<(exact_rational, AlgebraicNumber)>,
39}
40
41impl RealConstantForm {
42    pub(crate) fn zero() -> RealConstantForm {
43        // The additive identity of the form: every coefficient absent.
44        RealConstantForm {
45            rational_part: rational_zero(),
46            pi_coefficient: rational_zero(),
47            surd_terms: Vec::new(),
48            algebraic_terms: Vec::new(),
49        }
50    }
51
52    pub(crate) fn from_rational(value: exact_rational) -> RealConstantForm {
53        // A bare rational constant carries no basis term.
54        RealConstantForm {
55            rational_part: value,
56            ..RealConstantForm::zero()
57        }
58    }
59
60    pub(crate) fn as_rational(&self) -> Option<exact_rational> {
61        // The form's value as a rational when it has no basis term at all. Used by
62        // the product and power rules, which distribute only a rational scale.
63        if rational_is_zero(&self.pi_coefficient)
64            && self.surd_terms.is_empty()
65            && self.algebraic_terms.is_empty()
66        {
67            Some(self.rational_part.clone())
68        } else {
69            None
70        }
71    }
72
73    pub(crate) fn is_zero(&self) -> bool {
74        // Every pruned form with no basis term and a zero rational part is the
75        // exact zero. Pruning is what makes this an exact zero test rather than a
76        // structural one.
77        rational_is_zero(&self.rational_part)
78            && rational_is_zero(&self.pi_coefficient)
79            && self.surd_terms.is_empty()
80            && self.algebraic_terms.is_empty()
81    }
82
83    fn add_surd_term(&mut self, coefficient: exact_rational, radicand: exact_rational) {
84        // Accumulate a surd term into the basis, merging into an existing term with
85        // the same squarefree radicand so the basis stays linearly independent.
86        for (existing_coefficient, existing_radicand) in self.surd_terms.iter_mut() {
87            if *existing_radicand == radicand {
88                *existing_coefficient += coefficient;
89                return;
90            }
91        }
92        self.surd_terms.push((coefficient, radicand));
93    }
94
95    fn add_algebraic_term(&mut self, coefficient: exact_rational, value: AlgebraicNumber) {
96        // Accumulate an algebraic leaf into the basis, merging equal leaves.
97        for (existing_coefficient, existing_value) in self.algebraic_terms.iter_mut() {
98            if *existing_value == value {
99                *existing_coefficient += coefficient;
100                return;
101            }
102        }
103        self.algebraic_terms.push((coefficient, value));
104    }
105
106    fn prune(mut self) -> RealConstantForm {
107        // Drop basis terms whose coefficient has cancelled and order the surd terms
108        // by radicand, so structurally different constructions of the same value
109        // produce the same form.
110        self.surd_terms
111            .retain(|(coefficient, _)| !rational_is_zero(coefficient));
112        self.algebraic_terms
113            .retain(|(coefficient, _)| !rational_is_zero(coefficient));
114        self.surd_terms.sort_by(|left, right| left.1.cmp(&right.1));
115        self
116    }
117
118    pub(crate) fn add(self, other: RealConstantForm) -> RealConstantForm {
119        // Sum two forms coefficient by coefficient over the shared basis.
120        let mut result = self;
121        result.rational_part += other.rational_part;
122        result.pi_coefficient += other.pi_coefficient;
123        for (coefficient, radicand) in other.surd_terms {
124            result.add_surd_term(coefficient, radicand);
125        }
126        for (coefficient, value) in other.algebraic_terms {
127            result.add_algebraic_term(coefficient, value);
128        }
129        result.prune()
130    }
131
132    pub(crate) fn scale(self, factor: &exact_rational) -> RealConstantForm {
133        // Multiply every coefficient by a rational, which is the only product the
134        // Tier-1 basis is closed under.
135        let mut result = self;
136        result.rational_part *= factor.clone();
137        result.pi_coefficient *= factor.clone();
138        for (coefficient, _) in result.surd_terms.iter_mut() {
139            *coefficient *= factor.clone();
140        }
141        for (coefficient, _) in result.algebraic_terms.iter_mut() {
142            *coefficient *= factor.clone();
143        }
144        result.prune()
145    }
146
147    pub(crate) fn negate(self) -> RealConstantForm {
148        // Additive inverse of the form.
149        let minus_one = rational_from_integer(exact_integer::from(-1));
150        self.scale(&minus_one)
151    }
152}
153
154pub(crate) fn subtract(left: &RealConstantForm, right: &RealConstantForm) -> RealConstantForm {
155    // Pure data subtraction of two forms. This is the whole point of Tier 1: the
156    // difference of two operands is available without a context handle, so a
157    // consumer holding only a symbol table can still ask for the sign of
158    // `left - right`.
159    left.clone().add(right.clone().negate())
160}

162pub(crate) fn extract(
163    reference: &ExpressionReference,
164    table: &SymbolTable,
165    tier: SignTier,
166) -> Option<RealConstantForm> {
167    // Recursive descent from an expression onto the canonical real-constant form,
168    // declining (None) for anything outside the Tier-1 grammar rather than
169    // failing. The grammar is rational and integer literals, pi, real
170    // polynomial_root leaves, sums, products carrying at most one non-rational
171    // factor, integer powers of rationals, square roots of positive rationals,
172    // and the abs / signum / sqrt sugar over those.
173    match &reference.kind {
174        ExpressionKind::Integer(value) => Some(RealConstantForm::from_rational(
175            rational_from_integer(value.clone()),
176        )),
177        ExpressionKind::Rational(value) => Some(RealConstantForm::from_rational(value.clone())),
178        ExpressionKind::Constant(BuiltinConstant::Pi) => {
179            let mut form = RealConstantForm::zero();
180            form.pi_coefficient = rational_one();
181            Some(form)
182        }
183        ExpressionKind::PolynomialRoot(value) => extract_algebraic_leaf(value),
184        ExpressionKind::Sum(children) => {
185            let mut form = RealConstantForm::zero();
186            for child in children {
187                form = form.add(extract(child, table, tier)?);
188            }
189            Some(form)
190        }
191        ExpressionKind::Product(children) => extract_product(children, table, tier),
192        ExpressionKind::Power(base, exponent) => extract_power(base, exponent, table, tier),
193        ExpressionKind::FunctionCall(FunctionIdentifier::Builtin(builtin), arguments)
194            if arguments.len() == 1 =>
195        {
196            extract_unary_call(*builtin, &arguments[0], table, tier)
197        }
198        _ => None,
199    }
200}

202fn positive_verdict(
203    reference: &ExpressionReference,
204    table: &SymbolTable,
205    tier: SignTier,
206) -> TruthValue {
207    // Strict positivity at the requested tier. The structural handler already reads
208    // a symbol leaf's declared facts, so the ground tier keeps every verdict that
209    // does not need a model search to reach.
210    match tier {
211        SignTier::Ground => structural_is_positive(reference, table),
212        SignTier::Assumptions => is_positive(reference, table),
213    }
214}

216fn negative_verdict(
217    reference: &ExpressionReference,
218    table: &SymbolTable,
219    tier: SignTier,
220) -> TruthValue {
221    match tier {
222        SignTier::Ground => structural_is_negative(reference, table),
223        SignTier::Assumptions => is_negative(reference, table),
224    }

227fn extract_algebraic_leaf(value: &AlgebraicNumber) -> Option<RealConstantForm> {
228    // A polynomial_root leaf enters the basis only when its isolating region is
229    // real; a leaf isolated in the complex plane carries no order information and
230    // is declined. A degree-one leaf is a rational in disguise and folds into the
231    // rational part.
232    let AlgebraicIsolatingRegion::Real(_) = &value.isolating_region else {
233        return None;
234    };
235    if let Some(rational) = value.rational_value() {
236        return Some(RealConstantForm::from_rational(rational));
237    }
238    let mut form = RealConstantForm::zero();
239    form.algebraic_terms.push((rational_one(), value.clone()));
240    Some(form)
241}

243fn extract_product(
244    children: &[ExpressionReference],
245    table: &SymbolTable,
246    tier: SignTier,
247) -> Option<RealConstantForm> {
248    // A product distributes a rational scale over at most one non-rational form.
249    // Two non-rational factors would need a product of basis elements, which the
250    // linear Tier-1 basis cannot represent, so the extraction declines and leaves
251    // the comparison to Tier 2.
252    let mut scale = rational_one();
253    let mut non_rational: Option<RealConstantForm> = None;
254    for child in children {
255        let child_form = extract(child, table, tier)?;
256        if let Some(value) = child_form.as_rational() {
257            scale *= value;
258            continue;
259        }
260        if non_rational.is_some() {
261            return None;
262        }
263        non_rational = Some(child_form);
264    }
265    Some(match non_rational {
266        Some(form) => form.scale(&scale),
267        None => RealConstantForm::from_rational(scale),
268    })
269}

271fn extract_power(
272    base: &ExpressionReference,
273    exponent: &ExpressionReference,
274    table: &SymbolTable,
275    tier: SignTier,
276) -> Option<RealConstantForm> {
277    // Integer powers of a rational fold to a rational. A one-half exponent over a
278    // strictly positive rational base is a surd term, normalised so that its
279    // radicand is a squarefree integer; the positivity check is the realness gate
280    // of the whole procedure, because the principal square root of a negative
281    // rational is not a real number and carries no order verdict. Every other
282    // exponent is outside the Tier-1 basis.
283    match &exponent.kind {
284        ExpressionKind::Integer(exponent_value) => {
285            let base_value = extract(base, table, tier)?.as_rational()?;
286            let power = rational_power_checked(&base_value, exponent_value)?;
287            Some(RealConstantForm::from_rational(power))
288        }
289        ExpressionKind::Rational(exponent_value) if is_one_half(exponent_value) => {
290            let radicand = extract(base, table, tier)?.as_rational()?;
291            if radicand.sign_comparison() != Ordering::Greater {
292                return None;
293            }
294            let (scale, squarefree_radicand) = normalise_surd(&radicand)?;
295            let mut form = RealConstantForm::zero();
296            if squarefree_radicand == rational_one() {
297                form.rational_part = scale;
298            } else {
299                form.surd_terms.push((scale, squarefree_radicand));
300            }
301            Some(form)
302        }
303        _ => None,
304    }
305}

307fn extract_unary_call(
308    builtin: BuiltinFunction,
309    argument: &ExpressionReference,
310    table: &SymbolTable,
311    tier: SignTier,
312) -> Option<RealConstantForm> {
313    // The three unary heads the decided class admits. abs and signum are decided
314    // by recursing on the operand's own sign, falling back to the assumption
315    // queries when the form is undecided; sqrt is the surd sugar and is routed
316    // through the same one-half power rule so the two spellings cannot diverge.
317    match builtin {
318        BuiltinFunction::Abs => {
319            let operand = extract(argument, table, tier)?;
320            match sign(&operand) {
321                SignDecision::Positive | SignDecision::Zero => Some(operand),
322                SignDecision::Negative => Some(operand.negate()),
323                SignDecision::Unknown => {
324                    // The ground tier reads the declared facts about the operand's
325                    // symbol leaves through the structural handlers and stops there;
326                    // the assumption tier lets the same question reach the prover.
327                    if positive_verdict(argument, table, tier) == TruthValue::Yes {
328                        Some(operand)
329                    } else if negative_verdict(argument, table, tier) == TruthValue::Yes {
330                        Some(operand.negate())
331                    } else {
332                        None
333                    }
334                }
335            }
336        }
337        BuiltinFunction::Signum => {
338            let operand = extract(argument, table, tier)?;
339            let value = match sign(&operand) {
340                SignDecision::Positive => exact_integer::from(1),
341                SignDecision::Negative => exact_integer::from(-1),
342                SignDecision::Zero => exact_integer::from(0),
343                SignDecision::Unknown => return None,
344            };
345            Some(RealConstantForm::from_rational(rational_from_integer(
346                value,
347            )))
348        }
349        BuiltinFunction::Sqrt => {
350            let radicand = extract(argument, table, tier)?.as_rational()?;
351            if radicand.sign_comparison() != Ordering::Greater {
352                return None;
353            }
354            let (scale, squarefree_radicand) = normalise_surd(&radicand)?;
355            let mut form = RealConstantForm::zero();
356            if squarefree_radicand == rational_one() {
357                form.rational_part = scale;
358            } else {
359                form.surd_terms.push((scale, squarefree_radicand));
360            }
361            Some(form)
362        }
363        _ => None,
364    }
365}

367pub(crate) fn sign(form: &RealConstantForm) -> SignDecision {
368    // The exact sign of a Tier-1 form, ordered by cost. The cheap closed rules
369    // come first: a pure rational, a rational-pi-affine value against exact pi
370    // bounds, the single quadratic surd `p + q*sqrt(r)` decided by comparing p^2
371    // with q^2*r, and a single real algebraic leaf decided by isolating-interval
372    // refinement. What remains is a linear combination over several basis
373    // elements; it is decided by refining an exact rational enclosure until zero
374    // is excluded, which terminates because such a combination is provably
375    // nonzero — the square roots of distinct squarefree integers are linearly
376    // independent over the rationals, and pi is transcendental so it is
377    // independent of every algebraic value. Forms carrying more than one
378    // algebraic leaf have no such zero test until the minimal-polynomial tier
379    // exists, so they decline rather than refine forever.
380    if form.is_zero() {
381        return SignDecision::Zero;
382    }
383    if form.surd_terms.is_empty() && form.algebraic_terms.is_empty() {
384        if rational_is_zero(&form.pi_coefficient) {
385            return rational_to_sign(&form.rational_part);
386        }
387        return rational_pi_affine_sign(&form.rational_part, &form.pi_coefficient);
388    }
389    if rational_is_zero(&form.pi_coefficient) && form.algebraic_terms.is_empty() {
390        if let [(coefficient, radicand)] = form.surd_terms.as_slice() {
391            return surd_sum_sign(&form.rational_part, coefficient, radicand);
392        }
393    }
394    if rational_is_zero(&form.pi_coefficient) && form.surd_terms.is_empty() {
395        if let [(coefficient, value)] = form.algebraic_terms.as_slice() {
396            return single_algebraic_term_sign(&form.rational_part, coefficient, value);
397        }
398    }
399    if form.algebraic_terms.is_empty() {
400        return sign_by_enclosure(form);
401    }
402    SignDecision::Unknown
403}

405fn sign_by_enclosure(form: &RealConstantForm) -> SignDecision {
406    // Refine an exact rational enclosure of a provably nonzero form until the
407    // enclosure excludes zero. The loop stops early when a refinement fails to
408    // narrow the enclosure — which happens once the fixed rational pi bounds
409    // dominate the width — and it always stops at the iteration budget, so an
410    // undecided value is reported honestly instead of looping.
411    let two = rational_from_integer(exact_integer::from(2));
412    let mut target_width = rational_one();
413    let mut previous_width: Option<exact_rational> = None;
414    for _ in 0..ENCLOSURE_REFINEMENT_BUDGET {
415        let Some((lower, upper)) = enclose(form, &target_width) else {
416            return SignDecision::Unknown;
417        };
418        if lower.sign_comparison() == Ordering::Greater {
419            return SignDecision::Positive;
420        }
421        if upper.sign_comparison() == Ordering::Less {
422            return SignDecision::Negative;
423        }
424        let achieved_width = upper - lower;
425        if let Some(previous) = &previous_width {
426            if achieved_width >= *previous {
427                return SignDecision::Unknown;
428            }
429        }
430        previous_width = Some(achieved_width);
431        target_width /= two.clone();
432    }
433    SignDecision::Unknown
434}

436fn single_algebraic_term_sign(
437    rational_part: &exact_rational,
438    coefficient: &exact_rational,
439    value: &AlgebraicNumber,
440) -> SignDecision {
441    // The sign of `p + c*alpha` for one real algebraic leaf alpha: the value is
442    // `c*(alpha - (-p/c))`, so comparing alpha against the rational `-p/c` by
443    // isolating-interval refinement decides it, with the direction flipped when c
444    // is negative.
445    let coefficient_sign = coefficient.sign_comparison();
446    if coefficient_sign == Ordering::Equal {
447        return rational_to_sign(rational_part);
448    }
449    let target = -(rational_part.clone() / coefficient.clone());
450    let Ok(ordering) = super::compare_algebraic_with_rational(value, &target) else {
451        return SignDecision::Unknown;
452    };
453    let ordering = if coefficient_sign == Ordering::Less {
454        ordering.reverse()
455    } else {
456        ordering
457    };
458    match ordering {
459        Ordering::Greater => SignDecision::Positive,
460        Ordering::Equal => SignDecision::Zero,
461        Ordering::Less => SignDecision::Negative,
462    }
463}

465pub(crate) fn rational_pi_affine_sign(
466    constant: &exact_rational,
467    pi_coefficient: &exact_rational,
468) -> SignDecision {
469    // The exact sign of `constant + pi_coefficient*pi`, decided against an exact
470    // rational bracket for pi that is refined until the resulting interval sits
471    // strictly on one side of zero. With a nonzero pi_coefficient the value is
472    // provably nonzero — pi is irrational, so it can never equal the rational
473    // -constant/pi_coefficient — and the bracket narrows without limit, so the
474    // refinement always terminates with a verdict for any separation above the
475    // budget's reach. Both endpoints stay exact rationals throughout, so a
476    // verdict never rests on a rounded value.
477    if rational_is_zero(pi_coefficient) {
478        return rational_to_sign(constant);
479    }
480    if rational_is_zero(constant) {
481        return rational_to_sign(pi_coefficient);
482    }
483    let two = rational_from_integer(exact_integer::from(2));
484    let magnitude = rational_absolute_value(pi_coefficient);
485    let mut target_width = rational_one();
486    for _ in 0..PI_AFFINE_REFINEMENT_BUDGET {
487        let allowance = target_width.clone() / magnitude.clone();
488        let (lower_pi, upper_pi) = pi_bounds_within(&allowance);
489        let first = constant.clone() + pi_coefficient.clone() * lower_pi;
490        let second = constant.clone() + pi_coefficient.clone() * upper_pi;
491        let lower = std::cmp::min(first.clone(), second.clone());
492        let upper = std::cmp::max(first, second);
493        if lower.sign_comparison() == Ordering::Greater {
494            return SignDecision::Positive;
495        }
496        if upper.sign_comparison() == Ordering::Less {
497            return SignDecision::Negative;
498        }
499        target_width /= two.clone();
500    }
501    SignDecision::Unknown
502}

504pub(crate) fn surd_sum_sign(
505    rational_part: &exact_rational,
506    surd_coefficient: &exact_rational,
507    radicand: &exact_rational,
508) -> SignDecision {
509    // The exact sign of `p + q*sqrt(r)` with `r > 0`. Same-sign p and q give that
510    // sign directly; opposite signs are resolved by squaring — `p + q*sqrt(r)`
511    // compares to 0 exactly as `p^2` compares to `q^2*r` (with the direction set
512    // by which of p, q is positive). All arithmetic is exact rational, so the
513    // verdict is rigorous.
514    let rational_sign = rational_to_sign(rational_part);
515    let coefficient_sign = rational_to_sign(surd_coefficient);
516    if coefficient_sign == SignDecision::Zero {
517        return rational_sign;
518    }
519    if rational_sign != SignDecision::Negative && coefficient_sign == SignDecision::Positive {
520        return SignDecision::Positive;
521    }
522    if rational_sign != SignDecision::Positive && coefficient_sign == SignDecision::Negative {
523        return SignDecision::Negative;
524    }
525    let rational_squared = rational_part.clone() * rational_part.clone();
526    let scaled_radicand = surd_coefficient.clone() * surd_coefficient.clone() * radicand.clone();
527    let comparison = rational_squared.cmp(&scaled_radicand);
528    if rational_sign == SignDecision::Positive {
529        match comparison {
530            Ordering::Greater => SignDecision::Positive,
531            Ordering::Equal => SignDecision::Zero,
532            Ordering::Less => SignDecision::Negative,
533        }
534    } else {
535        match comparison {
536            Ordering::Less => SignDecision::Positive,
537            Ordering::Equal => SignDecision::Zero,
538            Ordering::Greater => SignDecision::Negative,
539        }
540    }
541}

543pub(crate) fn rational_to_sign(value: &exact_rational) -> SignDecision {
544    // The exact sign of a rational as a sign_decision.
545    match value.sign_comparison() {
546        Ordering::Greater => SignDecision::Positive,
547        Ordering::Less => SignDecision::Negative,
548        Ordering::Equal => SignDecision::Zero,
549    }
550}

552fn is_one_half(exponent: &exact_rational) -> bool {
553    // True when the exponent is exactly the rational 1/2 (a square root).
554    exponent.clone() * rational_from_integer(exact_integer::from(2)) == rational_one()
555}

557fn normalise_surd(radicand: &exact_rational) -> Option<(exact_rational, exact_rational)> {
558    // Rewrite `sqrt(radicand)` for a positive rational radicand as
559    // `scale*sqrt(squarefree)` with squarefree a squarefree positive integer,
560    // by clearing the denominator (`sqrt(n/d) = sqrt(n*d)/d`) and pulling the
561    // largest square factor out of the resulting integer. Distinct squarefree
562    // radicands are what make the surd basis linearly independent over the
563    // rationals, which is the zero test the multi-term sign rule relies on, so a
564    // radicand too large to factor is declined rather than left unnormalised.
565    let parts = rational_parts(radicand);
566    let numerator = parts.numerator;
567    let denominator = natural_to_integer(parts.denominator);
568    let product = numerator * denominator.clone();
569    if integer_bit_length(&product) > MAXIMUM_RADICAND_BIT_LENGTH {
570        return None;
571    }
572    let factors = factor_integer_internal(&product)?;
573    let mut outside = exact_integer::from(1);
574    let mut inside = exact_integer::from(1);
575    for (prime, multiplicity) in factors {
576        for _ in 0..(multiplicity / 2) {
577            outside *= prime.clone();
578        }
579        if multiplicity % 2 == 1 {
580            inside *= prime;
581        }
582    }
583    let scale = exact_rational::from_integers(outside, denominator);
584    Some((scale, rational_from_integer(inside)))
585}

603    fn sign_of(context: &Context, source: &str) -> SignDecision {
604        match form_of(context, source) {
605            Some(form) => sign(&form),
606            None => SignDecision::Unknown,
607        }
608    }
609
610    fn difference_sign(context: &Context, left: &str, right: &str) -> SignDecision {
611        let left_form = form_of(context, left).expect("left operand is a Tier-1 form");
612        let right_form = form_of(context, right).expect("right operand is a Tier-1 form");
613        sign(&subtract(&left_form, &right_form))
614    }
