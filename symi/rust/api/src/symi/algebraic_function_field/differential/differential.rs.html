18pub(crate) struct AlgebraicDifferential {
19    // A differential f dx on the curve, represented by its coefficient f as a
20    // field element. Wrapping the coefficient keeps function orders and
21    // differential orders from being confused: derivation produces this type

35pub(crate) struct HyperellipticModel {
36    // A certified hyperelliptic presentation alpha^2 = radicand(x) of an
37    // algebraic function field. The radicand is dense ascending, squarefree, and nonconstant; the
38    // genus follows the exact formula floor((degree - 1) / 2), and the parity
39    // of the degree decides the infinity-place structure: odd degree has one
40    // ramified place at infinity, even degree has two.
41    pub(crate) field: Arc<SimpleAlgebraicFunctionField>,
42    pub(crate) radicand: Vec<exact_rational>,
43    pub(crate) genus: usize,
44    pub(crate) infinity_is_ramified: bool,
45}

47pub(crate) fn hyperelliptic_model_from_field(
48    field: &Arc<SimpleAlgebraicFunctionField>,
49) -> Option<HyperellipticModel> {
50    // Recognise the binomial quadratic defining polynomial alpha^2 - s(x) and
51    // certify the hyperelliptic invariants. Quadratics with a nonzero linear
52    // term and higher-degree defining polynomials decline: the complete
53    // divisor backend currently implemented is pinned to the squarefree
54    // hyperelliptic family.
55    if field.degree != 2 {
56        return None;
57    }
58    let mut radicand_polynomial = None;
59    for (exponent, coefficient) in &field.defining_polynomial.terms {
60        match *exponent {
61            2 => {
62                let numeric = crate::polynomial::coefficient_to_number(coefficient)?;
63                if !crate::number::is_one(&numeric) {
64                    return None;
65                }
66            }
67            1 => return None,
68            0 => {
69                let base_polynomial =
70                    super::field::coefficient_to_base_polynomial(coefficient, field.base_variable)?;
71                radicand_polynomial = Some(crate::polynomial::negate(&base_polynomial));
72            }
73            _ => return None,
74        }
75    }
76    let radicand_dense = rational_polynomial_to_dense(&radicand_polynomial?)?;
77    let rationals = RationalConstantField;
78    let degree = polynomial_degree::<RationalConstantField>(&radicand_dense)?;
79    if degree == 0 {
80        return None;
81    }
82    if !polynomial_is_squarefree_vector(&rationals, &radicand_dense) {
83        return None;
84    }
85    Some(HyperellipticModel {
86        field: Arc::clone(field),
87        radicand: radicand_dense,
88        genus: (degree - 1) / 2,
89        infinity_is_ramified: degree % 2 == 1,
90    })
91}

97pub(crate) fn split_even_odd(
98    target: &AlgebraicFunctionFieldElement,
99) -> Option<(RationalFunction, RationalFunction)> {
100    // Split an element of a quadratic field into its even part (the
101    // coefficient of 1) and its odd coefficient g (the coefficient of alpha,
102    // so the odd part is g * alpha). The split is the exact k = 2
103    // specialization of the trace decomposition: the even part is a rational
104    // function of x whose integral is always elementary, so the logarithmic
105    // problem lives entirely in the odd part.
106    if target.field.degree != 2 || target.representative.len() != 2 {
107        return None;
108    }
109    Some((
110        target.representative[0].clone(),
111        target.representative[1].clone(),
112    ))
113}

116pub(crate) struct HyperellipticOddDifferential {
117    // The canonical form (numerator / denominator) * dx / alpha of an odd
118    // differential g * alpha dx, with numerator and denominator coprime dense
119    // rational polynomials, the denominator monic and squarefree, and the
120    // denominator coprime to the radicand. Under those certificates the
121    // differential has at worst simple poles at finite places, no poles at
122    // branch places, and every remaining pole at infinity.
123    pub(crate) numerator: Vec<exact_rational>,
124    pub(crate) denominator: Vec<exact_rational>,
125}

127pub(crate) fn odd_differential_from_coefficient(
128    model: &HyperellipticModel,
129    odd_coefficient: &RationalFunction,
130) -> Option<HyperellipticOddDifferential> {
131    // Convert g = numerator/denominator (the alpha coefficient) into the
132    // canonical (A/B) dx/alpha presentation through g*alpha = g*s/alpha. The
133    // construction verifies rather than assumes the reducedness certificates:
134    // the incoming denominator must be squarefree (the residual
135    // guarantee), and after cancelling gcd(denominator, radicand) once the
136    // result is certified coprime and squarefree.
137    let rationals = RationalConstantField;
138    let numerator_dense = rational_polynomial_to_dense(&odd_coefficient.numerator)?;
139    let denominator_dense = rational_polynomial_to_dense(&odd_coefficient.denominator)?;
140    if denominator_dense.is_empty() {
141        return None;
142    }
143    if numerator_dense.is_empty() {
144        return Some(HyperellipticOddDifferential {
145            numerator: Vec::new(),
146            denominator: vec![rational_one()],
147        });
148    }
149    if !polynomial_is_squarefree_vector(&rationals, &denominator_dense) {
150        return None;
151    }
152    let branch_common = polynomial_gcd_vector(&rationals, &denominator_dense, &model.radicand);
153    let reduced_denominator =
154        exact_polynomial_quotient(&rationals, &denominator_dense, &branch_common)?;
155    let radicand_cofactor = exact_polynomial_quotient(&rationals, &model.radicand, &branch_common)?;
156    let numerator = multiply_polynomials(&rationals, &numerator_dense, &radicand_cofactor);
157    let common = polynomial_gcd_vector(&rationals, &numerator, &reduced_denominator);
158    let numerator = exact_polynomial_quotient(&rationals, &numerator, &common)?;
159    let denominator = exact_polynomial_quotient(&rationals, &reduced_denominator, &common)?;
160    let leading = denominator.last()?.clone();
161    if leading == rational_zero() {
162        return None;
163    }
164    let leading_inverse = rational_one() / leading;
165    let numerator = trim_polynomial(
166        &rationals,
167        numerator
168            .iter()
169            .map(|value| value.clone() * leading_inverse.clone())
170            .collect(),
171    );
172    let denominator = trim_polynomial(
173        &rationals,
174        denominator
175            .iter()
176            .map(|value| value.clone() * leading_inverse.clone())
177            .collect(),
178    );
179    let certified = odd_differential_certificates_hold(model, &numerator, &denominator);
180    if !certified {
181        return None;
182    }
183    Some(HyperellipticOddDifferential {
184        numerator,
185        denominator,
186    })
187}

189fn odd_differential_certificates_hold(
190    model: &HyperellipticModel,
191    numerator: &[exact_rational],
192    denominator: &[exact_rational],
193) -> bool {
194    // Re-verify every reducedness certificate from scratch: denominator monic,
195    // squarefree, coprime to both the numerator and the radicand. Failing any
196    // certificate makes the caller decline rather than proceed unsoundly.
197    let rationals = RationalConstantField;
198    if denominator.is_empty() {
199        return false;
200    }
201    if *denominator.last().unwrap() != rational_one() {
202        return false;
203    }
204    if !polynomial_is_squarefree_vector(&rationals, denominator) {
205        return false;
206    }
207    let numerator_common = polynomial_gcd_vector(&rationals, numerator, denominator);
208    if polynomial_degree::<RationalConstantField>(&numerator_common) != Some(0)
209        && !numerator.is_empty()
210    {
211        return false;
212    }
213    let branch_common = polynomial_gcd_vector(&rationals, denominator, &model.radicand);
214    polynomial_degree::<RationalConstantField>(&branch_common) == Some(0) || denominator.len() == 1
215}

217pub(crate) fn odd_coefficient_from_differential(
218    model: &HyperellipticModel,
219    differential: &HyperellipticOddDifferential,
220) -> Option<RationalFunction> {
221    // Invert the canonical presentation: (A/B) dx/alpha corresponds to the
222    // alpha-coefficient g = A / (B * s). Used to reconstruct field elements
223    // for the mandatory exact decomposition checks.
224    let variable = model.field.base_variable;
225    let numerator_polynomial =
226        super::constant_extension::dense_to_rational_polynomial(variable, &differential.numerator);
227    let rationals = RationalConstantField;
228    let denominator_dense =
229        multiply_polynomials(&rationals, &differential.denominator, &model.radicand);
230    let denominator_polynomial =
231        super::constant_extension::dense_to_rational_polynomial(variable, &denominator_dense);
232    RationalFunction::from_polynomial(numerator_polynomial)
233        .divide(&RationalFunction::from_polynomial(denominator_polynomial))
234}

236pub(crate) fn element_from_even_odd(
237    field: &Arc<SimpleAlgebraicFunctionField>,
238    even_part: &RationalFunction,
239    odd_coefficient: &RationalFunction,
240) -> AlgebraicFunctionFieldElement {
241    // Assemble a quadratic-field element from its even part and odd
242    // coefficient.
243    let mut result = super::field::zero_element(field);
244    result.representative[0] = even_part.clone();
245    result.representative[1] = odd_coefficient.clone();
246    result
