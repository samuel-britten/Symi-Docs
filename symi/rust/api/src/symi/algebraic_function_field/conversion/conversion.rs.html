18pub(crate) fn tower_rational_to_algebraic_function_element(
19    numerator: &Polynomial,
20    denominator: &Polynomial,
21    field: &Arc<SimpleAlgebraicFunctionField>,
22) -> Option<AlgebraicFunctionFieldElement> {
23    if polynomial_is_zero(denominator) {
24        return None;
25    }
26    let numerator_element = tower_polynomial_to_element(numerator, field)?;
27    let denominator_element = tower_polynomial_to_element(denominator, field)?;
28    divide(&numerator_element, &denominator_element)
29}

31pub(crate) fn algebraic_function_element_to_expression(
32    context: &Context,
33    target: &AlgebraicFunctionFieldElement,
34) -> ExpressionReference {
35    let generator_expression = context
36        .interner
37        .intern(ExpressionKind::Symbol(target.field.generator));
38    let mut terms = Vec::new();
39    for (exponent, coefficient) in target.representative.iter().enumerate() {
40        let lifted_terms = lift_quadratic_polynomial_coefficient(target, exponent, coefficient)
41            .unwrap_or_else(|| vec![(exponent, coefficient.clone())]);
42        for (lifted_exponent, lifted_coefficient) in lifted_terms {
43            if lifted_coefficient.is_zero() {
44                continue;
45            }
46            let coefficient_expression =
47                rational_coefficient_to_expression(context, &lifted_coefficient);
48            if lifted_exponent == 0 {
49                terms.push(coefficient_expression);
50                continue;
51            }
52            let generator_power = if lifted_exponent == 1 {
53                Arc::clone(&generator_expression)
54            } else {
55                let exponent_expression =
56                    builders::integer(context, exact_integer::from(lifted_exponent));
57                builders::power(
58                    context,
59                    Arc::clone(&generator_expression),
60                    exponent_expression,
61                )
62            };
63            terms.push(builders::product(
64                context,
65                vec![coefficient_expression, generator_power],
66            ));
67        }
68    }
69    if terms.is_empty() {
70        builders::zero(context)
71    } else if terms.len() == 1 {
72        terms.remove(0)
73    } else {
74        builders::sum(context, terms)
75    }
76}

78fn lift_quadratic_polynomial_coefficient(
79    target: &AlgebraicFunctionFieldElement,
80    exponent: usize,
81    coefficient: &RationalFunction,
82) -> Option<Vec<(usize, RationalFunction)>> {
83    // The lift rewrites `q(x) * t` into odd powers of `t` via the identity
84    // `t^2 = radicand`, which holds only for a binomial defining polynomial
85    // `t^2 - radicand`. A quadratic field with a nonzero linear coefficient
86    // (admissible under make_simple_algebraic_function_field) must decline
87    // here and keep the unlifted presentation.
88    if target.field.degree != 2
89        || exponent != 1
90        || degree(&coefficient.denominator).unwrap_or(0) != 0
91        || target
92            .field
93            .defining_polynomial
94            .terms
95            .iter()
96            .any(|(term_exponent, _)| *term_exponent == 1)
97    {
98        return None;
99    }
100    let constant_coefficient = target
101        .field
102        .defining_polynomial
103        .terms
104        .iter()
105        .find(|(term_exponent, _)| *term_exponent == 0)
106        .map(|(_, value)| value)?;
107    let radicand = crate::polynomial::negate(&coefficient_to_base_polynomial(
108        constant_coefficient,
109        target.field.base_variable,
110    )?);
111    if degree(&radicand).unwrap_or(0) == 0 {
112        return None;
113    }
114    let denominator = RationalFunction::from_polynomial(coefficient.denominator.clone());
115    let mut working = coefficient.numerator.clone();
116    let mut lifted_exponent = exponent;
117    let mut result = Vec::new();
118    while !polynomial_is_zero(&working) {
119        let (quotient, remainder) = polynomial_divide_with_remainder(&working, &radicand)?;
120        if !polynomial_is_zero(&remainder) {
121            let lifted_coefficient =
122                RationalFunction::from_polynomial(remainder).divide(&denominator)?;
123            result.push((lifted_exponent, lifted_coefficient));
124        }
125        working = quotient;
126        lifted_exponent += 2;
127    }
128    Some(result)
129}

131fn polynomial_divide_with_remainder(
132    dividend: &Polynomial,
133    divisor: &Polynomial,
134) -> Option<(Polynomial, Polynomial)> {
135    if polynomial_is_zero(divisor) || dividend.main_variable != divisor.main_variable {
136        return None;
137    }
138    let variable = dividend.main_variable;
139    let divisor_degree = degree(divisor)?;
140    let divisor_leading = coefficient_to_number(&leading_coefficient(divisor)?)?;
141    let mut quotient = crate::polynomial::zero_polynomial(variable);
142    let mut remainder = dividend.clone();
143    while let Some(remainder_degree) = degree(&remainder) {
144        if remainder_degree < divisor_degree {
145            break;
146        }
147        let remainder_leading = coefficient_to_number(&leading_coefficient(&remainder)?)?;
148        let quotient_coefficient = crate::number::divide(&remainder_leading, &divisor_leading)?;
149        let term = Polynomial {
150            main_variable: variable,
151            terms: vec![(
152                remainder_degree - divisor_degree,
153                PolynomialCoefficient::Numeric(quotient_coefficient),
154            )],
155        };
156        quotient = crate::polynomial::arithmetic::add(&quotient, &term);
157        remainder = crate::polynomial::arithmetic::subtract(
158            &remainder,
159            &crate::polynomial::arithmetic::multiply(&term, divisor),
160        );
161    }
162    Some((quotient, remainder))
163}

165pub(crate) fn tower_polynomial_to_element(
166    target: &Polynomial,
167    field: &Arc<SimpleAlgebraicFunctionField>,
168) -> Option<AlgebraicFunctionFieldElement> {
169    if polynomial_is_zero(target) {
170        return Some(zero_element(field));
171    }
172    if target.main_variable == field.base_variable {
173        return Some(from_coefficient(
174            field,
175            RationalFunction::from_polynomial(target.clone()),
176        ));
177    }
178    if target.main_variable != field.generator {
179        return None;
180    }
181    let generator = generator_element(field);
182    let mut result = zero_element(field);
183    for (exponent, coefficient) in &target.terms {
184        let base_polynomial = coefficient_to_base_polynomial(coefficient, field.base_variable)?;
185        let coefficient_element =
186            from_coefficient(field, RationalFunction::from_polynomial(base_polynomial));
187        let term = multiply(&coefficient_element, &power(&generator, *exponent)?)?;
188        result = add(&result, &term)?;
189    }
190    Some(result)
191}

193fn rational_coefficient_to_expression(
194    context: &Context,
195    target: &RationalFunction,
196) -> ExpressionReference {
197    let numerator = polynomial_to_expression(context, &target.numerator);
198    if target.denominator.terms.len() == 1
199        && target.denominator.terms[0].0 == 0
200        && crate::polynomial::coefficient_is_one(&target.denominator.terms[0].1)
201    {
202        numerator
203    } else {
204        let denominator = polynomial_to_expression(context, &target.denominator);
205        builders::divide(context, numerator, denominator)
206    }
207}
