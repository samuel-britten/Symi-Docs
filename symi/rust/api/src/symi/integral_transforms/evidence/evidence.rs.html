1// Proof-carrying results for the shared integral-transform subsystem.
2//
3// The historical public transform entry points return only an expression, so a
4// caller cannot tell a closed table pair apart from a structural placeholder
5// without re-inspecting the returned syntax, and cannot tell which theorem
6// produced a closed image or which hypotheses that theorem required. Callers
7// that must replay a transform step — a transform-domain differential-equation
8// reduction is the motivating one — need both.
9//
10// This module adds a structured path beside the existing dispatchers. The
11// expression-valued entry points keep their exact behaviour and remain the
12// projection of the structured result; nothing here changes what a transform
13// evaluates to.
14

42/// Public native enumeration named `integral_transform_theorem`.
43#[derive(Debug, Clone, PartialEq, Eq, Hash)]
44pub enum IntegralTransformTheorem {
45    /// Represents the transform of the exact zero expression.
46    Zero,
47    /// Represents a direct entry of the shipped transform table.
48    DirectTablePair,
49    /// Represents linearity over a sum.
50    Linearity,
51    /// Represents extraction of a factor constant in the source variable.
52    ConstantFactor,
53    /// Represents the frequency-shift theorem.
54    FrequencyShift,
55    /// Represents the translation theorem in the source variable.
56    Translation,
57    /// Represents the scaling theorem with its nonzero-coefficient hypothesis.
58    Scaling,
59    /// Represents the derivative theorem at the recorded order.
60    Derivative(usize),
61    /// Represents the convolution theorem.
62    Convolution,
63    /// Represents the multiplication-by-source-coordinate theorem.
64    MultiplicationBySourceCoordinate(usize),
65    /// Represents the rational partial-fraction inverse path.
66    RationalPartialFractionInverse,
67}

75            IntegralTransformTheorem::Linearity => "linearity",
76            IntegralTransformTheorem::ConstantFactor => "constant_factor",

80            IntegralTransformTheorem::Derivative(_) => "derivative",

164/// Public native enumeration named `integral_transform_attempt`.
165#[derive(Debug, Clone, PartialEq, Eq)]
166pub enum IntegralTransformAttempt {
167    /// Represents a closed step carrying its theorem evidence.
168    Closed(IntegralTransformCertificate),
169    /// Represents a step that left a structural transform placeholder.
170    Unevaluated(ExpressionReference),
171    /// Represents a step refused before any transform was attempted.
172    Declined(IntegralTransformDecline),
173}
174
175impl IntegralTransformAttempt {
176    /// Returns the certificate of a closed attempt.
177    pub fn certificate(&self) -> Option<&IntegralTransformCertificate> {
178        match self {
179            IntegralTransformAttempt::Closed(certificate) => Some(certificate),
180            _ => None,
181        }
182    }
183
184    /// Projects the attempt onto the expression the historical entry points return.
185    pub fn expression(&self) -> Option<ExpressionReference> {
186        match self {
187            IntegralTransformAttempt::Closed(certificate) => Some(certificate.result.clone()),
188            IntegralTransformAttempt::Unevaluated(placeholder) => Some(placeholder.clone()),
189            IntegralTransformAttempt::Declined(_) => None,
190        }
191    }
192}

194/// Computes a forward transform and reports the theorem that closed it.
195pub fn analyze_forward(
196    context: &Context,
197    family: IntegralTransformFamily,
198    target: ExpressionReference,
199    source_variable: SymbolIdentifier,
200    transform_variable: SymbolIdentifier,
201) -> IntegralTransformAttempt {
202    if source_variable == transform_variable {
203        return IntegralTransformAttempt::Declined(
204            IntegralTransformDecline::CoincidentTransformVariable,
205        );
206    }
207    if zero_equivalent(context, &target, None) == ZeroEquivalenceVerdict::TrueValue {
208        return closed_attempt(
209            family,
210            IntegralTransformDirection::Forward,
211            target,
212            crate::expression::builders::zero(context),
213            source_variable,
214            transform_variable,
215            vec![IntegralTransformTheorem::Zero],
216            Vec::new(),
217        );
218    }
219    let table_image = match family {
220        IntegralTransformFamily::Fourier => fourier::table::transform_table_entry(
221            context,
222            &target,
223            source_variable,
224            transform_variable,
225        ),
226        IntegralTransformFamily::Laplace => laplace::table::transform_table_entry(
227            context,
228            &target,
229            source_variable,
230            transform_variable,
231        ),
232    };
233    if let Some(image) = table_image {
234        return closed_attempt(
235            family,
236            IntegralTransformDirection::Forward,
237            target,
238            image,
239            source_variable,
240            transform_variable,
241            vec![IntegralTransformTheorem::DirectTablePair],
242            Vec::new(),
243        );
244    }
245    let rule_image = match family {
246        IntegralTransformFamily::Fourier => fourier::rules::apply_forward_rules_with_identity(
247            context,
248            &target,
249            source_variable,
250            transform_variable,
251        ),
252        IntegralTransformFamily::Laplace => laplace::rules::apply_forward_rules_with_identity(
253            context,
254            &target,
255            source_variable,
256            transform_variable,
257        ),
258    };
259    if let Some((image, identity)) = rule_image {
260        let conditions = identity_conditions(context, &identity);
261        if conditions
262            .iter()
263            .any(|condition| condition.verdict != TruthValue::Yes)
264        {
265            return IntegralTransformAttempt::Declined(
266                IntegralTransformDecline::HypothesisNotProved,
267            );
268        }
269        return closed_attempt(
270            family,
271            IntegralTransformDirection::Forward,
272            target,
273            image,
274            source_variable,
275            transform_variable,
276            vec![identity.theorem],
277            conditions,
278        );
279    }
280    let placeholder = match family {
281        IntegralTransformFamily::Fourier => crate::simplify::simplify_fourier_transform(
282            context,
283            target,
284            source_variable,
285            transform_variable,
286        ),
287        IntegralTransformFamily::Laplace => crate::simplify::simplify_laplace_transform(
288            context,
289            target,
290            source_variable,
291            transform_variable,
292        ),
293    };
294    IntegralTransformAttempt::Unevaluated(placeholder)
295}

297/// Computes an inverse transform and reports the theorem that closed it.
298pub fn analyze_inverse(
299    context: &Context,
300    family: IntegralTransformFamily,
301    target: ExpressionReference,
302    transform_variable: SymbolIdentifier,
303    source_variable: SymbolIdentifier,
304) -> IntegralTransformAttempt {
305    if source_variable == transform_variable {
306        return IntegralTransformAttempt::Declined(
307            IntegralTransformDecline::CoincidentTransformVariable,
308        );
309    }
310    if zero_equivalent(context, &target, None) == ZeroEquivalenceVerdict::TrueValue {
311        return closed_attempt(
312            family,
313            IntegralTransformDirection::Inverse,
314            target,
315            crate::expression::builders::zero(context),
316            source_variable,
317            transform_variable,
318            vec![IntegralTransformTheorem::Zero],
319            Vec::new(),
320        );
321    }
322    let table_image = match family {
323        IntegralTransformFamily::Fourier => fourier::inverse::inverse_table_entry(
324            context,
325            &target,
326            transform_variable,
327            source_variable,
328        ),
329        IntegralTransformFamily::Laplace => laplace::inverse::inverse_table_entry(
330            context,
331            &target,
332            transform_variable,
333            source_variable,
334        ),
335    };
336    if let Some(image) = table_image {
337        return closed_attempt(
338            family,
339            IntegralTransformDirection::Inverse,
340            target,
341            image,
342            source_variable,
343            transform_variable,
344            vec![IntegralTransformTheorem::DirectTablePair],
345            Vec::new(),
346        );
347    }
348    if let ExpressionKind::Sum(children) = &target.kind {
349        let children = children.clone();
350        let mut inverted_terms = Vec::with_capacity(children.len());
351        let mut every_term_closed = true;
352        for child in children {
353            let attempt =
354                analyze_inverse(context, family, child, transform_variable, source_variable);
355            match attempt {
356                IntegralTransformAttempt::Closed(certificate) => {
357                    inverted_terms.push(certificate.result)
358                }
359                _ => {
360                    every_term_closed = false;
361                    break;
362                }
363            }
364        }
365        if every_term_closed {
366            let image = crate::expression::builders::sum(context, inverted_terms);
367            return closed_attempt(
368                family,
369                IntegralTransformDirection::Inverse,
370                target,
371                image,
372                source_variable,
373                transform_variable,
374                vec![IntegralTransformTheorem::Linearity],
375                Vec::new(),
376            );
377        }
378    }
379    if let Some(image) = match family {
380        IntegralTransformFamily::Fourier => fourier::inverse::inverse_product_sum_linearity(
381            context,
382            &target,
383            transform_variable,
384            source_variable,
385        ),
386        IntegralTransformFamily::Laplace => None,
387    } {
388        return closed_attempt(
389            family,
390            IntegralTransformDirection::Inverse,
391            target,
392            image,
393            source_variable,
394            transform_variable,
395            vec![IntegralTransformTheorem::Linearity],
396            Vec::new(),
397        );
398    }
399    if let Some(image) = match family {
400        IntegralTransformFamily::Fourier => fourier::inverse::inverse_constant_factor(
401            context,
402            &target,
403            transform_variable,
404            source_variable,
405        ),
406        IntegralTransformFamily::Laplace => laplace::inverse::inverse_constant_factor(
407            context,
408            &target,
409            transform_variable,
410            source_variable,
411        ),
412    } {
413        return closed_attempt(
414            family,
415            IntegralTransformDirection::Inverse,
416            target,
417            image,
418            source_variable,
419            transform_variable,
420            vec![IntegralTransformTheorem::ConstantFactor],
421            Vec::new(),
422        );
423    }
424    if let Some(image) = match family {
425        IntegralTransformFamily::Fourier => fourier::inverse::inverse_exponential_constant_factor(
426            context,
427            &target,
428            transform_variable,
429            source_variable,
430        ),
431        IntegralTransformFamily::Laplace => None,
432    } {
433        return closed_attempt(
434            family,
435            IntegralTransformDirection::Inverse,
436            target,
437            image,
438            source_variable,
439            transform_variable,
440            vec![IntegralTransformTheorem::ConstantFactor],
441            Vec::new(),
442        );
443    }
444    if let Some(image) = match family {
445        IntegralTransformFamily::Fourier => fourier::inverse::inverse_unit_width_gaussian(
446            context,
447            &target,
448            transform_variable,
449            source_variable,
450        ),
451        IntegralTransformFamily::Laplace => None,
452    } {
453        return closed_attempt(
454            family,
455            IntegralTransformDirection::Inverse,
456            target,
457            image,
458            source_variable,
459            transform_variable,
460            vec![IntegralTransformTheorem::DirectTablePair],
461            Vec::new(),
462        );
463    }
464    if let Some(image) = match family {
465        IntegralTransformFamily::Fourier => fourier::inverse::inverse_sifted_delta_product(
466            context,
467            &target,
468            transform_variable,
469            source_variable,
470        ),
471        IntegralTransformFamily::Laplace => None,
472    } {
473        return closed_attempt(
474            family,
475            IntegralTransformDirection::Inverse,
476            target,
477            image,
478            source_variable,
479            transform_variable,
480            vec![IntegralTransformTheorem::DirectTablePair],
481            Vec::new(),
482        );
483    }
484    if let Some(image) = match family {
485        IntegralTransformFamily::Fourier => fourier::inverse::inverse_translated_product(
486            context,
487            &target,
488            transform_variable,
489            source_variable,
490        ),
491        IntegralTransformFamily::Laplace => None,
492    } {
493        return closed_attempt(
494            family,
495            IntegralTransformDirection::Inverse,
496            target,
497            image,
498            source_variable,
499            transform_variable,
500            vec![IntegralTransformTheorem::Translation],
501            Vec::new(),
502        );
503    }
504    let partial_fraction_image = match family {
505        IntegralTransformFamily::Fourier => fourier::inverse::inverse_via_partial_fractions(
506            context,
507            &target,
508            transform_variable,
509            source_variable,
510        ),
511        IntegralTransformFamily::Laplace => laplace::inverse::inverse_via_partial_fractions(
512            context,
513            &target,
514            transform_variable,
515            source_variable,
516        ),
517    };
518    if let Some(image) = partial_fraction_image {
519        return closed_attempt(
520            family,
521            IntegralTransformDirection::Inverse,
522            target,
523            image,
524            source_variable,
525            transform_variable,
526            vec![IntegralTransformTheorem::RationalPartialFractionInverse],
527            Vec::new(),
528        );
529    }
530    let placeholder = match family {
531        IntegralTransformFamily::Fourier => crate::simplify::simplify_inverse_fourier_transform(
532            context,
533            target,
534            transform_variable,
535            source_variable,
536        ),
537        IntegralTransformFamily::Laplace => crate::simplify::simplify_inverse_laplace_transform(
538            context,
539            target,
540            transform_variable,
541            source_variable,
542        ),
543    };
544    IntegralTransformAttempt::Unevaluated(placeholder)
545}

547/// Independently recomputes a certificate and rechecks each recorded hypothesis.
548pub fn replay_certificate(context: &Context, certificate: &IntegralTransformCertificate) -> bool {
549    // A certificate is replayable when the shipped dispatcher, run again from
550    // the recorded source, returns an image exactly equal to the recorded
551    // result, and when every recorded hypothesis still queries as proved. The
552    // replay deliberately uses the expression-valued entry points rather than
553    // the structured ones so the two paths cannot share a defect.
554    for condition in &certificate.applicability_conditions {
555        if evaluate_condition(context, &condition.condition) != TruthValue::Yes {
556            return false;
557        }
558    }
559    let recomputed = match (certificate.family, certificate.direction) {
560        (IntegralTransformFamily::Fourier, IntegralTransformDirection::Forward) => {
561            fourier::transform_forward(
562                context,
563                certificate.source.clone(),
564                certificate.source_variable,
565                certificate.transform_variable,
566            )
567        }
568        (IntegralTransformFamily::Fourier, IntegralTransformDirection::Inverse) => {
569            fourier::transform_inverse(
570                context,
571                certificate.source.clone(),
572                certificate.transform_variable,
573                certificate.source_variable,
574            )
575        }
576        (IntegralTransformFamily::Laplace, IntegralTransformDirection::Forward) => {
577            laplace::transform_forward(
578                context,
579                certificate.source.clone(),
580                certificate.source_variable,
581                certificate.transform_variable,
582            )
583        }
584        (IntegralTransformFamily::Laplace, IntegralTransformDirection::Inverse) => {
585            laplace::transform_inverse(
586                context,
587                certificate.source.clone(),
588                certificate.transform_variable,
589                certificate.source_variable,
590            )
591        }
592    };
593    if contains_transform_placeholder(&recomputed) {
594        return false;
595    }
596    if recomputed == certificate.result {
597        return true;
598    }
599    let difference =
600        crate::expression::builders::subtract(context, recomputed, certificate.result.clone());
601    if zero_equivalent(context, &difference, None) == ZeroEquivalenceVerdict::TrueValue {
602        return true;
603    }
604    let cancelled = crate::simplify::rational::cancel(context, &difference);
605    let expanded = crate::expand::expand(context, &cancelled);
606    let combined = crate::simplify::rational::together(context, &expanded);
607    let expanded_combined = crate::expand::expand(context, &combined);
608    let simplified = crate::simplify::simplify(context, expanded_combined);
609    zero_equivalent(context, &simplified, None) == ZeroEquivalenceVerdict::TrueValue
610}

612/// Reports whether an expression still carries a forward or inverse transform node.
613pub fn contains_transform_placeholder(expression: &ExpressionReference) -> bool {
614    // A solved transform route may leave no unresolved transform node behind.
615    // The predicate is structural on purpose: a transform node is exactly the
616    // syntax the dispatchers emit when no closed-form path fires.
617    if matches!(
618        expression.kind,
619        ExpressionKind::FourierTransform(_, _, _)
620            | ExpressionKind::InverseFourierTransform(_, _, _)
621            | ExpressionKind::LaplaceTransform(_, _, _)
622            | ExpressionKind::InverseLaplaceTransform(_, _, _)
623    ) {
624        return true;
625    }
626    visit_children(expression, contains_transform_placeholder)
627}
628
629fn visit_children(
630    expression: &ExpressionReference,
631    mut predicate: impl FnMut(&ExpressionReference) -> bool,
632) -> bool {
633    // Structural any-child fold. The integral-transform stack has no shared
634    // traversal helper of its own and the placeholder search must not silently
635    // skip an expression variant, so every variant is listed explicitly.
636    match &expression.kind {
637        ExpressionKind::Sum(children)
638        | ExpressionKind::Product(children)
639        | ExpressionKind::FiniteSet(children)
640        | ExpressionKind::SetUnion(children)
641        | ExpressionKind::SetIntersection(children)
642        | ExpressionKind::SpecialFunctionParameterVector(children)
643        | ExpressionKind::FunctionCall(_, children)
644        | ExpressionKind::Derivative(_, _, children)
645        | ExpressionKind::Distribution(_, children) => children.iter().any(&mut predicate),
646        ExpressionKind::Power(left, right)
647        | ExpressionKind::SetComplement(left, right)
648        | ExpressionKind::Interval(left, right, _, _)
649        | ExpressionKind::ImageSet(_, left, right)
650        | ExpressionKind::ConditionSet(_, left, right)
651        | ExpressionKind::OrdinaryDifferentialEquation(left, right, _)
652        | ExpressionKind::RecurrenceEquation(left, right, _) => predicate(left) || predicate(right),
653        ExpressionKind::Factorial(argument)
654        | ExpressionKind::RandomVariable(_, argument)
655        | ExpressionKind::LaplaceTransform(argument, _, _)
656        | ExpressionKind::InverseLaplaceTransform(argument, _, _)
657        | ExpressionKind::FourierTransform(argument, _, _)
658        | ExpressionKind::InverseFourierTransform(argument, _, _) => predicate(argument),
659        ExpressionKind::Integral(argument, _, lower, upper)
660        | ExpressionKind::Summation(argument, _, lower, upper)
661        | ExpressionKind::IndexedProduct(argument, _, lower, upper) => {
662            predicate(argument)
663                || lower.as_ref().is_some_and(&mut predicate)
664                || upper.as_ref().is_some_and(&mut predicate)
665        }
666        ExpressionKind::Piecewise(branches) => branches
667            .iter()
668            .any(|(condition, value)| predicate(condition) || predicate(value)),
669        ExpressionKind::Integer(_)
670        | ExpressionKind::Rational(_)
671        | ExpressionKind::PolynomialRoot(_)
672        | ExpressionKind::Symbol(_)
673        | ExpressionKind::Constant(_) => false,
674    }
675}

677/// One closing theorem together with the hypotheses that theorem required.
678pub(crate) struct TransformRuleIdentity {
679    /// Stores the theorem that closed the step.
680    pub theorem: IntegralTransformTheorem,
681    /// Stores expressions the theorem required to be nonzero.
682    pub nonzero_requirements: Vec<ExpressionReference>,
683}
684
685impl TransformRuleIdentity {
686    pub(crate) fn plain(theorem: IntegralTransformTheorem) -> Self {
687        Self {
688            theorem,
689            nonzero_requirements: Vec::new(),
690        }
691    }
692
693    pub(crate) fn requiring_nonzero(
694        theorem: IntegralTransformTheorem,
695        requirement: ExpressionReference,
696    ) -> Self {
697        Self {
698            theorem,
699            nonzero_requirements: vec![requirement],
700        }
701    }
702}
703
704fn identity_conditions(
705    context: &Context,
706    identity: &TransformRuleIdentity,
707) -> Vec<IntegralTransformApplicabilityCondition> {
708    identity
709        .nonzero_requirements
710        .iter()
711        .map(|requirement| {
712            let condition = IntegralTransformCondition::Nonzero(requirement.clone());
713            let verdict = evaluate_condition(context, &condition);
714            IntegralTransformApplicabilityCondition { condition, verdict }
715        })
716        .collect()
717}
718
719fn evaluate_condition(context: &Context, condition: &IntegralTransformCondition) -> TruthValue {
720    match condition {
721        IntegralTransformCondition::Nonzero(expression) => {
722            queries::is_nonzero(expression, &context.symbol_table)
723        }
724        IntegralTransformCondition::Positive(expression) => {
725            queries::is_positive(expression, &context.symbol_table)
726        }
727        IntegralTransformCondition::Real(expression) => {
728            queries::is_real(expression, &context.symbol_table)
729        }
730    }
731}

733#[allow(clippy::too_many_arguments, reason = "one certificate constructor")]
734fn closed_attempt(
735    family: IntegralTransformFamily,
736    direction: IntegralTransformDirection,
737    source: ExpressionReference,
738    result: ExpressionReference,
739    source_variable: SymbolIdentifier,
740    transform_variable: SymbolIdentifier,
741    theorems: Vec<IntegralTransformTheorem>,
742    applicability_conditions: Vec<IntegralTransformApplicabilityCondition>,
743) -> IntegralTransformAttempt {
744    let proof_steps = theorems
745        .into_iter()
746        .map(|theorem| IntegralTransformProofStep {
747            theorem,
748            source: source.clone(),
749            result: result.clone(),
750            applicability_conditions: applicability_conditions.clone(),
751        })
752        .collect();
753    IntegralTransformAttempt::Closed(IntegralTransformCertificate {
