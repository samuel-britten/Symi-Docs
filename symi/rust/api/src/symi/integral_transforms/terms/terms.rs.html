1// Ordered execution of the independent additive terms of one integral transform.
2//
3// Linearity is the only coarse boundary an integral transform offers: a transform of a sum is the
4// sum of the transforms of its terms, and each term re-enters the whole dispatcher — the table,
5// then the algebraic rule stack, then for an inverse a partial-fraction decomposition — without
6// reading anything the sibling terms produce. This module owns the one coordinator that both the
7// Laplace and the Fourier families use for that boundary, so the two do not grow separate
8// schedulers and so the exactness rule below is stated once.
9//
10// The rule is that linearity commits all of its terms or none of them. A term that does not close
11// leaves the structural placeholder, and a sum of placeholders is not the transform of the sum: it
12// claims a decomposition the transform never proved. The rule is written once, in
13// `settle_transform_term` and `completed_linearity`, and the two execution adapters below differ
14// only in how they reach it: the serial adapter borrows the caller's terms and stops at the first
15// refusal, and the parallel adapter owns the terms it hands to the pool and merges their outcomes
16// in canonical term order. Which term refuses first is a property of canonical term order rather
17// than of completion order, so both adapters answer alike.
18

29/// Names the decline of one additive transform term that did not close.
30#[derive(Clone, Copy, Debug, Eq, PartialEq)]
31pub(crate) enum TransformTermDecline {
32    /// The term left the structural transform placeholder rather than a closed form.
33    NotRecognised,
34}
35
36/// The settled state of one additive term under the linearity rule.
37enum TransformTermSettlement {
38    /// The term closed, and its image is the term's contribution to the image of the sum.
39    Closed(ExpressionReference),
40    /// The term left a placeholder, which refuses the whole decomposition.
41    Refused,
42}
43
44pub(crate) fn transform_terms(
45    context: &Context,
46    terms: &[ExpressionReference],
47    operation_family: &'static str,
48    completeness_source: &'static str,
49    transform_term: impl Fn(&ExpressionReference) -> Option<ExpressionReference> + Send + Sync,
50) -> Option<Vec<ExpressionReference>> {
51    // Transform every additive term in canonical source order, returning None the moment any term
52    // declines. `transform_term` returns the closed form of one term or None when the term left a
53    // placeholder; it must not consult sibling terms, because the terms of a formed batch may run
54    // concurrently and in any order.
55    match transform_term_route(context, terms.len()) {
56        ExecutionRoute::Serial => transform_terms_serially(context, terms, &transform_term),
57        ExecutionRoute::Parallel => transform_terms_in_batch(
58            context,
59            terms,
60            operation_family,
61            completeness_source,
62            &transform_term,
63        ),
64    }
65}

67fn settle_transform_term(
68    term: &ExpressionReference,
69    branch_index: usize,
70    transform_term: &(impl Fn(&ExpressionReference) -> Option<ExpressionReference> + Send + Sync),
71) -> TransformTermSettlement {
72    // The one predicate that recognizes a closed term. Both adapters reach the linearity rule
73    // through here, so neither can decide on its own what counts as a term that closed, and the
74    // logical identity a delayed-work control names is the same on both routes.
75    crate::instrumentation::apply_structured_domain_test_control(
76        crate::instrumentation::StructuredDomainWorkKind::IntegralTransformTerm,
77        branch_index,
78    );
79    match transform_term(term) {
80        Some(image) => TransformTermSettlement::Closed(image),
81        None => TransformTermSettlement::Refused,
82    }
83}

85fn completed_linearity(
86    images: Vec<ExpressionReference>,
87    term_count: usize,
88) -> Option<Vec<ExpressionReference>> {
89    // The one completeness check. Linearity is only sound when every term of the sum was
90    // accounted for, so the count is checked rather than assumed on both routes. A settled serial
91    // batch reports no outcome at all for the terms it never ran, and today only a refusal settles
92    // this boundary — which both adapters have already turned into None before they arrive here.
93    // The check therefore does not fire on any settlement reachable now; it is what keeps a future
94    // settlement rule from silently publishing the transform of a proper subset of the sum as the
95    // transform of the sum.
96    (images.len() == term_count).then_some(images)
97}

99fn transform_terms_serially(
100    context: &Context,
101    terms: &[ExpressionReference],
102    transform_term: &(impl Fn(&ExpressionReference) -> Option<ExpressionReference> + Send + Sync),
103) -> Option<Vec<ExpressionReference>> {
104    // The borrowed adapter. It walks the caller's terms in canonical order and appends the closed
105    // images directly, so a transform that cannot fork forms no logical task identity, no owned
106    // task input, no priority batch, no evidence or condition ledger and no indexed outcome — the
107    // envelopes exist to carry work between threads, and there is no second thread here.
108    let mut images = Vec::with_capacity(terms.len());
109    for (branch_index, term) in terms.iter().enumerate() {
110        if context.operation().is_cancelled() {
111            return None;
112        }
113        match settle_transform_term(term, branch_index, transform_term) {
114            TransformTermSettlement::Closed(image) => images.push(image),
115            TransformTermSettlement::Refused => return None,
116        }
117    }
118    completed_linearity(images, terms.len())
119}

121fn transform_terms_in_batch(
122    context: &Context,
123    terms: &[ExpressionReference],
124    operation_family: &'static str,
125    completeness_source: &'static str,
126    transform_term: &(impl Fn(&ExpressionReference) -> Option<ExpressionReference> + Send + Sync),
127) -> Option<Vec<ExpressionReference>> {
128    // The owned adapter. Every term of a formed batch runs, and the outcomes are merged in
129    // canonical term order, so the refusal the serial adapter stops at is the same refusal this
130    // one finds among outcomes that may have completed in any order.
131    let tasks = terms
132        .iter()
133        .enumerate()
134        .map(|(branch_index, term)| {
135            SearchTask::new(
136                SearchTaskIdentifier::new(operation_family, 0, 1, branch_index, "additive term"),
137                Arc::clone(term),
138            )
139        })
140        .collect::<Vec<_>>();
141    crate::instrumentation::record_counter!(integral_transform_task_records, tasks.len());
142    let batch = SearchPriorityBatch::new(1, tasks);
143    crate::instrumentation::record_counter!(integral_transform_priority_batches);
144    let outcomes: Vec<
145        crate::parallelism::IndexedSearchTaskOutcome<ExpressionReference, (), TransformTermDecline>,
146    > = execute_search_batches_with_settlement(
147        ExecutionRoute::Parallel,
148        &[batch],
149        &context.operation().cancellation_token(),
150        |term, identifier| match settle_transform_term(
151            term,
152            identifier.branch_index,
153            transform_term,
154        ) {
155            TransformTermSettlement::Closed(image) => SearchTaskOutcome::Candidate {
156                value: image,
157                evidence: (),
158                conditions: SearchConditionLedger::empty(),
159                completeness_contribution: SearchCompletenessContribution::new(
160                    completeness_source,
161                    true,
162                    true,
163                ),
164                quality_key: vec![identifier.branch_index as u64],
165            },
166            TransformTermSettlement::Refused => SearchTaskOutcome::Declined {
167                reason: TransformTermDecline::NotRecognised,
168                explored_boundary: completeness_source,
169            },
170        },
171        |outcome| match outcome {
172            SearchTaskOutcome::Candidate { .. } => SearchSettlement::RequiresRemainingPeers,
173            _ => SearchSettlement::SettlesBatch,
174        },
175    )
176    .ok()?;
177    let mut images: Vec<ExpressionReference> = Vec::with_capacity(terms.len());
178    crate::instrumentation::record_counter!(integral_transform_result_vectors);
179    for indexed_outcome in outcomes {
180        match indexed_outcome.outcome {
181            SearchTaskOutcome::Candidate { value, .. } => images.push(value),
182            SearchTaskOutcome::ProvedEmpty { .. }
183            | SearchTaskOutcome::Declined { .. }
184            | SearchTaskOutcome::ResourceExhausted
185            | SearchTaskOutcome::Cancelled => return None,
186        }
187    }
188    completed_linearity(images, terms.len())
189}

191fn transform_term_route(context: &Context, term_count: usize) -> ExecutionRoute {
192    // The estimate is the term count squared: each term re-enters the dispatcher over a target
193    // whose own size grows with the sum it was taken from, so a wider sum costs more per term as
194    // well as more terms. The cutoff that reads this estimate is recorded with the measurement
195    // that placed it alongside the other work classes.
196    //
197    // A transform that cannot fork asks that first, from a handful of field reads, and never
198    // builds the estimate: this boundary is reached on calls of a few microseconds, so the
199    // estimate and the policy it feeds are a measurable share of a serial transform rather than a
200    // rounding error against the work they schedule.
201    if term_count < 2
202        || !context
203            .operation()
204            .may_fork_work_class("integral_transform_term")
205    {
206        return ExecutionRoute::Serial;
207    }
208    let work_estimate = WorkEstimate::new(
209        (term_count as u64).saturating_mul(term_count as u64),
210        term_count,
211        term_count.saturating_mul(std::mem::size_of::<ExpressionReference>()),
212        0,
213        term_count,
214        "integral_transform_term",
215    );
216    context
217        .operation()
218        .execution_decision_silent(
219            &work_estimate,
220            term_count,
221            crate::parallelism::runtime_memory_capacity(work_estimate.predicted_task_local_bytes),
222        )
223        .route
224}
