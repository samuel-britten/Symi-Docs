1// Forward Laplace table. Each entry recognises a textbook closed-form
2// time-domain shape and returns the frequency-domain image expressed over
3// the supplied `frequency_variable`. The table is purely structural — no
4// table entry consults the assumption engine beyond the positivity guards
5// the textbook entries themselves require. Unrecognised shapes return
6// None so the dispatcher in [super::transform_forward] falls through to
7// the algebraic rule layer.

22pub(crate) fn transform_table_entry(
23    context: &Context,
24    target: &ExpressionReference,
25    time_variable: SymbolIdentifier,
26    frequency_variable: SymbolIdentifier,
27) -> Option<ExpressionReference> {
28    // Dispatch a target expression against each table entry in priority
29    // order. The order matters only for inputs that match multiple shapes
30    // simultaneously (none currently do) — the per-entry recogniser is
31    // narrow enough to prevent overlap. Returns None on every unrecognised
32    // input so the caller can fall through to the algebraic rule layer.
33    if let Some(image) = match_constant_one(context, target, frequency_variable) {
34        return Some(image);
35    }
36    if let Some(image) = match_time_power(context, target, time_variable, frequency_variable) {
37        return Some(image);
38    }
39    if let Some(image) = match_exponential(context, target, time_variable, frequency_variable) {
40        return Some(image);
41    }
42    if let Some(image) = match_sine_or_cosine(context, target, time_variable, frequency_variable) {
43        return Some(image);
44    }
45    if let Some(image) = match_hyperbolic(context, target, time_variable, frequency_variable) {
46        return Some(image);
47    }
48    if let Some(image) = match_heaviside_step(context, target, time_variable, frequency_variable) {
49        return Some(image);
50    }
51    if let Some(image) = match_dirac_delta(context, target, time_variable, frequency_variable) {
52        return Some(image);
53    }
54    if let Some(image) = match_zero_order_bessel(context, target, time_variable, frequency_variable)
55    {
56        return Some(image);
57    }
58    None
59}

61fn match_zero_order_bessel(
62    context: &Context,
63    target: &ExpressionReference,
64    time_variable: SymbolIdentifier,
65    frequency_variable: SymbolIdentifier,
66) -> Option<ExpressionReference> {
67    let (function, arguments) = function_call_components(target)?;
68    if arguments.len() != 2
69        || !matches!(&arguments[0].kind, ExpressionKind::Integer(value) if *value == Integer::from(0))
70        || !is_time_symbol(&arguments[1], time_variable)
71    {
72        return None;
73    }
74    let frequency = symbol_expression(context, frequency_variable);
75    let one = builders::integer_value(context, 1);
76    let two = builders::integer_value(context, 2);
77    let frequency_square = builders::power(context, Arc::clone(&frequency), two);
78    let radicand = match function {
79        FunctionIdentifier::Builtin(BuiltinFunction::BesselJ) => {
80            if crate::assumptions::range_evidence::positive_verdict(
81                &context.symbol_table,
82                &frequency,
83            ) != TruthValue::Yes
84            {
85                return None;
86            }
87            builders::sum(context, vec![frequency_square, one])
88        }
89        FunctionIdentifier::Builtin(BuiltinFunction::BesselI) => {
90            let convergence_margin = builders::subtract(context, Arc::clone(&frequency), one);
91            if crate::assumptions::range_evidence::positive_verdict(
92                &context.symbol_table,
93                &convergence_margin,
94            ) != TruthValue::Yes
95            {
96                return None;
97            }
98            let one = builders::integer_value(context, 1);
99            builders::subtract(context, frequency_square, one)
100        }
101        _ => return None,
102    };
103    let negative_one_half = builders::rational_value(context, -1, 2);
104    Some(builders::power(context, radicand, negative_one_half))
105}

107fn match_constant_one(
108    context: &Context,
109    target: &ExpressionReference,
110    frequency_variable: SymbolIdentifier,
111) -> Option<ExpressionReference> {
112    // L{1} = 1/s. Recognised as the integer literal one (the only ASAE
113    // representation of the multiplicative identity at the top level).
114    if let ExpressionKind::Integer(value) = &target.kind {
115        if *value == Integer::from(1) {
116            let frequency = symbol_expression(context, frequency_variable);
117            let one = builders::integer_value(context, 1);
118            return Some(builders::divide(context, one, frequency));
119        }
120    }
121    None
122}

124fn match_time_power(
125    context: &Context,
126    target: &ExpressionReference,
127    time_variable: SymbolIdentifier,
128    frequency_variable: SymbolIdentifier,
129) -> Option<ExpressionReference> {
130    // L{t^n} = n! / s^{n+1} for nonnegative integer n; L{t} as a bare
131    // symbol degenerates to the n = 1 case. Non-integer or negative
132    // exponents fall through so the gamma-family entry (deferred) is the
133    // only natural extension point.
134    let (exponent_integer, is_bare) = match &target.kind {
135        ExpressionKind::Symbol(identifier) if *identifier == time_variable => {
136            (Integer::from(1), true)
137        }
138        ExpressionKind::Power(base, exponent) => {
139            if !is_time_symbol(base, time_variable) {
140                return None;
141            }
142            let ExpressionKind::Integer(value) = &exponent.kind else {
143                return None;
144            };
145            if value.sign_comparison() == std::cmp::Ordering::Less {
146                return None;
147            }
148            (value.clone(), false)
149        }
150        _ => return None,
151    };
152    let _ = is_bare;
153    let factorial_value = factorial_integer(&exponent_integer)?;
154    let frequency = symbol_expression(context, frequency_variable);
155    let numerator = builders::integer(context, factorial_value);
156    let new_exponent_integer = exponent_integer.clone() + Integer::from(1);
157    let denominator_exponent = builders::integer(context, new_exponent_integer);
158    let denominator = builders::power(context, frequency, denominator_exponent);
159    Some(builders::divide(context, numerator, denominator))
160}

162fn match_exponential(
163    context: &Context,
164    target: &ExpressionReference,
165    time_variable: SymbolIdentifier,
166    frequency_variable: SymbolIdentifier,
167) -> Option<ExpressionReference> {
168    // L{exp(a*t)} = 1/(s - a). The recognition allows a to be any
169    // expression that is constant with respect to `time_variable`. The
170    // bare-exp shape exp(t) corresponds to a = 1.
171    let (function, arguments) = function_call_components(target)?;
172    if !matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Exp)) {
173        return None;
174    }
175    if arguments.len() != 1 {
176        return None;
177    }
178    let coefficient = extract_constant_coefficient_of(context, &arguments[0], time_variable)?;
179    let frequency = symbol_expression(context, frequency_variable);
180    let denominator = builders::subtract(context, frequency, coefficient);
181    let one = builders::integer_value(context, 1);
182    Some(builders::divide(context, one, denominator))
183}

185fn match_sine_or_cosine(
186    context: &Context,
187    target: &ExpressionReference,
188    time_variable: SymbolIdentifier,
189    frequency_variable: SymbolIdentifier,
190) -> Option<ExpressionReference> {
191    // L{sin(omega*t)} = omega/(s^2 + omega^2) and
192    // L{cos(omega*t)} = s/(s^2 + omega^2). The angular frequency `omega`
193    // must be a constant with respect to `time_variable`; bare sin(t) and
194    // cos(t) correspond to omega = 1.
195    let (function, arguments) = function_call_components(target)?;
196    if arguments.len() != 1 {
197        return None;
198    }
199    let is_sine = matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Sin));
200    let is_cosine = matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Cos));
201    if !is_sine && !is_cosine {
202        return None;
203    }
204    let omega = extract_constant_coefficient_of(context, &arguments[0], time_variable)?;
205    let frequency = symbol_expression(context, frequency_variable);
206    let two = builders::integer_value(context, 2);
207    let frequency_squared = builders::power(context, Arc::clone(&frequency), Arc::clone(&two));
208    let omega_squared = builders::power(context, Arc::clone(&omega), two);
209    let denominator = builders::sum(context, vec![frequency_squared, omega_squared]);
210    if is_sine {
211        Some(builders::divide(context, omega, denominator))
212    } else {
213        Some(builders::divide(context, frequency, denominator))
214    }
215}

217fn match_hyperbolic(
218    context: &Context,
219    target: &ExpressionReference,
220    time_variable: SymbolIdentifier,
221    frequency_variable: SymbolIdentifier,
222) -> Option<ExpressionReference> {
223    // L{sinh(a*t)} = a/(s^2 - a^2) and L{cosh(a*t)} = s/(s^2 - a^2).
224    // The recognition mirrors the sin/cos arm but with the sign on the
225    // a-squared term flipped to negative.
226    let (function, arguments) = function_call_components(target)?;
227    if arguments.len() != 1 {
228        return None;
229    }
230    let is_sinh = matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Sinh));
231    let is_cosh = matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Cosh));
232    if !is_sinh && !is_cosh {
233        return None;
234    }
235    let coefficient = extract_constant_coefficient_of(context, &arguments[0], time_variable)?;
236    let frequency = symbol_expression(context, frequency_variable);
237    let two = builders::integer_value(context, 2);
238    let frequency_squared = builders::power(context, Arc::clone(&frequency), Arc::clone(&two));
239    let coefficient_squared = builders::power(context, Arc::clone(&coefficient), two);
240    let denominator = builders::subtract(context, frequency_squared, coefficient_squared);
241    if is_sinh {
242        Some(builders::divide(context, coefficient, denominator))
243    } else {
244        Some(builders::divide(context, frequency, denominator))
245    }
246}

248fn match_heaviside_step(
249    context: &Context,
250    target: &ExpressionReference,
251    time_variable: SymbolIdentifier,
252    frequency_variable: SymbolIdentifier,
253) -> Option<ExpressionReference> {
254    // L{heaviside_step(t - a)} = exp(-a*s)/s for a >= 0. The shift offset
255    // is required to be nonnegative — a negative offset would push the
256    // jump into the t < 0 region that the standard one-sided Laplace
257    // transform ignores. The recognition accepts the bare H(t) form
258    // (a = 0) and any t + (-a) where the constant arm is provably
259    // nonnegative.
260    let (function, arguments) = function_call_components(target)?;
261    if !matches!(
262        function,
263        FunctionIdentifier::Builtin(BuiltinFunction::HeavisideStep)
264    ) {
265        return None;
266    }
267    if arguments.len() != 1 {
268        return None;
269    }
270    let shift = extract_nonnegative_shift_of(context, &arguments[0], time_variable)?;
271    let frequency = symbol_expression(context, frequency_variable);
272    let neg_shift = builders::negate(context, shift);
273    let exponent_argument = builders::product(context, vec![neg_shift, Arc::clone(&frequency)]);
274    let exponential = builders::function_call(
275        context,
276        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
277        vec![exponent_argument],
278    );
279    Some(builders::divide(context, exponential, frequency))
280}

282fn match_dirac_delta(
283    context: &Context,
284    target: &ExpressionReference,
285    time_variable: SymbolIdentifier,
286    frequency_variable: SymbolIdentifier,
287) -> Option<ExpressionReference> {
288    // L{dirac_delta(t - a)} = exp(-a*s) for a >= 0. The same nonnegative
289    // shift guard applies as for heaviside_step.
290    let (function, arguments) = function_call_components(target)?;
291    if !matches!(
292        function,
293        FunctionIdentifier::Builtin(BuiltinFunction::DiracDelta)
294    ) {
295        return None;
296    }
297    if arguments.len() != 1 {
298        return None;
299    }
300    let shift = extract_nonnegative_shift_of(context, &arguments[0], time_variable)?;
301    let frequency = symbol_expression(context, frequency_variable);
302    let neg_shift = builders::negate(context, shift);
303    let exponent_argument = builders::product(context, vec![neg_shift, frequency]);
304    Some(builders::function_call(
305        context,
306        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
307        vec![exponent_argument],
308    ))
309}

311pub(crate) fn symbol_expression(
312    context: &Context,
313    identifier: SymbolIdentifier,
314) -> ExpressionReference {
315    // Build the bare-symbol expression for the given identifier. Used by
316    // every table entry that needs to splice the frequency variable into
317    // the image expression.
318    context.interner.intern(ExpressionKind::Symbol(identifier))
319}

321pub(crate) fn function_call_components(
322    target: &ExpressionReference,
323) -> Option<(FunctionIdentifier, Vec<ExpressionReference>)> {
324    // Project a function-call expression to its head function identifier
325    // and argument vector. Returns None for any other shape so callers
326    // can fail through cleanly when the input is not a call.
327    if let ExpressionKind::FunctionCall(function, arguments) = &target.kind {
328        return Some((*function, arguments.iter().cloned().collect()));
329    }
330    None
331}

333pub(crate) fn is_time_symbol(
334    target: &ExpressionReference,
335    time_variable: SymbolIdentifier,
336) -> bool {
337    // Predicate identifying the bare time-variable symbol leaf used by
338    // the table's `t^n` and `bare t` entries.
339    matches!(
340        &target.kind,
341        ExpressionKind::Symbol(identifier) if *identifier == time_variable
342    )
343}

345pub(crate) fn extract_constant_coefficient_of(
346    context: &Context,
347    argument: &ExpressionReference,
348    time_variable: SymbolIdentifier,
349) -> Option<ExpressionReference> {
350    // Decompose `argument` into a constant-coefficient * time_variable
351    // form and return the coefficient. Three structural shapes are
352    // recognised: the bare time symbol itself (coefficient = 1), a
353    // product whose factor list contains exactly one occurrence of the
354    // time symbol and whose remaining factors are constant with respect
355    // to it, and an arithmetic combination that does not depend on the
356    // time variable at all (treated as a zero-coefficient case is NOT
357    // applied here — the absence of the time variable would make the
358    // table entry's domain assumption fail). Returns None for any other
359    // shape.
360    if matches!(
361        &argument.kind,
362        ExpressionKind::Symbol(identifier) if *identifier == time_variable
363    ) {
364        return Some(builders::integer_value(context, 1));
365    }
366    if let ExpressionKind::Product(factors) = &argument.kind {
367        let mut time_count = 0usize;
368        let mut coefficient_factors: Vec<ExpressionReference> = Vec::new();
369        for factor in factors {
370            if matches!(
371                &factor.kind,
372                ExpressionKind::Symbol(identifier) if *identifier == time_variable
373            ) {
374                time_count += 1;
375                continue;
376            }
377            if expression_depends_on_variable(factor, time_variable) {
378                return None;
379            }
380            coefficient_factors.push(Arc::clone(factor));
381        }
382        if time_count != 1 {
383            return None;
384        }
385        if coefficient_factors.is_empty() {
386            return Some(builders::integer_value(context, 1));
387        }
388        return Some(builders::product(context, coefficient_factors));
389    }
390    None
391}

393pub(crate) fn extract_nonnegative_shift_of(
394    context: &Context,
395    argument: &ExpressionReference,
396    time_variable: SymbolIdentifier,
397) -> Option<ExpressionReference> {
398    // Decompose `argument` as `t + offset` or `t - offset` and return
399    // `offset` (always as a nonnegative-shift contribution: the rule
400    // requires the recognised inner argument to be t - a with a >= 0,
401    // and the negation is undone here so the caller receives the bare
402    // a). The bare t case returns offset = 0.
403    if matches!(
404        &argument.kind,
405        ExpressionKind::Symbol(identifier) if *identifier == time_variable
406    ) {
407        return Some(builders::integer_value(context, 0));
408    }
409    if let ExpressionKind::Sum(children) = &argument.kind {
410        let mut time_count = 0usize;
411        let mut residue: Vec<ExpressionReference> = Vec::new();
412        for child in children {
413            if matches!(
414                &child.kind,
415                ExpressionKind::Symbol(identifier) if *identifier == time_variable
416            ) {
417                time_count += 1;
418                continue;
419            }
420            if expression_depends_on_variable(child, time_variable) {
421                return None;
422            }
423            residue.push(Arc::clone(child));
424        }
425        if time_count != 1 {
426            return None;
427        }
428        let residue_expression = match residue.len() {
429            0 => return Some(builders::integer_value(context, 0)),
430            1 => residue.into_iter().next().unwrap(),
431            _ => builders::sum(context, residue),
432        };
433        // The residue is the constant offset c in t + c; the textbook
434        // convention is t - a so a = -c. Require a >= 0, i.e. c <= 0.
435        let negated_residue = builders::negate(context, Arc::clone(&residue_expression));
436        if let Some(value) = numeric_value(&negated_residue) {
437            if !number::is_negative(&value) {
438                return Some(negated_residue);
439            }
440            return None;
441        }
442        let positivity = crate::assumptions::range_evidence::positive_verdict(
443            &context.symbol_table,
444            &negated_residue,
445        );
446        if positivity == TruthValue::Yes {
447            return Some(negated_residue);
448        }
449        let zero_value = builders::zero(context);
450        if Arc::ptr_eq(&negated_residue, &zero_value) {
451            return Some(negated_residue);
452        }
453        return None;
454    }
455    None
456}

458pub(crate) fn expression_depends_on_variable(
459    target: &ExpressionReference,
460    variable: SymbolIdentifier,
461) -> bool {
462    // Predicate identifying any subtree that contains the named symbol
463    // as a free variable. Reuses the existing free-variable enumerator
464    // so binder-introducing nodes — integrals, summations, the Laplace
465    // pair — correctly mask their own bound variables.
466    crate::expression::free_variables::free_variables(target).contains(&variable)
467}

469fn factorial_integer(value: &Integer) -> Option<Integer> {
470    // Compute n! for a nonnegative integer value. Used by the time-power
471    // table entry. Returns None on negative input so the caller fails
472    // through to the next entry rather than fabricating a value.
473    if value.sign_comparison() == std::cmp::Ordering::Less {
474        return None;
475    }
476    let count = u64::try_from(value).ok()?;
477    let mut accumulator = Integer::from(1);
478    for index in 2..=count {
479        accumulator *= Integer::from(index);
480    }
481    Some(accumulator)
482}
483
484fn numeric_value(target: &ExpressionReference) -> Option<number_value> {
485    // Project a numeric leaf to its underlying number value. Returns None
486    // for any non-numeric expression so the caller can detect that the
487    // sign verdict has to come from the assumption engine instead.
488    match &target.kind {
489        ExpressionKind::Integer(value) => Some(number_value::Integer(value.clone())),
490        ExpressionKind::Rational(value) => Some(number_value::Rational(value.clone())),
491        _ => None,
492    }
493}

497    // Convenience accessor for the imaginary-unit constant. Used by the
