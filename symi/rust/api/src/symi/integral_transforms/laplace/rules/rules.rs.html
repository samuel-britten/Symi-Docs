1// Forward Laplace algebraic rules. Each rule recognises a structural shape on
2// the time-domain side and returns the frequency-domain image expressed
3// over the supplied `frequency_variable`. The rules form the second tier
4// of the forward dispatcher in [super::transform_forward]: they fire only
5// after the forward table fails to find a direct entry. Any rule whose
6// preconditions are not provably met returns None so the dispatcher
7// falls through to the structural placeholder rather than committing a
8// non-rigorous result.

25pub(crate) fn apply_forward_rules(
26    context: &Context,
27    target: &ExpressionReference,
28    time_variable: SymbolIdentifier,
29    frequency_variable: SymbolIdentifier,
30) -> Option<ExpressionReference> {
31    // Expression-valued projection of the identified dispatcher below. The
32    // historical entry point keeps its exact behaviour; the closing theorem is
33    // simply discarded for callers that do not need transform evidence.
34    apply_forward_rules_with_identity(context, target, time_variable, frequency_variable)
35        .map(|(image, _)| image)
36}

38pub(crate) fn apply_forward_rules_with_identity(
39    context: &Context,
40    target: &ExpressionReference,
41    time_variable: SymbolIdentifier,
42    frequency_variable: SymbolIdentifier,
43) -> Option<(ExpressionReference, TransformRuleIdentity)> {
44    // Walk the rule stack. The order is significant: linearity sits
45    // first so a top-level sum decomposes before any rule examines its
46    // shape; the frequency-shift, time-shift, and scaling rules come
47    // next because they reduce to a transform sub-problem whose shape is
48    // by construction simpler than the input. The rule that closes reports
49    // its own theorem identity so a caller can record which transform theorem
50    // produced the image and under which hypotheses.
51    if let Some(image) = apply_linearity(context, target, time_variable, frequency_variable) {
52        return Some((
53            image,
54            TransformRuleIdentity::plain(IntegralTransformTheorem::Linearity),
55        ));
56    }
57    if let Some(identified) =
58        apply_derivative_rule(context, target, time_variable, frequency_variable)
59    {
60        return Some(identified);
61    }
62    if let Some(image) = apply_constant_factor(context, target, time_variable, frequency_variable) {
63        return Some((
64            image,
65            TransformRuleIdentity::plain(IntegralTransformTheorem::ConstantFactor),
66        ));
67    }
68    if let Some(image) =
69        apply_exponential_constant_factor(context, target, time_variable, frequency_variable)
70    {
71        return Some((
72            image,
73            TransformRuleIdentity::plain(IntegralTransformTheorem::ConstantFactor),
74        ));
75    }
76    if let Some(identified) = apply_scaling(context, target, time_variable, frequency_variable) {
77        return Some(identified);
78    }
79    if let Some(image) = apply_frequency_shift(context, target, time_variable, frequency_variable) {
80        return Some((
81            image,
82            TransformRuleIdentity::plain(IntegralTransformTheorem::FrequencyShift),
83        ));
84    }
85    if let Some(image) = apply_time_shift(context, target, time_variable, frequency_variable) {
86        return Some((
87            image,
88            TransformRuleIdentity::plain(IntegralTransformTheorem::Translation),
89        ));
90    }
91    if let Some(image) = apply_convolution(context, target, time_variable, frequency_variable) {
92        return Some((
93            image,
94            TransformRuleIdentity::plain(IntegralTransformTheorem::Convolution),
95        ));
96    }
97    None
98}

100fn apply_convolution(
101    context: &Context,
102    target: &ExpressionReference,
103    time_variable: SymbolIdentifier,
104    frequency_variable: SymbolIdentifier,
105) -> Option<ExpressionReference> {
106    // L{(f * g)(t)} = F(s) * G(s), where the convolution
107    //   (f * g)(t) = ∫_0^t f(tau) * g(t - tau) d_tau
108    // is encoded as an `integral` AST node with bound variable `tau`,
109    // lower bound 0, upper bound the time variable, and an integrand
110    // that is a product whose factors split into one call dependent on
111    // `tau` (taking the role of f(tau)) and one call dependent on
112    // `t - tau` (taking the role of g(t - tau)). The recogniser:
113    //   (1) verifies the bounds: lower is the integer literal zero and
114    //       upper is the bare time variable;
115    //   (2) verifies the integrand is a binary product whose factor set
116    //       can be partitioned into a `tau`-only piece and a (t - tau)
117    //       piece — neither piece may depend on both variables, and
118    //       neither may depend on either of the two transform variables
119    //       outside the recognised role;
120    //   (3) reconstructs f(t) and g(t) by undoing the bound variable
121    //       substitution on each piece (replacing the dummy with the
122    //       time variable, replacing the dummy's complement with the
123    //       time variable), then transforms each through the forward
124    //       dispatcher;
125    //   (4) emits F(s) * G(s) if both sub-transforms close; refuses
126    //       (returns None) if either sub-transform produces a
127    //       placeholder, leaving the caller to fall back to the
128    //       structural placeholder.
129    let ExpressionKind::Integral(integrand, bound_variable, lower_bound, upper_bound) =
130        &target.kind
131    else {
132        return None;
133    };
134    let lower = lower_bound.as_ref()?;
135    let upper = upper_bound.as_ref()?;
136    if !is_integer_zero(lower) {
137        return None;
138    }
139    if !matches!(
140        &upper.kind,
141        ExpressionKind::Symbol(identifier) if *identifier == time_variable
142    ) {
143        return None;
144    }
145    let dummy_variable = *bound_variable;
146    if dummy_variable == time_variable || dummy_variable == frequency_variable {
147        return None;
148    }
149    let factors: Vec<ExpressionReference> = match &integrand.kind {
150        ExpressionKind::Product(children) => children.iter().map(Arc::clone).collect(),
151        _ => vec![Arc::clone(integrand)],
152    };
153    let (dummy_only_factor, complement_factor) = split_convolution_factors(
154        context,
155        &factors,
156        dummy_variable,
157        time_variable,
158        frequency_variable,
159    )?;
160    let f_of_time =
161        substitute_single_variable(context, &dummy_only_factor, dummy_variable, time_variable);
162    let g_of_time = recover_g_of_time(context, &complement_factor, dummy_variable, time_variable)?;
163    let f_image = super::transform_forward(context, f_of_time, time_variable, frequency_variable);
164    let g_image = super::transform_forward(context, g_of_time, time_variable, frequency_variable);
165    // No placeholder refusal here: the convolution rule's mathematical
166    // statement L{f * g} = F(s) * G(s) is rigorous even when F and G
167    // are themselves symbolic placeholders for unknown functions. The
168    // rigour boundary is enforced upstream by the shape recognition
169    // (matched bounds, dummy-only and (t - tau)-only factor split,
170    // round-trip verification in [recover_g_of_time]); once the shape
171    // is recognised, the product of the two sub-images is the textbook
172    // answer regardless of whether F and G have closed forms.
173    Some(builders::product(context, vec![f_image, g_image]))
174}

176fn split_convolution_factors(
177    context: &Context,
178    factors: &[ExpressionReference],
179    dummy_variable: SymbolIdentifier,
180    time_variable: SymbolIdentifier,
181    frequency_variable: SymbolIdentifier,
182) -> Option<(ExpressionReference, ExpressionReference)> {
183    // Partition a convolution integrand's factor list into a piece
184    // dependent only on the dummy variable `tau` and a piece dependent
185    // only on the bound shift `t - tau`. A factor that contains the
186    // frequency variable disqualifies the recognition because the
187    // convolution rule is a pure time-domain shape. Constant factors
188    // (depending on neither dummy nor time) attach to the dummy-only
189    // bucket so the rule's F(s) carries every constant; this is
190    // mathematically equivalent because constants commute through the
191    // transform. A factor that depends on both `tau` and `t` is held
192    // for the (t - tau) round-trip verification performed later in
193    // [recover_g_of_time]; a factor that depends on `t` alone (without
194    // `tau`) disqualifies the recognition because such a factor would
195    // be the wrong shape for a convolution integrand. When the
196    // partition leaves one bucket empty, the missing side is the
197    // degenerate "convolve with one" case and is synthesised as the
198    // integer literal one so the dispatcher can still emit F(s)*G(s).
199    let mut dummy_factors: Vec<ExpressionReference> = Vec::new();
200    let mut complement_factors: Vec<ExpressionReference> = Vec::new();
201    for factor in factors {
202        if table::expression_depends_on_variable(factor, frequency_variable) {
203            return None;
204        }
205        let depends_on_dummy = table::expression_depends_on_variable(factor, dummy_variable);
206        let depends_on_time = table::expression_depends_on_variable(factor, time_variable);
207        match (depends_on_dummy, depends_on_time) {
208            (true, true) => complement_factors.push(Arc::clone(factor)),
209            (true, false) => dummy_factors.push(Arc::clone(factor)),
210            (false, true) => return None,
211            (false, false) => dummy_factors.push(Arc::clone(factor)),
212        }
213    }
214    if dummy_factors.is_empty() && complement_factors.is_empty() {
215        return None;
216    }
217    let dummy_only = if dummy_factors.is_empty() {
218        builders::integer_value(context, 1)
219    } else {
220        collapse_product(dummy_factors)
221    };
222    let complement_only = if complement_factors.is_empty() {
223        builders::integer_value(context, 1)
224    } else {
225        collapse_product(complement_factors)
226    };
227    Some((dummy_only, complement_only))
228}

230fn collapse_product(factors: Vec<ExpressionReference>) -> ExpressionReference {
231    // Collapse a single-element factor list to its sole element;
232    // multi-element lists fall back to a structural product node built
233    // through make_expression so the convolution recogniser does not
234    // re-enter the simplifier mid-recognition.
235    if factors.len() == 1 {
236        factors.into_iter().next().unwrap()
237    } else {
238        Arc::new(crate::expression::make_expression(ExpressionKind::Product(
239            factors,
240        )))
241    }
242}

244fn recover_g_of_time(
245    context: &Context,
246    complement: &ExpressionReference,
247    dummy_variable: SymbolIdentifier,
248    time_variable: SymbolIdentifier,
249) -> Option<ExpressionReference> {
250    // Recover g(t) from a factor expressed in terms of (t - tau). The
251    // recovery has two phases. (1) Substitute `tau -> 0` to obtain a
252    // candidate g(t) under the assumption that the complement has the
253    // canonical convolution shape (every joint occurrence of t and tau
254    // appears as t - tau). (2) Verify the assumption by substituting
255    // `t -> t - tau` into the candidate; the result must structurally
256    // equal the original complement under the interner's hash-cons
257    // identity. The round-trip test rules out any factor that contains
258    // mixed shapes such as t * tau or t / tau, where the candidate
259    // would collapse to a constant under tau -> 0 and the recogniser
260    // would otherwise emit a non-rigorous L{g(t)} based on that
261    // constant. Returns None whenever the round-trip fails so the
262    // dispatcher falls back to the structural placeholder.
263    let zero_expression = builders::zero(context);
264    let candidate =
265        substitute_symbol_with_expression(context, complement, dummy_variable, zero_expression);
266    let time_expression = symbol_expression(context, time_variable);
267    let dummy_expression = symbol_expression(context, dummy_variable);
268    let time_minus_dummy = builders::subtract(context, time_expression, dummy_expression);
269    let round_tripped =
270        substitute_symbol_with_expression(context, &candidate, time_variable, time_minus_dummy);
271    if Arc::ptr_eq(&round_tripped, complement) {
272        Some(candidate)
273    } else {
274        None
275    }
276}

278fn substitute_single_variable(
279    context: &Context,
280    target: &ExpressionReference,
281    old_variable: SymbolIdentifier,
282    new_variable: SymbolIdentifier,
283) -> ExpressionReference {
284    // Replace every occurrence of `old_variable` in `target` with
285    // `new_variable`, using the existing substitute machinery so
286    // capture-avoiding hygiene is preserved. Used by the convolution
287    // recogniser to lift f(tau) to f(t).
288    let new_expression = symbol_expression(context, new_variable);
289    substitute_symbol_with_expression(context, target, old_variable, new_expression)
290}

292fn substitute_symbol_with_expression(
293    context: &Context,
294    target: &ExpressionReference,
295    old_variable: SymbolIdentifier,
296    replacement: ExpressionReference,
297) -> ExpressionReference {
298    // Wrap the substitute pipeline with a symbol-keyed mapping. The
299    // helper exists so the convolution recogniser's intent
300    // ("rename / specialise tau") reads more cleanly at the call site.
301    let old_expression = symbol_expression(context, old_variable);
302    let mut mapping: SubstitutionMap = HashMap::new();
303    mapping.insert(old_expression, replacement);
304    substitute(context, target, &mapping)
305}

307fn is_integer_zero(target: &ExpressionReference) -> bool {
308    // Predicate identifying the integer literal zero. Used by the
309    // convolution recogniser's lower-bound check.
310    matches!(
311        &target.kind,
312        ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::integer_zero()
313    )

316fn apply_linearity(
317    context: &Context,
318    target: &ExpressionReference,
319    time_variable: SymbolIdentifier,
320    frequency_variable: SymbolIdentifier,
321) -> Option<ExpressionReference> {
322    // L{sum_i f_i(t)} = sum_i L{f_i(t)}. The rule pulls every term
323    // through the forward dispatcher; if any term refuses (returns the
324    // structural placeholder) the rule itself refuses by returning None
325    // so the caller does not splice a half-recognised sum.
326    let ExpressionKind::Sum(children) = &target.kind else {
327        return None;
328    };
329    let transformed = crate::integral_transforms::terms::transform_terms(
330        context,
331        children,
332        "laplace forward",
333        "additive forward Laplace term",
334        |child| {
335            let candidate = super::transform_forward(
336                context,
337                Arc::clone(child),
338                time_variable,
339                frequency_variable,
340            );
341            (!super::is_laplace_placeholder(&candidate, time_variable, frequency_variable))
342                .then_some(candidate)
343        },
344    )?;
345    Some(builders::sum(context, transformed))
346}

348fn apply_derivative_rule(
349    context: &Context,
350    target: &ExpressionReference,
351    time_variable: SymbolIdentifier,
352    frequency_variable: SymbolIdentifier,
353) -> Option<(ExpressionReference, TransformRuleIdentity)> {
354    // L{f^(n)(t)} = s^n * F(s) − sum_{k=0}^{n-1} s^{n-1-k} * f^(k)(0)
355    //             = s^n * F(s) − s^(n-1) * f(0) − s^(n-2) * f'(0) − ... − f^(n-1)(0)
356    // The rule recognises the canonical n-th derivative shape produced
357    // by differentiation of an unknown function: the
358    // [`derivative`] AST variant with a single-element order vector and
359    // a single argument that is structurally the bare time variable.
360    // Compositions such as f(g(t)) produce a derivative whose argument
361    // is g(t) rather than t and so do not fire this rule — the chain
362    // rule would need to be applied first, which is outside the rigour
363    // boundary of what this module implements. The rule emits an unevaluated forward
364    // Laplace transform on the bare f(t) call because that quantity is
365    // by definition the unknown function's image F(s) in the textbook
366    // statement; downstream consumers (e.g. the constant-coefficient classifier's Laplace IVP path)
367    // can supply assumptions on the unknown function later by
368    // substituting a closed-form into F(s) at the end of the pipeline.
369    let ExpressionKind::Derivative(function, orders, arguments) = &target.kind else {
370        return None;
371    };
372    if orders.len() != 1 || arguments.len() != 1 {
373        return None;
374    }
375    let order = orders[0];
376    if order == 0 {
377        return None;
378    }
379    if !matches!(
380        &arguments[0].kind,
381        ExpressionKind::Symbol(identifier) if *identifier == time_variable
382    ) {
383        return None;
384    }
385    let function_identifier_value = *function;
386    let time = symbol_expression(context, time_variable);
387    let bare_call = builders::function_call(context, function_identifier_value, vec![time]);
388    let frequency_image = builders::laplace_transform_unevaluated(
389        context,
390        bare_call,
391        time_variable,
392        frequency_variable,
393    );
394    let frequency = symbol_expression(context, frequency_variable);
395    let order_as_integer = builders::integer_value(context, order as i64);
396    let s_to_n = builders::power(context, Arc::clone(&frequency), order_as_integer);
397    let positive_part = builders::product(context, vec![s_to_n, frequency_image]);
398    let mut terms: Vec<ExpressionReference> = Vec::with_capacity(order + 1);
399    terms.push(positive_part);
400    for derivative_order in 0..order {
401        let frequency_power_exponent = order - 1 - derivative_order;
402        let frequency_factor = if frequency_power_exponent == 0 {
403            builders::integer_value(context, 1)
404        } else {
405            let exponent_expression =
406                builders::integer_value(context, frequency_power_exponent as i64);
407            builders::power(context, Arc::clone(&frequency), exponent_expression)
408        };
409        let zero_expression = builders::zero(context);
410        let initial_value = if derivative_order == 0 {
411            builders::function_call(context, function_identifier_value, vec![zero_expression])
412        } else {
413            builders::derivative(
414                context,
415                function_identifier_value,
416                vec![derivative_order],
417                vec![zero_expression],
418            )
419        };
420        let term_product = builders::product(context, vec![frequency_factor, initial_value]);
421        let negated = builders::negate(context, term_product);
422        terms.push(negated);
423    }
424    Some((
425        builders::sum(context, terms),
426        TransformRuleIdentity::plain(IntegralTransformTheorem::Derivative(order)),
427    ))
428}

430fn apply_constant_factor(
431    context: &Context,
432    target: &ExpressionReference,
433    time_variable: SymbolIdentifier,
434    frequency_variable: SymbolIdentifier,
435) -> Option<ExpressionReference> {
436    // L{c * f(t)} = c * L{f(t)} for any c constant in the time variable.
437    // The rule scans a product, partitions the factors into those that
438    // depend on the time variable and those that do not, and recurses on
439    // the time-dependent residue. Refuses when the dependent residue is
440    // empty (which would degenerate to a constant Laplace transform of a
441    // constant — already handled by the table) or when the recursive
442    // transform falls back to a placeholder.
443    let ExpressionKind::Product(factors) = &target.kind else {
444        return None;
445    };
446    let mut constant_factors: Vec<ExpressionReference> = Vec::new();
447    let mut dependent_factors: Vec<ExpressionReference> = Vec::new();
448    for factor in factors {
449        if expression_depends_on_time(factor, time_variable) {
450            dependent_factors.push(Arc::clone(factor));
451        } else {
452            constant_factors.push(Arc::clone(factor));
453        }
454    }
455    if constant_factors.is_empty() || dependent_factors.is_empty() {
456        return None;
457    }
458    let dependent_payload = if dependent_factors.len() == 1 {
459        dependent_factors.into_iter().next().unwrap()
460    } else {
461        builders::product(context, dependent_factors)
462    };
463    let sub_image = super::transform_forward(
464        context,
465        dependent_payload,
466        time_variable,
467        frequency_variable,
468    );
469    if super::is_laplace_placeholder(&sub_image, time_variable, frequency_variable) {
470        return None;
471    }
472    let coefficient = if constant_factors.len() == 1 {
473        constant_factors.into_iter().next().unwrap()
474    } else {
475        builders::product(context, constant_factors)
476    };
477    Some(builders::product(context, vec![coefficient, sub_image]))
478}

480fn apply_exponential_constant_factor(
481    context: &Context,
482    target: &ExpressionReference,
483    time_variable: SymbolIdentifier,
484    frequency_variable: SymbolIdentifier,
485) -> Option<ExpressionReference> {
486    // Splits exp(A + B(t)) when A is independent of t, then transforms
487    // exp(B(t)) and restores the exact constant multiplier exp(A).
488    let ExpressionKind::FunctionCall(function, arguments) = &target.kind else {
489        return None;
490    };
491    if !matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Exp))
492        || arguments.len() != 1
493    {
494        return None;
495    }
496    let ExpressionKind::Sum(terms) = &arguments[0].kind else {
497        return None;
498    };
499    let (constant_terms, time_terms): (Vec<_>, Vec<_>) = terms
500        .iter()
501        .cloned()
502        .partition(|term| !expression_depends_on_time(term, time_variable));
503    if constant_terms.is_empty() || time_terms.is_empty() {
504        return None;
505    }
506    let constant_exponent = builders::sum(context, constant_terms);
507    let time_exponent = builders::sum(context, time_terms);
508    let constant_factor = builders::function_call(
509        context,
510        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
511        vec![constant_exponent],
512    );
513    let time_factor = builders::function_call(
514        context,
515        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
516        vec![time_exponent],
517    );
518    let image = super::transform_forward(context, time_factor, time_variable, frequency_variable);
519    if super::is_laplace_placeholder(&image, time_variable, frequency_variable) {
520        return None;
521    }
522    Some(builders::product(context, vec![constant_factor, image]))
523}

525fn expression_depends_on_time(
526    target: &ExpressionReference,
527    time_variable: SymbolIdentifier,
528) -> bool {
529    // Predicate identifying any subtree that contains the time variable
530    // as a free variable. Mirrors the helper in [super::table] so the
531    // partition in [apply_constant_factor] uses the same definition of
532    // "constant in time" as the rest of the Laplace pipeline.
533    super::table::expression_depends_on_variable(target, time_variable)
534}

536fn apply_scaling(
537    context: &Context,
538    target: &ExpressionReference,
539    time_variable: SymbolIdentifier,
540    frequency_variable: SymbolIdentifier,
541) -> Option<(ExpressionReference, TransformRuleIdentity)> {
542    // L{f(a*t)} = (1/a) * F(s/a) for a > 0. The rule detects an outer
543    // function call whose single argument is `scale * t` with scale
544    // provably positive and different from one. The image transform on
545    // f(t) is recovered through the forward dispatcher; the rule refuses
546    // when the sub-transform stays as a placeholder so the result stays
547    // rigorous.
548    let (function, arguments) = table::function_call_components(target)?;
549    if arguments.len() != 1 {
550        return None;
551    }
552    let inner = &arguments[0];
553    let scale = extract_positive_constant_factor(context, inner, time_variable)?;
554    if is_integer_one(&scale) {
555        return None;
556    }
557    let positivity =
558        crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &scale);
559    if positivity != TruthValue::Yes {
560        return None;
561    }
562    let bare_call = build_unary_call(context, function, time_variable);
563    let sub_image = super::transform_forward(context, bare_call, time_variable, frequency_variable);
564    if super::is_laplace_placeholder(&sub_image, time_variable, frequency_variable) {
565        return None;
566    }
567    let scaled_frequency = scale_frequency(context, &sub_image, &scale, frequency_variable);
568    Some((
569        builders::divide(context, scaled_frequency, Arc::clone(&scale)),
570        TransformRuleIdentity::requiring_nonzero(IntegralTransformTheorem::Scaling, scale),
571    ))
572}

574fn apply_frequency_shift(
575    context: &Context,
576    target: &ExpressionReference,
577    time_variable: SymbolIdentifier,
578    frequency_variable: SymbolIdentifier,
579) -> Option<ExpressionReference> {
580    // L{exp(a*t) * f(t)} = F(s - a). The rule scans a product for a
581    // single exp(a*t) factor whose exponent's t-coefficient is constant
582    // with respect to the time variable; the residue is interpreted as
583    // f(t) and its image is recovered through the forward dispatcher.
584    // The rule fires only when the residue's image is recognised.
585    let ExpressionKind::Product(factors) = &target.kind else {
586        return None;
587    };
588    let mut shift_offset: Option<ExpressionReference> = None;
589    let mut residue_factors: Vec<ExpressionReference> = Vec::new();
590    for factor in factors {
591        if shift_offset.is_none() {
592            if let Some(coefficient) = match_exp_coefficient(context, factor, time_variable) {
593                shift_offset = Some(coefficient);
594                continue;
595            }
596        }
597        residue_factors.push(Arc::clone(factor));
598    }
599    let shift = shift_offset?;
600    if residue_factors.is_empty() {
601        return None;
602    }
603    let residue = if residue_factors.len() == 1 {
604        residue_factors.into_iter().next().unwrap()
605    } else {
606        builders::product(context, residue_factors)
607    };
608    let residue_image =
609        super::transform_forward(context, residue, time_variable, frequency_variable);
610    if super::is_laplace_placeholder(&residue_image, time_variable, frequency_variable) {
611        return None;
612    }
613    let frequency = symbol_expression(context, frequency_variable);
614    let shifted_frequency_argument = builders::subtract(context, frequency, shift);
615    Some(substitute_frequency(
616        context,
617        &residue_image,
618        frequency_variable,
619        shifted_frequency_argument,
620    ))
621}

623fn apply_time_shift(
624    context: &Context,
625    target: &ExpressionReference,
626    time_variable: SymbolIdentifier,
627    frequency_variable: SymbolIdentifier,
628) -> Option<ExpressionReference> {
629    // L{heaviside_step(t - a) * f(t - a)} = exp(-a*s) * F(s) for a >= 0.
630    // The rule scans a product for exactly one heaviside_step(t - a)
631    // factor and treats the residue as f(t - a). Recovery of F(s) needs
632    // f(u) under u = t + a, then transformed through the forward
633    // dispatcher in u; the substitution u -> t restores the rule's
634    // shape. The rule refuses when the recovery fails to produce a
635    // closed-form image so the result stays rigorous.
636    let ExpressionKind::Product(factors) = &target.kind else {
637        return None;
638    };
639    let mut shift: Option<ExpressionReference> = None;
640    let mut residue_factors: Vec<ExpressionReference> = Vec::new();
641    for factor in factors {
642        if shift.is_none() {
643            if let Some(offset) = match_heaviside_shift(context, factor, time_variable) {
644                shift = Some(offset);
645                continue;
646            }
647        }
648        residue_factors.push(Arc::clone(factor));
649    }
650    let shift = shift?;
651    let positivity =
652        crate::assumptions::range_evidence::positive_verdict(&context.symbol_table, &shift);
653    let is_numeric_zero = matches!(
654        &shift.kind,
655        ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::integer_zero()
656    );
657    if positivity != TruthValue::Yes && !is_numeric_zero {
658        return None;
659    }
660    let residue = if residue_factors.is_empty() {
661        builders::integer_value(context, 1)
662    } else if residue_factors.len() == 1 {
663        residue_factors.into_iter().next().unwrap()
664    } else {
665        builders::product(context, residue_factors)
666    };
667    let recovered_function = substitute_time_variable(context, &residue, time_variable, &shift);
668    let recovered_image = super::transform_forward(
669        context,
670        recovered_function,
671        time_variable,
672        frequency_variable,
673    );
674    if super::is_laplace_placeholder(&recovered_image, time_variable, frequency_variable) {
675        return None;
676    }
677    let frequency = symbol_expression(context, frequency_variable);
678    let negated_shift = builders::negate(context, shift);
679    let exponent_argument = builders::product(context, vec![negated_shift, frequency]);
680    let exponential = builders::function_call(
681        context,
682        FunctionIdentifier::Builtin(BuiltinFunction::Exp),
683        vec![exponent_argument],
684    );
685    Some(builders::product(
686        context,
687        vec![exponential, recovered_image],
688    ))
689}

691fn extract_positive_constant_factor(
692    context: &Context,
693    argument: &ExpressionReference,
694    time_variable: SymbolIdentifier,
695) -> Option<ExpressionReference> {
696    // Mirror of [table::extract_constant_coefficient_of] used by the

698    // structural shape c * t with c constant in time_variable.
699    table::extract_constant_coefficient_of(context, argument, time_variable)
700}

702fn match_exp_coefficient(
703    context: &Context,
704    factor: &ExpressionReference,
705    time_variable: SymbolIdentifier,
706) -> Option<ExpressionReference> {
707    // Recognise factor = exp(a*t) and return the coefficient a. Used by
708    // the frequency-shift rule's scan.
709    let (function, arguments) = table::function_call_components(factor)?;
710    if !matches!(function, FunctionIdentifier::Builtin(BuiltinFunction::Exp)) {
711        return None;
712    }
713    if arguments.len() != 1 {
714        return None;
715    }
716    table::extract_constant_coefficient_of(context, &arguments[0], time_variable)
717}
718
719fn match_heaviside_shift(
720    context: &Context,
721    factor: &ExpressionReference,
722    time_variable: SymbolIdentifier,
723) -> Option<ExpressionReference> {
724    // Recognise factor = heaviside_step(t - a) and return the shift
725    // offset a (returned positive: the textbook convention is t - a so a
726    // is the inner argument's negated constant component).
727    let (function, arguments) = table::function_call_components(factor)?;
728    if !matches!(
729        function,
730        FunctionIdentifier::Builtin(BuiltinFunction::HeavisideStep)
731    ) {
732        return None;
733    }
734    if arguments.len() != 1 {
735        return None;
736    }
737    table::extract_nonnegative_shift_of(context, &arguments[0], time_variable)
738}

740fn build_unary_call(
741    context: &Context,
742    function: FunctionIdentifier,
743    time_variable: SymbolIdentifier,
744) -> ExpressionReference {
745    // Build f(t) from a function identifier and the time-variable symbol
746    // identifier. Used by the scaling rule to reconstruct the bare-call
747    // shape whose forward transform is the sub-problem to be solved.
748    let time = symbol_expression(context, time_variable);
749    builders::function_call(context, function, vec![time])
750}

752fn scale_frequency(
753    context: &Context,
754    image: &ExpressionReference,
755    scale: &ExpressionReference,
756    frequency_variable: SymbolIdentifier,
757) -> ExpressionReference {
758    // Replace every occurrence of the frequency variable in image with
759    // frequency_variable / scale. Used by the scaling rule which produces
760    // F(s/a) from F(s).
761    let frequency = symbol_expression(context, frequency_variable);
762    let new_argument = builders::divide(context, frequency, Arc::clone(scale));
763    substitute_frequency(context, image, frequency_variable, new_argument)
764}
765
766fn substitute_frequency(
767    context: &Context,
768    image: &ExpressionReference,
769    frequency_variable: SymbolIdentifier,
770    new_argument: ExpressionReference,
771) -> ExpressionReference {
772    // Replace every occurrence of the frequency variable inside the
773    // image with a supplied replacement expression. Wraps the existing
774    // [substitute] machinery so capture-avoidance is preserved.
775    let frequency = symbol_expression(context, frequency_variable);
776    let mut mapping: SubstitutionMap = HashMap::new();
777    mapping.insert(frequency, new_argument);
778    substitute(context, image, &mapping)
779}
780
781fn substitute_time_variable(
782    context: &Context,
783    body: &ExpressionReference,
784    time_variable: SymbolIdentifier,
785    shift: &ExpressionReference,
786) -> ExpressionReference {
787    // Replace every occurrence of the time variable in body with
788    // time_variable + shift. Used by the time-shift rule to recover f(t)
789    // from the f(t - a) residue: substituting t -> t + a undoes the
790    // shift so the recovered call's forward transform can be looked up
791    // directly.
792    let time = symbol_expression(context, time_variable);
793    let shifted = builders::sum(context, vec![Arc::clone(&time), Arc::clone(shift)]);
794    let mut mapping: SubstitutionMap = HashMap::new();
795    mapping.insert(time, shifted);
796    substitute(context, body, &mapping)
797}

799fn symbol_expression(context: &Context, identifier: SymbolIdentifier) -> ExpressionReference {
800    // Mirror of [table::symbol_expression]; kept here to avoid the
801    // module path noise at every call site.
802    table::symbol_expression(context, identifier)
803}

805fn is_integer_one(target: &ExpressionReference) -> bool {
806    // Predicate identifying the integer literal one. Used by the scaling
807    // rule to recognise the degenerate a = 1 case and decline so the
808    // dispatcher does not loop on the resulting trivial sub-problem.
809    matches!(
810        &target.kind,
811        ExpressionKind::Integer(value) if *value == crate::exact_arithmetic::integer_one()
812    )
