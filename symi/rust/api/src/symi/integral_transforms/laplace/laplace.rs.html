1// Forward and inverse Laplace transform. The forward pass
2// dispatches the standard-pair table and the algebraic rules; unrecognised inputs
3// fall back to the structural laplace_transform placeholder shipped by
4// [crate::simplify::simplify_laplace_transform]. The inverse pass routes
5// rational F(s) inputs through partial fractions and reverses
6// each summand against the table; non-rational inputs fall back to the
7// structural inverse_laplace_transform placeholder. The Bromwich contour
8// stays deferred to future work.
9//
10// State of the art. Doetsch (1974), Schiff (1999) for the canonical table
11// and algebraic rules. SymPy's `integrals.transforms.laplace_transform` is
12// the closest open-source structural reference for the table-and-rules
13// architecture adopted here.

35/// Performs the advanced native `transform_forward` operation.
36pub fn transform_forward(
37    context: &Context,
38    target: ExpressionReference,
39    time_variable: SymbolIdentifier,
40    frequency_variable: SymbolIdentifier,
41) -> ExpressionReference {
42    if Arc::ptr_eq(&target, &builders::zero(context)) {
43        return builders::zero(context);
44    }
45    // Eager forward-Laplace dispatcher. The order of the strategy stack is
46    // deliberate: a) the forward table fires first because it captures the
47    // closed-form atoms whose recognition does not require any rule
48    // composition (1, t^n, exp(a*t), trig, hyperbolic, heaviside, dirac);
49    // b) the algebraic rules fire next so structural shapes
50    // (linearity over sum, frequency shift through exp(a*t) * f(t), time
51    // shift through heaviside_step(t - a) * f(t - a), scaling f(a*t), the
52    // derivative rule, and the convolution product) collapse the input to
53    // the atom layer; c) if neither layer recognises the shape, the
54    // structural placeholder closes the dispatch so the result remains
55    // composable with later transformations through the execute pipeline.
56    if let Some(closed_form) =
57        table::transform_table_entry(context, &target, time_variable, frequency_variable)
58    {
59        return closed_form;
60    }
61    if let Some(closed_form) =
62        rules::apply_forward_rules(context, &target, time_variable, frequency_variable)
63    {
64        return closed_form;
65    }
66    simplify_laplace_transform(context, target, time_variable, frequency_variable)
67}

69/// Compute a forward Laplace transform while temporarily applying an immutable assumption formula.
70pub fn transform_forward_under_assumptions(
71    context: &Context,
72    target: ExpressionReference,
73    time_variable: SymbolIdentifier,
74    frequency_variable: SymbolIdentifier,
75    assumptions: AssumptionFormula,
76) -> Result<ExpressionReference, crate::assumptions::AssumptionMutationError> {
77    context.with_temporary_assumptions(assumptions, |scoped_context| {
78        transform_forward(scoped_context, target, time_variable, frequency_variable)
79    })
80}

82/// Performs the advanced native `transform_inverse` operation.
83pub fn transform_inverse(
84    context: &Context,
85    target: ExpressionReference,
86    frequency_variable: SymbolIdentifier,
87    time_variable: SymbolIdentifier,
88) -> ExpressionReference {
89    if Arc::ptr_eq(&target, &builders::zero(context)) {
90        return builders::zero(context);
91    }
92    // Eager inverse-Laplace dispatcher. The stack walks: a) a direct
93    // table reverse-lookup for the leaf shapes the forward table recognises
94    // (1/s, n!/s^{n+1}, 1/(s-a), the trig / hyperbolic pairs, exp(-a*s)/s
95    // for heaviside, exp(-a*s) for dirac); b) the linearity rule over
96    // sum so an inhomogeneous transform of a sum is decomposed term by
97    // term; c) the partial-fraction reverse-lookup pipeline for any
98    // remaining rational F(s); d) the structural placeholder otherwise.
99    if let Some(closed_form) =
100        inverse::inverse_table_entry(context, &target, frequency_variable, time_variable)
101    {
102        return closed_form;
103    }
104    if let crate::expression::ExpressionKind::Sum(children) = &target.kind {
105        if let Some(inverted_terms) = crate::integral_transforms::terms::transform_terms(
106            context,
107            children,
108            "laplace inverse",
109            "additive inverse Laplace term",
110            |child| {
111                let inverted = transform_inverse(
112                    context,
113                    Arc::clone(child),
114                    frequency_variable,
115                    time_variable,
116                );
117                (!is_inverse_laplace_placeholder(&inverted, frequency_variable, time_variable))
118                    .then_some(inverted)
119            },
120        ) {
121            return crate::expression::builders::sum(context, inverted_terms);
122        }
123    }
124    if let Some(closed_form) =
125        inverse::inverse_constant_factor(context, &target, frequency_variable, time_variable)
126    {
127        return closed_form;
128    }
129    if let Some(closed_form) =
130        inverse::inverse_via_partial_fractions(context, &target, frequency_variable, time_variable)
131    {
132        return closed_form;
133    }
134    simplify_inverse_laplace_transform(context, target, frequency_variable, time_variable)
135}

150pub(crate) fn is_inverse_laplace_placeholder(
151    candidate: &ExpressionReference,
152    frequency_variable: SymbolIdentifier,
153    time_variable: SymbolIdentifier,
154) -> bool {
155    // Recognise the structural placeholder returned by the inverse
156    // dispatcher when no closed-form path closes. Used by the linearity
157    // arm to detect that at least one summand could not be inverted, in
158    // which case the whole sum stays as a single placeholder rather than
159    // being torn apart into a sum of placeholders.
160    matches!(
161        &candidate.kind,
162        crate::expression::ExpressionKind::InverseLaplaceTransform(
163            _,
164            frequency,
165            time,
166        ) if *frequency == frequency_variable && *time == time_variable
167    )
168}

170pub(crate) fn is_laplace_placeholder(
171    candidate: &ExpressionReference,
172    time_variable: SymbolIdentifier,
173    frequency_variable: SymbolIdentifier,
174) -> bool {
175    // Recognise the structural placeholder returned by the forward
176    // dispatcher when no closed-form path closes. Used by the linearity
177    // arm of the forward rules to detect that at least one summand failed
178    // to transform, in which case the rule refuses to commit a partial
179    // result.
180    matches!(
181        &candidate.kind,
182        crate::expression::ExpressionKind::LaplaceTransform(
183            _,
184            time,
185            frequency,
186        ) if *time == time_variable && *frequency == frequency_variable
187    )
188}
