17/// Performs the advanced native `implicitize` operation.
18pub fn implicitize(
19    context: &Context,
20    coordinates: &[SymbolIdentifier],
21    parameters: &[SymbolIdentifier],
22    parametrisation: &[(SymbolIdentifier, ExpressionReference)],
23) -> Option<Vec<ExpressionReference>> {
24    // Implicitise a parametric curve or surface (Decision 5; Cox–Little–O'Shea ch. 3): turn a
25    // parametrisation xᵢ = fᵢ(parameters) into the generators of the implicit ideal
26    // ⟨xᵢ − fᵢ⟩ ∩ Q[coordinates]. A polynomial parametrisation eliminates the parameters
27    // directly; a rational one fᵢ = pᵢ/qᵢ clears denominators (the relation becomes
28    // qᵢ·xᵢ − pᵢ) and saturates by the denominator product through a fresh tag variable w with
29    // the relation w·∏qᵢ − 1, so the base-point / denominator locus is removed and the
30    // eliminant is the true implicit equation rather than a multiple carrying spurious
31    // components. None when any parametric expression is outside Q(parameters) (Decision 1)
32    // or there are no parameters to eliminate.
33    if parameters.is_empty() {
34        return None;
35    }
36    let parameter_arity = parameters.len();
37    let mut parameter_index: HashMap<SymbolIdentifier, usize> = HashMap::new();
38    for (index, parameter) in parameters.iter().enumerate() {
39        parameter_index.insert(*parameter, index);
40    }
41    let tag = context.symbol_table.fresh("implicitization_tag");
42    let mut generators: Vec<ExpressionReference> = Vec::new();
43    let mut denominators: Vec<ExpressionReference> = Vec::new();
44    for (coordinate, parametric) in parametrisation {
45        let (numerator, denominator) =
46            rational_pair(parametric, &parameter_index, parameter_arity)?;
47        let numerator_expression =
48            distributed_polynomial_to_expression(context, &numerator, parameters);
49        let coordinate_expression = context.interner.intern(ExpressionKind::Symbol(*coordinate));
50        let generator = if is_constant(&denominator) {
51            builders::subtract(context, coordinate_expression, numerator_expression)
52        } else {
53            let denominator_expression =
54                distributed_polynomial_to_expression(context, &denominator, parameters);
55            if !denominators
56                .iter()
57                .any(|existing| Arc::ptr_eq(existing, &denominator_expression))
58            {
59                denominators.push(Arc::clone(&denominator_expression));
60            }
61            let cleared =
62                builders::product(context, vec![coordinate_expression, denominator_expression]);
63            builders::subtract(context, cleared, numerator_expression)
64        };
65        generators.push(generator);
66    }
67    let mut all_variables: Vec<SymbolIdentifier> = coordinates.to_vec();
68    all_variables.extend_from_slice(parameters);
69    let mut eliminate: Vec<SymbolIdentifier> = parameters.to_vec();
70    if !denominators.is_empty() {
71        let product = builders::product(context, denominators);
72        let one = builders::one(context);
73        let tag_expression = context.interner.intern(ExpressionKind::Symbol(tag));
74        let scaled = builders::product(context, vec![tag_expression, product]);
75        let saturation = builders::subtract(context, scaled, one);
76        generators.push(saturation);
77        all_variables.push(tag);
78        eliminate.push(tag);
79    }
80    elimination_ideal(context, &generators, &all_variables, &eliminate)
81}

83fn rational_pair(
84    expression: &ExpressionReference,
85    parameter_index: &HashMap<SymbolIdentifier, usize>,
86    arity: usize,
87) -> Option<(DistributedPolynomial, DistributedPolynomial)> {
88    // Express a parametric expression as a (numerator, denominator) pair of distributed
89    // polynomials over the parameter ring, recursing through sums (cross-multiply), products
90    // (independent multiply), and integer powers (repeated multiply, inverting for a negative
91    // exponent). None whenever the structure leaves Q(parameters) — a transcendental call, a
92    // non-integer power, or a symbol that is not a parameter — which is the clean decline the
93    // ideal layer inherits (Decision 1). The pair is not reduced to lowest terms; the
94    // saturation removes the entire denominator locus, so a shared factor is harmless.
95    match &expression.kind {
96        ExpressionKind::Integer(value) => Some((
97            constant_polynomial(arity, exact_rational::from(value.clone())),
98            one_polynomial(arity),
99        )),
100        ExpressionKind::Rational(value) => Some((
101            constant_polynomial(arity, value.clone()),
102            one_polynomial(arity),
103        )),
104        ExpressionKind::Symbol(identifier) => {
105            let index = parameter_index.get(identifier)?;
106            let variable = monomial_operations::variable_monomial(arity, *index);
107            Some((
108                distributed::from_terms(arity, vec![(variable, rational_one())]),
109                one_polynomial(arity),
110            ))
111        }
112        ExpressionKind::Sum(children) => {
113            let mut numerator = zero_polynomial(arity);
114            let mut denominator = one_polynomial(arity);
115            for child in children {
116                let (child_numerator, child_denominator) =
117                    rational_pair(child, parameter_index, arity)?;
118                numerator = distributed::add(
119                    &distributed::multiply(&numerator, &child_denominator),
120                    &distributed::multiply(&child_numerator, &denominator),
121                );
122                denominator = distributed::multiply(&denominator, &child_denominator);
123            }
124            Some((numerator, denominator))
125        }
126        ExpressionKind::Product(children) => {
127            let mut numerator = one_polynomial(arity);
128            let mut denominator = one_polynomial(arity);
129            for child in children {
130                let (child_numerator, child_denominator) =
131                    rational_pair(child, parameter_index, arity)?;
132                numerator = distributed::multiply(&numerator, &child_numerator);
133                denominator = distributed::multiply(&denominator, &child_denominator);
134            }
135            Some((numerator, denominator))
136        }
137        ExpressionKind::Power(base, exponent) => {
138            let exponent_value = match &exponent.kind {
139                ExpressionKind::Integer(value) => i64::try_from(value).ok()?,
140                _ => return None,
141            };
142            let (base_numerator, base_denominator) = rational_pair(base, parameter_index, arity)?;
143            let magnitude = exponent_value.unsigned_abs() as u32;
144            let raised_numerator = power_polynomial(&base_numerator, magnitude, arity);
145            let raised_denominator = power_polynomial(&base_denominator, magnitude, arity);
146            if exponent_value >= 0 {
147                Some((raised_numerator, raised_denominator))
148            } else {
149                Some((raised_denominator, raised_numerator))
150            }
151        }
152        _ => None,
153    }
154}

156fn constant_polynomial(arity: usize, value: exact_rational) -> DistributedPolynomial {
157    distributed::constant(arity, value)
158}
159
160fn zero_polynomial(arity: usize) -> DistributedPolynomial {
161    distributed::zero(arity)
162}
163
164fn one_polynomial(arity: usize) -> DistributedPolynomial {
165    distributed::constant(arity, rational_one())
166}
167
168fn power_polynomial(
169    base: &DistributedPolynomial,
170    exponent: u32,
171    arity: usize,
172) -> DistributedPolynomial {
173    // Raise a distributed polynomial to a non-negative integer power by repeated multiplication.
174    let mut result = one_polynomial(arity);
175    for _ in 0..exponent {
176        result = distributed::multiply(&result, base);
177    }
178    result
179}

181fn is_constant(value: &DistributedPolynomial) -> bool {
182    // True when the polynomial is a constant — zero, or a single term at the identity monomial.
183    // A constant denominator is never zero on the parametric image, so it needs no saturation.
184    value
185        .terms
186        .iter()
187        .all(|(term_monomial, _)| is_constant_monomial(term_monomial))
188}
189
190fn is_constant_monomial(term_monomial: &Monomial) -> bool {
191    term_monomial
192        .exponents
193        .iter()
194        .all(|exponent| *exponent == 0)
195}

207    fn vanishes_on_parameter_point(
208        context: &Context,
209        generator: &ExpressionReference,
210        coordinate_values: &[(SymbolIdentifier, ExpressionReference)],
211    ) -> bool {
212        // Substitute coordinate expressions (themselves the parametric forms) into a candidate
213        // implicit generator and confirm it is numerically zero over random parameter samples —
214        // the elimination oracle reused from the Gröbner elimination path.
215        use crate::expression::substitute::{SubstitutionMap, substitute};
216        let mut mapping = SubstitutionMap::new();
217        for (coordinate, value) in coordinate_values {
218            let coordinate_expression =
219                context.interner.intern(ExpressionKind::Symbol(*coordinate));
220            mapping.insert(coordinate_expression, value.clone());
221        }
222        let substituted = substitute(context, generator, &mapping);
223        let zero = builders::zero(context);
224        !numerically_compare(
225            context,
226            &substituted,
227            &zero,
228            &OracleConfiguration::default(),
229        )
230        .is_distinct()
231    }

233    #[test]
234    fn twisted_cubic_implicitises_to_the_known_eliminant() {
235        // {x = t^2, y = t^3} has implicit equation y^2 - x^3 = 0.
236        let mut context = Context::new();
237        let x = variable(&mut context, "x");
238        let y = variable(&mut context, "y");
239        let t = variable(&mut context, "t");
240        let t_expression = builders::symbol(&mut context, "t");
241        let two = builders::integer_value(&mut context, 2);
242        let three = builders::integer_value(&mut context, 3);
243        let t_squared = builders::power(&mut context, Arc::clone(&t_expression), two);
244        let t_cubed = builders::power(&mut context, Arc::clone(&t_expression), three);
245        let generators = implicitize(
246            &mut context,
247            &[x, y],
248            &[t],
249            &[(x, Arc::clone(&t_squared)), (y, Arc::clone(&t_cubed))],
250        )
251        .unwrap();
252        assert!(!generators.is_empty());
253        for generator in &generators {
254            assert!(vanishes_on_parameter_point(
255                &mut context,
256                generator,
257                &[(x, Arc::clone(&t_squared)), (y, Arc::clone(&t_cubed))],
258            ));
259        }
260    }

263    fn parametric_circle_implicitises_to_unit_circle() {
264        // {x = (1 - t^2)/(1 + t^2), y = 2t/(1 + t^2)} has implicit equation x^2 + y^2 - 1 = 0.
265        // This exercises the rational-parametrisation saturation.
266        let mut context = Context::new();
267        let x = variable(&mut context, "x");
268        let y = variable(&mut context, "y");
269        let t = variable(&mut context, "t");
270        let t_expression = builders::symbol(&mut context, "t");
271        let two = builders::integer_value(&mut context, 2);
272        let t_squared = builders::power(&mut context, Arc::clone(&t_expression), two);
273        let one = builders::one(&mut context);
274        let denominator = builders::sum(&mut context, vec![one.clone(), Arc::clone(&t_squared)]);
275        let numerator_x = builders::subtract(&mut context, one, Arc::clone(&t_squared));
276        let coordinate_x = builders::divide(&mut context, numerator_x, Arc::clone(&denominator));
277        let two_literal = builders::integer_value(&mut context, 2);
278        let two_t = builders::product(&mut context, vec![two_literal, Arc::clone(&t_expression)]);
279        let coordinate_y = builders::divide(&mut context, two_t, denominator);
280        let generators = implicitize(
281            &mut context,
282            &[x, y],
283            &[t],
284            &[
285                (x, Arc::clone(&coordinate_x)),
286                (y, Arc::clone(&coordinate_y)),
287            ],
288        )
289        .unwrap();
290        assert!(!generators.is_empty());
291        for generator in &generators {
292            assert!(vanishes_on_parameter_point(

294                generator,
295                &[

297                    (y, Arc::clone(&coordinate_y))
