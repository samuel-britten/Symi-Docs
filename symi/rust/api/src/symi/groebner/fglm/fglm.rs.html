19// Deterministic work counters for the FGLM change of order, frozen by the performance ratchet
20// (Decision 7): the quotient dimension D, the total nonzeros of the multiplication matrices,
21// the number of monomials the main loop processed, the number of linear-dependence relations
22// emitted (the size of the raw target basis), and the number of independent standard monomials
23// accepted (which equals D on a correct run). They localise a regression to a specific kind of
24// extra work rather than to a wall-clock reading.
25#[derive(Debug, Clone, Default)]
26/// Public native data structure named `fglm_statistics`.
27pub struct FglmStatistics {
28    /// Stores the `quotient_dimension` value.
29    pub quotient_dimension: usize,
30    /// Stores the `multiplication_matrix_nonzeros` value.
31    pub multiplication_matrix_nonzeros: usize,
32    /// Stores the `monomials_processed` value.
33    pub monomials_processed: usize,
34    /// Stores the `linear_dependence_relations` value.
35    pub linear_dependence_relations: usize,
36    /// Stores the `standard_monomials_accepted` value.
37    pub standard_monomials_accepted: usize,
38}

40/// Performs the advanced native `change_monomial_order` operation.
41pub fn change_monomial_order(
42    context: &Context,
43    ring: &PolynomialRing,
44    basis: &[ExpressionReference],
45    target: MonomialOrder,
46) -> Option<Vec<ExpressionReference>> {
47    // Convert a Gröbner basis of a zero-dimensional ideal from the source order (ring.order)
48    // to the `target` order by FGLM (Faugère–Gianni–Lazard–Mora 1993), reified to expressions
49    // (Decision 1). The input generators are passed through the canonical source-order basis
50    // computation first, so the FGLM staircase reasoning is valid even when a caller hands in
51    // a non-reduced generating set (idempotent on a true reduced basis). None when the ideal is
52    // positive-dimensional (no finite quotient basis) or any generator is outside Q[ring.variables]
53    // (Decision 1). The returned basis is the canonical reduced target basis — identical to
54    // a direct groebner_basis(target) computation (the machine-checked Decision-1 invariant).
55    let mut distributed: Vec<DistributedPolynomial> = Vec::with_capacity(basis.len());
56    for generator in basis {
57        distributed.push(expression_to_distributed_polynomial(
58            generator,
59            &ring.variables,
60        )?);
61    }
62    let source_basis = default_groebner_basis(&distributed, ring.order);
63    let converted = change_order_distributed(&source_basis, ring.order, target)?;
64    Some(
65        converted
66            .iter()
67            .map(|polynomial| {
68                distributed_polynomial_to_expression(context, polynomial, &ring.variables)
69            })
70            .collect(),
71    )
72}

74/// Performs the advanced native `change_order_distributed` operation.
75pub fn change_order_distributed(
76    basis: &[DistributedPolynomial],
77    source_order: MonomialOrder,
78    target_order: MonomialOrder,
79) -> Option<Vec<DistributedPolynomial>> {
80    // The distributed-level FGLM change of order, assuming `basis` is a Gröbner basis of a
81    // zero-dimensional ideal under source_order. None when the ideal is positive-dimensional.
82    // This is the entry the solving path and the performance ratchet call directly (they
83    // already hold a fast source basis from the engine selector).
84    change_order_distributed_with_statistics(basis, source_order, target_order)
85        .map(|(converted, _)| converted)
86}

88/// Performs the advanced native `change_order_distributed_with_statistics` operation.
89pub fn change_order_distributed_with_statistics(
90    basis: &[DistributedPolynomial],
91    source_order: MonomialOrder,
92    target_order: MonomialOrder,
93) -> Option<(Vec<DistributedPolynomial>, FglmStatistics)> {
94    // The FGLM main loop. Present Q[x]/I as a vector space from the source basis (None when
95    // positive-dimensional); precompute the per-variable multiplication matrices; then
96    // enumerate monomials in increasing target order from 1, maintaining the normal-form
97    // coordinate vector of each (incrementally, by applying a multiplication matrix to a
98    // predecessor's vector). A monomial whose vector is independent of the accepted standard
99    // monomials joins the target staircase and spawns its variable multiples; a dependent one
100    // yields a linear relation whose target-leading monomial is itself, contributing a target
101    // basis element. The raw basis is finished through autoreduce for the canonical
102    // reduced target basis (Decision 1).
103    let structure = QuotientRingStructure::build(basis, source_order)?;
104    let arity = structure.arity;
105    let mut statistics = FglmStatistics::default();
106    statistics.quotient_dimension = structure.dimension();
107
108    let matrices: Vec<MultiplicationMatrix> = (0..arity)
109        .map(|variable_index| structure.multiplication_matrix(variable_index))
110        .collect();
111    statistics.multiplication_matrix_nonzeros = matrices
112        .iter()
113        .map(MultiplicationMatrix::nonzero_count)
114        .sum();
115
116    let mut echelon = EchelonBasis::new();
117    let mut accepted_monomials: Vec<Monomial> = Vec::new();
118    let mut target_basis: Vec<DistributedPolynomial> = Vec::new();
119    let mut target_leading_monomials: Vec<Monomial> = Vec::new();
120
121    let unit = monomial_operations::constant_monomial(arity);
122    let unit_polynomial = distributed::from_terms(arity, vec![(unit.clone(), rational_one())]);
123    let unit_vector = structure.normal_form_coordinates(&unit_polynomial);
124    let mut seen: HashSet<Vec<u32>> = HashSet::new();
125    seen.insert(unit.exponents.clone());
126    let mut pending: Vec<(Monomial, Vec<exact_rational>)> = vec![(unit, unit_vector)];
127
128    while !pending.is_empty() {
129        let chosen = select_minimum(&pending, target_order);
130        let (current, current_vector) = pending.remove(chosen);
131        statistics.monomials_processed += 1;
132        if target_leading_monomials
133            .iter()
134            .any(|leading| monomial_operations::divides(leading, &current))
135        {
136            continue;
137        }
138        match echelon.reduce(&current_vector) {
139            Reduction::Dependent { coefficients } => {
140                statistics.linear_dependence_relations += 1;
141                let mut terms: Vec<(Monomial, exact_rational)> =
142                    vec![(current.clone(), rational_one())];
143                for (index, coefficient) in coefficients.iter().enumerate() {
144                    if *coefficient != rational_zero() {
145                        terms.push((accepted_monomials[index].clone(), -coefficient.clone()));
146                    }
147                }
148                target_basis.push(distributed::from_terms(arity, terms));
149                target_leading_monomials.push(current);
150            }
151            Reduction::Independent {
152                pivot,
153                reduced,
154                coefficients,
155            } => {
156                echelon.accept(current_vector.clone(), pivot, reduced, coefficients);
157                accepted_monomials.push(current.clone());
158                for variable_index in 0..arity {
159                    let variable = monomial_operations::variable_monomial(arity, variable_index);
160                    let child = monomial_operations::multiply(&variable, &current);
161                    if seen.insert(child.exponents.clone()) {
162                        let child_vector = matrices[variable_index].apply(&current_vector);
163                        pending.push((child, child_vector));
164                    }
165                }
166            }
167        }
168    }
169
170    statistics.standard_monomials_accepted = accepted_monomials.len();
171    let canonical = autoreduce(&target_basis, target_order);
172    Some((canonical, statistics))
173}

175fn select_minimum(pending: &[(Monomial, Vec<exact_rational>)], order: MonomialOrder) -> usize {
176    // The index of the pending monomial least under the target order. FGLM must process
177    // monomials in increasing target order so that, when a monomial is found dependent, every
178    // standard monomial in its relation has already been accepted — which is what makes the
179    // emitted relation's leading monomial the processed monomial itself.
180    let mut best = 0;
181    for index in 1..pending.len() {
182        if compare_monomials(order, &pending[index].0, &pending[best].0) == std::cmp::Ordering::Less
183        {
184            best = index;
185        }
186    }
187    best
188}

190// The incremental Gaussian eliminator FGLM tests linear dependence against. It keeps the
191// accepted standard monomials' raw coordinate vectors and a parallel row-echelon basis. Each
192// echelon row records the combination of raw vectors it represents, so a dependent test
193// recovers the coefficients of the relation directly, with no second pass.
194struct EchelonBasis {
195    raw_vectors: Vec<Vec<exact_rational>>,
196    rows: Vec<EchelonRow>,
197}
198
199struct EchelonRow {
200    pivot: usize,
201    reduced: Vec<exact_rational>,
202    coefficients: Vec<exact_rational>,
203}
204
205enum Reduction {
206    Dependent {
207        coefficients: Vec<exact_rational>,
208    },
209    Independent {
210        pivot: usize,
211        reduced: Vec<exact_rational>,
212        coefficients: Vec<exact_rational>,
213    },
214}
215
216impl EchelonBasis {
217    fn new() -> EchelonBasis {
218        EchelonBasis {
219            raw_vectors: Vec::new(),
220            rows: Vec::new(),
221        }
222    }
223
224    fn reduce(&self, vector: &[exact_rational]) -> Reduction {
225        // Reduce a candidate's coordinate vector against the echelon rows, tracking the
226        // accumulated combination of accepted raw vectors. If the reduction reaches zero the
227        // candidate is dependent and the accumulator is exactly the relation's coefficients; if
228        // a nonzero residue remains the candidate is independent, and the residue (normalised to
229        // a unit pivot) together with its combination becomes a new echelon row.
230        let mut working = vector.to_vec();
231        let mut accumulator = vec![rational_zero(); self.raw_vectors.len()];
232        for row in &self.rows {
233            let factor = working[row.pivot].clone();
234            if factor == rational_zero() {
235                continue;
236            }
237            for (slot, entry) in working.iter_mut().zip(row.reduced.iter()) {
238                if *entry != rational_zero() {
239                    *slot -= &factor * entry;
240                }
241            }
242            for (index, coefficient) in row.coefficients.iter().enumerate() {
243                if *coefficient != rational_zero() {
244                    accumulator[index] += &factor * coefficient;
245                }
246            }
247        }
248        match working.iter().position(|entry| *entry != rational_zero()) {
249            None => Reduction::Dependent {
250                coefficients: accumulator,
251            },
252            Some(pivot) => {
253                let inverse = rational_one() / working[pivot].clone();
254                let reduced: Vec<exact_rational> =
255                    working.iter().map(|entry| entry * &inverse).collect();
256                let mut coefficients: Vec<exact_rational> = accumulator
257                    .iter()
258                    .map(|value| -(value * &inverse))
259                    .collect();
260                coefficients.push(inverse);
261                Reduction::Independent {
262                    pivot,
263                    reduced,
264                    coefficients,
265                }
266            }
267        }
268    }
269
270    fn accept(
271        &mut self,
272        raw: Vec<exact_rational>,
273        pivot: usize,
274        reduced: Vec<exact_rational>,
275        coefficients: Vec<exact_rational>,
276    ) {
277        // Record a newly accepted standard monomial: its raw coordinate vector and the echelon
278        // row that reduces it.
279        self.raw_vectors.push(raw);
280        self.rows.push(EchelonRow {
281            pivot,
282            reduced,
283            coefficients,
284        });
285    }
286}

312        // of the degrevlex basis equals the direct lex basis as Vecs after autoreduce.

324            MonomialOrder::DegreeReverseLexicographic,
