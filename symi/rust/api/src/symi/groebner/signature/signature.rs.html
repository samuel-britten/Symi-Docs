6// A signature: a module monomial x^a · e_position in the free module R^m that maps
7// the standard basis vector e_i to the i-th input generator f_i (Decision 5).
8// The signature engine processes S-pairs (J-pairs) in increasing
9// signature order and skips any pair provably reducing to zero, identified by
10// these module monomials. The signature is carried alongside the ordinary
11// polynomial part of every generator and J-pair.
12#[derive(Debug, Clone, PartialEq, Eq, Hash)]
13/// Public native data structure named `signature`.
14pub struct Signature {
15    /// Stores the `position` value.
16    pub position: usize,
17    /// Stores the `monomial` value.
18    pub monomial: Monomial,
19}
20
21/// Performs the advanced native `scale` operation.
22pub fn scale(value: &Signature, factor: &Monomial) -> Signature {
23    // Multiply a signature by a ring monomial — the module action x^b · (x^a e_i) =
24    // x^(a+b) e_i — used to lift a generator's signature to a J-pair's signature.
25    Signature {
26        position: value.position,
27        monomial: monomial_operations::multiply(&value.monomial, factor),
28    }
29}
30
31/// Performs the advanced native `compare_signatures` operation.
32pub fn compare_signatures(order: MonomialOrder, left: &Signature, right: &Signature) -> Ordering {
33    // The signature (module) order: position-over-term (POT). Compare the module
34    // position first (the generator index, lower index ranks lower), and on a tie
35    // compare the term under the ring's own monomial order. This is a total order
36    // compatible with the ring order (x^a e_i < x^b e_i iff x^a < x^b), which is the
37    // admissibility property the signature engine's termination and the two criteria
38    // rely on.
39    match left.position.cmp(&right.position) {
40        Ordering::Equal => compare_monomials(order, &left.monomial, &right.monomial),
41        other => other,
42    }
43}

45/// Performs the advanced native `signature_divides` operation.
46pub fn signature_divides(divisor: &Signature, dividend: &Signature) -> bool {
47    // Module divisibility: x^a e_i divides x^b e_j exactly when the positions match
48    // and the term divides. Both signature criteria are stated in terms of this
49    // relation.
50    divisor.position == dividend.position
51        && monomial_operations::divides(&divisor.monomial, &dividend.monomial)
52}

54/// Performs the advanced native `covered_by_syzygy` operation.
55pub fn covered_by_syzygy(value: &Signature, syzygies: &[Signature]) -> bool {
56    // The F5 / syzygy criterion: a J-pair whose signature is divisible by the leading
57    // term of a known syzygy reduces to zero and is skipped. The syzygy list is
58    // seeded with the principal syzygies and grows whenever a reduction reaches zero
59    // (its signature is then a syzygy leading term).
60    syzygies
61        .iter()
62        .any(|syzygy| signature_divides(syzygy, value))
63}

65/// Performs the advanced native `rewritable` operation.
66pub fn rewritable(
67    target: &Signature,
68    generator_signatures: &[Signature],
69    source_index: usize,
70) -> bool {
71    // The rewritten (GVW) criterion: a J-pair with signature `target`, contributed by
72    // the generator at `source_index`, is redundant when a generator added later
73    // (a higher index, since the list is append-only in discovery order) carries a
74    // signature dividing `target`. That later generator already represents this
75    // signature class with at least as much reduction, so the J-pair can be dropped
76    // without losing any basis element. This is the criterion that, unlike the
77    // original F5's row bookkeeping, makes the engine terminate regardless of the
78    // pair-processing order (Eder–Faugère survey).
79    generator_signatures
80        .iter()
81        .enumerate()
82        .any(|(index, candidate)| index > source_index && signature_divides(candidate, target))
83}

98    #[test]
99    fn signature_order_is_total_and_antisymmetric_on_a_grid() {
100        // POT must be a total order: every distinct pair compares strictly, and the
101        // comparison is antisymmetric. Checked across positions and a term grid for
102        // every ring order.
103        let orders = [
104            MonomialOrder::Lexicographic,
105            MonomialOrder::DegreeLexicographic,
106            MonomialOrder::DegreeReverseLexicographic,
107        ];
108        let mut grid: Vec<Signature> = Vec::new();
109        for position in 0..3usize {
110            for a in 0..3u32 {
111                for b in 0..3u32 {
112                    grid.push(signature_of(position, &[a, b]));
113                }
114            }
115        }
116        for order in orders {
117            for left in &grid {
118                for right in &grid {
119                    let forward = compare_signatures(order, left, right);
120                    let backward = compare_signatures(order, right, left);
121                    assert_eq!(forward, backward.reverse());
122                    if left == right {
123                        assert_eq!(forward, Ordering::Equal);
124                    } else {
125                        assert_ne!(forward, Ordering::Equal);
126                    }
127                }
128            }
129        }
130    }

132    #[test]
133    fn position_dominates_term_in_pot() {
134        // Position-over-term: a higher position outranks any term at a lower position.
135        let order = MonomialOrder::DegreeReverseLexicographic;
136        let low_position_high_term = signature_of(0, &[9, 9]);
137        let high_position_low_term = signature_of(1, &[0, 0]);
138        assert_eq!(
139            compare_signatures(order, &high_position_low_term, &low_position_high_term),
140            Ordering::Greater
141        );
142    }

144    #[test]
145    fn syzygy_criterion_covers_multiples() {
146        // A signature is covered when a syzygy of the same position divides it.
147        let syzygies = vec![signature_of(1, &[1, 0])];
148        assert!(covered_by_syzygy(&signature_of(1, &[2, 3]), &syzygies));
149        assert!(!covered_by_syzygy(&signature_of(0, &[2, 3]), &syzygies));
150        assert!(!covered_by_syzygy(&signature_of(1, &[0, 3]), &syzygies));
151    }

153    #[test]
154    fn rewrite_criterion_prefers_later_generators() {
155        // The signature at index 1 divides the target, and index 1 is later than the
156        // source index 0, so the J-pair is rewritable; with the source already the
157        // latest divisor it is not.
158        let generator_signatures = vec![signature_of(0, &[1, 0]), signature_of(0, &[2, 0])];
159        let target = signature_of(0, &[3, 1]);
160        assert!(rewritable(&target, &generator_signatures, 0));
161        assert!(!rewritable(&target, &generator_signatures, 1));
162    }
