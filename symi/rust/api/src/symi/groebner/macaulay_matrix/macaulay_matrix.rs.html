16// The sparse rational Macaulay matrix that is F4's batched-reduction core
17// (Decision 3). Columns are a shared index of monomials sorted
18// descending under the ring order, so column 0 is the largest monomial and a
19// row's leading term is its smallest-indexed entry. Rows are sparse
20// (column_index, coefficient) lists kept ascending by column index. Structured
21// Gaussian elimination over Q reduces the matrix to row-echelon form; the rows
22// carrying a leading monomial that was not a leading monomial of any input row
23// are F4's new basis elements. This component is deliberately decoupled from the
24// symbolic `matrix` module (whose entries are full expressions, reduced
25// fraction-free): here the entries are rationals and the columns are monomials,
26// which is the only shape F4's hot loop wants. It is also the natural home for a
27// future modular kernel (Decision 4) and a future parallel kernel (Decision 8).
28/// Public native data structure named `macaulay_matrix`.
29pub struct MacaulayMatrix {
30    arity: usize,
31    columns: Vec<Monomial>,
32    rows: Vec<MatrixRow>,
33    input_leading_columns: HashSet<usize>,
34}

53struct SparseAccumulator {
54    values: Vec<Option<exact_rational>>,
55    touched_columns: Vec<usize>,
56}
57
58impl SparseAccumulator {
59    /// Creates a reusable column-keyed accumulator for one row operation.
60    fn new(column_count: usize) -> Self {
61        Self {
62            values: (0..column_count).map(|_| None).collect(),
63            touched_columns: Vec::new(),
64        }
65    }
66
67    /// Adds one coefficient to a column and records whether it collided.
68    fn insert(&mut self, column: usize, coefficient: exact_rational) {
69        match self.values.get_mut(column) {
70            Some(Some(existing)) => {
71                record_counter!(f4_row_collisions);
72                *existing += coefficient;
73            }
74            Some(slot @ None) => {
75                *slot = Some(coefficient);
76                self.touched_columns.push(column);
77            }
78            None => {}
79        }
80    }
81
82    /// Consumes the accumulator into a canonical ascending sparse row.
83    fn finish(&mut self) -> SparseRow {
84        self.touched_columns.sort_unstable();
85        let touched_columns = std::mem::take(&mut self.touched_columns);
86        touched_columns
87            .into_iter()
88            .filter_map(|column| {
89                self.values[column]
90                    .take()
91                    .filter(|coefficient| *coefficient != rational_zero())
92                    .map(|coefficient| (column, coefficient))
93            })
94            .collect()
95    }
96}

98impl DenseRow {
99    /// Converts a sparse row to dense coefficients with explicit occupancy bits.
100    fn from_sparse(row: &SparseRow, column_count: usize) -> Self {
101        let mut coefficients = vec![rational_zero(); column_count];
102        let mut occupied = vec![false; column_count];
103        for (column, coefficient) in row {
104            if *column >= column_count || *coefficient == rational_zero() {
105                continue;
106            }
107            coefficients[*column] = coefficient.clone();
108            occupied[*column] = true;
109        }
110        Self {
111            coefficients,
112            occupied,
113            nonzero_count: row.len(),
114        }
115    }
116
117    /// Returns the first occupied column in the dense row.
118    fn leading_column(&self) -> Option<usize> {
119        self.occupied.iter().position(|occupied| *occupied)
120    }
121}

123impl MatrixRow {
124    /// Selects sparse or dense storage from live density and predicted fill-in.
125    fn from_sparse(row: SparseRow, column_count: usize) -> Self {
126        Self::from_sparse_with_prediction(row, column_count, 0)
127    }
128
129    /// Selects dense storage when either current occupancy or projected fill-in is high.
130    fn from_sparse_with_prediction(
131        row: SparseRow,
132        column_count: usize,
133        predicted_nonzero_count: usize,
134    ) -> Self {
135        if should_use_dense(row.len().max(predicted_nonzero_count), column_count) {
136            MatrixRow::Dense(DenseRow::from_sparse(&row, column_count))
137        } else {
138            MatrixRow::Sparse(row)
139        }
140    }
141
142    /// Returns the first occupied column without rebuilding the row.
143    fn leading_column(&self) -> Option<usize> {
144        match self {
145            MatrixRow::Sparse(row) => row.first().map(|(column, _)| *column),
146            MatrixRow::Dense(row) => row.leading_column(),
147        }
148    }
149
150    /// Applies a visitor to all occupied entries in column order.
151    fn visit(&self, mut visitor: impl FnMut(usize, &exact_rational)) {
152        match self {
153            MatrixRow::Sparse(row) => {
154                for (column, coefficient) in row {
155                    visitor(*column, coefficient);
156                }
157            }
158            MatrixRow::Dense(row) => {
159                for column in 0..row.coefficients.len() {
160                    if row.occupied[column] {
161                        visitor(column, &row.coefficients[column]);
162                    }
163                }
164            }
165        }
166    }
167
168    /// Reads an occupied coefficient at a pivot column.
169    fn coefficient_at(&self, target_column: usize) -> Option<exact_rational> {
170        match self {
171            MatrixRow::Sparse(row) => row
172                .iter()
173                .find(|(column, _)| *column == target_column)
174                .map(|(_, coefficient)| coefficient.clone()),
175            MatrixRow::Dense(row) => row
176                .occupied
177                .get(target_column)
178                .copied()
179                .filter(|occupied| *occupied)
180                .map(|_| row.coefficients[target_column].clone()),
181        }
182    }
183
184    /// Finds the first occupied column with an already registered pivot.
185    fn first_reducible_column(&self, pivots: &[Option<MatrixRow>]) -> Option<usize> {
186        let mut reducible_column = None;
187        self.visit(|column, _| {
188            if reducible_column.is_none() && matches!(pivots.get(column), Some(Some(_))) {
189                reducible_column = Some(column);
190            }
191        });
192        reducible_column
193    }
194
195    /// Returns the number of occupied coefficients in the row.
196    fn nonzero_count(&self) -> usize {
197        match self {
198            MatrixRow::Sparse(row) => row.len(),
199            MatrixRow::Dense(row) => row.nonzero_count,
200        }
201    }
202}

204fn should_use_dense(nonzero_count: usize, column_count: usize) -> bool {
205    column_count != 0
206        && nonzero_count.saturating_mul(100)
207            >= column_count.saturating_mul(DENSE_ROW_THRESHOLD_PERCENT)
208}

219pub(crate) fn build_with_route(
220    polynomials: &[DistributedPolynomial],
221    arity: usize,
222    order: MonomialOrder,
223    route: ExecutionRoute,
224) -> MacaulayMatrix {
225    // Assemble the matrix from a set of polynomials: collect every occurring
226    // monomial into a column index sorted descending under the order, then encode
227    // each polynomial as a sparse row over those columns. The leading column of each
228    // input row (its smallest column index) is recorded so the reduction can later
229    // tell a genuinely new leading monomial from one already present among the
230    // inputs.
231    let mut occurring: HashSet<Monomial> = HashSet::new();
232    for polynomial in polynomials {
233        for (term_monomial, _) in &polynomial.terms {
234            occurring.insert(term_monomial.clone());
235        }
236    }
237    let mut columns: Vec<Monomial> = occurring.into_iter().collect();
238    columns.sort_by(|left, right| compare_monomials(order, right, left));
239    let mut column_of: HashMap<Monomial, usize> = HashMap::new();
240    for (index, column) in columns.iter().enumerate() {
241        column_of.insert(column.clone(), index);
242    }
243    let encoded_rows = crate::parallelism::ordered_map(route, polynomials, |polynomial| {
244        if distributed::is_zero(polynomial) {
245            return None;
246        }
247        let mut row: SparseRow = polynomial
248            .terms
249            .iter()
250            .map(|(term_monomial, coefficient)| (column_of[term_monomial], coefficient.clone()))
251            .collect();
252        row.sort_by_key(|(column, _)| *column);
253        Some(row)
254    });
255    let mut rows: Vec<MatrixRow> = Vec::with_capacity(polynomials.len());
256    let mut input_leading_columns: HashSet<usize> = HashSet::new();
257    for row in encoded_rows.into_iter().flatten() {
258        input_leading_columns.insert(row[0].0);
259        rows.push(MatrixRow::from_sparse(row, columns.len()));
260    }
261    MacaulayMatrix {
262        arity,
263        columns,
264        rows,
265        input_leading_columns,
266    }
267}

292    pub(crate) fn reduce_with_route(&mut self, route: ExecutionRoute) {
293        // Structured sparse Gaussian elimination over Q to row-echelon form. Each
294        // input row is reduced against the pivots discovered so far — eliminating any
295        // entry sitting in an already-claimed pivot column — and, if a nonzero
296        // residue survives, it is normalised monic and registered as a new pivot
297        // keyed on its leading (smallest-index) column. The elimination terminates
298        // because each step removes the smallest pivot-column entry and only
299        // introduces entries in strictly larger columns. On return `self.rows` holds
300        // exactly the pivot rows, one per pivot column, the basis of the input row
301        // space in echelon form.
302        let mut pivot_of_column: Vec<Option<MatrixRow>> =
303            (0..self.columns.len()).map(|_| None).collect();
304        let mut pivot_columns: Vec<usize> = Vec::new();
305        let mut accumulator = SparseAccumulator::new(self.columns.len());
306        let mut original_rows = std::mem::take(&mut self.rows);
307        let block_width = if route == ExecutionRoute::Parallel {
308            PARALLEL_ROW_BLOCK_WIDTH
309        } else {
310            1
311        };
312        while !original_rows.is_empty() {
313            let block_length = block_width.min(original_rows.len());
314            let block: Vec<MatrixRow> = original_rows.drain(..block_length).collect();
315            let reduced_rows = if block.len() > 1 {
316                reduce_against_snapshot(route, block, &pivot_of_column, self.columns.len())
317            } else {
318                block
319            };
320            for mut current in reduced_rows {
321                loop {
322                    let Some(column) = current.first_reducible_column(&pivot_of_column) else {
323                        break;
324                    };
325                    let Some(pivot) = pivot_of_column.get(column).and_then(Option::as_ref) else {
326                        break;
327                    };
328                    let Some(factor) = current.coefficient_at(column) else {
329                        break;
330                    };
331                    current = subtract_scaled(&current, pivot, &factor, &mut accumulator);
332                }
333                let Some(leading_column) = current.leading_column() else {
334                    continue;
335                };
336                normalize_monic(&mut current);
337                if let Some(slot) = pivot_of_column.get_mut(leading_column) {
338                    if slot.is_none() {
339                        *slot = Some(current);
340                        pivot_columns.push(leading_column);
341                    }
342                }
343            }
344        }
345        pivot_columns.sort_unstable();
346        self.rows = pivot_columns
347            .into_iter()
348            .filter_map(|column| pivot_of_column[column].take())
349            .collect();
350    }
351
352    /// Provides the `extract_new_generators` operation on this native type.
353    pub fn extract_new_generators(&self) -> Vec<DistributedPolynomial> {
354        // The new basis elements F4 takes from a reduced matrix: every pivot row whose
355        // leading monomial was not a leading monomial of any input row carries
356        // genuinely new leading-term information (Faugère's F̃⁺). Rows whose leading
357        // column was an input leading column reproduce the old leading terms and are
358        // dropped.
359        let mut result: Vec<DistributedPolynomial> = Vec::new();
360        for row in &self.rows {
361            if row.leading_column().is_none() {
362                continue;
363            }
364            let Some(leading_column) = row.leading_column() else {
365                continue;
366            };
367            if self.input_leading_columns.contains(&leading_column) {
368                continue;
369            }
370            result.push(self.row_to_polynomial(row));
371        }
372        result
373    }
374
375    /// Provides the `reduced_rows_as_polynomials` operation on this native type.
376    pub fn reduced_rows_as_polynomials(&self) -> Vec<DistributedPolynomial> {
377        // Every pivot row re-expanded to a polynomial — used by the cross-checks that
378        // compare the matrix against a direct reduction.
379        self.rows
380            .iter()
381            .map(|row| self.row_to_polynomial(row))
382            .collect()
383    }
384
385    fn row_to_polynomial(&self, row: &MatrixRow) -> DistributedPolynomial {
386        let mut terms = Vec::with_capacity(row.nonzero_count());
387        row.visit(|column, coefficient| {
388            terms.push((self.columns[column].clone(), coefficient.clone()));
389        });
390        distributed::from_terms(self.arity, terms)
391    }
392}

394fn reduce_against_snapshot(
395    route: ExecutionRoute,
396    rows: Vec<MatrixRow>,
397    pivots: &[Option<MatrixRow>],
398    column_count: usize,
399) -> Vec<MatrixRow> {
400    // Reduce a block of rows independently against a pivot set that no task may
401    // extend. The coordinator finishes each returned row against the pivots the block
402    // itself installs, so the published row is the same normal form the row-at-a-time
403    // reduction produces.
404    crate::parallelism::ordered_map(route, &rows, |original| {
405        let mut current = original.clone();
406        let mut accumulator = SparseAccumulator::new(column_count);
407        loop {
408            let Some(column) = current.first_reducible_column(pivots) else {
409                break;
410            };
411            let Some(pivot) = pivots.get(column).and_then(Option::as_ref) else {
412                break;
413            };
414            let Some(factor) = current.coefficient_at(column) else {
415                break;
416            };
417            current = subtract_scaled(&current, pivot, &factor, &mut accumulator);
418        }
419        current
420    })
421}

423fn normalize_monic(row: &mut MatrixRow) {
424    // Scale a row so its leading (smallest-index) entry is one, the monic
425    // normalisation that makes the pivot set canonical.
426    let Some(leading_column) = row.leading_column() else {
427        return;
428    };
429    let Some(leading) = row.coefficient_at(leading_column) else {
430        return;
431    };
432    if leading == rational_one() {
433        return;
434    }
435    match row {
436        MatrixRow::Sparse(entries) => {
437            for (_, coefficient) in entries {
438                record_counter!(exact_rational_operations);
439                *coefficient /= &leading;
440                observe_rational(coefficient);
441            }
442        }
443        MatrixRow::Dense(dense) => {
444            for column in 0..dense.coefficients.len() {
445                if dense.occupied[column] {
446                    record_counter!(exact_rational_operations);
447                    dense.coefficients[column] /= &leading;
448                    observe_rational(&dense.coefficients[column]);
449                }
450            }
451        }
452    }
453}

455fn subtract_scaled(
456    target: &MatrixRow,
457    source: &MatrixRow,
458    factor: &exact_rational,
459    accumulator: &mut SparseAccumulator,
460) -> MatrixRow {
461    // Compute `target - factor * source` on two column-ascending sparse rows by a
462    // linear merge, dropping any coefficient that cancels to zero. The leading entry
463    // of `source` sits in the column being eliminated with coefficient one (sources
464    // are monic pivots), so that column vanishes from the result.
465    target.visit(|column, coefficient| accumulator.insert(column, coefficient.clone()));
466    source.visit(|column, coefficient| {
467        record_counter!(exact_rational_operations);
468        accumulator.insert(column, -(coefficient * factor));
469    });
470    let result = accumulator.finish();
471    let predicted_nonzero_count = target
472        .nonzero_count()
473        .saturating_add(source.nonzero_count())
474        .min(accumulator.values.len());
475    MatrixRow::from_sparse_with_prediction(
476        result,
477        accumulator.values.len(),
478        predicted_nonzero_count,
479    )
480}

504    fn dense_reduced_row_echelon(matrix: &[Vec<exact_rational>]) -> Vec<Vec<exact_rational>> {
505        // A direct dense Gauss–Jordan reference, the oracle the sparse kernel is
506        // differentially checked against.
507        let mut rows: Vec<Vec<exact_rational>> = matrix.to_vec();
508        let column_count = rows.first().map(|row| row.len()).unwrap_or(0);
509        let mut pivot_row = 0;
510        for column in 0..column_count {
511            let mut selected = None;
512            for candidate in pivot_row..rows.len() {
513                if rows[candidate][column] != rational_zero() {
514                    selected = Some(candidate);
515                    break;
516                }
517            }
518            let Some(selected) = selected else { continue };
519            rows.swap(pivot_row, selected);
520            let leading = rows[pivot_row][column].clone();
521            for value in rows[pivot_row].iter_mut() {
522                *value /= &leading;
523            }
524            for other in 0..rows.len() {
525                if other == pivot_row {
526                    continue;
527                }
528                let factor = rows[other][column].clone();
529                if factor == rational_zero() {
530                    continue;
531                }
532                for index in 0..column_count {
533                    let subtrahend = &factor * &rows[pivot_row][index].clone();
534                    rows[other][index] -= subtrahend;
535                }
536            }
537            pivot_row += 1;
538        }
539        rows.retain(|row| row.iter().any(|value| *value != rational_zero()));
540        rows
541    }

551    #[test]
552    fn sparse_reduction_pivot_columns_match_dense_gauss_jordan() {
553        // The differential test: the sparse kernel's pivot columns are exactly the
554        // dense Gauss–Jordan pivot columns of the same matrix.
555        let order = MonomialOrder::Lexicographic;
556        let polynomials = vec![
557            distributed::from_terms(
558                2,
559                vec![term(&[2, 0], 1), term(&[1, 1], 2), term(&[0, 0], 3)],
560            ),
561            distributed::from_terms(
562                2,
563                vec![term(&[2, 0], 2), term(&[0, 2], 1), term(&[0, 0], 1)],
564            ),
565            distributed::from_terms(2, vec![term(&[1, 1], 1), term(&[0, 1], 4)]),
566        ];
567        let mut matrix = build(&polynomials, 2, order);
568        let columns = matrix.columns.clone();
569        let dense: Vec<Vec<exact_rational>> = polynomials
570            .iter()
571            .map(|polynomial| {
572                columns
573                    .iter()
574                    .map(|column| {
575                        polynomial
576                            .terms
577                            .iter()
578                            .find(|(term_monomial, _)| term_monomial == column)
579                            .map(|(_, coefficient)| coefficient.clone())
580                            .unwrap_or_else(rational_zero)
581                    })
582                    .collect()
583            })
584            .collect();
585        let dense_reduced = dense_reduced_row_echelon(&dense);
586        let dense_pivots: Vec<usize> = dense_reduced
587            .iter()
588            .map(|row| {
589                row.iter()
590                    .position(|value| *value != rational_zero())
591                    .unwrap()
592            })
593            .collect();
594        matrix.reduce();
595        assert_eq!(sparse_pivot_columns(&matrix), dense_pivots);
596    }
