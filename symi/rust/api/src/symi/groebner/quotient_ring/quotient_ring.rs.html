16// The finite-dimensional quotient ring Q[x]/I of a zero-dimensional ideal I, presented as a
17// Q-vector space. It is the substrate FGLM consumes: the standard monomials
18// (the staircase) are a basis of the quotient, the dimension D is the with-multiplicity count
19// of complex solutions, and the per-variable multiplication matrices encode the ring structure
20// FGLM walks. It is built from a Gröbner basis of I under a source order; every leading-term
21// computation uses that order, so the structure is exact for the (ideal, order) the caller
22// passes. The whole module operates on `distributed_polynomial`; expression-level callers use
23// the convenience wrappers at the bottom (Decision 3, 7).
24#[derive(Debug, Clone)]
25/// Public native data structure named `quotient_ring_structure`.
26pub struct QuotientRingStructure {
27    /// Stores the `arity` value.
28    pub arity: usize,
29    /// Stores the `order` value.
30    pub order: MonomialOrder,
31    /// Stores the `basis` value.
32    pub basis: Vec<DistributedPolynomial>,
33    /// Stores the `standard_monomials` value.
34    pub standard_monomials: Vec<Monomial>,
35    index_of_monomial: HashMap<Vec<u32>, usize>,
36}
37
38impl QuotientRingStructure {
39    /// Provides the `build` operation on this native type.
40    pub fn build(
41        basis: &[DistributedPolynomial],
42        order: MonomialOrder,
43    ) -> Option<QuotientRingStructure> {
44        // Present Q[x]/I as a vector space from a Gröbner basis of I, returning None when the
45        // ideal is not zero-dimensional (the staircase is infinite, so no finite quotient
46        // basis exists — the clean FGLM decline of Decision 1). The standard monomials are
47        // enumerated below the per-variable staircase bounds and indexed for coordinate
48        // lookups.
49        let arity = match basis.first() {
50            Some(first) => first.arity,
51            None => return None,
52        };
53        let leading_monomials = leading_monomials_of(basis, order);
54        let standard_monomials = standard_monomials_of(arity, &leading_monomials)?;
55        let mut index_of_monomial: HashMap<Vec<u32>, usize> = HashMap::new();
56        for (index, term_monomial) in standard_monomials.iter().enumerate() {
57            index_of_monomial.insert(term_monomial.exponents.clone(), index);
58        }
59        Some(QuotientRingStructure {
60            arity,
61            order,
62            basis: basis.to_vec(),
63            standard_monomials,
64            index_of_monomial,
65        })
66    }
67
68    /// Provides the `dimension` operation on this native type.
69    pub fn dimension(&self) -> usize {
70        // The Q-dimension D of the quotient — the number of standard monomials, equal to the
71        // number of complex solutions of I counted with multiplicity.
72        self.standard_monomials.len()
73    }
74
75    /// Provides the `coordinate_vector` operation on this native type.
76    pub fn coordinate_vector(&self, value: &DistributedPolynomial) -> Option<Vec<exact_rational>> {
77        // Express a polynomial already in normal form (a Q-combination of standard monomials)
78        // as its coordinate vector in the standard-monomial basis. None when some monomial is
79        // not standard, which signals the input was not reduced modulo the basis.
80        let mut coordinates = vec![rational_zero(); self.dimension()];
81        for (term_monomial, coefficient) in &value.terms {
82            let index = self.index_of_monomial.get(&term_monomial.exponents)?;
83            coordinates[*index] = coefficient.clone();
84        }
85        Some(coordinates)
86    }
87
88    /// Provides the `normal_form_coordinates` operation on this native type.
89    pub fn normal_form_coordinates(&self, value: &DistributedPolynomial) -> Vec<exact_rational> {
90        // The coordinate vector of `value` reduced modulo the source basis — the composition
91        // of the normal-form reducer (a fast chokepoint) and the coordinate map.
92        // The normal form of any polynomial modulo a Gröbner basis is a Q-combination of
93        // standard monomials, so the coordinate map never fails here.
94        let normal = multivariate_normal_form(value, &self.basis, self.order);
95        self.coordinate_vector(&normal)
96            .expect("the normal form of a polynomial lies in the span of the standard monomials")
97    }
98
99    /// Provides the `multiplication_matrix` operation on this native type.
100    pub fn multiplication_matrix(&self, variable_index: usize) -> MultiplicationMatrix {
101        // The matrix of multiplication-by-x_{variable_index} acting on Q[x]/I in the standard
102        // basis. Column j is the coordinate vector of the normal form of x_k · (standard
103        // monomial j) — built through the same reducer FGLM's inner loop uses (Decision 9).
104        let dimension = self.dimension();
105        let mut columns: Vec<Vec<exact_rational>> = Vec::with_capacity(dimension);
106        let variable = monomial_operations::variable_monomial(self.arity, variable_index);
107        for standard in &self.standard_monomials {
108            let shifted = monomial_operations::multiply(&variable, standard);
109            let polynomial = distributed::from_terms(self.arity, vec![(shifted, rational_one())]);
110            columns.push(self.normal_form_coordinates(&polynomial));
111        }
112        MultiplicationMatrix { columns }
113    }
114}

116// A D×D rational matrix stored by columns, the shape FGLM consumes: column j is the action of
117// the operator on the j-th standard-monomial basis vector, so applying it to a coordinate
118// vector is a sparse-friendly linear combination of its columns.
119#[derive(Debug, Clone)]
120/// Public native data structure named `multiplication_matrix`.
121pub struct MultiplicationMatrix {
122    /// Stores the `columns` value.
123    pub columns: Vec<Vec<exact_rational>>,
124}
125
126impl MultiplicationMatrix {
127    /// Provides the `apply` operation on this native type.
128    pub fn apply(&self, vector: &[exact_rational]) -> Vec<exact_rational> {
129        // Apply the operator to a coordinate vector: sum over j of vector[j] · column_j. This
130        // is the incremental step FGLM uses to get the normal-form vector of x_k · m from that
131        // of m without a fresh reduction.
132        let dimension = self.columns.len();
133        let mut result = vec![rational_zero(); dimension];
134        for (column, scalar) in self.columns.iter().zip(vector.iter()) {
135            if *scalar == rational_zero() {
136                continue;
137            }
138            for (slot, entry) in result.iter_mut().zip(column.iter()) {
139                if *entry != rational_zero() {
140                    *slot += entry * scalar;
141                }
142            }
143        }
144        result
145    }
146
147    /// Provides the `compose` operation on this native type.
148    pub fn compose(&self, inner: &MultiplicationMatrix) -> MultiplicationMatrix {
149        // The operator composition self ∘ inner, column by column. Used only by the
150        // structural commutativity check (the multiplication matrices of a commutative
151        // quotient ring must commute pairwise — a free, planted-bug-catching invariant).
152        let columns = inner
153            .columns
154            .iter()
155            .map(|column| self.apply(column))
156            .collect();
157        MultiplicationMatrix { columns }
158    }
159
160    /// Provides the `nonzero_count` operation on this native type.
161    pub fn nonzero_count(&self) -> usize {
162        // The number of nonzero entries — a deterministic size counter the performance
163        // ratchet freezes (Decision 7).
164        self.columns
165            .iter()
166            .flat_map(|column| column.iter())
167            .filter(|entry| **entry != rational_zero())
168            .count()
169    }
170}

172/// Performs the advanced native `leading_monomials_of` operation.
173pub fn leading_monomials_of(
174    basis: &[DistributedPolynomial],
175    order: MonomialOrder,
176) -> Vec<Monomial> {
177    // The leading monomials of a basis under the given order — the generators of the
178    // leading-term ideal whose complement is the staircase.
179    basis
180        .iter()
181        .filter_map(|polynomial| distributed::leading_monomial(polynomial, order))
182        .collect()
183}

185/// Performs the advanced native `is_zero_dimensional_distributed` operation.
186pub fn is_zero_dimensional_distributed(
187    basis: &[DistributedPolynomial],
188    order: MonomialOrder,
189) -> bool {
190    // The zero-dimensionality test from the staircase (Cox–Little–O'Shea ch. 5): I is
191    // zero-dimensional iff for every ring variable some leading monomial of the Gröbner basis
192    // is a pure power of that variable, equivalently the staircase is finite. The empty basis
193    // (the zero ideal) and any ideal missing a pure power for some variable are
194    // positive-dimensional.
195    if basis.is_empty() {
196        return false;
197    }
198    let arity = basis[0].arity;
199    let leading_monomials = leading_monomials_of(basis, order);
200    (0..arity).all(|variable_index| {
201        leading_monomials
202            .iter()
203            .any(|term_monomial| is_pure_power(term_monomial, variable_index))
204    })
205}

207fn is_pure_power(term_monomial: &Monomial, variable_index: usize) -> bool {
208    // True when the monomial involves only the named variable — every other exponent is zero.
209    // The constant monomial (all zeros) is a pure power of every variable, which makes the
210    // unit ideal register as zero-dimensional with an empty staircase.
211    term_monomial
212        .exponents
213        .iter()
214        .enumerate()
215        .all(|(index, exponent)| index == variable_index || *exponent == 0)
216}
217
218fn standard_monomials_of(arity: usize, leading_monomials: &[Monomial]) -> Option<Vec<Monomial>> {
219    // Enumerate the staircase — the monomials divisible by no leading monomial — when it is
220    // finite. Each axis is bounded by the smallest pure power of that variable in the
221    // leading-term ideal; the box those bounds cut out contains every standard monomial, and
222    // filtering it by divisibility yields exactly the staircase. None when some axis has no
223    // pure-power bound (a positive-dimensional ideal).
224    let mut bounds: Vec<u32> = Vec::with_capacity(arity);
225    for variable_index in 0..arity {
226        let mut bound: Option<u32> = None;
227        for term_monomial in leading_monomials {
228            if is_pure_power(term_monomial, variable_index) {
229                let exponent = term_monomial.exponents[variable_index];
230                bound = Some(bound.map_or(exponent, |current| current.min(exponent)));
231            }
232        }
233        bounds.push(bound?);
234    }
235    let mut standard: Vec<Monomial> = Vec::new();
236    let mut exponents = vec![0u32; arity];
237    enumerate_box(arity, &bounds, &mut exponents, 0, &mut |candidate| {
238        let term_monomial = Monomial {
239            exponents: candidate.to_vec(),
240        };
241        let divisible = leading_monomials
242            .iter()
243            .any(|leading| monomial_operations::divides(leading, &term_monomial));
244        if !divisible {
245            standard.push(term_monomial);
246        }
247    });
248    Some(standard)
249}
250
251fn enumerate_box(
252    arity: usize,
253    bounds: &[u32],
254    exponents: &mut Vec<u32>,
255    position: usize,
256    visit: &mut impl FnMut(&[u32]),
257) {
258    // Visit every exponent vector in the box product [0, bounds[0]) × … × [0, bounds[n-1]).
259    // An axis with bound zero makes the whole box empty, which is how the unit ideal yields no
260    // standard monomials.
261    if position == arity {
262        visit(exponents);
263        return;
264    }
265    for value in 0..bounds[position] {
266        exponents[position] = value;
267        enumerate_box(arity, bounds, exponents, position + 1, visit);
268    }
269    exponents[position] = 0;
270}

272/// Performs the advanced native `is_zero_dimensional` operation.
273pub fn is_zero_dimensional(ring: &PolynomialRing, basis: &[ExpressionReference]) -> Option<bool> {
274    // Whether the ideal generated by `basis` over Q[ring.variables] is zero-dimensional.
275    // The leading-term test needs a Gröbner basis, so the input generators are passed through
276    // the canonical basis computation first (idempotent on a basis that is already reduced).
277    // None when any generator is outside Q[ring.variables] (Decision 1).
278    let distributed = convert_generators(basis, ring)?;
279    let groebner = default_groebner_basis(&distributed, ring.order);
280    Some(is_zero_dimensional_distributed(&groebner, ring.order))
281}

283/// Performs the advanced native `quotient_ring_dimension` operation.
284pub fn quotient_ring_dimension(
285    ring: &PolynomialRing,
286    basis: &[ExpressionReference],
287) -> Option<usize> {
288    // The Q-dimension D of Q[I]/ring — the number of standard monomials, the with-multiplicity
289    // count of complex solutions. None outside Q[x]; zero when the ideal is positive-dimensional
290    // (no finite quotient) so a caller can distinguish "infinite" from "empty variety" by
291    // pairing this with is_zero_dimensional.
292    let distributed = convert_generators(basis, ring)?;
293    let groebner = default_groebner_basis(&distributed, ring.order);
294    match QuotientRingStructure::build(&groebner, ring.order) {
295        Some(structure) => Some(structure.dimension()),
296        None => Some(0),
297    }
298}

300fn convert_generators(
301    basis: &[ExpressionReference],
302    ring: &PolynomialRing,
303) -> Option<Vec<DistributedPolynomial>> {
304    // Convert generators to the distributed representation over the ring variables, declining
305    // (None) as soon as one is outside Q[ring.variables].
306    let mut distributed: Vec<DistributedPolynomial> = Vec::with_capacity(basis.len());
307    for generator in basis {
308        distributed.push(expression_to_distributed_polynomial(
309            generator,
310            &ring.variables,
311        )?);
312    }
313    Some(distributed)
314}

341    fn circle_hyperbola_is_zero_dimensional_with_dimension_four() {
342        // <x^2 + y^2 - 1, xy - 1> is a zero-dimensional ideal; its quotient has dimension 4,
343        // the Bezout count of two conics.
344        let order = MonomialOrder::DegreeReverseLexicographic;
345        let generators = vec![
346            distributed::from_terms(
347                2,
348                vec![term(&[2, 0], 1), term(&[0, 2], 1), term(&[0, 0], -1)],
349            ),
350            distributed::from_terms(2, vec![term(&[1, 1], 1), term(&[0, 0], -1)]),
351        ];
352        let basis = groebner(&generators, order);
353        assert!(is_zero_dimensional_distributed(&basis, order));
354        let structure = QuotientRingStructure::build(&basis, order).unwrap();
355        assert_eq!(structure.dimension(), 4);
356    }

387    #[test]
388    fn multiplication_matrices_commute() {
389        // A free structural correctness check independent of FGLM: the multiplication
390        // operators of a commutative quotient ring commute pairwise.
391        let order = MonomialOrder::DegreeReverseLexicographic;
392        let generators = vec![
393            distributed::from_terms(
394                2,
395                vec![term(&[2, 0], 1), term(&[0, 2], 1), term(&[0, 0], -1)],
396            ),
397            distributed::from_terms(2, vec![term(&[1, 1], 1), term(&[0, 0], -1)]),
398        ];
399        let basis = groebner(&generators, order);
400        let structure = QuotientRingStructure::build(&basis, order).unwrap();
401        let matrix_x = structure.multiplication_matrix(0);
402        let matrix_y = structure.multiplication_matrix(1);
403        let left = matrix_x.compose(&matrix_y);
404        let right = matrix_y.compose(&matrix_x);
405        assert_eq!(left.columns, right.columns);
406    }
