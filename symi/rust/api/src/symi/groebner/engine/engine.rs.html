21// The internal engine selector (Decision 6). Every engine returns the
22// same canonical reduced Gröbner basis for an (ideal, order); the choice changes only
23// the number of machine operations, never the output, so it is unobservable to a
24// caller except in speed. F4 is the default (msolve's choice, the most robust matrix
25// engine); Buchberger is retained as the differential oracle; GVW is the
26// signature-based engine. This enum and the two `*_with_*` functions are internal /
27// test-only Rust API — the selector adds no public binding.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29/// Public native enumeration named `basis_engine`.
30pub enum BasisEngine {
31    /// Represents the `buchberger` case.
32    Buchberger,
33    /// Represents the `faugere_f4` case.
34    FaugereF4,
35    /// Represents the `gao_volny_wang` case.
36    GaoVolnyWang,
37    /// Represents the certified word-prime multimodular F4 case.
38    ModularF4,
39}
40
41/// Public native constant `default_engine`.
42pub const DEFAULT_ENGINE: BasisEngine = BasisEngine::FaugereF4;

44// Deterministic work counters gauged by the performance ratchet (Decision 7). They
45// count machine-independent units — S-pairs actually processed, reductions reaching
46// zero, the peak Macaulay-matrix dimensions, and the peak intermediate basis size —
47// so a regression localises to a specific kind of extra work rather than to a flaky
48// wall-clock reading.
49#[derive(Debug, Clone, Default)]
50/// Public native data structure named `basis_statistics`.
51pub struct BasisStatistics {
52    /// Stores the `s_pairs_processed` value.
53    pub s_pairs_processed: usize,
54    /// Stores the `reductions_to_zero` value.
55    pub reductions_to_zero: usize,
56    /// Stores the `matrix_rows_peak` value.
57    pub matrix_rows_peak: usize,
58    /// Stores the `matrix_columns_peak` value.
59    pub matrix_columns_peak: usize,
60    /// Stores the `matrix_nonzeros_peak` value.
61    pub matrix_nonzeros_peak: usize,
62    /// Stores the `intermediate_basis_size_peak` value.
63    pub intermediate_basis_size_peak: usize,
64}
65
66impl BasisStatistics {
67    /// Provides the `observe_matrix` operation on this native type.
68    pub fn observe_matrix(&mut self, rows: usize, columns: usize, nonzeros: usize) {
69        self.matrix_rows_peak = self.matrix_rows_peak.max(rows);
70        self.matrix_columns_peak = self.matrix_columns_peak.max(columns);
71        self.matrix_nonzeros_peak = self.matrix_nonzeros_peak.max(nonzeros);
72    }
73
74    /// Provides the `observe_basis_size` operation on this native type.
75    pub fn observe_basis_size(&mut self, size: usize) {
76        self.intermediate_basis_size_peak = self.intermediate_basis_size_peak.max(size);
77    }
78}

80/// Performs the advanced native `default_groebner_basis` operation.
81pub fn default_groebner_basis(
82    generators: &[DistributedPolynomial],
83    order: MonomialOrder,
84) -> Vec<DistributedPolynomial> {
85    // The default distributed-level canonical reduced basis, dispatched to F4. Every
86    // ideal operation (membership, sum, product, intersection, elimination) calls
87    // through here, so the whole surface accelerates with no signature change
88    // (Decision 6).
89    groebner_basis_distributed(generators, order, DEFAULT_ENGINE)
90}

108/// Performs the advanced native `groebner_basis_distributed` operation.
109pub fn groebner_basis_distributed(
110    generators: &[DistributedPolynomial],
111    order: MonomialOrder,
112    engine: BasisEngine,
113) -> Vec<DistributedPolynomial> {
114    // Compute the canonical reduced basis with a chosen engine, at the distributed
115    // level. Whatever the engine, the raw basis is funnelled through this module's
116    // `autoreduce`, so the result is the unique reduced basis for (ideal, order) —
117    // identical across engines (Decision 1).
118    groebner_basis_distributed_with_statistics(generators, order, engine).0
119}

121/// Apply the operation's execution policy to one engine dispatch.
122///
123/// The estimate is the generators' total term count, which is what decides both how many
124/// independent images a modular engine can form and how wide an F4 matrix row batch is worth
125/// scheduling. The route it yields is the operation's, so a sequential session computes the same
126/// basis on the serial backend and a cancelled operation stops at the next batch boundary.
127pub(crate) fn groebner_basis_distributed_with_operation(
128    generators: &[DistributedPolynomial],
129    order: MonomialOrder,
130    engine: BasisEngine,
131    operation: &OperationContext,
132) -> (Vec<DistributedPolynomial>, BasisStatistics) {
133    let cancellation = operation.cancellation_token();
134    // The estimate sums the terms of every generator, so a dispatch the shipping policy cannot
135    // admit is answered from the class disposition instead and the generators are read only by
136    // the engine that computes the basis.
137    let route = if operation.may_fork_work_class("groebner_basis") {
138        operation
139            .execution_decision(&engine_work_estimate(generators), 1, None)
140            .route
141    } else {
142        ExecutionRoute::Serial
143    };
144    groebner_basis_distributed_with_route_and_cancellation(
145        generators,
146        order,
147        engine,
148        route,
149        &cancellation,
150    )
151}

153/// The deterministic work estimate one engine dispatch is scheduled against.
154pub(crate) fn engine_work_estimate(generators: &[DistributedPolynomial]) -> WorkEstimate {
155    let term_count = generators
156        .iter()
157        .map(|generator| generator.terms.len())
158        .sum::<usize>();
159    WorkEstimate::new(
160        u64::try_from(term_count.saturating_mul(16)).unwrap_or(u64::MAX),
161        generators.len().max(2),
162        term_count.saturating_mul(std::mem::size_of::<DistributedPolynomial>()),
163        0,
164        1,
165        "groebner_basis",
166    )
167}

169fn groebner_basis_distributed_with_route_and_cancellation(
170    generators: &[DistributedPolynomial],
171    order: MonomialOrder,
172    engine: BasisEngine,
173    route: ExecutionRoute,
174    cancellation: &CancellationToken,
175) -> (Vec<DistributedPolynomial>, BasisStatistics) {
176    // The engine dispatch plus its deterministic work counters. The Buchberger arm is
177    // an instrumented copy of the reference loop (`buchberger.rs` itself stays
178    // untouched) so the oracle can report `reductions_to_zero`, the quantity the
179    // signature engine drives to zero (Decision 7). The route selects only which
180    // backend executes an engine's independent work; every arm returns the same
181    // canonical reduced basis on either route.
182    let mut statistics = BasisStatistics::default();
183    let raw = match engine {
184        BasisEngine::Buchberger => buchberger_with_statistics(generators, order, &mut statistics),
185        BasisEngine::FaugereF4 => faugere_f4_with_route(generators, order, &mut statistics, route),
186        BasisEngine::GaoVolnyWang => gao_volny_wang(generators, order, &mut statistics),
187        BasisEngine::ModularF4 => certified_modular_basis_with_route_and_cancellation(
188            generators,
189            order,
190            route,
191            cancellation,
192        )
193        .unwrap_or_else(|| faugere_f4_with_route(generators, order, &mut statistics, route)),
194    };
195    (autoreduce(&raw, order), statistics)
196}

198/// Performs the advanced native `groebner_basis_distributed_with_statistics` operation.
199pub fn groebner_basis_distributed_with_statistics(
200    generators: &[DistributedPolynomial],
201    order: MonomialOrder,
202    engine: BasisEngine,
203) -> (Vec<DistributedPolynomial>, BasisStatistics) {
204    // The forced-sequential entry to the engine dispatch, and the reference every
205    // parallel route is compared against.
206    let cancellation = CancellationToken::new();
207    groebner_basis_distributed_with_route_and_cancellation(
208        generators,
209        order,
210        engine,
211        ExecutionRoute::Serial,
212        &cancellation,
213    )
214}

216/// Performs the advanced native `groebner_basis_with_engine` operation.
217pub fn groebner_basis_with_engine(
218    context: &Context,
219    ring: &PolynomialRing,
220    generators: &[ExpressionReference],
221    engine: BasisEngine,
222) -> Option<Vec<ExpressionReference>> {
223    // The canonical reduced basis over Q[ring.variables] with a chosen engine, reified
224    // to expressions. None outside Q[x] (Decision 1, unchanged). The public
225    // `groebner_basis` equals this with the F4 default.
226    Some(groebner_basis_with_statistics(context, ring, generators, engine)?.0)
227}

229/// Performs the advanced native `groebner_basis_with_statistics` operation.
230pub fn groebner_basis_with_statistics(
231    context: &Context,
232    ring: &PolynomialRing,
233    generators: &[ExpressionReference],
234    engine: BasisEngine,
235) -> Option<(Vec<ExpressionReference>, BasisStatistics)> {
236    // The same computation, returning the basis and its statistics. Declines (None) as soon as a
237    // generator is outside Q[ring.variables]. This is the context-carrying entry, so the engine
238    // runs under the operation's execution mode and cancellation; the distributed-level
239    // `groebner_basis_distributed_with_statistics` stays forced-sequential and is the reference
240    // the ratchet and the benchmarks measure against.
241    let _conversion_timer = start_phase(InstrumentationPhase::PolynomialConversion);
242    let mut distributed: Vec<DistributedPolynomial> = Vec::with_capacity(generators.len());
243    for generator in generators {
244        distributed.push(expression_to_distributed_polynomial(
245            generator,
246            &ring.variables,
247        )?);
248    }
249    drop(_conversion_timer);
250    let (basis, statistics) = groebner_basis_distributed_with_operation(
251        &distributed,
252        ring.order,
253        engine,
254        context.operation(),
255    );
256    let _reification_timer = start_phase(InstrumentationPhase::ExpressionReification);
257    let reified = basis
258        .iter()
259        .map(|polynomial| {
260            distributed_polynomial_to_expression(context, polynomial, &ring.variables)
261        })
262        .collect();
263    Some((reified, statistics))
264}

266fn buchberger_with_statistics(
267    generators: &[DistributedPolynomial],
268    order: MonomialOrder,
269    statistics: &mut BasisStatistics,
270) -> Vec<DistributedPolynomial> {
271    // An instrumented copy of `buchberger_with_strategy` (normal selection)
272    // that counts S-pairs processed and reductions reaching zero. The behaviour is
273    // identical to the reference implementation — the same product / chain criteria and the same
274    // completeness sweep — so it is a faithful oracle; only the counters are added,
275    // for the Decision-7 assertion that Buchberger performs reductions to zero where
276    // the signature engine performs none.
277    let mut basis: Vec<DistributedPolynomial> = Vec::new();
278    for generator in generators {
279        let monic = distributed::make_monic(generator, order);
280        if !distributed::is_zero(&monic) && !basis.iter().any(|existing| *existing == monic) {
281            basis.push(monic);
282        }
283    }
284    let mut pairs = PairSet::new();
285    for index in 0..basis.len() {
286        pairs.insert_pairs_for_new_basis_element(&basis, index, order);
287    }
288    while let Some(pair) = pairs.select(SelectionStrategy::Normal, order) {
289        let leading_monomials: Vec<Monomial> = basis
290            .iter()
291            .map(|polynomial| distributed::leading_monomial(polynomial, order).unwrap())
292            .collect();
293        if let Some((witness_index, dividing_leading_monomial)) =
294            chain_criterion_witness(&leading_monomials, &pairs, pair.left, pair.right, &pair.lcm)
295        {
296            pairs.reject_by_chain_criterion(&pair, witness_index, dividing_leading_monomial);
297            continue;
298        }
299        statistics.s_pairs_processed += 1;
300        record_counter!(pairs_processed);
301        record_counter!(pair_state_transitions);
302        let s = s_polynomial(&basis[pair.left], &basis[pair.right], order);
303        let remainder = multivariate_normal_form(&s, &basis, order);
304        pairs.mark_processed(&pair);
305        if distributed::is_zero(&remainder) {
306            statistics.reductions_to_zero += 1;
307        } else {
308            let monic = distributed::make_monic(&remainder, order);
309            let new_index = basis.len();
310            basis.push(monic);
311            pairs.insert_pairs_for_new_basis_element(&basis, new_index, order);
312        }
313        statistics.observe_basis_size(basis.len());
314    }
315    statistics.observe_basis_size(basis.len());
316    basis
317}

338    fn every_engine_returns_the_same_canonical_basis() {
339        // Decision 1, this module's central invariant: faugere_f4 == gao_volny_wang
340        // == buchberger as Vecs after autoreduce.
341        let order = MonomialOrder::DegreeReverseLexicographic;
342        let f = distributed::from_terms(
343            2,
344            vec![term(&[2, 0], 1), term(&[0, 2], 1), term(&[0, 0], -1)],
345        );
346        let g = distributed::from_terms(2, vec![term(&[1, 1], 1), term(&[0, 0], -1)]);
347        let generators = vec![f, g];
348        let buchberger = groebner_basis_distributed(&generators, order, BasisEngine::Buchberger);
349        let f4 = groebner_basis_distributed(&generators, order, BasisEngine::FaugereF4);
350        let gvw = groebner_basis_distributed(&generators, order, BasisEngine::GaoVolnyWang);
351        assert_eq!(f4, buchberger);
352        assert_eq!(gvw, buchberger);
353    }

373    #[test]
374    fn the_modular_engine_reads_its_route_from_the_operation() {
375        // The engine dispatch is reached with a context, so the route it runs on is the
376        // operation's: a sequential session computes the basis on the serial backend, an
377        // admitted one reaches the pool, and the canonical reduced basis is the same either way.
378        //
379        // The class is serial-only, and its public entry is hard-wired serial besides, so the
380        // admission list is what keeps the modular image coordinator exercised at all.
381        //
382        // The observable is the parallel route itself. An operation that cannot fork — by its
383        // mode, by cancellation, or by the class disposition — is answered before it forms the
384        // estimate the policy would refuse, so it records no decision at all; only an operation
385        // that reaches the pool has one to record.
386        let order = MonomialOrder::DegreeReverseLexicographic;
387        let generators = vec![
388            distributed::from_terms(2, vec![term(&[2, 0], 1), term(&[0, 1], -1)]),
389            distributed::from_terms(2, vec![term(&[1, 1], 1), term(&[0, 0], -1)]),
390        ];
391        let _ = crate::parallelism::initialize_native_parallelism(4);
392
393        let sequential = OperationContext::new(0);
394        let (sequential_basis, _) = groebner_basis_distributed_with_operation(
395            &generators,
396            order,
397            BasisEngine::ModularF4,
398            &sequential,
399        );
400        assert!(
401            !sequential
402                .take_records()
403                .iter()
404                .any(|record| record.detail().contains("parallel route")),
405            "a sequential operation reached the parallel backend"
406        );
407
408        let automatic = OperationContext::with_configuration(
409            0,
410            OperationConfiguration {
411                execution_mode: ExecutionMode::Automatic,
412                admitted_serial_only_work_classes: &["groebner_basis"],
413                ..OperationConfiguration::default()
414            },
415        );
416        let (automatic_basis, _) = groebner_basis_distributed_with_operation(
417            &generators,
418            order,
419            BasisEngine::ModularF4,
420            &automatic,
421        );
422        assert_eq!(automatic_basis, sequential_basis);
423        assert!(
424            automatic.take_records().iter().any(|record| record
425                .detail()
426                .contains("parallel route: work class: groebner_basis")),
427            "an admitted automatic operation did not reach the parallel backend"
428        );
429
430        let cancelled = OperationContext::with_configuration(
431            0,
432            OperationConfiguration {
433                execution_mode: ExecutionMode::Automatic,
434                admitted_serial_only_work_classes: &["groebner_basis"],
435                ..OperationConfiguration::default()
436            },
437        );
438        cancelled.cancel();
439        let (cancelled_basis, _) = groebner_basis_distributed_with_operation(
440            &generators,
441            order,
442            BasisEngine::ModularF4,
443            &cancelled,
444        );
445        assert_eq!(cancelled_basis, sequential_basis);
446        assert!(
447            !cancelled
448                .take_records()
449                .iter()
450                .any(|record| record.detail().contains("parallel route")),
451            "a cancelled operation reached the parallel backend"
452        );
453    }
