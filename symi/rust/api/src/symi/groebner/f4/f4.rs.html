16/// Performs the advanced native `faugere_f4` operation.
17pub fn faugere_f4(
18    generators: &[DistributedPolynomial],
19    order: MonomialOrder,
20    statistics: &mut BasisStatistics,
21) -> Vec<DistributedPolynomial> {
22    faugere_f4_with_route(generators, order, statistics, ExecutionRoute::Serial)
23}

25pub(crate) fn faugere_f4_with_route(
26    generators: &[DistributedPolynomial],
27    order: MonomialOrder,
28    statistics: &mut BasisStatistics,
29    route: ExecutionRoute,
30) -> Vec<DistributedPolynomial> {
31    // No cancellation signal is presented to the run below, so its pair set is always driven to
32    // exhaustion and the basis it returns is always the complete one. The completeness answer is
33    // read rather than asserted, so this entry never has to fabricate a basis it did not compute.
34    let (basis, _completed) = run_faugere_f4(generators, order, statistics, route, None);
35    basis
36}

38/// Compute the F4 basis of an ideal, stopping if the operation is cancelled.
39///
40/// A partial pair set is not a Gröbner basis, so a run that stops early answers `None` rather
41/// than the basis it had reached: an interrupted computation declines, it does not publish an
42/// incomplete answer that a caller could mistake for a complete one.
43pub(crate) fn faugere_f4_with_route_and_cancellation(
44    generators: &[DistributedPolynomial],
45    order: MonomialOrder,
46    statistics: &mut BasisStatistics,
47    route: ExecutionRoute,
48    cancellation: &CancellationToken,
49) -> Option<Vec<DistributedPolynomial>> {
50    let (basis, completed) =
51        run_faugere_f4(generators, order, statistics, route, Some(cancellation));
52    completed.then_some(basis)
53}

55fn run_faugere_f4(
56    generators: &[DistributedPolynomial],
57    order: MonomialOrder,
58    statistics: &mut BasisStatistics,
59    route: ExecutionRoute,
60    cancellation: Option<&CancellationToken>,
61) -> (Vec<DistributedPolynomial>, bool) {
62    // F4 (Faugère 1999): batch a whole degree's worth of S-pairs into one Macaulay
63    // matrix and reduce it once, instead of reducing one S-polynomial at a time.
64    // The product criterion is applied through the same Update used by Buchberger
65    // when
66    // pairs are inserted and the chain criterion filters the batch; the surviving
67    // pairs' S-polynomial halves seed the matrix, symbolic preprocessing collects
68    // their reductors from the current basis, and a single Gaussian elimination over
69    // Q yields the batch's new leading terms. Every pair remains selected until its
70    // matrix contribution has been accounted for, after which it receives a resolved
71    // disposition. The loop therefore ends only after the explicit pair set is
72    // complete, and the caller then autoreduces to the canonical basis (Decision 1).
73    let mut basis = initial_basis(generators, order);
74    if basis.is_empty() {
75        return (basis, true);
76    }
77    let mut pairs = PairSet::new();
78    for index in 0..basis.len() {
79        pairs.insert_pairs_for_new_basis_element(&basis, index, order);
80    }
81    loop {
82        // The degree batch is F4's unit of progress, so it is also the granularity at which a
83        // cancellation request can be honoured without leaving the pair set half-consumed.
84        if cancellation.is_some_and(CancellationToken::is_cancelled) {
85            statistics.observe_basis_size(basis.len());
86            return (basis, false);
87        }
88        let batch = pairs.select_minimal_degree_batch(order);
89        if batch.is_empty() {
90            break;
91        }
92        let leading_monomials = leading_monomial_list(&basis, order);
93        let preparations = crate::parallelism::ordered_map(route, &batch, |pair| {
94            let chain_witness = chain_criterion_witness(
95                &leading_monomials,
96                &pairs,
97                pair.left,
98                pair.right,
99                &pair.lcm,
100            );
101            let rows = chain_witness.is_none().then(|| {
102                (
103                    s_polynomial_half(&basis[pair.left], &pair.lcm, order),
104                    s_polynomial_half(&basis[pair.right], &pair.lcm, order),
105                )
106            });
107            (pair.clone(), chain_witness, rows)
108        });
109        let mut rows: Vec<DistributedPolynomial> = Vec::new();
110        for (pair, chain_witness, pair_rows) in preparations {
111            if let Some((witness_index, dividing_leading_monomial)) = chain_witness {
112                pairs.reject_by_chain_criterion(&pair, witness_index, dividing_leading_monomial);
113                continue;
114            }
115            statistics.s_pairs_processed += 1;
116            record_counter!(pairs_processed);
117            record_counter!(pair_state_transitions);
118            if let Some((left_row, right_row)) = pair_rows {
119                rows.push(left_row);
120                rows.push(right_row);
121            }
122        }
123        if rows.is_empty() {
124            for pair in &batch {
125                pairs.mark_processed(pair);
126            }
127            continue;
128        }
129        let all_rows = symbolic_preprocessing_with_route(rows, &basis, order, route);
130        let arity = all_rows[0].arity;
131        let _matrix_build_timer = start_phase(InstrumentationPhase::MatrixBuild);
132        let mut matrix = macaulay_matrix::build_with_route(&all_rows, arity, order, route);
133        drop(_matrix_build_timer);
134        record_counter!(f4_rows_constructed, matrix.row_count());
135        statistics.observe_matrix(
136            matrix.row_count(),
137            matrix.column_count(),
138            matrix.nonzero_count(),
139        );
140        let _elimination_timer = start_phase(InstrumentationPhase::Elimination);
141        matrix.reduce_with_route(route);
142        drop(_elimination_timer);
143        let _insertion_timer = start_phase(InstrumentationPhase::Insertion);
144        let new_generators = matrix.extract_new_generators();
145        for pair in &batch {
146            pairs.mark_processed(pair);
147        }
148        for new_polynomial in new_generators {
149            if distributed::is_zero(&new_polynomial) {
150                continue;
151            }
152            let monic = distributed::make_monic(&new_polynomial, order);
153            let index = basis.len();
154            basis.push(monic);
155            pairs.insert_pairs_for_new_basis_element(&basis, index, order);
156            record_counter!(f4_output_generators);
157        }
158        drop(_insertion_timer);
159        statistics.observe_basis_size(basis.len());
160    }
161    statistics.observe_basis_size(basis.len());
162    (basis, true)
163}

165fn initial_basis(
166    generators: &[DistributedPolynomial],
167    order: MonomialOrder,
168) -> Vec<DistributedPolynomial> {
169    // The starting basis: every generator made monic, dropping zeros and duplicates,
170    // matching the Buchberger initialisation exactly so the two engines begin from
171    // the same set.
172    let mut basis: Vec<DistributedPolynomial> = Vec::new();
173    for generator in generators {
174        let monic = distributed::make_monic(generator, order);
175        if !distributed::is_zero(&monic) && !basis.iter().any(|existing| *existing == monic) {
176            basis.push(monic);
177        }
178    }
179    basis
180}

189fn s_polynomial_half(
190    polynomial: &DistributedPolynomial,
191    lcm: &Monomial,
192    order: MonomialOrder,
193) -> DistributedPolynomial {
194    // One side of an S-polynomial as an F4 row: (lcm / leading_monomial(f)) · f,
195    // scaled to a monic leading coefficient so the two halves of a pair share the
196    // same leading term and cancel under row reduction.
197    let leading_monomial = distributed::leading_monomial(polynomial, order).unwrap();
198    let leading_coefficient = distributed::leading_coefficient(polynomial, order).unwrap();
199    let factor_monomial = monomial_operations::divide(lcm, &leading_monomial).unwrap();
200    let factor_coefficient = rational_one() / leading_coefficient;
201    distributed::monomial_multiply(polynomial, &factor_monomial, &factor_coefficient)
202}

204fn symbolic_preprocessing_with_route(
205    rows: Vec<DistributedPolynomial>,
206    basis: &[DistributedPolynomial],
207    order: MonomialOrder,
208    route: ExecutionRoute,
209) -> Vec<DistributedPolynomial> {
210    // F4's symbolic preprocessing: close the row set under reduction. Every monomial
211    // appearing in any row that is divisible by some basis leading monomial — and is
212    // not already the leading monomial of a row — gets a reductor row
213    // (monomial / leading_monomial(g)) · g adjoined, whose own monomials are then
214    // examined in turn. On completion every reducible monomial has a pivot row, so the
215    // single Gaussian elimination performs exactly the reductions Buchberger would,
216    // sharing every common sub-reduction. The closure runs in two stages: the
217    // admission stage walks the monomial worklist and records which basis element and
218    // cofactor each new row comes from, using monomial arithmetic alone, and the
219    // construction stage then builds those rows. Only the construction stage carries
220    // coefficient arithmetic, so it is the stage handed to the route; the admitted
221    // sequence — and therefore the row order the elimination sees — is the same
222    // whichever backend builds the rows.
223    let _phase_timer = start_phase(InstrumentationPhase::SymbolicPreprocessing);
224    record_counter!(f4_rows_constructed, rows.len());
225    let arity = rows.first().map_or(0, |row| row.arity);
226    let mut arena = MonomialArena::new(arity);
227    let leading_identifiers: Vec<_> = basis
228        .iter()
229        .map(|polynomial| arena.intern(&distributed::leading_monomial(polynomial, order).unwrap()))
230        .collect();
231    let mut rows = rows;
232    let mut done: HashSet<usize> = rows
233        .iter()
234        .filter_map(|row| {
235            distributed::leading_monomial(row, order).map(|monomial| arena.intern(&monomial))
236        })
237        .collect();
238    let mut seen: HashSet<usize> = HashSet::new();
239    let mut worklist: Vec<usize> = Vec::new();
240    for row in &rows {
241        for (term_monomial, _) in &row.terms {
242            record_counter!(f4_symbolic_monomials_visited);
243            let identifier = arena.intern(term_monomial);
244            if seen.insert(identifier) {
245                worklist.push(identifier);
246            }
247        }
248    }
249    let mut admitted_reductors: Vec<(usize, Monomial)> = Vec::new();
250    while let Some(target) = worklist.pop() {
251        if done.contains(&target) {
252            continue;
253        }
254        for generator_index in 0..basis.len() {
255            record_counter!(f4_reductor_lookups);
256            let generator_leading_identifier = leading_identifiers[generator_index];
257            if !arena.divides(generator_leading_identifier, target) {
258                continue;
259            }
260            let Some(factor_identifier) = arena.divide(target, generator_leading_identifier) else {
261                continue;
262            };
263            let factor_monomial = arena.to_monomial(factor_identifier);
264            done.insert(target);
265            for (term_monomial, _) in &basis[generator_index].terms {
266                let shifted_monomial =
267                    monomial_operations::multiply(term_monomial, &factor_monomial);
268                let identifier = arena.intern(&shifted_monomial);
269                if seen.insert(identifier) {
270                    worklist.push(identifier);
271                }
272            }
273            admitted_reductors.push((generator_index, factor_monomial));
274            record_counter!(f4_rows_constructed);
275            break;
276        }
277    }
278    let reductors = crate::parallelism::ordered_map(
279        route,
280        &admitted_reductors,
281        |(generator_index, factor_monomial)| {
282            distributed::monomial_multiply(
283                &basis[*generator_index],
284                factor_monomial,
285                &rational_one(),
286            )
287        },
288    );
289    rows.extend(reductors);
290    rows
291}

313    fn buchberger_criterion_holds(basis: &[DistributedPolynomial], order: MonomialOrder) -> bool {
314        for i in 0..basis.len() {
315            for j in (i + 1)..basis.len() {
316                let s = s_polynomial(&basis[i], &basis[j], order);
317                let remainder = multivariate_normal_form(&s, basis, order);
318                if !distributed::is_zero(&remainder) {
319                    return false;
320                }
321            }
322        }
323        true
324    }

386    fn run_unpruned_buchberger(
387        generators: &[DistributedPolynomial],
388        order: MonomialOrder,
389    ) -> Vec<DistributedPolynomial> {
390        let mut basis: Vec<DistributedPolynomial> = generators
391            .iter()
392            .map(|generator| distributed::make_monic(generator, order))
393            .filter(|generator| !distributed::is_zero(generator))
394            .collect();
395        let mut pairs = Vec::new();
396        for right in 0..basis.len() {
397            for left in 0..right {
398                pairs.push((left, right));
399            }
400        }
401        while let Some((left, right)) = pairs.pop() {
402            let obstruction = s_polynomial(&basis[left], &basis[right], order);
403            let remainder = multivariate_normal_form(&obstruction, &basis, order);
404            if distributed::is_zero(&remainder) {
405                continue;
406            }
407            let new_index = basis.len();
408            basis.push(distributed::make_monic(&remainder, order));
409            for old_index in 0..new_index {
410                pairs.push((old_index, new_index));
411            }
412        }
413        autoreduce(&basis, order)
414    }

423    fn generated_polynomial(state: &mut u64, arity: usize) -> DistributedPolynomial {
424        let term_count = 2 + usize::try_from(next_generated_value(state) % 3).unwrap_or(0);
425        let mut terms = Vec::with_capacity(term_count);
426        for _ in 0..term_count {
427            let exponents: Vec<u32> = (0..arity)
428                .map(|_| u32::try_from(next_generated_value(state) % 4).unwrap_or(0))
429                .collect();
430            let coefficient = i64::try_from(next_generated_value(state) % 5).unwrap_or(0) - 2;
431            if coefficient != 0 {
432                terms.push(term(&exponents, coefficient));
433            }
434        }
435        distributed::from_terms(arity, terms)
436    }

612            distributed::from_terms(3, vec![term(&[1, 3, 0], 1), term(&[0, 0, 3], -1)]),
613        ];
614        let _ = crate::parallelism::initialize_native_parallelism(4);
615        let serial = symbolic_preprocessing_with_route(
616            seed_rows.clone(),
617            &basis,
618            order,
619            ExecutionRoute::Serial,
620        );
621        let parallel =
622            symbolic_preprocessing_with_route(seed_rows, &basis, order, ExecutionRoute::Parallel);
623        assert!(parallel.len() > 2);
624        assert_eq!(parallel, serial);
625    }
