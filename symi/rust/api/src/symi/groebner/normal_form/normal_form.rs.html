18struct SparseAccumulator {
19    coefficients: HashMap<MonomialIdentifier, exact_rational>,
20    positions: HashMap<MonomialIdentifier, usize>,
21    heap: Vec<MonomialIdentifier>,
22    order: MonomialOrder,
23}
24
25impl SparseAccumulator {
26    fn new(order: MonomialOrder) -> Self {
27        Self {
28            coefficients: HashMap::new(),
29            positions: HashMap::new(),
30            heap: Vec::new(),
31            order,
32        }
33    }
34
35    fn is_greater(
36        &self,
37        left: MonomialIdentifier,
38        right: MonomialIdentifier,
39        arena: &MonomialArena,
40    ) -> bool {
41        compare_identifiers(arena, self.order, left, right) == std::cmp::Ordering::Greater
42    }
43
44    fn add(
45        &mut self,
46        identifier: MonomialIdentifier,
47        coefficient: exact_rational,
48        arena: &MonomialArena,
49    ) {
50        if coefficient == rational_zero() {
51            return;
52        }
53        if let Some(existing) = self.coefficients.get_mut(&identifier) {
54            record_counter!(normal_form_accumulator_collisions);
55            *existing += coefficient;
56            if *existing == rational_zero() {
57                self.coefficients.remove(&identifier);
58                self.remove(identifier, arena);
59            }
60            return;
61        }
62        self.coefficients.insert(identifier, coefficient);
63        self.positions.insert(identifier, self.heap.len());
64        self.heap.push(identifier);
65        self.sift_up(self.heap.len() - 1, arena);
66        record_counter!(normal_form_accumulator_insertions);
67    }
68
69    fn sift_up(&mut self, mut index: usize, arena: &MonomialArena) {
70        while index > 0 {
71            let parent = (index - 1) / 2;
72            if !self.is_greater(self.heap[index], self.heap[parent], arena) {
73                break;
74            }
75            self.heap.swap(index, parent);
76            self.positions.insert(self.heap[index], index);
77            self.positions.insert(self.heap[parent], parent);
78            index = parent;
79        }
80    }
81
82    fn sift_down(&mut self, mut index: usize, arena: &MonomialArena) {
83        loop {
84            let left = index * 2 + 1;
85            let right = left + 1;
86            let mut greatest = index;
87            if left < self.heap.len()
88                && self.is_greater(self.heap[left], self.heap[greatest], arena)
89            {
90                greatest = left;
91            }
92            if right < self.heap.len()
93                && self.is_greater(self.heap[right], self.heap[greatest], arena)
94            {
95                greatest = right;
96            }
97            if greatest == index {
98                break;
99            }
100            self.heap.swap(index, greatest);
101            self.positions.insert(self.heap[index], index);
102            self.positions.insert(self.heap[greatest], greatest);
103            index = greatest;
104        }
105    }
106
107    fn remove(&mut self, identifier: MonomialIdentifier, arena: &MonomialArena) {
108        let Some(index) = self.positions.remove(&identifier) else {
109            return;
110        };
111        let Some(last) = self.heap.pop() else {
112            return;
113        };
114        if index < self.heap.len() {
115            self.heap[index] = last;
116            self.positions.insert(last, index);
117            self.sift_down(index, arena);
118            self.sift_up(index, arena);
119        }
120    }
121
122    fn pop_max(&mut self, arena: &MonomialArena) -> Option<(MonomialIdentifier, exact_rational)> {
123        let identifier = *self.heap.first()?;
124        let coefficient = self.coefficients.remove(&identifier)?;
125        self.positions.remove(&identifier);
126        let last = self.heap.pop()?;
127        if !self.heap.is_empty() {
128            self.heap[0] = last;
129            self.positions.insert(last, 0);
130            self.sift_down(0, arena);
131        }
132        Some((identifier, coefficient))
133    }
134}

136struct OrderedReducer {
137    polynomial: OrderedPolynomial,
138    leading_identifier: MonomialIdentifier,
139    leading_coefficient: exact_rational,
140}
141
142struct ReducerSet {
143    reducers: Vec<OrderedReducer>,
144}
145
146impl ReducerSet {
147    fn new(
148        divisors: &[&DistributedPolynomial],
149        arena: &mut MonomialArena,
150        order: MonomialOrder,
151    ) -> Option<Self> {
152        let mut reducers = Vec::new();
153        for divisor in divisors {
154            let polynomial = OrderedPolynomial::from_distributed(divisor, arena, order)?;
155            if let Some((leading_identifier, leading_coefficient)) = polynomial.leading() {
156                let leading_coefficient = leading_coefficient.clone();
157                reducers.push(OrderedReducer {
158                    polynomial,
159                    leading_identifier,
160                    leading_coefficient,
161                });
162            }
163        }
164        Some(Self { reducers })
165    }
166
167    fn find(
168        &self,
169        identifier: MonomialIdentifier,
170        arena: &mut MonomialArena,
171    ) -> Option<(usize, MonomialIdentifier)> {
172        let degree = arena.total_degree(identifier);
173        for (index, reducer) in self.reducers.iter().enumerate() {
174            record_counter!(normal_form_divisor_candidates_inspected);
175            if arena.total_degree(reducer.leading_identifier) > degree
176                || !arena.divides(reducer.leading_identifier, identifier)
177            {
178                continue;
179            }
180            return Some((index, arena.divide(identifier, reducer.leading_identifier)?));
181        }
182        None
183    }
184}

186fn ordered_multivariate_division(
187    dividend: &DistributedPolynomial,
188    divisors: &[&DistributedPolynomial],
189    order: MonomialOrder,
190) -> Option<DivisionResult> {
191    let mut arena = MonomialArena::new(dividend.arity);
192    let ordered_dividend = OrderedPolynomial::from_distributed(dividend, &mut arena, order)?;
193    let reducers = ReducerSet::new(divisors, &mut arena, order)?;
194    let mut accumulator = SparseAccumulator::new(order);
195    for (identifier, coefficient) in ordered_dividend.terms() {
196        accumulator.add(*identifier, coefficient.clone(), &arena);
197    }
198    let mut remainder_terms = Vec::new();
199    let mut quotient_terms: Vec<Vec<(MonomialIdentifier, exact_rational)>> =
200        vec![Vec::new(); reducers.reducers.len()];
201    while let Some((identifier, coefficient)) = accumulator.pop_max(&arena) {
202        let Some((reducer_index, quotient_identifier)) = reducers.find(identifier, &mut arena)
203        else {
204            remainder_terms.push((identifier, coefficient));
205            record_counter!(normal_form_accumulator_insertions);
206            continue;
207        };
208        let reducer = &reducers.reducers[reducer_index];
209        let quotient_coefficient = coefficient.clone() / reducer.leading_coefficient.clone();
210        record_counter!(exact_rational_operations);
211        quotient_terms[reducer_index].push((quotient_identifier, quotient_coefficient.clone()));
212        record_counter!(normal_form_quotient_terms_emitted);
213        record_counter!(normal_form_cancellations);
214        for (term_identifier, term_coefficient) in reducer.polynomial.terms() {
215            if *term_identifier == reducer.leading_identifier {
216                continue;
217            }
218            let product_identifier = arena.multiply(quotient_identifier, *term_identifier)?;
219            let contribution = -(quotient_coefficient.clone() * term_coefficient.clone());
220            record_counter!(exact_rational_operations);
221            accumulator.add(product_identifier, contribution, &arena);
222        }
223    }
224    let ring = ordered_dividend.ring;
225    let remainder = OrderedPolynomial {
226        ring,
227        terms: remainder_terms,
228    };
229    let quotients = quotient_terms
230        .into_iter()
231        .map(|terms| OrderedPolynomial { ring, terms }.to_distributed(&arena))
232        .collect();
233    Some(DivisionResult {
234        quotients,
235        remainder: remainder.to_distributed(&arena),
236    })
237}

239fn legacy_multivariate_division(
240    dividend: &DistributedPolynomial,
241    divisors: &[DistributedPolynomial],
242    order: MonomialOrder,
243) -> DivisionResult {
244    let arity = dividend.arity;
245    let mut quotients = vec![distributed::zero(arity); divisors.len()];
246    let mut remainder = distributed::zero(arity);
247    let mut current = dividend.clone();
248    while !distributed::is_zero(&current) {
249        let Some(lead_index) = distributed::leading_index(&current, order) else {
250            break;
251        };
252        let lead_monomial = current.terms[lead_index].0.clone();
253        let lead_coefficient = current.terms[lead_index].1.clone();
254        let mut divided = false;
255        for (divisor_index, divisor) in divisors.iter().enumerate() {
256            let Some(divisor_leading_monomial) = distributed::leading_monomial(divisor, order)
257            else {
258                continue;
259            };
260            if !crate::polynomial::monomial::divides(&divisor_leading_monomial, &lead_monomial) {
261                continue;
262            }
263            let divisor_leading_coefficient =
264                distributed::leading_coefficient(divisor, order).unwrap();
265            let factor_monomial =
266                crate::polynomial::monomial::divide(&lead_monomial, &divisor_leading_monomial)
267                    .unwrap();
268            let factor_coefficient = &lead_coefficient / &divisor_leading_coefficient;
269            let factor_term = distributed::from_terms(
270                arity,
271                vec![(factor_monomial.clone(), factor_coefficient.clone())],
272            );
273            quotients[divisor_index] = distributed::add(&quotients[divisor_index], &factor_term);
274            let subtrahend =
275                distributed::monomial_multiply(divisor, &factor_monomial, &factor_coefficient);
276            current = distributed::subtract(&current, &subtrahend);
277            divided = true;
278            break;
279        }
280        if !divided {
281            let lead_term = distributed::from_terms(arity, vec![(lead_monomial, lead_coefficient)]);
282            remainder = distributed::add(&remainder, &lead_term);
283            current = distributed::subtract(&current, &lead_term);
284        }
285    }
286    DivisionResult {
287        quotients,
288        remainder,
289    }
290}

292/// Performs the advanced native `multivariate_division` operation.
293pub fn multivariate_division(
294    dividend: &DistributedPolynomial,
295    divisors: &[DistributedPolynomial],
296    order: MonomialOrder,
297) -> DivisionResult {
298    let references: Vec<&DistributedPolynomial> = divisors.iter().collect();
299    ordered_multivariate_division(dividend, &references, order)
300        .unwrap_or_else(|| legacy_multivariate_division(dividend, divisors, order))
301}

303/// Reduces a value using borrowed divisors without cloning the divisor list.
304pub(crate) fn multivariate_normal_form_with_references(
305    value: &DistributedPolynomial,
306    divisors: &[&DistributedPolynomial],
307    order: MonomialOrder,
308) -> DistributedPolynomial {
309    ordered_multivariate_division(value, divisors, order)
310        .map(|result| result.remainder)
311        .unwrap_or_else(|| {
312            let owned: Vec<DistributedPolynomial> =
313                divisors.iter().map(|value| (*value).clone()).collect();
314            legacy_multivariate_division(value, &owned, order).remainder
315        })
316}

318/// Performs the advanced native `multivariate_normal_form` operation.
319pub fn multivariate_normal_form(
320    value: &DistributedPolynomial,
321    divisors: &[DistributedPolynomial],
322    order: MonomialOrder,
323) -> DistributedPolynomial {
324    multivariate_division(value, divisors, order).remainder
325}

348    fn reconstruct(
349        dividend: &DistributedPolynomial,
350        result: &DivisionResult,
351        divisors: &[DistributedPolynomial],
352    ) -> bool {
353        let mut total = distributed::zero(dividend.arity);
354        for (quotient, divisor) in result.quotients.iter().zip(divisors.iter()) {
355            total = distributed::add(&total, &distributed::multiply(quotient, divisor));
356        }
357        total = distributed::add(&total, &result.remainder);
358        distributed::is_zero(&distributed::subtract(dividend, &total))
359    }
