9/// Performs the advanced native `interval` operation.
10pub fn interval(
11    context: &Context,
12    lower_bound: ExpressionReference,
13    upper_bound: ExpressionReference,
14    lower_inclusive: bool,
15    upper_inclusive: bool,
16) -> ExpressionReference {
17    // Build a real-line interval routed through auto-simplification. Auto-
18    // simplification collapses degenerate shapes (empty interval, closed
19    // single-point interval to finite_set, the full real line interval to
20    // real_line) and forces an endpoint-at-infinity inclusivity flag to false
21    // so the canonical interner key is stable across every input phrasing.
22    simplify_interval(
23        context,
24        lower_bound,
25        upper_bound,
26        lower_inclusive,
27        upper_inclusive,
28    )
29}

111/// Performs the advanced native `fresh_integer_parameter` operation.
112pub fn fresh_integer_parameter(context: &Context, prefix: &str) -> SymbolIdentifier {
113    // Mint a brand-new binder for an image_set that ranges over the integers
114    // and record the integer domain on the new symbol. Without it the binder
115    // carries the neutral default domain, so every downstream query about a
116    // lattice point — realness, sign, definedness — sees an unconstrained
117    // unknown and a co-factor such as (sin(n*pi)^2 + 1)^(-2) cannot be proved
118    // finite there. The domain is recorded at the binding site rather than as a
119    // durable assertion: the binder belongs to the set being built, not to the
120    // caller's context, so a solve that mints one must leave the context's
121    // assumption base exactly as it found it.
122    let parameter = context.symbol_table.fresh(prefix);
123    context.symbol_table.declare_binder_predicates(
124        parameter,
125        &[crate::assumptions::CurrentScalarPredicate::Integer],
126    );
127    parameter
128}

130/// Performs the advanced native `image_set` operation.
131pub fn image_set(
132    context: &Context,
133    parameter_variable: SymbolIdentifier,
134    parameter_domain: ExpressionReference,
135    lambda_expression: ExpressionReference,
136) -> ExpressionReference {
137    // Build an image set { lambda_expression : parameter_variable in
138    // parameter_domain } routed through auto-simplification. The
139    // parameter_variable is a binder over lambda_expression; the simplifier
140    // collapses the trivial cases (empty parameter_domain, constant lambda
141    // not depending on parameter_variable, finite_set parameter_domain that
142    // can be enumerated element-by-element) before returning the structural
143    // placeholder.
144    simplify_image_set(
145        context,
146        parameter_variable,
147        parameter_domain,
148        lambda_expression,
149    )
150}
