1// The symbolic image of a set under a point map.
2//
3// A pre-image computation is the composition of two questions: what map inverts the
4// expression around the solve variable, and where does that map send the target set. The
5// second question has an answer in the language of set expressions themselves — the image
6// of `{ lambda(n) : n in Z }` under `t -> (t - b)/a` is `{ (lambda(n) - b)/a : n in Z }` —
7// so it needs neither an equation solver nor an enumerable set.
8//
9// This module answers the second question. It transports a set expression through a map
10// symbolically, never point by point, over the shapes the set language admits: the empty
11// set, a finite set, an image set, a union, an interval, the real line and the complex
12// plane. Union and intersection are not re-implemented here; `sets::builders::set_union`
13// and `sets::queries::intersect` already own them, and the image of a union is the union
14// of the images, which is the union rule below.
15//
16// Every rule is either total or declines. A map whose behaviour on a shape is not settled
17// by an exact premise — the orientation of an interval under a scale of unproved sign, the
18// reciprocal at a point whose vanishing is undecided — returns `None`, and the caller
19// keeps whatever route it already had. The algebra can therefore only make an answer
20// cheaper, never weaker.
21
22use std::sync::Arc;

24use crate::assumptions::TruthValue;
25use crate::expression::builders;
26use crate::expression::{BuiltinConstant, ExpressionKind, ExpressionReference};

28use crate::sets::builders::{
29    complex_plane, empty_set, finite_set, image_set, interval, real_line, set_union,

32pub(crate) enum PointMap {
33    /// `t -> t`.
34    Identity,
35    /// `t -> scale * t + offset`, for a variable-free, provably nonzero `scale`. Negation
36    /// is this map with a scale of minus one; it needs no variant of its own.
37    Affine {
38        scale: ExpressionReference,
39        offset: ExpressionReference,
40    },
41    /// `t -> 1 / t`, a bijection of the punctured line onto itself. Zero is not in its
42    /// domain and not in its range, so a zero point of the source set contributes nothing
43    /// to the image rather than contributing an undefined value.
44    Reciprocal,
45}
46
47impl PointMap {
48    /// The value of the map at one point, built through the auto-simplifying builders so
49    /// the result is in the canonical form every other route would have produced.
50    pub(crate) fn at(&self, context: &Context, point: &ExpressionReference) -> ExpressionReference {
51        match self {
52            PointMap::Identity => Arc::clone(point),
53            PointMap::Affine { scale, offset } => {
54                let scaled = builders::product(context, vec![Arc::clone(scale), Arc::clone(point)]);
55                builders::sum(context, vec![scaled, Arc::clone(offset)])
56            }
57            PointMap::Reciprocal => {
58                let one = builders::one(context);
59                builders::divide(context, one, Arc::clone(point))
60            }
61        }
62    }
63
64    fn admits_point(&self, context: &Context, point: &ExpressionReference) -> Option<bool> {
65        // Whether the map is defined at this point. `Some(true)` maps it, `Some(false)`
66        // drops it — a proof that the point has no pre-image, which keeps the image
67        // complete — and `None` declines the whole transport, because an image missing a
68        // point it could not classify would be a false completeness claim.
69        match self {
70            PointMap::Identity | PointMap::Affine { .. } => Some(true),
71            PointMap::Reciprocal => {
72                match crate::assumptions::range_evidence::nonzero_verdict(
73                    &context.symbol_table,
74                    point,
75                ) {
76                    TruthValue::Yes => Some(true),
77                    TruthValue::No => Some(false),
78                    TruthValue::Unknown => None,
79                }
80            }
81        }
82    }
83
84    fn preserves_the_whole_line(&self) -> bool {
85        // A map is a bijection of the whole line, and of the whole plane, exactly when it
86        // is affine with a nonzero scale; the construction site of `Affine` is where that
87        // scale was proved nonzero. The reciprocal misses zero, so it does not qualify.
88        matches!(self, PointMap::Identity | PointMap::Affine { .. })
89    }
90
91    fn interval_orientation(&self, context: &Context) -> Option<Orientation> {
92        // How the map reorders an ordered pair of endpoints. Only a proved sign settles
93        // it: an affine scale of undecided sign leaves the image endpoints in an unknown
94        // order, and an interval built from a reversed pair is a different set. The
95        // reciprocal is monotone on each side of zero but not across it, and deciding
96        // which side an interval lies on is a separate premise this rule does not carry.
97        match self {
98            PointMap::Identity => Some(Orientation::Increasing),
99            PointMap::Affine { scale, .. } => {
100                if crate::assumptions::range_evidence::positive_verdict(
101                    &context.symbol_table,
102                    scale,
103                ) == TruthValue::Yes
104                {
105                    return Some(Orientation::Increasing);
106                }
107                if crate::assumptions::range_evidence::negative_verdict(
108                    &context.symbol_table,
109                    scale,
110                ) == TruthValue::Yes
111                {
112                    return Some(Orientation::Decreasing);
113                }
114                let zero = builders::zero(context);
115                match crate::exact_real_order::compare_exact_real_expressions(context, scale, &zero)
116                {
117                    Ok(std::cmp::Ordering::Greater) => Some(Orientation::Increasing),
118                    Ok(std::cmp::Ordering::Less) => Some(Orientation::Decreasing),
119                    _ => None,
120                }
121            }
122            PointMap::Reciprocal => None,
123        }
124    }
125}
126
127#[derive(Clone, Copy, PartialEq, Eq)]
128enum Orientation {
129    Increasing,
130    Decreasing,
131}
132
133pub(crate) fn image_of_set(
134    context: &Context,
135    map: &PointMap,
136    set_expression: &ExpressionReference,
137) -> Option<ExpressionReference> {
138    // The image of a set expression under a point map, computed symbolically. `None` is a
139    // decline over a shape the algebra does not cover, never a claim that the image is
140    // empty.
141    match &set_expression.kind {
142        ExpressionKind::Constant(BuiltinConstant::EmptySet) => Some(empty_set(context)),
143        ExpressionKind::Constant(BuiltinConstant::RealLine) if map.preserves_the_whole_line() => {
144            Some(real_line(context))
145        }
146        ExpressionKind::Constant(BuiltinConstant::ComplexPlane)
147            if map.preserves_the_whole_line() =>
148        {
149            Some(complex_plane(context))
150        }
151        ExpressionKind::FiniteSet(points) => {
152            let points = points.clone();
153            let mut mapped: Vec<ExpressionReference> = Vec::with_capacity(points.len());
154            for point in &points {
155                if !map.admits_point(context, point)? {
156                    continue;
157                }
158                mapped.push(map.at(context, point));
159            }
160            Some(finite_set(context, mapped))
161        }
162        ExpressionKind::ImageSet(parameter_variable, parameter_domain, lambda_expression) => {
163            // The map composes with the parameterisation, so the family keeps exactly the
164            // binder and index set it had. The admissibility premise is discharged on the
165            // parameterisation itself, which stands for every point of the family at once.
166            let parameter_variable = *parameter_variable;
167            let parameter_domain = Arc::clone(parameter_domain);
168            let lambda_expression = Arc::clone(lambda_expression);
169            if !map.admits_point(context, &lambda_expression)? {
170                return Some(empty_set(context));
171            }
172            let mapped_lambda = map.at(context, &lambda_expression);
173            Some(image_set(
174                context,
175                parameter_variable,
176                parameter_domain,
177                mapped_lambda,
178            ))
179        }
180        ExpressionKind::SetUnion(operands) => {
181            // The image of a union is the union of the images, under any map at all.
182            let operands = operands.clone();
183            let mut pieces: Vec<ExpressionReference> = Vec::with_capacity(operands.len());
184            for operand in &operands {
185                pieces.push(image_of_set(context, map, operand)?);
186            }
187            Some(set_union(context, pieces))
188        }
189        ExpressionKind::Interval(lower_bound, upper_bound, lower_inclusive, upper_inclusive) => {
190            let lower_bound = Arc::clone(lower_bound);
191            let upper_bound = Arc::clone(upper_bound);
192            let lower_inclusive = *lower_inclusive;
193            let upper_inclusive = *upper_inclusive;
194            let orientation = map.interval_orientation(context)?;
195            let mapped_lower = map.at(context, &lower_bound);
196            let mapped_upper = map.at(context, &upper_bound);
197            Some(match orientation {
198                Orientation::Increasing => interval(
199                    context,
200                    mapped_lower,
201                    mapped_upper,
202                    lower_inclusive,
203                    upper_inclusive,
204                ),
205                Orientation::Decreasing => interval(
206                    context,
207                    mapped_upper,
208                    mapped_lower,
209                    upper_inclusive,
210                    lower_inclusive,
211                ),
212            })
213        }
214        _ => None,
215    }
