12/// Performs the advanced native `simplify_interval` operation.
13pub fn simplify_interval(
14    context: &Context,
15    lower_bound: ExpressionReference,
16    upper_bound: ExpressionReference,
17    lower_inclusive: bool,
18    upper_inclusive: bool,
19) -> ExpressionReference {
20    // Auto-simplify an interval. Detects the full-real-line shape, the
21    // empty-interval shapes, the closed-point collapse to finite_set, and
22    // forces an endpoint-at-infinity inclusivity flag to false (positive
23    // infinity is never an element). Anything else is interned as a
24    // structural interval with the (possibly tightened) inclusivity flags.
25    let lower_is_negative_infinity = is_negative_infinity(&lower_bound);
26    let upper_is_positive_infinity = is_positive_infinity(&upper_bound);
27    let lower_inclusive_final = if lower_is_negative_infinity {
28        false
29    } else {
30        lower_inclusive
31    };
32    let upper_inclusive_final = if upper_is_positive_infinity {
33        false
34    } else {
35        upper_inclusive
36    };
37    if lower_is_negative_infinity && upper_is_positive_infinity {
38        return real_line_constant(context);
39    }
40    if is_undefined(&lower_bound) || is_undefined(&upper_bound) {
41        return undefined_constant(context);
42    }
43    if lower_bound.kind == upper_bound.kind {
44        if !lower_inclusive_final || !upper_inclusive_final {
45            return empty_set_constant(context);
46        }
47        if is_finite_endpoint(&lower_bound) {
48            return simplify_finite_set(context, vec![lower_bound]);
49        }
50        return empty_set_constant(context);
51    }
52    if let Some(ordering) = numeric_endpoint_compare(&lower_bound, &upper_bound) {
53        match ordering {
54            Ordering::Greater => {
55                return empty_set_constant(context);
56            }
57            Ordering::Equal => {
58                if !lower_inclusive_final || !upper_inclusive_final {
59                    return empty_set_constant(context);
60                }
61                return simplify_finite_set(context, vec![lower_bound]);
62            }
63            Ordering::Less => {}
64        }
65    }
66    if lower_is_negative_infinity || upper_is_positive_infinity {
67        // The non-infinite endpoint is necessarily strictly inside the
68        // half-line that the infinity opens; nothing else to check.
69    }
70    context.interner.intern(ExpressionKind::Interval(
71        lower_bound,
72        upper_bound,
73        lower_inclusive_final,
74        upper_inclusive_final,
75    ))
76}

78fn numeric_endpoint_compare(
79    left: &ExpressionReference,
80    right: &ExpressionReference,
81) -> Option<Ordering> {
82    // Compare two interval endpoints numerically when both sides are exact
83    // numeric leaves (integer or rational). The infinity sentinels are
84    // resolved separately: negative_infinity sits strictly below every
85    // finite value, positive_infinity sits strictly above. Symbolic
86    // endpoints return None, leaving the interval structurally interned
87    // without collapse so the caller does not lose information.
88    if is_negative_infinity(left) {
89        if is_negative_infinity(right) {
90            return Some(Ordering::Equal);
91        }
92        return Some(Ordering::Less);
93    }
94    if is_positive_infinity(left) {
95        if is_positive_infinity(right) {
96            return Some(Ordering::Equal);
97        }
98        return Some(Ordering::Greater);
99    }
100    if is_negative_infinity(right) {
101        return Some(Ordering::Greater);
102    }
103    if is_positive_infinity(right) {
104        return Some(Ordering::Less);
105    }
106    let left_value = numeric_value(left)?;
107    let right_value = numeric_value(right)?;
108    Some(left_value.cmp(&right_value))
109}
110
111fn numeric_value(value: &ExpressionReference) -> Option<Rational> {
112    // Lift an exact numeric leaf to a Rational for the endpoint comparison.
113    // Returns None for any non-numeric kind so the interval simplifier
114    // refuses to apply the empty-interval collapse on symbolic input.
115    match &value.kind {
116        ExpressionKind::Integer(integer_value) => Some(Rational::from(integer_value.clone())),
117        ExpressionKind::Rational(rational_value) => Some(rational_value.clone()),
118        _ => None,
119    }
120}

122/// Performs the advanced native `simplify_finite_set` operation.
123pub fn simplify_finite_set(
124    context: &Context,
125    elements: Vec<ExpressionReference>,
126) -> ExpressionReference {
127    // Auto-simplify a finite set. Deduplicates and sorts the children under
128    // Cohen ordering and collapses the empty case to the empty_set constant.
129    if elements.is_empty() {
130        return empty_set_constant(context);
131    }
132    let mut canonical: Vec<ExpressionReference> = elements;
133    canonical.sort_by(|left, right| cohen_compare(left, right, &context.symbol_table));
134    canonical.dedup_by(|left, right| left.kind == right.kind);
135    if canonical.is_empty() {
136        return empty_set_constant(context);
137    }
138    context
139        .interner
140        .intern(ExpressionKind::FiniteSet(canonical))
141}

143/// Performs the advanced native `simplify_set_union` operation.
144pub fn simplify_set_union(
145    context: &Context,
146    operands: Vec<ExpressionReference>,
147) -> ExpressionReference {
148    // Auto-simplify a set union. Splices nested set_union children,
149    // removes empty_set operands, collapses to complex_plane whenever any
150    // operand is complex_plane (the ambient universe absorbs every other set)
151    // and to real_line whenever any operand is real_line and none is
152    // complex_plane, merges all finite_set operands into a single flat
153    // finite_set (union of finite sets is finite), and sorts the surviving
154    // operands under Cohen ordering with structural deduplication.
155    let mut flattened: Vec<ExpressionReference> = Vec::with_capacity(operands.len());
156    let mut finite_set_elements: Vec<ExpressionReference> = Vec::new();
157    let mut has_finite_sets = false;
158    let mut has_real_line = false;
159    for operand in operands {
160        match &operand.kind {
161            ExpressionKind::SetUnion(children) => {
162                for child in children {
163                    flattened.push(child.clone());
164                }
165            }
166            ExpressionKind::Constant(BuiltinConstant::EmptySet) => {}
167            ExpressionKind::Constant(BuiltinConstant::ComplexPlane) => {
168                return complex_plane_constant(context);
169            }
170            ExpressionKind::Constant(BuiltinConstant::RealLine) => {
171                has_real_line = true;
172            }
173            ExpressionKind::FiniteSet(elements) => {
174                has_finite_sets = true;
175                for element in elements {
176                    finite_set_elements.push(element.clone());
177                }
178            }
179            _ => {
180                flattened.push(operand);
181            }
182        }
183    }
184    if has_real_line {
185        return real_line_constant(context);
186    }
187    if has_finite_sets {
188        let mut remaining_points = Vec::new();
189        for point in finite_set_elements {
190            let mut absorbed = false;
191            for operand in flattened.iter_mut() {
192                let ExpressionKind::Interval(lower, upper, lower_inclusive, upper_inclusive) =
193                    &operand.kind
194                else {
195                    continue;
196                };
197                let mut new_lower_inclusive = *lower_inclusive;
198                let mut new_upper_inclusive = *upper_inclusive;
199                if !new_lower_inclusive && point.kind == lower.kind {
200                    new_lower_inclusive = true;
201                    absorbed = true;
202                }
203                if !new_upper_inclusive && point.kind == upper.kind {
204                    new_upper_inclusive = true;
205                    absorbed = true;
206                }
207                if absorbed {
208                    *operand = simplify_interval(
209                        context,
210                        lower.clone(),
211                        upper.clone(),
212                        new_lower_inclusive,
213                        new_upper_inclusive,
214                    );
215                    break;
216                }
217            }
218            if !absorbed {
219                remaining_points.push(point);
220            }
221        }
222        let merged = simplify_finite_set(context, remaining_points);
223        if !matches!(
224            &merged.kind,
225            ExpressionKind::Constant(BuiltinConstant::EmptySet)
226        ) {
227            flattened.push(merged);
228        }
229    }
230    if flattened.is_empty() {
231        return empty_set_constant(context);
232    }
233    flattened.sort_by(|left, right| cohen_compare(left, right, &context.symbol_table));
234    flattened.dedup_by(|left, right| left.kind == right.kind);
235    if flattened.len() == 1 {
236        return flattened.into_iter().next().unwrap();
237    }
238    context.interner.intern(ExpressionKind::SetUnion(flattened))
239}

241/// Performs the advanced native `simplify_set_intersection` operation.
242pub fn simplify_set_intersection(
243    context: &Context,
244    operands: Vec<ExpressionReference>,
245) -> ExpressionReference {
246    // Auto-simplify a set intersection. Splices nested set_intersection
247    // children, removes complex_plane operands (the ambient universe is the
248    // identity for intersection) and real_line operands (the identity across
249    // the real sets this crate builds), collapses to empty_set whenever any
250    // operand is empty_set, and sorts the surviving operands under Cohen
251    // ordering with structural deduplication. An intersection whose operands
252    // are all ambient collapses back to the narrowest ambient present, so
253    // real_line ∩ complex_plane is real_line rather than either identity's
254    // fallback.
255    let mut flattened: Vec<ExpressionReference> = Vec::with_capacity(operands.len());
256    let mut has_real_line = false;
257    let mut has_complex_plane = false;
258    for operand in operands {
259        match &operand.kind {
260            ExpressionKind::SetIntersection(children) => {
261                for child in children {
262                    flattened.push(child.clone());
263                }
264            }
265            ExpressionKind::Constant(BuiltinConstant::RealLine) => {
266                has_real_line = true;
267            }
268            ExpressionKind::Constant(BuiltinConstant::ComplexPlane) => {
269                has_complex_plane = true;
270            }
271            ExpressionKind::Constant(BuiltinConstant::EmptySet) => {
272                return empty_set_constant(context);
273            }
274            _ => {
275                flattened.push(operand);
276            }
277        }
278    }
279    if flattened.is_empty() {
280        if !has_real_line && has_complex_plane {
281            return complex_plane_constant(context);
282        }
283        return real_line_constant(context);
284    }
285    flattened.sort_by(|left, right| cohen_compare(left, right, &context.symbol_table));
286    flattened.dedup_by(|left, right| left.kind == right.kind);
287    if flattened.len() == 1 {
288        return flattened.into_iter().next().unwrap();
289    }
290    context
291        .interner
292        .intern(ExpressionKind::SetIntersection(flattened))
293}

295/// Performs the advanced native `simplify_set_complement` operation.
296pub fn simplify_set_complement(
297    context: &Context,
298    left_operand: ExpressionReference,
299    right_operand: ExpressionReference,
300) -> ExpressionReference {
301    // Auto-simplify the relative complement left \ right. Trivial
302    // identities collapse first: removing nothing returns left unchanged;
303    // removing from nothing or from oneself returns empty_set; removing the
304    // ambient universe complex_plane returns empty_set; removing the ambient
305    // real_line returns empty_set for every left operand except
306    // complex_plane, whose non-real points survive the removal.
307    if matches!(
308        &right_operand.kind,
309        ExpressionKind::Constant(BuiltinConstant::EmptySet)
310    ) {
311        return left_operand;
312    }
313    if matches!(
314        &left_operand.kind,
315        ExpressionKind::Constant(BuiltinConstant::EmptySet)
316    ) {
317        return empty_set_constant(context);
318    }
319    if matches!(
320        &right_operand.kind,
321        ExpressionKind::Constant(BuiltinConstant::ComplexPlane)
322    ) {
323        return empty_set_constant(context);
324    }
325    if matches!(
326        &right_operand.kind,
327        ExpressionKind::Constant(BuiltinConstant::RealLine)
328    ) && !matches!(
329        &left_operand.kind,
330        ExpressionKind::Constant(BuiltinConstant::ComplexPlane)
331    ) {
332        return empty_set_constant(context);
333    }
334    if left_operand.kind == right_operand.kind {
335        return empty_set_constant(context);
336    }
337    context
338        .interner
339        .intern(ExpressionKind::SetComplement(left_operand, right_operand))
340}

342fn is_negative_infinity(value: &ExpressionReference) -> bool {
343    // Recognise the negative_infinity constant for the interval endpoint
344    // rewrites; positive_infinity for the upper-side rewrite has its own
345    // sibling helper for symmetry of intent.
346    matches!(
347        &value.kind,
348        ExpressionKind::Constant(BuiltinConstant::NegativeInfinity)
349    )
350}
351
352fn is_positive_infinity(value: &ExpressionReference) -> bool {
353    // Recognise the positive_infinity constant for the upper-side interval
354    // rewrites.
355    matches!(
356        &value.kind,
357        ExpressionKind::Constant(BuiltinConstant::Infinity)
358    )
359}

361fn is_undefined(value: &ExpressionReference) -> bool {
362    // Recognise the undefined sentinel so the interval simplifier propagates
363    // it rather than synthesising a non-rigorous interval over an undefined
364    // bound.
365    matches!(
366        &value.kind,
367        ExpressionKind::Constant(BuiltinConstant::Undefined)
368    )
369}
370
371fn is_finite_endpoint(value: &ExpressionReference) -> bool {
372    // True when the endpoint is something other than +/-infinity — the
373    // closed-point collapse to finite_set requires a finite element on
374    // both sides.
375    !is_negative_infinity(value) && !is_positive_infinity(value)
376}

402/// Performs the advanced native `simplify_condition_set` operation.
403pub fn simplify_condition_set(
404    context: &Context,
405    parameter_variable: SymbolIdentifier,
406    base_set: ExpressionReference,
407    condition: ExpressionReference,
408) -> ExpressionReference {
409    // Auto-simplify a condition_set { parameter in base_set : condition }.
410    // Collapse arms:
411    //
412    //   1. base_set is empty_set → empty_set (nothing to filter).
413    //   2. condition does not bind parameter_variable → if condition
414    //      evaluates to yes, return base_set; if no, return empty_set;
415    //      otherwise retain the structural placeholder.
416    //   3. base_set is finite_set → substitute parameter_variable for each
417    //      element and evaluate condition; elements with yes survive;
418    //      elements with no drop; a single unknown blocks the closed-form
419    //      collapse and returns the placeholder.
420    //   4. Otherwise return the structural condition_set placeholder.
421    if matches!(
422        &base_set.kind,
423        ExpressionKind::Constant(BuiltinConstant::EmptySet)
424    ) {
425        return empty_set_constant(context);
426    }
427    let condition_free = free_variables(&condition);
428    if !condition_free.contains(&parameter_variable) {
429        match crate::simplify::evaluate_condition_in_context(context, &condition) {
430            crate::assumptions::TruthValue::Yes => return base_set,
431            crate::assumptions::TruthValue::No => return empty_set_constant(context),
432            crate::assumptions::TruthValue::Unknown => {}
433        }
434    }
435    if let ExpressionKind::FiniteSet(elements) = &base_set.kind {
436        let elements_clone = elements.clone();
437        let parameter_symbol = context
438            .interner
439            .intern(ExpressionKind::Symbol(parameter_variable));
440        let mut surviving: Vec<ExpressionReference> = Vec::new();
441        let mut all_decided = true;
442        for element in elements_clone {
443            let mut mapping: SubstitutionMap = std::collections::HashMap::new();
444            mapping.insert(parameter_symbol.clone(), element.clone());
445            let substituted_condition = substitute(context, &condition, &mapping);
446            match crate::simplify::evaluate_condition_in_context(context, &substituted_condition) {
447                crate::assumptions::TruthValue::Yes => {
448                    surviving.push(element);
449                }
450                crate::assumptions::TruthValue::No => {}
451                crate::assumptions::TruthValue::Unknown => {
452                    all_decided = false;
453                    break;
454                }
455            }
456        }
457        if all_decided {
458            if surviving.is_empty() {
459                return empty_set_constant(context);
460            }
461            return crate::sets::builders::finite_set(context, surviving);
462        }
463    }
464    context.interner.intern(ExpressionKind::ConditionSet(
465        parameter_variable,
466        base_set,
467        condition,
468    ))
469}

471/// Performs the advanced native `simplify_image_set` operation.
472pub fn simplify_image_set(
473    context: &Context,
474    parameter_variable: SymbolIdentifier,
475    parameter_domain: ExpressionReference,
476    lambda_expression: ExpressionReference,
477) -> ExpressionReference {
478    // Auto-simplify an image_set. The empty parameter_domain collapses
479    // to empty_set (the empty image of an empty domain is empty). A
480    // lambda_expression that does not bind the parameter_variable is
481    // constant over the domain and collapses to a finite_set carrying the
482    // one value (provided the domain is non-empty — undecidable shapes
483    // stay structural). A finite_set parameter_domain enumerates point-
484    // by-point via substitution, producing a finite_set of mapped values.
485    // Every other shape (interval, real_line, integer_set, set_union /
486    // set_intersection / set_complement of these, or a structurally
487    // unresolved expression) returns the structural image_set placeholder.
488    if matches!(
489        &parameter_domain.kind,
490        ExpressionKind::Constant(BuiltinConstant::EmptySet)
491    ) {
492        return empty_set_constant(context);
493    }
494    let body_free = free_variables(&lambda_expression);
495    if !body_free.contains(&parameter_variable) {
496        return crate::sets::builders::finite_set(context, vec![lambda_expression]);
497    }
498    if let ExpressionKind::FiniteSet(elements) = &parameter_domain.kind {
499        let elements_clone = elements.clone();
500        let parameter_symbol = context
501            .interner
502            .intern(ExpressionKind::Symbol(parameter_variable));
503        let mapped: Vec<ExpressionReference> = elements_clone
504            .into_iter()
505            .map(|element| {
506                let mut mapping: SubstitutionMap = std::collections::HashMap::new();
507                mapping.insert(parameter_symbol.clone(), element);
508                substitute(context, &lambda_expression, &mapping)
509            })
510            .collect();
511        return crate::sets::builders::finite_set(context, mapped);
512    }
513    context.interner.intern(ExpressionKind::ImageSet(
514        parameter_variable,
515        parameter_domain,
516        lambda_expression,
517    ))
